  18 structure OrdinaryPairHostEmbedding (G₀:SimpleGraph V₀) (H₀:V₀→U₀→Prop)
  19     (Q:SimpleGraph V) (H:V→U→Prop) where
  20   row : V₀↪V
  21   hub : U₀↪U
  22   rim : ∀ a b,G₀.Adj a b→Q.Adj (row a) (row b)
  23   spoke : ∀ a b,H₀ a b→H (row a) (hub b)

  25 def OrdinaryPairHostEmbedding.IsFresh {G₀:SimpleGraph V₀} {H₀:V₀→U₀→Prop}
  26     {Q:SimpleGraph V} {H:V→U→Prop}
  27     (e:OrdinaryPairHostEmbedding G₀ H₀ Q H) (F:SimpleGraph V)
  28     (A W R:Finset V) (B:Finset U) : Prop :=
  29   (∀ a,e.row a∈A∧e.row a∉W∧e.row a∉R∧∀ z∈R,¬F.Adj (e.row a) z)∧
  30   (∀ a,e.hub a∉B)∧(∀ a b,¬F.Adj (e.row a) (e.row b))

  32 def OrdinaryPairHostEmbedding.resources {G₀:SimpleGraph V₀} {H₀:V₀→U₀→Prop}
  33     {Q:SimpleGraph V} {H:V→U→Prop} (e:OrdinaryPairHostEmbedding G₀ H₀ Q H) :
  34     Finset (Sym2 V⊕(V×U)) :=
  35   ordinaryHostResources (G₀.map e.row) (ordinaryMappedSpokes H₀ e.row e.hub)

  37 theorem OrdinaryPairHostEmbedding.resources_subset {G₀:SimpleGraph V₀} {H₀:V₀→U₀→Prop}
  38     {Q:SimpleGraph V} {H:V→U→Prop} (e:OrdinaryPairHostEmbedding G₀ H₀ Q H) :
  39     e.resources⊆ordinaryHostResources Q H := by
  40   intro x hx
  41   cases x with
  42   | inl x=>
  43     induction x using Sym2.ind with
  44     | _ v w=>
  45       rw [OrdinaryPairHostEmbedding.resources,ordinaryHostResources_rim,SimpleGraph.mem_edgeSet,SimpleGraph.map_adj] at hx
  46       obtain ⟨a,b,hab,rfl,rfl⟩:=hx
  47       exact (ordinaryHostResources_rim _ _ _).mpr (e.rim a b hab)
  48   | inr p=>
  49     rcases p with ⟨v,u⟩
  50     obtain ⟨a,b,hab,rfl,rfl⟩:=(ordinaryHostResources_spoke _ _ _ _).mp hx
  51     exact (ordinaryHostResources_spoke _ _ _ _).mpr (e.spoke a b hab)
