  17     (Σ i,{v : V i // v∉T i}) ↪ (Σ i,V i) where
  18   toFun v := ⟨v.1,v.2.val⟩

  20     rintro ⟨i,v⟩ ⟨j,w⟩ he
  21     have hij : i=j := congrArg Sigma.fst he

  23     have hvw : v.val=w.val := eq_of_heq (Sigma.mk.inj_iff.mp he).2
  24     have hvw' : v=w := Subtype.ext hvw
