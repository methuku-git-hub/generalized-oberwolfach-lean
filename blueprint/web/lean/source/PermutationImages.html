  15 def permutationInverseEquiv (V : Type*) : Equiv.Perm V ≃ Equiv.Perm V where
  16   toFun := Equiv.symm
  17   invFun := Equiv.symm
  18   left_inv _ := rfl
  19   right_inv _ := rfl

  21 theorem uniformPermutation_images {I V : Type*} [Fintype I] [Fintype V]
  22     (x : I ↪ V) [Nonempty (I ↪ V)] :
  23     (FiniteLaw.uniform (Equiv.Perm V)).map (fun σ => x.trans σ.toEmbedding) =
  24       FiniteLaw.uniform (I ↪ V) := by
  25   apply finiteLaw_uniform_map_transitive
  26   intro f g
  27   obtain ⟨e, he⟩ := Equiv.Perm.exists_extending_pair f g f.injective g.injective
  28   refine ⟨permutationRightEquiv e, ?_⟩
  29   intro σ
  30   constructor
  31   · intro h
  32     ext i
  33     apply e.injective
  34     have hi := congrArg (fun z : I ↪ V => z i) h
  35     exact hi.trans (he i).symm
  36   · intro h
  37     ext i
  38     change e (σ (x i)) = g i
  39     rw [show σ (x i) = f i from congrArg (fun z : I ↪ V => z i) h]
  40     exact he i

  42 theorem uniformPermutation_inverse_images {I V : Type*} [Fintype I] [Fintype V]
  43     (x : I ↪ V) [Nonempty (I ↪ V)] :
  44     (FiniteLaw.uniform (Equiv.Perm V)).map (fun σ => x.trans σ.symm.toEmbedding) =
  45       FiniteLaw.uniform (I ↪ V) := by
  46   have h := uniformPermutation_images x
  47   rw [← finiteLaw_uniform_map_equiv (permutationInverseEquiv V)] at h
  48   rw [finiteLaw_map_map] at h
  49   convert! h using 1
