   1 import Mathlib.Combinatorics.SimpleGraph.Finite
   2 import Mathlib.Basic.Real.Basic
   3 /-! A simple underlying graph has degree at most two when the directed
   4 source has at most one predecessor and one successor at each vertex. -/
   5 set_option autoImplicit false
   6 noncomputable section
   7 namespace Oberwolfach
   8 attribute [local instance] Classical.propDecidable
   9 variable {X:Type} [Fintype X] [DecidableEq X]
  10 
  11 theorem prescribed_source_degree_from_directions (Q:SimpleGraph X) (D:X→X→Prop)
  12     (hrep:∀ x y,Q.Adj x y↔D x y ∨ D y x)
  13     (hout:∀ x y z,D x y→D x z→y=z)
  14     (hin:∀ x y z,D y x→D z x→y=z):
  15     ∀ x,Q.degree x≤2 := by
  16   intro x
  17   let out:X:=if h:∃ y,D x y then Classical.choose h else x
  18   let inn:X:=if h:∃ y,D y x then Classical.choose h else x
  19   have hsubset:Q.neighborFinset x⊆{out,inn}:=by
  20     intro y hy
  21     have hy':Q.Adj x y:=(SimpleGraph.mem_neighborFinset Q x y).mp hy
  22     rcases (hrep x y).mp hy' with hy'|hy'
  23     · have he:∃ z,D x z:=⟨y,hy'⟩
  24       have ho:D x out:=by simp only [out,dif_pos he];exact Classical.choose_spec he
  25       have hyout:y=out:=hout x y out hy' ho
  26       simp [hyout]
  27     · have he:∃ z,D z x:=⟨y,hy'⟩
  28       have hi:D inn x:=by simp only [inn,dif_pos he];exact Classical.choose_spec he
  29       have hyin:y=inn:=hin x y inn hy' hi
  30       simp [hyin]
  31   calc
  32     Q.degree x=(Q.neighborFinset x).card:=rfl
  33     _≤({out,inn}:Finset X).card:=Finset.card_le_card hsubset
  34     _≤({inn}:Finset X).card+1:=Finset.card_insert_le out {inn}
  35     _=2:=by simp
  36 end Oberwolfach
