  19     (j:Bool) (x y:V) (hx:x∉T) (hy:y∉T)
  20     (hxy:markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) j x y=1)
  21     (hD:¬DR.Adj x (if j then π y else y)):

  23       (fun e=>rawMarkedRestrictPair f g (rawMarkedDeleteRims σ DR.edgeFinset) (Sum.inl e)) j,
  24       f a.val.1=x∧f a.val.2=y:=by
  25   have hxmem:x∈Finset.univ.image f:=by rw [hfX];simp [hx]
  26   have hymem:y∈Finset.univ.image f:=by rw [hfX];simp [hy]
  27   obtain ⟨a,_,ha⟩:=Finset.mem_image.mp hxmem
  28   obtain ⟨b,_,hb⟩:=Finset.mem_image.mp hymem
  29   have hdeleted:markedPhysicalRim P π (3*d)
  30       (fun e=>rawMarkedDeleteRims σ DR.edgeFinset (Sum.inl e)) j x y=1:=by
  31     apply (markedPhysicalRim_delete_iff P π d σ DR.edgeFinset j x y).mpr
  32     exact ⟨hxy,by simpa only [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using hD⟩

  34       (fun e=>rawMarkedRestrictPair f g (rawMarkedDeleteRims σ DR.edgeFinset) (Sum.inl e)) j a b=1:=by
  35     apply (markedPhysicalRim_orderEmbedding f π ρ hf P (3*d)
  36       (fun e=>rawMarkedDeleteRims σ DR.edgeFinset (Sum.inl e)) j a b).mpr
  37     simpa only [ha,hb] using hdeleted
