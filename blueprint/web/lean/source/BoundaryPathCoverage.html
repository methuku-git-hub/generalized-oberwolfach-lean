  17 theorem boundary_path_exact_edges (C:I→OrdinaryWheel Q F H r)
  18     (X:Finset V) (x v w:I→V) (u:I→U)
  19     (hvw:∀ i,v i≠w i) (hhub:∀ i,ordinaryWheelHub (C i)=u i)
  20     (hxv:∀ i,(ordinaryWheelRim (C i)).Adj (x i) (v i))
  21     (hxw:∀ i,(ordinaryWheelRim (C i)).Adj (x i) (w i))
  22     (hsupport:∀ i,∀ a∈(ordinaryWheelRim (C i)).support,a=x i∨a∈X):
  23     (∀ a∉X,∀ b,(wheelFamilyRim C).Adj a b↔∃ i,a=x i ∧ (b=v i∨b=w i)) ∧
  24     (∀ a∉X,∀ z,wheelFamilySpokes C a z↔∃ i,a=x i ∧ u i=z):=by
  25   constructor
  26   · intro a ha b
  27     rw [wheelFamilyRim,SimpleGraph.iSup_adj]
  28     constructor
  29     · rintro ⟨i,hi⟩
  30       have he:a=x i:=(hsupport i a hi.mem_support_left).resolve_right ha
  31       refine ⟨i,he,?_⟩
  32       have hb: b∈(ordinaryWheelRim (C i)).neighborFinset (x i):=
  33         ((ordinaryWheelRim (C i)).mem_neighborFinset _ _).mpr (he ▸ hi)
  34       rw [ordinaryWheel_root_neighbors (C i) (x i) (v i) (w i) (hvw i) (hxv i) (hxw i)] at hb
  35       simpa only [Finset.mem_insert,Finset.mem_singleton] using hb
  36     · rintro ⟨i,rfl,hb|hb⟩
  37       · exact ⟨i,hb ▸ hxv i⟩
  38       · exact ⟨i,hb ▸ hxw i⟩
  39   · intro a ha z
  40     constructor
  41     · rintro ⟨i,hi,hia⟩
  42       exact ⟨i,(hsupport i a hia).resolve_right ha,(hhub i).symm.trans hi⟩
  43     · rintro ⟨i,rfl,hi⟩
  44       exact ⟨i,(hhub i).trans hi,(hxv i).mem_support_left⟩

  46 theorem boundary_path_physical_degrees (C:I→OrdinaryWheel Q F H r) (K D:ℕ)
  47     (hrim:∀ v,(wheelFamilyRim C).degree v≤2*(K+D))
  48     (hrow:∀ v,(Finset.univ.filter (wheelFamilySpokes C v)).card≤K+D)
  49     (hcol:∀ u,(Finset.univ.filter (fun v=>wheelFamilySpokes C v u)).card≤D*r):
  50     (∀ v,(ordinaryPhysicalIncidentEdges (wheelFamilyRim C) (wheelFamilySpokes C) (Sum.inl v)).card≤3*(K+D)) ∧
  51     (∀ u,(ordinaryPhysicalIncidentEdges (wheelFamilyRim C) (wheelFamilySpokes C) (Sum.inr u)).card≤D*r):=by
  52   constructor
  53   · intro v
  54     rw [ordinaryPhysicalIncidentEdges_rim_card]
  55     exact (Nat.add_le_add (hrim v) (hrow v)).trans_eq (by omega)
  56   · intro u
  57     rw [ordinaryPhysicalIncidentEdges_hub_card]
  58     exact hcol u

  60 theorem boundary_path_remainder_supported (C:I→OrdinaryWheel Q F H r)
  61     (X:Finset V) (J:Finset U) (x v w:I→V) (u:I→U)
  62     (hvw:∀ i,v i≠w i) (hhub:∀ i,ordinaryWheelHub (C i)=u i)
  63     (hxv:∀ i,(ordinaryWheelRim (C i)).Adj (x i) (v i))
  64     (hxw:∀ i,(ordinaryWheelRim (C i)).Adj (x i) (w i))
  65     (hsupport:∀ i,∀ a∈(ordinaryWheelRim (C i)).support,a=x i∨a∈X)
  66     (hQr:∀ a∉X,∀ b,Q.Adj a b→∃ i,a=x i ∧ (b=v i∨b=w i))
  67     (hHr:∀ a∉X,∀ z,H a z→∃ i,a=x i ∧ u i=z)
  68     (hHc:∀ z∉J,∀ a,¬H a z):
  69     (Q\wheelFamilyRim C).support⊆(X:Set V) ∧
  70       (∀ a z,H a z∧¬wheelFamilySpokes C a z→a∈X∧z∈J):=by
  71   have hex:=boundary_path_exact_edges C X x v w u hvw hhub hxv hxw hsupport
  72   constructor
  73   · intro a ha
  74     by_contra hn
  75     obtain ⟨b,hb⟩:=ha
  76     exact hb.2 ((hex.1 a hn b).mpr (hQr a hn b hb.1))
  77   · intro a z hz
  78     constructor
  79     · by_contra hn
  80       exact hz.2 ((hex.2 a hn z).mpr (hHr a hn z hz.1))
  81     · by_contra hn
  82       exact hHc z hn a hz.1
