  12 theorem fractional_resource_sparsification
  13     {V E A : Type} [Fintype V] [Fintype E] [Fintype A] [DecidableEq V]
  14     (R : E → Finset V) (w : E → ℝ) (g : A → E → ℝ)
  15     (hw : ∀ e,0≤w e) (hW : 0<∑ e,w e) :
  16     ∃ w' : E → ℝ,(∀ e,0≤w' e) ∧ (∀ e,w e=0 → w' e=0) ∧
  17       (∑ e,w' e=∑ e,w e) ∧
  18       (∀ v,hypergraphWeightedDegree R w' v=hypergraphWeightedDegree R w v) ∧
  19       (∀ u v,(∑ e,if u∈R e ∧ v∈R e then w' e else 0)=
  20         ∑ e,if u∈R e ∧ v∈R e then w e else 0) ∧
  21       (∀ i,∑ e,g i e*w' e=∑ e,g i e*w e) ∧
  22       (Finset.univ.filter (fun e=>0<w' e)).card≤
  23         Fintype.card V+Fintype.card V^2+Fintype.card A+1 := by
  24   classical
  25   let J := V ⊕ ((V×V)⊕A)
  26   let z : E → J → ℝ := fun e=>Sum.elim (fun v=>if v∈R e then 1 else 0)
  27     (Sum.elim (fun uv=>if uv.1∈R e ∧ uv.2∈R e then 1 else 0) (fun i=>g i e))
  28   obtain ⟨w',hw',hsupport,hmass,hmeasure,hcard⟩ := finite_measurement_sparsification z w hw hW
  29   refine ⟨w',hw',hsupport,hmass,?_,?_,?_,?_⟩
  30   · intro v
  31     simpa only [z,Sum.elim_inl,hypergraphWeightedDegree,mul_ite,mul_one,mul_zero] using hmeasure (Sum.inl v)
  32   · intro u v
  33     simpa only [z,Sum.elim_inr,Sum.elim_inl,mul_ite,mul_one,mul_zero] using hmeasure (Sum.inr (Sum.inl (u,v)))
  34   · intro i
  35     simpa only [z,Sum.elim_inr,mul_comm] using hmeasure (Sum.inr (Sum.inr i))
  36   · simpa only [J,Fintype.card_sum,Fintype.card_prod,pow_two,Nat.add_assoc] using hcard

  38 theorem sum_floor_weighted_bounds
  39     {E : Type*} [Fintype E] (w g : E → ℝ) (T : ℕ) (b : ℝ)
  40     (hw : ∀ e,0≤w e) (hg : ∀ e,0≤g e ∧ g e≤b) :
  41     (T : ℝ)*(∑ e,g e*w e)-Fintype.card E*b≤∑ e,g e*⌊(T : ℝ)*w e⌋₊ ∧
  42     (∑ e,g e*⌊(T : ℝ)*w e⌋₊)≤(T : ℝ)*(∑ e,g e*w e) := by
  43   have hlow (e : E) : (T : ℝ)*(g e*w e)-b≤g e*⌊(T : ℝ)*w e⌋₊ := by
  44     have hf := Nat.lt_floor_add_one ((T : ℝ)*w e)
  45     have hm := mul_le_mul_of_nonneg_left hf.le (hg e).1
  46     nlinarith [ (hg e).2 ]
  47   have hupp (e : E) : g e*⌊(T : ℝ)*w e⌋₊≤(T : ℝ)*(g e*w e) := by
  48     have hf := Nat.floor_le (mul_nonneg (Nat.cast_nonneg T) (hw e))
  49     have hm := mul_le_mul_of_nonneg_left hf (hg e).1
  50     nlinarith
  51   constructor
  52   · simpa only [Finset.sum_sub_distrib,← Finset.mul_sum,Finset.sum_const,Finset.card_univ,nsmul_eq_mul] using
  53       Finset.sum_le_sum (fun e (_ : e∈Finset.univ)=>hlow e)
  54   · simpa only [← Finset.mul_sum] using Finset.sum_le_sum (fun e (_ : e∈Finset.univ)=>hupp e)
