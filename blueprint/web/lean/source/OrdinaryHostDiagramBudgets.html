  10 theorem ordinary_host_diagram_budgets (β ε d:ℝ) (hβ:0<β) (hd:0≤d) (hε:ε≤β/4)
  11     (r m M f:ℕ) : ∃ n₀:ℕ,1≤n₀∧∀ n:ℕ,n₀≤n→∀ W R B:ℕ,
  12       W≤f→R≤m→(B:ℝ)≤ε*n+f→
  13       (W:ℝ)+((R:ℝ)+6*M+r+6)*(d+1)<(β/2)*n∧
  14       (B:ℝ)+6*M<(β/2)*n := by
  15   let P:ℝ:=f+(m+6*M+r+6)*(d+1)+f+6*M+1
  16   have hP:0<P:=by dsimp [P];positivity
  17   obtain ⟨N,hN⟩:=exists_nat_gt (4*P/β)
  18   refine ⟨max 1 N,le_max_left _ _,?_⟩
  19   intro n hn W R B hW hR hB
  20   have hnR:(N:ℝ)≤n:=by exact_mod_cast (le_max_right 1 N).trans hn
  21   have hlarge:4*P<β*n:=by simpa only [mul_comm] using (div_lt_iff₀ hβ).mp (hN.trans_le hnR)
  22   have hn0:0≤(n:ℝ):=by positivity
  23   have hW':(W:ℝ)≤f:=by exact_mod_cast hW
  24   have hR':(R:ℝ)≤m:=by exact_mod_cast hR
  25   have hterm:=mul_le_mul_of_nonneg_right (show (R:ℝ)+6*M+r+6≤m+6*M+r+6 by linarith only [hR']) (by linarith:(0:ℝ)≤d+1)
  26   have hP₁:(f:ℝ)+(m+6*M+r+6)*(d+1)≤P:=by
  27     dsimp [P]
  28     have h0:0≤(f:ℝ)+6*M+1:=by positivity
  29     linarith only [h0]
  30   have hP₂:(f:ℝ)+6*M≤P:=by
  31     dsimp [P]
  32     have h0:0≤(f:ℝ)+(m+6*M+r+6)*(d+1)+1:=by positivity
  33     linarith only [h0]
  34   have hεn:=mul_le_mul_of_nonneg_right hε hn0
  35   constructor
  36   · linarith only [hW',hterm,hP₁,hlarge,hP]
