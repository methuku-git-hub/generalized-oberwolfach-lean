  14     ∀ δ:ℝ,δ≤ζ*p*n/2→
  15       4*(C*p*Real.sqrt (C*p)*n)+4*(Nat.ceil (4*(C*p*Real.sqrt (C*p)*n)):ℝ)+δ≤ζ*p*n := by
  16   let a:=ζ/(80*C*Real.sqrt C)
  17   have ha:0<a:=by dsimp [a];positivity
  18   refine ⟨min 1 (a^2),lt_min (by norm_num) (pow_pos ha _),min_le_left _ _,?_⟩
  19   intro p hp hpp
  20   have hs:Real.sqrt p≤a:=(Real.sqrt_le_left ha.le).mpr (hpp.trans (min_le_right _ _))
  21   have hsmall:20*(C*p*Real.sqrt (C*p))≤ζ*p/4:=by
  22     rw [Real.sqrt_mul hC.le]
  23     have hmul:=mul_le_mul_of_nonneg_left hs (show 0≤80*C*Real.sqrt C by positivity)
  24     have heq:(80*C*Real.sqrt C)*a=ζ:=by dsimp [a];field_simp
  25     rw [heq] at hmul
  26     have hh:=mul_le_mul_of_nonneg_right hmul hp.le
  27     nlinarith only [hh]
  28   refine ⟨Nat.ceil (16/(ζ*p)),?_⟩
  29   intro n hn δ hδ

  33     have hh:=(div_le_iff₀ (mul_pos hζ hp)).mp hnlow
