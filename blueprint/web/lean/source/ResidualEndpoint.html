  11 def residualLeft {X Y Z : Type*} [Fintype Z]
  12     (A : X → Y → ℝ) (ρ : X → Y → Z → ℝ) : X → Y → ℝ :=
  13   fun x y => A x y - ∑ z, ρ x y z
  14 
  15 def residualRight {X Y Z : Type*} [Fintype X]
  16     (B : Y → Z → ℝ) (ρ : X → Y → Z → ℝ) : Y → Z → ℝ :=
  17   fun y z => B y z - ∑ x, ρ x y z

  19 theorem residual_interfaces_balanced {X Y Z : Type*}
  20     [Fintype X] [Fintype Z]
  21     (A : X → Y → ℝ) (B : Y → Z → ℝ) (ρ : X → Y → Z → ℝ)
  22     (hbal : ∀ y, ∑ x, A x y = ∑ z, B y z) (y : Y) :
  23     (∑ x, residualLeft A ρ x y) = ∑ z, residualRight B ρ y z := by
  24   unfold residualLeft residualRight
  25   rw [Finset.sum_sub_distrib, Finset.sum_sub_distrib, hbal y, Finset.sum_comm]

  27 def scaledResidualPath {X Y Z : Type*} [Fintype X] [Fintype Z]
  28     (A : X → Y → ℝ) (B : Y → Z → ℝ) (ρ : X → Y → Z → ℝ)
  29     (k : ℝ) (x : X) (y : Y) (z : Z) : ℝ :=
  30   residualPath (residualLeft A ρ) (residualRight B ρ) x y z / (k - 1)

  38 theorem scaledResidualEndpoint_mem {X Y Z : Type*}
  39     [Fintype X] [Fintype Y] [Fintype Z]
  40     (A : X → Y → ℝ) (B : Y → Z → ℝ) (ρ : X → Y → Z → ℝ)
  41     (dX : X → ℝ) (dZ : Z → ℝ) (k : ℝ) (hk : 1 < k)
  42     (hleft : ∀ x y, (∑ z, ρ x y z) ≤ A x y)
  43     (hright : ∀ y z, (∑ x, ρ x y z) ≤ B y z)
  44     (hbal : ∀ y, ∑ x, A x y = ∑ z, B y z)
  45     (hA : ∀ x, ∑ y, A x y = dX x)
  46     (hB : ∀ z, ∑ y, B y z = dZ z)
  47     (hρX : ∀ x, (∑ y, ∑ z, ρ x y z) = dX x / k)
  48     (hρZ : ∀ z, (∑ y, ∑ x, ρ x y z) = dZ z / k) :
  49     scaledResidualEndpoint A B ρ k ∈
  50       transportMatrices (fun x => dX x / k) (fun z => dZ z / k) := by
  51   have hL : ∀ x y, 0 ≤ residualLeft A ρ x y := fun x y => sub_nonneg.mpr (hleft x y)
  52   have hR : ∀ y z, 0 ≤ residualRight B ρ y z := fun y z => sub_nonneg.mpr (hright y z)
  53   have hbal' := residual_interfaces_balanced A B ρ hbal
  54   have hk1 : 0 < k - 1 := by linarith
  55   have hk0 : k ≠ 0 := by linarith
  56   have hid (d : ℝ) : (d - d / k) / (k - 1) = d / k := by
  57     apply (div_eq_iff hk1.ne').mpr
  58     field_simp
  59   refine ⟨?_, ?_, ?_⟩
  60   · intro x z
  61     exact Finset.sum_nonneg fun y _ =>
  62       div_nonneg (residualPath_bounds _ _ hL hR hbal' x y z).1 hk1.le
  63   · intro x
  64     unfold scaledResidualEndpoint scaledResidualPath
  65     rw [Finset.sum_comm]
  66     simp_rw [← Finset.sum_div, residualPath_left_load _ _ hL hbal']
  67     unfold residualLeft
  68     rw [Finset.sum_sub_distrib, hA x, hρX x, hid]
  69   · intro z
  70     unfold scaledResidualEndpoint scaledResidualPath
  71     rw [Finset.sum_comm]
  72     simp_rw [← Finset.sum_div, residualPath_right_load _ _ hR hbal']
  73     unfold residualRight
  74     rw [Finset.sum_sub_distrib, hB z, hρZ z, hid]

  94 theorem reserve_residual_reconstructs_left {X Y Z : Type*}
  95     [Fintype X] [Fintype Z]
  96     (A : X → Y → ℝ) (B : Y → Z → ℝ) (ρ : X → Y → Z → ℝ)
  97     (k : ℝ) (hk : 1 < k)
  98     (hleft : ∀ x y, (∑ z, ρ x y z) ≤ A x y)
  99     (hbal : ∀ y, ∑ x, A x y = ∑ z, B y z)
 100     (x : X) (y : Y) :
 101     (∑ z, (ρ x y z + (k - 1) * scaledResidualPath A B ρ k x y z)) = A x y := by
 102   have hk' : k - 1 ≠ 0 := by linarith
 103   unfold scaledResidualPath

 107     (residual_interfaces_balanced A B ρ hbal) x y
 108   rw [Finset.sum_add_distrib, hload]
 109   unfold residualLeft
