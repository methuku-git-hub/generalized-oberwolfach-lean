   8 theorem degree_centre_transfer (x p D d e δ : ℝ) (hp : 0≤p)
   9     (hx : |x-p*d|≤e) (hd : |d-D|≤δ) : |x-p*D|≤e+p*δ := by
  10   have he : x-p*D=(x-p*d)+p*(d-D) := by ring
  11   rw [he]
  12   apply (abs_add_le _ _).trans
  13   apply add_le_add hx
  14   rw [abs_mul,abs_of_nonneg hp]
  15   exact mul_le_mul_of_nonneg_left hd hp

  17 theorem restricted_vertex_balance_error (r0 rK h0 hK p k s t D B ε L δ : ℝ)
  18     (hD0 : p*D=s*B) (hDK : k*D=t*B)
  19     (hr0 : |r0-p*D|≤ε+L+p*δ) (hrK : |rK-k*D|≤ε+L+k*δ)
  20     (hh0 : |h0-s*B|≤ε) (hhK : |hK-t*B|≤ε) :
  21     |r0+rK-(h0+hK)|≤4*ε+2*L+(p+k)*δ := by
  22   have he : r0+rK-(h0+hK)=((r0-p*D)+(rK-k*D))-((h0-s*B)+(hK-t*B)) := by
  23     rw [hD0,hDK]
  24     ring
  25   rw [he]
  26   have hfirst := abs_sub_le ((r0-p*D)+(rK-k*D)) 0 ((h0-s*B)+(hK-t*B))
  27   simp only [sub_zero,zero_sub,abs_neg] at hfirst
  28   have hr := abs_add_le (r0-p*D) (rK-k*D)
  29   have hh := abs_add_le (h0-s*B) (hK-t*B)
  30   nlinarith only [hfirst,hr,hh,hr0,hrK,hh0,hhK]

  32 theorem restricted_marked_balance_error (r h k t D B ε L δ : ℝ)
  33     (hDK : k*D=t*B) (hr : |r-k*D|≤ε+L+k*δ) (hh : |h-t*B|≤ε) :
  34     |r-h|≤2*ε+L+k*δ := by
  35   have he := abs_sub_le r (k*D) h

  40 theorem restricted_hub_balance_error (h0 hK s t A ε m : ℝ)
  41     (hm : 0≤m) (hs : s=m*t) (hh0 : |h0-s*A|≤ε) (hhK : |hK-t*A|≤ε) :
  42     |h0-m*hK|≤(1+m)*ε := by
  43   have he : h0-m*hK=(h0-s*A)-m*(hK-t*A) := by rw [hs];ring
  44   rw [he]
  45   have hh := abs_sub_le (h0-s*A) 0 (m*(hK-t*A))

  47   have hmul := mul_le_mul_of_nonneg_left hhK hm
  48   nlinarith only [hh,hmul,hh0]
