  11 theorem long_protected_pair_edge_labels
  12     {n:ℕ} {U Color:Type} [Fintype U] [DecidableEq U] [Fintype Color]
  13     (π:Equiv.Perm (Fin n)) (d:ℕ) (hd:0<d)
  14     (σ:RawIntervalFamilyOutcome (Fin n) U Bool Color)
  15     (r:ℕ) (J:StartRetentionOutcome (rawStartHubSet π d σ false) r)
  16     (Old:U→Finset (Fin n)) (inner:Finset U) (θ ρ₀ ρ D:ℝ)
  17     (hθ:0<θ) (hρ₀:0<ρ₀) (hρlo:ρ₀≤ρ) (hDlo:(n:ℝ)/2≤D) (hsize:ρ*D=2*Fintype.card U)
  18     (hspace:1/(4*(d:ℝ))≤(5/64:ℝ)*θ^2)
  19     (hN:5*(n:ℝ)^(3/4:ℝ)≤(5*θ^2*ρ₀/1024)*n)
  20     (hinner:(inner.card:ℝ)≤(5*θ^2*ρ₀/1024)*n)
  21     (hOld:∀u,Disjoint (rawIntervalProtectedUnassignedList π d σ false true r J u) (Old u))
  22     (hpair:∀x y,x≠y→((5/32:ℝ)*θ^2-1/(4*(d:ℝ)))*Fintype.card U-5*(n:ℝ)^(3/4:ℝ)≤
  23       (rawProtectedPairHubs π d σ false true r J x y).card)
  24     (Kbad:SimpleGraph (Fin n)):
  25     let Lists:=rawIntervalProtectedUnassignedList π d σ false true r J
  26     let L:=protectedPairEdgeHubs Lists inner
  27     (∀e∈Kbad.edgeFinset,Disjoint (L e) inner) ∧
  28     (∀u e v,e∈Kbad.edgeFinset→v∈e.toFinset→v∈Old u→u∈L e→u∈(∅:Finset U)) ∧
  29     (∀e∈Kbad.edgeFinset,(2*(5*θ^2*ρ₀/1024))*(n:ℝ)≤(L e).card) ∧
  30     (∀v:Fin n,(((∅:Finset U).card):ℝ)≤(0:ℝ)*n):=by
  31   intro Lists L
  32   obtain ⟨hout,hshield,hcard⟩:=long_protected_pair_candidate_hubs π d hd σ r J Old inner θ ρ₀ ρ D
  33     hθ hρ₀ hρlo hDlo hsize hspace hN hinner hOld hpair
  34   refine ⟨fun e _=>hout e,?_,?_,by simp⟩
  35   · intro u e v he hv ho hu
  36     exact False.elim (hshield u e v hv ho hu)
  37   · intro e he
  38     induction e using Sym2.ind with
  39     | _ x y =>
  40       have hxy:x≠y:=by
  41         intro h
  42         subst y
  43         exact Kbad.loopless.irrefl x (by simpa only [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using he)
  44       exact hcard x y hxy

  59       (rawProtectedPairHubs π d σ false true r J x y).card)
  60     (hα:α≤2*(5*θ^2*ρ₀/1024)) (hb:0≤b) (Kbad:SimpleGraph (Fin n)):
