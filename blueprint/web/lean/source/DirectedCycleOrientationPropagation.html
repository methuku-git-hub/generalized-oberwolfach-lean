   8 theorem directed_orientation_propagates_on_orbit
   9     {X:Type*} (σ:Equiv.Perm X) (D:X→X→Prop)
  10     (hcover:∀x,D x (σ x) ∨ D (σ x) x)
  11     (hin:∀x y z,D y x→D z x→y=z)
  12     (hno2:∀x,σ (σ x)≠x) (a:X) (ha:D a (σ a)):
  13     ∀n:ℕ,D ((σ^n) a) (σ ((σ^n) a)):=by
  14   intro n
  15   induction n with
  16   | zero => simpa using ha
  17   | succ n ih =>
  18     have hnext:D (σ ((σ^n) a)) (σ (σ ((σ^n) a))) :=by
  19       rcases hcover (σ ((σ^n) a)) with hf | hb
  20       · exact hf
  21       · have heq:((σ^n) a)=σ (σ ((σ^n) a)):=hin (σ ((σ^n) a)) ((σ^n) a) (σ (σ ((σ^n) a))) ih hb
  22         exact False.elim (hno2 ((σ^n) a) heq.symm)
  23     simpa only [pow_succ',Equiv.Perm.mul_apply] using hnext
  24 end Oberwolfach
