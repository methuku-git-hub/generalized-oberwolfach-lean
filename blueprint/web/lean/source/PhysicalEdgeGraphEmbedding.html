  12 theorem physicalEdgeGraph_image_map {W V:Type} [Fintype W] [Fintype V] [DecidableEq W] [DecidableEq V]
  13     (f:W↪V) (S:Finset (Finset W)):
  14     physicalEdgeGraph (S.image (fun e=>e.image f))=(physicalEdgeGraph S).map f:=by
  15   ext v z
  16   change (v≠z∧{v,z}∈S.image (fun e=>e.image f))↔_
  17   rw [SimpleGraph.map_adj]
  18   constructor
  19   · rintro ⟨hvz,he⟩
  20     obtain ⟨e,he,hpair⟩:=Finset.mem_image.mp he
  21     have hv:v∈e.image f:=by rw [hpair];exact Finset.mem_insert_self _ _
  22     have hz:z∈e.image f:=by rw [hpair];exact Finset.mem_insert_of_mem (Finset.mem_singleton_self _)
  23     obtain ⟨x,hx,hfx⟩:=Finset.mem_image.mp hv
  24     obtain ⟨y,hy,hfy⟩:=Finset.mem_image.mp hz
  25     have hxy:x≠y:=by intro h;apply hvz;exact hfx.symm.trans ((congrArg f h).trans hfy)
  26     have heq:e={x,y}:=by
  27       apply Finset.image_injective f.injective
  28       simpa only [Finset.image_insert,Finset.image_singleton,hfx,hfy] using hpair
  29     exact ⟨x,y,⟨hxy,heq ▸ he⟩,hfx,hfy⟩
  30   · rintro ⟨x,y,hxy,rfl,rfl⟩
  31     refine ⟨fun he=>hxy.1 (f.injective he),Finset.mem_image.mpr ⟨{x,y},hxy.2,?_⟩⟩
  32     simp only [Finset.image_insert,Finset.image_singleton]
