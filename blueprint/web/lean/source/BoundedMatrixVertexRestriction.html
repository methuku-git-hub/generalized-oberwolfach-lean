  11 theorem bounded_matrix_vertex_restriction (f:W↪V) (M:V→V→ℝ)
  12     (hM:∀ x y,|M x y|≤1) (x:W):
  13     |(∑ y:W,M (f x) (f y))-(∑ y:V,M (f x) y)|≤(Finset.univ\Finset.univ.image f).card∧
  14     |(∑ y:W,M (f y) (f x))-(∑ y:V,M y (f x))|≤(Finset.univ\Finset.univ.image f).card:=by
  15   have hout:=finite_bounded_sum_restriction (Finset.univ.image f) Finset.univ
  16     (Finset.subset_univ _) (M (f x)) (fun y _=>hM (f x) y)
  17   have hin:=finite_bounded_sum_restriction (Finset.univ.image f) Finset.univ
  18     (Finset.subset_univ _) (fun y=>M y (f x)) (fun y _=>hM y (f x))
  19   rw [Finset.sum_image (fun _ _ _ _ he=>f.injective he)] at hout hin
  20   exact ⟨hout,hin⟩
