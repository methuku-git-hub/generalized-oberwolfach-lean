   3 /-! If the matching edge sets assigned to different factors are physically
   4 disjoint and contained in a graph, the number of matching endpoint sets
   5 containing a fixed vertex is at most its degree in that graph. -/

  12 theorem prescribed_matching_endpoint_occurrences (M:I→Finset (V×V)) (K:SimpleGraph V)
  13     (hsub:∀i,(M i).image prescribedArcPhysical⊆K.edgeFinset)
  14     (hpair:PairwisePhysicalDisjoint (fun i=>(M i).image prescribedArcPhysical)) (v:V):
  15     (Finset.univ.filter (fun i=>v∈prescribedMatchingEndpoints (M i))).card≤K.degree v := by
  16   let S:=Finset.univ.filter (fun i=>v∈prescribedMatchingEndpoints (M i))
  17   have hex:∀ i∈S,∃e,e∈(M i).image prescribedArcPhysical ∧ e∈K.incidenceFinset v:=by
  18     intro i hi
  19     have hv:v∈prescribedMatchingEndpoints (M i):=(Finset.mem_filter.mp hi).2
  20     obtain ⟨a,ha,hv⟩:=Finset.mem_biUnion.mp hv
  21     let e:Sym2 V:=prescribedArcPhysical a
  22     have he:e∈(M i).image prescribedArcPhysical:=Finset.mem_image.mpr ⟨a,ha,rfl⟩
  23     refine ⟨e,he,?_⟩
  24     rw [SimpleGraph.incidenceFinset_eq_filter,Finset.mem_filter]
  25     refine ⟨hsub i he,?_⟩
  26     simp only [Finset.mem_insert,Finset.mem_singleton] at hv
  27     simpa [e,prescribedArcPhysical,Sym2.mem_iff] using hv
  28   let pick:I→Sym2 V:=fun i=>if hi:i∈S then Classical.choose (hex i hi) else s(v,v)
  29   have hmem:∀ i∈S,pick i∈(M i).image prescribedArcPhysical ∧ pick i∈K.incidenceFinset v:=by
  30     intro i hi
  31     simpa [pick,hi] using (Classical.choose_spec (hex i hi))
  32   have hmap:Set.MapsTo pick (↑S:Set I) (↑(K.incidenceFinset v):Set (Sym2 V)):=by
  33     intro i hi;exact (hmem i hi).2
  34   have hinj:Set.InjOn pick (↑S:Set I):=by
  35     intro i hi j hj heq
  36     by_contra hij
  37     have heI:pick i∈(M i).image prescribedArcPhysical:=(hmem i hi).1
  38     have heJ:pick i∈(M j).image prescribedArcPhysical:=by rw [heq];exact (hmem j hj).1
  39     exact (Finset.disjoint_left.mp (hpair i j hij)) heI heJ
  40   have hc:=Finset.card_le_card_of_injOn pick hmap hinj
  41   rw [SimpleGraph.card_incidenceFinset_eq_degree] at hc
  42   exact hc
  43 end Oberwolfach
