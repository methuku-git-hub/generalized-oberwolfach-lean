   9 theorem colouring_round_ratio_progress
  10     {k : ℕ} (hk : 0<k) {l s ρ a b ε η M r l' s' : ℝ}
  11     (hs : 0<s) (ha : 0<a) (ha1 : a≤1) (hb : 0<b)
  12     (hε : 0<ε) (hε1 : ε≤1) (hη : 0≤η)
  13     (hρ : a≤ρ ∧ ρ≤1) (hr : 1+ε≤r) (hrM : r≤M)
  14     (hl : r*s≤l) (hlM : l≤M*s)
  15     (hκ : b≤1-s*ρ/l)
  16     (herror : η*M*(M+2)≤(a*ε/4)*b^(k-1))
  17     (hl' : l*(1-s*ρ/l)^k-η*l≤l')
  18     (hs' : s'≤s*(1-ρ)*(1-s*ρ/l)^(k-1)+η*l) :
  19     (r+a*ε/4)*s'≤l' := by
  20   let c := a*ε/4
  21   have hc : 0<c := by dsimp [c];positivity
  22   have hc1 : c≤1 := by
  23     dsimp [c]
  24     nlinarith [mul_le_mul ha1 hε1 hε.le (by norm_num : (0 : ℝ)≤1)]
  25   have hr0 : 0≤r := by linarith
  26   have hM0 : 0≤M := hr0.trans hrM
  27   have hl0 : 0<l := lt_of_lt_of_le (mul_pos (by linarith : 0<r) hs) hl
  28   have hρ0 : 0≤ρ := ha.le.trans hρ.1
  29   have hpow : b^(k-1)≤(1-s*ρ/l)^(k-1) := pow_le_pow_left₀ hb.le hκ _
  30   have hp0 : 0≤(1-s*ρ/l)^(k-1) := pow_nonneg (hb.le.trans hκ) _
  31   have hcoeff : 3*c≤(r-1)*ρ-c*(1-ρ) := by
  32     have h₁ := mul_le_mul_of_nonneg_right (show ε≤r-1 by linarith) hρ0
  33     have h₂ := mul_le_mul_of_nonneg_left hρ.1 hε.le
  34     have h₃ := mul_nonneg hc.le hρ0
  35     dsimp [c] at *
  36     nlinarith
  37   have hmargin : 3*c*s*b^(k-1)≤
  38       l*(1-s*ρ/l)^k-(r+c)*s*(1-ρ)*(1-s*ρ/l)^(k-1) := by
  39     have hid : l*(1-s*ρ/l)^k-(r+c)*s*(1-ρ)*(1-s*ρ/l)^(k-1)=
  40         (l-r*s+s*((r-1)*ρ-c*(1-ρ)))*(1-s*ρ/l)^(k-1) := by
  41       rw [show (1-s*ρ/l)^k=(1-s*ρ/l)^(k-1)*(1-s*ρ/l) from by
  42         rw [← pow_succ,Nat.sub_add_cancel (by omega : 1≤k)]]
  43       have he : l*(1-s*ρ/l)=l-s*ρ := by field_simp
  44       calc
  45         _ = (l*(1-s*ρ/l)-(r+c)*s*(1-ρ))*(1-s*ρ/l)^(k-1) := by ring
  46         _ = _ := by rw [he];ring
  47     rw [hid]
  48     calc
  49       _ ≤ (3*c*s)*(1-s*ρ/l)^(k-1) :=
  50         mul_le_mul_of_nonneg_left hpow (by positivity)
  51       _ ≤ (l-r*s+s*((r-1)*ρ-c*(1-ρ)))*(1-s*ρ/l)^(k-1) := by
  52         apply mul_le_mul_of_nonneg_right _ hp0
  53         have hh := mul_le_mul_of_nonneg_left hcoeff hs.le
  54         nlinarith
  55   have hloss : (r+c+1)*η*l≤c*s*b^(k-1) := by
  56     calc
  57       _ ≤ (M+2)*η*l := mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_right (by linarith) hη) hl0.le
  58       _ ≤ (M+2)*η*(M*s) := mul_le_mul_of_nonneg_left hlM (by positivity)
  59       _ = (η*M*(M+2))*s := by ring
  60       _ ≤ (c*b^(k-1))*s := mul_le_mul_of_nonneg_right herror hs.le
  61       _ = _ := by ring
  62   have hs'' := mul_le_mul_of_nonneg_left hs' (show 0≤r+c by positivity)
  63   have hposit : 0≤c*s*b^(k-1) := by positivity
  64   change (r+c)*s'≤l'
  65   nlinarith
