  11 theorem ordinary_actual_frequency_marginals
  12     (K:ℕ) (hK:0<K) (s θ c long:ℝ) (f:Fin K→ℝ)
  13     (hs0:0≤s) (hs1:s≤1/100) (hθ0:0≤θ) (hθ1:θ≤1/100) (hc:8≤c)
  14     (hlong:0≤long) (hf:∀ i,0≤f i)
  15     (hfreq:(∑ i,f i)+long=1-s):
  16     let cutoff:=θ/(100*(K:ℝ))
  17     let longKept:=if long<θ/100 then 0 else long
  18     let shortKept:=∑ i:Fin K,(if f i<cutoff then 0 else f i)
  19     let omitted:=(∑ i:Fin K,(if f i<cutoff then f i else 0))+
  20       (if long<θ/100 then long else 0)
  21     let a:=(1-θ)*longKept/c
  22     let b:=(c-1)*a+(1-θ)*shortKept
  23     let u:=1-s-a-b;
  24     0≤omitted ∧ omitted≤θ/50 ∧
  25     u=θ*(1-s)+(1-θ)*omitted ∧ θ/2≤u ∧ u≤2*θ ∧
  26     0≤a ∧ a≤1/c ∧ 1/2≤b ∧ s+a+b+u=1 := by
  27   intro cutoff longKept shortKept omitted a b u
  28   have hshort:=ordinary_discarded_short_frequency K hK θ hθ0 f hf
  29   have hlong0:0≤(if long<θ/100 then long else 0):=by split_ifs <;> positivity
  30   have hlongmax:(if long<θ/100 then long else 0)≤θ/100:=by
  31     split_ifs with h
  32     · exact h.le
  33     · positivity
  34   have homit0:0≤omitted:=by dsimp [omitted,cutoff];linarith [hshort.1]
  35   have homitmax:omitted≤θ/50:=by dsimp [omitted,cutoff];linarith [hshort.2]
  36   have hL:0≤longKept:=by dsimp [longKept];split_ifs <;> positivity
  37   have hS:0≤shortKept:=by
  38     dsimp [shortKept]
  39     apply Finset.sum_nonneg
  40     intro i _
  41     split_ifs <;> simp [hf]
  42   have hsplit:(∑ i:Fin K,(if f i<cutoff then 0 else f i))+
  43       (∑ i:Fin K,(if f i<cutoff then f i else 0))=∑ i:Fin K,f i:=by
  44     rw [←Finset.sum_add_distrib]
  45     apply Finset.sum_congr rfl
  46     intro i _
  47     split_ifs <;> ring
  48   have hlongsplit:longKept+(if long<θ/100 then long else 0)=long:=by dsimp [longKept];split_ifs <;> ring
  49   have htotal:longKept+shortKept+omitted=1-s:=by dsimp [shortKept,omitted];linarith
  50   have hh:=ordinary_frequency_marginal_bounds s θ c longKept shortKept omitted
  51     hs0 hs1 hθ0 hθ1 hc hL hS homit0 homitmax htotal
  52   exact ⟨homit0,homitmax,hh⟩
  53 end Oberwolfach
