  12 theorem factor_component_has_forward_cycle
  13     {X:Type} [Fintype X] [DecidableEq X]
  14     (G:SimpleGraph X) (hG:G.IsRegularOfDegree 2)
  15     (E:∀C:G.ConnectedComponent,SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph)
  16     (C:G.ConnectedComponent):
  17     ∃(z:X) (p:G.Walk z z),p.IsCycle ∧ p.length=Fintype.card C ∧
  18       (∀k<p.length,factorCycleOrientation G E (p.getVert k)=p.getVert (k+1)) ∧
  19       p.toSubgraph.verts=C.supp:=by
  20   classical
  21   have hN:3≤Fintype.card C:=(component_regular_two_cycle_iso G hG C).1
  22   obtain ⟨z,p,hp,hlen,hstep⟩:=forward_cycle_graph_walk (Fintype.card C) hN
  23   let f:SimpleGraph.cycleGraph (Fintype.card C) →g G :=
  24     ⟨fun i=>(E C i).val,by
  25       intro x y hxy
  26       exact ((E C).map_rel_iff).2 hxy⟩
  27   have hfinj:Function.Injective f:=by
  28     intro x y hxy
  29     apply (E C).injective
  30     exact Subtype.ext hxy
  31   let q:=p.map f
  32   refine ⟨f z,q,hp.map hfinj,by simpa [q] using hlen,?_,?_⟩
  33   · intro k hk
  34     have hk':k<p.length:=by simpa [q] using hk
  35     simp only [q,SimpleGraph.Walk.getVert_map]
  36     change factorCycleOrientation G E (E C (p.getVert k)).val=(E C (p.getVert (k+1))).val
  37     rw [factor_cycle_orientation_on_component G E C (E C (p.getVert k))]
  38     simp only [RelIso.symm_apply_apply]
  39     rw [hstep k hk']
  40   · ext x
  41     constructor
  42     · intro hx
  43       have hmem:x∈q.support:=q.mem_verts_toSubgraph.mp hx
  44       obtain ⟨i,hi,hxi⟩:=List.mem_map.mp (by simpa only [q,SimpleGraph.Walk.support_map] using hmem)
  45       rw [←hxi]
  46       exact (E C i).property
  47     · intro hx
  48       let i:Fin (Fintype.card C):=(E C).symm ⟨x,hx⟩
  49       have hfull:p.support.dropLast.toFinset=Finset.univ:=by
  50         apply Finset.eq_univ_of_card
  51         rw [List.toFinset_card_of_nodup hp.nodup_dropLast_support,List.length_dropLast,p.length_support,hlen,Fintype.card_fin]
  52         omega
  53       have hi:i∈p.support:=List.mem_of_mem_dropLast (List.mem_toFinset.mp (by rw [hfull];exact Finset.mem_univ _))
  54       apply q.mem_verts_toSubgraph.mpr
  55       change x∈(p.map f).support
  56       rw [SimpleGraph.Walk.support_map]
  57       apply List.mem_map.mpr
  58       refine ⟨i,hi,?_⟩
  59       change (E C i).val=x
  60       simp [i]
  61 end Oberwolfach
