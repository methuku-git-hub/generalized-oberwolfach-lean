  12 theorem degree_two_image_union_bound (H:I→SimpleGraph V) (occ:I→V→Prop)
  13     (hdeg:∀ i v,(H i).degree v≤(if occ i v then 2 else 0)) (v:V):
  14     ((Finset.univ.biUnion (fun i=>(H i).neighborFinset v)).card:ℕ)≤
  15       2*(Finset.univ.filter (fun i=>occ i v)).card := by
  16   calc
  17     _ ≤ ∑ i,(H i).degree v:=by
  18       have hh:=Finset.card_biUnion_le (s:=Finset.univ) (t:=fun i=>(H i).neighborFinset v)
  19       simpa only [SimpleGraph.card_neighborFinset_eq_degree] using hh
  20     _ ≤ ∑ i,(if occ i v then 2 else 0):=Finset.sum_le_sum (fun i _=>hdeg i v)
  21     _ = 2*(Finset.univ.filter (fun i=>occ i v)).card:=by simp [Finset.sum_ite, Nat.mul_comm]

  23 theorem bounded_degree_two_union_from_occurrences (H:I→SimpleGraph V) (occ:I→V→Prop)
  24     (hdeg:∀ i v,(H i).degree v≤(if occ i v then 2 else 0))
  25     (G:SimpleGraph V) (hcover:∀ v,(G.neighborFinset v)⊆Finset.univ.biUnion (fun i=>(H i).neighborFinset v))
  26     (M:ℕ) (hocc:∀ v,(Finset.univ.filter (fun i=>occ i v)).card≤M):
  27     ∀ v,G.degree v≤2*M := by
  28   intro v
  29   rw [←SimpleGraph.card_neighborFinset_eq_degree]
