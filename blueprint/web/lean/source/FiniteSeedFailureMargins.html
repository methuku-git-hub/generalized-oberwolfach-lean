   8 theorem eventually_polynomial_exponential_failure_below
   9     (C c δ:ℝ) (hc:0<c) (hδ:0<δ):
  10     ∃N:ℕ,∀n:ℕ,N≤n→C*(n:ℝ)^2*Real.exp (-c*(n:ℝ))<δ :=by
  11   obtain ⟨N₁,hN₁⟩:=eventually_stretchedExp_le_inv_pow C 2 (c:=c) (p:=1) hc (by norm_num) 1
  12   obtain ⟨N₂,hN₂⟩:=exists_nat_gt (1/δ)
  13   refine ⟨max N₁ N₂,?_⟩
  14   intro n hn
  15   have hn1:N₁≤n:=le_trans (le_max_left _ _) hn
  16   have hn2:N₂≤n:=le_trans (le_max_right _ _) hn
  17   have hn2r:(N₂:ℝ)≤n:=by exact_mod_cast hn2
  18   have hnpos:0<(n:ℝ):=by
  19     have hh:0<1/δ:=by positivity
  20     linarith
  21   have hinv:1/(n:ℝ)<δ:=by
  22     apply (div_lt_iff₀ hnpos).2
  23     have hh:(1/δ)<(n:ℝ):=lt_of_lt_of_le hN₂ hn2r
  24     have hh':1<δ*(n:ℝ):=by simpa [mul_comm] using (div_lt_iff₀ hδ).1 hh
  25     nlinarith
  26   have h:=hN₁ n hn1
  27   simp only [Real.rpow_two,Real.rpow_one,pow_one] at h
  28   exact h.trans_lt hinv
