  24     (hcount:|((restrictedResourceSubset e S).card:ℝ)-s*n|≤countErr)
  25     (hweight:∀j c,0≤w j c)
  26     (M:Finset (RestrictedResource (mixedCandidateResources R) e)):
  27     let g:=mixedHubIndicator hub i u
  28     let mass:=∑c:RestrictedResource (mixedCandidateResources R) e,
  29       g c.val.1 c.val.2*mixedCandidateWeight w c.val

  31     let error:=(restrictedResourceSubset e S).card*err+η*countErr
