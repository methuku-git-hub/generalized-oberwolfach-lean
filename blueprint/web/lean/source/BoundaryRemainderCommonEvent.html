  14 theorem boundary_remainder_selection_common_event (τ κ:ℝ)
  15     (hτ:0<τ) (hτ1:τ≤1) (hκ:0<κ) (hκ1:κ≤1) (q:ℕ):
  16     ∃ N:ℕ,∀ (V J:Type) [Fintype V] [DecidableEq V] [Fintype J] [DecidableEq J],
  17       N≤Fintype.card V→∀ (G:SimpleGraph V) (A Z:Finset V) (x:J↪V),
  18       (∀ j,x j∈Z)→κ*Fintype.card V≤(A.card:ℝ)→
  19       1-1/(Fintype.card V:ℝ)^q≤
  20         (FiniteLaw.pi (fun _:Sym2 V=>FiniteLaw.retentionCoin τ hτ.le hτ1)).probability
  21           (fun σ=>∀ (C D G':SimpleGraph V),G'=(G\C)\D→
  22             Disjoint C (retainedGraph G σ)→∀ ℓ d:ℝ,0≤ℓ→0≤d→
  23             (∀ v,(((G\retainedGraph G σ)\C).degree v:ℝ)≤ℓ)→
  24             (∀ v,(D.degree v:ℝ)≤d)→ℓ+d≤τ^2*A.card/4→
  25             (∀ v,((changed (A∩(retainedGraph G σ).neighborFinset v)
  26               (A∩G'.neighborFinset v)).card:ℝ)≤ℓ+d) ∧
  27             (∀ i,((innerBoundarySupport G' A (x i)).card:ℝ)≤2*τ*Fintype.card A) ∧
  28             (∀ i j,i≠j→((innerBoundarySupport G' A (x i)∩innerBoundarySupport G' A (x j)).card:ℝ)≤
  29               (3*τ/κ)^2*Fintype.card A) ∧
  30             (∀ v:A,((Finset.univ.filter (fun i=>v∈innerBoundarySupport G' A (x i))).card:ℝ)≤
  31               (3*τ/κ)*Fintype.card A)):=by

  33   have hε:0<ε:=by dsimp [ε];positivity
  34   obtain ⟨N,hN⟩:=fresh_boundary_support_common_event ε hε q

  38   apply hevent.trans

  42     τ (ε*Fintype.card V) ℓ d (retainedGraph_le G σ) hCR hleft hcover
  43     (fun v=>(hσ.1 v).le) (fun v w hvw=>(hσ.2.1 v w hvw).le) (fun v hv=>(hσ.2.2 v hv).le)
  44   have hεsmall:ε*Fintype.card V≤τ^2*A.card/4:=by

  53       (retainedGraph_le G σ) hCR A v

  55         changed (A∩(retainedGraph G σ).neighborFinset v) (A∩((G\C)\D).neighborFinset v):=by
  56       ext w;simp only [changed,Finset.mem_union,Finset.mem_sdiff,Finset.mem_inter,SimpleGraph.mem_neighborFinset,hrem]
  57     rw [he]
  58     have hh':((changed (A∩(retainedGraph G σ).neighborFinset v) (A∩((G\C)\D).neighborFinset v)).card:ℝ)≤
