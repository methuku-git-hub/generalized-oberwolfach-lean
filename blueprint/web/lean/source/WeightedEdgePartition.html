  13 theorem weighted_edge_partition_exists
  14     {V E I C A : Type} [Fintype V] [Fintype E] [Fintype I] [Fintype C] [Fintype A]
  15     [Nonempty C] (R : E → Finset V) (part : E → I)
  16     (w : A → E → ℝ) (b : A → ℝ) (W : I → A → ℝ)
  17     (d β : ℝ) (D B : ℕ) (T : I → A → ℝ) (s₀ s₁ s₂ : ℝ)
  18     (hb : ∀ a,0<b a) (hw : ∀ a e,0≤w a e ∧ w a e≤b a)
  19     (hdegree : ∀ i v,((Finset.univ.filter (fun e=>part e=i ∧ v∈R e)).card : ℝ)≤d)
  20     (hcodeg : ∀ i v u,v≠u →
  21       ((Finset.univ.filter (fun e=>part e=i ∧ v∈R e ∧ u∈R e)).card : ℝ)≤β)
  22     (hβ : 0≤β) (hW : ∀ i a,(∑ e,if part e=i then w a e else 0)≤W i a)
  23     (hs₀ : 0≤s₀) (hs₁ : 0≤s₁) (hs₂ : 0≤s₂)
  24     (hsmall :
  25       (∑ _i : I,∑ _v : V,∑ _c : C,Real.exp (d/Fintype.card C*(Real.exp s₀-1)-s₀*((D : ℝ)+1)))+

  27       (∑ i : I,∑ a : A,∑ _c : C,Real.exp ((W i a/Fintype.card C*(Real.exp s₂-1)-s₂*T i a)/b a)))<1) :
  28     ∃ split : E → C,
  29       (∀ i c v,(Finset.univ.filter (fun e=>part e=i ∧ split e=c ∧ v∈R e)).card≤D) ∧
  30       (∀ i c v u,v≠u →
  31         (Finset.univ.filter (fun e=>part e=i ∧ split e=c ∧ v∈R e ∧ u∈R e)).card≤B) ∧
  32       (∀ i a c,(∑ e,if part e=i ∧ split e=c then w a e else 0)<T i a) := by

  36     (fun iv e=>if part e=iv.1 ∧ iv.2∈R e then 1 else 0)

  51     · exact hb ia.2

  54     rcases j with iv|ivu|ia <;> simp only [f,bound,Sum.elim_inl,Sum.elim_inr]

  58       · exact hw ia.2 e

  69     · exact div_le_div_of_nonneg_right (hW ia.1 ia.2) (Nat.cast_nonneg _)
