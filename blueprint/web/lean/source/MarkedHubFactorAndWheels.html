  19     ∃ η : ℝ,0<η ∧ ∃ N : ℕ,∀ (A U E K : Type) [Fintype A] [DecidableEq A] [DecidableEq U],
  20       N≤Fintype.card A →
  21       ∀ (tail head : E → A) (ktail khead : K → A) (π : Equiv.Perm A)
  22         (b : ℕ) (H₀ Hₖ : A → U → Prop) (w : U),1≤b →
  23       Function.Injective (markedPhysicalEdge tail head ktail khead π) →
  24       (∀ v,¬(H₀ v w ∧ Hₖ v w)) →
  25       (∀ v z,H₀ v w → Hₖ z w → v≠π z) →
  26       β*Fintype.card A≤Fintype.card (hubNeighbourClass H₀ Hₖ w true) →
  27       Fintype.card (hubNeighbourClass H₀ Hₖ w false)=m*Fintype.card (hubNeighbourClass H₀ Hₖ w true) →
  28       β*(Fintype.card A:ℝ)^(m+1)≤Fintype.card (HubMarkedCycle tail head ktail khead m π b H₀ Hₖ w) →
  29       (∀ j : Bool,∀ v v' : A,v≠v' →
  30         (if j then Hₖ v w else H₀ v w) → (if j then Hₖ v' w else H₀ v' w) →
  31         β*(Fintype.card A:ℝ)^m≤Fintype.card (RootFirstCommonMarkedPath tail head ktail khead m π b H₀ Hₖ j v v' w w)) →
  32       ∀ wt : HubMarkedCycle tail head ktail khead m π b H₀ Hₖ w → ℝ,
  33       (∀ C,bLo/(Fintype.card A:ℝ)^m≤wt C ∧ wt C≤bHi/(Fintype.card A:ℝ)^m) →
  34       (∀ v,H₀ v w ∨ Hₖ v w → |1-hypergraphWeightedDegree (hubMarkedCycleVertices w) wt v|≤η) →
  35       ∃ M : Finset (HubMarkedCycle tail head ktail khead m π b H₀ Hₖ w),
  36         (∀ C∈M,∀ D∈M,C≠D → Disjoint (hubMarkedCycleVertices w C) (hubMarkedCycleVertices w D)) ∧
  37         M.biUnion (hubMarkedCycleVertices w)=Finset.univ.filter (fun v => H₀ v w ∨ Hₖ v w) ∧
  38         (∀ v,H₀ v w ∨ Hₖ v w → ∃! C : M,Sum.inr (v,w)∈markedWheelPhysicalResources (hubCycleWheel w C.val)) ∧
  39         (∀ C∈M,∀ D∈M,C≠D → Disjoint (hubCyclePhysicalEdges w C) (hubCyclePhysicalEdges w D)) ∧
  40         (∀ C∈M,∀ e∈hubCyclePhysicalEdges w C,e.card=2) ∧
  41         ∀ v,(physicalEdgeGraph (M.biUnion (hubCyclePhysicalEdges w))).degree v≤2 := by

  48   let e := hubNeighbourEmbedding H₀ Hₖ w hd

  50   have hnpos : (0:ℝ)<Fintype.card A := by linarith [le_max_left (1:ℝ) ((k₀:ℝ)/β)]
  51   have hkpos : (0:ℝ)<k := (mul_pos hβ hnpos).trans_le hsize
  52   have hk₀ : k₀≤k := by

  66     rcases v with ⟨i,v⟩

  69     · exact Or.inr v.property
