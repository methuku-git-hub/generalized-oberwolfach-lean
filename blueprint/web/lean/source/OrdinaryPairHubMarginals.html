  21     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ) (hd:2≤d)
  22     (ρ:Sym2 V→ℝ) (hρ:∀ e,0≤ρ e) (hρ1:∀ e,ρ e≤1)
  23     (q:U→Γ→ℝ) (hq:∀ u g,0≤q u g) (hQ:∀ u,∑ g,q u g≤1/8)
  24     (s:U→Color→ℝ) (hs:∀ u j,0≤s u j)
  25     (hS:∀ u,∑ j,s u j ≤ intervalUncoveredProbability d (∑ g,q u g))
  26     (hnoshort:∀ x k,0<k→k<2*d+1→(π^k) x≠x)
  27     (j:Color) (u u':U) (hne:u≠u') (a:Sym2 V⊕(V×Unit)):
  28     (FiniteLaw.pi (rawWheelLaw ρ hρ hρ1 d hd q hq hQ s hs hS)).probability

  32   cases a with
  33   | inl e=>
  34     have hh:=rawWheelResource_probability P π ρ hρ hρ1 d hd q hq hQ s hs hS hnoshort j (Sum.inl e)
  35     simpa [ordinaryResourceState,ordinaryPairHubMarginal,rawWheelMarginal] using hh
  36   | inr a=>
  37     rcases a with ⟨v,⟨⟩⟩
  38     let a:Sym2 V⊕(V×U):=Sum.inr (v,u)
  39     let b:Sym2 V⊕(V×U):=Sum.inr (v,u')
  40     have hab:a≠b:=by intro h;exact hne (congrArg Prod.snd (Sum.inr.inj h))
  41     have hdis:=rawWheelResourceCoordinates_disjoint π d a b hab (by
  42       intro x h y k hx hy
  43       have hx':(x,h)=(v,u):=Sum.inr.inj hx.symm
  44       have hy':(y,k)=(v,u'):=Sum.inr.inj hy.symm
  45       left
  46       have hh:h=u:=congrArg Prod.snd hx'
  47       have hk:k=u':=congrArg Prod.snd hy'
  48       simpa only [hh,hk] using hne)
  49     let Ev:=fun (z:RawWheelOutcome V U Γ Color)=>rawWheelSpokes π d z j v u

  51     have hEv:∀ z z',(∀ e∈rawWheelResourceCoordinates π d a,z e=z' e)→(Ev z↔Ev z'):=by
  52       intro z z' hz
  53       have hh:=rawWheelResourceState_supported P π d j a z z' hz
  54       simpa [a,Ev,ordinaryResourceState,decide_eq_decide] using hh

  56       intro z z' hz
  57       have hh:=rawWheelResourceState_supported P π d j b z z' hz
  58       simpa [b,Ev',ordinaryResourceState,decide_eq_decide] using hh
  59     have hind:=finiteLaw_disjoint_supported_events μ _ _ hdis Ev Ev' hEv hEv'
  60     have hp:=rawWheelResource_probability P π ρ hρ hρ1 d hd q hq hQ s hs hS hnoshort j a
  61     have hp':=rawWheelResource_probability P π ρ hρ hρ1 d hd q hq hQ s hs hS hnoshort j b
  62     have hp0:(FiniteLaw.pi μ).probability Ev=s u j:=by
  63       simpa [μ,Ev,a,ordinaryResourceState,rawWheelMarginal] using hp
  64     have hp0':(FiniteLaw.pi μ).probability Ev'=s u' j:=by
  65       simpa [μ,Ev',b,ordinaryResourceState,rawWheelMarginal] using hp'
  66     rw [hp0,hp0'] at hind
