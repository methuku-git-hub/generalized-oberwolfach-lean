  19 def markedReferenceDomain (P : SimpleGraph V) (π : Equiv.Perm V) : Option (V⊕V)→Finset V
  20   | none=>Finset.univ
  21   | some (Sum.inl x)=>P.neighborFinset x
  22   | some (Sum.inr x)=>Finset.univ.filter (fun v=>P.Adj x (π v))
