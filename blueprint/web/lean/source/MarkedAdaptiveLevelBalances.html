   7 theorem marked_adaptive_level_balances
   8     (ρ s₀ m Dbal Dload B H:ℝ) (hm:0<m)
   9     (hcentre:(ρ*s₀/2)*Dbal=s₀*B)
  10     (hload:H*s₀=Dload*(ρ*s₀/2)):
  11     (ρ*(s₀/m)/2)*Dbal=(s₀/m)*B ∧
  12     H*(s₀/m)=Dload*(ρ*(s₀/m)/2) ∧
  13     H*(s₀/m)*m=Dload*(ρ*s₀/2) := by
  14   have hcentre':=((congrArg (fun x:ℝ=>x/m) hcentre))
  15   have hload':=((congrArg (fun x:ℝ=>x/m) hload))
  16   refine ⟨?_,?_,?_⟩
  17   · convert hcentre' using 1 <;> ring
  18   · convert hload' using 1 <;> ring
  19   · have hm0:m≠0:=ne_of_gt hm
  20     field_simp
  21     nlinarith only [hload]
  22 end Oberwolfach
