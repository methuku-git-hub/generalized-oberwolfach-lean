  12 theorem fractional_resource_matching_positive (r B:ℕ) (hr:2≤r) (hB:1≤B)
  13     (K a γ:ℝ) (hK:0<K) (ha:0<a) (hγ:0<γ):
  14     ∃η₀:ℝ,0<η₀ ∧ ∃n₀:ℕ,∀n:ℕ,n₀≤n →
  15     ∀(V E A I:Type) [Fintype V] [Fintype E] [Fintype A] [Fintype I] [DecidableEq V],
  16     ∀(R:E→Finset V) (w:E→ℝ) (g:A→E→ℝ) (S:I→Finset V),
  17     Fintype.card V≤n^B → Fintype.card A≤n^B → Fintype.card I≤n^B →
  18     (∀e,0<w e→2≤(R e).card ∧ (R e).card≤r) → (∀e,0≤w e) →
  19     (∀v,1-η₀≤hypergraphWeightedDegree R w v ∧ hypergraphWeightedDegree R w v≤1+η₀) →
  20     (∀u v,u≠v → (∑e,if u∈R e ∧ v∈R e then w e else 0)≤K/n) →
  21     (∀i,a*n≤(S i).card) → (∀i e,0≤g i e ∧ g i e≤1) →
  22     (∀i,a*n≤∑e,g i e*w e) →
  23     ∃M:Finset E,(∀e∈M,0<w e) ∧ (∀e∈M,∀f∈M,e≠f→Disjoint (R e) (R f)) ∧
  24       (∀i,((S i\M.biUnion R).card:ℝ)≤γ*(S i).card) ∧
  25       (∀i,(1-γ)*(∑e,g i e*w e)≤∑e∈M,g i e) ∧
  26       (∀i,(∑e∈M,g i e)≤(1+γ)*(∑e,g i e*w e)) := by
  27   obtain ⟨η,hη,n₀,hmatch⟩:=fractional_resource_matching r B hr hB K a γ hK ha hγ
  28   refine ⟨η,hη,n₀,?_⟩
  29   intro n hn V E A I _ _ _ _ _ R w g S hV hA hI hsize hw hdeg hpair hS hg hmass
  30   let R' : PositiveResource w→Finset V := fun e=>R e.val
  31   let w' : PositiveResource w→ℝ := fun e=>w e.val
  32   let g' : A→PositiveResource w→ℝ := fun i e=>g i e.val
  33   have hdeg':∀v,1-η≤hypergraphWeightedDegree R' w' v ∧ hypergraphWeightedDegree R' w' v≤1+η := by
  34     intro v
  35     rw [show hypergraphWeightedDegree R' w' v=hypergraphWeightedDegree R w v from positive_resource_degree R w hw v]
  36     exact hdeg v
  37   have hpair':∀u v,u≠v → (∑e:PositiveResource w,if u∈R' e ∧ v∈R' e then w' e else 0)≤K/n := by
  38     intro u v huv
  39     rw [show (∑e:PositiveResource w,if u∈R' e ∧ v∈R' e then w' e else 0)=
  40       ∑e,if u∈R e ∧ v∈R e then w e else 0 from positive_resource_codegree R w hw u v]
  41     exact hpair u v huv
  42   have hmass':∀i,a*n≤∑e:PositiveResource w,g' i e*w' e:=by
  43     intro i
  44     rw [show (∑e:PositiveResource w,g' i e*w' e)=∑e,g i e*w e from positive_resource_weighted_sum w hw (g i)]
  45     exact hmass i
  46   obtain ⟨M,hdis,hcover,hlower,hupper⟩:=hmatch n hn V (PositiveResource w) A I R' w' g' S
  47     hV hA hI (fun e=>hsize e.val e.property) (fun e=>(e.property).le) hdeg' hpair' hS (fun i e=>hg i e.val) hmass'
  48   let N:Finset E:=M.image Subtype.val
  49   have hsum:∀i,(∑e∈N,g i e)=∑e∈M,g' i e:=by
  50     intro i
  51     dsimp [N,g']
  52     rw [Finset.sum_image]
  53     intro a _ b _ hab
  54     exact Subtype.ext hab
  55   have hunion:N.biUnion R=M.biUnion R':=by
  56     ext v
  57     simp [N,R',Finset.mem_biUnion]
  58   refine ⟨N,?_,?_,?_,?_,?_⟩
  59   · intro e he
  60     obtain ⟨x,_,rfl⟩:=Finset.mem_image.mp he
  61     exact x.property
  62   · intro e he f hf hne
  63     obtain ⟨x,hx,rfl⟩:=Finset.mem_image.mp he
  64     obtain ⟨y,hy,rfl⟩:=Finset.mem_image.mp hf
  65     exact hdis x hx y hy (fun h=>hne (congrArg Subtype.val h))
