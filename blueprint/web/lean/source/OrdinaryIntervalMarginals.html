  10 theorem ordinary_interval_marginals_feasible (s a b u θ:ℝ) (d:ℕ) (hd:2≤d)
  11     (hθ:0<θ) (hda:1/θ≤d) (hs:0≤s) (ha:0≤a) (ha8:a≤1/8) (hb:0≤b)
  12     (hu:θ/2≤u) (hrow:s+a+b+u=1):
  13     let q:Bool→ℝ:=fun j=>if j then a else 0
  14     let t:Bool→ℝ:=fun j=>if j then b else s;
  15     (∀ j,0≤q j) ∧ (∑ j,q j)≤1/8 ∧ (∀ j,0≤t j) ∧
  16     (∑ j,t j) ≤ intervalUncoveredProbability d (∑ j,q j) := by
  17   intro q t
  18   have hd0:(0:ℝ)<d:=by exact_mod_cast (show 0<d by omega)

  20   have habound:a/(d:ℝ)≤θ/8:=by
  21     apply (div_le_iff₀ hd0).2
  22     nlinarith
  23   have hroom:a/(d:ℝ)≤u:=by linarith
  24   refine ⟨?_,?_,?_,?_⟩
  25   · intro j;cases j <;> simp [q,ha]
  26   · simpa [q] using ha8
  27   · intro j;cases j <;> simp [t,hs,hb]
  28   · simp [q,t,intervalUncoveredProbability]
