  11 theorem ordinary_relative_transformer_budgets (β ε d:ℝ) (hβ:0<β) (hd:0≤d)
  12     (hε:ε≤β/4) (r m f:ℕ) :
  13     ∃ n₀:ℕ,∀ n:ℕ,n₀≤n→∀ W R B e s v:ℕ,
  14       W≤f→R≤m→e≤m*m→s≤m*m→v≤2*(m*m)→
  15       (B:ℝ)≤ε*n+f→
  16       (W:ℝ)+((R:ℝ)+(e+s)*(r-1:ℕ)+r+7)*(d+1)<(β/2)*n ∧
  17       (W:ℝ)+(e+s)*(r-1:ℕ)+v*r+((r:ℝ)+4)*(d+1)<(β/2)*n ∧
  18       (W:ℝ)+(e+s)*(r-1:ℕ)+2*(r:ℝ)*v≤f+2*(m*m)*(r-1:ℕ)+4*(r:ℝ)*(m*m) ∧
  19       (B:ℝ)+e<(β/2)*n := by
  20   let P:ℝ:=f+(m+2*(m*m)*(r-1:ℕ)+r+7)*(d+1)+2*(m*m)*(r-1:ℕ)+2*(m*m)*r+((r:ℝ)+4)*(d+1)+f+m*m+1
  21   have hP:0<P:=by dsimp [P];positivity
  22   obtain ⟨n₀,hn₀⟩:=exists_nat_gt (4*P/β)
  23   refine ⟨n₀,?_⟩
  24   intro n hn W R B e s v hW hR he hs hv hB
  25   have hnR:(n₀:ℝ)≤n:=by exact_mod_cast hn
  26   have hlarge:4*P<β*n:=by simpa only [mul_comm] using (div_lt_iff₀ hβ).mp (hn₀.trans_le hnR)
  27   have hn0:0≤(n:ℝ):=by positivity
  28   have hW':(W:ℝ)≤f:=by exact_mod_cast hW
  29   have hR':(R:ℝ)≤m:=by exact_mod_cast hR
  30   have he':(e:ℝ)≤m*m:=by exact_mod_cast he
  31   have hs':(s:ℝ)≤m*m:=by exact_mod_cast hs
  32   have hv':(v:ℝ)≤2*(m*m):=by exact_mod_cast hv
  33   have hsum:(e:ℝ)+s≤2*(m*m):=by linarith only [he',hs']
  34   have hsumr:=mul_le_mul_of_nonneg_right hsum (by positivity:(0:ℝ)≤(r-1:ℕ))
  35   have hvR:=mul_le_mul_of_nonneg_right hv' (by positivity:(0:ℝ)≤r)
  36   have hv2R:=mul_le_mul_of_nonneg_left hv' (by positivity:(0:ℝ)≤2*r)
  37   have hpar:=mul_le_mul_of_nonneg_right
  38     (show (R:ℝ)+((e:ℝ)+s)*(r-1:ℕ)+r+7≤m+2*(m*m)*(r-1:ℕ)+r+7 by linarith only [hR',hsumr])
  39     (by linarith:(0:ℝ)≤d+1)
  40   have hP₁:(f:ℝ)+(m+2*(m*m)*(r-1:ℕ)+r+7)*(d+1)≤P:=by
  41     dsimp [P]
  42     have h1:0≤2*(m*m:ℝ)*(r-1:ℕ)+2*(m*m)*r+((r:ℝ)+4)*(d+1)+f+m*m+1:=by positivity
  43     linarith only [h1]
  44   have hP₂:(f:ℝ)+2*(m*m)*(r-1:ℕ)+2*(m*m)*r+((r:ℝ)+4)*(d+1)≤P:=by
  45     dsimp [P]
  46     have h1:0≤((m:ℝ)+2*(m*m)*(r-1:ℕ)+r+7)*(d+1)+f+m*m+1:=by positivity
  47     linarith only [h1]
  48   have hP₃:(f:ℝ)+m*m≤P:=by
  49     dsimp [P]
  50     have h1:0≤(f:ℝ)+(m+2*(m*m)*(r-1:ℕ)+r+7)*(d+1)+2*(m*m)*(r-1:ℕ)+2*(m*m)*r+((r:ℝ)+4)*(d+1)+1:=by positivity
  51     linarith only [h1]
  52   have hεn:=mul_le_mul_of_nonneg_right hε hn0
