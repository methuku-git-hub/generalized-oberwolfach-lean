  12 theorem finite_weight_combination_test {I W : Type*} [Fintype I] [Fintype W]
  13     (a : I → ℝ) (T : I → W → ℝ) (t : W → ℝ) :
  14     (∑ w,(∑ i,a i*T i w)*t w)=∑ i,a i*(∑ w,T i w*t w) := by
  15   calc
  16     _ = ∑ w,∑ i,a i*(T i w*t w) := by
  17       apply Finset.sum_congr rfl
  18       intro w _
  19       rw [Finset.sum_mul]
  20       apply Finset.sum_congr rfl
  21       intro i _
  22       ring
  23     _ = ∑ i,∑ w,a i*(T i w*t w) := Finset.sum_comm
  24     _ = _ := by simp only [Finset.mul_sum]

  26 theorem ordinaryRimLoad_combination {V U I : Type*} [Fintype V] [Fintype U] [Fintype I]
  27     [DecidableEq V] [DecidableEq U]
  28     {Q F : SimpleGraph V} {H : V → U → Prop} {r : ℕ}
  29     (a : I → ℝ) (T : I → OrdinaryWheel Q F H r → ℝ) (x y : V) :
  30     ordinaryRimLoad (fun W => ∑ i,a i*T i W) x y=∑ i,a i*ordinaryRimLoad (T i) x y := by
  31   let t (W : OrdinaryWheel Q F H r) : ℝ := if (ordinaryWheelRim W).Adj x y then 1 else 0
  32   have ht (w : OrdinaryWheel Q F H r → ℝ) :
  33       (∑ W,w W*t W)=ordinaryRimLoad w x y := by
  34     unfold ordinaryRimLoad
  35     apply Finset.sum_congr rfl
  36     intro W _
  37     dsimp only [t]
  38     split_ifs <;> simp
  39   rw [← ht,finite_weight_combination_test]
  40   simp only [ht]
