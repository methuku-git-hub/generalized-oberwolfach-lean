  15 theorem bounded_fractional_matching_on_subset (P:SimpleGraph V) (S:Finset V) (β:ℝ)
  16     (hβ:0<β) (hβhalf:β<1/2) (hn:4/β≤(S.card:ℝ))
  17     (hdeg:∀ v∈S,(1/2+β)*S.card≤((S∩P.neighborFinset v).card:ℝ)) :
  18     ∃ weight:Sym2 V→ℝ,
  19       (∀ v∈S,∀ w∈S,0≤weight s(v,w) ∧ weight s(v,w)≤5/(β*S.card)) ∧
  20       (∀ v∈S,∑ w∈S,weight s(v,w)=1) ∧
  21       (∀ v w,¬P.Adj v w → weight s(v,w)=0) ∧
  22       (∀ v∈S,∀ w∈S,P.Adj v w → β/(5*S.card)≤weight s(v,w)) := by
  23   obtain ⟨w,hsym,hzero,hbound,hsum⟩:=bounded_perfect_fractional_matching
  24     (P.induce (S:Set V)) β hβ hβhalf (by simpa using hn) (by
  25       intro v
  26       rw [induced_degree_intersection]
  27       simpa only [Finset.coe_sort_coe,Fintype.card_coe,Finset.inter_comm] using hdeg v.val v.property)
  28   let f:V→V→ℝ:=fun v u=>if hv:v∈S then if hu:u∈S then w ⟨v,hv⟩ ⟨u,hu⟩ else 0 else 0
  29   have hf (v u:V):f v u=f u v:=by
  30     by_cases hv:v∈S <;> by_cases hu:u∈S
  31     · simp only [f,dite_eq_left hv,dite_eq_left hu]
  32       exact hsym _ _
  33     all_goals simp [f,hv,hu]
  34   let weight:Sym2 V→ℝ:=Sym2.lift ⟨f,hf⟩
  35   have he (v u:V):weight s(v,u)=f v u:=rfl
  36   have hcard:(0:ℝ)<S.card:=(div_pos (by norm_num) hβ).trans_le hn
  37   refine ⟨weight,?_,?_,?_,?_⟩
  38   · intro v hv u hu
  39     rw [he]
  40     simp only [f,dite_eq_left hv,dite_eq_left hu]
  41     by_cases ha:P.Adj v u
  42     · have hb:=hbound ⟨v,hv⟩ ⟨u,hu⟩ ha
  43       simp only [Finset.coe_sort_coe,Fintype.card_coe] at hb
  44       exact ⟨(by positivity : 0≤β/(5*S.card)).trans hb.1,hb.2⟩
  45     · rw [hzero ⟨v,hv⟩ ⟨u,hu⟩ ha]
  46       exact ⟨le_refl _,by positivity⟩
  47   · intro v hv
  48     have hh:=Finset.sum_subtype (F:=inferInstance) (p:=fun u:V=>u∈S)
  49       S (fun _=>Iff.rfl) (fun u=>weight s(v,u))
  50     rw [hh]
  51     calc
  52       _ = ∑ u:S,w ⟨v,hv⟩ u:=by
  53         apply Finset.sum_congr rfl
  54         intro u _
  55         simp only [he,f,dite_eq_left hv,dite_eq_left u.property]
  56         rfl
  57       _ = 1:=hsum ⟨v,hv⟩
  58   · intro v u ha
  59     rw [he]
  60     by_cases hv:v∈S <;> by_cases hu:u∈S
  61     · simp only [f,dite_eq_left hv,dite_eq_left hu]
  62       exact hzero ⟨v,hv⟩ ⟨u,hu⟩ ha
  63     all_goals simp [f,hv,hu]
  64   · intro v hv u hu ha
  65     rw [he]
  66     simp only [f,dite_eq_left hv,dite_eq_left hu]
  67     simpa only [Finset.coe_sort_coe,Fintype.card_coe] using (hbound ⟨v,hv⟩ ⟨u,hu⟩ ha).1
