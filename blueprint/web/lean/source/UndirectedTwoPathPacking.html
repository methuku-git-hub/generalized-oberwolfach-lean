  11 def twoPathEdges {V : Type*} [DecidableEq V] (x y z : V) : Finset (Sym2 V) :=

  14 theorem undirectedTwoPathPacking_next {I V : Type*} [Fintype V] [DecidableEq V]
  15     (S : Finset I) (x y z : I → V) (u v : V) (C : Finset V)
  16     (L : ℕ) (hL : 0<L) {a : ℝ}
  17     (hx : ∀ w,((S.filter (fun i => x i=w)).card:ℝ)≤a)
  18     (hy : ∀ w,((S.filter (fun i => y i=w)).card:ℝ)≤a)
  19     (hm : ∀ w,middleCount S z w≤L)
  20     (hlarge : (S.card:ℝ)/L+4*a+4*L<C.card) :
  21     ∃ w∈C,middleCount S z w<L ∧
  22       (u,w)∉packedArcs S x y z ∧ (w,v)∉packedArcs S x y z ∧
  23       (w,u)∉packedArcs S x y z ∧ (v,w)∉packedArcs S x y z := by
  24   classical
  25   let Fu := Finset.univ.filter (fun w => (w,u)∈packedArcs S x y z)
  26   let Fv := Finset.univ.filter (fun w => (v,w)∈packedArcs S x y z)
  27   have hFu : (Fu.card:ℝ)≤a+L := by
  28     have hh := packedArcs_in_bound S x y z u
  29     have hr : (Fu.card:ℝ)≤((S.filter (fun i => y i=u)).card:ℝ)+middleCount S z u := by
  30       exact_mod_cast hh
  31     have hm' : (middleCount S z u:ℝ)≤L := by exact_mod_cast hm u
  32     linarith [hy u]
  33   have hFv : (Fv.card:ℝ)≤a+L := by
  34     have hh := packedArcs_out_bound S x y z v
  35     have hr : (Fv.card:ℝ)≤((S.filter (fun i => x i=v)).card:ℝ)+middleCount S z v := by
  36       exact_mod_cast hh
  37     have hm' : (middleCount S z v:ℝ)≤L := by exact_mod_cast hm v
  38     linarith [hx v]
  39   obtain ⟨w,hw,hmw,hu,hv,hwu,hvw⟩ := twoPathPacking_next S x y z u v C Fu Fv L hL
  40     (hx u) (hy v) hm hFu hFv (by linarith)
  41   exact ⟨w,hw,hmw,hu,hv,by simpa [Fu] using hwu,by simpa [Fv] using hvw⟩

  43 theorem twoPathEdges_disjoint_of_unused {I V : Type*} [DecidableEq V]
  44     (S : Finset I) (x y z : I → V) (u v w : V)
  45     (huw : (u,w)∉packedArcs S x y z) (hwv : (w,v)∉packedArcs S x y z)
  46     (hwu : (w,u)∉packedArcs S x y z) (hvw : (v,w)∉packedArcs S x y z)
  47     (j : I) (hj : j∈S) :
  48     Disjoint (twoPathEdges u v w) (twoPathEdges (x j) (y j) (z j)) := by
  49   classical
  50   have h1 : (x j,z j)∈packedArcs S x y z := Finset.mem_biUnion.mpr ⟨j,hj,by simp [twoPathArcs]⟩
  51   have h2 : (z j,y j)∈packedArcs S x y z := Finset.mem_biUnion.mpr ⟨j,hj,by simp [twoPathArcs]⟩
  52   apply Finset.disjoint_left.mpr
  53   intro e he hej
  54   simp only [twoPathEdges,Finset.mem_insert,Finset.mem_singleton] at he hej
  55   rcases he with rfl | rfl <;> rcases hej with he | he <;>
  56     rw [Sym2.eq_iff] at he <;> rcases he with ⟨ha,hb⟩ | ⟨ha,hb⟩ <;>
  57     simp_all

  59 theorem undirectedTwoPathPacking_exists
  60     {I V : Type*} [Fintype I] [DecidableEq I]
  61     [Fintype V] [DecidableEq V] [Nonempty V]
  62     (x y : I → V) (C : I → Finset V)
  63     (L : ℕ) (hL : 0 < L) {a : ℝ}
  64     (hx : ∀ v, ((Finset.univ.filter (fun i => x i = v)).card : ℝ) ≤ a)
  65     (hy : ∀ v, ((Finset.univ.filter (fun i => y i = v)).card : ℝ) ≤ a)
  66     (hlarge : ∀ i, (Fintype.card I : ℝ) / L + 4 * a + 4 * L < (C i).card) :
  67     ∃ z : I → V,
  68       (∀ i, z i ∈ C i) ∧
  69       (∀ i j, i ≠ j → Disjoint (twoPathEdges (x i) (y i) (z i))
  70         (twoPathEdges (x j) (y j) (z j))) ∧
  71       ∀ v, middleCount Finset.univ z v ≤ L := by
  72   classical
  73   have hall (S : Finset I) : ∃ z : I → V,
  74       (∀ i ∈ S, z i ∈ C i) ∧
  75       (∀ i ∈ S, ∀ j ∈ S, i ≠ j → Disjoint (twoPathEdges (x i) (y i) (z i))
  76         (twoPathEdges (x j) (y j) (z j))) ∧
  77       ∀ v, middleCount S z v ≤ L := by
  78     induction S using Finset.induction_on with
  79     | empty =>
  80       exact ⟨fun _ => Classical.choice inferInstance, by simp, by simp,
  81         fun v => by simp [middleCount]⟩
  82     | @insert i S hi ih =>
  83       obtain ⟨z, hz, hdisj, hmiddle⟩ := ih
  84       have hxp (v : V) : ((S.filter (fun j => x j = v)).card : ℝ) ≤ a := by
  85         apply le_trans _ (hx v)
  86         exact_mod_cast Finset.card_le_card
  87           (Finset.filter_subset_filter _ (Finset.subset_univ S))
  88       have hyp (v : V) : ((S.filter (fun j => y j = v)).card : ℝ) ≤ a := by
  89         apply le_trans _ (hy v)
  90         exact_mod_cast Finset.card_le_card
  91           (Finset.filter_subset_filter _ (Finset.subset_univ S))
  92       have hcp : (S.card : ℝ) / L + 4 * a + 4 * L < (C i).card := by
  93         have hc : (S.card : ℝ) ≤ Fintype.card I := by exact_mod_cast Finset.card_le_univ S
  94         have hd := div_le_div_of_nonneg_right hc (Nat.cast_nonneg L)
  95         linarith [hlarge i]
  96       obtain ⟨w,hw,hsat,hout,hin,hinrev,houtrev⟩ :=
  97         undirectedTwoPathPacking_next S x y z (x i) (y i) (C i) L hL hxp hyp hmiddle hcp
  98       let z' := Function.update z i w
  99       have heq (j : I) (hj : j ∈ S) : z' j = z j := by
 100         exact Function.update_of_ne (by intro hji; subst j; exact hi hj) _ _
 101       have hnew : z' i = w := Function.update_self _ _ _
 102       have hn (j : I) (hj : j ∈ S) :
 103           Disjoint (twoPathEdges (x i) (y i) w) (twoPathEdges (x j) (y j) (z j)) := by
 104         exact twoPathEdges_disjoint_of_unused S x y z (x i) (y i) w hout hin hinrev houtrev j hj
 105       have hcount (v : V) : middleCount (insert i S) z' v =
 106           (if w = v then 1 else 0) + middleCount S z v := by
 107         have hfilter : S.filter (fun j => z' j = v) = S.filter (fun j => z j = v) := by
 108           apply Finset.filter_congr
 109           intro j hj
 110           rw [heq j hj]
 111         unfold middleCount
 112         rw [Finset.filter_insert, hfilter, hnew]
 113         by_cases hwv : w = v <;> simp [hwv, hi, Nat.add_comm]
 114       refine ⟨z', ?_, ?_, ?_⟩
 115       · intro j hj
 116         rcases Finset.mem_insert.mp hj with rfl | hj
 117         · simpa only [hnew] using hw
 118         · simpa only [heq j hj] using hz j hj
 119       · intro j hj k hk hjk
 120         by_cases hji : j = i
 121         · subst j
 122           have hkS : k ∈ S := (Finset.mem_insert.mp hk).resolve_left (Ne.symm hjk)
 123           rw [hnew, heq k hkS]
 124           exact hn k hkS
 125         · have hjS : j ∈ S := (Finset.mem_insert.mp hj).resolve_left hji
 126           by_cases hki : k = i
 127           · subst k
 128             rw [heq j hjS, hnew]
 129             exact (hn j hjS).symm
 130           · have hkS : k ∈ S := (Finset.mem_insert.mp hk).resolve_left hki
 131             rw [heq j hjS, heq k hkS]
 132             exact hdisj j hjS k hkS hjk
 133       · intro v
 134         rw [hcount]
 135         by_cases hwv : w = v
 136         · subst v
 137           simp only [ite_true]
 138           omega
 139         · simp only [hwv, ite_false, zero_add]
 140           exact hmiddle v
 141   obtain ⟨z, hz, hdisj, hmiddle⟩ := hall Finset.univ
 142   exact ⟨z, fun i => hz i (Finset.mem_univ i),
 143     fun i j h => hdisj i (Finset.mem_univ i) j (Finset.mem_univ j) h, hmiddle⟩
