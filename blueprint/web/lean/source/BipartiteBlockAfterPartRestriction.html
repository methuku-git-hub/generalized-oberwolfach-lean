  13 theorem finite_filter_card_le_restriction_plus_loss {X:Type} [DecidableEq X]
  14     (A A':Finset X) (p:X→Prop):
  15     (A.filter p).card≤(A'.filter p).card+(A\A').card:=by
  16   have hs:A.filter p⊆A'.filter p∪(A\A'):=by
  17     intro x hx
  18     obtain ⟨hx,hp⟩:=Finset.mem_filter.mp hx
  19     by_cases hx':x∈A'
  20     · exact Finset.mem_union_left _ (Finset.mem_filter.mpr ⟨hx',hp⟩)
  21     · exact Finset.mem_union_right _ (Finset.mem_sdiff.mpr ⟨hx,hx'⟩)
  22   exact (Finset.card_le_card hs).trans (Finset.card_union_le _ _)

  26 theorem bipartite_block_after_part_restriction
  27     (E:X→Y→Prop) (A A':Finset X) (B B':Finset Y) (hA:A'⊆A) (hB:B'⊆B)
  28     (d tau lam loss:ℝ)
  29     (H:BipartiteBlockBounds E (fun x=>x∈A) (fun y=>y∈B) (d+loss) tau lam)
  30     (hLA:((A\A').card:ℝ)≤loss) (hLB:((B\B').card:ℝ)≤loss):
  31     BipartiteBlockBounds E (fun x=>x∈A') (fun y=>y∈B') d tau lam:=by
  32   constructor
  33   · intro x hx
  34     have hold:=H.row x (hA hx)
  35     have hloss:((B.filter (E x)).card:ℝ)≤(B'.filter (E x)).card+(B\B').card:=by
  36       exact_mod_cast finite_filter_card_le_restriction_plus_loss B B' (E x)
  37     simp only [←Finset.filter_filter,Finset.filter_mem_eq_inter,Finset.univ_inter] at hold ⊢
  38     linarith only [hold,hloss,hLB]
  39   · intro y hy
  40     have hold:=H.col y (hB hy)
  41     have hloss:((A.filter (fun x=>E x y)).card:ℝ)≤(A'.filter (fun x=>E x y)).card+(A\A').card:=by
  42       exact_mod_cast finite_filter_card_le_restriction_plus_loss A A' (fun x=>E x y)
  43     simp only [←Finset.filter_filter,Finset.filter_mem_eq_inter,Finset.univ_inter] at hold ⊢
  44     linarith only [hold,hloss,hLA]
  45   · intro S T hS hT hs ht
  46     exact H.cut S T (fun x hx=>hA (hS x hx)) (fun y hy=>hB (hT y hy)) hs ht
