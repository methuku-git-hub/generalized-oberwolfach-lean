  13 theorem attachmentRowEmbedding_hom (φ : V₀ ↪ V) (f : I → W ↪ V)
  14     (hroot : ∀ i w v,f i w≠φ v) (hsep : ∀ i j,i≠j → ∀ a b,f i a≠f j b)
  15     (G₀ : SimpleGraph V₀) (C : SimpleGraph W) (z : W) (ρ : I → V₀) (G : SimpleGraph V)
  16     (hold : ∀ a b,G₀.Adj a b → G.Adj (φ a) (φ b))
  17     (hcycle : ∀ i w w',w≠z → w'≠z → C.Adj w w' → G.Adj (f i w) (f i w'))
  18     (hcross : ∀ i w,C.Adj z w → G.Adj (φ (ρ i)) (f i w)) :
  19     ∀ a b,(ordinaryAttachmentRim G₀ C z ρ).Adj a b →
  20       G.Adj (expansionRowEmbedding φ f hroot hsep a) (expansionRowEmbedding φ f hroot hsep b) := by
  21   intro a b h
  22   cases a with
  23   | inl a =>
  24     cases b with
  25     | inl b => exact hold a b h
  26     | inr p =>
  27       rcases p with ⟨i,w⟩
  28       change ρ i=a ∧ w≠z ∧ C.Adj z w at h
  29       rcases h with ⟨rfl,_,hc⟩
  30       exact hcross i w hc
  31   | inr p =>
  32     rcases p with ⟨i,w⟩
  33     cases b with
  34     | inl b =>
  35       change ρ i=b ∧ w≠z ∧ C.Adj z w at h
  36       rcases h with ⟨rfl,_,hc⟩
  37       exact (hcross i w hc).symm
  38     | inr p =>
  39       rcases p with ⟨j,w'⟩
  40       change i=j ∧ w≠z ∧ w'≠z ∧ C.Adj w w' at h
  41       obtain ⟨rfl,hw,hw',hc⟩ := h
  42       exact hcycle i w w' hw hw' hc

  44 theorem attachmentEmbedding_spokes (φ : V₀ ↪ V) (f : I → W ↪ V)
  45     (hroot : ∀ i w v,f i w≠φ v) (hsep : ∀ i j,i≠j → ∀ a b,f i a≠f j b)
  46     (ψ : U₀ ↪ U) (a : I → U) (huroot : ∀ i u,a i≠ψ u) (husep : Function.Injective a)
  47     (H₀ : V₀ → U₀ → Prop) (z : W) (ρ : I → V₀) (H : V → U → Prop)
  48     (hold : ∀ v u,H₀ v u → H (φ v) (ψ u))
  49     (hH : ∀ i w,w≠z → H (f i w) (a i)) (hcross : ∀ i,H (φ (ρ i)) (a i)) :
  50     ∀ b c,ordinaryAttachmentSpokes H₀ z ρ b c →
  51       H (expansionRowEmbedding φ f hroot hsep b) (expansionHubEmbedding ψ a huroot husep c) := by
  52   intro b c h
  53   cases b with
  54   | inl b =>
  55     cases c with
  56     | inl c => exact hold b c h
  57     | inr i =>
  58       change ρ i=b at h
  59       subst b
  60       exact hcross i
  61   | inr p =>
  62     rcases p with ⟨i,w⟩
  63     cases c with
  64     | inl c => exact False.elim h
  65     | inr j =>
  66       change w≠z ∧ i=j at h
  67       obtain ⟨hw,rfl⟩ := h
  68       exact hH i w hw
