  13 def assignedSetIndices {V C J : Type*}
  14     (Q : Finset J) (T : J → Finset V) (σ : V → C×Bool) (c : C) : Finset J :=
  15   Q.filter (fun j=>colourAssignedIn (T j) σ c)

  17 def rejectedSetIndices {V C J : Type*} (G : SimpleGraph V)
  18     (Q : Finset J) (T : J → Finset V) (σ : V → C×Bool) (c : C) : Finset J :=
  19   Q.filter (fun j=>colourRejectedIn G (T j) σ c)

  21 def nibbleColourCount {V C J : Type*} (G : SimpleGraph V)
  22     (Q : Finset J) (T : J → Finset V) (L : Finset C) (a b : ℕ)
  23     (σ : V → C×Bool) : ℝ :=
  24   (L.filter (fun c=>a≤(assignedSetIndices Q T σ c).card ∧
  25     b≤(rejectedSetIndices G Q T σ c).card)).card

  27 theorem rejectedSetIndices_subset_assigned {V C J : Type*} (G : SimpleGraph V)
  28     (Q : Finset J) (T : J → Finset V) (σ : V → C×Bool) (c : C) :
  29     rejectedSetIndices G Q T σ c⊆assignedSetIndices Q T σ c := by
  30   intro j hj
  31   obtain ⟨hj,hv⟩ := Finset.mem_filter.mp hj
  32   obtain ⟨v,hv,hc,_⟩ := hv
  33   exact Finset.mem_filter.mpr ⟨hj,v,hv,hc⟩

  35 theorem nibbleColourCount_witness
  36     {V C J : Type*} [DecidableEq V] [Fintype J] (G : SimpleGraph V)
  37     (Q : Finset J) (T : J → Finset V) (σ : V → C×Bool) (c : C) {a b : ℕ}
  38     (ha : a≤(assignedSetIndices Q T σ c).card)
  39     (hb : b≤(rejectedSetIndices G Q T σ c).card) :
  40     ∃ W : Finset V,W.card≤3*Q.card ∧ (∀ v∈W,(σ v).1=c) ∧
  41       ∀ τ,(∀ v∈W,σ v=τ v) →
  42         a≤(assignedSetIndices Q T τ c).card ∧
  43         b≤(rejectedSetIndices G Q T τ c).card := by
  44   classical
  45   obtain ⟨A,hA,hcolA,hkeepA⟩ := witnessed_count_threshold
  46     (fun j τ=>colourAssignedIn (T j) τ c) Q Prod.fst σ c
  47     (fun j _ hj=>colourAssignedIn_witness (T j) σ c hj) ha
  48   obtain ⟨B,hB,hcolB,hkeepB⟩ := witnessed_count_threshold
  49     (fun j τ=>colourRejectedIn G (T j) τ c) Q Prod.fst σ c
  50     (fun j _ hj=>colourRejectedIn_witness G (T j) σ c hj) hb
  51   refine ⟨A∪B,?_,?_,?_⟩
  52   · have hcard := Finset.card_union_le A B
  53     omega
  54   · intro v hv
  55     rcases Finset.mem_union.mp hv with hv|hv
  56     · exact hcolA v hv
  57     · exact hcolB v hv
  58   · intro τ he
  59     exact ⟨hkeepA τ (fun v hv=>he v (Finset.mem_union_left _ hv)),
  60       hkeepB τ (fun v hv=>he v (Finset.mem_union_right _ hv))⟩

  62 theorem nibbleColourCount_certificate
  63     {V C J : Type*} [Fintype V] [Fintype C] [Fintype J]
  64     (G : SimpleGraph V) (Q : Finset J) (T : J → Finset V)
  65     (L : Finset C) (a b : ℕ) :
  66     HasCoordinateCertificateBound (nibbleColourCount G Q T L a b)
  67       1 ((3*(Q.card : ℝ))*(L.card : ℝ)) := by
  68   classical
  69   let P : C → (V → C×Bool) → Prop := fun c σ=>
  70     c∈L ∧ a≤(assignedSetIndices Q T σ c).card ∧
  71       b≤(rejectedSetIndices G Q T σ c).card
  72   have hex (σ : V → C×Bool) (c : C) (h : P c σ) :
  73       ∃ W : Finset V,W.card≤3*Q.card ∧ (∀ v∈W,(σ v).1=c) ∧
  74         ∀ τ,(∀ v∈W,σ v=τ v) → P c τ := by
  75     obtain ⟨W,hsize,hcol,hkeep⟩ := nibbleColourCount_witness G Q T σ c h.2.1 h.2.2
  76     exact ⟨W,hsize,hcol,fun τ he=>⟨h.1,hkeep τ he⟩⟩
  77   let W : (V → C×Bool) → C → Finset V := fun σ c=>
  78     if h : P c σ then Classical.choose (hex σ c h) else ∅
  79   have hW (σ : V → C×Bool) (c : C) (h : P c σ) :
  80       (W σ c).card≤3*Q.card ∧ (∀ v∈W σ c,(σ v).1=c) ∧
  81         ∀ τ,(∀ v∈W σ c,σ v=τ v) → P c τ := by
  82     dsimp only [W]
  83     rw [dif_pos h]
  84     exact Classical.choose_spec (hex σ c h)
  85   have hc := monochromatic_witness_certificate P W Prod.fst L
  86     (fun _ _ h=>h.1) (fun σ c h=>(hW σ c h).1)
  87     (fun σ c h=>(hW σ c h).2.1) (fun σ c h=>(hW σ c h).2.2)
  88   convert hc using 1
  89   ·
  90     funext σ
  91     unfold nibbleColourCount
  92     apply congrArg (fun n : ℕ=>(n : ℝ))
  93     apply congrArg Finset.card
  94     ext c
  95     simp [P]
  96   · push_cast
  97     ring
