  14 theorem bipartite_no_odd_decomposition {n r : ℕ} {G : SimpleGraph (Fin n)}
  15     (hG : G.IsBipartite) (hr : Odd r) {u v : Fin n} (huv : G.Adj u v) :
  16     ¬ HasCycleDecomposition G r := by
  17   rintro ⟨q, H, hpart, hH⟩
  18   obtain ⟨i, hi⟩ := (hpart u v).1.mp huv
  19   have hsub : H i ≤ G := by
  20     intro x y hxy
  21     exact (hpart x y).1.mpr ⟨i, hxy⟩
  22   exact (hH i).not_bipartite hr (SimpleGraph.Colorable.mono_left hsub hG)

  24 def balancedCompleteBipartite (t : ℕ) : SimpleGraph (Fin (2 * t)) :=
  25   (SimpleGraph.completeEquipartiteGraph 2 t).map
  26     (finProdFinEquiv : Fin 2 × Fin t ≃ Fin (2 * t)).toEmbedding
  27 
  28 theorem balancedCompleteBipartite_regular (t : ℕ) :
  29     (balancedCompleteBipartite t).IsRegularOfDegree t := by
  30   classical
  31   intro x
  32   let e : Fin 2 × Fin t ≃ Fin (2 * t) := finProdFinEquiv
  33   obtain ⟨y, rfl⟩ := e.surjective x
  34   have hh := SimpleGraph.degree_map_apply (SimpleGraph.completeEquipartiteGraph 2 t)
  35     e.toEmbedding y
  36   rw [SimpleGraph.degree_completeEquipartiteGraph] at hh
  37   simpa only [balancedCompleteBipartite, e, ← SimpleGraph.ncard_neighborSet, Equiv.toEmbedding_apply,
  38     Nat.reduceSub, one_mul] using hh
  39 
  40 theorem balancedCompleteBipartite_edge_card (t : ℕ) :
  41     (balancedCompleteBipartite t).edgeFinset.card = t ^ 2 := by
  42   classical
  43   have hh := SimpleGraph.card_edgeFinset_map
  44     (finProdFinEquiv : Fin 2 × Fin t ≃ Fin (2 * t)).toEmbedding
  45     (SimpleGraph.completeEquipartiteGraph 2 t)
  46   rw [SimpleGraph.card_edgeFinset_completeEquipartiteGraph] at hh
  47   simpa only [balancedCompleteBipartite, ← Set.ncard_coe_finset,
  48     SimpleGraph.coe_edgeFinset, Nat.choose_self, one_mul] using hh

  50 theorem balancedCompleteBipartite_bipartite (t : ℕ) :
  51     (balancedCompleteBipartite t).IsBipartite :=
  52   SimpleGraph.Colorable.map _ SimpleGraph.completeEquipartiteGraph_colorable

  54 theorem balancedCompleteBipartite_has_edge {t : ℕ} (ht : 0 < t) :
  55     ∃ u v, (balancedCompleteBipartite t).Adj u v := by
  56   let e : Fin 2 × Fin t ≃ Fin (2 * t) := finProdFinEquiv
  57   let a : Fin t := ⟨0, ht⟩
  58   refine ⟨e (0, a), e (1, a), ?_⟩
  59   apply SimpleGraph.map_adj_apply.mpr
  60   change (0 : Fin 2) ≠ 1
  61   decide

  63 theorem odd_cycle_divisible_counterexample {r : ℕ} (hr3 : 3 ≤ r) (hr : Odd r)
  64     (N : ℕ) :
  65     ∃ n ≥ N, ∃ G : SimpleGraph (Fin n), CycleDivisible G r ∧
  66       (∀ v, (n : ℝ) / 2 = (G.degree v : ℝ)) ∧ ¬ HasCycleDecomposition G r := by
  67   classical
  68   let t := 2 * r * (N + 1)
  69   have ht : 0 < t := by dsimp [t]; positivity
  70   refine ⟨2 * t, ?_, balancedCompleteBipartite t, ?_, ?_, ?_⟩
  71   · dsimp only [t]
  72     nlinarith
  73   · constructor
  74     · intro v
  75       rw [balancedCompleteBipartite_regular t v]
  76       exact ⟨r * (N + 1), by dsimp only [t]; ring⟩
  77     · rw [balancedCompleteBipartite_edge_card]
  78       exact ⟨4 * r * (N + 1) ^ 2, by dsimp only [t]; ring⟩
  79   · intro v
  80     rw [balancedCompleteBipartite_regular t v]
  81     push_cast
  82     ring
  83   · obtain ⟨u, v, huv⟩ := balancedCompleteBipartite_has_edge ht
  84     exact bipartite_no_odd_decomposition (balancedCompleteBipartite_bipartite t) hr huv

  86 theorem cycleThreshold_half_le_of_odd {r : ℕ} (hr3 : 3 ≤ r) (hr : Odd r) :
  87     (1 / 2 : ℝ) ≤ cycleThreshold r := by
  88   classical
  89   apply le_csInf (show {d | AdmissibleCycleBound r d}.Nonempty from ⟨1, one_admissible r⟩)
  90   intro d hd
  91   by_contra hnot
  92   have hdlt : d < 1 / 2 := lt_of_not_ge hnot
  93   obtain ⟨N, hN⟩ := hd.2
  94   obtain ⟨n, hn, G, hdiv, hdegree, hno⟩ := odd_cycle_divisible_counterexample hr3 hr N
  95   apply hno (hN n hn G hdiv ?_)
  96   intro v
  97   rw [← hdegree v]
  98   nlinarith [Nat.cast_nonneg (α := ℝ) n]
