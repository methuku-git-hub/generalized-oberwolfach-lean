  16     let fraction:Option (Fin K)→ℝ:=fun j=>
  17       ((Finset.univ.filter (fun v=>longPreparationCycleClass F hF K v=j)).card:ℝ)/n

  19     let long:=fraction none
  20     let cutoff:=θ/(100*(K:ℝ))
  21     let longKept:=if long<θ/100 then 0 else long
  22     let kept:=fun i:Fin K=>if f i<cutoff then 0 else f i
  23     let a:=(1-θ)*longKept/c
  24     let b:=(c-1)*a+(1-θ)*∑ i,kept i
  25     let η:=fun i:Fin K=>(1-θ)*kept i/b;
  26     0<b ∧ (∀ i,0≤η i ∧ η i≤1/b) ∧ (∑ i,η i)*b+(c-1)*a=b := by
