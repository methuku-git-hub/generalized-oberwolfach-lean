  13 def pairSampleProbability (n m : ℕ) : ℝ := (m.choose 2:ℝ)/(n.choose 2:ℝ)

  15 theorem pairSampleProbability_pos {n m : ℕ} (hm : 2≤m) (hmn : m≤n) :
  16     0<pairSampleProbability n m := by
  17   apply div_pos
  18   · exact_mod_cast Nat.choose_pos hm
  19   · exact_mod_cast Nat.choose_pos (hm.trans hmn)

  21 theorem pairSampleProbability_formula {n m : ℕ} (hm : 2≤m) (hmn : m≤n) :
  22     pairSampleProbability n m=(m:ℝ)*(m-1)/((n:ℝ)*(n-1)) := by
  23   unfold pairSampleProbability
  24   rw [Nat.cast_choose_two,Nat.cast_choose_two]
  25   field_simp

  28 theorem pairSampleProbability_lower {n m : ℕ} (hm : 2≤m) (hmn : m≤n) :
  29     ((m:ℝ)/n)^2/2≤pairSampleProbability n m := by
  30   rw [pairSampleProbability_formula hm hmn]
  31   have hn : (0:ℝ)<n := Nat.cast_pos.mpr (by omega)
  32   have hn1 : (0:ℝ)<(n:ℝ)-1 := by
  33     have hh : (2:ℝ)≤n := by exact_mod_cast hm.trans hmn
  34     linarith
  35   have hm2 : (2:ℝ)≤m := by exact_mod_cast hm
  36   have hd : 0<(n:ℝ)*(n-1) := mul_pos hn hn1
  37   apply (le_div_iff₀ hd).mpr
  38   field_simp
  39   nlinarith [sq_nonneg (m:ℝ),mul_nonneg (show 0≤(m:ℝ)*(m-2) by positivity) hn.le]

  49 theorem cycle_sample_coefficient_scale {n m r : ℕ} (hm : 2≤m) (hmn : m≤n)
  50     (hr : 2≤r) (κ : ℝ) (hκ : 0≤κ) (hκ1 : κ≤1) :
  51     ((m:ℝ)/n)^r/(pairSampleProbability n m/(1+κ))≤
  52       (4*(m:ℝ)^(r-2))/(n:ℝ)^(r-2) := by
  53   have hn : (0:ℝ)<n := Nat.cast_pos.mpr (by omega)
  54   have hm' : (0:ℝ)<m := Nat.cast_pos.mpr (by omega)
  55   have hx : 0<(m:ℝ)/n := div_pos hm' hn
  56   have hκ' : 0<1+κ := by linarith
  57   have hs : 0<((m:ℝ)/n)^2/2 := by positivity
  58   have hpow : ((m:ℝ)/n)^r=((m:ℝ)/n)^(r-2)*((m:ℝ)/n)^2 := by
  59     rw [← pow_add,Nat.sub_add_cancel hr]
  60   calc
  61     _ = (1+κ)*(((m:ℝ)/n)^r/pairSampleProbability n m) := by
  62       simp only [div_div_eq_mul_div]
  63       ring
  64     _ ≤ (1+κ)*(((m:ℝ)/n)^r/(((m:ℝ)/n)^2/2)) := by
  65       apply mul_le_mul_of_nonneg_left _ hκ'.le
  66       exact div_le_div_of_nonneg_left (by positivity) hs (pairSampleProbability_lower hm hmn)
  67     _ = 2*(1+κ)*((m:ℝ)/n)^(r-2) := by
  68       rw [hpow]
  69       field_simp
  70     _ ≤ 4*((m:ℝ)/n)^(r-2) := by
  71       have hp : 0≤((m:ℝ)/n)^(r-2) := by positivity
  72       nlinarith [mul_le_mul_of_nonneg_right hκ1 hp]
  73     _ = _ := by rw [div_pow];ring
