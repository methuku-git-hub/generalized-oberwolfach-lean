  11 theorem edge_assignment_greedy (S:Finset E) (L Bad:E→Finset W) (C:E→E→Prop)
  12     (hsym:∀x y,C x y→C y x) (K:ℕ) (hK:0<K)
  13     (hbudget:∀ e∈S,(Bad e).card+((S.filter (C e)).card:ℝ)+(S.card:ℝ)/K<((L e).card:ℝ)):
  14     ∃f:E→W,
  15       (∀e∈S,f e∈L e ∧ f e∉Bad e) ∧
  16       (∀e∈S,∀e'∈S,e≠e'→C e e'→f e≠f e') ∧
  17       (∀w,(S.filter (fun e=>f e=w)).card≤K) := by
  18   have hconstruct:∀T:Finset E,T⊆S→∃f:E→W,
  19       (∀e∈T,f e∈L e ∧ f e∉Bad e) ∧
  20       (∀e∈T,∀e'∈T,e≠e'→C e e'→f e≠f e') ∧
  21       (∀w,(T.filter (fun e=>f e=w)).card≤K) := by
  22     intro T
  23     induction T using Finset.induction_on with
  24     | empty =>
  25       intro hsub
  26       refine ⟨fun _=>Classical.choice inferInstance,?_,?_,?_⟩
  27       · simp
  28       · simp
  29       · simp
  30     | @insert e T he ih =>
  31       intro hsub
  32       have heS:e∈S:=hsub (Finset.mem_insert_self _ _)
  33       have hTS:T⊆S:=fun x hx=>hsub (Finset.mem_insert_of_mem hx)
  34       obtain ⟨f,hallowed,hconf,hcap⟩:=ih hTS
  35       obtain ⟨w,hw,hwb,hwconf,hwcap⟩:=edge_assignment_available_index S T hTS f e (L e) (Bad e) (C e) K hK (hbudget e heS)
  36       obtain ⟨ha,hc,hcapa,_⟩:=edge_assignment_update T f e he L Bad C hsym K hallowed hconf hcap w hw hwb hwconf hwcap
  37       exact ⟨Function.update f e w,ha,hc,hcapa⟩
  38   exact hconstruct S (Finset.Subset.refl _)
