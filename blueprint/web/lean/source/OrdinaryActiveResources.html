  18 abbrev OrdinaryActiveEdge (Q : SimpleGraph V) (H : V → U → Prop) :=
  19   {e : Sym2 V⊕(V×U) // ordinaryResourceState Q H e=true}

  21 def ordinaryActiveResources (W : OrdinaryWheel Q F H r) : Finset (OrdinaryActiveEdge Q H) :=
  22   (ordinaryWheelResources W).subtype (fun e=>ordinaryResourceState Q H e=true)

  24 theorem ordinaryWheelResources_present (W : OrdinaryWheel Q F H r)
  25     (e : Sym2 V⊕(V×U)) (he : e∈ordinaryWheelResources W) : ordinaryResourceState Q H e=true := by
  26   cases e with
  27   | inl e =>
  28     rw [ordinaryWheelResources_rim] at he
  29     have hq : e∈Q.edgeSet := SimpleGraph.edgeSet_mono W.val.1.property he
  30     simpa only [ordinaryResourceState,Sum.elim_inl,decide_eq_true_eq] using hq
  31   | inr vu =>
  32     rcases vu with ⟨v,u⟩
  33     rw [ordinaryWheelResources_spoke] at he
  34     have hh : H v u := by
  35       rw [← he.1]
  36       exact W.property.1 v he.2
  37     simpa only [ordinaryResourceState,Sum.elim_inr,decide_eq_true_eq] using hh
  38 
  39 theorem ordinaryActiveResources_mem (W : OrdinaryWheel Q F H r) (e : OrdinaryActiveEdge Q H) :
  40     e∈ordinaryActiveResources W ↔ e.val∈ordinaryWheelResources W := Finset.mem_subtype

  42 theorem ordinaryActiveResources_map (W : OrdinaryWheel Q F H r) :
  43     (ordinaryActiveResources W).map (Function.Embedding.subtype _) =ordinaryWheelResources W :=

  46 theorem ordinaryActiveResources_card (W : OrdinaryWheel Q F H r) :

  48   have hh := congrArg Finset.card (ordinaryActiveResources_map W)

  57 theorem ordinary_active_exact_loads (w : OrdinaryWheel Q F H r → ℝ)
  58     (hrim : ∀ x y,ordinaryRimLoad w x y=if Q.Adj x y then 1 else 0)
  59     (hspoke : ∀ v u,ordinarySpokeLoad w v u=if H v u then 1 else 0)
  60     (e : OrdinaryActiveEdge Q H) : (∑ W,if e∈ordinaryActiveResources W then w W else 0)=1 := by
  61   rcases e with ⟨e,he⟩
  62   simp only [ordinaryActiveResources_mem]
  63   cases e with
  64   | inl e =>
  65     induction e using Sym2.ind with
  66     | _ x y =>
  67       have hq : Q.Adj x y := by simpa only [ordinaryResourceState,Sum.elim_inl,decide_eq_true_eq,SimpleGraph.mem_edgeSet] using he
  68       rw [ordinaryWheelResources_load_rim,hrim,if_pos hq]
  69   | inr vu =>
  70     rcases vu with ⟨v,u⟩
  71     have hh : H v u := by simpa only [ordinaryResourceState,Sum.elim_inr,decide_eq_true_eq] using he
  72     rw [ordinaryWheelResources_load_spoke,hspoke,if_pos hh]
