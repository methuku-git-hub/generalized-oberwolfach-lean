  14     (h:∀ x,f (ρ x)=π (f x)) (tail head:E→W) (ktail khead:K→W) (a:E⊕K):

  18     Finset.image_insert,Finset.image_singleton,h]

  28 theorem spokeRelationImage_disjoint (f:W↪V) (H₀ HK:W→U→Prop) (u:U)
  29     (h:∀ x,¬(H₀ x u∧HK x u)):∀ v,¬(spokeRelationImage f H₀ v u∧spokeRelationImage f HK v u):=by
  30   rintro v ⟨⟨x,hx,hxH⟩,⟨y,hy,hyH⟩⟩
  31   have he:x=y:=f.injective (hx.trans hy.symm)
  32   exact h x ⟨hxH,he.symm ▸ hyH⟩

  34 theorem spokeRelationImage_separated (f:W↪V) (ρ:Equiv.Perm W) (π:Equiv.Perm V)
  35     (h:∀ x,f (ρ x)=π (f x)) (H₀ HK:W→U→Prop) (u:U)
  36     (hs:∀ x y,H₀ x u→HK y u→x≠ρ y):
  37     ∀ v z,spokeRelationImage f H₀ v u→spokeRelationImage f HK z u→v≠π z:=by
  38   rintro v z ⟨x,hx,hxH⟩ ⟨y,hy,hyH⟩ he
  39   apply hs x y hxH hyH
  40   apply f.injective
  41   rw [h,hx,hy]
  42   exact he

  44 theorem spokeRelationImage_neighbourhood (f:W↪V) (H₀ HK:W→U→Prop) (u:U):
  45     Finset.univ.filter (fun v=>spokeRelationImage f H₀ v u∨spokeRelationImage f HK v u)=
  46       (Finset.univ.filter (fun x=>H₀ x u∨HK x u)).image f:=by
  47   ext v
  48   simp only [Finset.mem_filter,Finset.mem_univ,true_and,Finset.mem_image,spokeRelationImage]
  49   constructor
  50   · rintro (⟨x,hx,hH⟩|⟨x,hx,hH⟩)
  51     · exact ⟨x,Or.inl hH,hx⟩
  52     · exact ⟨x,Or.inr hH,hx⟩
  53   · rintro ⟨x,hH,hx⟩
  54     rcases hH with hH|hH
  55     · exact Or.inl ⟨x,hx,hH⟩
  56     · exact Or.inr ⟨x,hx,hH⟩
