  10 theorem marked_iteration_global_tolerance_choice
  11     (r h : ℕ) (κ γ p k omega C θ b : ℝ)
  12     (hκ : 0<κ) (hγ : 0<γ) (hp : 0<p) (hk : 0<k) (ho : 0<omega)
  13     (hC : 0≤C) (hθ : 0<θ) (hb : 0<b) :
  14     ∃ ζmax : ℝ,0<ζmax ∧ ∀ ε ζ : ℝ,ε≤κ/4→0≤ζ→ζ≤ζmax→
  15       ε+ζ≤κ ∧ ε+4*ζ≤κ ∧ ε+2*h*ζ≤κ ∧ ε+h*ζ≤κ ∧
  16       ζ≤(κ-ε)*omega ∧ ε≤κ ∧ 2*ζ≤(3/4)*γ*p^2 ∧ ζ≤γ*min p k/8 ∧
  17       C*ζ≤κ/4 ∧ C*ζ≤θ/4 ∧ 2*(r+5:ℝ)*ζ≤b ∧ (2*r+11:ℝ)*ζ≤b := by
  18   obtain ⟨z0,hz0,hbase⟩ := marked_iteration_reference_scale h κ γ p k omega C hκ hγ hp hk ho hC
  19   let zmax := min z0 (min (θ/(4*(C+1))) (b/(2*(r:ℝ)+12)))
  20   refine ⟨zmax,by dsimp [zmax];positivity,?_⟩
  21   intro ε ζ he hz hzmax
  22   obtain ⟨h1,h2,h3,h4,h5,h6,h7,h8,h9⟩ := hbase ε ζ he hz (hzmax.trans (min_le_left _ _))

  32 theorem marked_iteration_local_tolerance_choice
  33     (m h : ℕ) (x eps C t β η : ℝ)
  34     (hx : 0<x) (heps : 0<eps) (hC : 0≤C) (ht : 0<t) (hβ : 0<β) (hη : 0<η) :
  35     ∃ sigmaMax : ℝ,0<sigmaMax ∧ ∀ sigma : ℝ,0≤sigma→sigma≤sigmaMax→
  36       (m+2:ℝ)*sigma≤x/4 ∧ 2*(h:ℝ)*sigma≤x/2 ∧
  37       4*C*sigma/eps≤min (t/8) (min β (η/4)) := by
  38   let b := min (t/8) (min β (η/4))
  39   have hb : 0<b := by dsimp [b];positivity
  40   let sigmaMax := min (x/(4*((m:ℝ)+2)))
  41     (min (x/(4*((h:ℝ)+1))) (eps*b/(4*(C+1))))
  42   refine ⟨sigmaMax,by dsimp [sigmaMax];positivity,?_⟩
  43   intro sigma hs hmax
  44   have h1 := hmax.trans (min_le_left _ _)
  45   have h2 := (hmax.trans (min_le_right _ _)).trans (min_le_left _ _)
  46   have h3 := (hmax.trans (min_le_right _ _)).trans (min_le_right _ _)

  48   have h2' := (le_div_iff₀ (show 0<4*((h:ℝ)+1) by positivity)).mp h2
