  14 def initialIndexDensity (U : Type c) [Fintype U] (D : ℕ) : ℝ :=
  15   2 * (Fintype.card U : ℝ) / D

  17 structure Vortex {V : Type b} [Fintype V] (U : Type c) [Fintype U]
  18     (G : SimpleGraph V) (D : ℕ) (β ε : ℝ) (m : ℕ)
  19     extends VertexVortex G D (β / 2) ε m where
  20   indices : ℕ → Finset U
  21   indices_initial : indices 0 = Finset.univ
  22   index_sizes : ∀ i, (indices i).card = floorOrbit ε (Fintype.card U) i
  23   indices_nested : ∀ i, indices (i + 1) ⊆ indices i
  24   centre_estimates : ∀ i ≤ length, 0 < i →
  25     |2 * ((indices i).card : ℝ) / initialIndexDensity U D -
  26       ((D : ℝ) / Fintype.card V) * (vertices i).card| ≤ β * (vertices i).card ∧
  27     ∀ x ∈ vertices (i - 2),
  28       |((G.neighborFinset x ∩ vertices i).card : ℝ) -
  29         2 * (indices i).card / initialIndexDensity U D| ≤ β * (vertices i).card

  31 theorem vortex_exists {β ε ρ₀ : ℝ} (hb : 0 < β) (he : 0 < ε)
  32     (heq : ε < 1 / 4) (hp : 0 < ρ₀) :
  33     ∃ m₀ : ℕ, 1 ≤ m₀ ∧ ∀ (V : Type b) [Fintype V] (U : Type c) [Fintype U]
  34       (G : SimpleGraph V) (D m : ℕ),
  35       G.IsRegularOfDegree D → (Fintype.card V : ℝ) / 2 ≤ D →
  36       ρ₀ ≤ initialIndexDensity U D → initialIndexDensity U D ≤ 1 →
  37       m₀ ≤ m → m ≤ Fintype.card V → Nonempty (Vortex U G D β ε m) := by
  38   have he1 : ε < 1 := by linarith
  39   obtain ⟨N₁, hN₁1, hN₁⟩ := vertexVortex_exists.{b} (η := β / 2) (by positivity) he he1
  40   obtain ⟨N₂, hN₂1, hN₂⟩ := eventually_vortex_error_budget hb he he1 hp
  41   refine ⟨max N₁ N₂, by omega, ?_⟩
  42   intro V _ U _ G D m hreg hDmin hρmin hρmax hm hmn
  43   have hmpos : 0 < m := by omega
  44   have hnpos : 0 < Fintype.card V := by omega
  45   have hnreal : (0 : ℝ) < Fintype.card V := by exact_mod_cast hnpos
  46   have hDpos : (0 : ℝ) < D := by linarith
  47   let ρ := initialIndexDensity U D
  48   have hρpos : 0 < ρ := hp.trans_le hρmin
  49   have hcentre : 2 * (Fintype.card U : ℝ) = ρ * D := by
  50     dsimp only [ρ, initialIndexDensity]
  51     field_simp
  52   have hDupper : (D : ℝ) ≤ Fintype.card V := by
  53     let : Nonempty V := Fintype.card_pos_iff.mp hnpos
  54     let x : V := Classical.choice (inferInstance : Nonempty V)
  55     have hh := G.degree_lt_card_verts x
  56     rw [hreg x] at hh
  57     exact_mod_cast hh.le
  58   have hu : (Fintype.card U : ℝ) ≤ (Fintype.card V : ℝ) / 2 := by
  59     have hh := (div_le_iff₀ hDpos).mp hρmax
  60     change 2 * (Fintype.card U : ℝ) ≤ (1 : ℝ) * D at hh
  61     linarith
  62   obtain ⟨Q⟩ := hN₁ V G D m hreg (by omega) hmn
  63   obtain ⟨I, hI0, hIcard, hIsub⟩ := nested_finsets_of_sizes (V := U)
  64     (floorOrbit ε (Fintype.card U)) rfl (floorOrbit_antitone he1 _)
  65   obtain ⟨_, _, hmargin⟩ := hN₂ m (by omega)
  66   refine ⟨{
  67     toVertexVortex := Q
  68     indices := I
  69     indices_initial := hI0
  70     index_sizes := hIcard
  71     indices_nested := hIsub
  72     centre_estimates := ?_ }⟩
  73   intro i hi hipos
  74   have herr := rounded_vortex_centre_error he.le heq hp hρmin hnpos hu hcentre i
  75   have hsmall : |2 * ((I i).card : ℝ) / ρ -
  76       ((D : ℝ) / Fintype.card V) * (Q.vertices i).card| ≤ β / 2 * (Q.vertices i).card := by
  77     have hlow : ε * m / 2 ≤ ((Q.vertices i).card : ℝ) := by
  78       have hsub := finset_chain_subset Q.vertices Q.length Q.nested hi (le_refl _)
  79       exact Q.terminal_lower.trans (by exact_mod_cast Finset.card_le_card hsub)
  80     have hbound : 3 / ρ₀ ≤ β / 2 * (Q.vertices i).card := by
  81       have hh := mul_le_mul_of_nonneg_left hlow hb.le
  82       nlinarith [hmargin]
  83     rw [hIcard i, Q.sizes i hi]
  84     exact herr.le.trans (by simpa only [Q.sizes i hi] using hbound)
  85   refine ⟨hsmall.trans ?_, ?_⟩
  86   · have hnonneg : (0 : ℝ) ≤ (Q.vertices i).card := by positivity
  87     nlinarith
  88   · intro x hx
  89     have hbase := Q.degrees i hi hipos x hx
  90     have ht := abs_sub_le ((G.neighborFinset x ∩ Q.vertices i).card : ℝ)
  91       (((D : ℝ) / Fintype.card V) * (Q.vertices i).card)
  92       (2 * ((I i).card : ℝ) / ρ)
  93     rw [abs_sub_comm (((D : ℝ) / Fintype.card V) * (Q.vertices i).card)] at ht
  94     dsimp only [ρ] at hsmall ht
  95     linarith
