  10 inductive PrescribedSourcePiece (X:Type) where
  11   | path (vertices:List X)
  12   | cycle (first:X) (rest:List X)
  13 
  14 def PrescribedSourcePiece.vertices:PrescribedSourcePiece X→List X
  15   | .path xs=>xs
  16   | .cycle a xs=>a::xs
  17 
  18 def PrescribedSourcePiece.edges:PrescribedSourcePiece X→Finset (Sym2 X)
  19   | .path xs=>prescribedOrderedPathEdges xs
  20   | .cycle a xs=>prescribedOrderedPathEdges ((a::xs)++[a])
  21 
  22 def PrescribedSourcePiece.arcs:PrescribedSourcePiece X→Finset (X×X)
  23   | .path xs=>prescribedDirectedPathEdges xs
  24   | .cycle a xs=>prescribedDirectedPathEdges ((a::xs)++[a])
  25 
  26 def PrescribedSourcePiece.Valid (p:PrescribedSourcePiece X) (D:X→X→Prop) (F:SimpleGraph X):Prop:=
  27   match p with
  28   | .path xs=>xs.Nodup ∧ List.IsChain (fun x y=>D x y ∧ ¬F.Adj x y) xs
  29   | .cycle a xs=>(a::xs).Nodup ∧ 3≤(a::xs).length ∧ DirectedCycleThrough D F (a::xs)

  31 theorem prescribed_source_piece_arc_endpoints (p:PrescribedSourcePiece X) (e:X×X)
  32     (he:e∈p.arcs):e.1∈p.vertices ∧ e.2∈p.vertices := by
  33   cases p with
  34   | path xs =>exact prescribed_directed_path_edge_endpoints xs e he
  35   | cycle a xs =>
  36     have hh:=prescribed_directed_path_edge_endpoints ((a::xs)++[a]) e he
  37     change e.1∈a::xs ∧ e.2∈a::xs
  38     simpa [or_assoc,or_left_comm,or_comm] using hh
