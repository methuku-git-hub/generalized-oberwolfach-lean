  15 theorem marked_restricted_deletion_budget
  16     (f:W↪V) (g:Z↪U) (σ:RawMarkedOutcome V U Γ Color)
  17     (H:Bool→V→U→Prop) (DR:SimpleGraph V) (DS:SimpleGraph (V⊕U))
  18     (ζ K:ℝ) (hζ:0≤ζ) (hK:(Fintype.card V:ℝ)≤K*Fintype.card W)
  19     (hr:∀ v,(DR.degree v:ℝ)≤ζ*Fintype.card V)
  20     (hs:∀ z,(DS.degree z:ℝ)≤ζ*Fintype.card V):
  21     rawMarkedRestrictPair f g (rawMarkedDeleteRims σ DR.edgeFinset)=
  22       rawMarkedDeleteRims (rawMarkedRestrictPair f g σ) (DR.comap f).edgeFinset ∧
  23     (fun b x u=>markedDeletePhysicalSpokes H DS b (f x) (g u))=
  24       markedDeletePhysicalSpokes (fun b x u=>H b (f x) (g u)) (DS.comap (Sum.map f g)) ∧
  25     (∀ x,( (DR.comap f).degree x:ℝ)≤(ζ*K)*Fintype.card W) ∧
  26     (∀ z,( (DS.comap (Sum.map f g)).degree z:ℝ)≤(ζ*K)*Fintype.card W) := by
  27   have mult:ζ*(Fintype.card V:ℝ)≤(ζ*K)*Fintype.card W := by
  28     nlinarith only [mul_le_mul_of_nonneg_left hK hζ]
  29   have hg : (Sum.map f g : W⊕Z → V⊕U).Injective := by
  30     intro x y hxy
  31     cases x <;> cases y <;> simp_all
  32   let fg:W⊕Z ↪ V⊕U := ⟨Sum.map f g,hg⟩
  33   refine ⟨rawMarkedRestrictPair_deleteRims f g σ DR,
  34     markedDeletePhysicalSpokes_restrict f g H DS,?_,?_⟩
  35   · intro x
  36     have hdeg:(DR.comap f).degree x≤DR.degree (f x) := by
  37       rw [graph_comap_degree_into_image DR f x]
  38       exact (Finset.card_le_card Finset.inter_subset_left).trans_eq (SimpleGraph.card_neighborFinset_eq_degree _ _)
  39     exact (Nat.cast_le.mpr hdeg).trans ((hr (f x)).trans mult)
  40   · intro z
  41     have hdeg:(DS.comap fg).degree z≤DS.degree (fg z) := by
  42       rw [graph_comap_degree_into_image DS fg z]
  43       exact (Finset.card_le_card Finset.inter_subset_left).trans_eq (SimpleGraph.card_neighborFinset_eq_degree _ _)
  44     exact (Nat.cast_le.mpr hdeg).trans ((hs (fg z)).trans mult)
