  27 theorem marked_boundary_selection_supports (tail head:E→V) (ktail khead:K→V) (π:Equiv.Perm V)
  28     (H₀ HK:V→U→Prop) (X:Finset V) (Z:Finset U) (σ:Sym2 (V⊕U)→Bool)
  29     (hub:J↪U) (root:I↪V) (hhub:∀ j,hub j∉Z) (hroot:∀ i,root i∉X)
  30     (Y:J→Finset V) (A₀ A₁ X₀ X₁:I→Finset V) (Y₀ Y₁:I→Finset U)
  31     (hY:∀ j,Y j⊆markedBoundarySpokeRows H₀ HK X σ (hub j))
  32     (hA:∀ i,A₀ i∪A₁ i∪(X₀ i∪X₁ i)⊆markedBoundaryRimNeighbours tail head ktail khead π X σ (root i))
  33     (hB:∀ i,Y₀ i∪Y₁ i⊆markedBoundarySpokeHubs H₀ HK Z σ (root i))
  34     (P M:ℝ)
  35     (hP:∀ x y,x≠y→((markedBoundarySupport tail head ktail khead π H₀ HK X Z σ x∩
  36       markedBoundarySupport tail head ktail khead π H₀ HK X Z σ y).card:ℝ)≤P)
  37     (hM:∀ v,((Finset.univ.filter (fun x=>v∈markedBoundarySupport tail head ktail khead π H₀ HK X Z σ x)).card:ℝ)≤M):
  38     let S:J⊕I→Finset (V⊕U):=Sum.elim
  39       (fun j=>(Y j∪(Y j).image π).image Sum.inl)
  40       (fun i=>ordinaryInnerVertices (A₀ i∪A₁ i∪(X₀ i∪X₁ i)) (Y₀ i∪Y₁ i))
  41     (∀ i j,i≠j→((S i∩S j).card:ℝ)≤P)∧
  42     (∀ v,((Finset.univ.filter (fun j=>v∈S j)).card:ℝ)≤M) := by
  43   intro S
  44   apply indexed_support_bounds (markedBoundarySupport tail head ktail khead π H₀ HK X Z σ)
  45     (markedBoundaryRootEmbedding hub root) S _ P M hP hM
  46   intro i
  47   cases i with
  48   | inl j=>
  49     change (Y j∪(Y j).image π).image Sum.inl⊆
  50       markedBoundarySupport tail head ktail khead π H₀ HK X Z σ (Sum.inr (hub j))
  51     rw [marked_boundary_support_at_hub tail head ktail khead π H₀ HK X Z σ (hub j) (hhub j)]
  52     exact Finset.image_subset_image (Finset.union_subset_union (hY j) (Finset.image_subset_image (hY j)))
  53   | inr i=>
  54     change ordinaryInnerVertices (A₀ i∪A₁ i∪(X₀ i∪X₁ i)) (Y₀ i∪Y₁ i)⊆
  55       markedBoundarySupport tail head ktail khead π H₀ HK X Z σ (Sum.inl (root i))
  56     rw [marked_boundary_support_at_row tail head ktail khead π H₀ HK X Z σ (root i) (hroot i)]
  57     exact Finset.union_subset_union
  58       (Finset.image_subset_image ((hA i).trans Finset.subset_union_left)) (Finset.image_subset_image (hB i))
