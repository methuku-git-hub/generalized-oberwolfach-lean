  13 theorem residual_actual_prescribed_set (F Q:SimpleGraph V)
  14     (hF:F.IsRegularOfDegree 2) (hQ:Q≤F)
  15     (s₁ s₂:Finset I) (hs:Disjoint s₁ s₂)
  16     (u v:I→V) (P:∀i,F.Walk (u i) (v i))
  17     (hp:∀i∈s₁∪s₂,(P i).IsPath)
  18     (hP:(↑(s₁∪s₂):Set I).PairwiseDisjoint (fun i=>(P i).support.toFinset))
  19     (c d:ℕ) (hc:3≤c) (hd:0<d) (hlen:∀i∈s₁∪s₂,(P i).length=c*d+4)
  20     (T:I→Finset ℕ) (hT:∀i,T i⊆Finset.range d) (C:Finset V)
  21     (hC:∀i∈s₁∪s₂,Disjoint (P i).support.toFinset C):
  22     let R:=Finset.univ \ (C ∪ (s₁∪s₂).biUnion (pathFamilyLocalDeletion F s₁ u v P c d T hT))
  23     let A₀ := (s₁.biUnion (fun i=>{(P i).getVert 2,(P i).getVert (2+c*d)})) ∪
  24       (R ∩ s₂.biUnion (fun i=>(Finset.range (d+1)).image (fun j=>(P i).getVert (2+c*j))))
  25     let missing:=∑i∈s₂,(d-(T i).card)
  26     let runs:=∑i∈s₂,(selectedSegmentRuns d (T i)).card;
  27     A₀⊆R ∧ (∀ x∈A₀,∀ y∈A₀,¬Q.Adj x y) ∧
  28       A₀.card=2*s₁.card+s₂.card+missing+runs := by
  29   let s:=s₁∪s₂
  30   let D:=pathFamilyLocalDeletion F s₁ u v P c d T hT
  31   let R:=Finset.univ \ (C∪s.biUnion D)
  32   let A₁:=s₁.biUnion (fun i=>{(P i).getVert 2,(P i).getVert (2+c*d)})
  33   let M:=s₂.biUnion (fun i=>(Finset.range (d+1)).image (fun j=>(P i).getVert (2+c*j)))
  34   let A₀:=A₁∪(R∩M)
  35   have hD:∀i∈s,D i⊆(P i).support.toFinset := by
  36     intro i hi
  37     exact path_family_local_deletion_subset F s₁ u v P c d T hT i (hlen i hi)
  38   have hA₁:A₁⊆R := by
  39     intro x hx
  40     obtain ⟨i,hi,hx⟩:=Finset.mem_biUnion.mp hx
  41     have hi':i∈s := Finset.mem_union_left _ hi
  42     have hxin:x∈(P i).support.toFinset := by
  43       simp only [Finset.mem_insert,Finset.mem_singleton] at hx
  44       rcases hx with rfl|rfl <;> simp
  45     have hnot:x∉D i := by
  46       simp only [D,pathFamilyLocalDeletion,hi,ite_true,centralPathInterior,Finset.mem_sdiff]
  47       intro h
  48       exact h.2 hx
  49     have hloc:=disjoint_family_deletion_localization s (fun k=>(P k).support.toFinset)
  50       D C hP hD hC i hi' x hxin
  51     exact Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,hloc.2 hnot⟩
  52   have hA₀:A₀⊆R := Finset.union_subset hA₁ Finset.inter_subset_left
  53   have hind:=two_factor_residual_prescribed_markers_independent F Q hF hQ s₁ s₂ u v P hp hP c d hc hlen R
  54   have hset:=selected_marker_global_set_identity F s₁ s₂ hs u v P hp hP c d (by omega) hlen T hT C hC
  55   have hcount:=residual_prescribed_marker_family_count F s₁ s₂ hs u v P hp hP c d (by omega) hd hlen T (by intro i hi;exact hT i)
  56   dsimp only at hset hcount hind ⊢
  57   refine ⟨hA₀,hind,?_⟩
  58   change (A₁∪(R∩M)).card=2*s₁.card+s₂.card+
  59     (∑i∈s₂,(d-(T i).card))+(∑i∈s₂,(selectedSegmentRuns d (T i)).card)
  60   rw [hset]
  61   exact hcount
  62 end Oberwolfach
