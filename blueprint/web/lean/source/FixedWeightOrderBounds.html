   9 theorem fixed_weight_order_bounds (n a A B w:ℝ) (m:ℕ)
  10     (ha:0<a) (han:a≤n) (hna:n≤2*a) (hA:0≤A) (hB:0≤B)
  11     (hw:A/n^m≤w∧w≤B/n^m):
  12     (A/2^m)/a^m≤w∧w≤B/a^m:=by
  13   constructor
  14   · have hn:0<n:=ha.trans_le han
  15     have hh:=div_le_div_of_nonneg_left hA (pow_pos hn m)
  16       (pow_le_pow_left₀ (le_of_lt hn) hna m)
  17     rw [mul_pow,←div_div] at hh
  18     exact hh.trans hw.1
  19   · exact hw.2.trans (div_le_div_of_nonneg_left hB (pow_pos ha m)
  20       (pow_le_pow_left₀ ha.le han m))
