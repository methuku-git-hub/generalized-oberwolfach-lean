  44 theorem graph_le (P : MarkedEndpointPairing A₀ A₁ X₀ X₁ Y₀ Y₁ H₀ HK π b) :
  45     P.graph≤ordinaryPhysicalGraph ⊥ (fun v u=>H₀ v u∨HK v u) := by
  46   apply marked_endpoint_spoke_subgraph
  47   · intro y
  48     have hh:=P.incoming_adj (P.incoming.symm y)
  49     rw [P.incoming.apply_symm_apply] at hh
  50     exact Or.inl hh.2
  51   · intro y
  52     have hh:=P.outgoing_adj (P.outgoing.symm y)
  53     rw [P.outgoing.apply_symm_apply] at hh
  54     dsimp [markedOutgoingAssignment] at hh
  55     split_ifs at hh with h
  56     · exact Or.inr hh.2
  57     · exact Or.inl hh.2
