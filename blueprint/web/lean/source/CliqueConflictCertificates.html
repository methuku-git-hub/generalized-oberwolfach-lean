   4 /-! Certificates for the number of vertices in a clique whose tentative
   5 colours conflict with a neighbour. Witnesses are chosen from the actual
   6 conflicts of the realised colouring. -/

  16 def cliqueConflictSet (G : SimpleGraph V) (T : Finset V) (σ : V → C) : Finset V :=
  17   T.filter (fun v=>∃ u,G.Adj v u ∧ σ u=σ v)
  18 
  19 def cliqueConflictWitness (G : SimpleGraph V) (T : Finset V) (σ : V → C) (v : V) : V :=
  20   if hv : v∈cliqueConflictSet G T σ then
  21     Classical.choose (Finset.mem_filter.mp hv).2 else v
  22 
  23 theorem cliqueConflictWitness_spec
  24     (G : SimpleGraph V) (T : Finset V) (σ : V → C)
  25     {v : V} (hv : v∈cliqueConflictSet G T σ) :
  26     G.Adj v (cliqueConflictWitness G T σ v) ∧
  27       σ (cliqueConflictWitness G T σ v)=σ v := by
  28   unfold cliqueConflictWitness
  29   rw [dif_pos hv]
  30   exact Classical.choose_spec (Finset.mem_filter.mp hv).2
  31 
  32 def cliqueWitnessWeight (T W : Finset V) (v : V) : ℝ :=
  33   if v∈T then 2 else if v∈W then 1 else 0
  34 
  35 theorem cliqueWitnessWeight_bounds (T W : Finset V) (v : V) :
  36     0≤cliqueWitnessWeight T W v ∧ cliqueWitnessWeight T W v≤2 := by
  37   unfold cliqueWitnessWeight
  38   split_ifs <;> norm_num

  40 theorem cliqueWitnessWeight_changed_sum (T W E : Finset V) :
  41     (∑ v,if v∈E then cliqueWitnessWeight T W v else 0)=
  42       2*((T∩E).card : ℝ)+(((W\T)∩E).card : ℝ) := by
  43   classical
  44   have he (v : V) :
  45       (if v∈E then cliqueWitnessWeight T W v else 0)=
  46         2*(if v∈T∩E then (1 : ℝ) else 0)+
  47           (if v∈(W\T)∩E then (1 : ℝ) else 0) := by
  48     by_cases ht : v∈T <;> by_cases hw : v∈W <;> by_cases hE : v∈E <;>
  49       simp [cliqueWitnessWeight,ht,hw,hE]
  50   simp_rw [he]
  51   rw [Finset.sum_add_distrib,← Finset.mul_sum]
  52   simp
  53   have hT : Finset.univ.filter (fun v=>v∈T ∧ v∈E)=T∩E := by ext;simp
  54   have hW : Finset.univ.filter (fun v=>(v∈W ∧ v∉T) ∧ v∈E)=(W\T)∩E := by ext;simp
  55   rw [hT,hW]

  57 theorem cliqueConflict_certificate
  58     (G : SimpleGraph V) (T : Finset V)
  59     (hclique : ∀ u∈T,∀ v∈T,u≠v → G.Adj u v) :
  60     HasCoordinateCertificateBound (fun σ : V → C=>(cliqueConflictSet G T σ).card) 2
  61       (3*(T.card : ℝ)) := by
  62   classical
  63   intro σ
  64   let B := cliqueConflictSet G T σ
  65   let f := cliqueConflictWitness G T σ
  66   let W := B.image f
  67   let w := cliqueWitnessWeight T W
  68   have hBT : B⊆T := Finset.filter_subset _ _
  69   refine ⟨w,cliqueWitnessWeight_bounds T W,?_,?_⟩
  70   · have hh := cliqueWitnessWeight_changed_sum T W Finset.univ
  71     simp only [Finset.mem_univ,ite_true,Finset.inter_univ] at hh
  72     have hW : W.card≤T.card := (Finset.card_image_le).trans (Finset.card_le_card hBT)
  73     have hd : (W\T).card≤W.card := Finset.card_le_card Finset.sdiff_subset
  74     change (∑ v,cliqueWitnessWeight T W v)≤_
  75     rw [hh]
  76     have hWr : (W.card : ℝ)≤T.card := by exact_mod_cast hW
  77     have hdr : ((W\T).card : ℝ)≤W.card := by exact_mod_cast hd
  78     linarith
  79   · intro τ
  80     let B' := cliqueConflictSet G T τ
  81     let E := Finset.univ.filter (fun v=>σ v≠τ v)
  82     let L := (B\B').filter (fun v=>σ v=τ v)
  83     have hL (v : V) (hv : v∈L) : v∈B ∧ v∉B' ∧ σ v=τ v := by
  84       obtain ⟨hs,he⟩ := Finset.mem_filter.mp hv
  85       exact ⟨(Finset.mem_sdiff.mp hs).1,(Finset.mem_sdiff.mp hs).2,he⟩
  86     have hchanged (v : V) (hv : v∈L) : f v∈E := by
  87       have h := hL v hv
  88       have hw := cliqueConflictWitness_spec G T σ h.1
  89       apply Finset.mem_filter.mpr
  90       refine ⟨Finset.mem_univ _,?_⟩
  91       intro heq
  92       apply h.2.1
  93       apply Finset.mem_filter.mpr
  94       refine ⟨hBT h.1,f v,hw.1,?_⟩
  95       change τ (f v)=τ v
  96       have hs : σ (f v)=σ v := hw.2
  97       exact heq.symm.trans (hs.trans h.2.2)
  98     have hinj : Set.InjOn f (L : Set V) := by
  99       intro u hu v hv huv
 100       by_contra hne
 101       have hU := hL u hu
 102       have hV := hL v hv
 103       have hwu := cliqueConflictWitness_spec G T σ hU.1
 104       have hwv := cliqueConflictWitness_spec G T σ hV.1
 105       have hcol : σ u=σ v := by
 106         calc
 107           _ = σ (f u) := hwu.2.symm
 108           _ = σ (f v) := congrArg σ huv
 109           _ = _ := hwv.2
 110       apply hU.2.1
 111       apply Finset.mem_filter.mpr
 112       refine ⟨hBT hU.1,v,hclique u (hBT hU.1) v (hBT hV.1) hne,?_⟩
 113       exact hV.2.2.symm.trans (hcol.symm.trans hU.2.2)
 114     have himage : L.image f⊆(T∩E)∪((W\T)∩E) := by
 115       intro u hu
 116       obtain ⟨v,hv,rfl⟩ := Finset.mem_image.mp hu
 117       have he := hchanged v hv
 118       by_cases ht : f v∈T
 119       · exact Finset.mem_union_left _ (Finset.mem_inter.mpr ⟨ht,he⟩)
 120       · apply Finset.mem_union_right
 121         exact Finset.mem_inter.mpr ⟨Finset.mem_sdiff.mpr
 122           ⟨Finset.mem_image.mpr ⟨v,(hL v hv).1,rfl⟩,ht⟩,he⟩
 123     have hLcard : L.card≤(T∩E).card+((W\T)∩E).card := by
 124       rw [← Finset.card_image_of_injOn hinj]
 125       exact (Finset.card_le_card himage).trans (Finset.card_union_le _ _)
 126     have hcover : B⊆(B'∪(T∩E))∪L := by
 127       intro v hv
 128       by_cases hv' : v∈B'
 129       · exact Finset.mem_union_left _ (Finset.mem_union_left _ hv')
 130       · by_cases he : σ v=τ v
 131         · exact Finset.mem_union_right _ (Finset.mem_filter.mpr
 132             ⟨Finset.mem_sdiff.mpr ⟨hv,hv'⟩,he⟩)
 133         · exact Finset.mem_union_left _ (Finset.mem_union_right _
 134             (Finset.mem_inter.mpr ⟨hBT hv,Finset.mem_filter.mpr ⟨Finset.mem_univ _,he⟩⟩))
 135     have hBcard : B.card≤B'.card+(T∩E).card+L.card :=
 136       (Finset.card_le_card hcover).trans
 137         ((Finset.card_union_le _ _).trans
 138           (Nat.add_le_add_right (Finset.card_union_le _ _) _))
 139     have hcount : B.card≤B'.card+2*(T∩E).card+((W\T)∩E).card := by omega
 140     have heq : (∑ i,if σ i≠τ i then w i else 0)=
 141         2*((T∩E).card : ℝ)+(((W\T)∩E).card : ℝ) := by
 142       have hh := cliqueWitnessWeight_changed_sum T W E
 143       simpa only [E,Finset.mem_filter,Finset.mem_univ,true_and] using hh
 144     change (B.card : ℝ)≤(B'.card : ℝ)+_
 145     rw [heq]
 146     exact_mod_cast (show B.card≤B'.card+(2*(T∩E).card+((W\T)∩E).card) by omega)
