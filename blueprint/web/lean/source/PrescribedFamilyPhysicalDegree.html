  16 theorem prescribed_family_physical_degree
  17     (st:PrescribedFamilyState I X V) (QV:I→Finset X) (Q:I→SimpleGraph X)
  18     (A₀:I→Finset X) (ψ:I→X→V) (E₀:I→Finset (Sym2 V)) (J:SimpleGraph V)
  19     (β δ n:ℝ) (hβ:0<β) (hβ1:β≤1) (hδ:0<δ) (hδbound:δ≤(β/100)^2)
  20     (hn:2/Real.sqrt δ≤n) (hQdegree:∀ i x,(Q i).degree x≤2)
  21     (hold:∀ v,((Finset.univ.filter (fun i=>∃ x,x∈A₀ i ∧ ψ i x=v)).card:ℝ)≤δ*n)
  22     (hJ:∀ v,(J.degree v:ℝ)≤δ*n)
  23     (hvalid:st.Valid QV Q A₀ ψ E₀ J (Real.sqrt δ*n)):
  24     (prescribedPhysicalUnion st.physicalEdges).edgeFinset=Finset.univ.biUnion st.physicalEdges ∧
  25     ∀ v,(prescribedPhysicalUnion st.physicalEdges |>.degree v:ℝ)≤5*Real.sqrt δ*n := by
  26   have hfree:∀ v,((st.freeOccurrences.filter (fun p=>st.vertexMap p.1 p.2=v)).card:ℝ)≤Real.sqrt δ*n+1:=by
  27     intro v
  28     have heq:st.freeOccurrences.filter (fun p=>familyVertexImage st.vertexMap p=v) =
  29         st.freeOccurrences.filter (fun p=>st.vertexMap p.1 p.2=v):=by
  30       ext p
  31       simp only [familyVertexImage, Finset.mem_filter]
  32       exact Finset.mem_filter
  33     rw [←heq]
  34     exact hvalid.2.2.2.2.1 v
  35   obtain ⟨G,hG,hcontains⟩:=prescribed_actual_used_graph Q st.assigned A₀ st.vertexMap ψ st.freeOccurrences
  36     hvalid.1 hvalid.2.1 hvalid.2.2.2.1 hQdegree δ n (Real.sqrt δ*n) hold hfree J hJ
  37   have hE:∀ i,st.physicalEdges i⊆G.edgeFinset:=by
  38     intro i
  39     rw [hvalid.2.2.2.2.2.2.2.1 i]
  40     exact (prescribed_source_edges_image_contained (Q i) (st.assigned i) (st.vertexMap i) (hvalid.1 i)
  41       (st.sourceEdges i) (hvalid.2.2.2.2.2.1 i) (hvalid.2.2.2.2.2.2.1 i)).trans (hcontains i)
  42   have hedge:(prescribedPhysicalUnion st.physicalEdges).edgeFinset=Finset.univ.biUnion st.physicalEdges:=by
  43     ext e
  44     constructor
  45     · intro he
  46       have hh:e∈(prescribedPhysicalUnion st.physicalEdges).edgeSet:=SimpleGraph.mem_edgeFinset.mp he
  47       change e∈(SimpleGraph.fromEdgeSet (↑(Finset.univ.biUnion st.physicalEdges):Set (Sym2 V))).edgeSet at hh
  48       rw [SimpleGraph.edgeSet_fromEdgeSet] at hh
  49       exact hh.1
  50     · intro he
  51       obtain ⟨i,_,hi⟩:=Finset.mem_biUnion.mp he
  52       have hnot:¬e.IsDiag:=G.not_isDiag_of_mem_edgeSet (SimpleGraph.mem_edgeFinset.mp (hE i hi))
  53       apply SimpleGraph.mem_edgeFinset.mpr
  54       change e∈(SimpleGraph.fromEdgeSet (↑(Finset.univ.biUnion st.physicalEdges):Set (Sym2 V))).edgeSet
  55       rw [SimpleGraph.edgeSet_fromEdgeSet]
  56       exact ⟨he,hnot⟩
  57   have hle:prescribedPhysicalUnion st.physicalEdges≤G:=by
  58     intro x y hxy
  59     have he:s(x,y)∈Finset.univ.biUnion st.physicalEdges:=by
  60       have hh: s(x,y)∈(↑(Finset.univ.biUnion st.physicalEdges):Set (Sym2 V)):=
  61         ((SimpleGraph.fromEdgeSet_adj _).mp hxy).1
  62       exact hh
  63     obtain ⟨i,_,hi⟩:=Finset.mem_biUnion.mp he
  64     exact SimpleGraph.mem_edgeFinset.mp (hE i hi)
  65   have hnum:=prescribed_embedding_margins_from_hypotheses β δ n hβ hβ1 hδ hδbound hn
  66   refine ⟨hedge,?_⟩
  67   intro v
  68   have hd:((prescribedPhysicalUnion st.physicalEdges).degree v:ℝ)≤G.degree v:=by
  69     exact_mod_cast SimpleGraph.degree_le_of_le hle
  70   have hh:=hG v
  71   nlinarith [hnum.2.2.1]
