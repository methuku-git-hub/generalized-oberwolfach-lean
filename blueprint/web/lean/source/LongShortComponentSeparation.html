   9 theorem long_paths_disjoint_short_component_sets
  10     {X I J:Type*} [Fintype X] [DecidableEq X]
  11     (G:SimpleGraph X) (K:ℕ) (long:I→G.ConnectedComponent) (short:J→G.ConnectedComponent)
  12     (hLong:∀i,K<Fintype.card (long i)) (hShort:∀j,Fintype.card (short j)≤K)
  13     (len:I→ℕ) (ext:∀i,Fin (len i)↪X) (C:J→Finset X)
  14     (hExt:∀i p,ext i p∈(long i).supp)
  15     (hC:∀j x,x∈C j→x∈(short j).supp) :
  16     ∀i j,Disjoint (Finset.univ.image (ext i)) (C j) := by
  17   intro i j
  18   apply Finset.disjoint_left.mpr
  19   intro x hx hy
  20   obtain ⟨p,_,rfl⟩:=Finset.mem_image.mp hx
  21   have heq:G.connectedComponentMk (ext i p)=long i:=
  22     (SimpleGraph.ConnectedComponent.mem_supp_iff _ _).mp (hExt i p)
  23   have heq':G.connectedComponentMk (ext i p)=short j:=
  24     (SimpleGraph.ConnectedComponent.mem_supp_iff _ _).mp (hC j (ext i p) hy)
  25   have hcomp:long i=short j:=heq.symm.trans heq'
  26   have hlarge:=hLong i
  27   rw [hcomp] at hlarge
  28   exact (not_lt_of_ge (hShort j)) hlarge
  29 end Oberwolfach
