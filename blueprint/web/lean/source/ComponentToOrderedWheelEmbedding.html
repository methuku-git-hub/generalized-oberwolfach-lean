  14 theorem component_to_ordered_ordinary_wheel_embedding
  15     (F:SimpleGraph (Fin n)) (hF:F.IsRegularOfDegree 2) (C:F.ConnectedComponent)
  16     (π:Equiv.Perm (Fin n)) (b:ℕ) (W:OrderedOrdinaryWheel (U:=U) π b r)
  17     (hsize:Fintype.card C=r):
  18     ∃ f:C↪Fin n,
  19       (∀x,∃i:Fin r,f x=W.1.val i) ∧
  20       (∀i:Fin r,∃x:C,f x=W.1.val i) ∧
  21       (∀x y,C.toSimpleGraph.Adj x y →
  22         ∃i:Fin r,(f x=W.1.val i ∧ f y=W.1.val (cycleNext i)) ∨
  23           (f y=W.1.val i ∧ f x=W.1.val (cycleNext i))) ∧
  24       (∀x,Sum.inr (f x,W.2)∈orderedOrdinaryResources W) :=by
  25   have hthree:3≤r:=by rw [←hsize];exact (component_regular_two_cycle_iso F hF C).1
  26   obtain ⟨e0⟩:Nonempty (SimpleGraph.cycleGraph r ≃g C.toSimpleGraph):=by
  27     rw [←hsize]
  28     exact (component_regular_two_cycle_iso F hF C).2
  29   let f:C↪Fin n:=⟨fun x=>W.1.val (e0.symm x),by
  30     intro x y he
  31     apply e0.symm.injective
  32     exact (permutationSeparated_injective π b W.1.val W.1.property) he⟩
  33   refine ⟨f,?_,?_,?_,?_⟩
  34   · intro x
  35     exact ⟨e0.symm x,rfl⟩
  36   · intro i
  37     refine ⟨e0 i,?_⟩
  38     simp [f]
  39   · intro x y hxy
  40     have hc:(SimpleGraph.cycleGraph r).Adj (e0.symm x) (e0.symm y):=e0.symm.map_rel_iff.mpr hxy
  41     rcases (cycleGraph_adj_next (by omega) _ _).mp hc with h|h
  42     · refine ⟨e0.symm x,Or.inl ⟨rfl,?_⟩⟩
  43       change W.1.val (e0.symm y)=W.1.val (cycleNext (e0.symm x))
  44       rw [h]
  45     · refine ⟨e0.symm y,Or.inr ⟨rfl,?_⟩⟩
  46       change W.1.val (e0.symm x)=W.1.val (cycleNext (e0.symm y))
  47       rw [h]
  48   · intro x
  49     apply (orderedOrdinaryResources_spoke W _).mpr
  50     exact ⟨e0.symm x,rfl⟩
  51 end Oberwolfach
