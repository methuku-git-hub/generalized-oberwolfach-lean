  13 theorem pruned_ordinary_class_threshold {W:Type} [Fintype W] {n:ℕ} (hn:0<n)
  14     (K:ℕ) (hK:0<K) (F:W→SimpleGraph (Fin n)) (hF:∀w,(F w).IsRegularOfDegree 2)
  15     (Allowed:Set ℕ) (hAllowed:∀w,AllowedCycleLengths (F w) Allowed)
  16     (D:ℕ) (hEven:Even D) (hsize:Fintype.card W=D/2) (hDn:D≤n)
  17     (δ α κL κS:ℝ) (hδ:1/2≤δ) (hδA:allowedThreshold Allowed≤δ) (hα:0<α)
  18     (hL0:0<κL) (hS0:0<κS) (hL:κL≤α/1000) (hS:κS≤α/(1000*(K:ℝ)))
  19     (hdegree:(δ+α)*(n:ℝ)≤D):
  20     let classify:=fun w=>factorPreparationClass (F w) K
  21     let fibres:=fun j:Option (Fin K)=>Finset.univ.filter (fun w=>classify w=j)
  22     let cutoff:=fun j:Option (Fin K)=>if j=none then κL else κS
  23     let small:=fun j:Option (Fin K)=>((fibres j).card:ℝ)<cutoff j*n
  24     let removed:=Finset.univ.filter (fun w=>small (classify w))
  25     let remaining:={w:W // w∉removed}
  26     let cf:=fun w:remaining=>classify w.val
  27     let D0:=D-2*removed.card
  28     let fibres0:=fun j:Option (Fin K)=>Finset.univ.filter (fun w:remaining=>cf w=j)
  29     let C:=Finset.univ.filter (fun j=>(fibres0 j).Nonempty)
  30     let k:=C.card
  31     let e:Fin k≃↥C:=(Fintype.equivFinOfCardEq (Fintype.card_coe C)).symm;
  32     ∀ a:Fin k,∀i:Fin K,(e a).val=some i→
  33       3 ≤ i.val+1 ∧ i.val+1 ≤ K ∧ i.val+1 ∈ Allowed ∧
  34       cycleThreshold (i.val+1)≤allowedThreshold Allowed ∧
  35       ∀R:SimpleGraph (Fin n),R.IsRegularOfDegree D0→∀v,
  36         (cycleThreshold (i.val+1)+α/2)*(n:ℝ)≤(R.degree v:ℝ) :=by
  37   intro classify fibres cutoff small removed remaining cf D0 fibres0 C k e a i hi
  38   have hmem:(fibres0 (e a).val).Nonempty:=(Finset.mem_filter.mp (e a).property).2
  39   obtain ⟨w,hw⟩:=hmem
  40   have hwclass:classify w.val=some i:=by
  41     have hh:cf w=(e a).val:=(Finset.mem_filter.mp hw).2
  42     exact hh.trans hi
  43   have hcls:=((factor_preparation_class hn (F w.val) (hF w.val) Allowed (hAllowed w.val) K hK).2 i hwclass)
  44   have hbdd:BddAbove (cycleThreshold '' Allowed):=by
  45     refine ⟨1,?_⟩
  46     rintro x ⟨r,hr,rfl⟩
  47     exact cycleThreshold_le_one r
  48   have hle:cycleThreshold (i.val+1)≤allowedThreshold Allowed:=le_csSup hbdd ⟨_,hcls.2.1,rfl⟩
  49   have hsmall0:=factor_small_class_budget K hK classify (n:ℝ) α κL κS (by positivity) hα hL0.le hS0.le hL hS
  50   have hsmall:(removed.card:ℝ)≤α*n/500:=by simpa [fibres,cutoff,small,removed,ite_mul] using hsmall0
  51   have hmarg:(δ+α/2)*(n:ℝ)≤(D0:ℝ):=
  52     (factor_small_removal_margin D removed.card n δ α (by linarith) hα hdegree hsmall removed.card (le_refl _)).2
  53   refine ⟨hcls.1,by omega,hcls.2.1,hle,?_⟩
  54   intro R hR v
  55   have hn' :=mul_le_mul_of_nonneg_right (show cycleThreshold (i.val+1)+α/2≤δ+α/2 by linarith) (Nat.cast_nonneg n)
  56   have hh:=hn'.trans hmarg
  57   simpa only [hR v] using hh
  58 end Oberwolfach
