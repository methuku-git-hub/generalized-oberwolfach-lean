  17 theorem marked_finite_physical_incidence_bound
  18     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ) (σ:RawMarkedOutcome V U Γ Color):
  19     let E₀:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) false
  20     let E₁:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) true
  21     ∀ (A:Finset (E₀⊕E₁)) (δ:ℝ),
  22     (∀ x,((markedSampleArcSetGraph P π d σ A).degree x:ℝ)≤δ)→∀ v,
  23     ((A.filter (fun a=>Sum.elim (fun e=>e.val.1) (fun e=>e.val.1) a=v∨
  24       Sum.elim (fun e=>e.val.2) (fun e=>e.val.2) a=v)).card:ℝ)≤4*δ:=by
  25   intro E₀ E₁ A δ hD v
  26   let D:=markedSampleArcSetGraph P π d σ A
  27   have hAdj0 (a:E₀) (ha:Sum.inl a∈A):D.Adj a.val.1 a.val.2:=by
  28     have h:=(markedSampleArcSetGraph_coordinate P π d σ A (Sum.inl a)).mpr ha
  29     simpa only [Sum.elim_inl,SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using h
  30   have hAdj1 (a:E₁) (ha:Sum.inr a∈A):D.Adj a.val.1 (π a.val.2):=by
  31     have h:=(markedSampleArcSetGraph_coordinate P π d σ A (Sum.inr a)).mpr ha
  32     simpa only [Sum.elim_inr,SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using h
  33   obtain ⟨h0,h1,h2,h3⟩:=marked_physical_rim_deletion_degrees
  34     (fun e:E₀=>e.val.1) (fun e:E₀=>e.val.2) (fun e:E₁=>e.val.1) (fun e:E₁=>e.val.2) π
  35     (markedSampleArc_physical_injective P π (3*d) _) D δ hD
  36   have hset:(A.filter (fun a=>Sum.elim (fun e:E₀=>e.val.1) (fun e:E₁=>e.val.1) a=v∨
  37       Sum.elim (fun e:E₀=>e.val.2) (fun e:E₁=>e.val.2) a=v))=
  38       Finset.univ.filter (fun a:E₀⊕E₁=>a∈A∧(Sum.elim (fun e=>e.val.1) (fun e=>e.val.1) a=v∨
  39       Sum.elim (fun e=>e.val.2) (fun e=>e.val.2) a=v)):=by
  40     ext a;simp only [Finset.mem_filter,Finset.mem_univ,true_and]
  41   rw [hset,←Finset.sum_boole,Fintype.sum_sum_type]
  42   have hE:(∑ a:E₀,if Sum.inl a∈A∧(a.val.1=v∨a.val.2=v) then (1:ℝ) else 0)≤2*δ:=by
  43     calc
  44       _ ≤ ∑ a:E₀,((if D.Adj a.val.1 a.val.2∧a.val.1=v then (1:ℝ) else 0)+
  45         (if D.Adj a.val.1 a.val.2∧a.val.2=v then 1 else 0)):=by
  46         apply Finset.sum_le_sum
  47         intro a _
  48         by_cases ha:Sum.inl a∈A
  49         · have hd:=hAdj0 a ha
  50           by_cases ht:a.val.1=v <;> by_cases hh:a.val.2=v <;> simp_all
  51         · simp only [ha,false_and,ite_false];positivity
  52       _ ≤ 2*δ:=by rw [Finset.sum_add_distrib];linarith only [h0 v,h1 v]
  53   have hK:(∑ a:E₁,if Sum.inr a∈A∧(a.val.1=v∨a.val.2=v) then (1:ℝ) else 0)≤2*δ:=by
  54     calc
  55       _ ≤ ∑ a:E₁,((if D.Adj a.val.1 (π a.val.2)∧a.val.1=v then (1:ℝ) else 0)+
  56         (if D.Adj a.val.1 (π a.val.2)∧a.val.2=v then 1 else 0)):=by
  57         apply Finset.sum_le_sum
  58         intro a _
  59         by_cases ha:Sum.inr a∈A
  60         · have hd:=hAdj1 a ha
  61           by_cases ht:a.val.1=v <;> by_cases hh:a.val.2=v <;> simp_all
  62         · simp only [ha,false_and,ite_false];positivity
  63       _ ≤ 2*δ:=by rw [Finset.sum_add_distrib];linarith only [h2 v,h3 v]
  64   simp only [Sum.elim_inl,Sum.elim_inr]
  65   linarith only [hE,hK]
