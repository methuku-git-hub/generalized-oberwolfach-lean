  14 def twoPathEdge {I V : Type*} (x y z : I → V) (p : I×Bool) : Sym2 V :=
  15   if p.2 then s(z p.1,y p.1) else s(x p.1,z p.1)

  17 theorem twoPathEdge_mem {I V : Type*} [DecidableEq V] (x y z : I → V) (i : I) (b : Bool) :
  18     twoPathEdge x y z (i,b)∈twoPathEdges (x i) (y i) (z i) := by
  19   cases b <;> simp [twoPathEdge,twoPathEdges]

  21 def twoPathEdgeEmbedding {I V : Type*} [DecidableEq V] (x y z : I → V)
  22     (hxy : ∀ i,x i≠y i)
  23     (hdisj : ∀ i j,i≠j → Disjoint (twoPathEdges (x i) (y i) (z i))
  24       (twoPathEdges (x j) (y j) (z j))) : (I×Bool)↪Sym2 V where
  25   toFun := twoPathEdge x y z
  26   inj' := by
  27     intro ⟨i,a⟩ ⟨j,b⟩ he
  28     have hij : i=j := by
  29       by_contra hij
  30       exact Finset.disjoint_left.mp (hdisj i j hij) (twoPathEdge_mem x y z i a)
  31         (he.symm ▸ twoPathEdge_mem x y z j b)
  32     subst j
  33     have hn : s(x i,z i)≠s(z i,y i) := by
  34       intro hh
  35       rcases Sym2.eq_iff.mp hh with ⟨h1,h2⟩ | ⟨h1,h2⟩ <;> exact hxy i (by aesop)
  36     cases a <;> cases b <;> simp only [twoPathEdge,Bool.false_eq_true,ite_false,ite_true] at he
  37     · rfl
  38     · exact False.elim (hn he)
  39     · exact False.elim (hn he.symm)
  40     · rfl

  49 theorem edgeColourGraph_degree_edge_sum {V R : Type*} [Fintype V] [DecidableEq V]
  50     [CommRing R] {k : ℕ} (G : SimpleGraph V) (σ : Sym2 V → Fin k) (a : Fin k) (v : V) :
  51     ((edgeColourGraph G σ a).degree v:R)=
  52       ∑ e : Sym2 V,(if e∈G.edgeFinset ∧ v∈e then (1:R) else 0)*(if σ e=a then 1 else 0) := by
  53   rw [← SimpleGraph.card_incidenceFinset_eq_degree,
  54     SimpleGraph.incidenceFinset_eq_filter,edgeColourGraph_edges,Finset.filter_filter,
  55     Finset.card_filter,Nat.cast_sum]
  56   have hf : (∑ e : Sym2 V,(if e∈G.edgeFinset ∧ v∈e then (1:R) else 0)*(if σ e=a then 1 else 0))=
  57       ∑ e∈G.edgeFinset,(if v∈e then (1:R) else 0)*(if σ e=a then 1 else 0) := by
  58     have hi (e : Sym2 V) : (if e∈G.edgeFinset ∧ v∈e then (1:R) else 0)*(if σ e=a then 1 else 0)=
  59         if e∈G.edgeFinset then (if v∈e then (1:R) else 0)*(if σ e=a then 1 else 0) else 0 := by
  60       by_cases he : e∈G.edgeFinset <;> simp [he]
  61     simp only [hi]
  62     rw [← Finset.sum_filter]
  63     have he : Finset.univ.filter (fun e => e∈G.edgeFinset)=G.edgeFinset := by ext e;simp only [Finset.mem_filter,Finset.mem_univ,true_and]
  64     rw [he]
  65   rw [hf]
  66   apply Finset.sum_congr rfl
  67   intro e _
  68   split_ifs <;> simp_all

  90 theorem twoPath_switch_degree_change {I V R : Type*} [Fintype I]
  91     [Fintype V] [DecidableEq V] [CommRing R] {k : ℕ}
  92     (G : SimpleGraph V) (σ : Sym2 V → Fin k) (x y z : I → V)
  93     (hxy : ∀ i,x i≠y i)
  94     (hdisj : ∀ i j,i≠j → Disjoint (twoPathEdges (x i) (y i) (z i))
  95       (twoPathEdges (x j) (y j) (z j)))
  96     (hxz : ∀ i,G.Adj (x i) (z i)) (hyz : ∀ i,G.Adj (z i) (y i))
  97     (a : Fin k) (v : V) :
  98     let ι := twoPathEdgeEmbedding x y z hxy hdisj
  99     ((edgeColourGraph G (σ ∘ pairSwitchPermutation ι) a).degree v:R)-
 100       ((edgeColourGraph G σ a).degree v:R)=
 101       ∑ i,((if x i=v then (1:R) else 0)-(if y i=v then 1 else 0))*
 102         ((if σ s(z i,y i)=a then 1 else 0)-(if σ s(x i,z i)=a then 1 else 0)) := by
 103   intro ι
 104   rw [edgeColourGraph_degree_edge_sum,edgeColourGraph_degree_edge_sum]
 105   have hh := pair_switch_weighted_difference ι σ
 106     (fun e => if e∈G.edgeFinset ∧ v∈e then (1:R) else 0) (fun b => if b=a then (1:R) else 0)
 107   change (∑ e,(if e∈G.edgeFinset ∧ v∈e then (1:R) else 0)*(if σ (pairSwitchPermutation ι e)=a then 1 else 0))-
 108     (∑ e,(if e∈G.edgeFinset ∧ v∈e then (1:R) else 0)*(if σ e=a then 1 else 0))=_
 109   rw [hh]
 110   apply Finset.sum_congr rfl
 111   intro i _
 112   have he1 : s(x i,z i)∈G.edgeFinset := SimpleGraph.mem_edgeFinset.mpr (hxz i)
 113   have he2 : s(z i,y i)∈G.edgeFinset := SimpleGraph.mem_edgeFinset.mpr (hyz i)
 114   have hi0 : ι (i,false)=s(x i,z i) := rfl
 115   have hi1 : ι (i,true)=s(z i,y i) := rfl
 116   simp only [hi0,hi1,he1,he2,true_and]
 117   rw [edge_incidence_difference (x i) (y i) (z i) v (hxz i).ne (hyz i).symm.ne]

 119 theorem twoPath_switch_preserves_card {I V : Type*} [Fintype I]
 120     [Fintype V] [DecidableEq V] {k : ℕ}
 121     (G : SimpleGraph V) (σ : Sym2 V → Fin k) (x y z : I → V)
 122     (hxy : ∀ i,x i≠y i)
 123     (hdisj : ∀ i j,i≠j → Disjoint (twoPathEdges (x i) (y i) (z i))
 124       (twoPathEdges (x j) (y j) (z j)))
 125     (hxz : ∀ i,G.Adj (x i) (z i)) (hyz : ∀ i,G.Adj (z i) (y i)) (a : Fin k) :
 126     let ι := twoPathEdgeEmbedding x y z hxy hdisj
 127     (edgeColourGraph G (σ ∘ pairSwitchPermutation ι) a).edgeFinset.card=
 128       (edgeColourGraph G σ a).edgeFinset.card := by
 129   intro ι
 130   have he : ((edgeColourGraph G (σ ∘ pairSwitchPermutation ι) a).edgeFinset.card:ℤ)=
 131       ((edgeColourGraph G σ a).edgeFinset.card:ℤ) := by
 132     rw [edgeColourGraph_card_edge_sum,edgeColourGraph_card_edge_sum]
 133     apply pair_switch_preserves_weighted_sum ι σ _ (fun b => if b=a then (1:ℤ) else 0)
 134     intro i
 135     have he1 : s(x i,z i)∈G.edgeFinset := SimpleGraph.mem_edgeFinset.mpr (hxz i)
 136     have he2 : s(z i,y i)∈G.edgeFinset := SimpleGraph.mem_edgeFinset.mpr (hyz i)
 137     have hi0 : ι (i,false)=s(x i,z i) := rfl
 138     have hi1 : ι (i,true)=s(z i,y i) := rfl
 139     simp only [hi0,hi1,he1,he2,ite_true]
 140   exact_mod_cast he
