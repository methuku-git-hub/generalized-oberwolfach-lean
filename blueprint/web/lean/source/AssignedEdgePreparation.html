  14 theorem prepare_assigned_bad_edges (G:SimpleGraph V) (Dstar:V→V→Prop)
  15     (hDstar:∀x y,G.Adj x y→Dstar x y ∨ Dstar y x)
  16     (L:Sym2 V→Finset W) (B:V→Finset W) (old:W→Finset V) (inner:Finset W)
  17     (hOld:∀v i,v∈old i→i∈B v) (hOutside:∀e∈G.edgeFinset,Disjoint (L e) inner)
  18     (a b:ℝ) (ha:0<a) (hb:0≤b) (hbsmall:b≤a/16) (hn:0<Fintype.card V)
  19     (hL:∀e∈G.edgeFinset,a*Fintype.card V≤((L e).card:ℝ))
  20     (hB:∀v,((B v).card:ℝ)≤b*Fintype.card V)
  21     (hG:∀v,(G.degree v:ℝ)≤b*Fintype.card V):
  22     let K:=⌈4*(G.edgeFinset.card:ℝ)/(a*Fintype.card V)⌉₊;
  23     ∃ (assign:Sym2 V→W) (orient:Sym2 V→V×V),
  24       (∀e∈G.edgeFinset,prescribedArcPhysical (orient e)=e ∧ Dstar (orient e).1 (orient e).2) ∧
  25       (∀e∈G.edgeFinset,assign e∈L e ∧ ∀v∈e.toFinset,assign e∉B v) ∧
  26       (∀i,∀e∈G.edgeFinset,∀e'∈G.edgeFinset,assign e=i→assign e'=i→e≠e'→Disjoint e.toFinset e'.toFinset) ∧
  27       (∀i,(G.edgeFinset.filter (fun e=>assign e=i)).card≤K) ∧
  28       (∀i,Disjoint (old i) (prescribedMatchingEndpoints (prescribedAssignedArcs G assign orient i))) ∧
  29       (∀i,i∈inner→prescribedAssignedArcs G assign orient i=∅) ∧
  30       (∀i,IsPrescribedOrientedMatching (prescribedAssignedArcs G assign orient i)) ∧
  31       (∀i,(prescribedAssignedArcs G assign orient i).card≤K) := by
  32   intro K
  33   obtain ⟨f,hallowed,hmatching,hcap⟩:=graph_edge_assignment_original G L B a b ha hb hbsmall hn hL hB hG
  34   obtain ⟨orient,horient⟩:=assigned_edge_orientation G Dstar hDstar
  35   let assign:=extendEdgeAssignment G f
  36   have ha':∀e∈G.edgeFinset,assign e∈L e ∧ ∀v∈e.toFinset,assign e∉B v:=by
  37     intro e he
  38     simpa [assign,extend_edge_assignment_value G f e he] using hallowed ⟨e,he⟩
  39   have hm:=extend_edge_assignment_matching G f (by
  40     intro e e' hne hc
  41     exact hmatching e e' hne hc)
  42   have hc:∀i,(G.edgeFinset.filter (fun e=>assign e=i)).card≤K:=by
  43     intro i
  44     rw [show assign=extendEdgeAssignment G f from rfl,extend_edge_assignment_fiber_card]
  45     exact hcap i
  46   have hphys:∀e∈G.edgeFinset,prescribedArcPhysical (orient e)=e:=fun e he=>(horient e he).1
  47   have hold:=assigned_edges_avoid_old_images G f orient hphys B old hOld (fun e=>(hallowed e).2)
  48   have hinner:=assigned_edges_avoid_inner_indices G f orient L inner (fun e=>(hallowed e).1) hOutside
  49   have hM:=prescribed_assigned_arcs_are_matchings G assign orient hphys hm
  50   have hcard:∀i,(prescribedAssignedArcs G assign orient i).card≤K:=by
  51     intro i
  52     rw [prescribed_assigned_arc_card G assign orient hphys]
  53     exact hc i
  54   exact ⟨assign,orient,horient,ha',hm,hc,hold,hinner,hM,hcard⟩
  55 end Oberwolfach
