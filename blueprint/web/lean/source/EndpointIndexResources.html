  15 def endpointIndexResources (x:V) (u:U) (e:Sym2 V):Finset (Sym2 V⊕(V×U)):=
  16   insert (Sum.inl e) (e.toFinset.image (fun v=>Sum.inl s(x,v)) ∪
  17     e.toFinset.image (fun v=>Sum.inr (v,u)))

  28     (endpointIndexResources x u e).card≤5:=by
  29   have he:e.toFinset.card≤2:=by rw [Sym2.card_toFinset];split_ifs <;> omega
  30   have hi:=Finset.card_insert_le (Sum.inl e)
  31     (e.toFinset.image (fun v=>Sum.inl (β:=V×U) s(x,v)) ∪ e.toFinset.image (fun v=>Sum.inr (α:=Sym2 V) (v,u)))
  32   have hu:=Finset.card_union_le (e.toFinset.image (fun v=>Sum.inl (β:=V×U) s(x,v)))
  33     (e.toFinset.image (fun v=>Sum.inr (α:=Sym2 V) (v,u)))
  34   have h₁:=Finset.card_image_le (s:=e.toFinset) (f:=fun v=>Sum.inl (β:=V×U) s(x,v))

  36   change (endpointIndexResources x u e).card≤_ at hi
