  16  theorem separated_directed_prefix (F:SimpleGraph V)
  17     (D:ℕ→V→V→Prop) (A:ℕ→Finset V) (W Z:Finset V)
  18     (d c:ℝ) (hd:0≤d) (hF:∀ v,(F.degree v:ℝ)≤d)
  19     (k:ℕ) (x:V) (hxA:x∈A 0) (hxW:x∉W) (hxZ:x∉Z)
  20     (hxF:∀ z∈Z,¬F.Adj x z)
  21     (hcount:∀ i<k,∀ v∈A i,c≤(((A (i+1)).filter (D i v)).card:ℝ))
  22     (hbudget:W.card+((Z.card:ℝ)+k+1)*(d+1)<c):
  23     ∃ p:Fin (k+1)↪V,p 0=x∧
  24       (∀ i,p i∈A i.val∧p i∉W∧p i∉Z∧∀ z∈Z,¬F.Adj (p i) z)∧
  25       (∀ i:Fin k,D i.val (p i.castSucc) (p i.succ))∧
  26       ∀ i j,¬F.Adj (p i) (p j):=by
  27   induction k with
  28   | zero=>
  29     let p:Fin 1↪V:=⟨fun _=>x,fun _ _ _=>Subsingleton.elim _ _⟩
  30     refine ⟨p,rfl,?_,?_,?_⟩
  31     · intro i
  32       have hi:i=0:=by apply Fin.ext;omega
  33       subst i
  34       exact ⟨hxA,hxW,hxZ,hxF⟩
  35     · intro i;exact Fin.elim0 i
  36     · intro i j;exact F.loopless.irrefl _
  37   | succ k ih=>
  38     have hbudget':W.card+((Z.card:ℝ)+k+1)*(d+1)<c:=by
  39       push_cast at hbudget
  40       nlinarith only [hbudget,hd]
  41     obtain ⟨p,hp0,hpA,hpD,hpF⟩:=ih (fun i hi=>hcount i (by omega)) hbudget'
  42     let S:Finset V:=Finset.univ.image p
  43     have hS:S.card=k+1:=by
  44       dsimp only [S]
  45       rw [Finset.card_image_of_injective _ p.injective,Finset.card_univ,Fintype.card_fin]
  46     have hZS:((Z∪S).card:ℝ)≤Z.card+(k+1:ℕ):=by
  47       exact_mod_cast (Finset.card_union_le Z S).trans_eq (by rw [hS])
  48     have hchoice:W.card+((Z∪S).card:ℝ)*(d+1)<c:=by
  49       have hh:=mul_le_mul_of_nonneg_right hZS (show 0≤d+1 by linarith)
  50       push_cast at hh hbudget
  51       nlinarith only [hh,hbudget,hd]
  52     obtain ⟨y,hy,hyW,hyZS,hyF⟩:=candidate_vertex_avoiding F ((A (k+1)).filter (D k (p (Fin.last k))))
  53       W (Z∪S) d c hF (hcount k (by omega) _ (hpA (Fin.last k)).1) hchoice
  54     have hyp:y∉Set.range p:=by
  55       rintro ⟨i,hi⟩
  56       exact hyZS (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨i,Finset.mem_univ _,hi⟩))
  57     let q:Fin (k+2)↪V:=⟨Fin.snoc p y,Fin.snoc_injective_of_injective p.injective hyp⟩
  58     have hqold (i:Fin (k+1)):q i.castSucc=p i:=by simp only [q,Function.Embedding.coeFn_mk,Fin.snoc_castSucc]
  59     have hqlast:q (Fin.last (k+1))=y:=by simp only [q,Function.Embedding.coeFn_mk,Fin.snoc_last]
  60     refine ⟨q,?_,?_,?_,?_⟩
  61     · exact (hqold 0).trans hp0
  62     · intro i
  63       refine Fin.lastCases ?_ (fun j=>?_) i
  64       · rw [hqlast]
  65         exact ⟨(Finset.mem_filter.mp hy).1,hyW,
  66           fun hz=>hyZS (Finset.mem_union_left _ hz),
  67           fun z hz=>hyF z (Finset.mem_union_left _ hz)⟩
  68       · rw [hqold]
  69         exact hpA j
  70     · intro i
  71       refine Fin.lastCases ?_ (fun j=>?_) i
  72       · rw [show (Fin.last k).succ=Fin.last (k+1) from rfl,hqold,hqlast]
  73         exact (Finset.mem_filter.mp hy).2
  74       · rw [hqold,show j.castSucc.succ=j.succ.castSucc from rfl,hqold]
  75         exact hpD j
  76     · intro i j
  77       refine Fin.lastCases ?_ (fun a=>?_) i
  78       · rw [hqlast]
  79         refine Fin.lastCases ?_ (fun b=>?_) j
  80         · rw [hqlast];exact F.loopless.irrefl _
  81         · rw [hqold]
  82           exact hyF _ (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨b,Finset.mem_univ _,rfl⟩))
  83       · rw [hqold]
  84         refine Fin.lastCases ?_ (fun b=>?_) j
  85         · rw [hqlast]
  86           exact fun h=>hyF _ (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨a,Finset.mem_univ _,rfl⟩)) h.symm
  87         · rw [hqold];exact hpF a b
