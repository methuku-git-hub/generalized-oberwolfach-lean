  10 theorem successive_root_parameters (r:ℕ) (C β κ lam:ℝ)
  11     (hC:0≤C) (hβ:0<β) (hκ:0<κ) (hlam:0<lam) :
  12     ∃ K:ℕ,0<K ∧ ∃ θ:ℝ,0<θ ∧ ∀ d M:ℕ,∃ N:ℕ,∀ a m t:ℕ,
  13       N≤a → κ*a≤(m:ℝ) → (m:ℝ)≤C*a → (t:ℝ)≤θ*a →
  14       let L:ℝ:=t*max (2*(K+1)) r
  15       M≤m ∧ L≤lam*m ∧ L≤lam^2*m/16 ∧
  16         (m*r:ℕ)/(K:ℝ)+((r:ℝ)+3)*(d+1)+4*(K+1)+3*L<β*a := by
  17   obtain ⟨K,hK,ζ,hζ,N₀,hcap⟩:=root_wheel_cap_parameters r C (β/2) hC (by positivity)
  18   let D:ℝ:=(max (2*(K+1)) r:ℕ)
  19   have hD:0<D:=by
  20     dsimp [D]
  21     exact_mod_cast (show 0<max (2*(K+1)) r from lt_of_lt_of_le (by omega) (le_max_left _ _))
  22   let θ:=min (β/(12*D)) (min (lam*κ/D) (lam^2*κ/(16*D)))
  23   have hθ:0<θ:=by dsimp [θ];positivity
  24   refine ⟨K,hK,θ,hθ,?_⟩
  25   intro d M
  26   obtain ⟨N₁,hN₁⟩:=exists_nat_gt ((d:ℝ)/ζ)
  27   obtain ⟨N₂,hN₂⟩:=exists_nat_gt ((M:ℝ)/κ)
  28   refine ⟨max N₀ (max N₁ (max N₂ 1)),?_⟩
  29   intro a m t ha hmlo hmhi ht
  30   dsimp only
  31   have haR:(0:ℝ)<a:=Nat.cast_pos.mpr (by omega)
  32   have hd:(d:ℝ)≤ζ*a:=by
  33     have hh:=(div_lt_iff₀ hζ).mp (hN₁.trans_le (Nat.cast_le.mpr (show N₁≤a by omega)))
  34     nlinarith only [hh]
  35   have hM:M≤m:=by
  36     have hh:=(div_lt_iff₀ hκ).mp (hN₂.trans_le (Nat.cast_le.mpr (show N₂≤a by omega)))
  37     have hh':(M:ℝ)<m:=by nlinarith only [hh,hmlo]
  38     exact Nat.le_of_lt (Nat.cast_lt.mp hh')
  39   have htd:(t:ℝ)*D≤θ*a*D:=mul_le_mul_of_nonneg_right ht hD.le
  40   have hbθ:θ≤β/(12*D):=min_le_left _ _
  41   have hrθ:θ≤lam*κ/D:=(min_le_right _ _).trans (min_le_left _ _)
  42   have hcθ:θ≤lam^2*κ/(16*D):=(min_le_right _ _).trans (min_le_right _ _)
  43   have hbudgetL:(t:ℝ)*D≤β*a/12:=by
  44     have hh:=(le_div_iff₀ (by positivity:0<12*D)).mp hbθ
  45     have hh':=mul_le_mul_of_nonneg_right hh haR.le
  46     nlinarith only [htd,hh']
  47   have hrowL:(t:ℝ)*D≤lam*m:=by
  48     have hh:=(le_div_iff₀ hD).mp hrθ
  49     have hh':=mul_le_mul_of_nonneg_right hh haR.le
  50     have hm:=mul_le_mul_of_nonneg_left hmlo hlam.le
  51     nlinarith only [htd,hh',hm]
  52   have hcutL:(t:ℝ)*D≤lam^2*m/16:=by
  53     have hh:=(le_div_iff₀ (by positivity:0<16*D)).mp hcθ
  54     have hh':=mul_le_mul_of_nonneg_right hh haR.le
  55     have hm:=mul_le_mul_of_nonneg_left hmlo (sq_nonneg lam)
  56     nlinarith only [htd,hh',hm]
  57   have hc:=hcap a m (by omega) hmhi d (by positivity) hd
  58   refine ⟨hM,hrowL,hcutL,?_⟩
  59   change (m*r:ℕ)/(K:ℝ)+((r:ℝ)+3)*(d+1)+4*(K+1)+3*((t:ℝ)*D)<β*a
  60   nlinarith only [hc,hbudgetL,mul_pos hβ haR]
