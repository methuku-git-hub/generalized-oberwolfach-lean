  22     let μ := FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (m i)))

  24       2*Real.exp (-(2*ξ^2/(c*L^2))*t) := by

  26   have hvar : (∑ i,(m i:ℝ))*(L/t)^2 ≤ c*L^2/t := by

  28       _ ≤ (c*t)*(L/t)^2 := mul_le_mul_of_nonneg_right hsteps (sq_nonneg _)
