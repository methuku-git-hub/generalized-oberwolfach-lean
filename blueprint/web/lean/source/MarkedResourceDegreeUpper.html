  13 theorem marked_incident_card_le_graph_degree (G:SimpleGraph (V⊕U))
  14     (S:Finset (Finset V⊕(V×U)))
  15     (hpairs:∀ e,Sum.inl e∈S→e.card=2)
  16     (hrim:∀ v w,Sum.inl ({v,w}:Finset V)∈S→G.Adj (Sum.inl v) (Sum.inl w))
  17     (hspoke:∀ v u,Sum.inr (v,u)∈S→G.Adj (Sum.inl v) (Sum.inr u))
  18     (x:V⊕U):
  19     (S.filter (markedPhysicalIncident x)).card≤G.degree x:=by
  20   have hsub:S.filter (markedPhysicalIncident x)⊆(G.neighborFinset x).image (markedPairResource x):=by
  21     intro e he
  22     obtain ⟨hmem,hi⟩:=Finset.mem_filter.mp he
  23     cases x with
  24     | inl v=>
  25       cases e with
  26       | inl e=>
  27         obtain ⟨a,b,hab,rfl⟩:=Finset.card_eq_two.mp (hpairs e hmem)
  28         have hv:v=a∨v=b:=by simpa only [markedPhysicalIncident,Finset.mem_insert,Finset.mem_singleton] using hi
  29         rcases hv with hv|hv
  30         · subst v
  31           exact Finset.mem_image.mpr ⟨Sum.inl b,(G.mem_neighborFinset _ _).mpr (hrim a b hmem),rfl⟩
  32         · subst v
  33           refine Finset.mem_image.mpr ⟨Sum.inl a,(G.mem_neighborFinset _ _).mpr (hrim a b hmem).symm,?_⟩
  34           simp only [markedPairResource,Finset.pair_comm]
  35       | inr z=>
  36         have hz:z.1=v:=hi
  37         refine Finset.mem_image.mpr ⟨Sum.inr z.2,(G.mem_neighborFinset _ _).mpr (hz ▸ hspoke z.1 z.2 hmem),?_⟩
  38         exact congrArg Sum.inr (Prod.ext hz.symm rfl)
  39     | inr u=>
  40       cases e with
  41       | inl e=>exact False.elim hi
  42       | inr z=>
  43         have hz:z.2=u:=hi
  44         refine Finset.mem_image.mpr ⟨Sum.inl z.1,(G.mem_neighborFinset _ _).mpr (hz ▸ (hspoke z.1 z.2 hmem).symm),?_⟩
  45         exact congrArg Sum.inr (Prod.ext rfl hz.symm)
  46   exact (Finset.card_le_card hsub).trans ((Finset.card_image_le).trans (G.card_neighborFinset_eq_degree x).le)
