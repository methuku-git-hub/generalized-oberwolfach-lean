  10 theorem reserve_capacity_inequality {k n δ d : ℝ}
  11     (hk : 2 ≤ k) (hn : 0 < n) (_hδ : 0 ≤ δ)
  12     (hthreshold : k * n ≤ (2 * k - 1) * δ) (hd : δ ≤ d) :
  13     0 < d + δ - n ∧ (d / k) / (d + δ - n) ≤ 1 := by
  14   have hk0 : 0 < k := by linarith
  15   have hhalf : n < 2 * δ := by
  16     by_contra h
  17     have hh : 2 * δ ≤ n := le_of_not_gt h
  18     have hp := mul_nonneg (by linarith : 0 ≤ k - 1 / 2) (sub_nonneg.mpr hh)
  19     nlinarith
  20   have hden : 0 < d + δ - n := by linarith
  21   refine ⟨hden, (div_le_one hden).mpr ?_⟩
  22   apply (div_le_iff₀ hk0).mpr
  23   have hp := mul_nonneg (by linarith : 0 ≤ k - 1) (sub_nonneg.mpr hd)
  24   nlinarith

  26 def intervalCommon {X Y Z : Type*} [Fintype Y]
  27     (L : X → Y → Prop) (R : Y → Z → Prop) (x : X) (z : Z) : ℕ :=
  28   (Finset.univ.filter (fun y => L x y ∧ R y z)).card

  30 def endpointReserve {X Y Z : Type*} [Fintype Y]
  31     (L : X → Y → Prop) (R : Y → Z → Prop) (Q : X → Z → ℝ)
  32     (x : X) (y : Y) (z : Z) : ℝ :=
  33   if L x y ∧ R y z then Q x z / intervalCommon L R x z else 0

  44 theorem endpointReserve_endpoints {X Y Z : Type*} [Fintype Y]
  45     (L : X → Y → Prop) (R : Y → Z → Prop) (Q : X → Z → ℝ)
  46     (hc : ∀ x z, 0 < intervalCommon L R x z) (x : X) (z : Z) :
  47     (∑ y, endpointReserve L R Q x y z) = Q x z := by
  48   unfold endpointReserve
  49   rw [← Finset.sum_filter]
  50   simp only [Finset.sum_const, nsmul_eq_mul]
  51   change (intervalCommon L R x z : ℝ) * (Q x z / intervalCommon L R x z) = _
  52   exact mul_div_cancel₀ _ (by exact_mod_cast (hc x z).ne')

  54 theorem intervalCommon_lower {X Y Z : Type*} [Fintype Y]
  55     (L : X → Y → Prop) (R : Y → Z → Prop) (x : X) (z : Z) :
  56     (Finset.univ.filter (L x)).card + (Finset.univ.filter (fun y => R y z)).card ≤
  57       intervalCommon L R x z + Fintype.card Y := by
  58   let A := Finset.univ.filter (L x)
  59   let B := Finset.univ.filter (fun y => R y z)
  60   have hc := Finset.card_union_add_card_inter A B
  61   have hu : (A ∪ B).card ≤ Fintype.card Y := Finset.card_le_univ _
  62   have hi : A ∩ B = Finset.univ.filter (fun y => L x y ∧ R y z) := by
  63     ext y
  64     simp only [A, B, Finset.mem_inter, Finset.mem_filter, Finset.mem_univ, true_and]
  65   unfold intervalCommon
  66   rw [hi] at hc
  67   dsimp only [A, B] at hc hu
  68   omega

  70 theorem endpointReserve_left_le {X Y Z : Type*} [Fintype Y] [Fintype Z]
  71     (L : X → Y → Prop) (R : Y → Z → Prop) (Q : X → Z → ℝ)
  72     (dX : X → ℝ) (k δ : ℝ)
  73     (hk : 2 ≤ k) (hn : 0 < Fintype.card Y) (hδ : 0 ≤ δ)
  74     (hthreshold : k * Fintype.card Y ≤ (2 * k - 1) * δ)
  75     (hQ : ∀ x z, 0 ≤ Q x z)
  76     (hrow : ∀ x, ∑ z, Q x z = dX x / k)
  77     (hdX : ∀ x, dX x = (Finset.univ.filter (L x)).card)
  78     (hd : ∀ x, δ ≤ dX x)
  79     (hdR : ∀ z, δ ≤ (Finset.univ.filter (fun y => R y z)).card)
  80     (x : X) (y : Y) :
  81     (∑ z, endpointReserve L R Q x y z) ≤ 1 := by
  82   have hn' : (0 : ℝ) < Fintype.card Y := by exact_mod_cast hn
  83   obtain ⟨hden, hcap⟩ := reserve_capacity_inequality hk hn' hδ hthreshold (hd x)
  84   have hbound (z : Z) :
  85       endpointReserve L R Q x y z ≤ Q x z / (dX x + δ - Fintype.card Y) := by
  86     have hc := intervalCommon_lower L R x z
  87     have hc' : dX x + (Finset.univ.filter (fun y => R y z)).card ≤
  88         (intervalCommon L R x z : ℝ) + Fintype.card Y := by
  89       rw [hdX x]
  90       exact_mod_cast hc
  91     have hdenle : dX x + δ - Fintype.card Y ≤ (intervalCommon L R x z : ℝ) := by
  92       linarith [hdR z]
  93     unfold endpointReserve
  94     split_ifs
  95     · exact div_le_div_of_nonneg_left (hQ x z) hden hdenle
  96     · exact div_nonneg (hQ x z) hden.le
  97   calc
  98     (∑ z, endpointReserve L R Q x y z) ≤
  99         ∑ z, Q x z / (dX x + δ - Fintype.card Y) :=
 100       Finset.sum_le_sum fun z _ => hbound z
 101     _ = (dX x / k) / (dX x + δ - Fintype.card Y) := by rw [← Finset.sum_div, hrow x]
 102     _ ≤ 1 := hcap

 104 theorem endpointReserve_right_le {X Y Z : Type*} [Fintype X] [Fintype Y]
 105     (L : X → Y → Prop) (R : Y → Z → Prop) (Q : X → Z → ℝ)
 106     (dZ : Z → ℝ) (k δ : ℝ)
 107     (hk : 2 ≤ k) (hn : 0 < Fintype.card Y) (hδ : 0 ≤ δ)
 108     (hthreshold : k * Fintype.card Y ≤ (2 * k - 1) * δ)
 109     (hQ : ∀ x z, 0 ≤ Q x z)
 110     (hcol : ∀ z, ∑ x, Q x z = dZ z / k)
 111     (hdZ : ∀ z, dZ z = (Finset.univ.filter (fun y => R y z)).card)
 112     (hd : ∀ z, δ ≤ dZ z)
 113     (hdL : ∀ x, δ ≤ (Finset.univ.filter (L x)).card)
 114     (y : Y) (z : Z) :
 115     (∑ x, endpointReserve L R Q x y z) ≤ 1 := by
 116   have hh := endpointReserve_left_le (fun z y => R y z) (fun y x => L x y)

 118     hcol hdZ hd hdL z y
 119   have heq (x : X) :
 120       endpointReserve (fun z y => R y z) (fun y x => L x y) (fun z x => Q x z) z y x =
 121       endpointReserve L R Q x y z := by
