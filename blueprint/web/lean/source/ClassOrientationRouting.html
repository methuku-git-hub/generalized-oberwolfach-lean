  11 theorem class_orientation_routing
  12     {V:Type} [Fintype V] [DecidableEq V]
  13     (S E Protected:SimpleGraph V) (M₀:V→V→ℕ) (m L:ℕ)
  14     (hregular:S.IsRegularOfDegree (2*m)) (hL:0<L) (hn:0<Fintype.card V)
  15     (a f β:ℝ) (ha:0≤a) (hf:0≤f) (hβ:0<β)
  16     (hphysical:∀x y,M₀ x y+M₀ y x=if S.Adj x y then 1 else 0)
  17     (herr:∀v,|((∑w,M₀ v w:ℕ):ℝ)-m|≤a)
  18     (hforbid:∀v,(Protected.degree v:ℝ)≤f)
  19     (hpaths:∀x y,x≠y→β*Fintype.card V≤
  20       ((Finset.univ.filter (fun z=>0<M₀ x z ∧ 0<M₀ z y ∧ E.Adj x z ∧ E.Adj z y)).card:ℝ))
  21     (hmargin:(Fintype.card V:ℝ)*a/L+2*a+4*L+2*f<β*Fintype.card V):
  22     ∃M:V→V→ℕ,
  23       (∀x y,M x y+M y x=if S.Adj x y then 1 else 0) ∧
  24       (∀v,(∑w,M v w)=m ∧ (∑w,M w v)=m) ∧
  25       (∀x y,M x y≠M₀ x y ∨ M y x≠M₀ y x→E.Adj x y ∧ ¬Protected.Adj x y) ∧
  26       (∀v,((Finset.univ.filter (fun w=>M v w≠M₀ v w ∨ M w v≠M₀ w v)).card:ℝ)≤a+2*L):=by
  27   have htotal:(∑_v:V,m)=S.edgeFinset.card:=by
  28     have hh:=S.sum_degrees_eq_twice_card_edges
  29     have hdeg:∀v,S.degree v=2*m:=hregular
  30     simp only [hdeg,Finset.sum_const,Finset.card_univ,nsmul_eq_mul] at hh ⊢
  31     nlinarith [hh]
  32   obtain ⟨M,hphys,hout,havoid,hchange⟩:=orientationRouting S E Protected M₀ (fun _=>m) L hL hn
  33     ha hf hβ hphysical htotal herr hforbid hpaths hmargin
  34   refine ⟨M,hphys,?_,havoid,hchange⟩
  35   intro v
  36   have hd:=orientationMatrix_degree S M hphys v
  37   rw [hout v,hregular v] at hd
  38   exact ⟨hout v,by omega⟩
  39 end Oberwolfach
