  19 def localOrdinaryParent {P Q F : SimpleGraph V} {T : Finset V} (hQP : Q≤P)
  20     (C : LocalOrdinaryCycle Q F T r) : OrdinaryCycle P r := ⟨C.val.val,C.val.property.trans hQP⟩

  22 def localOrdinaryParentEquiv (P Q F : SimpleGraph V) (T : Finset V) (hQP : Q≤P) :
  23     LocalOrdinaryCycle Q F T r ≃ {C : OrdinaryCycle P r // C.val.val≤Q ∧
  24       (∀ v∈C.val.val.support,v∈T) ∧ ∀ x∈C.val.val.support,∀ y∈C.val.val.support,¬F.Adj x y} where
  25   toFun C := ⟨localOrdinaryParent hQP C,C.val.property,C.property⟩
  26   invFun C := ⟨⟨C.val.val,C.property.1⟩,C.property.2⟩
  27   left_inv _ := rfl
  28   right_inv _ := rfl

  30 theorem localOrdinary_sum (P Q F : SimpleGraph V) (T : Finset V) (hQP : Q≤P)
  31     (f : OrdinaryCycle P r→ℝ) :
  32     (∑ C : LocalOrdinaryCycle Q F T r,f (localOrdinaryParent hQP C))=
  33       ∑ C : OrdinaryCycle P r,if C.val.val≤Q ∧ (∀ v∈C.val.val.support,v∈T) ∧
  34         (∀ x∈C.val.val.support,∀ y∈C.val.val.support,¬F.Adj x y) then f C else 0 := by
  35   let good := fun C : OrdinaryCycle P r=>C.val.val≤Q ∧ (∀ v∈C.val.val.support,v∈T) ∧
  36     (∀ x∈C.val.val.support,∀ y∈C.val.val.support,¬F.Adj x y)
  37   have hh := finite_sum_restrict_of_zero good (fun C=>if good C then f C else 0) (fun C h=>if_neg h)
  38   have hs := (localOrdinaryParentEquiv (r:=r) P Q F T hQP).symm.sum_comp
  39     (fun C=>f (localOrdinaryParent hQP C))
  40   calc
  41     _ = ∑ C : {C : OrdinaryCycle P r // good C},f C.val := hs.symm
  42     _ = ∑ C : {C : OrdinaryCycle P r // good C},if good C.val then f C.val else 0 := by
  43       apply Finset.sum_congr rfl;intro C _;exact (if_pos C.property).symm
  44     _ = _ := by convert hh.symm using 1 <;> congr!

  46 def ordinaryLocalPotential (P Q F : SimpleGraph V) (T : Finset V)
  47     (φ : OrdinaryCycle P r→ℝ) (v : V) : ℝ :=
  48   ∑ C,if v∈C.val.val.support ∧ C.val.val≤Q ∧
  49     (∀ x∈C.val.val.support,∀ y∈C.val.val.support,¬F.Adj x y) ∧
  50     (∀ z∈C.val.val.support,z≠v → z∈T) then φ C else 0

  56 theorem ordinaryLocalPotential_present (P Q F : SimpleGraph V) (T : Finset V) (hQP : Q≤P)
  57     (φ : OrdinaryCycle P r→ℝ) (v : V) (hv : v∈T) :
  58     ordinaryLocalPotential P Q F T φ v=
  59       hypergraphWeightedDegree localOrdinaryVertices (fun C : LocalOrdinaryCycle Q F T r=>φ (localOrdinaryParent hQP C)) v := by
  60   have hh := localOrdinary_sum P Q F T hQP (fun C=>if v∈C.val.val.support then φ C else 0)
  61   unfold hypergraphWeightedDegree
  62   simp only [localOrdinaryVertices,Set.mem_toFinset]
  63   change ordinaryLocalPotential P Q F T φ v=∑ C : LocalOrdinaryCycle Q F T r,
  64     if v∈(localOrdinaryParent hQP C).val.val.support then φ (localOrdinaryParent hQP C) else 0
  65   rw [hh]
  66   unfold ordinaryLocalPotential
  67   apply Finset.sum_congr rfl
  68   intro C _
  69   have ht : (∀ z∈C.val.val.support,z≠v → z∈T)↔∀ z∈C.val.val.support,z∈T := by
  70     constructor
  71     · intro h z hz;by_cases he : z=v
  72       · exact he ▸ hv
  73       · exact h z hz he
  74     · intro h z hz _;exact h z hz
  75   simp only [ht]
  76   by_cases hroot : v∈C.val.val.support <;> by_cases hQ : C.val.val≤Q <;>
  77     by_cases hT : ∀ z∈C.val.val.support,z∈T <;>
  78     by_cases hF : ∀ x∈C.val.val.support,∀ y∈C.val.val.support,¬F.Adj x y <;>
