   4 /-! Weighted incidence sums count each physical undirected edge twice.
   5 This identity keeps unordered endpoint-edge weights from being doubled. -/

  15 theorem weighted_graph_handshake (G : SimpleGraph V) (weight : Sym2 V→ℝ) :
  16     (∑ v,∑ w,if G.Adj v w then weight s(v,w) else 0)=2*∑ e∈G.edgeFinset,weight e := by
  17   let e : (Σ v,G.neighborSet v)≃G.Dart :=
  18     { toFun := fun p=>⟨(p.1,p.2.val),p.2.property⟩
  19       invFun := fun d=>⟨d.fst,⟨d.snd,d.adj⟩⟩
  20       left_inv := by intro p;cases p;rfl
  21       right_inv := by intro d;cases d;rfl }
  22   have hrow (v : V) : (∑ w:G.neighborSet v,weight s(v,w.val))=
  23       ∑ w,if G.Adj v w then weight s(v,w) else 0 := by
  24     have hh:=Finset.sum_subtype (F:=inferInstance) (p:=fun w=>G.Adj v w)
  25       (G.neighborFinset v) (fun w=>by simp) (fun w=>weight s(v,w))
  26     have hx : (∑ w:G.neighborSet v,weight s(v,w.val))=∑ w∈G.neighborFinset v,weight s(v,w) := by
  27       convert hh.symm using 1 <;> congr!
  28     rw [hx,SimpleGraph.neighborFinset_eq_filter,Finset.sum_filter]
  29   have hdarts : (∑ v,∑ w,if G.Adj v w then weight s(v,w) else 0)=
  30       ∑ d:G.Dart,weight d.edge := by
  31     rw [←e.sum_comp (fun d=>weight d.edge),Fintype.sum_sigma]
  32     apply Finset.sum_congr rfl
  33     intro v hv
  34     exact (hrow v).symm
  35   rw [hdarts]
  36   have hf:=Finset.sum_fiberwise_of_maps_to (s:=Finset.univ) (t:=G.edgeFinset)
  37     (g:=SimpleGraph.Dart.edge) (fun d _=>by rw [SimpleGraph.mem_edgeFinset];exact d.edge_mem) (fun d=>weight d.edge)
  38   rw [←hf,Finset.mul_sum]
  39   apply Finset.sum_congr rfl
  40   intro edge hedge
  41   have hsum : (∑ d:G.Dart with d.edge=edge,weight d.edge)=
  42       (Finset.univ.filter (fun d:G.Dart=>d.edge=edge)).card*weight edge := by
  43     calc
  44       _ = ∑ _d:G.Dart with _d.edge=edge,weight edge := by
  45         apply Finset.sum_congr rfl
  46         intro d hd
  47         rw [(Finset.mem_filter.mp hd).2]
  48       _ = _ := by simp
  49   rw [hsum,G.dart_edge_fiber_card edge (by simpa using hedge)]
  50   norm_num
