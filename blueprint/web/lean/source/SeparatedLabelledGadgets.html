  15 theorem separated_labelled_gadgets (r N : ℕ) (Fgraph : SimpleGraph V) (R : Finset V) (B : Finset U)
  16     (P : Fin N → (Fin r ↪ V) → U → Prop)
  17     (hlocal : ∀ i (S : Finset V) (T : Finset U),R⊆S → B⊆T →
  18       S.card≤R.card+N*r → T.card≤B.card+N →
  19       ∃ (f : Fin r ↪ V) (u : U),P i f u ∧ (∀ a,f a∉S) ∧ u∉T ∧ (∀ a z,z∈S → ¬Fgraph.Adj (f a) z)) :
  20     ∃ (f : Fin N → Fin r ↪ V) (u : Fin N → U),
  21       (∀ i,P i (f i) (u i) ∧ (∀ a,f i a∉R) ∧ u i∉B) ∧
  22       (∀ i j,i≠j → u i≠u j ∧ ∀ a b,f i a≠f j b ∧ ¬Fgraph.Adj (f i a) (f j b)) := by
  23   let allowed := fun i (o : Option ((Fin r ↪ V)×U)) =>
  24     ∃ f u,o=some (f,u) ∧ P i f u ∧ (∀ a,f a∉R) ∧ u∉B
  25   let compatible := fun (o p : Option ((Fin r ↪ V)×U)) =>
  26     ∀ f u g v,o=some (f,u) → p=some (g,v) → u≠v ∧ ∀ a b,f a≠g b ∧ ¬Fgraph.Adj (f a) (g b)
  27   obtain ⟨F,hF,hcomp⟩ := sequential_select_prefix N allowed compatible none (by
  28     intro i F _hprefix
  29     let J := Finset.univ.filter (fun j : Fin N => j.val < i.val)
  30     let S := R∪J.biUnion (fun j => optionLabelledRows (F j))
  31     let T := B∪J.biUnion (fun j => optionLabelledHubs (F j))
  32     have hJ : J.card≤N := (Finset.card_le_univ _).trans_eq (Fintype.card_fin N)
  33     have hS : S.card≤R.card+N*r := by
  34       have hsum : (J.biUnion (fun j => optionLabelledRows (F j))).card≤J.card*r := by
  35         calc
  36           _ ≤ ∑ j∈J,(optionLabelledRows (F j)).card := Finset.card_biUnion_le
  37           _ ≤ ∑ _j∈J,r := Finset.sum_le_sum (fun _ _ => optionLabelledRows_card _)
  38           _ = _ := by simp
  39       have hh := Finset.card_union_le R (J.biUnion (fun j => optionLabelledRows (F j)))
  40       dsimp [S]
  41       nlinarith
  42     have hT : T.card≤B.card+N := by
  43       have hsum : (J.biUnion (fun j => optionLabelledHubs (F j))).card≤J.card := by
  44         calc
  45           _ ≤ ∑ j∈J,(optionLabelledHubs (F j)).card := Finset.card_biUnion_le
  46           _ ≤ ∑ _j∈J,1 := Finset.sum_le_sum (fun _ _ => optionLabelledHubs_card _)
  47           _ = _ := by simp
  48       have hh := Finset.card_union_le B (J.biUnion (fun j => optionLabelledHubs (F j)))
  49       dsimp [T]
  50       omega
  51     obtain ⟨f,u,hP,hf,hu,hFsep⟩ := hlocal i S T Finset.subset_union_left Finset.subset_union_left hS hT
  52     refine ⟨some (f,u),⟨f,u,rfl,hP,?_,?_⟩,?_⟩
  53     · exact fun a h => hf a (Finset.mem_union_left _ h)
  54     · exact fun h => hu (Finset.mem_union_left _ h)
  55     · intro j hj g v f' u' hg hfu
  56       have he : (f,u)=(f',u') := Option.some.inj hfu
  57       cases he
  58       have hjJ : j∈J := Finset.mem_filter.mpr ⟨Finset.mem_univ _,hj⟩
  59       refine ⟨?_,?_⟩
  60       · intro he
  61         apply hu
  62         apply Finset.mem_union_right
  63         apply Finset.mem_biUnion.mpr
  64         refine ⟨j,hjJ,?_⟩
  65         rw [hg]
  66         simpa [optionLabelledHubs] using he.symm
  67       · intro a b
  68         have haS : g a∈S := by
  69           apply Finset.mem_union_right
  70           apply Finset.mem_biUnion.mpr
  71           refine ⟨j,hjJ,?_⟩
  72           rw [hg]
  73           exact Finset.mem_image.mpr ⟨a,Finset.mem_univ _,rfl⟩
  74         exact ⟨fun he => hf b (he ▸ haS),fun hadj => hFsep b (g a) haS hadj.symm⟩)
  75   choose f u he hP hf hu using hF
  76   refine ⟨f,u,(fun i => ⟨hP i,hf i,hu i⟩),?_⟩
  77   intro i j hij
  78   rcases lt_or_gt_of_ne hij with h|h
  79   · exact hcomp i j h (f i) (u i) (f j) (u j) (he i) (he j)
  80   · have hh := hcomp j i h (f j) (u j) (f i) (u i) (he j) (he i)
  81     exact ⟨Ne.symm hh.1,fun a b => ⟨Ne.symm (hh.2 b a).1,fun he => (hh.2 b a).2 he.symm⟩⟩

  83 theorem separated_labelled_gadgets_finite {I : Type*} [Fintype I]
  84     (r : ℕ) (Fgraph : SimpleGraph V) (R : Finset V) (B : Finset U) (P : I → (Fin r ↪ V) → U → Prop)
  85     (hlocal : ∀ i (S : Finset V) (T : Finset U),R⊆S → B⊆T →
  86       S.card≤R.card+Fintype.card I*r → T.card≤B.card+Fintype.card I →
  87       ∃ (f : Fin r ↪ V) (u : U),P i f u ∧ (∀ a,f a∉S) ∧ u∉T ∧ (∀ a z,z∈S → ¬Fgraph.Adj (f a) z)) :
  88     ∃ (f : I → Fin r ↪ V) (u : I → U),
  89       (∀ i,P i (f i) (u i) ∧ (∀ a,f i a∉R) ∧ u i∉B) ∧
  90       (∀ i j,i≠j → u i≠u j ∧ ∀ a b,f i a≠f j b ∧ ¬Fgraph.Adj (f i a) (f j b)) := by
  91   let e := Fintype.equivFin I
  92   obtain ⟨f,u,hP,hsep⟩ := separated_labelled_gadgets r (Fintype.card I) Fgraph R B
  93     (fun i => P (e.symm i)) (fun i => hlocal (e.symm i))
  94   refine ⟨fun i => f (e i),fun i => u (e i),?_,?_⟩
  95   · intro i
  96     simpa only [e.symm_apply_apply] using hP (e i)
  97   · intro i j hij
  98     exact hsep (e i) (e j) (fun he => hij (e.injective he))
