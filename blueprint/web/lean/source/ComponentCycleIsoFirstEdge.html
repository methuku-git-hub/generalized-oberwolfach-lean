  20 theorem component_cycle_iso_with_first_edge
  21     {X:Type} [Fintype X] [DecidableEq X] (G:SimpleGraph X) (hG:G.IsRegularOfDegree 2)
  22     (C:G.ConnectedComponent) (a b:C) (hab:G.Adj a.val b.val):
  23     ∃ e:SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph,
  24       e ⟨0,by have h:3≤Fintype.card C:=(component_regular_two_cycle_iso G hG C).1;omega⟩=a ∧
  25       e ⟨1,by have h:3≤Fintype.card C:=(component_regular_two_cycle_iso G hG C).1;omega⟩=b:=by
  26   classical
  27   have h3:3≤Fintype.card C:=(component_regular_two_cycle_iso G hG C).1
  28   let m:=Fintype.card C-2
  29   have hN:Fintype.card C=m+2:=by dsimp [m];omega
  30   obtain ⟨e₀⟩:Nonempty (SimpleGraph.cycleGraph (m+2) ≃g C.toSimpleGraph):=by
  31     have hh:=(component_regular_two_cycle_iso G hG C).2
  32     rw [hN] at hh
  33     exact hh
  34   let f:Fin (m+2)↪C:=e₀.toEquiv.toEmbedding
  35   have hmap:(SimpleGraph.cycleGraph (m+2)).map f=C.toSimpleGraph:=by
  36     ext x y
  37     obtain ⟨x,rfl⟩:=e₀.surjective x
  38     obtain ⟨y,rfl⟩:=e₀.surjective y
  39     change ((SimpleGraph.cycleGraph (m+2)).map f).Adj (f x) (f y) ↔ C.toSimpleGraph.Adj (e₀ x) (e₀ y)
  40     rw [SimpleGraph.map_adj_apply]
  41     exact (e₀.map_adj_iff (v:=x) (w:=y)).symm
  42   have hab':((SimpleGraph.cycleGraph (m+2)).map f).Adj a b:=by
  43     rw [hmap]
  44     exact hab
  45   obtain ⟨g,hgmap,hg0,hg1⟩:=ordered_cycle_representation_first_edge_exists m f a b hab'
  46   rw [hmap] at hgmap
  47   have hcard:Fintype.card (Fin (m+2))=Fintype.card C:=by simp [hN]
  48   have hbij:Function.Bijective g:=(Fintype.bijective_iff_injective_and_card g).2 ⟨g.injective,hcard⟩
  49   let E:Fin (m+2)≃C:=Equiv.ofBijective g hbij
  50   let e:SimpleGraph.cycleGraph (m+2) ≃g C.toSimpleGraph:={
  51     toEquiv:=E
  52     map_rel_iff':=by
  53       intro x y
  54       rw [←hgmap]
  55       exact SimpleGraph.map_adj_apply }
  56   have he0:e 0=a:=hg0
  57   have he1:e 1=b:=hg1
  58   exact cycle_iso_first_edges_transport C.toSimpleGraph (Fintype.card C) (m+2) (by omega) (by omega) hN e a b
  59     (by simpa using he0) (by simpa using he1)
