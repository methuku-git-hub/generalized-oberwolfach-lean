  12     (ext:Fin (L+5)↪X) (p:F.Walk (ext ⟨0,by omega⟩) (ext ⟨L+4,by omega⟩))
  13     (hsupport:p.support=List.ofFn ext) : p.support.toFinset=Finset.univ.image ext := by
