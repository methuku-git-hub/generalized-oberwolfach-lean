  12 theorem two_factor_residual_count_after_disjoint_deletions
  13     (F:SimpleGraph V) (hF:F.IsRegularOfDegree 2)
  14     (s:Finset I) (u v:I→V) (P:∀ i,F.Walk (u i) (v i))
  15     (hp:∀i∈s,(P i).IsPath) (hne:∀i∈s,u i≠v i)
  16     (hP:(↑s:Set I).PairwiseDisjoint (fun i=>(P i).support.toFinset))
  17     (t:Finset J) (root:J→V) (C:∀ j,F.Walk (root j) (root j))
  18     (hc:∀j∈t,(C j).IsCycle)
  19     (hC:(↑t:Set J).PairwiseDisjoint (fun j=>(C j).support.toFinset))
  20     (hcross:∀i∈s,∀j∈t,Disjoint (P i).support.toFinset (C j).support.toFinset):
  21     let deletedV := (t.biUnion (fun j=>(C j).support.toFinset)) ∪
  22       (s.biUnion (fun i=>(P i).support.toFinset \ {u i,v i}))
  23     let deletedE := (t.biUnion (fun j=>(C j).edges.toFinset)) ∪
  24       (s.biUnion (fun i=>(P i).edges.toFinset));
  25     (F.edgeFinset \ deletedE).card + s.card = (Finset.univ \ deletedV).card := by
  26   let CV := t.biUnion (fun j=>(C j).support.toFinset)
  27   let CE := t.biUnion (fun j=>(C j).edges.toFinset)
  28   let PV := s.biUnion (fun i=>(P i).support.toFinset \ {u i,v i})
  29   let PE := s.biUnion (fun i=>(P i).edges.toFinset)
  30   have hcvpe : Disjoint CV PV := by
  31     apply Finset.disjoint_left.mpr
  32     intro x hx hy
  33     obtain ⟨j,hj,hx⟩:=Finset.mem_biUnion.mp hx
  34     obtain ⟨i,hi,hy⟩:=Finset.mem_biUnion.mp hy
  35     exact Finset.disjoint_left.mp (hcross i hi j hj) (Finset.mem_sdiff.mp hy).1 hx
  36   have hcepe : Disjoint CE PE := by
  37     apply Finset.disjoint_left.mpr
  38     intro e hx hy
  39     obtain ⟨j,hj,hx⟩:=Finset.mem_biUnion.mp hx
  40     obtain ⟨i,hi,hy⟩:=Finset.mem_biUnion.mp hy
  41     induction e using Sym2.ind with
  42     | _ x y =>
  43       have hx' : x∈(C j).support.toFinset := by
  44         apply List.mem_toFinset.mpr
  45         exact (C j).fst_mem_support_of_mem_edges (List.mem_toFinset.mp hx)
  46       have hy' : x∈(P i).support.toFinset := by
  47         apply List.mem_toFinset.mpr
  48         exact (P i).fst_mem_support_of_mem_edges (List.mem_toFinset.mp hy)
  49       exact Finset.disjoint_left.mp (hcross i hi j hj) hy' hx'
  50   have hEsub : CE∪PE⊆F.edgeFinset := by
  51     intro e he
  52     rcases Finset.mem_union.mp he with he|he
  53     · obtain ⟨j,hj,hj'⟩:=Finset.mem_biUnion.mp he
  54       exact SimpleGraph.mem_edgeFinset.mpr ((C j).edges_subset_edgeSet (List.mem_toFinset.mp hj'))
  55     · obtain ⟨i,hi,hi'⟩:=Finset.mem_biUnion.mp he
  56       exact SimpleGraph.mem_edgeFinset.mpr ((P i).edges_subset_edgeSet (List.mem_toFinset.mp hi'))
  57   have hVsub:CV∪PV⊆(Finset.univ:Finset V) := Finset.subset_univ _
  58   have hEc := Finset.card_sdiff_add_card_eq_card hEsub
  59   have hVc := Finset.card_sdiff_add_card_eq_card hVsub
  60   have hCE:CE.card=CV.card:=disjoint_cycle_walks_edge_vertex_count F t root C hc hC
  61   have hPE:PE.card=PV.card+s.card:=disjoint_simple_paths_edges_and_internal_vertices F s u v P hp hne hP
  62   have hEunion:(CE∪PE).card=CE.card+PE.card:=Finset.card_union_of_disjoint hcepe
  63   have hVunion:(CV∪PV).card=CV.card+PV.card:=Finset.card_union_of_disjoint hcvpe
  64   have hFcard:=two_factor_edge_count F hF
  65   change (F.edgeFinset \ (CE∪PE)).card+s.card = ((Finset.univ:Finset V) \ (CV∪PV)).card
  66   rw [hEunion,hCE,hPE,hFcard] at hEc
  67   rw [hVunion,Finset.card_univ] at hVc
  68   omega
  69 end Oberwolfach
