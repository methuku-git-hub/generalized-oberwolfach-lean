  15  theorem finRotate_ne_self_of_two_le {n:ℕ} (hn:2≤n) (i:Fin n):finRotate n i≠i:=by
  16   obtain ⟨k,rfl⟩:=Nat.exists_eq_succ_of_ne_zero (by omega: n≠0)
  17   intro he
  18   have hh:=congrArg Fin.val he
  19   rw [coe_finRotate] at hh
  20   split_ifs at hh with hi
  21   · have hi':i.val=k:=congrArg Fin.val hi

  25  theorem finite_permutation_without_fixed_points (I:Type*) [Fintype I]
  26     (hcard:Fintype.card I≠1):∃ σ:Equiv.Perm I,∀ i,σ i≠i:=by
  27   let e:=Fintype.equivFin I
  28   refine ⟨(e.trans (finRotate (Fintype.card I))).trans e.symm,?_⟩
  29   intro i hi
  30   have hpos:0<Fintype.card I:=Fintype.card_pos_iff.mpr ⟨i⟩
  31   have hh:=congrArg e hi
  32   simp only [Equiv.trans_apply,Equiv.apply_symm_apply] at hh
  33   exact finRotate_ne_self_of_two_le (by omega) (e i) hh

  37  theorem transformer_endpoint_assignments (Q:SimpleGraph V) (H:V→U→Prop)
  38     (hbalance:∀ v,Q.degree v=2*((Finset.univ:Finset U).filter (H v)).card):
  39     ∃ (σ:∀ v,Equiv.Perm (Q.neighborSet v))
  40       (b:∀ v,Q.neighborSet v≃({u:U // H v u}×Fin 2)),
  41       (∀ v w,σ v w≠w) ∧
  42       (∀ v w,s(v,(σ v w).val)≠s(v,w.val)):=by
  43   have hcard (v:V):Fintype.card (Q.neighborSet v)=2*Fintype.card {u:U // H v u}:=by
  44     rw [Set.fintypeCard_eq_ncard,SimpleGraph.ncard_neighborSet]
  45     simpa only [Fintype.card_subtype,Finset.card_filter] using hbalance v
  46   have hperm (v:V):∃ σ:Equiv.Perm (Q.neighborSet v),∀ w,σ w≠w:=by
  47     apply finite_permutation_without_fixed_points
  48     rw [hcard]
  49     omega
  50   choose σ hσ using hperm
  51   have hb (v:V):Fintype.card (Q.neighborSet v)=Fintype.card ({u:U // H v u}×Fin 2):=by
  52     rw [Fintype.card_prod,Fintype.card_fin,hcard,Nat.mul_comm]
  53   refine ⟨σ,fun v=>Fintype.equivOfCardEq (hb v),hσ,?_⟩
  54   intro v w he
  55   exact hσ v w (Subtype.ext (sym2_fixed_endpoint_injective v he))
