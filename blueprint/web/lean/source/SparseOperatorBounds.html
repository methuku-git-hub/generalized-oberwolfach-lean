  13 theorem sparseRectangleCoefficient_bound (H : A → B → Prop) (q : B → ℝ) (z : A×B → ℝ)
  14     (L : ℝ) (hL : 0≤L) (hq : ∀ u,0≤q u ∧ q u≤L) (x y : A) (u v : B) :
  15     |sparseRectangleCoefficient H q z x y u v|≤
  16       L^3*‖z‖/((Fintype.card A:ℝ)*Fintype.card B) := by
  17   have hz : |z (x,u)|≤‖z‖ := by simpa only [Real.norm_eq_abs] using norm_le_pi_norm z (x,u)
  18   have hq0 := (hq u).1
  19   have hq1 := (hq v).1
  20   have hp : q u*(q v)^2≤L^3 := by
  21     calc
  22       _ ≤ L*L^2 := mul_le_mul (hq u).2 (pow_le_pow_left₀ hq1 (hq v).2 2) (sq_nonneg _) hL
  23       _ = _ := by ring
  24   unfold sparseRectangleCoefficient bipAdj
  25   split_ifs <;> simp only [one_mul,mul_one,zero_mul,mul_zero,zero_div,abs_zero]
  26   all_goals first
  27   | positivity
  28   | simp only [abs_div,abs_mul,abs_of_nonneg hq0,abs_of_nonneg (sq_nonneg (q v)),
  29         abs_of_nonneg (show (0:ℝ)≤(Fintype.card A:ℝ)*Fintype.card B by positivity)]
  30     apply div_le_div_of_nonneg_right _ (by positivity)
  31     calc
  32       _ = |z (x,u)| *(q u*(q v)^2) := by ring
  33       _ ≤ ‖z‖*L^3 := mul_le_mul hz hp (by positivity) (norm_nonneg _)
  34       _ = _ := by ring

  49 theorem sparseOppositeMap_norm (H : A → B → Prop) (q : B → ℝ) (z : A×B → ℝ)
  50     (L : ℝ) (hL : 0≤L) (hq : ∀ u,0≤q u ∧ q u≤L)
  51     (ha : 0<Fintype.card A) (hb : 0<Fintype.card B) :
  52     ‖sparseOppositeMap H q z‖≤L^3*‖z‖ := by
  53   have ha' : (0:ℝ)<Fintype.card A := Nat.cast_pos.mpr ha
  54   have hb' : (0:ℝ)<Fintype.card B := Nat.cast_pos.mpr hb
  55   apply (pi_norm_le_iff_of_nonneg (show 0≤L^3*‖z‖ by positivity)).mpr
  56   intro e
  57   rcases e with ⟨x,u⟩
  58   have hq0 := (hq u).1
  59   have hsum : |∑ y,∑ v,bipAdj H x v*bipAdj H y u*q v*z (y,v)|≤
  60       (Fintype.card A:ℝ)*(Fintype.card B:ℝ)*(L*‖z‖) := by
  61     calc
  62       _ ≤ ∑ y,|∑ v,bipAdj H x v*bipAdj H y u*q v*z (y,v)| := Finset.abs_sum_le_sum_abs _ _
  63       _ ≤ ∑ y,∑ v,|bipAdj H x v*bipAdj H y u*q v*z (y,v)| := by
  64         apply Finset.sum_le_sum;intro y _;exact Finset.abs_sum_le_sum_abs _ _
  65       _ ≤ ∑ _y : A,∑ _v : B,L*‖z‖ := by
  66         apply Finset.sum_le_sum;intro y _
  67         apply Finset.sum_le_sum;intro v _
  68         have hqv := (hq v).1
  69         have hz : |z (y,v)|≤‖z‖ := by simpa only [Real.norm_eq_abs] using norm_le_pi_norm z (y,v)
  70         unfold bipAdj
  71         split_ifs <;> simp only [one_mul,zero_mul,abs_zero]
  72         all_goals first
  73         | positivity
  74         | rw [abs_mul,abs_of_nonneg hqv]
  75           exact mul_le_mul (hq v).2 hz (abs_nonneg _) hL
  76       _ = _ := by simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul];ring
