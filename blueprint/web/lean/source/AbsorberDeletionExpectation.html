  32   let F := fun d : D => labelledAbsorberFree R (S d.1) X d.2
  33   let w := fun d : D => if Disjoint (F d) Z then (1:ℝ) else 0
  34   have hw (d : D) (hn : w d≠0) : Disjoint (F d) Z := by
  35     by_contra hd
  36     simp only [w,hd,ite_false,ne_eq,not_true_eq_false] at hn
  37   have hF : ∀ d : D,w d≠0 → F d⊆Finset.univ.map (profileComplementEmbedding Z) := by
  38     intro d hn v hv
  39     have hz : v∉Z := fun hz => Finset.disjoint_left.mp (hw d hn) hv hz
  40     obtain ⟨u,hu⟩ := (profileComplementEmbedding_range Z v).mpr hz
  41     exact Finset.mem_map.mpr ⟨u,Finset.mem_univ _,hu⟩
  42   have ha : ∀ d : D,w d≠0 → ∀ i,
  43       (Finset.univ.filter (fun u : ProfileRemainingClass Z i => profileComplementEmbedding Z ⟨i,u⟩∈F d)).card=c*a i := by
  44     intro d hn i
  45     exact (profile_avoiding_preimage_class_card (F d) Z (hw d hn) i).trans
  46       (labelled_absorber_free_class_card R (S d.1) X d.2 a hR i)
  47   have hex := embedded_profile_weighted_support_expectation (profileComplementEmbedding Z) F w
  48     (fun i => c*a i) m hF ha hm
