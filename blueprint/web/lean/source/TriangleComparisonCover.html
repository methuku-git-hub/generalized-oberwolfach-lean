  16 theorem triangle_comparison_cover_of_factor (Q H F : SimpleGraph V) (A B X : Finset V)
  17     (hAB : Disjoint A B) (hXA : X⊆A) (x : V) (hxA : x∉X) (hxB : x∉B)
  18     (hA : Q.neighborFinset x⊆X) (hB : H.neighborFinset x⊆B)
  19     (hHA : ∀ a∈X,H.neighborFinset a⊆B)
  20     (hF : ∀ a,Q.Adj x a → ¬F.Adj x a)
  21     (M : Finset (OrdinaryWheel Q F H.Adj 3))
  22     (hM : ∀ W∈M,x∉(ordinaryWheelRim W).support)
  23     (hfactor : HasResourceDecomposition
  24       (endpointTriangleVertices
  25         (Q:=endpointHostGraph
  26           (Q\spanningRestriction (ordinaryWheelRimGraph M) X)
  27           (H\bipartiteRelationGraph X B (hAB.mono_left hXA) (ordinaryPackingSpokes M)) A B)
  28         (F:=F) (S:=Q.neighborFinset x) (T:=H.neighborFinset x))
  29       (Q.neighborFinset x∪H.neighborFinset x)) :
  30     ∃ N : Finset (OrdinaryWheel Q F H.Adj 3), OrdinaryRowCover N x X 0 ∧
  31       Disjoint (N.biUnion ordinaryWheelResources) (M.biUnion ordinaryWheelResources) := by
  32   let D:=spanningRestriction (ordinaryWheelRimGraph M) X
  33   let E:=bipartiteRelationGraph X B (hAB.mono_left hXA) (ordinaryPackingSpokes M)
  34   have hxD (a:V):¬D.Adj x a:=by
  35     intro h;exact hxA ((spanningRestriction_adj _ X x a).mp h).2.1
  36   have hxE (a:V):¬E.Adj x a:=bipartiteRelationGraph_outside X B (hAB.mono_left hXA) _ hxA hxB
  37   have hQr (a:V):(Q\D).Adj x a ↔ Q.Adj x a:=by simp [SimpleGraph.sdiff_adj,hxD]
  38   have hHr (a:V):(H\E).Adj x a ↔ H.Adj x a:=by simp [SimpleGraph.sdiff_adj,hxE]
  39   have hQN:(Q\D).neighborFinset x=Q.neighborFinset x:=by
  40     ext a;simp only [SimpleGraph.mem_neighborFinset,hQr]
  41   have hHN:(H\E).neighborFinset x=H.neighborFinset x:=by
  42     ext a;simp only [SimpleGraph.mem_neighborFinset,hHr]
  43   obtain ⟨N,hN⟩:=endpoint_factor_row_cover (Q\D) (H\E) F A B X
  44     (Q.neighborFinset x) (H.neighborFinset x) hAB hXA x hA hB
  45     (fun a=>(hQr a).trans (Q.mem_neighborFinset x a).symm)
  46     (fun u=>(hHr u).trans (H.mem_neighborFinset x u).symm)
  47     (fun a ha=>hF a ha.1) hfactor
  48   have hspoke (a u:V) (h:(H\E).Adj a u):H.Adj a u ∧ ¬(a∈X∧ordinaryPackingSpokes M a u):=by
  49     refine ⟨h.1,?_⟩
  50     rintro ⟨ha,hold⟩
  51     have hu:u∈B:=hHA a ha ((H.mem_neighborFinset a u).mpr h.1)
  52     exact h.2 (Or.inl ⟨ha,hu,hold⟩)
  53   let C:N→OrdinaryWheel (Q\D) F (fun a u=>H.Adj a u∧¬(a∈X∧ordinaryPackingSpokes M a u)) 3:=
  54     fun W=>ordinaryWheelToLarger le_rfl hspoke W.val
  55   have he:Finset.univ.image (fun W:N=>W.val)=N:=by ext W;simp
  56   have hr:wheelFamilyRim C=ordinaryWheelRimGraph N:=by
  57     change wheelFamilyRim (fun W:N=>W.val)=_
  58     rw [←ordinary_family_image_rim,he]
  59   have hs:wheelFamilySpokes C=ordinaryPackingSpokes N:=by
  60     change wheelFamilySpokes (fun W:N=>W.val)=_
  61     rw [←ordinary_family_image_spokes,he]
  62   apply row_comparison_cover_image M X x 0 hM C
  63   · intro W a ha;exact hN.support W.val W.property a ha
  64   · intro W Z hne
  65     exact hN.disjoint W.val W.property Z.val Z.property (fun h=>hne (Subtype.ext h))
  66   · intro a;rw [hr];exact (hN.rims a).trans (hQr a)
  67   · intro u;rw [hs];exact (hN.spokes u).trans (hHr u)
  68   · intro a ha;rw [hr];exact hN.rimDegree a ha
  69   · intro a ha;rw [hs];exact hN.rowDegree a ha
  70   · intro u;rw [hs];exact hN.columnDegree u
