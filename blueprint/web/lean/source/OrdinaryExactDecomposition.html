  23 theorem ordinaryHostResources_disjoint (Q P : SimpleGraph V) (H K : V → U → Prop)
  24     (hQP : Disjoint Q P) (hHK : ∀ v u,¬(H v u ∧ K v u)) :
  25     Disjoint (ordinaryHostResources Q H) (ordinaryHostResources P K) := by
  26   apply Finset.disjoint_left.mpr
  27   intro e he hf
  28   cases e with
  29   | inl e =>
  30     rw [ordinaryHostResources_rim] at he hf
  31     induction e using Sym2.ind with
  32     | _ v w => exact (disjoint_iff_inf_le.mp hQP) ⟨he,hf⟩
  33   | inr vu =>
  34     rcases vu with ⟨v,u⟩
  35     rw [ordinaryHostResources_spoke] at he hf
  36     exact hHK v u ⟨he,hf⟩

  46 theorem finite_wheel_spoke_column {W : Type*} [Fintype W]
  47     (C : W → SimpleGraph V) (hub : W → U) (r : ℕ)
  48     (hC : ∀ w,IsEmbeddedCycle (C w) r) (ψ : W → ℝ) (u : U) :
  49     (∑ v,∑ w,if hub w=u ∧ v∈(C w).support then ψ w else 0)=
  50       (r:ℝ)*(∑ w,if hub w=u then ψ w else 0) := by
  51   rw [Finset.sum_comm,Finset.mul_sum]
  52   apply Finset.sum_congr rfl
  53   intro w _
  54   by_cases hu : hub w=u
  55   · simp only [hu,true_and,ite_true]
  56     rw [← Finset.sum_filter]
  57     have he : Finset.univ.filter (fun v => v∈(C w).support)=(C w).support.toFinset := by
  58       ext v; simp
  59     have hc : (C w).support.toFinset.card=r := (hC w).support_card
  60     rw [he,Finset.sum_const,nsmul_eq_mul,hc]
  61   · simp only [hu,false_and,ite_false,Finset.sum_const_zero,mul_zero]

  63 theorem ordinary_wheel_decomposition_exact_loads (Q : SimpleGraph V) (H : V → U → Prop) (r : ℕ)
  64     (h : HasOrdinaryWheelDecomposition Q H r) :

  66       (∀ v w,(∑ W : M,if (ordinaryWheelRim W.val).Adj v w then (1:ℝ) else 0)=if Q.Adj v w then 1 else 0) ∧
  67       (∀ v u,(∑ W : M,if ordinaryWheelHub W.val=u ∧ v∈(ordinaryWheelRim W.val).support then (1:ℝ) else 0)=if H v u then 1 else 0) := by
  68   obtain ⟨M,hM,hcover⟩ := h
  69   refine ⟨M,?_,?_⟩
  70   · intro v w
  71     have he : (∑ W : M,if (ordinaryWheelRim W.val).Adj v w then (1:ℝ) else 0)=
  72         ∑ W∈M,if (ordinaryWheelRim W).Adj v w then (1:ℝ) else 0 := by
  73       convert (Finset.sum_coe_sort M (fun W : UniversalOrdinaryWheel V U r => if (ordinaryWheelRim W).Adj v w then (1:ℝ) else 0)) using 1 <;> congr!

  75     have hh := resource_decomposition_indicator (ordinaryWheelResources (P:=(⊤ : SimpleGraph V)) (F:=⊥) (Hbase:=fun (_ : V) (_ : U) => True) (r:=r)) M (ordinaryHostResources Q H) hM hcover (Sum.inl s(v,w))
  76     simpa only [ordinaryWheelResources_rim,ordinaryHostResources_rim,SimpleGraph.mem_edgeSet] using hh
  77   · intro v u
  78     have he : (∑ W : M,if ordinaryWheelHub W.val=u ∧ v∈(ordinaryWheelRim W.val).support then (1:ℝ) else 0)=
  79         ∑ W∈M,if ordinaryWheelHub W=u ∧ v∈(ordinaryWheelRim W).support then (1:ℝ) else 0 := by

  85 theorem ordinary_wheel_decomposition_balance (Q : SimpleGraph V) (H : V → U → Prop) (r : ℕ)
  86     (h : HasOrdinaryWheelDecomposition Q H r) :
  87     (∀ v,Q.degree v=2*(Finset.univ.filter (fun u => H v u)).card) ∧
  88     (∀ u,r∣(Finset.univ.filter (fun v => H v u)).card) := by
  89   obtain ⟨M,hrim,hspoke⟩ := ordinary_wheel_decomposition_exact_loads Q H r h
  90   let C : M → SimpleGraph V := fun w => ordinaryWheelRim w.val
  91   let hub : M → U := fun w => ordinaryWheelHub w.val
  92   have hC (w : M) : IsEmbeddedCycle (C w) r := w.val.val.1.val.property
  93   constructor
  94   · intro v
  95     have hh := finite_wheel_load_balance C hub r hC (fun _ => (1:ℝ)) v
  96     change 2*(∑ u,∑ W : M,if ordinaryWheelHub W.val=u ∧ v∈(ordinaryWheelRim W.val).support then (1:ℝ) else 0)=
  97       ∑ w,∑ W : M,if (ordinaryWheelRim W.val).Adj v w then (1:ℝ) else 0 at hh
  98     simp_rw [hspoke,hrim] at hh
  99     have hs : (∑ u,if H v u then (1:ℝ) else 0)=((Finset.univ.filter (fun u => H v u)).card:ℝ) := by
 100       rw [← Finset.sum_filter];simp
 101     rw [hs,graph_adjacency_row_sum] at hh
 102     exact_mod_cast hh.symm

 104     have hh := finite_wheel_spoke_column C hub r hC (fun _ => (1:ℝ)) u
 105     change (∑ v,∑ W : M,if ordinaryWheelHub W.val=u ∧ v∈(ordinaryWheelRim W.val).support then (1:ℝ) else 0)=
 106       (r:ℝ)*(∑ W : M,if ordinaryWheelHub W.val=u then (1:ℝ) else 0) at hh
 107     simp_rw [hspoke] at hh
 108     have hs : (∑ v,if H v u then (1:ℝ) else 0)=((Finset.univ.filter (fun v => H v u)).card:ℝ) := by
 109       rw [← Finset.sum_filter];simp
 110     have ht : (∑ W : M,if ordinaryWheelHub W.val=u then (1:ℝ) else 0)=
 111       ((M.filter (fun W => ordinaryWheelHub W=u)).card:ℝ) := by
 112       have he : (∑ W : M,if ordinaryWheelHub W.val=u then (1:ℝ) else 0)=
 113           ∑ W∈M,if ordinaryWheelHub W=u then (1:ℝ) else 0 := by
