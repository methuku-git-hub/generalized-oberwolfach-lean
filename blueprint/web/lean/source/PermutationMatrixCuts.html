  14 theorem matrixCut_permute_columns {V : Type*} [DecidableEq V]
  15     (M : V→V→ℕ) (τ : Equiv.Perm V) (S T : Finset V) :
  16     matrixCut (fun x y=>M x (τ y)) S T=matrixCut M S (T.image τ) := by
  17   unfold matrixCut
  18   apply Finset.sum_congr rfl
  19   intro x _
  20   rw [Finset.sum_image]
  21   exact fun x _ y _ h=>τ.injective h

  34 theorem permutationSeparatedMatrix_cut_loss {V : Type*} [Fintype V]
  35     (π : Equiv.Perm V) (b : ℕ) (M : V→V→ℕ) (hM : ∀ x y,M x y≤1)
  36     (S T : Finset V) :
  37     (matrixCut M S T:ℝ)-(2*(b:ℝ)+1)*min (S.card:ℝ) T.card≤
  38       (matrixCut (permutationSeparatedMatrix π b M) S T:ℝ) := by
  39   let W := fun x y=>if PermutationClose π b x y then (1:ℕ) else 0
  40   have hrow (x : V) : ((∑ y,W x y:ℕ):ℝ)≤2*(b:ℝ)+1 := by
  41     have h := permutationClose_neighbour_card π b x
  42     simp only [Fintype.card_subtype,Finset.card_filter] at h
  43     exact_mod_cast h
  44   have hcol (y : V) : ((∑ x,W x y:ℕ):ℝ)≤2*(b:ℝ)+1 := by
  45     simpa only [W,permutationClose_symm π b] using hrow y
  46   exact matrixCut_loss_bound M (permutationSeparatedMatrix π b M) W
  47     (by intro x y;dsimp [W,permutationSeparatedMatrix];split_ifs <;> simp_all)
  48     hrow hcol S T
