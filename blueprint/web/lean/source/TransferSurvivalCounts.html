  18 def transferRetainedEquiv (σ : Sym2 V → Bool) :
  19     {T : OrdinaryTransferConfiguration hr P F H v u u' // ∀ e∈transferRequiredEdges T,σ e=true} ≃
  20     OrdinaryTransferConfiguration hr (retainedGraph P σ) F H v u u' where
  21   toFun T := {
  22     labels := T.val.labels
  23     omitted := T.val.omitted
  24     rim_edges := by
  25       intro j x y hxy
  26       refine ⟨T.val.rim_edges j hxy,?_⟩
  27       apply T.property s(x,y)
  28       exact Finset.mem_biUnion.mpr ⟨j,Finset.mem_univ _,by
  29         rw [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet]
  30         exact hxy⟩
  31     spokes := T.val.spokes
  32     independent := T.val.independent }
  33   invFun T := ⟨{
  34     labels := T.labels
  35     omitted := T.omitted
  36     rim_edges := fun j _ _ hxy=>(T.rim_edges j hxy).1
  37     spokes := T.spokes
  38     independent := T.independent },by
  39       intro e he
  40       obtain ⟨j,_,he⟩ := Finset.mem_biUnion.mp he
  41       induction e using Sym2.ind with
  42       | _ x y=>
  43         rw [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] at he
  44         exact (T.rim_edges j he).2⟩
  45   left_inv T := by
  46     apply Subtype.ext
  47     apply OrdinaryTransferConfiguration.labels_injective
  48     rfl
  49   right_inv T := by
  50     apply OrdinaryTransferConfiguration.labels_injective
  51     rfl

  53 theorem transferSurvivalSum_card (σ : Sym2 V → Bool) :
  54     transferSurvivalSum r hr P F H v u u' σ=
  55       (Fintype.card (OrdinaryTransferConfiguration hr (retainedGraph P σ) F H v u u'):ℝ)/
  56         (Fintype.card V:ℝ)^(r^2) := by
  57   have hcard := Fintype.card_congr (transferRetainedEquiv r hr P F H v u u' σ)
  58   rw [← hcard]
  59   unfold transferSurvivalSum
  60   simp_rw [patternValue_indicator]
  61   rw [← Finset.mul_sum,Fintype.card_subtype]
  62   simp only [Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  63   rw [div_eq_mul_inv,mul_comm]
  64   simp only [one_mul]
  65   congr 1
  66   apply Finset.sum_congr rfl
  67   intro T _
  68   split_ifs <;> rfl

  70 def transferRootSpokesEquiv (hroot : H v u ∧ H v u') :
  71     OrdinaryTransferCandidate hr P F H v u u' ≃ OrdinaryTransferConfiguration hr P F H v u u' where
  72   toFun T := {
  73     labels := T.labels
  74     omitted := T.omitted
  75     rim_edges := T.rim_edges
  76     spokes := by
  77       intro z hz
  78       have hh := T.spokes z hz
  79       by_cases he : z=v
  80       · simpa only [he] using hroot
  81       · exact ⟨hh.1.resolve_left he,hh.2.resolve_left he⟩
  82     independent := T.independent }
  83   invFun T := {
  84     labels := T.labels
  85     omitted := T.omitted
  86     rim_edges := T.rim_edges
  87     spokes := fun z hz=>⟨Or.inr (T.spokes z hz).1,Or.inr (T.spokes z hz).2⟩
  88     independent := T.independent }
  89   left_inv T := by apply OrdinaryTransferConfiguration.labels_injective;rfl
  90   right_inv T := by apply OrdinaryTransferConfiguration.labels_injective;rfl
