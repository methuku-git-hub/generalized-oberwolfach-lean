  16  theorem localOrdinaryRestriction_of_avoidance (Q D F : SimpleGraph V) (T Z : Finset V)
  17     (C : LocalOrdinaryCycle Q F T r) (hZ : Disjoint (localOrdinaryVertices C) Z)
  18     (hD : ∀ e∈canonicalGraphPairs D,¬(e.1∈localOrdinaryVertices C ∧ e.2∈localOrdinaryVertices C)) :
  19     localOrdinaryRestrictionGood (Q\D) (T\Z) C := by
  20   constructor
  21   · intro x y hxy
  22     refine ⟨C.val.property hxy,?_⟩
  23     intro hDxy
  24     let e : Sym2 V := s(x,y)
  25     have heD : e.out∈canonicalGraphPairs D :=
  26       Finset.mem_image.mpr ⟨e,SimpleGraph.mem_edgeFinset.mpr hDxy,rfl⟩
  27     have heC : C.val.val.val.Adj e.out.1 e.out.2 := by
  28       change s(e.out.1,e.out.2)∈C.val.val.val.edgeSet
  29       apply SimpleGraph.mem_edgeFinset.mp
  30       have he : s(e.out.1,e.out.2)=e := by rw [Sym2.mk];exact e.out_eq
  31       rw [he]
  32       exact SimpleGraph.mem_edgeFinset.mpr hxy
  33     exact hD e.out heD ⟨Set.mem_toFinset.mpr heC.mem_support_left,Set.mem_toFinset.mpr heC.mem_support_right⟩
  34   · intro x hx
  35     exact Finset.mem_sdiff.mpr ⟨C.property.1 x (Set.mem_toFinset.mp hx),
  36       fun hz=>Finset.disjoint_left.mp hZ hx hz⟩
