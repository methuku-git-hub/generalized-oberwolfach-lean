  12 theorem prescribed_one_ended_path_realization
  13     (as:List X) (ha:as.Nodup) (A:Finset X) (f:X→V)
  14     (hinj:Set.InjOn f A) (hsource:∀ a∈as,a∉A)
  15     (a:X) (haA:a∈A) (S L:Finset V) (β:ℝ) (D:V→V→Prop)
  16     (hext:IsExtendingList D L β) (F:SimpleGraph V) (M:ℝ) (hM:0≤M)
  17     (hdeg:∀ v,(F.degree v:ℝ)≤M) (forward:Bool)
  18     (hbudget:((A.image f∪S).card:ℝ)+as.length+1+M<β*Fintype.card V):
  19     ∃ g:X→V,Set.InjOn g (↑(A∪as.toFinset):Set X) ∧ (∀ u∈A,g u=f u) ∧
  20     (∀ u∈as,g u∈L ∧ g u∉S) ∧ DirectedPathFrom D F forward (g a) (as.map g) := by
  21   let B:=A.image f∪S
  22   obtain ⟨ys,hyslen,hysnodup,hys,hpath⟩:=extending_list_embed_one_sided_path D L β hext F M hM hdeg forward as.length B (f a) hbudget
  23   have hlen:as.length=ys.length:=hyslen.symm
  24   have htarget:∀ z∈ys,z∉A.image f:=by
  25     intro z hz hmem
  26     exact (hys z hz).2.1 (Finset.mem_union_left S hmem)
  27   let g:=updatePairedImages f (as.zip ys)
  28   have hb:=prescribed_embedding_zip_update as ys hlen ha hysnodup A f hinj hsource htarget
  29   have hmap:=prescribed_embedding_update_map_list as ys hlen ha hysnodup A f hinj hsource htarget
  30   refine ⟨g,hb.1,hb.2.1,?_,?_⟩
  31   · intro u hu
  32     obtain ⟨i,hi⟩:=List.mem_iff_get.mp hu
  33     have hp:=prescribed_embedding_zip_update_at_position as ys hlen ha hysnodup A f hinj hsource htarget i
  34     have hz:ys.get ⟨i.val,by omega⟩∈ys:=List.get_mem _ _
  35     have hgood:=hys _ hz
  36     have hnotS:(ys.get ⟨i.val,by omega⟩)∉S:=fun h=>hgood.2.1 (Finset.mem_union_right _ h)
  37     rw [←hi]
  38     change updatePairedImages f (as.zip ys) (as.get i)∈L ∧
  39       updatePairedImages f (as.zip ys) (as.get i)∉S
  40     rw [hp]
  41     exact ⟨hgood.1,hnotS⟩
  42   · have hga:g a=f a:=hb.2.1 a haA
  43     rw [hga,hmap]
  44     exact hpath
  45 end Oberwolfach
