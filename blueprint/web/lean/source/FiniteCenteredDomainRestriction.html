  11 theorem finite_bounded_sum_restriction (A B:Finset X) (hAB:A⊆B) (f:X→ℝ)
  12     (hf:∀ x∈B,|f x|≤1):
  13     |(∑ x∈A,f x)-(∑ x∈B,f x)|≤((B\A).card:ℝ):=by
  14   have he: (∑ x∈B\A,f x)+(∑ x∈A,f x)=∑ x∈B,f x:=Finset.sum_sdiff hAB
  15   have ha:|(∑ x∈A,f x)-(∑ x∈B,f x)|=|∑ x∈B\A,f x|:=by
  16     rw [←he]
  17     ring_nf
  18     exact abs_neg _
  19   rw [ha]

  21     _ ≤∑ x∈B\A,|f x|:=Finset.abs_sum_le_sum_abs _ _
  22     _ ≤∑ x∈B\A,(1:ℝ):=Finset.sum_le_sum (fun x hx=>hf x (Finset.mem_sdiff.mp hx).1)

  25 theorem finite_centered_count_restriction (A B:Finset X) (hAB:A⊆B)
  26     (eligible present:X→Prop) [DecidablePred eligible] [DecidablePred present]
  27     (target:X→ℝ) (ht:∀ x∈B,0≤target x∧target x≤1) (ε:ℝ)
  28     (hcount:|(∑ x∈B.filter eligible,if present x then (1:ℝ) else 0)-
  29       ∑ x∈B.filter eligible,target x|<ε):
  30     |(∑ x∈A.filter eligible,if present x then (1:ℝ) else 0)-
  31       ∑ x∈A.filter eligible,target x|<ε+(B\A).card:=by
  32   let f:=fun x=>if eligible x then ((if present x then (1:ℝ) else 0)-target x) else 0
  33   have hf:∀ x∈B,|f x|≤1:=by
  34     intro x hx
  35     obtain ⟨h0,h1⟩:=ht x hx
  36     dsimp [f]
  37     split_ifs <;> rw [abs_le] <;> constructor <;> linarith
  38   have he (C:Finset X):(∑ x∈C,f x)=
  39       (∑ x∈C.filter eligible,if present x then (1:ℝ) else 0)-∑ x∈C.filter eligible,target x:=by
  40     dsimp [f]
  41     rw [←Finset.sum_filter,Finset.sum_sub_distrib]
  42   have hd:=finite_bounded_sum_restriction A B hAB f hf
  43   rw [←he B] at hcount
  44   rw [←he A]
  45   have hh:|(∑ x∈A,f x)|≤|(∑ x∈A,f x)-(∑ x∈B,f x)|+|(∑ x∈B,f x)|:=by
  46     simpa only [sub_add_cancel] using abs_add_le ((∑ x∈A,f x)-(∑ x∈B,f x)) (∑ x∈B,f x)
  47   exact hh.trans_lt ((add_lt_add_of_le_of_lt hd hcount).trans_eq (add_comm _ _))
