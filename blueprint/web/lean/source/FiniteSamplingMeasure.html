  24 theorem finiteLaw_integral
  25     {Ω : Type*} [Fintype Ω] [MeasurableSpace Ω] [MeasurableSingletonClass Ω]
  26     (μ : FiniteLaw Ω) (X : Ω → ℝ) :
  27     ∫ ω, X ω ∂μ.toPMF.toMeasure = μ.expectation X := by
  28   rw [PMF.integral_eq_sum]
  29   simp only [FiniteLaw.toPMF, PMF.ofFintype_apply,
  30     ENNReal.toReal_ofReal (μ.nonneg _), smul_eq_mul, FiniteLaw.expectation]

  32 theorem finiteLaw_probability_measure
  33     {Ω : Type*} [Fintype Ω] [MeasurableSpace Ω] [MeasurableSingletonClass Ω]
  34     (μ : FiniteLaw Ω) (A : Ω → Prop) :
  35     μ.toPMF.toMeasure.real {ω | A ω} = μ.probability A := by
  36   classical
  37   have hnonneg (ω : Ω) : 0 ≤ (if A ω then μ.mass ω else 0) := by
  38     split_ifs <;> (first | exact μ.nonneg ω | exact le_refl _)
  39   have heq : μ.toPMF.toMeasure {ω | A ω} = ENNReal.ofReal (μ.probability A) := by
  40     rw [PMF.toMeasure_apply_fintype, FiniteLaw.probability,
  41       ENNReal.ofReal_sum_of_nonneg (fun ω _ => hnonneg ω)]
  42     apply Finset.sum_congr rfl
  43     intro ω _
  44     by_cases ha : A ω <;> simp [Set.indicator, FiniteLaw.toPMF, ha]
  45   rw [measureReal_def, heq, ENNReal.toReal_ofReal]
  46   exact Finset.sum_nonneg (fun ω _ => hnonneg ω)

  48 theorem finiteLaw_pi_measure
  49     {I : Type*} [Fintype I] [DecidableEq I]
  50     {State : I → Type*} [∀ i, Fintype (State i)]
  51     [∀ i, MeasurableSpace (State i)] [∀ i, MeasurableSingletonClass (State i)]
  52     (μ : ∀ i, FiniteLaw (State i)) :
  53     (FiniteLaw.pi μ).toPMF.toMeasure = Measure.pi (fun i => (μ i).toPMF.toMeasure) := by
  54   apply Measure.ext_of_singleton
  55   intro σ
  56   rw [PMF.toMeasure_apply_singleton _ _ (measurableSet_singleton _), Measure.pi_singleton]
  57   simp_rw [PMF.toMeasure_apply_singleton _ _ (measurableSet_singleton _)]
  58   change ENNReal.ofReal (∏ i, (μ i).mass (σ i)) =
  59     ∏ i, ENNReal.ofReal ((μ i).mass (σ i))
  60   exact ENNReal.ofReal_prod_of_nonneg (fun i _ => (μ i).nonneg (σ i))

  62 theorem finiteLaw_coordinate_expectation
  63     {I : Type*} [Fintype I] [DecidableEq I]
  64     {State : I → Type*} [∀ i, Fintype (State i)]
  65     [∀ i, MeasurableSpace (State i)] [∀ i, MeasurableSingletonClass (State i)]
  66     (μ : ∀ i, FiniteLaw (State i)) (i : I) (X : State i → ℝ) :
  67     (FiniteLaw.pi μ).expectation (fun σ => X (σ i)) = (μ i).expectation X := by
  68   rw [← finiteLaw_integral, finiteLaw_pi_measure, ← finiteLaw_integral]
  69   have hm := measurePreserving_eval (fun j => (μ j).toPMF.toMeasure) i
  70   rw [← hm.map_eq]
  71   exact (integral_map_of_stronglyMeasurable hm.measurable
  72     (measurable_of_finite X).stronglyMeasurable).symm

  74 /-- The coordinate variables of the explicitly constructed categorical
  75 law are independent. Arbitrary functions of one coordinate retain this

  77 theorem finiteLaw_coordinate_independence
  78     {I : Type*} [Fintype I] [DecidableEq I]
  79     {State : I → Type*} [∀ i, Fintype (State i)]
  80     [∀ i, MeasurableSpace (State i)] [∀ i, MeasurableSingletonClass (State i)]
  81     (μ : ∀ i, FiniteLaw (State i)) (X : ∀ i, State i → ℝ) :
  82     iIndepFun (fun i σ => X i (σ i)) (FiniteLaw.pi μ).toPMF.toMeasure := by
  83   rw [finiteLaw_pi_measure]
  84   exact iIndepFun_pi (fun i => (measurable_of_finite (X i)).aemeasurable)

  86 /-- Hoeffding's inequality for the exact finite product law. Each summand
  87 depends on one original coordinate; the displayed centre is its finite
  88 expectation under that same law. -/
  89 theorem finiteLaw_bounded_coordinate_sum
  90     {I : Type*} [Fintype I] [DecidableEq I]
  91     {State : I → Type*} [∀ i, Fintype (State i)]
  92     [∀ i, MeasurableSpace (State i)] [∀ i, MeasurableSingletonClass (State i)]
  93     (μ : ∀ i, FiniteLaw (State i)) (X : ∀ i, State i → ℝ)
  94     (a b : I → ℝ) (s : Finset I)
  95     (hbounded : ∀ i ∈ s, ∀ x, X i x ∈ Set.Icc (a i) (b i))
  96     {t : ℝ} (ht : 0 ≤ t) :
  97     (FiniteLaw.pi μ).probability
  98       (fun σ => t ≤ |(∑ i ∈ s, X i (σ i)) -
  99         ∑ i ∈ s, (μ i).expectation (X i)|) ≤
 100       2 * Real.exp (-2 * t ^ 2 / (∑ i ∈ s, (b i - a i) ^ 2)) := by
 101   have hind := finiteLaw_coordinate_independence μ X
 102   have htail := boundedIndependentSum_tail (μ := (FiniteLaw.pi μ).toPMF.toMeasure)
 103     (fun i (σ : ∀ j, State j) => X i (σ i)) a b s hind
 104     (fun i _ => (measurable_of_finite (fun (σ : ∀ j, State j) => X i (σ i))).aemeasurable)
 105     (fun i hi => Filter.Eventually.of_forall (fun σ => hbounded i hi (σ i))) ht
 106   simp only [finiteLaw_integral, finiteLaw_probability_measure,
 107     finiteLaw_coordinate_expectation] at htail
 108   exact htail

 110 /-- The raw directed-colour cut estimate is a concentration statement over
 111 physical edges. `forward e` and `reverse e` may both hold; the probability
 112 law still has exactly one colour and one direction for edge `e`. -/
 113 theorem categoricalDirection_cut_tail
 114     {E : Type*} [Fintype E] [DecidableEq E] {k : ℕ}
 115     (ρ : Fin k → ℝ) (hρ : ∀ a, 0 ≤ ρ a) (htotal : ∑ a, ρ a = 1)
 116     (a : Fin k) (forward reverse : E → Prop)
 117     [DecidablePred forward] [DecidablePred reverse]
 118     {t : ℝ} (ht : 0 ≤ t) :
 119     (FiniteLaw.pi (fun _ : E => FiniteLaw.categoricalDirection ρ hρ htotal)).probability
 120       (fun σ => t ≤ |(∑ e, directedCutContribution a (forward e) (reverse e) (σ e)) -
 121         ρ a / 2 * (∑ e : E, ((if forward e then (1 : ℝ) else 0) +
 122           (if reverse e then (1 : ℝ) else 0)))|) ≤
 123       2 * Real.exp (-2 * t ^ 2 / Fintype.card E) := by
 124   classical
 125   have hh := finiteLaw_bounded_coordinate_sum
 126     (fun _ : E => FiniteLaw.categoricalDirection ρ hρ htotal)
 127     (fun e => directedCutContribution a (forward e) (reverse e))
 128     (fun _ => (0 : ℝ)) (fun _ => (1 : ℝ)) Finset.univ
 129     (fun e _ s => directedCutContribution_bounds a (forward e) (reverse e) s) ht
 130   simpa only [categoricalDirection_expected_cut, ← Finset.mul_sum, sub_zero, one_pow,
 131     Finset.sum_const, Finset.card_univ, nsmul_eq_mul, mul_one] using hh
