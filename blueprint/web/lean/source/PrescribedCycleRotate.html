  11 theorem prescribed_cycle_rotate (a b:X) (xs:List X) (hnodup:(a::xs).Nodup) (hb:b∈xs)
  12     (R:X→X→Prop):
  13     ∃ ys:List X,(b::ys).Nodup ∧ (a::xs).length=(b::ys).length ∧ (a::xs).toFinset=(b::ys).toFinset ∧
  14       (List.IsChain R ((a::xs)++[a]) ↔ List.IsChain R ((b::ys)++[b])) ∧
  15       prescribedOrderedPathEdges ((a::xs)++[a])=prescribedOrderedPathEdges ((b::ys)++[b]) ∧
  16       prescribedDirectedPathEdges ((a::xs)++[a])=prescribedDirectedPathEdges ((b::ys)++[b]) := by
  17   obtain ⟨pre,post,hxs,hleft,hright,hinter,hphys,hdir⟩:=prescribed_cycle_two_boundary_split a b xs
  18     (List.nodup_cons.mp hnodup).1 (List.nodup_cons.mp hnodup).2 hb
  19   let ys:=post++a::pre
  20   have hperm:(a::xs).Perm (b::ys):=by
  21     simpa [ys,hxs,List.cons_append,List.append_assoc] using
  22       (List.perm_append_comm (l₁:=a::pre) (l₂:=b::post))
  23   have hnodup':(b::ys).Nodup:=(hperm.nodup_iff).mp hnodup
  24   have hsets:(a::xs).toFinset=(b::ys).toFinset:=by
  25     ext x
  26     simp only [List.mem_toFinset]
  27     exact hperm.mem_iff
  28   have hp1:List.IsChain R ((a::xs)++[a]) ↔
  29       List.IsChain R (a::pre++[b]) ∧ List.IsChain R (b::post++[a]):=by
  30     simpa [hxs,List.cons_append,List.append_assoc] using
  31       (List.isChain_split (R:=R) (c:=b) (l₁:=a::pre) (l₂:=post++[a]))
  32   have hp2:List.IsChain R ((b::ys)++[b]) ↔
  33       List.IsChain R (b::post++[a]) ∧ List.IsChain R (a::pre++[b]):=by
  34     simpa [ys,List.cons_append,List.append_assoc] using
  35       (List.isChain_split (R:=R) (c:=a) (l₁:=b::post) (l₂:=pre++[b]))
  36   have hphys':prescribedOrderedPathEdges ((b::ys)++[b])=
  37       prescribedOrderedPathEdges (b::post++[a]) ∪ prescribedOrderedPathEdges (a::pre++[b]):=by
  38     simpa [ys,List.cons_append,List.append_assoc] using prescribed_path_edges_pivot a (pre++[b]) (b::post)
  39   have hdir':prescribedDirectedPathEdges ((b::ys)++[b])=
  40       prescribedDirectedPathEdges (b::post++[a]) ∪ prescribedDirectedPathEdges (a::pre++[b]):=by
  41     simpa [ys,List.cons_append,List.append_assoc] using prescribed_directed_path_edges_pivot a (pre++[b]) (b::post)
  42   refine ⟨ys,hnodup',hperm.length_eq,hsets,?_,?_,?_⟩
  43   · exact hp1.trans (and_comm.trans hp2.symm)
  44   · rw [hphys,hphys',Finset.union_comm]
  45   · rw [hdir,hdir',Finset.union_comm]
  46 end Oberwolfach
