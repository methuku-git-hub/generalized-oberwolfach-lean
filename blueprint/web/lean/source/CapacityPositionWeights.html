  12 def capacityPositionWeights {E : Type*} (k : ℕ) (w : E → ℝ) (u : Fin k → E → ℝ) :
  13     Fin (k+1) → E → ℝ :=
  14   Fin.snoc (fun j => (1/(k+1:ℝ)) • w+u j)
  15     (w-∑ j,((1/(k+1:ℝ)) • w+u j))

  50     (w : E → ℝ) (u : Fin k → E → ℝ) (κ U : ℝ)
  51     (hU : 0≤U) (hw : ∀ e,|w e-1|≤κ) (hu : ∀ j e,|u j e|≤U) :
  52     ∀ j e,|capacityPositionWeights k w u j e-1/(k+1:ℝ)| ≤ κ/(k+1:ℝ)+k*U := by
  53   have hkp : (0:ℝ)<k+1 := by positivity
  54   have hkr : (1:ℝ)≤k := by exact_mod_cast hk

  56   refine Fin.lastCases ?_ (fun j => ?_) j
  57   · rw [capacityPositionWeights_last_formula]

  60       rw [abs_div,abs_of_pos hkp]
  61       exact div_le_div_of_nonneg_right (hw e) hkp.le
