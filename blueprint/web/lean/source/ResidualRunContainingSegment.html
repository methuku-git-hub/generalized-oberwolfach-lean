   8 theorem selected_segment_in_maximal_run (d:ℕ) (T:Finset ℕ)
   9     (hT:T⊆Finset.range d) (j:ℕ) (hj:j∈T):
  10     ∃ a b,(a,b)∈selectedSegmentRuns d T ∧ a≤j ∧ j≤b := by
  11   let good := (Finset.range (j+1)).filter (fun a=>∀ k,a≤k→k≤j→k∈T)
  12   have hgood:good.Nonempty := by
  13     refine ⟨j,Finset.mem_filter.mpr ⟨Finset.mem_range.mpr (by omega),?_⟩⟩
  14     intro k h1 h2
  15     have : k=j := by omega
  16     simpa [this] using hj
  17   let a:=good.min' hgood
  18   have ha:=Finset.min'_mem good hgood
  19   have had:a≤j := by
  20     have hh: a<j+1 := Finset.mem_range.mp (Finset.mem_filter.mp ha).1
  21     omega
  22   have hagood:∀k,a≤k→k≤j→k∈T := (Finset.mem_filter.mp ha).2
  23   have haT:a∈T := hagood a le_rfl had
  24   have hstart:a∈selectedRunStarts T := by
  25     apply (selected_run_start_mem T a).mpr
  26     refine ⟨haT,?_⟩
  27     by_cases ha0:a=0
  28     · exact Or.inl ha0
  29     · right
  30       intro hapred
  31       have hpGood:a-1∈good := by
  32         apply Finset.mem_filter.mpr
  33         refine ⟨Finset.mem_range.mpr (by omega),?_⟩
  34         intro k hk hjk
  35         by_cases heq:k=a-1
  36         · simpa [heq] using hapred
  37         · exact hagood k (by omega) hjk
  38       have hm:=Finset.min'_le good (a-1) hpGood
  39       omega
  40   obtain ⟨b,hb,_⟩:=selected_run_exists_unique_from_start d T hT a hstart
  41   have hjb:j≤b := by
  42     by_contra h
  43     have hnext: b+1≤j := by omega
  44     have hnextT:=hagood (b+1) (by have hh:=hb.1;omega) hnext
  45     exact hb.2.2.2 hnextT
  46   exact ⟨a,b,(selected_run_mem d T hT a b).mpr hb,had,hjb⟩
