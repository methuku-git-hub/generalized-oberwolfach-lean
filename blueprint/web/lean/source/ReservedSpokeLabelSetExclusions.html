  12     (hOld:∀i,Disjoint ((Old i).image (ψ i)) (Outer i))
  13     (hForbidden:∀i,Forbidden i⊆Outer i)
  14     (hRequired:∀i,Required i⊆Inner\Reserve i)
  15     (hReserve:∀i,Forbidden i∩Inner⊆Reserve i):
  16     (∀i x,x∈Old i→ψ i x∉Forbidden i) ∧
  17     (∀i,Disjoint (Required i) (Forbidden i)) :=by
