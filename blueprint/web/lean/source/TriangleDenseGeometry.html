  12 theorem triangle_four_graph_neighbours {V:Type} [Fintype V] [DecidableEq V]
  13     (G:SimpleGraph V) (β:ℝ)
  14     (hdeg:∀ v,(3/4+β)*(Fintype.card V:ℝ)≤(G.degree v:ℝ)) (a b c d:V):
  15     4*β*(Fintype.card V:ℝ)≤
  16       (((Finset.univ:Finset V).filter (fun x=>G.Adj a x∧G.Adj b x∧G.Adj c x∧G.Adj d x)).card:ℝ):=by
  17   let N:Fin 4→Finset V:=fun i=>G.neighborFinset (![a,b,c,d] i)
  18   have hd:∀ i∈(Finset.univ:Finset (Fin 4)),(3/4+β)*((Finset.univ:Finset V).card:ℝ)≤
  19       (((Finset.univ:Finset V)∩N i).card:ℝ):=by
  20     intro i _
  21     simpa [N,SimpleGraph.card_neighborFinset_eq_degree] using hdeg (![a,b,c,d] i)
  22   have h:=common_neighbour_card_lower (Finset.univ:Finset V) (Finset.univ:Finset (Fin 4)) N (3/4+β) hd
  23   have he:common (Finset.univ:Finset V) (Finset.univ:Finset (Fin 4)) N=
  24       (Finset.univ:Finset V).filter (fun x=>G.Adj a x∧G.Adj b x∧G.Adj c x∧G.Adj d x):=by
  25     ext x
  26     simp only [common,Finset.mem_filter,Finset.mem_univ,true_and,N,SimpleGraph.mem_neighborFinset]
  27     constructor
  28     · intro h
  29       exact ⟨h 0 (by simp),h 1 (by simp),h 2 (by simp),h 3 (by simp)⟩
  30     · rintro ⟨ha,hb,hc,hd⟩ i _
  31       fin_cases i <;> simp_all
  32   rw [he] at h
  33   simp only [Finset.card_univ,Fintype.card_fin,Nat.cast_ofNat] at h
  34   nlinarith

  37     (G:SimpleGraph V) (H:V→U→Prop) (β ε:ℝ) (hε:ε≤β)
  38     (hdeg:∀ v,(3/4+β)*(Fintype.card V:ℝ)≤(G.degree v:ℝ))
  39     (hnear:∀ u,(((Finset.univ:Finset V).filter (fun v=>¬H v u)).card:ℝ)≤ε*(Fintype.card V:ℝ))

  42       (((Finset.univ:Finset V).filter (fun x=>H x u∧H x u'∧G.Adj a x∧G.Adj b x∧G.Adj c x∧G.Adj d x)).card:ℝ):=by
  43   let C:Finset V:=(Finset.univ:Finset V).filter (fun x=>G.Adj a x∧G.Adj b x∧G.Adj c x∧G.Adj d x)
  44   let W:Finset V:=((Finset.univ:Finset V).filter (fun x=>¬H x u))∪((Finset.univ:Finset V).filter (fun x=>¬H x u'))
  45   have he:C\W=(Finset.univ:Finset V).filter (fun x=>H x u∧H x u'∧G.Adj a x∧G.Adj b x∧G.Adj c x∧G.Adj d x):=by
  46     ext x
  47     simp only [C,W,Finset.mem_sdiff,Finset.mem_filter,Finset.mem_univ,true_and,Finset.mem_union]
  48     tauto
  49   have hw:(W.card:ℝ)≤2*ε*(Fintype.card V:ℝ):=by
  50     have hc:(W.card:ℝ)≤(((Finset.univ:Finset V).filter (fun x=>¬H x u)).card:ℝ)+
  51       (((Finset.univ:Finset V).filter (fun x=>¬H x u')).card:ℝ):=by exact_mod_cast Finset.card_union_le _ _
  52     have h1:=hnear u
  53     have h2:=hnear u'
  54     linarith
  55   have hC:4*β*(Fintype.card V:ℝ)≤(C.card:ℝ):=triangle_four_graph_neighbours G β hdeg a b c d
  56   have hsplit:(C.card:ℝ)≤((C\W).card:ℝ)+(W.card:ℝ):=by exact_mod_cast Finset.card_le_card_sdiff_add_card (s:=C) (t:=W)
  57   have hn:0≤(Fintype.card V:ℝ):=by positivity
  58   rw [←he]

  60 theorem triangle_common_hub_neighbours {V U:Type} [Fintype U] [DecidableEq U] [DecidableEq V]
  61     (H:V→U→Prop) (ε:ℝ) (hε:0≤ε)
  62     (hnear:∀ v,(((Finset.univ:Finset U).filter (fun u=>¬H v u)).card:ℝ)≤ε*(Fintype.card U:ℝ))
  63     (T:Finset V) (hT:T.card≤9):
  64     (1-9*ε)*(Fintype.card U:ℝ)≤
  65       ((common (Finset.univ:Finset U) T (fun v=>(Finset.univ:Finset U).filter (fun u=>H v u))).card:ℝ):=by
  66   let N:V→Finset U:=fun v=>(Finset.univ:Finset U).filter (fun u=>H v u)
  67   have hd:∀ v∈T,(1-ε)*((Finset.univ:Finset U).card:ℝ)≤(((Finset.univ:Finset U)∩N v).card:ℝ):=by
  68     intro v _
  69     have hc:((N v).card:ℝ)+(((Finset.univ:Finset U).filter (fun u=>¬H v u)).card:ℝ)=Fintype.card U:=by
  70       exact_mod_cast Finset.card_filter_add_card_filter_not (s:=(Finset.univ:Finset U)) (fun u=>H v u)
  71     have hh:=hnear v
  72     simpa only [Finset.univ_inter,Finset.card_univ] using (show (1-ε)*(Fintype.card U:ℝ)≤(N v).card by linarith)
  73   have hh:=common_neighbour_card_lower (Finset.univ:Finset U) T N (1-ε) hd
  74   simp only [Finset.card_univ] at hh
  75   have ht:(T.card:ℝ)≤9:=by exact_mod_cast hT
  76   have hn:0≤(Fintype.card U:ℝ):=by positivity
  77   have hprod:(T.card:ℝ)*ε≤9*ε:=mul_le_mul_of_nonneg_right ht hε
  78   have hprod':(T.card:ℝ)*ε*(Fintype.card U:ℝ)≤9*ε*(Fintype.card U:ℝ):=
  79     mul_le_mul_of_nonneg_right hprod hn
  80   dsimp [N] at hh
  81   nlinarith
