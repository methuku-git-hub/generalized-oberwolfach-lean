  16     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ)
  17     (tail head:E → V) (ktail khead:K → V)
  18     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  19     (hE:∀ e,¬PermutationClose π (3*d) (tail e) (head e)∧P.Adj (tail e) (head e))
  20     (hK:∀ e,¬PermutationClose π (3*d) (ktail e) (khead e)∧P.Adj (ktail e) (π (khead e)))
  21     (σ:RawMarkedOutcome V U Γ Color) (I:Finset V) (hI:∀ v,π v∈I↔v∈I)
  22     (hAgree:∀ mark x y,x∈I → y∈I →
  23       (markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) mark x y=1 ↔
  24         (if mark then ∃ e:K,ktail e=x∧khead e=y else ∃ e:E,tail e=x∧head e=y))) :
  25     let τ:=rawMarkedEncodeHost tail head ktail khead π σ I;
  26     markedPhysicalRim P π (3*d) (fun e=>τ (Sum.inl e)) false=occurrenceMultiplicity tail head ∧
  27     markedPhysicalRim P π (3*d) (fun e=>τ (Sum.inl e)) true=occurrenceMultiplicity ktail khead := by
  28   intro τ
  29   have heinj:Function.Injective (fun e:E=>(tail e,head e)):=by
  30     intro a b hab
  31     have hh:(Sum.inl a:E⊕K)=Sum.inl b:=hp (by
  32       have ht:tail a=tail b:=congrArg Prod.fst hab
  33       have hh:head a=head b:=congrArg Prod.snd hab
  34       simpa only [markedPhysicalEdge,Sum.elim_inl,ht,hh])
  35     exact Sum.inl.inj hh
  36   have hkinj:Function.Injective (fun e:K=>(ktail e,khead e)):=by

  39       have ht:ktail a=ktail b:=congrArg Prod.fst hab
  40       have hh:khead a=khead b:=congrArg Prod.snd hab
  41       simpa only [markedPhysicalEdge,Sum.elim_inr,ht,hh])
  42     exact Sum.inr.inj hh
