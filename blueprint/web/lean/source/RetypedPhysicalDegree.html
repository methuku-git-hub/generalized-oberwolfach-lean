  14 def retypedPhysicalEdges (tail head:E→V) (M M₀:E→Bool) : Finset (Finset V) :=
  15   (Finset.univ.filter (fun e=>M e≠M₀ e)).image (fun e=>({tail e,head e}:Finset V))

  17 theorem retyped_physical_degree_le (tail head:E→V) (M M₀:E→Bool) (v:V) :
  18     (physicalEdgeGraph (retypedPhysicalEdges tail head M M₀)).degree v ≤
  19       (∑ e, if head e=v ∧ M e≠M₀ e then (1:ℕ) else 0) +
  20       (∑ e, if tail e=v ∧ M e≠M₀ e then (1:ℕ) else 0) := by
  21   let D := Finset.univ.filter (fun e=>M e≠M₀ e)
  22   let H := D.filter (fun e=>head e=v)
  23   let T := D.filter (fun e=>tail e=v)
  24   let pair := fun e:E=>({tail e,head e}:Finset V)
  25   have hsub : ((retypedPhysicalEdges tail head M M₀).filter (fun s=>v∈s)) ⊆
  26       H.image pair ∪ T.image pair := by
  27     intro s hs
  28     obtain ⟨hs,hv⟩ := Finset.mem_filter.mp hs
  29     obtain ⟨e,he,rfl⟩ := Finset.mem_image.mp hs
  30     simp only [Finset.mem_insert,Finset.mem_singleton] at hv
  31     rcases hv with htail|hhead
  32     · apply Finset.mem_union_right
  33       exact Finset.mem_image.mpr ⟨e,Finset.mem_filter.mpr ⟨he,htail.symm⟩,rfl⟩
  34     · apply Finset.mem_union_left
  35       exact Finset.mem_image.mpr ⟨e,Finset.mem_filter.mpr ⟨he,hhead.symm⟩,rfl⟩
  36   have hbound := (physicalEdgeGraph_degree_le_incident (retypedPhysicalEdges tail head M M₀) v).trans
  37     ((Finset.card_le_card hsub).trans (Finset.card_union_le _ _))
  38   have h1 : (H.image pair).card ≤ H.card := Finset.card_image_le
  39   have h2 : (T.image pair).card ≤ T.card := Finset.card_image_le
  40   have hH : H.card = ∑ e, if head e=v ∧ M e≠M₀ e then (1:ℕ) else 0 := by
  41     have heq : H = Finset.univ.filter (fun e=>head e=v ∧ M e≠M₀ e) := by
  42       ext e
  43       simp only [H,D,Finset.mem_filter,Finset.mem_univ,true_and]
  44       tauto
  45     rw [heq,Finset.card_filter]
  46   have hT : T.card = ∑ e, if tail e=v ∧ M e≠M₀ e then (1:ℕ) else 0 := by
  47     have heq : T = Finset.univ.filter (fun e=>tail e=v ∧ M e≠M₀ e) := by
  48       ext e
  49       simp only [T,D,Finset.mem_filter,Finset.mem_univ,true_and]
  50       tauto
  51     rw [heq,Finset.card_filter]
  52   omega
  53 theorem marked_indegree_correction_physical
  54     [Nonempty V] (tail head:E→V) (M₀:E→Bool) (A:E→Prop) (b:ℕ) (rho eta:ℝ)
  55     (hrho:0<rho) (heta:0<eta) (hetaR:eta≤rho)
  56     (hpair:Function.Injective (fun e:E=>(tail e,head e)))
  57     (hdev:∀ v, |(markedHeadCount head M₀ v:ℝ)-(b:ℝ)|≤eta*Fintype.card V)
  58     (hmarked:∀ v, rho*Fintype.card V≤(markedFlexibleIncoming head M₀ A v true:ℝ))
  59     (hunmarked:∀ v, rho*Fintype.card V≤(markedFlexibleIncoming head M₀ A v false:ℝ)) :
  60     ∃ M:E→Bool,
  61       (∀ v, markedHeadCount head M v=b) ∧
  62       (∀ e, M e≠M₀ e → A e) ∧
  63       (∀ v, ((physicalEdgeGraph (retypedPhysicalEdges tail head M M₀)).degree v:ℝ) ≤
  64         (1+1/rho)*eta*Fintype.card V+1) := by
  65   obtain ⟨M,hM,hA,hhead,htail⟩ := marked_flexible_retyping tail head M₀ A b rho eta hrho heta hetaR
  66     hpair hdev hmarked hunmarked
  67   refine ⟨M,hM,hA,?_⟩
  68   let n : ℝ := Fintype.card V
  69   have hn : 0<n := by dsimp [n]; exact_mod_cast Fintype.card_pos
  70   have hs : n*(eta*n/(rho*n))=eta*n/rho := by
  71     field_simp
  72   have ha : 0≤eta*n/rho := div_nonneg (mul_nonneg heta.le hn.le) hrho.le
  73   have hc : ((⌈eta*n/rho⌉₊:ℕ):ℝ) ≤ eta*n/rho+1 := (Nat.ceil_lt_add_one ha).le
  74   intro v
  75   have hd := retyped_physical_degree_le tail head M M₀ v
  76   have hd' : ((physicalEdgeGraph (retypedPhysicalEdges tail head M M₀)).degree v:ℝ) ≤
  77       ((∑ e, if head e=v ∧ M e≠M₀ e then (1:ℕ) else 0):ℝ) +
  78       ((∑ e, if tail e=v ∧ M e≠M₀ e then (1:ℕ) else 0):ℝ) := by exact_mod_cast hd
  79   have hh := hhead v
  80   change _ ≤ eta*n at hh
  81   have htail' : ((∑ e, if tail e=v ∧ M e≠M₀ e then (1:ℕ) else 0):ℝ) ≤
  82       ((⌈eta*n/rho⌉₊:ℕ):ℝ) := by
  83     have ht := htail v
  84     have hce : ⌈n*(eta*n/(rho*n))⌉₊ = ⌈eta*n/rho⌉₊ := congrArg Nat.ceil hs
  85     rw [hce] at ht
  86     exact_mod_cast ht
  87   calc
  88     _ ≤ eta*n + (⌈eta*n/rho⌉₊:ℕ) := hd'.trans (add_le_add hh htail')
  89     _ ≤ eta*n + (eta*n/rho+1) := by linarith
  90     _ = (1+1/rho)*eta*Fintype.card V+1 := by dsimp [n]; ring
