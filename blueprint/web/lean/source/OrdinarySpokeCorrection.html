  15 theorem ordinary_spoke_correction {V U : Type*} [Fintype V] [Fintype U]
  16     [DecidableEq V] [DecidableEq U] {r : ℕ} (hr : 3≤r)
  17     (Q F : SimpleGraph V) (H : V → U → Prop)
  18     (γ η : ℝ) (hγ : 0<γ) (hη : 0≤η) (hn : 0<Fintype.card V)
  19     (ψ : OrdinaryWheel Q F H r → ℝ)
  20     (hdeg : ∀ v,Q.degree v=2*(Finset.univ.filter (H v)).card)
  21     (hR : ∀ x y,Q.Adj x y → ordinaryRimLoad ψ x y=1)
  22     (herr : ∀ v u,|ordinarySpokeResidual ψ v u|≤η)
  23     (hT : ∀ v (i j : {u // H v u}),i.val≠j.val →
  24       γ*(Fintype.card V:ℝ)^(r^2)≤
  25         (Fintype.card (OrdinaryTransferConfiguration hr Q F H v i.val j.val):ℝ)) :
  26     ∃ Φ : OrdinaryWheel Q F H r → ℝ,
  27       (∀ x y,ordinaryRimLoad Φ x y=0) ∧
  28       (∀ v u,ordinarySpokeLoad Φ v u=ordinarySpokeResidual ψ v u) ∧
  29       (∀ D,|Φ D|≤(2*η*(6*((r:ℝ)^2)^r/γ)*(r+1))/(Fintype.card V:ℝ)^(r-1)) := by
  30   have hn' : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr hn
  31   let T (v : V) := ordinaryNeighbourTransfer hr Q F H v
  32   let R (v : V) := ordinaryRowCorrection v (ordinarySpokeResidual ψ v) (T v)
  33   have hpos (v : V) (i j : {u // H v u}) (hij : i.val≠j.val) :
  34       0<Fintype.card (OrdinaryTransferConfiguration hr Q F H v i.val j.val) := by
  35     have hp : (0:ℝ)<γ*(Fintype.card V:ℝ)^(r^2) := by positivity
  36     exact_mod_cast hp.trans_le (hT v i j hij)
  37   have hrows (v : V) :
  38       (∀ x y,ordinaryRimLoad (R v) x y=0) ∧
  39       (∀ x u,ordinarySpokeLoad (R v) x u=
  40         (if x=v then (1:ℝ) else 0)*ordinarySpokeResidual ψ v u) := by
  41     have hz : (∑ u : {u // H v u},ordinarySpokeResidual ψ v u)=0 := by
  42       rw [← finite_sum_restrict_of_zero (H v) (ordinarySpokeResidual ψ v)
  43         (fun u hu => ordinarySpokeResidual_outside ψ v u hu)]
  44       exact ordinarySpokeResidual_balanced ψ hdeg hR v
  45     exact ordinaryRowCorrection_loads v (ordinarySpokeResidual ψ v) hz
  46       (fun u hu => ordinarySpokeResidual_outside ψ v u hu) (T v)
  47       (fun i j => (ordinaryNeighbourTransfer_loads hr Q F H v i j (hpos v)).1)
  48       (fun i j => (ordinaryNeighbourTransfer_loads hr Q F H v i j (hpos v)).2)
  49   let Φ (D : OrdinaryWheel Q F H r) : ℝ := ∑ v, (1:ℝ)*R v D
  50   refine ⟨Φ,?_,?_,?_⟩
  51   · intro x y
  52     change ordinaryRimLoad (fun D => ∑ v,(1:ℝ)*R v D) x y=0
  53     rw [ordinaryRimLoad_combination]
  54     simp only [fun v => (hrows v).1 x y,mul_zero,Finset.sum_const_zero]
  55   · intro x u
  56     change ordinarySpokeLoad (fun D => ∑ v,(1:ℝ)*R v D) x u=ordinarySpokeResidual ψ x u
  57     rw [ordinarySpokeLoad_combination]
  58     simp only [fun v => (hrows v).2 x u,one_mul]
  59     simp [eq_comm]
  60   · intro D
  61     let K : ℝ := 6*((r:ℝ)^2)^r/γ
  62     let B (v : V) : ℝ := if v∈(ordinaryWheelRim D).support then
  63       K/(Fintype.card V:ℝ)^(r-1) else K/(Fintype.card V:ℝ)^r
  64     have hK : 0≤K := by dsimp [K];positivity
  65     have hB (v : V) : 0≤B v := by dsimp [B];split_ifs <;> positivity
  66     have hb (v : V) : |R v D|≤2*η*B v := by
  67       exact ordinaryRowCorrection_bound v (ordinarySpokeResidual ψ v) (T v) D η (B v)
  68         hη (hB v) (herr v)
  69         (fun i j => ordinaryNeighbourTransfer_bound hr Q F H v i j γ hγ hn (hT v) D)
  70         (fun i j hi hj => ordinaryNeighbourTransfer_zero_other_hubs hr Q F H v i j D hi hj)
  71     have hsum : (∑ v,B v)≤K*(r+1)/(Fintype.card V:ℝ)^(r-1) := by
  72       have hh := finite_vertex_coefficient_sum (ordinaryWheelRim D).support.toFinset
  73         (show 1≤r by omega) D.val.1.val.property.support_card K hK hn
  74       simpa only [B,Set.mem_toFinset] using hh
  75     calc
  76       |Φ D| = |∑ v,R v D| := by simp only [Φ,one_mul]
  77       _ ≤ ∑ v,|R v D| := Finset.abs_sum_le_sum_abs _ _
  78       _ ≤ ∑ v,2*η*B v := Finset.sum_le_sum (fun v _ => hb v)
  79       _ = 2*η*∑ v,B v := by rw [Finset.mul_sum]
  80       _ ≤ 2*η*(K*(r+1)/(Fintype.card V:ℝ)^(r-1)) :=
  81         mul_le_mul_of_nonneg_left hsum (by positivity)
  82       _ = _ := by dsimp [K];ring
