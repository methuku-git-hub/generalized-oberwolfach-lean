  16 theorem endpoint_fixed_weight_estimates
  17     (P R H F : SimpleGraph V) (A B : Finset V) (x : V)
  18     (hAB : Disjoint A B) (hR : R≤P\F)
  19     (hFB : ∀ u∈B,∀ w,¬F.Adj u w)
  20     (κ p q s c C L δ ξ : ℝ)
  21     (ha : 0<(A.card:ℝ)) (hκ : 0<κ) (hp : 0<p)
  22     (hc : 0≤c) (hC : 0≤C) (hL : 0≤L) (hδ : 0≤δ)
  23     (hb : κ*A.card≤(B.card:ℝ)) (hba : B.card≤A.card)
  24     (hq : p≤q) (hq1 : q≤1) (hs : p≤s) (hs1 : s≤1)
  25     (hF : ∀ u,(F.degree u:ℝ)≤L)
  26     (weight : Sym2 V→ℝ)
  27     (hweight : ∀ v∈A∩P.neighborFinset x,∀ w∈A∩P.neighborFinset x,
  28       0≤weight s(v,w) ∧ weight s(v,w)≤C/A.card)
  29     (hsum : ∀ v∈A∩P.neighborFinset x,
  30       ∑ w∈A∩P.neighborFinset x,weight s(v,w)=1)
  31     (hzero : ∀ v∈A∩P.neighborFinset x,∀ w∈A∩P.neighborFinset x,
  32       ¬P.Adj v w → weight s(v,w)=0)
  33     (hpositive : ∀ e∈(endpointEligibleGraph P F (A∩P.neighborFinset x) x).edgeSet,
  34       c/A.card≤weight e ∧ weight e≤C/A.card)
  35     (hbalance : |q*(A∩P.neighborFinset x).card/(2*B.card*s)-1|≤δ)
  36     (hrow : ∀ v∈common A {x} (fun y=>(P\F).neighborFinset y),
  37       |endpointRowPotential P R H F A B x v weight (B.card*q^2*s^3)-
  38         endpointEligibleRowWeight F (A∩P.neighborFinset x) x v (fun v w=>weight s(v,w))|≤ξ)
  39     (hindex : ∀ u∈B,
  40       |endpointIndexPotential P R H F A x u weight (B.card*q^2*s^3)-
  41         q/(B.card*s)*(∑ e∈(endpointEligibleGraph P F (A∩P.neighborFinset x) x).edgeFinset,weight e)|≤ξ) :
  42     let S:=common A {x} (fun y=>R.neighborFinset y)
  43     let T:=common B {x} (fun y=>H.neighborFinset y)
  44     let wt:=fun D : EndpointTriangle (endpointHostGraph R H A B) F S T=>
  45       weight (endpointTriangleRowEdge D)/(B.card*q^2*s^3)
  46     (∀ D,c/(A.card:ℝ)^2≤wt D ∧ wt D≤(C/(κ*p^5))/(A.card:ℝ)^2) ∧
  47     (∀ v∈S∪T,|1-hypergraphWeightedDegree endpointTriangleVertices wt v|≤
  48       δ+(2*C+(1+2*C)/(2*κ*p))*L/A.card+ξ) := by
  49   dsimp only
  50   let Px:=A∩P.neighborFinset x
  51   have hPa : (Px.card:ℝ)≤A.card := by exact_mod_cast Finset.card_le_card Finset.inter_subset_left
  52   have hmass:=endpoint_unordered_matching_mass_error P F Px x weight A.card C L ha hC hPa hF hweight hsum hzero
  53   have hindexTarget:=endpoint_index_target_error (A.card:ℝ) B.card κ p q s C L Px.card
  54     (∑ e∈(endpointEligibleGraph P F Px x).edgeFinset,weight e) δ
  55     ha hκ hp hC hL hb (hp.le.trans hq) hq1 hs hmass hbalance
  56   have hrowLoss : 2*C*L/A.card≤(2*C+(1+2*C)/(2*κ*p))*L/A.card := by
  57     apply div_le_div_of_nonneg_right _ ha.le
  58     apply mul_le_mul_of_nonneg_right _ hL
  59     have : 0≤(1+2*C)/(2*κ*p) := by positivity
  60     linarith
  61   have hindexLoss : ((1+2*C)/(2*κ*p))*L/A.card≤(2*C+(1+2*C)/(2*κ*p))*L/A.card := by
  62     apply div_le_div_of_nonneg_right _ ha.le
  63     apply mul_le_mul_of_nonneg_right _ hL
  64     linarith
  65   constructor
  66   · intro D
  67     exact endpoint_triangle_coefficient_bounds A.card B.card κ p q s c C _ ha hκ hp hc hC hb
  68       (by exact_mod_cast hba) hq hq1 hs hs1
  69       (hpositive _ (endpoint_triangle_parent_edge P R H F A B x hAB hR hFB D))
  70   · intro v hv
  71     rcases Finset.mem_union.mp hv with hv|hv
  72     · have hv' := (mem_singleton_common_graph R A x v).mp hv
  73       have hvP : v∈Px := Finset.mem_inter.mpr ⟨hv'.1,by simpa using (hR hv'.2).1⟩
  74       have hvE:= (mem_singleton_common_graph (P\F) A x v).mpr ⟨hv'.1,hR hv'.2⟩
  75       have htarget:=endpoint_matching_row_target F Px x v (fun v w=>weight s(v,w)) A.card C L
  76         ha hC hF (hweight v hvP) (hsum v hvP)
  77       rw [endpoint_row_potential_present P R H F A B x v hAB hR hFB hv weight _]
  78       rw [abs_sub_comm]
  79       have hh:=abs_sub_le (endpointRowPotential P R H F A B x v weight (B.card*q^2*s^3))
  80         (endpointEligibleRowWeight F Px x v (fun v w=>weight s(v,w))) 1
  81       have hrow' := hrow v (by convert hvE using 1 <;> congr!)
  82       linarith only [hh,hrow',htarget,hrowLoss,hδ]
  83     · have hvB := ((mem_singleton_common_graph H B x v).mp hv).1
  84       rw [endpoint_index_potential_present P R H F A B x v hAB hR hFB hv weight _]
  85       rw [abs_sub_comm]
  86       have hh:=abs_sub_le (endpointIndexPotential P R H F A x v weight (B.card*q^2*s^3))
  87         (q/(B.card*s)*(∑ e∈(endpointEligibleGraph P F Px x).edgeFinset,weight e)) 1
  88       linarith only [hh,hindex v hvB,hindexTarget,hindexLoss]
