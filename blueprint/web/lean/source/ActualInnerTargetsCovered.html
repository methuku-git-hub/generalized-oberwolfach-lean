  10 theorem actual_inner_targets_covered_by_complement
  11     {X V I:Type} [Fintype X] [Fintype V] [Fintype I]
  12     [DecidableEq X] [DecidableEq V] [DecidableEq I]
  13     (K J:I→Type) [∀w,DecidableEq (K w)] [∀w,DecidableEq (J w)]
  14     (F:I→SimpleGraph X) (data:∀w,ActualResidualData (K:=K w) (J:=J w) (F w))
  15     (D:I→Finset X) (ψ:I→X→V)
  16     (fC:∀w,{x // x∉D w}↪V) (Φ:∀w,{x // x∈(data w).remaining}→V)
  17     (hDR:∀w,Disjoint (D w) (data w).remaining)
  18     (hold:∀w (x:{x // x∉D w}),x.val∈(data w).oldDomain→fC w x=ψ w x.val)
  19     (hnew:∀w (x:{x // x∈(data w).remaining}) (hx:x.val∉D w),fC w ⟨x.val,hx⟩=Φ w x)
  20     (H:Finset I) (A:Finset V) (reserved selected absorber:I→Finset V):
  21     let occupied:=fun w=>((data w).oldDomain\D w).image (ψ w)
  22     let Z:=innerRequiredTargets H A reserved selected absorber occupied
  23     (∀w v,v∈Z w→∃x,Φ w x=v)→
  24     ∀w∈H,A⊆reserved w∪selected w∪absorber w∪Finset.univ.image (fC w):=by
  25   intro occupied Z hZ w hw v hv
  26   by_cases hmem:v∈reserved w∪selected w∪absorber w
  27   · exact Finset.mem_union_left _ hmem
  28   · apply Finset.mem_union_right
  29     by_cases hOcc:v∈occupied w
  30     · obtain ⟨x,hx,hxv⟩:=Finset.mem_image.mp hOcc
  31       have hxOld: x∈(data w).oldDomain:=(Finset.mem_sdiff.mp hx).1
  32       have hxD:x∉D w:=(Finset.mem_sdiff.mp hx).2
  33       exact Finset.mem_image.mpr ⟨⟨x,hxD⟩,Finset.mem_univ _,(hold w ⟨x,hxD⟩ hxOld).trans hxv⟩
  34     · have hvZ:v∈Z w:=by
  35         simp only [Z,innerRequiredTargets,hw,if_true,Finset.mem_sdiff,Finset.mem_union]
  36         refine ⟨hv,?_⟩
  37         intro hh
  38         rcases hh with hh|hh
  39         · exact hmem (by simpa only [Finset.mem_union] using hh)
  40         · exact hOcc hh
  41       obtain ⟨x,hx⟩:=hZ w v hvZ
  42       have hxD:x.val∉D w:=by intro h;exact Finset.disjoint_left.mp (hDR w) h x.property
  43       exact Finset.mem_image.mpr ⟨⟨x.val,hxD⟩,Finset.mem_univ _,(hnew w x hxD).trans hx⟩
  44 end Oberwolfach
