  12 theorem marked_hub_factor_vertex_count
  13     {V U E K:Type*} [Fintype V] [DecidableEq V]
  14     (tail head:E→V) (ktail khead:K→V) (m b:ℕ) (π:Equiv.Perm V)
  15     (H₀ Hₖ:V→U→Prop) (u:U)
  16     (M:Finset (HubMarkedCycle tail head ktail khead m π b H₀ Hₖ u))
  17     (hM:∀C∈M,∀D∈M,C≠D→Disjoint (hubMarkedCycleVertices u C) (hubMarkedCycleVertices u D))
  18     (S Y:Finset V) (hcover:M.biUnion (hubMarkedCycleVertices u)=S)
  19     (hYS:Y⊆S) (hmarked:∀x,Hₖ x u↔x∈Y) (hsep:∀x,Hₖ x u→¬H₀ x u):
  20     S.card=(m+1)*Y.card := by
  21   obtain ⟨select,hselected,honto⟩:=marked_hub_factor_root_bijection tail head ktail khead m b π H₀ Hₖ u M hM S Y hcover hYS hmarked hsep
  22   have himg:Finset.univ.image select=M:=by
  23     ext C
  24     simp only [Finset.mem_image,Finset.mem_univ,true_and]
  25     constructor
  26     · rintro ⟨x,rfl⟩;exact (hselected x).1
  27     · intro hC;exact honto C hC
  28   have hmcard:M.card=Y.card:=by
  29     rw [←himg,Finset.card_image_of_injective _ select.injective]
  30     simp
  31   have hcard:∀C:HubMarkedCycle tail head ktail khead m π b H₀ Hₖ u,
  32       (hubMarkedCycleVertices u C).card=m+1:=by
  33     intro C
  34     rw [hubMarkedCycleVertices,Finset.card_image_of_injective _
  35       (permutationSeparated_injective π b C.val C.property.1.1)]
  36     simp
  37   have hpair:(↑M:Set (HubMarkedCycle tail head ktail khead m π b H₀ Hₖ u)).PairwiseDisjoint (hubMarkedCycleVertices u):=by
  38     intro C hC D hD hne
  39     exact hM C hC D hD hne
  40   calc
  41     S.card=(M.biUnion (hubMarkedCycleVertices u)).card:=by rw [hcover]
  42     _=∑C∈M,(hubMarkedCycleVertices u C).card:=Finset.card_biUnion hpair
  43     _=M.card*(m+1):=by simp [hcard]
  44     _=Y.card*(m+1):=by rw [hmcard]
  45     _=(m+1)*Y.card:=Nat.mul_comm _ _
  46 end Oberwolfach
