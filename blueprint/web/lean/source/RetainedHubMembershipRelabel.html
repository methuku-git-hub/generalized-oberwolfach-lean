  13     (f : W ≃ V) (g : Z ≃ U) (A : V → Finset U) (B : W → Finset Z)
  14     (h : ∀ w z, z ∈ B w ↔ g z ∈ A (f w)) (t : ℕ)
  15     (J : StartRetentionOutcome B t) (w : W) (z : Z) :
  16     z ∈ startRetainedHubs B t J w ↔
  17       g z ∈ startRetainedHubs A t (startRetentionRelabelEquiv f g A B h t J) (f w) := by
  18   simp only [startRetainedHubs, startRetentionRelabelEquiv, Equiv.trans_apply,
  19     Equiv.piCongrLeft_apply_apply, Equiv.piCongrRight_apply, Pi.map_apply,
  20     fixedSubsetRelabelEquiv, Equiv.coe_fn_mk, Finset.mem_map, Function.Embedding.subtype_apply]
  21   constructor
  22   · rintro ⟨v,hv,hz⟩
  23     refine ⟨eligibleHubEquiv f g A B h w v, ?_, ?_⟩
  24     · exact ⟨v,hv,rfl⟩
  25     · change g v.val = g z
  26       rw [hz]
  27   · rintro ⟨u,hu,huval⟩
  28     obtain ⟨v,hv,hvu⟩ := hu
  29     refine ⟨v,hv,?_⟩
