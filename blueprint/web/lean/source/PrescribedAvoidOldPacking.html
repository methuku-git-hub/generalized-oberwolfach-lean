  10 theorem prescribed_embedding_avoids_physical_packing
  11     (Q:SimpleGraph X) (Dsrc:X→X→Prop) (hrep:∀x y,Q.Adj x y→Dsrc x y ∨ Dsrc y x)
  12     (B:Finset X) (g Φ:X→V) (hold:∀x∈B,Φ x=g x)
  13     (M:Finset (V×V)) (hM:(prescribedInducedArcs Dsrc B).image (Prod.map g g)=M)
  14     (D:V→V→Prop) (harc:∀x y,Dsrc x y→(x∉B ∨ y∉B)→D (Φ x) (Φ y))
  15     (K:Finset (Sym2 V)) (hMavoid:Disjoint (M.image prescribedArcPhysical) K)
  16     (hDavoid:∀x y,D x y→s(x,y)∉K):
  17     Disjoint (Q.edgeFinset.image (Sym2.map Φ)) K := by
  18   have done:∀x y,Dsrc x y→s(Φ x,Φ y)∉K:=by
  19     intro x y hxy
  20     by_cases hx:x∈B
  21     · by_cases hy:y∈B
  22       · have hm:(g x,g y)∈M:=by
  23           rw [←hM]
  24           apply Finset.mem_image.mpr
  25           refine ⟨(x,y),?_,rfl⟩
  26           simp [prescribedInducedArcs,hx,hy,hxy]
  27         have hphys:s(g x,g y)∈M.image prescribedArcPhysical:=by
  28           exact Finset.mem_image.mpr ⟨(g x,g y),hm,rfl⟩
  29         rw [←hold x hx,←hold y hy] at hphys
  30         exact (Finset.disjoint_left.mp hMavoid) hphys
  31       · exact hDavoid _ _ (harc x y hxy (Or.inr hy))
  32     · exact hDavoid _ _ (harc x y hxy (Or.inl hx))
  33   apply Finset.disjoint_left.mpr
  34   intro e he hK
  35   obtain ⟨a,ha,hae⟩:=Finset.mem_image.mp he
  36   induction a using Sym2.ind with
  37   | _ x y =>
  38     have hxy:Q.Adj x y:=SimpleGraph.mem_edgeFinset.mp ha
  39     have h':s(Φ x,Φ y)∈K:=by
  40       rw [←hae] at hK
  41       exact hK
  42     rcases hrep x y hxy with h|h
  43     · exact done x y h h'
  44     · apply done y x h
  45       simpa [Sym2.eq_swap] using h'
  46 end Oberwolfach
