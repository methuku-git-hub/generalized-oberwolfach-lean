  22 theorem ordinary_isolator_potential_lower (P F : SimpleGraph V) (H : V → U → Prop)
  23     (k : ℕ) (θ α : ℝ) (x y : V) (σ : Sym2 V → Bool) (hn : 0<Fintype.card V)
  24     (c η : ℝ) (hc : 0≤c)

  26     (hpotential : ∀ u,H x u → H y u →
  27       c≤rootedInducedSurvivalSum P
  28         (fun S : OrdinaryIsolatorCandidate P F H k θ α x y u=>S.val) s(x,y) σ) :
  29     c*η≤ordinaryIsolatorPotential P F H k θ α x y σ := by
  30   unfold ordinaryIsolatorPotential
  31   apply (le_div_iff₀ (Nat.cast_pos.mpr hn)).mpr
  32   have hsum := Finset.sum_le_sum (fun u (_ : u∈(Finset.univ : Finset U))=>
  33     show (if H x u ∧ H y u then c else 0)≤
  34       (if H x u ∧ H y u then rootedInducedSurvivalSum P
  35         (fun S : OrdinaryIsolatorCandidate P F H k θ α x y u=>S.val) s(x,y) σ else 0) by
  36       split_ifs with h
  37       · exact hpotential u h.1 h.2
  38       · exact le_rfl)
  39   have he : (∑ u : U,if H x u ∧ H y u then c else 0)=
  40       (Fintype.card {u : U // H x u ∧ H y u}:ℝ)*c := by
  41     rw [← Finset.sum_filter,Finset.sum_const,nsmul_eq_mul,Fintype.card_subtype]
  42   rw [he] at hsum
  43   have hh := mul_le_mul_of_nonneg_right hhubs hc
  44   nlinarith only [hsum,hh]
