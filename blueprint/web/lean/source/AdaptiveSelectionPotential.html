  14 def selectionRemaining (active : Fin N→I→Prop) (k : ℕ) (x : I) : ℕ :=
  15   (Finset.univ.filter (fun i : Fin N=>k ≤ i.val ∧ active i x)).card

  17 def selectionPotential (active : Fin N→I→Prop) (q : ℝ) (k : ℕ) (Z : I→ℕ) : ℝ :=
  18   ∑ x,(2:ℝ)^(Z x)*(1+q)^selectionRemaining active k x

  20 theorem selectionRemaining_step (active : Fin N→I→Prop) (i : Fin N) (x : I) :
  21     selectionRemaining active i.val x=
  22       selectionRemaining active (i.val+1) x+(if active i x then 1 else 0) := by
  23   let S := Finset.univ.filter (fun j : Fin N=>i.val+1≤j.val ∧ active j x)
  24   have hi : i∉S := by simp [S]
  25   by_cases ha : active i x
  26   · have he : Finset.univ.filter (fun j : Fin N=>i.val≤j.val ∧ active j x)=insert i S := by
  27       ext j
  28       simp only [Finset.mem_filter,Finset.mem_univ,true_and,Finset.mem_insert,S]
  29       constructor
  30       · rintro ⟨hj,hja⟩
  31         by_cases hji : j=i
  32         · exact Or.inl hji
  33         · right
  34           have hne : j.val≠i.val := fun hh=>hji (Fin.ext hh)
  35           exact ⟨by omega,hja⟩
  36       · rintro (rfl|⟨hj,hja⟩)
  37         · exact ⟨le_rfl,ha⟩
  38         · exact ⟨by omega,hja⟩
  39     simp only [selectionRemaining,he,Finset.card_insert_of_notMem hi,if_pos ha]
  40     rfl
  41   · have he : Finset.univ.filter (fun j : Fin N=>i.val≤j.val ∧ active j x)=S := by
  42       ext j
  43       simp only [S,Finset.mem_filter,Finset.mem_univ,true_and]
  44       constructor
  45       · rintro ⟨hj,hja⟩
  46         have hne : j.val≠i.val := by intro he;exact ha (Fin.ext he ▸ hja)
  47         exact ⟨by omega,hja⟩
  48       · rintro ⟨hj,hja⟩;exact ⟨by omega,hja⟩
  49     simp only [selectionRemaining,he,if_neg ha,Nat.add_zero]
  50     rfl

  52 theorem sum_two_pow_increment (z : ℕ) (hit : A→Prop) :
  53     (∑ a,(2:ℝ)^(z+(if hit a then 1 else 0)))=
  54       (2:ℝ)^z*((Fintype.card A:ℝ)+(Finset.univ.filter hit).card) := by
  55   have hp (a : A) : (2:ℝ)^(z+(if hit a then 1 else 0))=
  56       (2:ℝ)^z*(1+(if hit a then 1 else 0)) := by
  57     by_cases h : hit a <;> simp [h,pow_succ] <;> ring
  58   simp_rw [hp]
  59   rw [← Finset.mul_sum,Finset.sum_add_distrib]
  60   simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul,mul_one]
  61   rw [Finset.sum_boole]

  63 theorem selectionPotential_step_average (active : Fin N→I→Prop) (i : Fin N)
  64     (q : ℝ) (hq : 0≤q) (Z : I→ℕ) (hit : A→I→Prop)
  65     (hinactive : ∀ x,¬active i x → ∀ a,¬hit a x)
  66     (hcount : ∀ x,((Finset.univ.filter (fun a=>hit a x)).card:ℝ)≤q*Fintype.card A) :
  67     (∑ a,selectionPotential active q (i.val+1) (fun x=>Z x+(if hit a x then 1 else 0)))≤
  68       (Fintype.card A:ℝ)*selectionPotential active q i.val Z := by
  69   unfold selectionPotential
  70   rw [Finset.sum_comm,Finset.mul_sum]
  71   apply Finset.sum_le_sum
  72   intro x hx
  73   rw [← Finset.sum_mul,sum_two_pow_increment]
  74   rw [selectionRemaining_step]
  75   by_cases ha : active i x
  76   · rw [if_pos ha,pow_succ]
  77     have hh := mul_le_mul_of_nonneg_left (add_le_add_left (hcount x) (Fintype.card A))
  78       (by positivity : (0:ℝ)≤2^(Z x)*(1+q)^selectionRemaining active (i.val+1) x)
  79     nlinarith only [hh]
  80   · simp only [if_neg ha,Nat.add_zero]
  81     have he : Finset.univ.filter (fun a=>hit a x)=∅ := by
  82       ext a
  83       simp only [Finset.mem_filter,Finset.mem_univ,true_and,Finset.notMem_empty,iff_false]
  84       exact hinactive x ha a
  85     rw [he,Finset.card_empty,Nat.cast_zero,add_zero]
  86     ring_nf
  87     exact le_rfl

  90     (active : Fin N→I→Prop) (i : Fin N) (q : ℝ) (hq : 0≤q) (Z : I→ℕ)

 102 theorem selectionPotential_counter_le (active : Fin N→I→Prop) (q : ℝ) (hq : 0≤q)
 103     (k : ℕ) (Z : I→ℕ) (x : I) : (2:ℝ)^(Z x)≤selectionPotential active q k Z := by
 104   have hp : (1:ℝ)≤(1+q)^selectionRemaining active k x := one_le_pow₀ (by linarith)
 105   calc
 106     _ ≤ (2:ℝ)^(Z x)*(1+q)^selectionRemaining active k x := by nlinarith only [hp,pow_pos (by norm_num : (0:ℝ)<2) (Z x)]

 108       (fun y _=>mul_nonneg (pow_nonneg (by norm_num : (0:ℝ)≤2) (Z y))
