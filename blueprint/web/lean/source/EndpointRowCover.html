  15 theorem endpoint_factor_row_cover (Q H F : SimpleGraph V) (A B X S T : Finset V)
  16     (hAB : Disjoint A B) (hXA : X⊆A) (x : V)
  17     (hA : S⊆X) (hB : T⊆B)
  18     (hS : ∀ a,Q.Adj x a ↔ a∈S) (hT : ∀ u,H.Adj x u ↔ u∈T)
  19     (hF : ∀ a,Q.Adj x a → ¬F.Adj x a)
  20     (hfactor : HasResourceDecomposition
  21       (endpointTriangleVertices (Q:=endpointHostGraph Q H A B) (F:=F)
  22         (S:=S) (T:=T)) (S∪T)) :
  23     ∃ M : Finset (OrdinaryWheel Q F H.Adj 3), OrdinaryRowCover M x X 0 := by
  24   obtain ⟨M,W,hW,hdis,hrim,hspoke,hdeg,hrow,hcol⟩ :=
  25     endpoint_factor_completed_wheels Q H F A B S T
  26       hAB (hA.trans hXA) hB x
  27       (fun a ha=>⟨(hS a).mpr ha,hF a ((hS a).mpr ha)⟩)
  28       (fun u hu=>(hT u).mpr hu) hfactor
  29   refine ⟨Finset.univ.image W,ordinaryRowCover_of_family W x X 0 ?_ hdis ?_ ?_ ?_ ?_ hcol⟩
  30   · intro C a ha
  31     rw [←Set.mem_toFinset,(hW C).2,Finset.mem_insert] at ha
  32     exact ha.elim Or.inl (fun h=>Or.inr (hA (Finset.mem_inter.mp h).2))
  33   · intro a
  34     constructor
  35     · intro ha
  36       obtain ⟨C,hC⟩ := SimpleGraph.iSup_adj.mp ha
  37       exact (W C).val.1.property hC
  38     · intro ha
  39       obtain ⟨C,hC,_⟩ := hrim a ((hS a).mp ha)
  40       apply SimpleGraph.iSup_adj.mpr
  41       exact ⟨C,(ordinaryWheelResources_rim (W C) s(x,a)).mp hC⟩
  42   · intro u
  43     constructor
  44     · rintro ⟨C,hu,hx⟩
  45       exact hu ▸ (W C).property.1 x hx
  46     · intro hu
  47       obtain ⟨C,hC,_⟩ := hspoke u ((hT u).mp hu)
  48       exact ⟨C,(ordinaryWheelResources_spoke (W C) x u).mp hC⟩
  49   · simpa using hdeg
  50   · simpa using hrow
