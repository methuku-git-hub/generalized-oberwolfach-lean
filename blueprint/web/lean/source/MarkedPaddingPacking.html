  55 theorem marked_padding_avoids_old_resources (hm:0<m)
  56     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  57     (W:MarkedWheel tail head ktail khead m π b H₀ HK)
  58     (v:V) (p:Fin m↪V) (Z:Finset V) (B:Finset U)
  59     (hW:∀ a,W.1.val a=Fin.cons (α:=fun _=>V) v p a) (hav:∀ a,p a∉Z)
  60     (hh:W.2.val∉B) (L:Finset (Finset V⊕(V×U)))
  61     (hL:∀ e∈L,match e with
  62       | Sum.inl e=>∃ a:E⊕K,markedPhysicalEdge tail head ktail khead π a=e∧
  63           Sum.elim tail ktail a∈Z∧Sum.elim head khead a∈Z
  64       | Sum.inr z=>z.2∈B):
  65     Disjoint (markedWheelPhysicalResources W) L:=by
  66   apply Finset.disjoint_left.mpr
  67   intro e he hle
  68   cases e with
  69   | inl e=>
  70     obtain ⟨a,ha,hat,hah⟩:=hL _ hle
  71     have hw: e∈markedCyclePhysicalEdges tail head ktail khead m π b W.1:=
  72       (markedWheelPhysicalResources_rim W e).mp he
  73     obtain ⟨k,_,hk⟩:=Finset.mem_image.mp hw
  74     have hka:=hp (hk.trans ha.symm)

  76     rw [hka] at ht

  78   | inr z=>
