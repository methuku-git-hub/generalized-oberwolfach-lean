  11 def remainingSwap {V : Type*} [DecidableEq V] (x y : V) :
  12     {z : V // z ≠ x} ≃ {z : V // z ≠ y} where
  13   toFun z := ⟨Equiv.swap x y z, fun h => z.2 ((Equiv.swap x y).injective (h.trans (Equiv.swap_apply_left x y).symm))⟩
  14   invFun z := ⟨Equiv.swap x y z, fun h => z.2 ((Equiv.swap x y).injective (h.trans (Equiv.swap_apply_right x y).symm))⟩
  15   left_inv z := by apply Subtype.ext; exact Equiv.swap_apply_self x y z
  16   right_inv z := by apply Subtype.ext; exact Equiv.swap_apply_self x y z
