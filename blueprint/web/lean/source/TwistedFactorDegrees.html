  17 theorem twistedFactorHead_injective (π : Equiv.Perm V) (f : I → Fin (m+1) → V)
  18     (hf : Function.Injective (fun p : I×Fin (m+1) => f p.1 p.2))
  19     (hsep : ∀ i k (j : Fin m),f i j.succ≠π (f k 0)) :
  20     Function.Injective (fun p : I×Fin (m+1) => twistedTupleHead π (f p.1) p.2) := by
  21   rintro ⟨i,j⟩ ⟨k,l⟩
  22   refine Fin.lastCases ?_ (fun j => ?_) j
  23   · refine Fin.lastCases ?_ (fun l => ?_) l
  24     · intro he
  25       simp only [twistedTupleHead,Fin.snoc_last,Fin.snoc_castSucc] at he
  26       have hh := hf (a₁:=(i,0)) (a₂:=(k,0)) (π.injective he)
  27       have hik : i=k := congrArg (fun p : I×Fin (m+1) => p.1) hh
  28       exact Prod.ext hik rfl
  29     · intro he
  30       simp only [twistedTupleHead,Fin.snoc_last,Fin.snoc_castSucc] at he
  31       exact (hsep k i l he.symm).elim
  32   · refine Fin.lastCases ?_ (fun l => ?_) l
  33     · intro he
  34       simp only [twistedTupleHead,Fin.snoc_last,Fin.snoc_castSucc] at he
  35       exact (hsep i k j he).elim
  36     · intro he
  37       simp only [twistedTupleHead,Fin.snoc_last,Fin.snoc_castSucc] at he
  38       have hh := hf (a₁:=(i,j.succ)) (a₂:=(k,l.succ)) he
  39       have hik : i=k := congrArg Prod.fst hh
  40       have hjl : j=l := Fin.succ_injective m (congrArg Prod.snd hh)
  41       subst k;subst l;rfl
