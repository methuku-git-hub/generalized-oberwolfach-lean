   9 theorem finiteLaw_sigma_reference_extension
  10     {Ω Λ:Type} [Fintype Ω] [Fintype Λ] {F:Λ→Type} [∀ y,Fintype (F y)]
  11     (μ:FiniteLaw Ω) (f:Ω→Λ) (ν:∀ y,FiniteLaw (F y))
  12     (B:(Σ x,F (f x))→Prop) (Ref:(Σ y,F y)→Prop) (p:ℝ):
  13     let joint:=μ.sigma (fun x=>ν (f x))
  14     let selected:=(μ.map f).sigma ν
  15     let project:=fun z:(Σ x,F (f x))=>(⟨f z.1,z.2⟩:(Σ y,F y));
  16     p≤joint.probability B→
  17     (∀ E:(Σ y,F y)→Prop,p≤selected.probability E→∃ y,Ref y ∧ E y)→
  18     ∃ z,Ref (project z) ∧ B z ∧ 0<joint.mass z:=by
  19   intro joint selected project hB hRef
  20   let Extra:=fun y:(Σ y,F y)=>∃ z,project z=y ∧ B z ∧ 0<joint.mass z
  21   have hprob:p≤selected.probability Extra:=
  22     hB.trans (finiteLaw_sigma_positive_extension_probability μ f ν B)
  23   obtain ⟨y,hy,z,hproject,hgood,hmass⟩:=hRef Extra hprob
  24   refine ⟨z,?_,hgood,hmass⟩
  25   simpa only [hproject] using hy
