  16 theorem sparse_rim_expansion_cycle (Q F : SimpleGraph V) (A W R : Finset V)
  17     (x y : V) (r : ℕ) (hr : 3≤r) (d c : ℝ) (hd : 0≤d)
  18     (hF : ∀ v,(F.degree v:ℝ)≤d)
  19     (hcount : ∀ v w,c≤((A∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  20     (hbudget : W.card+((R.card:ℝ)+r+6)*(d+1)<c) :
  21     ∃ f : Fin r ↪ V,
  22       (∀ i j,(SimpleGraph.cycleGraph r).Adj i j → Q.Adj (f i) (f j)) ∧
  23       Q.Adj y (f ⟨0,by omega⟩) ∧ Q.Adj x (f ⟨r-1,by omega⟩) ∧
  24       (∀ i j,¬F.Adj (f i) (f j)) ∧
  25       ∀ i,f i∈A ∧ f i∉W ∧ f i∉R ∧ ∀ v∈R,¬F.Adj (f i) v := by
  26   have hd1 : 0≤d+1 := by linarith
  27   have hrR : (0:ℝ)≤r := Nat.cast_nonneg _
  28   obtain ⟨a,haA,haW,haR,hya,_hya,haF⟩ := sparse_common_neighbour_avoiding Q F A W R y y d c hd hF
  29     (hcount y y) (by nlinarith only [hbudget,hd1,hrR])
  30   have hRa : ((insert a R).card:ℝ)≤R.card+1 := by exact_mod_cast Finset.card_insert_le a R
  31   obtain ⟨b,hbA,hbW,hbRa,hxb,hab,hbF⟩ := sparse_common_neighbour_avoiding Q F A W (insert a R) x a d c hd hF
  32     (hcount x a) (by nlinarith only [hbudget,hd1,hrR,hRa])
  33   have hbR : b∉R := fun h => hbRa (Finset.mem_insert_of_mem h)
  34   let D := R∪R.biUnion (fun v => F.neighborFinset v)
  35   let W' := W∪D
  36   let Z : Finset V := {a,b}
  37   have hZ : ∀ v∈Z,∀ w∈Z,¬F.Adj v w := by
  38     apply graph_independent_insert F ({b}:Finset V) a
  39     · intro v hv w hw
  40       rw [Finset.mem_singleton.mp hv,Finset.mem_singleton.mp hw]
  41       exact F.loopless.irrefl b
  42     · intro w hw h
  43       rw [Finset.mem_singleton.mp hw] at h
  44       exact hbF a (Finset.mem_insert_self _ _) h.symm
  45   have hW' : (W'.card:ℝ)≤W.card+(R.card:ℝ)*(d+1) := by
  46     have hD := forbidden_used_neighbours_card F R d hF
  47     have hh : (W'.card:ℝ)≤W.card+(D.card:ℝ) := by exact_mod_cast Finset.card_union_le W D
  48     exact hh.trans (by dsimp [D];linarith only [hD])
  49   have hZc : (Z.card:ℝ)≤2 := by
  50     have hh := Finset.card_insert_le a ({b}:Finset V)
  51     simp only [Finset.card_singleton] at hh
  52     exact_mod_cast hh
  53   have hrsub : ((r-1:ℕ):ℝ)≤r := by exact_mod_cast Nat.sub_le r 1
  54   obtain ⟨p,hp,hlen,hclass,hind,_hZp⟩ := sparse_fixed_length_forbidden_path Q F A W' Z a b
  55     hab.ne (by simp [Z]) (by simp [Z]) hZ (r-1) (by omega) d c hd hF hcount
  56     (by nlinarith only [hbudget,hW',hZc,hrsub,hd1])
  57   let f := pathCycleEmbedding p hp hlen
  58   refine ⟨f,pathCycleEmbedding_hom hr p hp hlen hab,?_,?_,?_,?_⟩
  59   · simpa only [f,pathCycleEmbedding_first] using hya
  60   · simpa only [f,pathCycleEmbedding_last (r:=r) (by omega) p hp hlen] using hxb
  61   · intro i j
  62     exact hind _ (p.getVert_mem_support _) _ (p.getVert_mem_support _)
  63   · intro i
  64     rcases hclass (f i) (p.getVert_mem_support _) with ha|hb|hrest
  65     · rw [ha];exact ⟨haA,haW,haR,haF⟩
  66     · rw [hb];exact ⟨hbA,hbW,hbR,fun v hv => hbF v (Finset.mem_insert_of_mem hv)⟩
  67     · have hnotW : f i∉W := fun h => hrest.2.1 (Finset.mem_union_left _ h)
  68       have hnotR : f i∉R := fun h => hrest.2.1
  69         (Finset.mem_union_right _ (Finset.mem_union_left _ h))
  70       refine ⟨hrest.1,hnotW,hnotR,?_⟩
  71       intro v hv hFiv
  72       apply hrest.2.1
  73       exact Finset.mem_union_right _ (Finset.mem_union_right _ (Finset.mem_biUnion.mpr
  74         ⟨v,hv,(F.mem_neighborFinset v (f i)).mpr hFiv.symm⟩))
