  17 theorem marked_two_families_packing
  18     (R:I→MarkedWheel tail head ktail khead m π b H₀ HK)
  19     (S:J→MarkedWheel tail head ktail khead m π b H₀ HK)
  20     (hRR:∀ i j,i≠j→Disjoint (markedWheelPhysicalResources (R i)) (markedWheelPhysicalResources (R j)))
  21     (hSS:∀ i j,i≠j→Disjoint (markedWheelPhysicalResources (S i)) (markedWheelPhysicalResources (S j)))
  22     (hRS:∀ i j,Disjoint (markedWheelPhysicalResources (R i)) (markedWheelPhysicalResources (S j))):
  23     ∀ W∈Finset.univ.image R∪Finset.univ.image S,
  24       ∀ Z∈Finset.univ.image R∪Finset.univ.image S,W≠Z→
  25         Disjoint (markedWheelPhysicalResources W) (markedWheelPhysicalResources Z):=by
  26   intro W hW Z hZ hne
  27   rcases Finset.mem_union.mp hW with hW|hW <;>
  28     rcases Finset.mem_union.mp hZ with hZ|hZ
  29   · obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hW
  30     obtain ⟨j,_,rfl⟩:=Finset.mem_image.mp hZ
  31     exact hRR i j (fun h=>hne (congrArg R h))
  32   · obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hW
  33     obtain ⟨j,_,rfl⟩:=Finset.mem_image.mp hZ
  34     exact hRS i j
  35   · obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hW
  36     obtain ⟨j,_,rfl⟩:=Finset.mem_image.mp hZ
  37     exact (hRS j i).symm
  38   · obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hW
  39     obtain ⟨j,_,rfl⟩:=Finset.mem_image.mp hZ
  40     exact hSS i j (fun h=>hne (congrArg S h))

  43     (R:I→MarkedWheel tail head ktail khead m π b H₀ HK)
  44     (S:J→MarkedWheel tail head ktail khead m π b H₀ HK)
  45     (X:Finset V) (Z:Finset U) (er:I→Finset V) (es:J→V×U)
  46     (hR:∀ i,(markedWheelPhysicalResources (R i)).filter (markedInnerResource X Z)={Sum.inl (er i)})
  47     (hS:∀ i,(markedWheelPhysicalResources (S i)).filter (markedInnerResource X Z)={Sum.inr (es i)}):

  50   ext e
  51   have hr (i:I):e∈markedWheelPhysicalResources (R i)∧markedInnerResource X Z e↔e=Sum.inl (er i):=by
  52     simpa only [Finset.mem_filter,Finset.mem_singleton] using Iff.of_eq (congrArg (fun A=>e∈A) (hR i))
  53   have hs (i:J):e∈markedWheelPhysicalResources (S i)∧markedInnerResource X Z e↔e=Sum.inr (es i):=by
  54     simpa only [Finset.mem_filter,Finset.mem_singleton] using Iff.of_eq (congrArg (fun A=>e∈A) (hS i))
  55   simp only [Finset.mem_filter,Finset.mem_biUnion,Finset.mem_union,Finset.mem_image,Finset.mem_univ,true_and]
  56   constructor
  57   · rintro ⟨⟨W,(⟨i,rfl⟩|⟨j,rfl⟩),he⟩,hi⟩
  58     · exact Or.inl ⟨i,((hr i).mp ⟨he,hi⟩).symm⟩
  59     · exact Or.inr ⟨j,((hs j).mp ⟨he,hi⟩).symm⟩
  60   · rintro (⟨i,rfl⟩|⟨j,rfl⟩)

  62       exact ⟨⟨R i,Or.inl ⟨i,rfl⟩,h.1⟩,h.2⟩
  63     · have h:=(hs j).mpr rfl
  64       exact ⟨⟨S j,Or.inr ⟨j,rfl⟩,h.1⟩,h.2⟩
