   9 theorem matching_physical_lift {C X Y W:Type} [DecidableEq C] [DecidableEq X] [DecidableEq Y]
  10     (S:Finset C) (R:C→Finset X) (φ:X→Y) (lift:C→W) (B:W→Finset Y)
  11     (hφ:Function.Injective φ)
  12     (hmatch:∀c∈S,∀d∈S,c≠d→Disjoint (R c) (R d))
  13     (hphysical:∀c∈S,B (lift c)=(R c).image φ):
  14     ∀c∈S,∀d∈S,c≠d→Disjoint (B (lift c)) (B (lift d)) :=by
  15   intro c hc d hd hcd
  16   rw [hphysical c hc,hphysical d hd]
  17   apply Finset.disjoint_left.mpr
  18   intro y hyc hyd
  19   obtain ⟨x,hxc,hxy⟩:=Finset.mem_image.mp hyc
  20   obtain ⟨x',hx'd,hx'y⟩:=Finset.mem_image.mp hyd
  21   have hxx':x=x':=hφ (hxy.trans hx'y.symm)
  22   subst x'
  23   exact (Finset.disjoint_left.mp (hmatch c hc d hd hcd)) hxc hx'd
