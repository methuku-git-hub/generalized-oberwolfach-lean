  19 theorem marked_endpoint_column_estimates
  20     (P : SimpleGraph V) (π : Equiv.Perm V) (d h : ℕ) (hh : 1≤h) (A : Finset V) (B : Finset U)
  21     (p k : ℝ) (q : U→Γ→ℝ) (s : U→Color→ℝ) (g : Γ) (j : Color) (err : ℝ) (herr : 0≤err)
  22     (σ : RawMarkedOutcome V U Γ Color) (H : Bool→V→U→Prop)
  23     (hr : MarkedRowsWithSpokes P π d h A p k q s g j err σ H)
  24     (w : Bool→ℝ) (hq : ∀ u∈B,q u g=w true) (hs : ∀ u∈B,s u j=w false)
  25     (hw : ∀ t,0≤w t ∧ w t≤1) (a : Bool×Bool×V) :
  26     (|((markedEndpointLabels P π d A a σ).card:ℝ)-
  27       (if a.1 then k else p)*(markedRowDomain P π (3*d) A {a} ∅).card|≤err) ∧
  28     (∀ (t : Bool) u,u∈B →
  29       |(((markedEndpointLabels P π d A a σ).filter (fun z=>H t z u)).card:ℝ)-
  30         w t*(markedEndpointLabels P π d A a σ).card|≤2*err) := by
  31   let C := markedRowDomain P π (3*d) A {a} ∅
  32   let L := markedEndpointLabels P π d A a σ
  33   have hbase : |(L.card:ℝ)-(if a.1 then k else p)*C.card|≤err := by
  34     have he := hr {a} ∅ ∅ (by simpa using hh) (by simp) (by simp) (by simp)
  35     simpa [markedRowCountWithSpokes,markedEndpointLabels,L,C,Finset.card_filter,
  36       Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero,-Finset.sum_boole] using he.le
  37   refine ⟨hbase,?_⟩
  38   intro t u hu
  39   have he := hr {a} ∅ {(t,u)} (by simpa using hh) (by simp) (by simpa using hh)
  40     (by intro b hb c hc he;simpa using (Finset.mem_singleton.mp hb).trans (Finset.mem_singleton.mp hc).symm)
  41   have hew : (if t then q u g else s u j)=w t := by
  42     cases t <;> simp [hq u hu,hs u hu]
  43   have he' : |((L.filter (fun z=>H t z u)).card:ℝ)-
  44       w t*((if a.1 then k else p)*C.card)|≤err := by
  45     simpa [markedRowCountWithSpokes,markedEndpointLabels,L,C,Finset.card_filter,
  46       Finset.filter_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero,hew,mul_left_comm,mul_assoc,-Finset.sum_boole]
  47       using he.le
  48   exact count_error_after_comparing_target _ _ _ _ err (hw t).1 (hw t).2 herr he' hbase
