  13     (as:List X) (ha:as.Nodup) (hlen:2≤as.length) (A:Finset X) (f:X→V)
  14     (hinj:Set.InjOn f A) (hsource:∀ u∈as,u∉A) (a:X) (haA:a∈A)

  16     (F:SimpleGraph V) (M:ℝ) (hM:0≤M) (hdeg:∀ v,(F.degree v:ℝ)≤M)

  19     (∀ u∈as,g u∈L ∧ g u∉S) ∧ DirectedPathThrough D F (g a) (g a) (as.map g) := by
  20   let B:=A.image f∪S
  21   obtain ⟨ys,hyslen,hysnodup,hys,hcycle⟩:=extending_list_embed_pinned_cycle D L β hext F M hM hdeg B (f a) (as.length+1) (by omega) hbudget
  22   have hlen':as.length=ys.length:=by simp [hyslen]
  23   have htarget:∀ z∈ys,z∉A.image f:=by
  24     intro z hz hmem
  25     exact (hys z hz).2.1 (Finset.mem_union_left S hmem)
  26   have hgood:∀ z∈ys,z∈L ∧ z∉S:=by
  27     intro z hz
  28     exact ⟨(hys z hz).1,fun h=>(hys z hz).2.1 (Finset.mem_union_right _ h)⟩
  29   let g:=updatePairedImages f (as.zip ys)
  30   have hb:=prescribed_embedding_zip_update as ys hlen' ha hysnodup A f hinj hsource htarget
  31   have hmap:=prescribed_embedding_update_map_list as ys hlen' ha hysnodup A f hinj hsource htarget
  32   have hnew:=prescribed_embedding_zip_targets as ys hlen' ha hysnodup A f hinj hsource htarget (fun z=>z∈L ∧ z∉S) hgood
  33   refine ⟨g,hb.1,hb.2.1,hnew,?_⟩
  34   have hga:g a=f a:=hb.2.1 a haA
