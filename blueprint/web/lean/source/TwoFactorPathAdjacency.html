   3 its immediate predecessor and successor on that path. -/

  10 theorem two_factor_path_adjacent_indices (F:SimpleGraph V)
  11     (hF:F.IsRegularOfDegree 2) {u v:V} (p:F.Walk u v) (hp:p.IsPath)
  12     (i j:ℕ) (hi0:i≠0) (hilt:i<p.length) (hjle:j≤p.length)
  13     (hij:F.Adj (p.getVert i) (p.getVert j)):
  14     j+1=i ∨ i+1=j := by
  15   have hsub:=p.toSubgraph.neighborSet_subset (p.getVert i)
  16   have hsmall:(p.toSubgraph.neighborSet (p.getVert i)).ncard=2 :=
  17     hp.ncard_neighborSet_toSubgraph_internal_eq_two hi0 hilt
  18   have hbig:(F.neighborSet (p.getVert i)).ncard=2 := by rw [F.ncard_neighborSet,hF (p.getVert i)]
  19   have hsets:p.toSubgraph.neighborSet (p.getVert i)=F.neighborSet (p.getVert i) := by
  20     apply Set.eq_of_subset_of_ncard_le hsub
  21     rw [hbig,hsmall]
  22   have hmem:p.getVert j∈p.toSubgraph.neighborSet (p.getVert i) := by rw [hsets];exact hij
  23   rw [hp.neighborSet_toSubgraph_internal hi0 hilt] at hmem
  24   simp only [Set.mem_insert_iff,Set.mem_singleton_iff] at hmem
  25   rcases hmem with h|h
  26   · left
  27     have hj:j=i-1 := hp.getVert_injOn (by simpa using hjle) (by simp;omega) h
  28     omega
  29   · right
  30     exact (hp.getVert_injOn (by simp;omega) (by simpa using hjle) h.symm)
  31 end Oberwolfach
