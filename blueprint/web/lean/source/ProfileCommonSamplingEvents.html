  16 theorem finiteLaw_vertices_bad_probability {Ω W : Type*} [Fintype Ω] [Fintype W] [DecidableEq W]
  17     (μ : FiniteLaw Ω) (Y : Ω → Finset W) (M : ℕ) (hM : ∀ ω,(Y ω).card ≤ M)
  18     (Bad : W → Ω → Prop) (ε : ℝ) (hε : 0 ≤ ε)
  19     (h : ∀ v,μ.probability (fun ω => v∈Y ω ∧ Bad v ω) ≤
  20       ε*μ.probability (fun ω => v∈Y ω)) :
  21     μ.probability (fun ω => ∃ v∈Y ω,Bad v ω) ≤ ε*M := by
  22   have hcard (ω) : ((Finset.univ.filter (fun v : W => v∈Y ω)).card:ℝ) ≤ M := by
  23     have he : Finset.univ.filter (fun v : W => v∈Y ω)=Y ω := by ext v;simp
  24     rw [he]
  25     exact_mod_cast hM ω
  26   have hh := finiteLaw_eligible_bad_probability μ Finset.univ (fun v ω => v∈Y ω) Bad M ε hε
  27     (fun ω => by convert hcard ω using 1 <;> congr!) (fun v _ => h v)
  28   simpa only [Finset.mem_univ,exists_const,true_and] using hh

  42 theorem profile_all_sampling_tests_exists {Ω : Type*} [Fintype Ω]
  43     (μ : FiniteLaw Ω) (a : I → ℕ) (c : ℕ) (ha : ∑ i,a i=c)
  44     (Y : Ω → Finset (Σ i,V i)) (M : ℕ) (hM : ∀ ω,(Y ω).card ≤ M)
  45     (BadV : (Σ i,V i) → Ω → Prop)
  46     (BadR : (∀ i,FixedSubset (V i) (a i)) → Ω → Prop) (BadOther : Ω → Prop)
  47     (εv εr δ : ℝ) (hεv : 0 ≤ εv) (hεr : 0 ≤ εr)
  48     (hV : ∀ v,μ.probability (fun ω => v∈Y ω ∧ BadV v ω) ≤
  49       εv*μ.probability (fun ω => v∈Y ω))
  50     (hR : ∀ ρ,μ.probability (fun ω => profileRootSet ρ⊆Y ω ∧ BadR ρ ω) ≤
  51       εr*μ.probability (fun ω => profileRootSet ρ⊆Y ω))
  52     (hOther : μ.probability BadOther ≤ δ)
  53     (hsmall : εv*M+εr*(M:ℝ)^c+δ < 1) :
  54     ∃ ω,0 < μ.mass ω ∧
  55       (∀ v∈Y ω,¬BadV v ω) ∧
  56       (∀ ρ,profileRootSet ρ⊆Y ω → ¬BadR ρ ω) ∧ ¬BadOther ω := by
  57   let A := fun j : Fin 3 => if j=0 then (fun ω => ∃ v∈Y ω,BadV v ω)
  58     else if j=1 then (fun ω => ∃ ρ,profileRootSet ρ⊆Y ω ∧ BadR ρ ω) else BadOther
  59   let p := fun j : Fin 3 => if j=0 then εv*M else if j=1 then εr*(M:ℝ)^c else δ
  60   have htests : ∀ j∈Finset.univ,μ.probability (A j) ≤ p j := by
  61     intro j _
  62     fin_cases j <;> norm_num [A,p]
  63     · simpa only [Sigma.exists] using finiteLaw_vertices_bad_probability μ Y M hM BadV εv hεv hV
  64     · exact profile_roots_bad_probability μ a c ha Y M hM BadR εr hεr hR
  65     · exact hOther
  66   have hp : ∑ j∈(Finset.univ : Finset (Fin 3)),p j < 1 := by
  67     simpa [p,Fin.sum_univ_succ,add_assoc] using hsmall
  68   obtain ⟨ω,hω,hgood⟩ := finiteLaw_common_event_exists μ Finset.univ A p htests hp
  69   have hv := hgood 0 (Finset.mem_univ _)
  70   have hr := hgood 1 (Finset.mem_univ _)
  71   have ho := hgood 2 (Finset.mem_univ _)
  72   simp only [A,ite_true,not_exists,not_and] at hv
  73   simp only [A,show (1 : Fin 3)≠0 by decide,ite_false,ite_true,not_exists,not_and] at hr
  74   simp only [A,show (2 : Fin 3)≠0 by decide,show (2 : Fin 3)≠1 by decide,ite_false] at ho
  75   exact ⟨ω,hω,hv,hr,ho⟩
