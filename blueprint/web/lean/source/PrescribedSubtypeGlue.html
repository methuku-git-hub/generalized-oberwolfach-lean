  11 theorem prescribed_subtype_embeddings_glue (embed:X↪Y) (A:Finset Y) (ψ:Y→V) (Φ:X→V)
  12     (hψ:Set.InjOn ψ (↑A:Set Y)) (hΦ:Function.Injective Φ)
  13     (hAgree:∀x,embed x∈A→Φ x=ψ (embed x))
  14     (hAvoid:∀x,embed x∉A→Φ x∉A.image ψ):
  15     ∃g:Y→V,Set.InjOn g (↑(A∪Finset.univ.image embed):Set Y) ∧
  16       (∀y∈A,g y=ψ y) ∧ (∀x,g (embed x)=Φ x) := by
  17   let Q:Finset Y:=Finset.univ.image embed
  18   let v₀:V:=Classical.choice inferInstance
  19   let lift:Y→V:=fun y=>if hy:y∈Q then Φ (Classical.choose (Finset.mem_image.mp hy)) else v₀
  20   have hlift:∀x,lift (embed x)=Φ x:=by
  21     intro x
  22     have hx:embed x∈Q:=Finset.mem_image.mpr ⟨x,Finset.mem_univ x,rfl⟩
  23     dsimp [lift]
  24     rw [dif_pos hx]
  25     have hh:=Classical.choose_spec (Finset.mem_image.mp hx)
  26     have heq:Classical.choose (Finset.mem_image.mp hx)=x:=embed.injective hh.2
  27     rw [heq]
  28   have hliftInj:Set.InjOn lift (↑Q:Set Y):=by
  29     intro y hy z hz heq
  30     obtain ⟨x,_,rfl⟩:=Finset.mem_image.mp hy
  31     obtain ⟨x',_,rfl⟩:=Finset.mem_image.mp hz
  32     rw [hlift,hlift] at heq
  33     exact congrArg embed (hΦ heq)
  34   have hag:∀y∈A∩Q,ψ y=lift y:=by
  35     intro y hy
  36     obtain ⟨hyA,hyQ⟩:=Finset.mem_inter.mp hy
  37     obtain ⟨x,_,rfl⟩:=Finset.mem_image.mp hyQ
  38     rw [hlift]
  39     exact (hAgree x hyA).symm
  40   have hav:∀y∈Q\A,∀z∈A,lift y≠ψ z:=by
  41     intro y hy z hz
  42     obtain ⟨hyQ,hyA⟩:=Finset.mem_sdiff.mp hy
  43     obtain ⟨x,_,rfl⟩:=Finset.mem_image.mp hyQ
  44     rw [hlift]
  45     intro heq
  46     exact hAvoid x hyA (Finset.mem_image.mpr ⟨z,hz,heq.symm⟩)
  47   obtain ⟨g,hginj,hgA,hgQ⟩:=prescribed_embeddings_glue A Q ψ lift hψ hliftInj hag hav
  48   refine ⟨g,hginj,hgA,?_⟩
  49   intro x
  50   have hx:embed x∈Q:=Finset.mem_image.mpr ⟨x,Finset.mem_univ x,rfl⟩
  51   rw [hgQ (embed x) hx,hlift]
  52 end Oberwolfach
