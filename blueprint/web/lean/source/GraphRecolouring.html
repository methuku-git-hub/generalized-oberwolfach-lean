  16 theorem recolour_colour_card_difference {I V : Type*} [Fintype I]
  17     [Fintype V] [DecidableEq V] {k : ℕ}
  18     (G : SimpleGraph V) (ι : I↪Sym2 V) (σ : Sym2 V → Fin k) (c : I → Fin k)
  19     (b a : Fin k) (hG : ∀ i,ι i∈G.edgeFinset) (hb : ∀ i,σ (ι i)=b) :
  20     ((edgeColourGraph G (recolourCoordinates ι σ c) a).edgeFinset.card:ℤ)-
  21       ((edgeColourGraph G σ a).edgeFinset.card:ℤ)=
  22       ∑ i,((if c i=a then (1:ℤ) else 0)-(if b=a then 1 else 0)) := by
  23   rw [edgeColourGraph_card_edge_sum,edgeColourGraph_card_edge_sum]
  24   have hh := recolour_weighted_difference ι σ c
  25     (fun e => if e∈G.edgeFinset then (1:ℤ) else 0) (fun d => if d=a then (1:ℤ) else 0)
  26   simpa only [hG,ite_true,one_mul,hb] using hh

  28 theorem recolour_nonreserve_card {I V : Type*} [Fintype I]
  29     [Fintype V] [DecidableEq V] {k : ℕ}
  30     (G : SimpleGraph V) (ι : I↪Sym2 V) (σ : Sym2 V → Fin k) (c : I → Fin k)
  31     (b a : Fin k) (hG : ∀ i,ι i∈G.edgeFinset) (hb : ∀ i,σ (ι i)=b) (hab : a≠b) :
  32     (edgeColourGraph G (recolourCoordinates ι σ c) a).edgeFinset.card=
  33       (edgeColourGraph G σ a).edgeFinset.card+(Finset.univ.filter (fun i => c i=a)).card := by
  34   have hh := recolour_colour_card_difference G ι σ c b a hG hb
  35   simp only [Ne.symm hab,ite_false,sub_zero] at hh
  36   have hc : (∑ i,if c i=a then (1:ℤ) else 0)=
  37       ((Finset.univ.filter (fun i => c i=a)).card:ℤ) := by
  38     simp only [Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  39   rw [hc] at hh
  40   have he : ((edgeColourGraph G (recolourCoordinates ι σ c) a).edgeFinset.card:ℤ)=
  41       ((edgeColourGraph G σ a).edgeFinset.card:ℤ)+((Finset.univ.filter (fun i => c i=a)).card:ℤ) := by
  42     omega
  43   exact_mod_cast he

  45 theorem recolour_path_count_stability {I V : Type*} [Fintype I]
  46     [Fintype V] [DecidableEq V] {k : ℕ}
  47     (G : SimpleGraph V) (ι : I↪Sym2 V) (σ : Sym2 V → Fin k) (c : I → Fin k)
  48     (u v : V) (huv : u≠v) (a b : Fin k) :
  49     |(colouredTwoPathCount G (recolourCoordinates ι σ c) u v a b:ℝ)-
  50       (colouredTwoPathCount G σ u v a b:ℝ)|≤Fintype.card I := by
  51   have hh := disjoint_pattern_stability (G.neighborFinset u∩G.neighborFinset v)
  52     (twoPathEdges u v) (fun w e => if e=s(u,w) then a else b)
  53     (Finset.univ.image ι) σ (recolourCoordinates ι σ c)
  54     (common_two_path_edges_disjoint G u v huv) (by
  55       intro e he
  56       symm
  57       apply recolourCoordinates_fixed
  58       simpa only [Finset.mem_image,Finset.mem_univ,true_and,Set.mem_range] using he)
  59   have hc : (Finset.univ.image ι).card=Fintype.card I := by
  60     rw [Finset.card_image_of_injective _ ι.injective,Finset.card_univ]
  61   simpa only [hc,← colouredTwoPathCount_patterns G _ u v huv a b] using hh

  63 theorem recolour_degree_stability {I V : Type*} [Fintype I]
  64     [Fintype V] [DecidableEq V] {k : ℕ}
  65     (G : SimpleGraph V) (ι : I↪Sym2 V) (σ : Sym2 V → Fin k) (c : I → Fin k)
  66     (a : Fin k) (v : V) :
  67     |((edgeColourGraph G (recolourCoordinates ι σ c) a).degree v:ℝ)-
  68       ((edgeColourGraph G σ a).degree v:ℝ)|≤Fintype.card I := by
  69   have hd : ∀ x∈G.neighborFinset v,∀ y∈G.neighborFinset v,x≠y →
  70       Disjoint ({s(v,x)}:Finset (Sym2 V)) {s(v,y)} := by
  71     intro x _ y _ hxy
  72     simp only [Finset.disjoint_singleton_left,Finset.mem_singleton]
  73     exact fun h => hxy (incidentPair_injective v h)
  74   have hh := disjoint_pattern_stability (G.neighborFinset v)
  75     (fun w => {s(v,w)}) (fun _ _ => a) (Finset.univ.image ι) σ (recolourCoordinates ι σ c) hd (by
  76       intro e he
  77       symm
  78       apply recolourCoordinates_fixed
  79       simpa only [Finset.mem_image,Finset.mem_univ,true_and,Set.mem_range] using he)
  80   have hc : (Finset.univ.image ι).card=Fintype.card I := by
  81     rw [Finset.card_image_of_injective _ ι.injective,Finset.card_univ]
