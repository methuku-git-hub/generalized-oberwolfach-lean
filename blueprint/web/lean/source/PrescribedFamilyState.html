  21 def PrescribedFamilyState.Valid (st:PrescribedFamilyState I X V)
  22     (QV:I→Finset X) (Q:I→SimpleGraph X) (A₀:I→Finset X) (ψ:I→X→V)
  23     (E₀:I→Finset (Sym2 V)) (J:SimpleGraph V) (t:ℝ):Prop:=
  24   FamilyPartialInjections st.assigned st.vertexMap ∧
  25   (∀ i x,x∈st.assigned i ↔ x∈A₀ i ∨ (i,x)∈st.freeOccurrences) ∧
  26   (∀ i,st.assigned i⊆QV i) ∧
  27   (∀ i x,x∈A₀ i→st.vertexMap i x=ψ i x) ∧
  28   (∀ v,((st.freeOccurrences.filter (fun p=>familyVertexImage st.vertexMap p=v)).card:ℝ)≤t+1) ∧
  29   (∀ i,st.sourceEdges i⊆(Q i).edgeFinset) ∧
  30   (∀ i e,e∈st.sourceEdges i→∀ x∈e.toFinset,x∈st.assigned i) ∧
  31   (∀ i,st.physicalEdges i=(st.sourceEdges i).image (Sym2.map (st.vertexMap i))) ∧
  32   PairwisePhysicalDisjoint st.physicalEdges ∧
  33   (∀ i,st.physicalEdges i=E₀ i∪st.newEdges i) ∧
  34   (∀ i,Disjoint (st.newEdges i) J.edgeFinset)
