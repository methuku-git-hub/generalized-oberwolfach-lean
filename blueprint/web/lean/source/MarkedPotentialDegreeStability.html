  38   have hM : 0≤M := by dsimp [M,C];positivity

  40   have hS : (S.card : ℝ)≤2 := by
  41     have hh : S.card≤2 := by dsimp [S]; exact (Finset.card_insert_le _ _).trans (by simp)

  46   simp only [Finset.notMem_empty,or_false,Finset.card_empty,Nat.cast_zero,add_zero] at hHcost

  48   apply hstart.trans
