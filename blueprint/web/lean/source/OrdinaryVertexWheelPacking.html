   5 /-! Physical disjointness of the constructed vertex-wheel pairs. Their
   6 fixed endpoints are disjoint, their internal vertices avoid every fixed
   7 endpoint and old root, and their root spokes are distinct. -/

  25       (rootWheelInternal (D j).first (x j) (a j) (b j)))

  53   have hSS (i j:I) (hij:i≠j):Disjoint (S i) (S j):=by
  54     apply Finset.disjoint_left.mpr
  55     intro v hv hv'
