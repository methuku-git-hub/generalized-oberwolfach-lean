  12 theorem Vortex.preserving_permutation {V U : Type*} [Fintype V] [Fintype U]
  13     {G : SimpleGraph V} {D m : ℕ} {β ε : ℝ} (Q : Vortex U G D β ε m)
  14     (he : 0 ≤ ε) (heq : ε ≤ 1 / 4) (h : ℕ) (hm : (h : ℝ) ≤ 3 * ε * m / 8) :
  15     ∃ π : Equiv.Perm V,
  16       (∀ i, ∀ x, π x ∈ Q.extendedVertices i ↔ x ∈ Q.extendedVertices i) ∧
  17       (∀ x k, nestedLevel Q.vertices Q.length ((π ^ k) x) =
  18         nestedLevel Q.vertices Q.length x) ∧
  19       (∀ x k, 0 < k → k < h → (π ^ k) x ≠ x) := by
  20   have hs : ∀ i < Q.length, (Q.vertices i \ Q.vertices (i + 1)).Nonempty →
  21       h ≤ (Q.vertices i \ Q.vertices (i + 1)).card := by
  22     intro i hi _
  23     have hh := Q.difference_size_lower he heq (Nat.lt_succ_self i)
  24       (by omega : i + 1 ≤ Q.length + 1)
  25     rw [Q.extendedVertices_eq (by omega : i ≤ Q.length),
  26       Q.extendedVertices_eq (by omega : i + 1 ≤ Q.length)] at hh
  27     exact_mod_cast hm.trans hh
  28   have ht : h ≤ (Q.vertices Q.length).card := by
  29     have hm0 : 0 ≤ ε * (m : ℝ) := mul_nonneg he (Nat.cast_nonneg m)
  30     have hh : (h : ℝ) ≤ (Q.vertices Q.length).card := by
  31       have := Q.terminal_lower
  32       linarith
  33     exact_mod_cast hh
  34   obtain ⟨π, hπ, hlevel, hreturn⟩ :=
  35     nested_shell_permutation Q.vertices Q.length h Q.initial Q.nested hs ht
  36   refine ⟨π, ?_, hlevel, hreturn⟩
  37   intro i x
  38   by_cases hi : i ≤ Q.length
  39   · rw [Q.extendedVertices_eq hi]
  40     exact hπ i hi x
  41   · rw [Q.extendedVertices_empty (by omega)]
  42     simp

  44 theorem vortex_with_permutation_exists {β ε ρ₀ : ℝ} (hb : 0 < β)
  45     (he : 0 < ε) (heq : ε < 1 / 4) (hp : 0 < ρ₀) (h : ℕ) :
  46     ∃ m₀ : ℕ, 1 ≤ m₀ ∧ ∀ (V : Type b) [Fintype V] (U : Type c) [Fintype U]
  47       (G : SimpleGraph V) (D m : ℕ),
  48       G.IsRegularOfDegree D → (Fintype.card V : ℝ) / 2 ≤ D →
  49       ρ₀ ≤ initialIndexDensity U D → initialIndexDensity U D ≤ 1 →
  50       m₀ ≤ m → m ≤ Fintype.card V →
  51       ∃ Q : Vortex U G D β ε m, ∃ π : Equiv.Perm V,
  52         (∀ i, ∀ x, π x ∈ Q.extendedVertices i ↔ x ∈ Q.extendedVertices i) ∧
  53         (∀ x k, nestedLevel Q.vertices Q.length ((π ^ k) x) =
  54           nestedLevel Q.vertices Q.length x) ∧
  55         (∀ x k, 0 < k → k < h → (π ^ k) x ≠ x) := by
  56   obtain ⟨N, hN1, hN⟩ := vortex_exists.{b,c} hb he heq hp
  57   refine ⟨max N ⌈8 * (h : ℝ) / (3 * ε)⌉₊, by omega, ?_⟩
  58   intro V _ U _ G D m hreg hD hρ hρ1 hm hmn
  59   obtain ⟨Q⟩ := hN V U G D m hreg hD hρ hρ1 (by omega) hmn
  60   have hceil : 8 * (h : ℝ) / (3 * ε) ≤ (m : ℝ) :=
  61     (Nat.le_ceil _).trans (by exact_mod_cast (show ⌈8 * (h : ℝ) / (3 * ε)⌉₊ ≤ m by omega))
  62   have hsize : (h : ℝ) ≤ 3 * ε * m / 8 := by
  63     have hh := (div_le_iff₀ (show 0 < 3 * ε by positivity)).mp hceil
  64     nlinarith
  65   obtain ⟨π, hπ⟩ := Q.preserving_permutation he.le heq.le h hsize
  66   exact ⟨Q, π, hπ⟩
