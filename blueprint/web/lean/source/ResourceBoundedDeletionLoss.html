  16 theorem ambient_resource_pair_weight (R : C → Finset V) (r J : ℕ) (hr : 2≤r)
  17     (hR : ∀ C₀,(R C₀).card=r) (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y}≤J)
  18     (w : C → ℝ) (B : ℝ) (hB : 0≤B) (hn : 0<Fintype.card V)
  19     (hw : ∀ C₀,w C₀≤B/(Fintype.card V:ℝ)^(r-1))
  20     (x y : V) (hxy : x≠y) :
  21     (∑ C₀,if x∈R C₀ ∧ y∈R C₀ then w C₀ else 0)≤J*B/Fintype.card V := by
  22   have hc := uniform_resources_pair_count R r J hr hR hJ x y hxy
  23   have hnR : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr hn
  24   calc
  25     _ ≤ ∑ C₀,if x∈R C₀ ∧ y∈R C₀ then B/(Fintype.card V:ℝ)^(r-1) else 0 := by
  26       apply Finset.sum_le_sum
  27       intro C₀ _;split_ifs <;> first | exact hw C₀ | exact le_rfl
  28     _ = (Fintype.card {C₀ : C // x∈R C₀ ∧ y∈R C₀}:ℝ)*(B/(Fintype.card V:ℝ)^(r-1)) := by
  29       simp only [Fintype.card_subtype,←Finset.sum_filter,Finset.sum_const,nsmul_eq_mul]
  30     _ ≤ (J*(Fintype.card V:ℝ)^(r-2))*(B/(Fintype.card V:ℝ)^(r-1)) := by
  31       apply mul_le_mul_of_nonneg_right _ (by positivity)
  32       exact_mod_cast hc
  33     _ = _ := by
  34       rw [show r-1=(r-2)+1 by omega,pow_succ]
  35       field_simp

  53       ∑ C₀,if x∈R C₀ ∧ ¬P C₀ then w C₀ else 0 := by
  54     unfold hypergraphWeightedDegree
  55     rw [←Finset.sum_sub_distrib]
  56     apply Finset.sum_congr rfl
  57     intro C₀ _
  58     by_cases hv : x∈R C₀ <;> by_cases hp : P C₀ <;> simp [hv,hp]
  59   rw [he]
