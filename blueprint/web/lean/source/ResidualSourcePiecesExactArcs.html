  11 theorem two_factor_subgraph_source_pieces_exact_arcs (F H:SimpleGraph V)
  12     (hF:F.IsRegularOfDegree 2) (hHF:H≤F) (R:Finset V)
  13     (hisol:∀x,x∉R→∀y,¬H.Adj x y):
  14     ∃ (D:V→V→Prop) (orig ps:List (PrescribedSourcePiece V)),
  15       orig.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset) ∧
  16       (∀p∈orig,∃a xs,p=.cycle a xs ∧ p.OriginalValid D) ∧
  17       sourcePieceVerticesUnion orig=Finset.univ ∧ sourcePieceEdgesUnion orig=F.edgeFinset ∧
  18       (∀x y,D x y ↔ (x,y)∈sourcePieceArcsUnion orig) ∧
  19       ps.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset) ∧
  20       (∀p∈ps,p.OriginalValid D) ∧ sourcePieceVerticesUnion ps=R ∧
  21       sourcePieceEdgesUnion ps=H.edgeFinset ∧
  22       sourcePieceArcsUnion ps=(sourcePieceArcsUnion orig).filter (fun e=>H.Adj e.1 e.2) := by
  23   obtain ⟨D₀,orig,hpair,hcycles,hvertices,hedges,hD₀⟩:=two_factor_directed_source_cycles F hF
  24   let D:V→V→Prop:=fun x y=>(x,y)∈sourcePieceArcsUnion orig
  25   have hcycles':∀p∈orig,∃a xs,p=.cycle a xs ∧ p.OriginalValid D := by
  26     intro p hp
  27     obtain ⟨a,xs,rfl,hvalid⟩:=hcycles p hp
  28     refine ⟨a,xs,rfl,hvalid.1,hvalid.2.1,?_⟩
  29     apply (prescribed_directed_edges_chain D ((a::xs)++[a])).2
  30     intro e he
  31     exact Finset.mem_biUnion.mpr ⟨.cycle a xs,List.mem_toFinset.mpr hp,he⟩
  32   obtain ⟨ps,hps,hvalid,hv,he,ha⟩:=retained_cycle_family_source_pieces_with_arcs H R hisol D orig hpair hcycles'
  33   refine ⟨D,orig,ps,hpair,hcycles',hvertices,hedges,by intro x y;rfl,hps,hvalid,?_,?_,ha⟩
  34   · ext x
  35     rw [hv x,hvertices]
  36     simp
  37   · rw [he,hedges]
  38     ext e
  39     simp only [Finset.mem_filter]
  40     have hsub:H.edgeFinset⊆F.edgeFinset := SimpleGraph.edgeFinset_mono hHF
  41     constructor
  42     · exact And.right
  43     · intro hx;exact ⟨hsub hx,hx⟩
  44 end Oberwolfach
