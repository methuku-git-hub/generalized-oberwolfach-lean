  14 theorem marked_extension_target_lower (a domain target count γ p ε loss : ℝ)
  15     (ha : 0≤a) (hγ : 0<γ) (hp : 0<p) (hdomain : γ*a-loss≤domain)
  16     (hloss : loss≤γ*a/2) (htarget : p^3≤target)
  17     (hε : ε≤p^3*γ/4) (hcount : target*domain-ε*a≤count) :
  18     (p^3*γ/4)*a≤count := by
  19   have hdom : γ*a/2≤domain := by linarith only [hdomain,hloss]
  20   have ht0 : 0≤target := (pow_nonneg hp.le _).trans htarget
  21   have hh:=mul_le_mul htarget hdom (by positivity : 0≤γ*a/2) ht0
  22   have he:=mul_le_mul_of_nonneg_right hε ha
  23   nlinarith only [hh,he,hcount]

  25 theorem marked_one_extension_lower (P : SimpleGraph V) (π : Equiv.Perm V) (b : ℕ)
  26     (R : (V×V)⊕(V×V)→Prop) (H : Bool→V→U→Prop) (A : Finset V)
  27     (hA : A.image π=A) (j : Bool) (v : V) (u : U) (γ p pj sj ε : ℝ)
  28     (hγ : 0<γ) (hγ1 : γ≤1) (hp : 0<p) (hp1 : p≤1) (hpj : p≤pj) (hsj : p≤sj)
  29     (hloss : 2*(2*(b:ℝ)+1)≤γ*A.card/2) (hε : ε≤p^3*γ/4)
  30     (hdeg : ∀ x,(1/2+γ/2)*A.card≤((A∩P.neighborFinset x).card:ℝ))
  31     (hcount : pj*sj*(markedRowDomain P π b A (markedOneExtensionRequests j v) ∅).card-ε*A.card≤
  32       markedRelationRowCount R H (markedRowDomain P π b A (markedOneExtensionRequests j v) ∅)
  33         (markedOneExtensionRequests j v) {(j,u)}) :
  34     (p^3*γ/4)*A.card≤((markedOneExtensionSet R H A j v u).card:ℝ) := by
  35   have ht : p^3≤pj*sj := by
  36     calc
  37       _ ≤ p^2 := pow_le_pow_of_le_one hp.le hp1 (by decide)
  38       _ = p*p := by ring
  39       _ ≤ _ := mul_le_mul hpj hsj hp.le (hp.le.trans hpj)
  40   have hd:=marked_one_extension_domain_lower P π b A hA j v γ hγ1 hdeg
  41   have hh:=marked_extension_target_lower A.card _ _ _ γ p ε (2*(b:ℝ)+1)
  42     (Nat.cast_nonneg _) hγ hp hd (by linarith only [hloss,Nat.cast_nonneg (α:=ℝ) b]) ht hε hcount
  43   exact hh.trans (markedRelationRowCount_one_le R H A _ (Finset.filter_subset _ _) j v u)

  46     (R : (V×V)⊕(V×V)→Prop) (H : Bool→V→U→Prop) (A : Finset V)
  47     (j : Bool) (v w : V) (u : U) (hne : v≠if j then π w else w) (γ p p₀ pj s₀ ε : ℝ)
  48     (hγ : 0<γ) (hγ1 : γ≤1) (hp : 0<p) (hp₀ : p≤p₀) (hpj : p≤pj) (hs₀ : p≤s₀)
  49     (hloss : 2*(2*(b:ℝ)+1)≤γ*A.card/2) (hε : ε≤p^3*γ/4)

  51     (hcount : p₀*pj*s₀*(markedRowDomain P π b A (markedTwoExtensionRequests j v w) ∅).card-ε*A.card≤
  52       markedRelationRowCount R H (markedRowDomain P π b A (markedTwoExtensionRequests j v w) ∅)
  53         (markedTwoExtensionRequests j v w) {(false,u)}) :
  54     (p^3*γ/4)*A.card≤((markedTwoExtensionSet R H A j v w u).card:ℝ) := by

  64 theorem marked_extension_index_lower (H : Bool→V→U→Prop) (B : Finset U) (X Y : Finset V)
  65     (c : ℕ) (p s t ε : ℝ) (hp : 0<p) (hp1 : p≤1) (hs : p≤s) (ht : p≤t)
  66     (hXY : X.card+Y.card≤c+6) (hε : ε≤p^(c+6)/2)
  67     (hcount : (s^X.card*t^Y.card-ε)*B.card≤
  68       ((common (common B X (fun v=>Finset.univ.filter (H false v))) Y
  69         (fun v=>Finset.univ.filter (H true v))).card:ℝ)) :
  70     (p^(c+6)/2)*B.card≤((common (common B X (fun v=>Finset.univ.filter (H false v))) Y
  71         (fun v=>Finset.univ.filter (H true v))).card:ℝ) := by
  72   have hprod : p^(c+6)≤s^X.card*t^Y.card := by
  73     calc
  74       _ ≤ p^(X.card+Y.card) := pow_le_pow_of_le_one hp.le hp1 hXY
  75       _ = p^X.card*p^Y.card := pow_add _ _ _
  76       _ ≤ _ := mul_le_mul (pow_le_pow_left₀ hp.le hs _) (pow_le_pow_left₀ hp.le ht _)
  77         (pow_nonneg hp.le _) (pow_nonneg (hp.le.trans hs) _)
  78   exact (mul_le_mul_of_nonneg_right (by linarith only [hprod,hε]) (Nat.cast_nonneg B.card)).trans hcount
