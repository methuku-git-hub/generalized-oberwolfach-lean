  19 theorem ordinaryResourceRows_card (L : Finset (Sym2 V⊕(V×U))) :
  20     (ordinaryResourceRows L).card≤2*L.card := by
  21   apply Finset.card_biUnion_le.trans
  22   calc
  23     _ ≤ ∑ _e∈L,2 := by

  25       intro e _

  40 theorem ordinary_transformer_forbidden_edges (K L : Finset (Sym2 V⊕(V×U)))
  41     (R W : Finset V) (B : Finset U)
  42     (hrows : ordinaryResourceRows L⊆W) (hhubs : ordinaryResourceHubs L⊆B)
  43     (havoid : ∀ v∈W,∀ w,Sum.inl s(v,w)∈K→v∈R)
  44     (hspoke : ∀ v∈W,∀ u,Sum.inr (v,u)∈K→v∈R)
  45     (hRR : ∀ v∈R,∀ w∈R,Sum.inl s(v,w)∉K)
  46     (hRB : ∀ v∈R,∀ u∈B,Sum.inr (v,u)∉K) : Disjoint K L := by
  47   apply Finset.disjoint_left.mpr
  48   intro e he hL
  49   cases e with
  50   | inl e=>
  51     induction e using Sym2.ind with
  52     | _ v w=>
  53       have hv : v∈W := hrows (Finset.mem_biUnion.mpr ⟨.inl s(v,w),hL,by simp⟩)
  54       have hw : w∈W := hrows (Finset.mem_biUnion.mpr ⟨.inl s(v,w),hL,by simp⟩)
  55       have hev : Sum.inl s(w,v)∈K := by simpa only [Sym2.eq_swap] using he
  56       exact hRR v (havoid v hv w he) w (havoid w hw v hev) he
  57   | inr p=>
  58     rcases p with ⟨v,u⟩
  59     have hv : v∈W := hrows (Finset.mem_biUnion.mpr ⟨.inr (v,u),hL,by simp⟩)
  60     have hu : u∈B := hhubs (Finset.mem_biUnion.mpr ⟨.inr (v,u),hL,by simp⟩)
  61     exact hRB v (hspoke v hv u he) u hu he
