  14 theorem physical_bipartite_row_count (H:V→U→Prop) (D:SimpleGraph (V⊕U)) (B:Finset U) (v:V) :
  15     ((B.image Sum.inr)∩((ordinaryPhysicalGraph ⊥ H)\D).neighborFinset (Sum.inl v)).card=
  16       (B.filter (fun u=>H v u∧¬D.Adj (Sum.inl v) (Sum.inr u))).card := by
  17   have he : (B.image Sum.inr)∩((ordinaryPhysicalGraph ⊥ H)\D).neighborFinset (Sum.inl v)=
  18       (B.filter (fun u=>H v u∧¬D.Adj (Sum.inl v) (Sum.inr u))).image Sum.inr := by
  19     ext z
  20     cases z <;> simp [ordinaryPhysicalGraph]
  21   rw [he,Finset.card_image_of_injective _ Sum.inr_injective]

  52     (A:Finset V) (B:Finset U) (d tau lam:ℝ)
  53     (hr:∀ v∈A,d≤(((B.image Sum.inr)∩((ordinaryPhysicalGraph ⊥ H)\D).neighborFinset (Sum.inl v)).card:ℝ))
  54     (hc:∀ u∈B,d≤(((A.image Sum.inl)∩((ordinaryPhysicalGraph ⊥ H)\D).neighborFinset (Sum.inr u)).card:ℝ))

  56       lam*S.card*T.card≤(bipartiteMatrixCut (fun z w=>if ((ordinaryPhysicalGraph ⊥ H)\D).Adj z w then 1 else 0) S T:ℝ)) :
  57     BipartiteBlockBounds (fun v u=>H v u∧¬D.Adj (Sum.inl v) (Sum.inr u))
  58       (fun v=>v∈A) (fun u=>u∈B) d tau lam := by
