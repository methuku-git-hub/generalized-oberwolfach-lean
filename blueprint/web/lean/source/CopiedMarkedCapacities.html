  13 def copiedMarkedCapacity {V E K : Type*} [Fintype E] [Fintype K] [DecidableEq V]
  14     (tail head : E → V) (ktail khead : K → V) (m : ℕ)
  15     (f : Fin m → E → ℝ) (wK : K → ℝ) : Fin (m+1) → V → V → ℝ :=
  16   Fin.snoc (fun j => occurrenceCapacityMatrix tail head (f j))
  17     (occurrenceCapacityMatrix ktail khead wK)

  21     (f : Fin m → E → ℝ) (wK : K → ℝ) (j : Fin m) (x y : V) :

  37 theorem copiedMarkedCapacity_balance {V E K : Type*} [Fintype V] [Fintype E] [Fintype K] [DecidableEq V]
  38     (tail head : E → V) (ktail khead : K → V) (m : ℕ)
  39     (f : Fin m → E → ℝ) (wK : K → ℝ) (d : Fin (m+1) → V → ℝ)
  40     (hfr : ∀ j v,occurrenceSumLinear tail (f j) v=d j.castSucc v)
  41     (hfc : ∀ j v,occurrenceSumLinear head (f j) v=d j.succ v)
  42     (hkr : ∀ v,occurrenceSumLinear ktail wK v=d (Fin.last m) v)
  43     (hkc : ∀ v,occurrenceSumLinear khead wK v=d 0 v) :
  44     (∀ i x,(∑ y,copiedMarkedCapacity tail head ktail khead m f wK i x y)=d i x) ∧
  45     (∀ i y,(∑ x,copiedMarkedCapacity tail head ktail khead m f wK i x y)=d (finRotate (m+1) i) y) := by
  46   constructor
  47   · intro i x
  48     refine Fin.lastCases ?_ (fun j => ?_) i
  49     · simp only [copiedMarkedCapacity_marked,occurrenceCapacityMatrix_row,hkr]
  50     · simp only [copiedMarkedCapacity_ordinary,occurrenceCapacityMatrix_row,hfr]
  51   · intro i y
  52     refine Fin.lastCases ?_ (fun j => ?_) i
  53     · simp only [copiedMarkedCapacity_marked,occurrenceCapacityMatrix_col,finRotate_last,hkc]
  54     · simp only [copiedMarkedCapacity_ordinary,occurrenceCapacityMatrix_col,finRotate_castSucc_eq_succ,hfc]

  56 theorem occurrenceCapacityMatrix_pos_iff {V E : Type*} [Fintype E] [DecidableEq V]
  57     (tail head : E → V) (hinj : Function.Injective (fun e => (tail e,head e)))
  58     (w : E → ℝ) (hw : ∀ e,0<w e) (x y : V) :
  59     0<occurrenceCapacityMatrix tail head w x y ↔ ∃ e,tail e=x ∧ head e=y := by
  60   constructor
  61   · intro h
  62     by_contra hn
  63     rw [occurrenceCapacityMatrix_zero tail head w x y hn] at h
  64     exact (lt_irrefl _ h)
  65   · rintro ⟨e,rfl,rfl⟩
  66     rw [occurrenceCapacityMatrix_at_edge tail head hinj w]
  67     exact hw e

  69 theorem copiedMarkedCapacity_bounds {V E K : Type*} [Fintype E] [Fintype K] [DecidableEq V]
  70     (tail head : E → V) (ktail khead : K → V)
  71     (hinj : Function.Injective (fun e => (tail e,head e)))
  72     (hkinj : Function.Injective (fun e => (ktail e,khead e)))
  73     (m : ℕ) (hm : 1≤m) (f : Fin m → E → ℝ) (wK : K → ℝ)
  74     (hf : ∀ j e,1/(2*(m:ℝ))≤f j e ∧ f j e≤2/(m:ℝ))
  75     (hwK : ∀ e,1/2≤wK e ∧ wK e≤3/2) :
  76     (∀ i x y,0≤copiedMarkedCapacity tail head ktail khead m f wK i x y) ∧
  77     (∀ i x y,0<copiedMarkedCapacity tail head ktail khead m f wK i x y →
  78       1/(2*(m:ℝ))≤copiedMarkedCapacity tail head ktail khead m f wK i x y ∧
  79       copiedMarkedCapacity tail head ktail khead m f wK i x y≤2) := by
  80   have hmr : (1:ℝ)≤m := by exact_mod_cast hm
  81   have hmpos : (0:ℝ)<m := by linarith
  82   constructor
  83   · intro i x y
  84     refine Fin.lastCases ?_ (fun j => ?_) i
  85     · simp only [copiedMarkedCapacity_marked]
  86       exact occurrenceCapacityMatrix_nonneg ktail khead wK (fun e => by linarith [(hwK e).1]) x y
  87     · simp only [copiedMarkedCapacity_ordinary]
  88       exact occurrenceCapacityMatrix_nonneg tail head (f _) (fun e =>
  89         (by positivity : (0:ℝ)≤1/(2*m)).trans (hf _ e).1) x y
  90   · intro i x y
  91     refine Fin.lastCases ?_ (fun j => ?_) i
  92     · simp only [copiedMarkedCapacity_marked]
  93       intro hpos
  94       obtain ⟨hl,hu⟩ := occurrenceCapacityMatrix_positive_bounds ktail khead hkinj wK (1/2) (3/2) hwK x y hpos
  95       constructor
  96       · apply le_trans _ hl
  97         apply (div_le_iff₀ (by positivity)).mpr
  98         linarith
  99       · linarith
 100     · simp only [copiedMarkedCapacity_ordinary]
 101       intro hpos
 102       obtain ⟨hl,hu⟩ := occurrenceCapacityMatrix_positive_bounds tail head hinj (f j) (1/(2*m)) (2/m) (hf j) x y hpos
 103       refine ⟨hl,hu.trans ?_⟩
 104       apply (div_le_iff₀ hmpos).mpr
