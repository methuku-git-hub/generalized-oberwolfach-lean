  14 theorem positive_ordinary_wheel_weights (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  15     ∃ N : ℕ,∀ k : ℕ,N≤k → ∀ θ γ bminus bplus : ℝ,
  16       max (1/2) (cycleThreshold r)≤θ → 0<γ → 0<bminus → 0<bplus →
  17       ∃ η₀ : ℝ,0<η₀ ∧
  18       ∀ (V U : Type) [Fintype V] [DecidableEq V] [Fintype U] [DecidableEq U],
  19       0<Fintype.card V → Fintype.card U≤Fintype.card V →
  20       ∀ (P Q F : SimpleGraph V) (H : V → U → Prop),Q≤P\F →
  21       (∀ v,Q.degree v=2*(Finset.univ.filter (H v)).card) →
  22       (∀ v,γ*Fintype.card V≤((Finset.univ.filter (H v)).card:ℝ)) →
  23       (∀ x y,Q.Adj x y → γ*(Fintype.card V:ℝ)^(k-1)≤
  24         (Fintype.card (OrdinaryIsolatorConfiguration P Q F H k θ (α/4) x y):ℝ)) →
  25       (∀ v (i j : {u // H v u}),i.val≠j.val →
  26         γ*(Fintype.card V:ℝ)^(r^2)≤
  27           (Fintype.card (OrdinaryTransferConfiguration hr Q F H v i.val j.val):ℝ)) →
  28       ∀ η : ℝ,0≤η → η≤η₀ → ∀ ψ : OrdinaryWheel Q F H r → ℝ,
  29       (∀ D,bminus/(Fintype.card V:ℝ)^(r-1)≤ψ D ∧
  30         ψ D≤bplus/(Fintype.card V:ℝ)^(r-1)) →
  31       (∀ x y,Q.Adj x y → |ordinaryRimLoad ψ x y-1|≤η) →
  32       (∀ v u,H v u → |ordinarySpokeLoad ψ v u-1|≤η) →
  33       ∃ ω : OrdinaryWheel Q F H r → ℝ,
  34         (∀ x y,ordinaryRimLoad ω x y=if Q.Adj x y then 1 else 0) ∧
  35         (∀ v u,ordinarySpokeLoad ω v u=if H v u then 1 else 0) ∧
  36         (∀ D,(bminus/2)/(Fintype.card V:ℝ)^(r-1)≤ω D ∧
  37           ω D≤(bplus+bminus/2)/(Fintype.card V:ℝ)^(r-1)) := by
  38   obtain ⟨N,hN⟩ := ordinary_wheel_correction r hr (α/4) (by positivity)
  39   refine ⟨N,?_⟩
  40   intro k hk θ γ bminus bplus hθ hγ hbm _hbp
  41   let C := ordinaryWheelCorrectionConstant r γ
  42   have hC : 0<C := ordinaryWheelCorrectionConstant_pos r γ hγ
  43   refine ⟨bminus/(2*C),by positivity,?_⟩
  44   intro V U _ _ _ _ hn _hU P Q F H hQP hdeg _hH hcount hT η hη hη₀ ψ hψ hR hS
  45   have hRerr (x y : V) : |ordinaryRimResidual ψ x y|≤η := by
  46     by_cases hxy : Q.Adj x y
  47     · simp only [ordinaryRimResidual,if_pos hxy,abs_sub_comm]
  48       exact hR x y hxy
  49     · rw [ordinaryRimResidual_outside ψ x y hxy,abs_zero]
  50       exact hη
  51   have hSerr (v : V) (u : U) : |ordinarySpokeResidual ψ v u|≤η := by
  52     by_cases hvu : H v u
  53     · simp only [ordinarySpokeResidual,if_pos hvu,abs_sub_comm]
  54       exact hS v u hvu
  55     · rw [ordinarySpokeResidual_outside ψ v u hvu,abs_zero]
  56       exact hη
  57   obtain ⟨ω,hωR,hωS,hωb⟩ := hN V U P Q F H k hk θ ((le_max_right _ _).trans hθ)
  58     γ η hγ hη hn (hQP.trans sdiff_le) hdeg hcount hT ψ hRerr hSerr
  59   have hsmall : C*η≤bminus/2 := by
  60     have hh := (le_div_iff₀ (show 0<2*C by positivity)).mp hη₀
  61     nlinarith
  62   have hn' : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr hn
  63   refine ⟨ω,hωR,hωS,?_⟩
  64   intro D
  65   have hdiff : |ω D-ψ D|≤(bminus/2)/(Fintype.card V:ℝ)^(r-1) :=
  66     (hωb D).trans (div_le_div_of_nonneg_right hsmall (by positivity))
  67   have hh := abs_le.mp hdiff
  68   have hlow := (hψ D).1
  69   have hhigh := (hψ D).2
  70   constructor
  71   · have he : bminus/(Fintype.card V:ℝ)^(r-1)=
  72         2*((bminus/2)/(Fintype.card V:ℝ)^(r-1)) := by ring
  73     linarith
  74   · rw [add_div]
  75     linarith
