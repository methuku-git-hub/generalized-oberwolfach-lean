   9 theorem source_piece_list_lift_to_vertex_subtype (R:Finset V):
  10     ∀ ps:List (PrescribedSourcePiece V),
  11       (∀p∈ps,p.vertices.toFinset⊆R)→
  12       ∃ qs:List (PrescribedSourcePiece {x // x∈R}),
  13         qs.map (fun q=>q.map Subtype.val)=ps := by
  14   intro ps
  15   induction ps with
  16   | nil => intro _;exact ⟨[],rfl⟩
  17   | cons p ps ih =>
  18     intro hR
  19     have hp:p.vertices.toFinset⊆R:=hR p (by simp)
  20     have htail:∀q∈ps,q.vertices.toFinset⊆R:=by
  21       intro q hq;exact hR q (by simp [hq])
  22     obtain ⟨q,hq⟩:=source_piece_lift_to_vertex_subtype R p hp
  23     obtain ⟨qs,hqs⟩:=ih htail
  24     exact ⟨q::qs,by simp [hq,hqs]⟩
  25 end Oberwolfach
