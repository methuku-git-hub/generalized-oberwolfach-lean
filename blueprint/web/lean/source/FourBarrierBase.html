  12 def fourBarrierBase (t : ℕ) : SimpleGraph (Fin ((t + t) + (t - 1))) :=

  15 def fourBarrierA (t : ℕ) : Fin t ↪ Fin ((t + t) + (t - 1)) :=
  16   (sumLeftEmbedding t t).trans (sumLeftEmbedding (t + t) (t - 1))

  18 def fourBarrierC (t : ℕ) : Fin t ↪ Fin ((t + t) + (t - 1)) :=

  23 def fourBarrierColour (t : ℕ) (v : Fin ((t + t) + (t - 1))) : Fin 3 :=

  28     ((finSumFinEquiv : Fin (t + t) ⊕ Fin (t - 1) ≃ Fin ((t + t) + (t - 1))).symm v)

  42 theorem fourBarrierColour_zero_iff (t : ℕ) (v : Fin ((t + t) + (t - 1))) :
  43     fourBarrierColour t v = 0 ↔ ∃ a, fourBarrierA t a = v := by
  44   constructor
  45   · intro hv
  46     obtain ⟨z, rfl⟩ := (finSumFinEquiv :
  47       Fin (t + t) ⊕ Fin (t - 1) ≃ Fin ((t + t) + (t - 1))).surjective v
  48     cases z with
  49     | inr b =>
  50       simp only [fourBarrierColour, Equiv.symm_apply_apply, Sum.elim_inr] at hv
  51       norm_num at hv
  52     | inl z =>
  53       obtain ⟨z, rfl⟩ := (finSumFinEquiv : Fin t ⊕ Fin t ≃ Fin (t + t)).surjective z
  54       cases z with
  55       | inr c =>
  56         simp only [fourBarrierColour, Equiv.symm_apply_apply, Sum.elim_inl, Sum.elim_inr] at hv
  57         norm_num at hv
  58       | inl a => exact ⟨a, rfl⟩

  60     exact fourBarrierColour_A t a

 110 theorem fourBarrierBase_internal_edges (t : ℕ) :
 111     (fourBarrierBase t).edgeFinset.filter (colourInternalEdge (fourBarrierColour t)) =
 112       ((⊤ : SimpleGraph (Fin t)).map (fourBarrierA t)).edgeFinset := by
 113   ext e
 114   induction e using Sym2.ind with
 115   | _ u v =>
 116     simp only [Finset.mem_filter, SimpleGraph.mem_edgeFinset, SimpleGraph.mem_edgeSet,

 118     constructor
 119     · rintro ⟨huv, hu, hv⟩
