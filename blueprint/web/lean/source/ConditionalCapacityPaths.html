   9 theorem mixed_capacity_transition_scale {X : Type*} [Fintype X]
  10     (d : ℕ → X → ℝ) (M n d₀ d₁ : ℝ)
  11     (hn : (Fintype.card X : ℝ)=n) (hnpos : 0<n) (hd₀ : 0<d₀) (hd₁ : 0<d₁)
  12     (hms : ∀ i, ∑ x, d i x=M)
  13     (hd : ∀ i x, d₀*n ≤ d i x ∧ d i x ≤ d₁*n)
  14     (i : ℕ) (z : X) (T : ℝ)
  15     (hmix : |T-capacityProfile d M i z| ≤ (1/10)*capacityProfile d M i z) :
  16     ((9/10)*(d₀/d₁))/n ≤ T ∧ T ≤ ((11/10)*(d₁/d₀))/n := by
  17   obtain ⟨hlo,hup⟩ := capacity_profile_scale (d i) n M d₀ d₁ hn hnpos hd₀ hd₁ (hms i) (hd i) z
  18   obtain ⟨hml,hmu⟩ := abs_le.mp hmix
  19   change -((1/10)*(d i z/M)) ≤ T-d i z/M at hml
  20   change T-d i z/M ≤ (1/10)*(d i z/M) at hmu
  21   simp only [mul_div_assoc]
  22   constructor <;> linarith

  24 theorem capacity_transition_scale {X : Type*}
  25     (a : ℕ → X → X → ℝ) (d : ℕ → X → ℝ)
  26     (n a₀ a₁ d₀ d₁ : ℝ) (hn : 0<n) (ha₀ : 0<a₀) (hd₀ : 0<d₀)
  27     (j : ℕ) (x y : X)
  28     (ha : a₀ ≤ a j x y ∧ a j x y ≤ a₁)
  29     (hd : d₀*n ≤ d j x ∧ d j x ≤ d₁*n) :
  30     a₀/(d₁*n) ≤ capacityTransition a d j x y ∧
  31       capacityTransition a d j x y ≤ a₁/(d₀*n) := by
  32   have hdp : 0<d j x := (mul_pos hd₀ hn).trans_le hd.1
  33   exact ⟨div_le_div₀ (ha₀.le.trans ha.1) ha.1 hdp hd.2,
  34     div_le_div₀ (ha₀.le.trans (ha.1.trans ha.2)) ha.2 (mul_pos hd₀ hn) hd.1⟩

  36 theorem conditional_capacity_path_bounds {X : Type*} [Fintype X]
  37     (a : ℕ → X → X → ℝ) (d : ℕ → X → ℝ)
  38     (M n a₀ a₁ d₀ d₁ : ℝ)
  39     (hn : (Fintype.card X : ℝ)=n) (hnpos : 0<n)
  40     (ha₀ : 0<a₀) (ha₁ : 0<a₁) (hd₀ : 0<d₀) (hd₁ : 0<d₁)
  41     (hms : ∀ i, ∑ x, d i x=M)
  42     (hd : ∀ i x, d₀*n ≤ d i x ∧ d i x ≤ d₁*n)
  43     (i k : ℕ) (hk : 0<k) (p : Fin (k+1) → X)
  44     (ha : ∀ j : Fin k, a₀ ≤ a (i+j.val) (p j.castSucc) (p j.succ) ∧
  45       a (i+j.val) (p j.castSucc) (p j.succ) ≤ a₁)
  46     (hmix : |transitionWalk (capacityTransition a d) i k (p 0) (p (Fin.last k))-
  47       capacityProfile d M (i+k) (p (Fin.last k))| ≤
  48       (1/10)*capacityProfile d M (i+k) (p (Fin.last k))) :
  49     conditionalPathLower a₀ d₀ d₁ k/n^(k-1) ≤
  50       transitionPathWeight (capacityTransition a d) i k p /
  51         transitionWalk (capacityTransition a d) i k (p 0) (p (Fin.last k)) ∧
  52     transitionPathWeight (capacityTransition a d) i k p /
  53         transitionWalk (capacityTransition a d) i k (p 0) (p (Fin.last k)) ≤
  54       conditionalPathUpper a₁ d₀ d₁ k/n^(k-1) := by
  55   exact conditional_path_scale_bounds (capacityTransition a d) i k p n a₀ a₁ d₀ d₁ _
  56     hnpos hk ha₀ ha₁ hd₀ hd₁
  57     (fun j => capacity_transition_scale a d n a₀ a₁ d₀ d₁ hnpos ha₀ hd₀ _ _ _ (ha j) (hd _ _))
  58     (mixed_capacity_transition_scale d M n d₀ d₁ hn hnpos hd₀ hd₁ hms hd _ _ _ hmix)
