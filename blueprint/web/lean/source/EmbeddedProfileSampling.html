  18 def profileFreePreimage (e : (Σ i,V i) ↪ W) (Z A : Finset W) : Finset (Σ i,V i) :=
  19   Finset.univ.filter (fun x => e x∈A ∧ e x∉Z)

  21 theorem embedded_profile_subset_iff (e : (Σ i,V i) ↪ W) (Z A : Finset W)
  22     (hA : A⊆Z∪Finset.univ.map e) {m : I → ℕ} (σ : ∀ i,FixedSubset (V i) (m i)) :
  23     A⊆Z∪(profileRootSet σ).map e ↔ profileFreePreimage e Z A⊆profileRootSet σ := by
  24   constructor
  25   · intro h x hx
  26     have hx' := (Finset.mem_filter.mp hx).2
  27     have hm := (Finset.mem_union.mp (h hx'.1)).resolve_left hx'.2
  28     obtain ⟨y,hy,he⟩ := Finset.mem_map.mp hm
  29     have he' := e.injective he
  30     subst y
  31     exact hy
  32   · intro h x hx
  33     by_cases hz : x∈Z
  34     · exact Finset.mem_union_left _ hz
  35     · obtain ⟨y,_,rfl⟩ := Finset.mem_map.mp ((Finset.mem_union.mp (hA hx)).resolve_left hz)
  36       exact Finset.mem_union_right _ (Finset.mem_map.mpr ⟨y,
  37         h (Finset.mem_filter.mpr ⟨Finset.mem_univ _,hx,hz⟩),rfl⟩)

  39 theorem embedded_profile_inclusion_probability (e : (Σ i,V i) ↪ W) (Z A : Finset W)
  40     (hA : A⊆Z∪Finset.univ.map e) (m : I → ℕ)
  41     [∀ i,Nonempty (FixedSubset (V i) (m i))]
  42     (hm : ∀ i,m i ≤ Fintype.card (V i)) :
  43     (FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (m i)))).probability
  44       (fun σ => A⊆Z∪(profileRootSet σ).map e)=
  45     ∏ i,((m i).choose (Finset.univ.filter (fun v : V i => e ⟨i,v⟩∈A ∧ e ⟨i,v⟩∉Z)).card:ℝ)/
  46       (Fintype.card (V i)).choose (Finset.univ.filter (fun v : V i => e ⟨i,v⟩∈A ∧ e ⟨i,v⟩∉Z)).card := by
  47   have hp : (fun σ : ∀ i,FixedSubset (V i) (m i) => A⊆Z∪(profileRootSet σ).map e)=
  48       (fun σ => profileFreePreimage e Z A⊆profileRootSet σ) :=
