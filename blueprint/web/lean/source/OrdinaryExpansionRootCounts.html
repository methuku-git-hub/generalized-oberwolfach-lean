  15 theorem actual_edge_endpoint_count (G : SimpleGraph V) (a : V) :
  16     (Finset.univ.filter (fun e : G.edgeFinset => a=e.val.out.1)).card+
  17     (Finset.univ.filter (fun e : G.edgeFinset => a=e.val.out.2)).card=G.degree a := by
  18   rw [card_filter_naturalIndicator,card_filter_naturalIndicator,←Finset.sum_add_distrib]
  19   have hs : (∑ e : G.edgeFinset,naturalIndicator (a∈e.val))=G.degree a := by
  20     rw [Finset.sum_coe_sort G.edgeFinset (fun e : Sym2 V => naturalIndicator (a∈e))]
  21     calc
  22       _ = ∑ e∈G.edgeFinset,if a∈e then 1 else 0 := by
  23         apply Finset.sum_congr rfl
  24         intro e _
  25         by_cases h : a∈e <;> simp [naturalIndicator,h]
  26       _ = (G.edgeFinset.filter (fun e => a∈e)).card := by
  27         rw [Finset.card_eq_sum_ones,Finset.sum_filter]
  28       _ = G.degree a := by rw [←G.incidenceFinset_eq_filter,G.card_incidenceFinset_eq_degree]
  29   rw [←hs]
  30   apply Finset.sum_congr rfl
  31   intro e _
  32   have he : s(e.val.out.1,e.val.out.2)=e.val := by rw [Sym2.mk];exact e.val.out_eq
  33   have hm : a∈e.val ↔ a=e.val.out.1 ∨ a=e.val.out.2 := by
  34     calc
  35       (a∈e.val) ↔ a∈s(e.val.out.1,e.val.out.2) := by rw [he]
  36       _ ↔ _ := Sym2.mem_iff
  37   have hn := (edge_out_adj G e).ne
  38   by_cases h1 : a=e.val.out.1 <;> by_cases h2 : a=e.val.out.2
  39   all_goals simp_all [naturalIndicator]

  41 theorem actual_spoke_row_count (H : V → U → Prop) (a : V) :
  42     (Finset.univ.filter (fun j : {e : V×U // H e.1 e.2} => a=j.val.1)).card=
  43       (Finset.univ.filter (H a)).card := by
  44   let e : {j : {e : V×U // H e.1 e.2} // a=j.val.1} ≃ {b : U // H a b} := {
  45     toFun := fun j => ⟨j.val.val.2,by have h:=j.val.property;rw [←j.property] at h;exact h⟩
  46     invFun := fun b => ⟨⟨(a,b.val),b.property⟩,rfl⟩
  47     left_inv := by intro j;apply Subtype.ext;apply Subtype.ext;exact Prod.ext j.property rfl
  48     right_inv := by intro b;rfl }
  49   simpa only [Fintype.card_subtype] using Fintype.card_congr e
