  11 theorem localScale_exponential_tail (a n : ℕ) (ha : 0<a) (han : a≤n)
  12     (K β D : ℝ) (hK : 0<K) (hβ : 0<β) (hD : 0<D) (hnK : (n:ℝ)≤K*a) (k : ℕ) :
  13     2*Real.exp (-2*(β*((a:ℝ)/n)^k)^2/(D/n))≤
  14       2*Real.exp (-(2*(β/K^k)^2/D)*a) := by
  15   have haR : (0:ℝ)<a := by exact_mod_cast ha
  16   have hnR : (0:ℝ)<n := by exact_mod_cast ha.trans_le han
  17   have hanR : (a:ℝ)≤n := by exact_mod_cast han
  18   have hratio : 1/K≤(a:ℝ)/n := (div_le_div_iff₀ hK hnR).mpr (by simpa only [one_mul,mul_one,mul_comm] using hnK)
  19   have ht : β/K^k≤β*((a:ℝ)/n)^k := by
  20     have hh := mul_le_mul_of_nonneg_left (pow_le_pow_left₀ (by positivity) hratio k) hβ.le
  21     simpa only [div_pow,one_pow,mul_one_div] using hh
  22   have hcmp : 2*(β/K^k)^2*(a:ℝ)/D≤2*(β*((a:ℝ)/n)^k)^2*n/D := by
  23     gcongr
  24   apply mul_le_mul_of_nonneg_left (Real.exp_le_exp.mpr ?_) (by norm_num)
  25   have he : -2*(β*((a:ℝ)/n)^k)^2/(D/n)=-(2*(β*((a:ℝ)/n)^k)^2*n/D) := by
  26     field_simp
  27   rw [he]
  28   exact (neg_le_neg hcmp).trans (le_of_eq (by ring))
  29 
  30 end Oberwolfach
