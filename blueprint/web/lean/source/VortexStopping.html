   9 theorem floorOrbit_stopping {ε : ℝ} (he : 0 < ε) (he1 : ε < 1)
  10     {n m : ℕ} (hm : 0 < m) (hmn : m ≤ n) (hlarge : 2 ≤ ε * m) :
  11     ∃ ℓ : ℕ, 0 < ℓ ∧ floorOrbit ε n ℓ < m ∧
  12       (∀ i < ℓ, m ≤ floorOrbit ε n i) ∧
  13       (∀ i ≤ ℓ, ε * m / 2 ≤ (floorOrbit ε n i : ℝ)) := by
  14   have hex : ∃ i : ℕ, floorOrbit ε n i < m :=
  15     ⟨n, by rw [floorOrbit_eventually_zero he1]; exact hm⟩
  16   let ℓ := Nat.find hex
  17   have hlast : floorOrbit ε n ℓ < m := Nat.find_spec hex
  18   have hbefore : ∀ i < ℓ, m ≤ floorOrbit ε n i := by
  19     intro i hi
  20     exact Nat.le_of_not_gt (Nat.find_min hex hi)
  21   have hlpos : 0 < ℓ := by
  22     by_contra hh
  23     have hz : ℓ = 0 := by omega
  24     rw [hz, floorOrbit_zero] at hlast
  25     omega
  26   have hprev := hbefore (ℓ - 1) (by omega)
  27   have hprevReal : (m : ℝ) ≤ floorOrbit ε n (ℓ - 1) := by exact_mod_cast hprev
  28   have hmstep := mul_le_mul_of_nonneg_left hprevReal he.le
  29   have hsuc : ℓ - 1 + 1 = ℓ := by omega
  30   have hround := Nat.lt_floor_add_one (ε * (floorOrbit ε n (ℓ - 1) : ℝ))
  31   rw [← floorOrbit_succ, hsuc] at hround
  32   have hterm : ε * m / 2 ≤ (floorOrbit ε n ℓ : ℝ) := by linarith
  33   refine ⟨ℓ, hlpos, hlast, hbefore, ?_⟩
  34   intro i hi
  35   have hh : floorOrbit ε n ℓ ≤ floorOrbit ε n i := floorOrbit_antitone he1 n hi
  36   exact hterm.trans (by exact_mod_cast hh)
