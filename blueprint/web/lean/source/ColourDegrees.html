  14 theorem incidentPair_injective
  15     {V : Type*} (x : V) : Function.Injective (fun y => s(x, y)) := by
  16   intro y z h
  17   rcases Sym2.eq_iff.mp h with ⟨_, he⟩ | ⟨h1, h2⟩
  18   · exact he
  19   · exact h2.trans h1

  21 theorem sampledColourOrientation_degree_patterns
  22     {V : Type*} [Fintype V] [LinearOrder V] {k : ℕ} (G : SimpleGraph V)
  23     (σ : Sym2 V → Fin k × Bool) (a : Fin k) (x : V) :
  24     ((∑ y, sampledColourOrientation G σ a x y : ℕ) : ℝ) =
  25       ∑ y ∈ G.neighborFinset x,
  26         patternValue {s(x, y)} (fun _ => (a, orientationBit x y)) σ := by
  27   have hn : G.neighborFinset x = Finset.univ.filter (G.Adj x) := by ext y; simp
  28   rw [hn, Finset.sum_filter]
  29   push_cast
  30   apply Finset.sum_congr rfl
  31   intro y _
  32   by_cases ha : G.Adj x y <;> simp [sampledColourOrientation, patternValue, ha]

  34 theorem sampledColourOrientation_degree_tail
  35     {V : Type*} [Fintype V] [LinearOrder V] {k : ℕ} (G : SimpleGraph V)
  36     (ρ : Fin k → ℝ) (hρ : ∀ a, 0 ≤ ρ a) (htotal : ∑ a, ρ a = 1)
  37     (a : Fin k) (x : V) {t : ℝ} (ht : 0 ≤ t) :
  38     (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.categoricalDirection ρ hρ htotal)).probability
  39       (fun σ => t ≤ |((∑ y, sampledColourOrientation G σ a x y : ℕ) : ℝ) -
  40         ρ a / 2 * G.degree x|) ≤
  41       2 * Real.exp (-2 * t ^ 2 / Fintype.card V) := by
  42   have hh := finiteLaw_singleton_patterns_tail_at_scale
  43     (fun _ : Sym2 V => FiniteLaw.categoricalDirection ρ hρ htotal)
  44     (G.neighborFinset x) (fun y => s(x, y)) (fun y _ => (a, orientationBit x y))
  45     (fun _ _ _ _ h => incidentPair_injective x h)
  46     (n := Fintype.card V) (by exact_mod_cast (G.degree_lt_card_verts x).le) ht
  47   have hmass (b : Bool) :
  48       (FiniteLaw.categoricalDirection ρ hρ htotal).mass (a, b) = ρ a / 2 := rfl
  49   simp only [hmass, Finset.sum_const, SimpleGraph.card_neighborFinset_eq_degree, nsmul_eq_mul,
  50     ← sampledColourOrientation_degree_patterns] at hh
  51   have heq : (G.degree x : ℝ) * (ρ a / 2) = ρ a / 2 * G.degree x := mul_comm _ _
  52   simpa only [heq] using hh

  94 /-- Undirected colour degree inside a fixed graph Z. The graph Z is
  95 chosen before sampling; no direction is tested. -/
  96 def sampledColourDegree {V : Type*} [Fintype V] {k : ℕ}
  97     (Z : SimpleGraph V) (c : Sym2 V → Fin k) (a : Fin k) (x : V) : ℕ :=
  98   ((Z.neighborFinset x).filter (fun y => c s(x, y) = a)).card
