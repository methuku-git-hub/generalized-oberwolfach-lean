  13 theorem prescribed_positions_from_lists (Q:SimpleGraph X) (D:X→X→Prop)
  14     (hdeg:∀x,Q.degree x≤2) (hrep:∀x y,Q.Adj x y→D x y ∨ D y x)
  15     (hDQ:∀x y,D x y→Q.Adj x y) (horiented:∀x y,D x y→¬D y x)
  16     (A:Finset X) (hA:∀x∈A,∀y∈A,¬Q.Adj x y) (f:X→V) (hinj:Set.InjOn f (↑A:Set X))
  17     (ms:List (V×V)) (zs:List V)
  18     (hM:(ms.flatMap prescribedPairVertices).Nodup) (hZ:zs.Nodup)
  19     (hMZ:List.Disjoint (ms.flatMap prescribedPairVertices) zs)
  20     (hOld:Disjoint (A.image f) (ms.flatMap prescribedPairVertices++zs).toFinset)
  21     (hedges:Fintype.card X≤Q.edgeFinset.card+A.card)
  22     (hsize:8*A.card+8*ms.length+4*zs.length+10≤Fintype.card X):
  23     ∃ (B:Finset X) (g:X→V),A⊆B ∧ Set.InjOn g (↑B:Set X) ∧ (∀x∈A,g x=f x) ∧
  24       B.image g=A.image f∪(ms.flatMap prescribedPairVertices).toFinset∪zs.toFinset ∧
  25       (prescribedInducedArcs D B).image (Prod.map g g)=ms.toFinset := by
  26   obtain ⟨es,vs,heslen,hvslen,he,hpair,hvs,hcross,hsingle,hnodup⟩:=
  27     prescribed_position_selection_original Q D hdeg hrep A ms.length zs.length hedges hsize
  28   have htarget:(ms.flatMap prescribedPairVertices++zs).Nodup:=List.nodup_append'.mpr ⟨hM,hZ,hMZ⟩
  29   have hdisS:∀ x∈es.flatMap prescribedPairVertices++vs,x∉A:=by
  30     intro x hx
  31     have hout:x∉prescribedClosedNeighbourhoods Q A:=by
  32       rcases List.mem_append.mp hx with hx|hx
  33       · obtain ⟨e,he',hx⟩:=List.mem_flatMap.mp hx
  34         simp only [prescribedPairVertices,List.mem_cons,List.not_mem_nil,or_false] at hx
  35         rcases hx with hx|hx
  36         · subst x;exact (he e he').2.2.1
  37         · subst x;exact (he e he').2.2.2
  38       · exact hvs x hx
  39     exact (prescribed_outside_closed_neighbourhood Q A x hout).1
  40   have hdisT:∀ y∈ms.flatMap prescribedPairVertices++zs,y∉A.image f:=by
  41     intro y hy hOldY
  42     exact (Finset.disjoint_left.mp hOld hOldY) (List.mem_toFinset.mpr hy)
  43   have hsel:∀e∈es,D e.1 e.2:=by intro e he';exact (he e he').1
  44   have he':∀e∈es,D e.1 e.2 ∧ e.1∉prescribedClosedNeighbourhoods Q A ∧ e.2∉prescribedClosedNeighbourhoods Q A:=by
  45     intro e he';exact ⟨(he e he').1,(he e he').2.2.1,(he e he').2.2.2⟩
  46   have honly:=prescribed_selected_induced_arcs Q D A hDQ horiented hA es vs he' hpair hvs hcross hsingle
  47   let B:=A∪(es.flatMap prescribedPairVertices++vs).toFinset
  48   obtain ⟨g,hginj,hgpres,hgimage,hgarcs⟩:=prescribed_position_pairing D A f hinj es ms vs zs
  49     heslen hvslen hnodup htarget hdisS hdisT hsel honly
  50   refine ⟨B,g,Finset.subset_union_left,hginj,hgpres,?_,hgarcs⟩
  51   simpa [B,List.toFinset_append,Finset.union_assoc] using hgimage
