  12 theorem linear_deviation_failure_eventually_small (C A ε : ℝ)
  13     (hA : 0<A) (hε : 0<ε) (K q : ℕ) :
  14     ∃ N : ℕ,∀ n : ℕ,N≤n → ∀ M D : ℝ,0≤M → M≤C*(n:ℝ)^K →
  15       0<D → D≤A*n →
  16       M*(2*Real.exp (-2*(ε*n)^2/D))≤1/(n:ℝ)^q := by
  17   obtain ⟨N,hN⟩ := eventually_stretchedExp_le_inv_pow (2*C) K
  18     (c:=2*ε^2/A) (p:=1) (by positivity) (by norm_num) q
  19   refine ⟨max N 1,?_⟩
  20   intro n hn M D hM hMb hD hDb
  21   have hn0 : (0:ℝ)<n := by exact_mod_cast (show 0<n by omega)
  22   have hex : -2*(ε*n)^2/D≤-(2*ε^2/A)*n := by
  23     apply (div_le_iff₀ hD).mpr
  24     have hmul := mul_le_mul_of_nonneg_left hDb (by positivity : 0≤2*ε^2/A*n)
  25     have heq : (2*ε^2/A*n)*(A*n)=2*(ε*n)^2 := by field_simp <;> ring
  26     rw [heq] at hmul
  27     nlinarith only [hmul]
  28   calc
  29     _ ≤ M*(2*Real.exp (-(2*ε^2/A)*n)) :=
  30       mul_le_mul_of_nonneg_left (mul_le_mul_of_nonneg_left (Real.exp_le_exp.mpr hex) (by norm_num)) hM
  31     _ ≤ (C*(n:ℝ)^K)*(2*Real.exp (-(2*ε^2/A)*n)) :=
  32       mul_le_mul_of_nonneg_right hMb (by positivity)
  33     _ = (2*C)*(n:ℝ)^K*Real.exp (-(2*ε^2/A)*n) := by ring
  34     _ ≤ _ := by simpa only [Real.rpow_natCast,Real.rpow_one] using hN n (by omega)
