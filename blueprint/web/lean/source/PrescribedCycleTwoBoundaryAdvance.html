  12 theorem prescribed_cycle_two_boundary_advance
  13     (QV:I→Finset X) (Q:I→SimpleGraph X) (A₀:I→Finset X) (ψ:I→X→V)
  14     (E₀:I→Finset (Sym2 V)) (J:SimpleGraph V)
  15     (β δ:ℝ) (hβ:0<β) (hβ1:β≤1) (hδ:0<δ) (hδbound:δ≤(β/100)^2)
  16     (hnβ:100/β≤(Fintype.card V:ℝ)) (hn:2/Real.sqrt δ≤(Fintype.card V:ℝ))
  17     (hI:(Fintype.card I:ℝ)≤Fintype.card V)
  18     (hQsize:∀ i,((QV i).card:ℝ)≤δ*Fintype.card V) (hQdegree:∀ i x,(Q i).degree x≤2)
  19     (hold:∀ v,((Finset.univ.filter (fun i=>∃ x,x∈A₀ i ∧ ψ i x=v)).card:ℝ)≤δ*Fintype.card V)
  20     (hJ:∀ v,(J.degree v:ℝ)≤δ*Fintype.card V)
  21     (k:I) (Dsrc:X→X→Prop) (Fsrc:SimpleGraph X) (hDQ:∀ x y,Dsrc x y→(Q k).Adj x y)
  22     (D:V→V→Prop) (L:Finset V) (hext:IsExtendingList D L β)
  23     (a b:X) (xs:List X) (hnodup:(a::xs).Nodup) (ha:a∈A₀ k) (hb:b∈A₀ k) (hbin:b∈xs)
  24     (hsourcecycle:DirectedCycleThrough Dsrc Fsrc (a::xs))
  25     (st:PrescribedFamilyState I X V)
  26     (hvalid:st.Valid QV Q A₀ ψ E₀ J (Real.sqrt δ*Fintype.card V))
  27     (hinit:prescribedInducedSourceEdges (Q k) (A₀ k)⊆st.sourceEdges k)
  28     (hfree:∀ x∈a::xs,x∉A₀ k→x∉st.assigned k) (hsubset:∀ x∈a::xs,x∈QV k):
  29     ∃ st':PrescribedFamilyState I X V,
  30       st'.Valid QV Q A₀ ψ E₀ J (Real.sqrt δ*Fintype.card V) ∧
  31       st'.assigned=Function.update st.assigned k (st.assigned k∪(a::xs).toFinset) ∧
  32       (∀ i,st.assigned i⊆st'.assigned i) ∧ (∀ i,st.sourceEdges i⊆st'.sourceEdges i) ∧
  33       (∀ i x,x∈st.assigned i→st'.vertexMap i x=st.vertexMap i x) ∧
  34       (∀ x∈a::xs,x∉A₀ k→st'.vertexMap k x∈L) ∧
  35       (∀ e∈prescribedDirectedPathEdges ((a::xs)++[a]),
  36         (e.1∈A₀ k ∧ e.2∈A₀ k) ∨ D (st'.vertexMap k e.1) (st'.vertexMap k e.2)) ∧
  37       (a::xs).toFinset⊆st'.assigned k ∧
  38       prescribedOrderedPathEdges ((a::xs)++[a])⊆st'.sourceEdges k := by

  42   let p2:List X:=b::post++[a]
  43   have hmem1:∀ x∈p1,x∈a::xs:=by
  44     intro x hx;simp [p1,hxs] at hx ⊢;tauto
  45   have hmem2:∀ x∈p2,x∈a::xs:=by
  46     intro x hx;simp [p2,hxs] at hx ⊢;tauto
  47   have hcover:∀ x∈a::xs,x∈p1 ∨ x∈p2:=by
  48     intro x hx;simp [p1,p2,hxs] at hx ⊢;tauto
  49   have hc:List.IsChain (fun x y=>Dsrc x y ∧ ¬Fsrc.Adj x y) ((a::xs)++[a]):=by

  52     apply hc.infix

  54     simp [p1,hxs,List.append_assoc]

  56     apply hc.infix
  57     refine ⟨a::pre,[],?_⟩
  58     simp [p2,hxs,List.append_assoc]

  60   have hs1:∀ x∈p1,x∈QV k:=by intro x hx;exact hsubset x (hmem1 x hx)
  61   obtain ⟨mid,hm,hmExact,hmA,hmE,hmPres,hmL,_,hmArcs,hmVertices,hmEdges⟩:=

  64   have hinitmid:prescribedInducedSourceEdges (Q k) (A₀ k)⊆mid.sourceEdges k:=hinit.trans (hmE k)
  65   have hf2:∀ x∈p2,x∉A₀ k→x∉mid.assigned k:=by

  67     rw [hmExact,Function.update_self] at hmid
  68     rcases Finset.mem_union.mp hmid with hold'|hnew
  69     · exact hfree x (hmem2 x hx) hnot hold'
  70     · have hh:=hinter x (List.mem_toFinset.mp hnew) hx
  71       rcases hh with rfl|rfl <;> contradiction
  72   have hs2:∀ x∈p2,x∈QV k:=by intro x hx;exact hsubset x (hmem2 x hx)
  73   obtain ⟨last,hl,hlExact,hlA,hlE,hlPres,hlL,_,hlArcs,hlVertices,hlEdges⟩:=

  76   have hunion:p1.toFinset∪p2.toFinset=(a::xs).toFinset:=by
  77     ext x
  78     simp only [Finset.mem_union,List.mem_toFinset]
  79     exact ⟨fun h=>h.elim (hmem1 x) (hmem2 x),hcover x⟩
  80   refine ⟨last,hl,?_,?_,?_,?_,?_,?_,?_,?_⟩
  81   · rw [hlExact,hmExact]
  82     funext i
  83     by_cases hi:i=k
  84     · subst i
  85       simp only [Function.update_self,Finset.union_assoc,hunion]
  86     · simp [Function.update_of_ne hi]
  87   · intro i;exact (hmA i).trans (hlA i)
  88   · intro i;exact (hmE i).trans (hlE i)
  89   · intro i x hx;exact (hlPres i x (hmA i hx)).trans (hmPres i x hx)
  90   · intro x hx hnot

  92     · rw [hlPres k x (hmVertices (List.mem_toFinset.mpr hx1))]
  93       exact hmL x hx1 hnot
  94     · exact hlL x hx2 hnot
  95   · intro e he
  96     rw [hdirected] at he
  97     rcases Finset.mem_union.mp he with he1|he2
  98     · rcases hmArcs e he1 with hboth|hnew
  99       · exact Or.inl hboth

 103             hlPres k e.2 (hmVertices (List.mem_toFinset.mpr hend.2))]
 104         exact hnew
 105     · exact hlArcs e he2
 106   · intro x hx
 107     rcases hcover x (List.mem_toFinset.mp hx) with hx1|hx2
 108     · exact hlA k (hmVertices (List.mem_toFinset.mpr hx1))
 109     · exact hlVertices (List.mem_toFinset.mpr hx2)
