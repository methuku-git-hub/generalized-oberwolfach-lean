  14 theorem graph_independent_insert (F:SimpleGraph V) (Z:Finset V) (a:V)
  15     (hZ:∀ v∈Z,∀ w∈Z,¬F.Adj v w) (ha:∀ v∈Z,¬F.Adj a v):
  16     ∀ v∈insert a Z,∀ w∈insert a Z,¬F.Adj v w:=by
  17   intro v hv w hw
  18   rcases Finset.mem_insert.mp hv with he|hvZ <;> rcases Finset.mem_insert.mp hw with he'|hwZ
  19   · rw [he,he'];exact F.loopless.irrefl _
  20   · rw [he];exact ha w hwZ
  21   · rw [he'];exact fun h=>ha v hvZ h.symm
  22   · exact hZ v hvZ w hwZ

  24 theorem transformer_rim_common_path (Q F:SimpleGraph V) (A W Ω Z:Finset V)
  25     (hAΩ:A⊆Ω) (x y x' y':V) (hroots:x∈Z∧y∈Z∧x'∈Z∧y'∈Z)
  26     (hΩ:x∈Ω∧y∈Ω∧x'∈Ω∧y'∈Ω)
  27     (hZ:∀ v∈Z,∀ w∈Z,¬F.Adj v w)
  28     (r:ℕ) (hr:4≤r) (L b:ℝ) (hL:0≤L) (hF:∀ v,(F.degree v:ℝ)≤L)
  29     (hcount:∀ v∈Ω,∀ w∈Ω,b≤((A∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  30     (hthree:r=4→∀ a∈A,b≤(((A∩Q.neighborFinset a)∩Q.neighborFinset y∩Q.neighborFinset y').card:ℝ))
  31     (hbudget:W.card+((Z.card:ℝ)+r+2)*(L+1)<b):
  32     ∃ a b',∃ p:Q.Walk a b',
  33       p.IsPath ∧ p.length=r-3 ∧ Q.Adj x a ∧ Q.Adj x' a ∧ Q.Adj y b' ∧ Q.Adj y' b' ∧
  34       (∀ v∈p.support,v∈A∧v∉W∧v∉Z) ∧
  35       (∀ v∈p.support,∀ w∈p.support,¬F.Adj v w) ∧
  36       (∀ v∈p.support,∀ w∈Z,¬F.Adj v w):=by
  37   have hL1:0≤L+1:=by linarith
  38   obtain ⟨a,haA,haW,haZ,hxa,hx'a,hFa⟩:=sparse_common_neighbour_avoiding Q F A W Z x x' L b
  39     hL hF (hcount x hΩ.1 x' hΩ.2.2.1) (by nlinarith only [hbudget,hL1,(show (0:ℝ)≤r from Nat.cast_nonneg r)])
  40   let A₁:=if r=4 then A∩Q.neighborFinset a else A
  41   have hcount₁:b≤((A₁∩Q.neighborFinset y∩Q.neighborFinset y').card:ℝ):=by
  42     dsimp only [A₁]
  43     split_ifs with hr4
  44     · exact hthree hr4 a haA
  45     · exact hcount y hΩ.2.1 y' hΩ.2.2.2
  46   have hZ₁:((insert a Z).card:ℝ)≤Z.card+1:=by exact_mod_cast Finset.card_insert_le a Z
  47   obtain ⟨c,hcA₁,hcW,hcZ₁,hyc,hy'c,hFc⟩:=sparse_common_neighbour_avoiding Q F A₁ W (insert a Z) y y' L b
  48     hL hF hcount₁ (by
  49       have hh:=mul_le_mul_of_nonneg_right hZ₁ hL1
  50       nlinarith only [hh,hbudget,hL1,(show (0:ℝ)≤r from Nat.cast_nonneg r)])
  51   have hcA:c∈A:=by
  52     dsimp only [A₁] at hcA₁
  53     split_ifs at hcA₁ with hr4
  54     · exact (Finset.mem_inter.mp hcA₁).1
  55     · exact hcA₁
  56   have hca:c≠a:=fun h=>hcZ₁ (by simp [h])
  57   have hcZ:c∉Z:=fun h=>hcZ₁ (Finset.mem_insert_of_mem h)
  58   let Z₁:=insert c (insert a Z)
  59   have hind:∀ v∈Z₁,∀ w∈Z₁,¬F.Adj v w:=
  60     graph_independent_insert F (insert a Z) c (graph_independent_insert F Z a hZ hFa) hFc
  61   have hex:∃ p:Q.Walk a c,p.IsPath ∧ p.length=r-3 ∧
  62       (∀ v∈p.support,v=a∨v=c∨v∈A∧v∉W∧v∉Z₁) ∧
  63       (∀ v∈p.support,∀ w∈p.support,¬F.Adj v w) ∧
  64       (∀ v∈p.support,∀ w∈Z₁,¬F.Adj v w):=by
  65     by_cases hr4:r=4
  66     · have hac:Q.Adj a c:=by
  67         have hh:c∈A∩Q.neighborFinset a:=by simpa only [A₁,hr4,ite_true] using hcA₁
  68         exact (Q.mem_neighborFinset a c).mp (Finset.mem_inter.mp hh).2
  69       let p:Q.Walk a c:=SimpleGraph.Walk.cons hac SimpleGraph.Walk.nil
  70       have hm (v:V):v∈p.support↔v=a∨v=c:=by simp [p]
  71       refine ⟨p,?_,by simp [p,hr4],?_,?_,?_⟩
  72       · exact SimpleGraph.Walk.IsPath.cons (by simp) (by simpa using hca.symm)
  73       · intro v hv
  74         rcases (hm v).mp hv with h|h
  75         · exact Or.inl h
  76         · exact Or.inr (Or.inl h)
  77       · intro v hv w hw
  78         apply hind v _ w _
  79         · rcases (hm v).mp hv with rfl|rfl <;> simp [Z₁]
  80         · rcases (hm w).mp hw with rfl|rfl <;> simp [Z₁]
  81       · intro v hv w hw
  82         apply hind v _ w hw
  83         rcases (hm v).mp hv with rfl|rfl <;> simp [Z₁]
  84     · have hZ₁card:(Z₁.card:ℝ)≤Z.card+2:=by
  85         have hh:Z₁.card≤Z.card+2:=(Finset.card_insert_le c (insert a Z)).trans (by have hh:=Finset.card_insert_le a Z;omega)
  86         exact_mod_cast hh
  87       exact sparse_fixed_length_forbidden_path_within Q F A W Z₁ Ω hAΩ a c (hAΩ haA) (hAΩ hcA)
  88         hca.symm (by simp [Z₁]) (by simp [Z₁]) hind (r-3) (by omega) L b hL hF hcount (by
  89           have hrc:((r-3:ℕ):ℝ)≤r:=by exact_mod_cast Nat.sub_le r 3
  90           have hh:=mul_le_mul_of_nonneg_right (show (Z₁.card:ℝ)+(r-3:ℕ)≤Z.card+r+2 by linarith) hL1
  91           linarith only [hh,hbudget])
  92   obtain ⟨p,hp,hlen,hclass,hindp,hrootp⟩:=hex
  93   refine ⟨a,c,p,hp,hlen,hxa,hx'a,hyc,hy'c,?_,hindp,?_⟩
  94   · intro v hv
  95     rcases hclass v hv with rfl|rfl|hv
  96     · exact ⟨haA,haW,haZ⟩
  97     · exact ⟨hcA,hcW,hcZ⟩
  98     · exact ⟨hv.1,hv.2.1,fun h=>hv.2.2 (by simp [Z₁,h])⟩
  99   · intro v hv w hw
 100     exact hrootp v hv w (by simp [Z₁,hw])
