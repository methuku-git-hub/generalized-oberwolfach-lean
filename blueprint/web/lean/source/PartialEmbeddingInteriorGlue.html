   9 theorem partial_embeddings_glue_across_interior
  10     {X Y:Type*} [DecidableEq X] (G:SimpleGraph X) (H:SimpleGraph Y)
  11     (A D:Finset X) (hDA:D⊆A)
  12     (fA:{x // x∈A}↪Y) (fC:{x // x∉D}↪Y)
  13     (hA:∀x y (hx:x∈A) (hy:y∈A),G.Adj x y→H.Adj (fA ⟨x,hx⟩) (fA ⟨y,hy⟩))
  14     (hC:∀x y (hx:x∉D) (hy:y∉D),G.Adj x y→H.Adj (fC ⟨x,hx⟩) (fC ⟨y,hy⟩))
  15     (hbound:∀x (hx:x∈A) (hxD:x∉D),fA ⟨x,hx⟩=fC ⟨x,hxD⟩)
  16     (hdis:∀x (hx:x∈D) y (hy:y∉D),fA ⟨x,hDA hx⟩≠fC ⟨y,hy⟩)
  17     (hneigh:∀x,x∈D→∀y,G.Adj x y→y∈A) :
  18     ∃f:X↪Y,(∀x (hx:x∈D),f x=fA ⟨x,hDA hx⟩) ∧
  19       (∀x (hx:x∉D),f x=fC ⟨x,hx⟩) ∧ (∀x y,G.Adj x y→H.Adj (f x) (f y)) := by
  20   classical
  21   let f:X→Y:=fun x=>if hx:x∈D then fA ⟨x,hDA hx⟩ else fC ⟨x,hx⟩
  22   have hf:Function.Injective f:=by
  23     intro x y hxy
  24     by_cases hx:x∈D <;> by_cases hy:y∈D
  25     · have h:fA ⟨x,hDA hx⟩=fA ⟨y,hDA hy⟩:=by simpa [f,hx,hy] using hxy
  26       exact congrArg Subtype.val (fA.injective h)
  27     · have h:fA ⟨x,hDA hx⟩=fC ⟨y,hy⟩:=by simpa [f,hx,hy] using hxy
  28       exact False.elim (hdis x hx y hy h)
  29     · have h:fA ⟨y,hDA hy⟩=fC ⟨x,hx⟩:=by simpa [f,hx,hy] using hxy.symm
  30       exact False.elim (hdis y hy x hx h)
  31     · have h:fC ⟨x,hx⟩=fC ⟨y,hy⟩:=by simpa [f,hx,hy] using hxy
  32       exact congrArg Subtype.val (fC.injective h)
  33   refine ⟨⟨f,hf⟩,?_,?_,?_⟩
  34   · intro x hx; simp [f,hx]
  35   · intro x hx; simp [f,hx]
  36   · intro x y hxy
  37     by_cases hx:x∈D <;> by_cases hy:y∈D
  38     · simpa [f,hx,hy] using hA x y (hDA hx) (hDA hy) hxy
  39     · have hyA:=hneigh x hx y hxy
  40       have h:=hA x y (hDA hx) hyA hxy
  41       rw [hbound y hyA hy] at h
  42       simpa [f,hx,hy] using h
  43     · have hxA:=hneigh y hy x hxy.symm
  44       have h:=hA y x (hDA hy) hxA hxy.symm
  45       rw [hbound x hxA hx] at h
  46       simpa [f,hx,hy] using h.symm
  47     · simpa [f,hx,hy] using hC x y hx hy hxy
  48 end Oberwolfach
