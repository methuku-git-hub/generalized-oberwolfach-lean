  15 def ordinaryIsolatorDeleteConflictsEquiv (P Q F : SimpleGraph V) (H : V → U → Prop)
  16     (k : ℕ) (θ α : ℝ) (x y : V) (u : U) :
  17     {C : OrdinaryIsolatorConfiguration P Q F H k θ α x y // C.hub=u} ≃
  18     {C : OrdinaryIsolatorConfiguration P (Q\F) F H k θ α x y // C.hub=u} where
  19   toFun C := ⟨{ C.val with
  20     rim_edges := by
  21       intro z t hzt
  22       have hh := (spanningRestriction_adj P C.val.vertices.val z t).mp hzt
  23       exact ⟨C.val.rim_edges hzt,C.val.independent z hh.2.1 t hh.2.2⟩ },C.property⟩
  24   invFun C := ⟨{ C.val with
  25     rim_edges := by
  26       intro z t hzt
  27       exact (C.val.rim_edges hzt).1 },C.property⟩
  28   left_inv C := by apply Subtype.ext;apply OrdinaryIsolatorConfiguration.encoding_injective;rfl
  29   right_inv C := by apply Subtype.ext;apply OrdinaryIsolatorConfiguration.encoding_injective;rfl

  33     (P : SimpleGraph V) (S : I → FixedSubset V k) (root : Sym2 V) (σ : Sym2 V → Bool) :
  34     rootedInducedSurvivalSum P S root σ=
  35       (Fintype.card {i : I // ∀ e∈rootOmittedInducedEdges P (S i).val root,σ e=true}:ℝ)/
  36         (Fintype.card V:ℝ)^(k-2) := by
