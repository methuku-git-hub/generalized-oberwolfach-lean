  14 theorem uniform_profile_mgf (m : Fin s → ℕ)
  15     [∀ i,Nonempty (FixedSubset (V i) (m i))]
  16     (hm : ∀ i,m i ≤ Fintype.card (V i)) (f : (∀ i,FixedSubset (V i) (m i)) → ℝ)
  17     (K : ℝ) (hK : 0 ≤ K)
  18     (hchange : ∀ i (σ τ : ∀ j,FixedSubset (V j) (m j)),
  19       (∀ j,j≠i → σ j=τ j) → ∀ x y,τ i=fixedSubsetMap (Equiv.swap x y) (σ i) → |f σ-f τ| ≤ K)
  20     (u : ℝ) :
  21     let μ := FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (m i)))
  22     μ.expectation (fun σ => Real.exp (u*(f σ-μ.expectation f))) ≤
  23       Real.exp (u^2*((∑ i,(m i:ℝ))*K^2)/8) := by
  24   let spec := fun i => UniformSampleSpec.subset (m i) (hm i)
  25   have hchange' : ∀ i (σ τ : ∀ j,(spec j).State),
  26       (∀ j,j≠i → σ j=τ j) → ∀ x y,τ i=(spec i).exchange x y (σ i) → |f σ-f τ| ≤ K := by
  27     intro i σ τ ha x y he
  28     exact hchange i σ τ ha x y he
  29   have hh := uniformSamples_mgf spec f (fun _ => K) (fun _ => hK) hchange' u
  30   have hvar : (∑ i,((spec i).exposures:ℝ)*K^2) ≤ (∑ i,(m i:ℝ))*K^2 := by
  31     rw [Finset.sum_mul]
  32     apply Finset.sum_le_sum
  33     intro i _
  34     apply mul_le_mul_of_nonneg_right _ (sq_nonneg K)
  35     exact_mod_cast (Nat.min_le_left (m i) (Fintype.card (V i)-m i))
  36   have hh' := hh.trans (Real.exp_le_exp.mpr (div_le_div_of_nonneg_right
  37       (mul_le_mul_of_nonneg_left hvar (sq_nonneg u)) (by norm_num)))
  38   simpa only [spec,UniformSampleSpec.law,UniformSampleSpec.State] using hh'
