  11 theorem combine_class_matchings {A Y : Type*} [Fintype A] [Fintype Y]
  12     (P : A → Prop) (Q : Y → Prop) (E : A → Y → Prop)
  13     (f₁ : {a // P a} ≃ {y // Q y}) (f₀ : {a // ¬P a} ≃ {y // ¬Q y})
  14     (h₁ : ∀ a, E a.val (f₁ a).val) (h₀ : ∀ a, E a.val (f₀ a).val) :
  15     ∃ f : A ≃ Y, (∀ a, E a (f a)) ∧ (∀ a, Q (f a) ↔ P a) := by
  16   classical
  17   let f : A ≃ Y := (Equiv.sumCompl P).symm.trans
  18     ((Equiv.sumCongr f₁ f₀).trans (Equiv.sumCompl Q))
  19   refine ⟨f,?_,?_⟩
  20   · intro a
  21     by_cases ha : P a
  22     · change E a ((Equiv.sumCompl Q) ((Equiv.sumCongr f₁ f₀) ((Equiv.sumCompl P).symm a)))
  23       rw [Equiv.sumCompl_symm_apply_of_pos ha]
  24       exact h₁ ⟨a,ha⟩
  25     · change E a ((Equiv.sumCompl Q) ((Equiv.sumCongr f₁ f₀) ((Equiv.sumCompl P).symm a)))
  26       rw [Equiv.sumCompl_symm_apply_of_neg ha]
  27       exact h₀ ⟨a,ha⟩
  28   · intro a
  29     by_cases ha : P a
  30     · have he : Q (f a) := by
  31         change Q ((Equiv.sumCompl Q) ((Equiv.sumCongr f₁ f₀) ((Equiv.sumCompl P).symm a)))
  32         rw [Equiv.sumCompl_symm_apply_of_pos ha]
  33         exact (f₁ ⟨a,ha⟩).property
  34       exact ⟨fun _ => ha,fun _ => he⟩
  35     · have he : ¬Q (f a) := by
  36         change ¬Q ((Equiv.sumCompl Q) ((Equiv.sumCongr f₁ f₀) ((Equiv.sumCompl P).symm a)))
  37         rw [Equiv.sumCompl_symm_apply_of_neg ha]
  38         exact (f₀ ⟨a,ha⟩).property
  39       exact ⟨fun h => (he h).elim,fun h => (ha h).elim⟩
