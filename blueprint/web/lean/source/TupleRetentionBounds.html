  13 theorem conflicting_retention_probability_zero
  14     {V C J : Type*} [Fintype V] [Fintype C]
  15     (G : SimpleGraph V) (μ : FiniteLaw (V → C×Bool)) (f : J → V) (c : C)
  16     (hbad : ∃ i j,G.Adj (f i) (f j)) :
  17     μ.probability (fun σ=>∀ j,coinRetainedColour G σ (f j)=some c)=0 := by
  18   obtain ⟨i,j,hij⟩ := hbad
  19   have hn (σ : V → C×Bool) : ¬∀ j,coinRetainedColour G σ (f j)=some c := by

  21     exact coinRetainedColour_proper G σ hij c (h i) (h j)

  26     {V C J : Type*} [Fintype V] [DecidableEq V] [Fintype C] [Fintype J]
  27     (G : SimpleGraph V) (L : V → Finset C) (hL : ∀ v,(L v).Nonempty)
  28     {l K B : ℕ} (hl : 1<l) (hcard : ∀ v,(L v).card=l)
  29     (f : J → V) (hfi : Function.Injective f)
  30     (hgood : ∀ i j,¬G.Adj (f i) (f j)) (c : C) (hc : ∀ j,c∈L (f j))
  31     (hbound : ∀ j,(colourNeighbourSet G L (f j) c).card≤K)
  32     (hpair : ∀ i j,i≠j →
  33       (colourNeighbourSet G L (f i) c∩colourNeighbourSet G L (f j) c).card≤B)
  34     (hsmall : 2*((Fintype.card J)^2*B)≤l) :
  35     let p := (FiniteLaw.pi (fun u=>(FiniteLaw.uniformOn (L u) (hL u)).withCoin
  36       (equalizationCoin G L l K u)
  37       (equalizationCoin_bounds G L (by omega) K u))).probability
  38       (fun σ=>∀ j,coinRetainedColour G σ (f j)=some c)
  39     let b := (((1-1/(l : ℝ))^K)/(l : ℝ))^(Fintype.card J)
  40     b≤p ∧ p≤(1+2*(((Fintype.card J)^2*B : ℕ) : ℝ)/(l : ℝ))*b := by

  42   let S := Finset.univ.image f

  44     rw [Finset.card_image_of_injective _ hfi,Finset.card_univ]
  45   have hS : ∀ u∈S,∀ v∈S,¬G.Adj u v := by
  46     intro u hu v hv
  47     obtain ⟨i,_,rfl⟩ := Finset.mem_image.mp hu

  59       (colourNeighbourSet G L u c∩colourNeighbourSet G L v c).card≤B := by
  60     intro u hu v hv huv

  63     exact hpair i j (fun he=>huv (congrArg f he))
