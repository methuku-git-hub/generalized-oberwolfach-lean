  10 theorem long_assigned_selected_density_uniform (m:ℕ) (hm:7≤m)
  11     (ρ₀ ρ:ℝ) (hρ₀:0<ρ₀) (hρlo:ρ₀≤ρ) (hρhi:ρ≤1):
  12     let s:ℝ:=1/100
  13     let t:ℝ:=s/m
  14     let p:ℝ:=ρ*s/2
  15     let k:ℝ:=ρ*t/2
  16     let kstar:ℝ:=ρ₀*t/2
  17     let pstarRef:ℝ:=ρ₀*t/8;
  18     0<s ∧ s≤1/100 ∧ 0<t ∧ t≤s ∧ t≤1 ∧
  19     0<kstar ∧ kstar≤p ∧ kstar≤k ∧
  20     0<pstarRef ∧ pstarRef≤ρ₀*t/4 ∧ k=p/m ∧
  21     0≤p ∧ 0≤k ∧ 2*p+2*k≤ρ :=by
  22   intro s t p k kstar pstarRef
  23   have hmpos:(0:ℝ)<m:=by exact_mod_cast (show 0<m by omega)
  24   have hmone:(1:ℝ)≤m:=by exact_mod_cast (show 1≤m by omega)
  25   have hrho:0<ρ:=hρ₀.trans_le hρlo
  26   have hs:0<s:=by dsimp [s];norm_num
  27   have ht:0<t:=by dsimp [t];positivity
  28   have hts:t≤s:=by
  29     dsimp [t]
  30     exact (div_le_iff₀ hmpos).2 (by nlinarith)
  31   have ht1:t≤1:=by dsimp [s] at hts;linarith
  32   have hk0:0<kstar:=by dsimp [kstar];positivity
  33   have hkp:kstar≤p:=by
  34     have h₁:=mul_le_mul_of_nonneg_right hρlo ht.le
  35     have h₂:=mul_le_mul_of_nonneg_left hts hrho.le
  36     dsimp [kstar,p]
  37     exact div_le_div_of_nonneg_right (h₁.trans h₂) (by norm_num)
  38   have hkk:kstar≤k:=by dsimp [kstar,k];nlinarith
  39   have hRef:0<pstarRef:=by dsimp [pstarRef];positivity
  40   have hRefU:pstarRef≤ρ₀*t/4:=by dsimp [pstarRef];nlinarith
  41   have heq:k=p/m:=by dsimp [k,p,t];ring
  42   have hp:0≤p:=by dsimp [p];positivity
  43   have hk:0≤k:=by dsimp [k];positivity
  44   have hsum:2*p+2*k≤ρ:=by
  45     dsimp [p,k,s] at *
