  14  theorem common_class_prebalance (π:Equiv.Perm V) (A₀ B₀ I ZK ZO:Finset V)
  15     (hZA:ZK⊆A₀) (hZI:Disjoint (ZK∪ZO) I)
  16     (hsep:Disjoint ZK (ZO.image π.symm)):
  17     let A:=(A₀\I)\ZO.image π.symm
  18     let B:=((B₀\I)∪ZO)\A.image π
  19     ZK⊆A ∧ ZO⊆B ∧ A⊆A₀ ∧ B⊆B₀∪ZO ∧
  20       Disjoint (A∪B) I ∧ Disjoint B (A.image π) ∧
  21       ((changed A A₀).card+(changed B B₀).card≤
  22         (B₀∩A₀.image π).card+2*I.card+2*ZO.card):=by
  23   intro A B
  24   have hAA:A⊆A₀:=Finset.sdiff_subset.trans Finset.sdiff_subset
  25   have hZA':ZK⊆A:=by
  26     intro x hx
  27     exact Finset.mem_sdiff.mpr ⟨Finset.mem_sdiff.mpr ⟨hZA hx,
  28       fun hi=>Finset.disjoint_left.mp hZI (Finset.mem_union_left _ hx) hi⟩,
  29       fun hy=>Finset.disjoint_left.mp hsep hx hy⟩
  30   have hZO':ZO⊆B:=by
  31     intro x hx
  32     refine Finset.mem_sdiff.mpr ⟨Finset.mem_union_right _ hx,?_⟩
  33     rintro hh
  34     obtain ⟨y,hy,he⟩:=Finset.mem_image.mp hh
  35     apply (Finset.mem_sdiff.mp hy).2
  36     exact Finset.mem_image.mpr ⟨x,hx,by rw [←he];exact π.symm_apply_apply y⟩
  37   have hBB:B⊆B₀∪ZO:=by
  38     intro x hx
  39     rcases (Finset.mem_sdiff.mp hx).1 with hx
  40     rcases Finset.mem_union.mp hx with hx|hx
  41     · exact Finset.mem_union_left _ (Finset.mem_sdiff.mp hx).1
  42     · exact Finset.mem_union_right _ hx
  43   have hAI:Disjoint (A∪B) I:=by
  44     apply Finset.disjoint_left.mpr
  45     intro x hx hi
  46     rcases Finset.mem_union.mp hx with hx|hx
  47     · exact (Finset.mem_sdiff.mp (Finset.mem_sdiff.mp hx).1).2 hi
  48     · rcases Finset.mem_union.mp (Finset.mem_sdiff.mp hx).1 with hx|hx
  49       · exact (Finset.mem_sdiff.mp hx).2 hi
  50       · exact Finset.disjoint_left.mp hZI (Finset.mem_union_right _ hx) hi
  51   have hBπ:Disjoint B (A.image π):=by
  52     apply Finset.disjoint_left.mpr
  53     intro x hx hy
  54     exact (Finset.mem_sdiff.mp hx).2 hy
  55   have hAc:changed A A₀⊆I∪ZO.image π.symm:=by
  56     intro x hx
  57     have hh:(x∈A∧x∉A₀)∨(x∈A₀∧x∉A):=by simpa only [changed,Finset.mem_union,Finset.mem_sdiff] using hx
  58     rcases hh with ⟨hx,hnot⟩|⟨hx,hnot⟩
  59     · exact False.elim (hnot (hAA hx))
  60     · by_cases hi:x∈I
  61       · exact Finset.mem_union_left _ hi
  62       · apply Finset.mem_union_right
  63         by_contra hh
  64         exact hnot (Finset.mem_sdiff.mpr ⟨Finset.mem_sdiff.mpr ⟨hx,hi⟩,hh⟩)
  65   have hBc:changed B B₀⊆(I∪ZO)∪(B₀∩A₀.image π):=by
  66     intro x hx
  67     have hh:(x∈B∧x∉B₀)∨(x∈B₀∧x∉B):=by simpa only [changed,Finset.mem_union,Finset.mem_sdiff] using hx
  68     rcases hh with ⟨hx,hnot⟩|⟨hx,hnot⟩
  69     · have hzo: x∈ZO:=(Finset.mem_union.mp (hBB hx)).resolve_left hnot
  70       exact Finset.mem_union_left _ (Finset.mem_union_right _ hzo)
  71     · by_cases hi:x∈I
  72       · exact Finset.mem_union_left _ (Finset.mem_union_left _ hi)
  73       · have hh:x∈A.image π:=by
  74           by_contra hh
  75           exact hnot (Finset.mem_sdiff.mpr ⟨Finset.mem_union_left _ (Finset.mem_sdiff.mpr ⟨hx,hi⟩),hh⟩)
  76         exact Finset.mem_union_right _ (Finset.mem_inter.mpr ⟨hx,Finset.image_subset_image hAA hh⟩)
  77   have hAc':(changed A A₀).card≤I.card+ZO.card:=by
  78     have hh:=(Finset.card_le_card hAc).trans (Finset.card_union_le _ _)
  79     simpa only [Finset.card_image_of_injective _ π.symm.injective] using hh
  80   have hBc':(changed B B₀).card≤I.card+ZO.card+(B₀∩A₀.image π).card:=by
  81     exact (Finset.card_le_card hBc).trans ((Finset.card_union_le _ _).trans
  82       (Nat.add_le_add_right (Finset.card_union_le _ _) _))
  83   exact ⟨hZA',hZO',hAA,hBB,hAI,hBπ,by omega⟩
