  15 def markedExtensionIndexRequests (X Y : Finset V) : Finset (Bool×V) :=
  16   X.image (fun v=>(false,v)) ∪ Y.image (fun v=>(true,v))

  18 theorem markedExtensionIndexRequests_mem (X Y : Finset V) (a : Bool×V) :
  19     a∈markedExtensionIndexRequests X Y ↔ (a.1=false ∧ a.2∈X) ∨ (a.1=true ∧ a.2∈Y) := by
  20   obtain ⟨b,v⟩:=a
  21   cases b <;> simp [markedExtensionIndexRequests]

  23 theorem markedExtensionIndexRequests_card (X Y : Finset V) :
  24     (markedExtensionIndexRequests X Y).card=X.card+Y.card := by
  25   have hdis : Disjoint (X.image (fun v=>(false,v))) (Y.image (fun v=>(true,v))) := by
  26     apply Finset.disjoint_left.mpr
  27     intro a ha hb
  28     obtain ⟨v,hv,he⟩:=Finset.mem_image.mp ha
  29     obtain ⟨w,hw,hf⟩:=Finset.mem_image.mp hb
  30     have hh:=congrArg Prod.fst (he.trans hf.symm)
  31     cases hh
  32   rw [markedExtensionIndexRequests,Finset.card_union_of_disjoint hdis]
  33   rw [Finset.card_image_of_injective _ (by intro v w h;exact Prod.mk.inj h |>.2),
  34     Finset.card_image_of_injective _ (by intro v w h;exact Prod.mk.inj h |>.2)]

  36 theorem markedExtensionIndexRequests_separated (π : Equiv.Perm V) (b : ℕ) (X Y : Finset V)
  37     (hXY : Disjoint X Y) (hsep : ((X∪Y):Set V).Pairwise (fun v w=>¬PermutationClose π b v w)) :
  38     ∀ a∈markedExtensionIndexRequests X Y,∀ e∈markedExtensionIndexRequests X Y,
  39       a≠e → ¬PermutationClose π b a.2 e.2 := by
  40   intro a ha e he hne
  41   have ha':=(markedExtensionIndexRequests_mem X Y a).mp ha
  42   have he':=(markedExtensionIndexRequests_mem X Y e).mp he
  43   have ham : a.2∈X∪Y := by rcases ha' with h|h;exact Finset.mem_union_left _ h.2;exact Finset.mem_union_right _ h.2
  44   have hem : e.2∈X∪Y := by rcases he' with h|h;exact Finset.mem_union_left _ h.2;exact Finset.mem_union_right _ h.2
  45   have hroot : a.2≠e.2 := by
  46     intro h
  47     rcases ha' with ha'|ha' <;> rcases he' with he'|he'
  48     · exact hne (Prod.ext (ha'.1.trans he'.1.symm) h)
  49     · exact Finset.disjoint_left.mp hXY ha'.2 (h.symm ▸ he'.2)
  50     · exact Finset.disjoint_left.mp hXY (h.symm ▸ he'.2) ha'.2
  51     · exact hne (Prod.ext (ha'.1.trans he'.1.symm) h)
  52   exact hsep (by simpa using ham) (by simpa using hem) hroot
