  33 theorem interface_family_random_regularization (β : ℝ) (hβ : 0 < β) :
  34     ∃ C η₀ : ℝ, 0 < C ∧ 0 < η₀ ∧ ∀ η : ℝ, 0 < η → η < η₀ →
  35       ∃ m₀ : ℕ, ∀ (I V : Type) [Fintype I] [DecidableEq I] [Fintype V] [DecidableEq V],
  36       m₀ ≤ Fintype.card V → ∀ (M : I → V → V → ℕ),
  37       (∀ i x y, M i x y ≤ 1) → ∀ d : ℝ,
  38       (∀ i x, (1 / 2 + β) * Fintype.card V ≤ ((∑ y, M i x y : ℕ) : ℝ)) →
  39       (∀ i y, (1 / 2 + β) * Fintype.card V ≤ ((∑ x, M i x y : ℕ) : ℝ)) →
  40       (∀ i x, d - η * Fintype.card V ≤ ((∑ y, M i x y : ℕ) : ℝ) ∧
  41         ((∑ y, M i x y : ℕ) : ℝ) ≤ d + η * Fintype.card V) →
  42       (∀ i y, d - η * Fintype.card V ≤ ((∑ x, M i x y : ℕ) : ℝ) ∧
  43         ((∑ x, M i x y : ℕ) : ℝ) ≤ d + η * Fintype.card V) →
  44       let q := ⌊d - Real.sqrt η * Fintype.card V⌋₊
  45       ∃ ν : FiniteLaw (∀ i, BipartiteEdges (M i) → Bool),
  46         (∀ D, 0 < ν.mass D → ∀ i,
  47           (∀ x, (∑ y, retainedBipartiteMatrix (M i) (D i) x y) = q) ∧
  48           (∀ y, (∑ x, retainedBipartiteMatrix (M i) (D i) x y) = q)) ∧
  49         (∀ (D : ∀ i, BipartiteEdges (M i) → Bool) i x y, retainedBipartiteMatrix (M i) (D i) x y ≤ M i x y) ∧
  50         (∀ i (e : BipartiteEdges (M i)),
  51           ν.probability (fun D => retainedBipartiteMatrix (M i) (D i) e.1 e.2.1 = 0) ≤
  52             C * Real.sqrt η) := by
  53   obtain ⟨C, η₀, hC, hη₀, hall⟩ := bipartite_matrix_random_regularization β hβ
  54   refine ⟨C, η₀, hC, hη₀, ?_⟩
  55   intro η hη hηsmall
  56   obtain ⟨m₀, hm₀⟩ := hall η hη hηsmall
  57   refine ⟨m₀, ?_⟩
  58   intro I V _ _ _ _ hm M hM d hl hr hnl hnr
  59   have hex (i : I) := hm₀ V hm (M i) (hM i) d (hl i) (hr i) (hnl i) (hnr i)
  60   choose μ hdegrees hsub hprob using hex
  61   refine ⟨FiniteLaw.pi μ, ?_, ?_, ?_⟩
  62   · intro D hD i
  63     exact hdegrees i (D i) (finiteLaw_pi_positive_coordinates μ D hD i)
  64   · intro D i x y
  65     exact hsub i (D i) x y
  66   · intro i e
  67     rw [finiteLaw_pi_coordinate_event μ i (fun D => retainedBipartiteMatrix (M i) D e.1 e.2.1 = 0)]
  68     exact hprob i e
