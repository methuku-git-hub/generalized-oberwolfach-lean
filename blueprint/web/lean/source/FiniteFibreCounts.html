  14 theorem finite_sigma_predicate_card {I : Type*} [Fintype I] {D : I → Type*}
  15     [∀ i,Fintype (D i)] (P : ∀ i,D i → Prop) :
  16     Fintype.card {d : Σ i,D i // P d.1 d.2}=∑ i,Fintype.card {d : D i // P i d} := by
  17   let e : {d : Σ i,D i // P d.1 d.2} ≃ Σ i,{d : D i // P i d} := {
  18     toFun d := ⟨d.val.1,⟨d.val.2,d.property⟩⟩
  19     invFun d := ⟨⟨d.1,d.2.val⟩,d.2.property⟩
  20     left_inv := by rintro ⟨⟨i,d⟩,h⟩;rfl
  21     right_inv := by rintro ⟨i,⟨d,h⟩⟩;rfl }
  22   simpa only [Fintype.card_sigma] using Fintype.card_congr e

  24 theorem finite_fibre_card_identity {A B : Type*} [Fintype A] [Fintype B] (f : A → B) :
  25     Fintype.card A=∑ b,Fintype.card {a : A // f a=b} := by
  26   let e : A ≃ Σ b,{a : A // f a=b} := {
  27     toFun a := ⟨f a,⟨a,rfl⟩⟩
  28     invFun b := b.2.val
  29     left_inv := fun _ => rfl
  30     right_inv := by rintro ⟨b,⟨a,h⟩⟩;subst b;rfl }
  31   simpa only [Fintype.card_sigma] using Fintype.card_congr e

  33 theorem finite_fibre_card_bound {A B : Type*} [Fintype A] [Fintype B]
  34     (f : A → B) (M : ℕ) (hf : ∀ b,Fintype.card {a : A // f a=b} ≤ M) :
  35     Fintype.card A ≤ Fintype.card B*M := by
  36   rw [finite_fibre_card_identity f]
  37   calc
  38     _ ≤ ∑ _b : B,M := Finset.sum_le_sum (fun b _ => hf b)
