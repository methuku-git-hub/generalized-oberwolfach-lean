  10 theorem chainJoint_product_with_last {I X : Type*}
  11     (m : I → X → ℝ) (K : I → X → X → ℝ) (s : I → I)
  12     (i : I) (j : ℕ) (w : Fin (j + 1) → X) :
  13     chainJoint m K s i j w =
  14       m (s^[j] i) (w (Fin.last j)) *
  15         ∏ t : Fin j, K (s^[t.val] i) (w t.castSucc) (w t.succ) /
  16           m (s^[t.val + 1] i) (w t.succ) := by
  17   induction j generalizing i with
  18   | zero => simp [chainJoint]
  19   | succ j ih =>
  20     change K i (w 0) (w 1) / m (s i) (w 1) *
  21       chainJoint m K s (s i) j (Fin.tail w) = _
  22     rw [ih, Fin.prod_univ_succ]
  23     simp only [Fin.val_zero, Function.iterate_zero_apply, Nat.zero_add,
  24       Fin.castSucc_zero, Fin.succ_zero_eq_one,
  25       Fin.val_succ, Function.iterate_succ_apply, Fin.tail, Fin.succ_castSucc, Fin.succ_last]
  26     ring

  28 theorem chainJoint_product {I X : Type*}
  29     (m : I → X → ℝ) (K : I → X → X → ℝ) (s : I → I)
  30     (hm : ∀ i x, 0 < m i x)
  31     (i : I) (j : ℕ) (w : Fin (j + 2) → X) :
  32     chainJoint m K s i (j + 1) w =
  33       (∏ t : Fin (j + 1), K (s^[t.val] i) (w t.castSucc) (w t.succ)) /
  34       ∏ t : Fin j, m (s^[t.val + 1] i) (w t.succ.castSucc) := by
  35   rw [chainJoint_product_with_last, Finset.prod_div_distrib]
  36   have hden :
  37       (∏ t : Fin (j + 1), m (s^[t.val + 1] i) (w t.succ)) =
  38         (∏ t : Fin j, m (s^[t.val + 1] i) (w t.succ.castSucc)) *
  39           m (s^[j + 1] i) (w (Fin.last (j + 1))) := by
  40     rw [Fin.prod_univ_castSucc]
  41     rfl
  42   rw [hden]
