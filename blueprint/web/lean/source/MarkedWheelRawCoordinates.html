  26   let ρ:=canonicalMarkedRotation k
  27   have hi:ρ.symm i=Fin.last m↔i=k:=by

  32     · intro he

  34       simpa only [Equiv.apply_symm_apply,ρ,canonicalMarkedRotation_last] using he
  35   have hz:W.1.val (ρ.symm i)=z i:=by rw [hW];simp only [ρ,Equiv.apply_symm_apply]
  36   have hs:W.1.val (finRotate (m+1) (ρ.symm i))=z (finRotate (m+1) i):=by
