  16 def markedPairResource: (V⊕U)→(V⊕U)→(Finset V⊕(V×U))
  17   | Sum.inl x,Sum.inl y=>Sum.inl {x,y}
  18   | Sum.inl x,Sum.inr u=>Sum.inr (x,u)
  19   | Sum.inr u,Sum.inl x=>Sum.inr (x,u)
  20   | Sum.inr _,Sum.inr _=>Sum.inl ∅

  22 theorem graph_degree_le_marked_incident (G:SimpleGraph (V⊕U))
  23     (S:Finset (Finset V⊕(V×U))) (hU:∀ u v,¬G.Adj (Sum.inr u) (Sum.inr v))
  24     (hS:∀ x y,G.Adj x y→markedPairResource x y∈S) (x:V⊕U):
  25     G.degree x≤(S.filter (markedPhysicalIncident x)).card:=by
  26   let f:{y // G.Adj x y}→S.filter (markedPhysicalIncident x):=fun y=>
  27     ⟨markedPairResource x y.val,Finset.mem_filter.mpr ⟨hS x y.val y.property,by
  28       cases x <;> rcases y with ⟨(_|_),hy⟩
  29       all_goals simp only [markedPairResource,markedPhysicalIncident,Finset.mem_insert,
  30         Finset.mem_singleton,true_or]
  31       exact (hU _ _ hy).elim⟩⟩
  32   have hi:Function.Injective f:=by
  33     intro y z he
  34     have he':markedPairResource x y.val=markedPairResource x z.val:=congrArg Subtype.val he
  35     apply Subtype.ext
  36     cases x with
  37     | inl v=>
  38       rcases y with ⟨(y|y),hy⟩ <;> rcases z with ⟨(z|z),hz⟩
  39       · have hp:({v,y}:Finset V)={v,z}:=Sum.inl.inj he'
  40         have hyz:y=v∨y=z:=by
  41           have h:y∈({v,z}:Finset V):=hp ▸ (by simp)
  42           simpa only [Finset.mem_insert,Finset.mem_singleton] using h
  43         rcases hyz with hyz|hyz
  44         · exact (hy.ne (congrArg Sum.inl hyz.symm)).elim
  45         · exact congrArg Sum.inl hyz
  46       · exact (Sum.inl_ne_inr he').elim
  47       · exact (Sum.inr_ne_inl he').elim
  48       · exact congrArg Sum.inr (Prod.mk.inj (Sum.inr.inj he')).2
  49     | inr u=>
  50       rcases y with ⟨(y|y),hy⟩ <;> rcases z with ⟨(z|z),hz⟩
  51       · exact congrArg Sum.inl (Prod.mk.inj (Sum.inr.inj he')).1
  52       · exact (hU _ _ hz).elim
  53       · exact (hU _ _ hy).elim
  54       · exact (hU _ _ hy).elim
  55   have hc:=Fintype.card_le_of_injective f hi
  56   have he:Fintype.card {y // G.Adj x y}=G.degree x:=by
  57     convert G.card_neighborSet_eq_degree x using 1 <;> congr!
  58   rw [he,Fintype.card_coe] at hc
  59   exact hc

  61 theorem graph_degree_le_unused_marked_incident (G:SimpleGraph (V⊕U))
  62     (S C:Finset (Finset V⊕(V×U))) (hU:∀ u v,¬G.Adj (Sum.inr u) (Sum.inr v))
  63     (hS:∀ x y,G.Adj x y→markedPairResource x y∈S∧markedPairResource x y∉C) (x:V⊕U):
  64     G.degree x≤((S.filter (markedPhysicalIncident x))\C).card:=by
  65   have h:=graph_degree_le_marked_incident G (S\C) hU
  66     (fun x y h=>Finset.mem_sdiff.mpr (hS x y h)) x
  67   convert h using 1
