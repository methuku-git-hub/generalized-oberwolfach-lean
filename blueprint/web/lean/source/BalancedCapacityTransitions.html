  15 theorem capacityTransition_isTransition {X : Type*} [Fintype X]
  16     (a : ℕ → X → X → ℝ) (d : ℕ → X → ℝ)
  17     (ha : ∀ i x y, 0 ≤ a i x y) (hd : ∀ i x, 0 < d i x)
  18     (hrow : ∀ i x, ∑ y, a i x y=d i x) :
  19     ∀ i, IsTransition (capacityTransition a d i) := by
  20   intro i
  21   constructor
  22   · intro x y
  23     exact div_nonneg (ha i x y) (hd i x).le
  24   · intro x
  25     change (∑ y, a i x y/d i x)=1
  26     rw [← Finset.sum_div,hrow,div_self (hd i x).ne']

  28 theorem capacityProfile_stationary {X : Type*} [Fintype X]
  29     (a : ℕ → X → X → ℝ) (d : ℕ → X → ℝ) (M : ℝ)
  30     (hd : ∀ i x, 0 < d i x) (hcol : ∀ i y, ∑ x, a i x y=d (i+1) y) :
  31     ∀ i, transitionApply (capacityProfile d M i) (capacityTransition a d i)=capacityProfile d M (i+1) := by
  32   intro i
  33   funext y
  34   change (∑ x, (d i x/M)*(a i x y/d i x))=d (i+1) y/M
  35   have heq (x : X) : (d i x/M)*(a i x y/d i x)=a i x y/M := by
  36     field_simp [(hd i x).ne']
  37   simp_rw [heq]
  38   rw [← Finset.sum_div,hcol]

  40 theorem capacityProfile_distribution {X : Type*} [Fintype X]
  41     (d : ℕ → X → ℝ) (M : ℝ) (hM : 0 < M)
  42     (hd : ∀ i x, 0 < d i x) (hs : ∀ i, ∑ x, d i x=M) :
  43     (∀ i x, 0 < capacityProfile d M i x) ∧ (∀ i, ∑ x, capacityProfile d M i x=1) := by
  44   constructor
  45   · intro i x
  46     exact div_pos (hd i x) hM
  47   · intro i
  48     change (∑ x, d i x/M)=1
  49     rw [← Finset.sum_div,hs,div_self hM.ne']

  51 theorem capacityTransition_reconstruct {X : Type*}
  52     (a : ℕ → X → X → ℝ) (d : ℕ → X → ℝ) (M : ℝ) (hM : M ≠ 0)
  53     (hd : ∀ i x, d i x ≠ 0) (i : ℕ) (x y : X) :
