  16 theorem marked_remaining_endpoint_class
  17     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ) (σ:RawMarkedOutcome V U Gamma Color)
  18     (H:Bool→V→U→Prop) (X:Finset V) (Z:Finset U) (hX:∀ x,π x∈X↔x∈X)
  19     (DR:SimpleGraph V) (HD:V→U→Prop) (mark incoming:Bool) (x:V) (hx:x∉X):
  20     let G:=ordinaryBoundaryGraph (markedSampleArcSetGraph P π d σ Finset.univ)
  21       (fun x u=>H false x u∨H true x u) X Z
  22     let C:=ordinaryPhysicalGraph DR HD;
  23     markedEndpointLabels P π d X (mark,incoming,x) (rawMarkedDeleteRims σ DR.edgeFinset)=
  24       (markedEndpointLabels P π d X (mark,incoming,x) σ).filter
  25         (fun v=>markedBoundaryEndpointCoin π mark incoming x (Sum.inl v)∈(G\C).edgeSet):=by
  26   intro G C
  27   ext v
  28   have hg: v∈markedEndpointLabels P π d X (mark,incoming,x) σ→
  29       markedBoundaryEndpointCoin π mark incoming x (Sum.inl v)∈G.edgeSet:=
  30     marked_sample_endpoint_coin_mem_boundary P π d σ H X Z hX mark incoming x hx v
  31   have he:v∈markedEndpointLabels P π d X (mark,incoming,x) (rawMarkedDeleteRims σ DR.edgeFinset)↔
  32       v∈markedEndpointLabels P π d X (mark,incoming,x) σ∧
  33       s((if incoming then v else x),if mark then π (if incoming then x else v) else (if incoming then x else v))∉DR.edgeFinset:=by
  34     rw [marked_endpoint_labels_actual,marked_endpoint_labels_actual]
  35     simp only [Finset.mem_filter,markedPhysicalRim_delete_iff]
  36     tauto
  37   rw [he,Finset.mem_filter]
  38   have hc:markedBoundaryEndpointCoin π mark incoming x (Sum.inl v)∈C.edgeSet↔
  39       s((if incoming then v else x),if mark then π (if incoming then x else v) else (if incoming then x else v))∈DR.edgeFinset:=by
  40     rw [marked_boundary_endpoint_rim_coin]
  41     cases mark <;> cases incoming <;>
  42       simp only [Bool.false_eq_true,ite_false,ite_true,SimpleGraph.mem_edgeFinset,
  43         SimpleGraph.mem_edgeSet,C,ordinaryPhysicalGraph]
