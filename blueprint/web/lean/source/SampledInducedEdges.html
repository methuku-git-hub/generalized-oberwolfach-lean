  12 theorem finiteLaw_prod_map_fst {Ω Γ : Type*} [Fintype Ω] [Fintype Γ]
  13     (μ : FiniteLaw Ω) (ν : FiniteLaw Γ) : (μ.prod ν).map Prod.fst=μ := by
  14   apply finiteLaw_ext
  15   intro ω
  16   simp only [FiniteLaw.map,FiniteLaw.prod,Fintype.sum_prod_type]
  17   have hh (x : Ω) : (∑ y,if x=ω then μ.mass x*ν.mass y else 0)=
  18       if x=ω then μ.mass x else 0 := by
  19     by_cases hx : x=ω
  20     · simp only [hx,ite_true,← Finset.mul_sum,ν.total,mul_one]
  21     · simp [hx]
  22   simp only [hh,Finset.sum_ite_eq',Finset.mem_univ,ite_true]

  24 theorem sampledInducedGraph_edge_probability {V : Type*} [Fintype V] [DecidableEq V]
  25     {m : ℕ} [Nonempty (FixedSubset V m)] (hm : m≤Fintype.card V)
  26     (G : SimpleGraph V) (μ : Sym2 V → FiniteLaw Bool) (u v : V) (huv : G.Adj u v) :
  27     ((FiniteLaw.uniform (FixedSubset V m)).prod (FiniteLaw.pi μ)).probability
  28       (fun z => (sampledInducedGraph G z).Adj u v)=
  29       pairSampleProbability (Fintype.card V) m*(μ s(u,v)).mass true := by
  30   rw [finiteLaw_prod_probability_slices (FiniteLaw.uniform (FixedSubset V m)) (FiniteLaw.pi μ)
  31     (fun S σ => (sampledInducedGraph G (S,σ)).Adj u v)]
  32   have hp (S : FixedSubset V m) : (FiniteLaw.pi μ).probability
  33       (fun σ => (sampledInducedGraph G (S,σ)).Adj u v)=
  34       if u∈S.val ∧ v∈S.val then (μ s(u,v)).mass true else 0 := by
  35     by_cases hS : u∈S.val ∧ v∈S.val
  36     · have he : (fun σ => (sampledInducedGraph G (S,σ)).Adj u v)=
  37           (fun σ => σ s(u,v)=true) := by
  38         funext σ
  39         apply propext
  40         simp only [sampledInducedGraph,spanningRestriction_adj,retainedGraph,huv,true_and,hS.1,hS.2,and_true]
  41       rw [he,finiteLaw_pi_coordinate_probability,if_pos hS]
  42     · have he : (fun σ => (sampledInducedGraph G (S,σ)).Adj u v)=(fun _ => False) := by
  43         funext σ
  44         apply propext
  45         simp only [sampledInducedGraph,spanningRestriction_adj,retainedGraph,huv,true_and]
  46         tauto
  47       rw [he,finiteLaw_probability_false,if_neg hS]
  48   simp only [hp]
  49   calc
  50     _ = (FiniteLaw.uniform (FixedSubset V m)).probability
  51         (fun S => u∈S.val ∧ v∈S.val)*(μ s(u,v)).mass true := by
  52       unfold FiniteLaw.probability
  53       rw [Finset.sum_mul]
  54       apply Finset.sum_congr rfl
  55       intro S _
  56       split_ifs <;> simp
  57     _ = _ := by rw [fixedSubset_pair_probability hm u v huv.ne]
