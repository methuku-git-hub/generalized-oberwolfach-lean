  15 structure PartitionTupleWeightConditions {V : Type*} [Fintype V] {ℓ k : ℕ} {d : Fin k → ℕ}
  16     (G : SimpleGraph V) (σ : Fin (ℓ + 2) → Equiv.Perm V) (P : FixedPartition V k d)
  17     (a : Fin (ℓ + 2) ⊕ Fin (ℓ + 2) → Fin k) (p q A B : ℝ)
  18     (w : (Fin (ℓ + 2) → V × V) → ℝ) : Prop where
  19   nonneg : ∀ t, 0 ≤ w t
  20   support : ∀ t, 0 < w t → TwistedCyclicTupleValid G σ t ∧
  21     ∀ i, P.1 (t i).1 = a (Sum.inl i) ∧ P.1 (t i).2 = a (Sum.inr i)
  22   lower : ∀ t, TwistedCyclicTupleValid G σ t →
  23     (∀ i, P.1 (t i).1 = a (Sum.inl i) ∧ P.1 (t i).2 = a (Sum.inr i)) → A ≤ w t
  24   upper : ∀ t, w t ≤ B
  25   left : ∀ i x y,
  26     (1-p) * (if P.1 x = a (Sum.inl i) ∧ P.1 y = a (Sum.inr i) ∧ G.Adj x y then 1 else 0) ≤
  27       (∑ t, if t i = (x,y) then w t else 0) ∧
  28     (∑ t, if t i = (x,y) then w t else 0) ≤
  29       (if P.1 x = a (Sum.inl i) ∧ P.1 y = a (Sum.inr i) ∧ G.Adj x y then 1 else 0)
  30   right : ∀ i x y,
  31     (1-p) * (if P.1 x = a (Sum.inr i) ∧ P.1 y = a (Sum.inl (finRotate (ℓ + 2) i)) ∧
  32       G.Adj x (σ i y) then 1 else 0) ≤
  33       (∑ t, if ((t i).2,(t (finRotate (ℓ + 2) i)).1) = (x,y) then w t else 0) ∧
  34     (∑ t, if ((t i).2,(t (finRotate (ℓ + 2) i)).1) = (x,y) then w t else 0) ≤
  35       (if P.1 x = a (Sum.inr i) ∧ P.1 y = a (Sum.inl (finRotate (ℓ + 2) i)) ∧
  36         G.Adj x (σ i y) then 1 else 0)
  37   positions : ∀ i v,
  38     (∑ t, if (t i).1 = v then w t else 0) = (if P.1 v = a (Sum.inl i) then q else 0) ∧
  39     (∑ t, if (t i).2 = v then w t else 0) = (if P.1 v = a (Sum.inr i) then q else 0)

  41 theorem near_regular_partition_tuple_weights (β : ℝ) (hβ : 0 < β) :
  42     ∃ C η₀ : ℝ, 0 < C ∧ 0 < η₀ ∧ ∀ η : ℝ, 0 < η → η < η₀ →
  43       ∃ m₀ : ℕ, ∀ (V : Type) [Fintype V] (ℓ k : ℕ) (d : Fin k → ℕ)
  44       (G : SimpleGraph V) (σ : Fin (ℓ + 2) → Equiv.Perm V) (P : FixedPartition V k d)
  45       (a : Fin (ℓ + 2) ⊕ Fin (ℓ + 2) → Fin k) (m : ℕ) (hm : ∀ s, d (a s) = m),
  46       0 < m → m₀ ≤ m → ∀ d₀ α : ℝ, 0 < α →
  47       (1/2+β) * m ≤ d₀ - η * m →
  48       (∀ s v, |(graphCellDegree G (Equiv.refl V) P v (a s) : ℝ) - d₀| ≤ η * m) →
  49       (∀ i v, |(graphCellDegree G (σ i) P v (a (Sum.inl (finRotate (ℓ + 2) i))) : ℝ) - d₀| ≤ η * m) →
  50       let q := ⌊d₀ - Real.sqrt η * m⌋₊
  51       ((ℓ + 2) + (2 * (ℓ + 2) - 1) * α) * m ≤ (2 * (ℓ + 2) - 1) * (q : ℝ) →
  52       ∃ w, PartitionTupleWeightConditions G σ P a (C * Real.sqrt η) q
  53         ((1 - 2 * (ℓ + 2) * (C * Real.sqrt η)) *
  54           (((ℓ + 2 : ℝ) * cyclicSectorLower ℓ α) / (m : ℝ) ^ (2 * ℓ + 2)))
  55         (((ℓ + 2 : ℝ) * cyclicSectorUpper ℓ α) / (m : ℝ) ^ (2 * ℓ + 2)) w := by
  56   obtain ⟨C,η₀,hC,hη₀,hall⟩ := near_regular_cyclic_matrix_weights β hβ
  57   refine ⟨C,η₀,hC,hη₀,?_⟩
  58   intro η hη hηsmall
  59   obtain ⟨m₀,hm₀⟩ := hall η hη hηsmall
  60   refine ⟨m₀,?_⟩
  61   intro V _ ℓ k d G σ P a m hm hmpos hmlarge d₀ α hα hmin hid hσ q hthreshold
  62   letI : NeZero m := ⟨by omega⟩
  63   let M := partitionCyclicMatrices G σ P a m hm
  64   obtain ⟨hl,hr⟩ := partitionCyclicMatrices_near_degrees G σ P a m hm d₀ η hid hσ
  65   have hb := partitionCyclicMatrices_binary G σ P a m hm
  66   have ht := hm₀ ℓ (Fin m) (by simpa using hmlarge) M hb d₀ α hα
  67     (by intro i x; simpa only [Fintype.card_fin] using hmin.trans (hl i x).1)
  68     (by intro i y; simpa only [Fintype.card_fin] using hmin.trans (hr i y).1)
  69     (by simpa only [Fintype.card_fin] using hl)
  70     (by simpa only [Fintype.card_fin] using hr)
  71     (by simpa only [Fintype.card_fin] using hthreshold)
  72   simp only [Fintype.card_fin] at ht
  73   obtain ⟨w,hwn,hws,hwl,hwr,hwp,hwu,hwb⟩ := ht
  74   let W := embeddedWeight (partitionCyclicTupleEmbedding P a m hm) w
  75   have hB : 0 ≤ ((ℓ + 2 : ℝ) * cyclicSectorUpper ℓ α) / (m : ℝ) ^ (2 * ℓ + 2) := by
  76     exact le_trans (hwn (fun _ => (0,0))) (hwu _)
  77   obtain ⟨hbound,hsupport,hlower⟩ := partitionTuple_weight_support_and_bounds G σ P a m hm
  78     w _ _ hB hwn hws hwb hwu
  79   refine ⟨W,⟨fun t => (hbound t).1, hsupport, hlower, fun t => (hbound t).2, ?_, ?_, ?_⟩⟩
  80   · exact partitionTuple_left_edge_bounds G σ P a m hm w (C * Real.sqrt η) (by
  81       intro i x y
  82       convert hwl i x y using 2 <;> apply Finset.sum_congr rfl <;> intro v _ <;> split_ifs <;> rfl)
  83   · exact partitionTuple_cross_edge_bounds G σ P a m hm w (C * Real.sqrt η) (by
  84       intro i x y
  85       convert hwr i x y using 2 <;> apply Finset.sum_congr rfl <;> intro v _ <;> split_ifs <;> rfl)
  86   · exact partitionTuple_position_loads P a m hm w q (by
  87       intro i x
  88       convert hwp i x using 2 <;> apply Finset.sum_congr rfl <;> intro v _ <;> split_ifs <;> rfl)
