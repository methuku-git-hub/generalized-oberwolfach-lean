  10 theorem factor_nonempty_class_indices {W B:Type} [Fintype W] [Fintype B] [DecidableEq B]
  11     (classify:W→B) (D:ℕ) (hD:0<D) (hsize:2*Fintype.card W=D):
  12     let fibres:=fun j:B=>Finset.univ.filter (fun w=>classify w=j)
  13     let C:=Finset.univ.filter (fun j=>(fibres j).Nonempty)
  14     let k:=C.card
  15     let e:Fin k≃↥C:=(Fintype.equivFinOfCardEq (Fintype.card_coe C)).symm
  16     let m:=fun a:Fin k=>(fibres (e a).val).card
  17     let ρ:=fun a:Fin k=>2*(m a:ℝ)/(D:ℝ);
  18     0<k ∧ (∀ a,0<m a) ∧ (∑ a,m a)=D/2 ∧
  19     (∀ a,0<ρ a) ∧ (∑ a,ρ a)=1 ∧ (∀ a,ρ a/2*D=m a):=by
  20   intro fibres C k e m ρ
  21   have hW:0<Fintype.card W:=by omega
  22   obtain ⟨w⟩:=Fintype.card_pos_iff.mp hW
  23   have hC:C.Nonempty:=by
  24     refine ⟨classify w,Finset.mem_filter.mpr ⟨Finset.mem_univ _,?_⟩⟩
  25     exact ⟨w,Finset.mem_filter.mpr ⟨Finset.mem_univ _,rfl⟩⟩
  26   have hk:0<k:=Finset.card_pos.mpr hC
  27   have hm:∀ a,0<m a:=by
  28     intro a
  29     exact Finset.card_pos.mpr (Finset.mem_filter.mp (e a).property).2
  30   have hsumall:(∑ j:B,(fibres j).card)=Fintype.card W:=by
  31     simpa [fibres] using Finset.sum_card_fiberwise_eq_card_filter
  32       (Finset.univ:Finset W) (Finset.univ:Finset B) classify
  33   have hsumC:(∑ j:↥C,(fibres j.val).card)=Fintype.card W:=by
  34     rw [Finset.sum_coe_sort C (fun j=>(fibres j).card)]
  35     calc
  36       _ = ∑ j:B,(fibres j).card:=by
  37         apply Finset.sum_subset (Finset.subset_univ C)
  38         intro j _ hj
  39         have hempty:¬(fibres j).Nonempty:=by
  40           intro h
  41           apply hj
  42           exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,h⟩
  43         exact Finset.card_eq_zero.mpr (Finset.not_nonempty_iff_eq_empty.mp hempty)
  44       _ = _:=hsumall
  45   have hsumm:(∑ a,m a)=Fintype.card W:=by
  46     calc
  47       _ = ∑ j:↥C,(fibres j.val).card:=e.sum_comp (fun j=>(fibres j.val).card)
  48       _ = _:=hsumC
  49   have hD':(0:ℝ)<D:=by exact_mod_cast hD
  50   have hmR:(∑ a:Fin k,(m a:ℝ))=Fintype.card W:=by exact_mod_cast hsumm
  51   have hsize':(2:ℝ)*Fintype.card W=D:=by exact_mod_cast hsize
  52   refine ⟨hk,hm,?_,?_,?_,?_⟩
  53   · rw [hsumm]
  54     omega
  55   · intro a
  56     dsimp [ρ]
  57     have hh:(0:ℝ)<m a:=by exact_mod_cast hm a
  58     positivity
  59   · dsimp [ρ]
  60     rw [←Finset.sum_div,←Finset.mul_sum,hmR,hsize']
  61     exact div_self (ne_of_gt hD')
  62   · intro a
  63     dsimp [ρ]
  64     field_simp
  65 end Oberwolfach
