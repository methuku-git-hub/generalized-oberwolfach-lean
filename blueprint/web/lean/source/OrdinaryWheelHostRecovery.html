  21 theorem ordinary_decomposition_recover_host (Q F : SimpleGraph V) (H : V → U → Prop)
  22     (L : Finset (Sym2 V⊕(V×U))) (hL : L⊆ordinaryHostResources Q H)
  23     (D : Finset (UniversalOrdinaryWheel V U r))
  24     (hD : (∀ W∈D,∀ W'∈D,W≠W'→Disjoint (ordinaryWheelResources W) (ordinaryWheelResources W')) ∧
  25       D.biUnion ordinaryWheelResources=L)
  26     (hF : ∀ W∈D,∀ v∈(ordinaryWheelRim W).support,∀ w∈(ordinaryWheelRim W).support,¬F.Adj v w) :
  27     HasResourceDecomposition (ordinaryWheelResources (P:=Q) (F:=F) (Hbase:=H) (r:=r)) L := by
  28   have hsub (W : D) : ordinaryWheelResources W.val⊆ordinaryHostResources Q H := by
  29     intro e he
  30     apply hL
  31     rw [←hD.2]
  32     exact Finset.mem_biUnion.mpr ⟨W.val,W.property,he⟩
  33   let f : D → OrdinaryWheel Q F H r := fun W=>ordinaryWheelInHost Q F H W.val (hsub W) (hF W.val W.property)
  34   refine ⟨Finset.univ.image f,?_,?_⟩
  35   · intro A hA B hB hAB
  36     obtain ⟨W,_,rfl⟩ := Finset.mem_image.mp hA
  37     obtain ⟨W',_,rfl⟩ := Finset.mem_image.mp hB
  38     exact hD.1 W.val W.property W'.val W'.property (fun he=>hAB (congrArg f (Subtype.ext he)))
  39   · ext e
  40     rw [←hD.2]
  41     simp only [Finset.mem_biUnion,Finset.mem_image,Finset.mem_univ,true_and]
  42     constructor
  43     · rintro ⟨W,⟨W',rfl⟩,he⟩
  44       exact ⟨W'.val,W'.property,he⟩
  45     · rintro ⟨W,hW,he⟩
  46       exact ⟨f ⟨W,hW⟩,⟨⟨W,hW⟩,rfl⟩,he⟩
