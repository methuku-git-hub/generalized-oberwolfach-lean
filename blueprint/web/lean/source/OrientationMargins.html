  12 theorem orientation_failure_sum_bound
  13     {V : Type*} [Fintype V] (G E : SimpleGraph V)
  14     (hdeg : ∀ v, 0 < G.degree v) {a α : ℝ}
  15     (hcommon : ∀ x y, x ≠ y → 2 * α * Fintype.card V ≤
  16       ((E.neighborFinset x ∩ E.neighborFinset y).card : ℝ)) :
  17     ((∑ x : V, 2 * Real.exp (-2 * a ^ 2 / G.degree x)) +
  18       (∑ p : {p : V × V // p.1 ≠ p.2},
  19         2 * Real.exp (-((E.neighborFinset p.1.1 ∩ E.neighborFinset p.1.2).card : ℝ) / 64))) ≤
  20       2 * Fintype.card V * Real.exp (-2 * a ^ 2 / Fintype.card V) +
  21       2 * (Fintype.card V : ℝ) ^ 2 * Real.exp (-α * Fintype.card V / 32) := by
  22   have hd : (∑ x : V, 2 * Real.exp (-2 * a ^ 2 / G.degree x)) ≤
  23       ∑ _x : V, 2 * Real.exp (-2 * a ^ 2 / Fintype.card V) := by
  24     apply Finset.sum_le_sum
  25     intro x _
  26     apply mul_le_mul_of_nonneg_left _ (by norm_num)
  27     apply Real.exp_le_exp.mpr
  28     have hpos : (0 : ℝ) < G.degree x := by exact_mod_cast hdeg x
  29     have hle : (G.degree x : ℝ) ≤ Fintype.card V := by
  30       exact_mod_cast (G.degree_lt_card_verts x).le
  31     have hh := div_le_div_of_nonneg_left (show 0 ≤ (2 : ℝ) * a ^ 2 by positivity) hpos hle
  32     simpa only [neg_mul, neg_div] using neg_le_neg hh
  33   have hp : (∑ p : {p : V × V // p.1 ≠ p.2},
  34       2 * Real.exp (-((E.neighborFinset p.1.1 ∩ E.neighborFinset p.1.2).card : ℝ) / 64)) ≤
  35       ∑ _p : {p : V × V // p.1 ≠ p.2},
  36         2 * Real.exp (-α * Fintype.card V / 32) := by
  37     apply Finset.sum_le_sum
  38     intro p _
  39     apply mul_le_mul_of_nonneg_left _ (by norm_num)
  40     apply Real.exp_le_exp.mpr
  41     linarith [hcommon p.1.1 p.1.2 p.2]
  42   have hcard : (Fintype.card {p : V × V // p.1 ≠ p.2} : ℝ) ≤ (Fintype.card V : ℝ) ^ 2 := by
  43     have hh := Fintype.card_subtype_le (fun p : V × V => p.1 ≠ p.2)
  44     simpa only [Fintype.card_prod, Nat.cast_mul, pow_two] using (Nat.cast_le (α := ℝ)).mpr hh
  45   simp only [Finset.sum_const, Finset.card_univ, nsmul_eq_mul] at hd hp
  46   have hlast := mul_le_mul_of_nonneg_right hcard
  47     (show 0 ≤ 2 * Real.exp (-α * Fintype.card V / 32) by positivity)
  48   nlinarith

  50 theorem threeFifths_tail_exponent {n : ℝ} (hn : 0 < n) :
  51     -2 * (n ^ (3 / 5 : ℝ)) ^ 2 / n = -2 * n ^ (1 / 5 : ℝ) := by
  52   have hp : (n ^ (3 / 5 : ℝ)) ^ 2 / n = n ^ (1 / 5 : ℝ) := by
  53     rw [← Real.rpow_natCast, ← Real.rpow_mul hn.le, ← Real.rpow_sub_one hn.ne']
  54     norm_num
  55   calc
  56     _ = -2 * ((n ^ (3 / 5 : ℝ)) ^ 2 / n) := by ring
  57     _ = _ := by rw [hp]

  59 theorem eventually_orientation_failure_small_pow {α : ℝ} (hα : 0 < α) (q : ℕ) :
  60     ∃ N : ℕ, ∀ n : ℕ, N ≤ n →
  61       2 * n * Real.exp (-2 * (n : ℝ) ^ (1 / 5 : ℝ)) +
  62       2 * (n : ℝ) ^ 2 * Real.exp (-α * n / 32) ≤ 1 / (n : ℝ) ^ q := by
  63   obtain ⟨N₁, h₁⟩ := eventually_stretchedExp_le_inv_pow 4 1
  64     (c := 2) (p := 1 / 5) (by norm_num) (by norm_num) q
  65   obtain ⟨N₂, h₂⟩ := eventually_stretchedExp_le_inv_pow 4 2
  66     (c := α / 32) (p := 1) (by positivity) (by norm_num) q
  67   refine ⟨max N₁ N₂, fun n hn => ?_⟩
  68   have h1 := h₁ n (le_trans (le_max_left _ _) hn)
  69   have h2 := h₂ n (le_trans (le_max_right _ _) hn)
  70   simp only [Real.rpow_one, Real.rpow_two] at h1 h2
  71   have heq : -(α / 32) * (n : ℝ) = -α * n / 32 := by ring
  72   rw [heq] at h2

  81 theorem orientation_numeric_bounds {n α : ℝ} (hn : 1 ≤ n)
  82     (hb : 2 ≤ n ^ (4 / 5 : ℝ))
  83     (hsmall : 7 * n ^ (4 / 5 : ℝ) + 4 < 23 * α * n / 100) :
  84     0 < Nat.ceil (n ^ (4 / 5 : ℝ)) ∧
  85     n * n ^ (3 / 5 : ℝ) / Nat.ceil (n ^ (4 / 5 : ℝ)) +
  86         2 * n ^ (3 / 5 : ℝ) + 4 * Nat.ceil (n ^ (4 / 5 : ℝ)) + 2 * (α * n / 100) <
  87       (α / 4) * n ∧
  88     n ^ (3 / 5 : ℝ) + 2 * Nat.ceil (n ^ (4 / 5 : ℝ)) ≤ 4 * n ^ (4 / 5 : ℝ) := by
  89   have hnpos : 0 < n := lt_of_lt_of_le zero_lt_one hn
  90   have hLpos : 0 < Nat.ceil (n ^ (4 / 5 : ℝ)) := Nat.ceil_pos.mpr (Real.rpow_pos_of_pos hnpos _)
  91   have hLr : (0 : ℝ) < Nat.ceil (n ^ (4 / 5 : ℝ)) := by exact_mod_cast hLpos
  92   have ha : n ^ (3 / 5 : ℝ) ≤ n ^ (4 / 5 : ℝ) :=
  93     Real.rpow_le_rpow_of_exponent_le hn (by norm_num)
  94   have hceil := Nat.le_ceil (n ^ (4 / 5 : ℝ))
  95   have hceil' := Nat.ceil_lt_add_one (Real.rpow_nonneg hnpos.le (4 / 5))
  96   have heq : n * n ^ (3 / 5 : ℝ) = (n ^ (4 / 5 : ℝ)) ^ 2 := by
  97     rw [← Real.rpow_natCast, ← Real.rpow_mul hnpos.le]
  98     have hsum := Real.rpow_add hnpos 1 (3 / 5)
  99     norm_num only [Real.rpow_one] at hsum
 100     rw [← hsum]
 101     norm_num
 102   have hdiv : n * n ^ (3 / 5 : ℝ) / Nat.ceil (n ^ (4 / 5 : ℝ)) ≤ n ^ (4 / 5 : ℝ) := by
 103     apply (div_le_iff₀ hLr).mpr
 104     rw [heq]
 105     nlinarith [Real.rpow_nonneg hnpos.le (4 / 5)]
 106   exact ⟨hLpos, by nlinarith, by linarith⟩

 108 theorem eventually_orientation_numeric {α : ℝ} (hα : 0 < α) :
 109     ∃ N : ℕ, ∀ n : ℕ, N ≤ n →
 110       0 < n ∧ 0 < Nat.ceil ((n : ℝ) ^ (4 / 5 : ℝ)) ∧
 111       (n : ℝ) * (n : ℝ) ^ (3 / 5 : ℝ) / Nat.ceil ((n : ℝ) ^ (4 / 5 : ℝ)) +
 112           2 * (n : ℝ) ^ (3 / 5 : ℝ) + 4 * Nat.ceil ((n : ℝ) ^ (4 / 5 : ℝ)) +
 113           2 * (α * n / 100) < (α / 4) * n ∧
 114       (n : ℝ) ^ (3 / 5 : ℝ) + 2 * Nat.ceil ((n : ℝ) ^ (4 / 5 : ℝ)) ≤
 115         4 * (n : ℝ) ^ (4 / 5 : ℝ) := by
 116   have h₁ := (tendsto_rpow_neg_atTop (y := 1 / 5) (by norm_num)).const_mul 7
 117   have h₂ := (tendsto_rpow_neg_atTop (y := 1) (by norm_num)).const_mul 4
 118   have hh := (h₁.add h₂).comp tendsto_natCast_atTop_atTop
 119   simp only [mul_zero, add_zero] at hh
 120   have hs : ∀ᶠ n : ℕ in atTop,
 121       7 * (n : ℝ) ^ (-(1 / 5 : ℝ)) + 4 * (n : ℝ) ^ (-(1 : ℝ)) < 23 * α / 100 :=
 122     hh.eventually (eventually_lt_nhds (by positivity))
 123   have ht := (tendsto_rpow_atTop (y := (4 / 5 : ℝ)) (by norm_num)).comp tendsto_natCast_atTop_atTop
 124   have hb : ∀ᶠ n : ℕ in atTop, 2 ≤ (n : ℝ) ^ (4 / 5 : ℝ) := ht.eventually (eventually_ge_atTop 2)
 125   have hall : ∀ᶠ n : ℕ in atTop,
 126       0 < n ∧ 0 < Nat.ceil ((n : ℝ) ^ (4 / 5 : ℝ)) ∧
 127       (n : ℝ) * (n : ℝ) ^ (3 / 5 : ℝ) / Nat.ceil ((n : ℝ) ^ (4 / 5 : ℝ)) +
 128           2 * (n : ℝ) ^ (3 / 5 : ℝ) + 4 * Nat.ceil ((n : ℝ) ^ (4 / 5 : ℝ)) +
 129           2 * (α * n / 100) < (α / 4) * n ∧
 130       (n : ℝ) ^ (3 / 5 : ℝ) + 2 * Nat.ceil ((n : ℝ) ^ (4 / 5 : ℝ)) ≤
 131         4 * (n : ℝ) ^ (4 / 5 : ℝ) := by
 132     filter_upwards [hs, hb, eventually_ge_atTop 1] with n hs hb hn
 133     have hnpos : (0 : ℝ) < n := by exact_mod_cast (lt_of_lt_of_le Nat.zero_lt_one hn)
 134     have hnreal : (1 : ℝ) ≤ n := by exact_mod_cast hn
 135     have heq : (n : ℝ) ^ (-(1 / 5 : ℝ)) = (n : ℝ) ^ (4 / 5 : ℝ) / n := by
 136       rw [← Real.rpow_sub_one hnpos.ne']
 137       norm_num
 138     have hsmall : 7 * (n : ℝ) ^ (4 / 5 : ℝ) + 4 < 23 * α * n / 100 := by
 139       rw [heq, Real.rpow_neg_one] at hs
 140       have hmul := mul_lt_mul_of_pos_right hs hnpos
 141       field_simp at hmul
 142       nlinarith
 143     exact ⟨by omega, orientation_numeric_bounds hnreal hb hsmall⟩
 144   exact eventually_atTop.mp hall
