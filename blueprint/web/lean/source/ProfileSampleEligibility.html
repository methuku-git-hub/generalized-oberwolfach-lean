  17 def profileRequiredRoots (F Z X : Finset (Σ i,V i)) (i : I) : Finset (ProfileRemainingClass F i) :=
  18   Finset.univ.filter (fun v => (⟨i,v.val⟩ : Σ i,V i)∈X ∧ (⟨i,v.val⟩ : Σ i,V i)∉Z)

  20 theorem profileRequiredRoots_map_subset (F Z X : Finset (Σ i,V i)) :
  21     (Finset.univ.sigma (profileRequiredRoots F Z X)).map (profileComplementEmbedding F)⊆X := by
  22   intro x hx
  23   obtain ⟨⟨i,v⟩,hv,rfl⟩ := Finset.mem_map.mp hx
  24   exact (Finset.mem_filter.mp (Finset.mem_sigma.mp hv).2).2.1

  27     (profileRequiredRoots F Z X i).card≤X.card := by

  33 theorem profileRequiredRoots_eligibility (F Z X : Finset (Σ i,V i))
  34     (hcompat : X∩F⊆Z) (m : I → ℕ) (σ : ∀ i,FixedSubset (ProfileRemainingClass F i) (m i)) :
  35     X⊆(profileRootSet σ).map (profileComplementEmbedding F)∪Z ↔
  36       ∀ i,profileRequiredRoots F Z X i⊆(σ i).val := by
  37   constructor
  38   · intro h i v hv
  39     obtain ⟨hvX,hvZ⟩ := (Finset.mem_filter.mp hv).2
  40     have hvS := (Finset.mem_union.mp (h hvX)).resolve_right hvZ
  41     have hvσ : (⟨i,v⟩ : Σ j,ProfileRemainingClass F j)∈profileRootSet σ := by
  42       obtain ⟨u,hu,he⟩ := Finset.mem_map.mp hvS
  43       have he' : u=(⟨i,v⟩ : Σ j,ProfileRemainingClass F j) := (profileComplementEmbedding F).injective he
  44       simpa only [he'] using hu
  45     exact (profileRootSet_mem σ i v).mp hvσ
  46   · intro h x hx
  47     by_cases hz : x∈Z
  48     · exact Finset.mem_union_right _ hz
  49     · have hf : x∉F := fun hf => hz (hcompat (Finset.mem_inter.mpr ⟨hx,hf⟩))
  50       obtain ⟨u,hu⟩ := (profileComplementEmbedding_range F x).mpr hf
  51       rcases u with ⟨i,v⟩
  52       have hv : v∈profileRequiredRoots F Z X i := by
  53         apply Finset.mem_filter.mpr
  54         exact ⟨Finset.mem_univ _,by change profileComplementEmbedding F ⟨i,v⟩∈X ∧ profileComplementEmbedding F ⟨i,v⟩∉Z;simpa only [hu] using And.intro hx hz⟩
  55       exact Finset.mem_union_left _ (Finset.mem_map.mpr ⟨⟨i,v⟩,(profileRootSet_mem σ i v).mpr (h i hv),hu⟩)
