  14 theorem ordinary_pair_common_geometry (Q:SimpleGraph V) (H:V→U→Prop) (β:ℝ)
  15     (hpair:∀ u u' v w,β*Fintype.card V≤
  16       (((Finset.univ.filter (fun a=>H a u∧H a u'))∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ)):
  17     ∀ u u':U,∀ S:Finset V,S.card≤2→
  18       β*Fintype.card V≤((common (Finset.univ.filter (fun v=>H v u∧H v u')) S
  19         (fun x=>Q.neighborFinset x)).card:ℝ) := by
  20   intro u u' S hS
  21   have hpad:∃ v w:V,S⊆{v,w}:=by
  22     by_cases h:S.Nonempty
  23     · obtain ⟨v,hv⟩:=h
  24       have hc:(S.erase v).card≤1:=by
  25         have he:=Finset.card_erase_of_mem hv
  26         omega
  27       obtain ⟨w,hw⟩:=Finset.card_le_one_iff_subset_singleton.mp hc
  28       refine ⟨v,w,?_⟩
  29       intro x hx
  30       by_cases h:x=v
  31       · simp [h]
  32       · have hx':x∈S.erase v:=Finset.mem_erase.mpr ⟨h,hx⟩
  33         have hxw:=hw hx'
  34         simpa using Or.inr (Finset.mem_singleton.mp hxw)
  35     · let v:V:=Classical.arbitrary V
  36       exact ⟨v,v,by simp [Finset.not_nonempty_iff_eq_empty.mp h]⟩
  37   obtain ⟨v,w,hsub⟩:=hpad
  38   have hinc:
  39       (Finset.univ.filter (fun a=>H a u∧H a u'))∩Q.neighborFinset v∩Q.neighborFinset w ⊆
  40       common (Finset.univ.filter (fun a=>H a u∧H a u')) S (fun x=>Q.neighborFinset x) := by
  41     intro x hx
  42     simp only [Finset.mem_inter,Finset.mem_filter,Finset.mem_univ,true_and,SimpleGraph.mem_neighborFinset] at hx
  43     simp only [common,Finset.mem_filter,Finset.mem_univ,true_and]
  44     refine ⟨hx.1.1,?_⟩
  45     intro a ha
  46     have ha':a=v∨a=w:=by simpa using hsub ha
  47     rcases ha' with rfl|rfl
  48     · simpa only [SimpleGraph.mem_neighborFinset] using hx.1.2
  49     · simpa only [SimpleGraph.mem_neighborFinset] using hx.2
  50   exact (hpair u u' v w).trans (by exact_mod_cast Finset.card_le_card hinc)
