   9 theorem factor_component_cycles_disjoint_from_reserved_paths
  10     {X I J:Type} [Fintype X] [DecidableEq X]
  11     (G:SimpleGraph X) (pathComponent:I→G.ConnectedComponent) (cycleComponent:J→G.ConnectedComponent)
  12     (m:I→ℕ) (ext:∀i,Fin (m i)→X)
  13     (u v:I→X) (P:∀i,G.Walk (u i) (v i))
  14     (hpath:∀i,(P i).support=List.ofFn (ext i))
  15     (hpathComponent:∀i k,ext i k∈(pathComponent i).supp)
  16     (root:J→X) (C:∀j,G.Walk (root j) (root j))
  17     (hcycleComponent:∀j,(C j).toSubgraph.verts=(cycleComponent j).supp)
  18     (hseparate:∀i j,pathComponent i≠cycleComponent j):
  19     ∀i j,Disjoint (P i).support.toFinset (C j).support.toFinset:=by
  20   intro i j
  21   apply Finset.disjoint_left.mpr
  22   intro x hx hy
  23   have hx':x∈(pathComponent i).supp:=by
  24     have hh: x∈List.ofFn (ext i):=by simpa only [←hpath i] using List.mem_toFinset.mp hx
  25     obtain ⟨k,hk⟩:=List.mem_ofFn.mp hh
  26     rw [←hk]
  27     exact hpathComponent i k
  28   have hy':x∈(cycleComponent j).supp:=by
  29     rw [←hcycleComponent j]
  30     exact (C j).mem_verts_toSubgraph.mpr (List.mem_toFinset.mp hy)
  31   exact Set.disjoint_left.mp (G.pairwise_disjoint_supp_connectedComponent (hseparate i j)) hx' hy'
  32 end Oberwolfach
