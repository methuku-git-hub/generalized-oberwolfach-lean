  20 abbrev DenseMarkedBaseFamily := {f : Fin (m+1)→V // DenseMarkedBase P π b T₀ TK f}

  22 def markedBaseRequiredEdges (C : DenseMarkedBaseFamily P π b m T₀ TK) : Finset (Sym2 V) :=
  23   Finset.univ.image (twistedTupleEdge C.val (π (C.val 0)))

  25 def markedBasePositionState (C : DenseMarkedBaseFamily P π b m T₀ TK) :
  26     Fin m⊕Unit→Option (Bool×Bool)
  27   | Sum.inl i=>some (false,orientationBit (C.val i.castSucc) (C.val i.succ))
  28   | Sum.inr _=>some (true,orientationBit (C.val (Fin.last m)) (π (C.val 0)))

  30 def markedBaseRequiredState (C : DenseMarkedBaseFamily P π b m T₀ TK)
  31     (e : Sym2 V) : Option (Bool×Bool) :=
  32   if h : ∃ i,twistedTupleEdge C.val (π (C.val 0)) i=e then
  33     markedBasePositionState P π b m T₀ TK C (Classical.choose h) else none

  35 theorem markedBaseRequiredState_apply (hb : 1≤b) (hm : 2≤m)
  36     (C : DenseMarkedBaseFamily P π b m T₀ TK) (i : Fin m⊕Unit) :
  37     markedBaseRequiredState P π b m T₀ TK C (twistedTupleEdge C.val (π (C.val 0)) i)=
  38       markedBasePositionState P π b m T₀ TK C i := by
  39   unfold markedBaseRequiredState
  40   rw [dif_pos ⟨i,rfl⟩]
  41   congr 1
  42   exact denseMarkedBase_physical_injective P π b m hb hm T₀ TK C.val C.property
  43     (Classical.choose_spec (show ∃ j,twistedTupleEdge C.val (π (C.val 0)) j=
  44       twistedTupleEdge C.val (π (C.val 0)) i from ⟨i,rfl⟩))

  52 theorem markedBaseRequiredEdges_tail_in_range
  53     (C : DenseMarkedBaseFamily P π b m T₀ TK) (e : Sym2 V)
  54     (he : e∈markedBaseRequiredEdges P π b m T₀ TK C) :
  55     e.out.1∈Set.range C.val ∨ e.out.2∈Set.range C.val := by
  56   obtain ⟨i,_,hi⟩ := Finset.mem_image.mp he
  57   have hout := e.out_eq
  58   cases i with
  59   | inl i=>
  60     change s(C.val i.castSucc,C.val i.succ)=e at hi
  61     rw [← hout] at hi
  62     rcases Sym2.eq_iff.mp hi with ⟨hx,hy⟩|⟨hx,hy⟩
  63     · exact Or.inl ⟨i.castSucc,hx⟩
  64     · exact Or.inr ⟨i.castSucc,hx⟩
  65   | inr i=>
  66     change s(C.val (Fin.last m),π (C.val 0))=e at hi
  67     rw [← hout] at hi
  68     rcases Sym2.eq_iff.mp hi with ⟨hx,hy⟩|⟨hx,hy⟩
  69     · exact Or.inl ⟨Fin.last m,hx⟩
  70     · exact Or.inr ⟨Fin.last m,hx⟩
