  15 theorem positive_ordinary_cycle_weights_with_pair_loads
  16     (r : ℕ) (hr : 3 ≤ r) (α : ℝ) (hα : 0 < α) :
  17     ∃ c C : ℝ, 0 < c ∧ 0 < C ∧ ∃ N : ℕ, 0 < N ∧
  18       ∀ (V : Type) [Fintype V] [DecidableEq V], N ≤ Fintype.card V →
  19       ∀ G : SimpleGraph V,
  20         (∀ v,(cycleThreshold r+α)*Fintype.card V ≤ (G.degree v:ℝ)) →
  21       ∃ a : OrdinaryCycle G r → ℝ,
  22         (∀ D,c/(Fintype.card V:ℝ)^(r-2) ≤ a D ∧
  23           a D ≤ C/(Fintype.card V:ℝ)^(r-2)) ∧
  24         (∀ e : G.edgeSet,∑ D,a D*ordinaryCycleIncidence G r D e=1) ∧
  25         (∀ x y z : V, x ≠ y → x ≠ z → y ≠ z →
  26           (∑ D,if D.val.val.Adj x y ∧ D.val.val.Adj x z then a D else 0) ≤
  27             C/Fintype.card V) ∧
  28         (∀ x y z t : V, x ≠ y → x ≠ z → x ≠ t → y ≠ z → y ≠ t → z ≠ t →
  29           (∑ D,if D.val.val.Adj x y ∧ D.val.val.Adj z t then a D else 0) ≤
  30             C/(Fintype.card V:ℝ)^2) := by
  31   obtain ⟨c,B,hc,hB,N,hN,hw⟩ := positive_ordinary_fractional_cycle_weights r hr α hα
  32   have hrR : (1:ℝ) ≤ r := by exact_mod_cast (show 1 ≤ r by omega)
  33   let C := (r:ℝ)^4*B
  34   have hC : 0<C := by dsimp [C];positivity
  35   have hBC : B≤C := by
  36     dsimp [C]
  37     have hpow : (1:ℝ) ≤ (r:ℝ)^4 := one_le_pow₀ hrR
  38     nlinarith
  39   have hB3 : (r:ℝ)^3*B ≤ C := by
  40     dsimp [C]
  41     have hpow : (r:ℝ)^3 ≤ (r:ℝ)^4 := by nlinarith [sq_nonneg ((r:ℝ)^2)]
  42     exact mul_le_mul_of_nonneg_right hpow hB.le
  43   refine ⟨c,C,hc,hC,N,hN,?_⟩
  44   intro V _ _ hn G hdeg
  45   have hn0 : 0<Fintype.card V := hN.trans_le hn
  46   have hnR : (0:ℝ)≤Fintype.card V := Nat.cast_nonneg _
  47   obtain ⟨a,ha,hload⟩ := hw V hn G hdeg
  48   refine ⟨a,?_,hload,?_,?_⟩
  49   · intro D
  50     exact ⟨(ha D).1,(ha D).2.trans (div_le_div_of_nonneg_right hBC (by positivity))⟩
  51   · intro x y z hxy hxz hyz
  52     apply (ordinary_cycle_adjacent_pair_load G r (by omega) hn0 a
  53       (fun D => le_trans (by positivity) (ha D).1) B hB.le (fun D => (ha D).2)
  54       x y z hxy hxz hyz).trans
  55     exact div_le_div_of_nonneg_right hB3 hnR
  56   · intro x y z t hxy hxz hxt hyz hyt hzt
  57     apply (ordinary_cycle_disjoint_pair_load G r (by omega) hn0 a
  58       (fun D => le_trans (by positivity) (ha D).1) B hB.le (fun D => (ha D).2)
  59       x y z t hxy hxz hxt hyz hyt hzt).trans
  60     exact div_le_div_of_nonneg_right (le_refl C) (by positivity)
