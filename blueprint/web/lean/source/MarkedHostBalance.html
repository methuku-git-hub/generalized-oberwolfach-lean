  21 theorem marked_unit_balance_of_spoke_degrees {V U E K : Type*} [Fintype V] [Fintype U]
  22     [Fintype E] [Fintype K] [DecidableEq V]
  23     (tail head : E → V) (ktail khead : K → V) (m : ℕ)
  24     (H₀ Hₖ : V → U → Prop)
  25     (hdir : ∀ v,occurrenceDegree (Sum.elim tail ktail) v=occurrenceDegree (Sum.elim head khead) v)
  26     (hK : ∀ v,occurrenceDegree khead v=∑ u,if Hₖ v u then (1:ℝ) else 0)
  27     (hout : ∀ v,occurrenceDegree (Sum.elim tail ktail) v=
  28       (∑ u,if H₀ v u then (1:ℝ) else 0)+(∑ u,if Hₖ v u then (1:ℝ) else 0))
  29     (hhub : ∀ u,(∑ v,if H₀ v u then (1:ℝ) else 0)=(m:ℝ)*(∑ v,if Hₖ v u then (1:ℝ) else 0)) :
  30     (fun _ : E⊕K => (1:ℝ))∈markedBalanceSubspace tail head ktail khead m := by
  31   constructor
  32   · funext v
  33     rw [occurrenceSumLinear_one,occurrenceSumLinear_one]
  34     exact hdir v
  35   · have hksum : (Fintype.card K:ℝ)=∑ v,∑ u,if Hₖ v u then (1:ℝ) else 0 := by
  36       rw [← occurrenceDegree_total khead]
  37       exact Finset.sum_congr rfl (fun v _ => hK v)
  38     have houtsum : (Fintype.card E:ℝ)+Fintype.card K=
  39         (∑ v,∑ u,if H₀ v u then (1:ℝ) else 0)+(∑ v,∑ u,if Hₖ v u then (1:ℝ) else 0) := by
  40       calc
  41         _ = (Fintype.card (E⊕K):ℝ) := by simp
  42         _ = ∑ v,occurrenceDegree (Sum.elim tail ktail) v := (occurrenceDegree_total _).symm
  43         _ = _ := by simp only [hout,Finset.sum_add_distrib]
  44     have hratio : (∑ v,∑ u,if H₀ v u then (1:ℝ) else 0)=
  45         (m:ℝ)*(∑ v,∑ u,if Hₖ v u then (1:ℝ) else 0) := by
  46       calc
  47         _ = ∑ u,∑ v,if H₀ v u then (1:ℝ) else 0 := Finset.sum_comm
  48         _ = ∑ u,(m:ℝ)*(∑ v,if Hₖ v u then (1:ℝ) else 0) := Finset.sum_congr rfl (fun u _ => hhub u)
  49         _ = (m:ℝ)*(∑ u,∑ v,if Hₖ v u then (1:ℝ) else 0) := by rw [Finset.mul_sum]
  50         _ = _ := by rw [Finset.sum_comm]
  51     change (∑ _e : E,(1:ℝ))=(m:ℝ)*(∑ _e : K,(1:ℝ))
  52     simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul,mul_one]
  53     rw [← hksum] at hratio houtsum
  54     rw [hratio] at houtsum
