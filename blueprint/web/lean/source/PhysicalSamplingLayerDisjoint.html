  13 theorem marked_sample_graph_edge_has_physical_state
  14     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ) (raw:RawMarkedOutcome V U Γ Color)
  15     (x y:V) (h:(markedSampleArcSetGraph P π d raw Finset.univ).Adj x y):
  16     ∃b o,raw (Sum.inl s(x,y))=some (b,o):=by
  17   change x≠y ∧ ({x,y}:Finset V)∈
  18     (Finset.univ.image (markedPhysicalEdge
  19       (fun e:MarkedSampleArc P π (3*d) (fun e=>raw (Sum.inl e)) false=>e.val.1)
  20       (fun e:MarkedSampleArc P π (3*d) (fun e=>raw (Sum.inl e)) false=>e.val.2)
  21       (fun e:MarkedSampleArc P π (3*d) (fun e=>raw (Sum.inl e)) true=>e.val.1)
  22       (fun e:MarkedSampleArc P π (3*d) (fun e=>raw (Sum.inl e)) true=>e.val.2) π)) at h
  23   obtain ⟨a,_,ha⟩:=Finset.mem_image.mp h.2
  24   have hpair {a b c e:V} (hh:({a,b}:Finset V)={c,e}):s(a,b)=s(c,e):=by
  25     apply Sym2.eq_iff.mpr
  26     apply Set.pair_eq_pair_iff.mp
  27     simpa only [Finset.coe_pair] using congrArg (fun S:Finset V=>(S:Set V)) hh
  28   cases a with
  29   | inl e=>
  30     have hval:raw (Sum.inl s(e.val.1,e.val.2))=some (false,orientationBit e.val.1 e.val.2):=
  31       ((markedPhysicalRim_eq_one_iff P π (3*d) _ false e.val.1 e.val.2).mp e.property).2.2
  32     have hp:s(e.val.1,e.val.2)=s(x,y):=hpair (by simpa [markedPhysicalEdge] using ha)
  33     exact ⟨false,orientationBit e.val.1 e.val.2,hp ▸ hval⟩
  34   | inr e=>
  35     have hval:raw (Sum.inl s(e.val.1,π e.val.2))=some (true,orientationBit e.val.1 (π e.val.2)):=
  36       ((markedPhysicalRim_eq_one_iff P π (3*d) _ true e.val.1 e.val.2).mp e.property).2.2
  37     have hp:s(e.val.1,π e.val.2)=s(x,y):=hpair (by simpa [markedPhysicalEdge] using ha)
  38     exact ⟨true,orientationBit e.val.1 (π e.val.2),hp ▸ hval⟩

  40 theorem selected_and_approximate_rim_graphs_disjoint
  41     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ)
  42     (σ:∀e:RawWheelCoordinate V U,RawExpandedMarkedCoordinateState Γ Color e):
  43     Disjoint (longSelectedPhysicalGraph P π d σ) (longApproximatePhysicalGraph P π d σ):=by
  44   apply disjoint_iff_inf_le.mpr
  45   intro x y h
  46   obtain ⟨b,o,hsel⟩:=marked_sample_graph_edge_has_physical_state P π d (fun e=>rawExpandedMarkedCollapse e (σ e)) x y h.1
  47   obtain ⟨b',o',happ⟩:=marked_sample_graph_edge_has_physical_state P π d (fun e=>rawExpandedApproximateCollapse e (σ e)) x y h.2
  48   have hh:¬(markedExpandedPhysicalCollapse (σ (Sum.inl s(x,y)))≠none ∧
  49       markedExpandedApproximateCollapse (σ (Sum.inl s(x,y)))≠none):=by
  50     rcases σ (Sum.inl s(x,y)) with ⟨z,o⟩
  51     fin_cases z <;> simp [markedExpandedPhysicalCollapse,markedExpandedApproximateCollapse]
  52   apply hh
  53   constructor
  54   · simp only [rawExpandedMarkedCollapse] at hsel
  55     rw [hsel];exact Option.some_ne_none _
  56   · simp only [rawExpandedApproximateCollapse] at happ
  57     rw [happ];exact Option.some_ne_none _
