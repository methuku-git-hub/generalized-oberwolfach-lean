  10 theorem ordinary_cycle_sampling_scaled_probability {V:Type} [Fintype V] [DecidableEq V]
  11     (G:SimpleGraph V) (r:ℕ) (hr:3≤r) (hn:0<Fintype.card V) (B:ℝ) (hB:0<B)
  12     (w:OrdinaryCycle G r→ℝ)
  13     (hw:∀ C,0≤w C ∧ w C≤B/(Fintype.card V:ℝ)^(r-2)):
  14     ∀ C,0≤((Fintype.card V:ℝ)/(2*B))*w C ∧ ((Fintype.card V:ℝ)/(2*B))*w C≤1/2 := by
  15   intro C
  16   have hn1:(1:ℝ)≤Fintype.card V:=by exact_mod_cast hn
  17   have hnpos:(0:ℝ)<Fintype.card V:=by exact_mod_cast hn
  18   have hpow:(Fintype.card V:ℝ)≤(Fintype.card V:ℝ)^(r-2):=by
  19     calc
  20       _ = (Fintype.card V:ℝ)^1:=by simp
  21       _ ≤ (Fintype.card V:ℝ)^(r-2):=pow_le_pow_right₀ hn1 (by omega)
  22   have hp:0<(Fintype.card V:ℝ)^(r-2):=pow_pos hnpos _
  23   have htotal:w C*(Fintype.card V:ℝ)^(r-2)≤B:=(le_div_iff₀ hp).mp (hw C).2
  24   have hh:=mul_le_mul_of_nonneg_right hpow (hw C).1
  25   have hnum:(Fintype.card V:ℝ)*w C≤B:=by nlinarith
  26   have hden:0<2*B:=by positivity
  27   have hratio:((Fintype.card V:ℝ)*w C)/(2*B)≤1/2:=
  28     (div_le_iff₀ hden).2 (by nlinarith)
  29   constructor
  30   · exact mul_nonneg (div_nonneg hnpos.le hden.le) (hw C).1
  31   · convert hratio using 1 <;> ring
