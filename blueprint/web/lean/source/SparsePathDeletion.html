  15 theorem two_common_neighbours_after_deletion (Q D : SimpleGraph V) (A : Finset V)
  16     (x y : V) :
  17     (A∩Q.neighborFinset x∩Q.neighborFinset y).card≤
  18       (A∩(Q\D).neighborFinset x∩(Q\D).neighborFinset y).card+D.degree x+D.degree y := by
  19   let C:=A∩(Q\D).neighborFinset x∩(Q\D).neighborFinset y
  20   have hs : A∩Q.neighborFinset x∩Q.neighborFinset y⊆C∪(D.neighborFinset x∪D.neighborFinset y):=by
  21     intro v hv
  22     have hvA:=(Finset.mem_inter.mp (Finset.mem_inter.mp hv).1).1
  23     have hx:=(Q.mem_neighborFinset x v).mp (Finset.mem_inter.mp (Finset.mem_inter.mp hv).1).2
  24     have hy:=(Q.mem_neighborFinset y v).mp (Finset.mem_inter.mp hv).2
  25     by_cases hDx:D.Adj x v
  26     · exact Finset.mem_union_right _ (Finset.mem_union_left _ ((D.mem_neighborFinset x v).mpr hDx))
  27     by_cases hDy:D.Adj y v
  28     · exact Finset.mem_union_right _ (Finset.mem_union_right _ ((D.mem_neighborFinset y v).mpr hDy))
  29     exact Finset.mem_union_left _ (Finset.mem_inter.mpr ⟨Finset.mem_inter.mpr
  30       ⟨hvA,((Q\D).mem_neighborFinset x v).mpr ⟨hx,hDx⟩⟩,((Q\D).mem_neighborFinset y v).mpr ⟨hy,hDy⟩⟩)
  31   have hh:=(Finset.card_le_card hs).trans ((Finset.card_union_le _ _).trans
  32     (Nat.add_le_add_left (Finset.card_union_le _ _) C.card))
  33   simpa only [SimpleGraph.card_neighborFinset_eq_degree,Nat.add_assoc] using hh

  35 theorem ordinary_root_wheel_after_local_deletion (Q D F : SimpleGraph V) (H : V→U→Prop)
  36     (A W : Finset V) (x v w : V) (u:U) (hxA:x∉A) (hvw:v≠w)
  37     (hxv:Q.Adj x v) (hxw:Q.Adj x w) (hDxv:¬D.Adj x v) (hDxw:¬D.Adj x w)
  38     (hFroot:∀ a∈({x,v,w}:Finset V),∀ b∈({x,v,w}:Finset V),¬F.Adj a b)
  39     (hHroot:H x u∧H v u∧H w u) (hHA:∀ a∈A,H a u)
  40     (r:ℕ) (hr:4≤r) (L d b:ℝ) (hL:0≤L)
  41     (hF:∀ a,(F.degree a:ℝ)≤L) (hD:∀ a,a≠x → (D.degree a:ℝ)≤d)
  42     (hcount:∀ a,a≠x → ∀ b',b'≠x → b≤((A∩Q.neighborFinset a∩Q.neighborFinset b').card:ℝ))
  43     (hbudget:W.card+((r:ℝ)+3)*(L+1)+2*d<b) :
  44     ∃ C:OrdinaryWheel Q F H r,ordinaryWheelHub C=u ∧
  45       (ordinaryWheelRim C).Adj x v ∧ (ordinaryWheelRim C).Adj x w ∧
  46       Disjoint (ordinaryWheelRim C) D ∧
  47       (∀ a∈(ordinaryWheelRim C).support,a∈({x,v,w}:Finset V) ∨ a∈A∧a∉W) := by
  48   let Ω:Finset V:=Finset.univ\{x}
  49   have hAΩ:A⊆Ω:=by
  50     intro a ha
  51     simp only [Ω,Finset.mem_sdiff,Finset.mem_univ,Finset.mem_singleton,true_and]
  52     intro he
  53     exact hxA (he ▸ ha)
  54   have hcounts (a:V) (ha:a∈Ω) (b':V) (hb':b'∈Ω) :
  55       b-2*d≤((A∩(Q\D).neighborFinset a∩(Q\D).neighborFinset b').card:ℝ):=by
  56     have hax:a≠x:=by simpa only [Ω,Finset.mem_sdiff,Finset.mem_univ,Finset.mem_singleton,true_and] using ha
  57     have hbx:b'≠x:=by simpa only [Ω,Finset.mem_sdiff,Finset.mem_univ,Finset.mem_singleton,true_and] using hb'
  58     have hh:=two_common_neighbours_after_deletion Q D A a b'
  59     have hhR : ((A∩Q.neighborFinset a∩Q.neighborFinset b').card:ℝ)≤
  60       (A∩(Q\D).neighborFinset a∩(Q\D).neighborFinset b').card+D.degree a+D.degree b':=by exact_mod_cast hh
  61     linarith only [hhR,hcount a hax b' hbx,hD a hax,hD b' hbx]
  62   obtain ⟨C,hCu,hCxv,hCxw,hCs⟩:=ordinary_root_wheel_from_common_counts_within (Q\D) F H A W Ω hAΩ x v w
  63     (by simp [Ω,hxv.ne.symm]) (by simp [Ω,hxw.ne.symm]) u hvw ⟨hxv,hDxv⟩ ⟨hxw,hDxw⟩ hFroot hHroot hHA
  64     r hr L (b-2*d) hL hF (by intro a ha b' hb';convert hcounts a ha b' hb' using 1 <;> congr!) (by linarith only [hbudget])
  65   refine ⟨ordinaryWheelInclude sdiff_le C,hCu,hCxv,hCxw,?_,hCs⟩
  66   apply disjoint_iff_inf_le.mpr
  67   intro a b' he
  68   exact ((C.val.1.property he.1).2 he.2).elim
