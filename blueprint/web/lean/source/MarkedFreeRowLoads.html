  41 theorem markedFamilyRowLoad_le_free_add_forced
  42     (C : I → MarkedWheel tail head ktail khead m π b H₀ HK) (a z : I → V) (v : V) :
  43     markedFamilyRowLoad C v≤markedFamilyFreeRowLoad C a z v+
  44       (Finset.univ.filter (fun i => v=a i ∨ v=z i)).card := by
  45   have hs : Finset.univ.filter (fun i => v∈markedWheelRows (C i)) ⊆
  46       Finset.univ.filter (fun i => v∈markedWheelRows (C i) ∧ v≠a i ∧ v≠z i) ∪
  47       Finset.univ.filter (fun i => v=a i ∨ v=z i) := by
  48     intro i hi
  49     by_cases h : v=a i ∨ v=z i
  50     · exact Finset.mem_union_right _ (Finset.mem_filter.mpr ⟨Finset.mem_univ _,h⟩)
  51     · exact Finset.mem_union_left _ (Finset.mem_filter.mpr
  52         ⟨Finset.mem_univ _,(Finset.mem_filter.mp hi).2,fun ha => h (Or.inl ha),fun hz => h (Or.inr hz)⟩)
  53   exact (Finset.card_le_card hs).trans (Finset.card_union_le _ _)
