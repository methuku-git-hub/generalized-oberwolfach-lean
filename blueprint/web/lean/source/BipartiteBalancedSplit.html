  14 theorem bipartite_balanced_split (lam : ℝ) (hlam : 0<lam) :
  15     ∃ N : ℕ,∀ (X Y : Type) [Fintype X] [Fintype Y],
  16       ∀ m : ℕ,N≤m → Fintype.card X=2*m → Fintype.card Y=m →
  17       ∀ E : X→Y→Prop,(∀ y,2*lam*m≤((Finset.univ.filter (fun x=>E x y)).card:ℝ)) →
  18       ∃ P : FixedPartition X 2 (fun _=>m),∀ y i,
  19         lam*m/2≤(((Finset.univ.filter (fun x=>E x y))∩(fixedPartitionCell P i).val).card:ℝ) := by
  20   obtain ⟨N,hN⟩:=partition_failure_eventually_small 2 (by norm_num) lam hlam
  21   refine ⟨max N 2,?_⟩
  22   intro X Y _ _ m hm hX hY E hdeg
  23   have hm2 : 2≤m:=(le_max_right N 2).trans hm
  24   have hmR : (0:ℝ)<m:=by exact_mod_cast (show 0<m by omega)
  25   letI : Nonempty (FixedPartition X 2 (fun _=>m)):=fixedPartition_nonempty _ (by simpa using hX.symm)
  26   let μ:=FiniteLaw.uniform (FixedPartition X 2 (fun _=>m))
  27   let A (y:Y):=Finset.univ.filter (fun x=>E x y)
  28   let Bad (z : Y×Fin 2) (P : FixedPartition X 2 (fun _=>m)):=
  29     lam*m/2≤|((A z.1∩(fixedPartitionCell P z.2).val).card:ℝ)-(A z.1).card/2|
  30   let p:=2*Real.exp (-lam^2*m/2)
  31   have htail (z:Y×Fin 2):μ.probability (Bad z)≤p:=by
  32     have ht:=fixedPartition_cell_count_tail (d:=fun _ : Fin 2=>m) z.2 (A z.1)
  33       (show (0:ℝ)≤lam*m/2 by positivity)
  34     have hmean : (m:ℝ)/Fintype.card X=1/2:=by rw [hX];push_cast;field_simp
  35     have hexp : -2*(lam*m/2)^2/(m:ℝ)=-lam^2*m/2:=by field_simp <;> ring
  36     simpa only [hmean,mul_one_div,hexp] using ht
  37   have hsum : ∑ _z:Y×Fin 2,p<1:=by
  38     have hf:=(hN (2*m) (by omega)).2.2
  39     have hd : 2*m/2=m:=by omega
  40     rw [partitionFailure,hd] at hf
  41     have hinv : 1/(2*m:ℝ)<1:=(div_lt_one (by positivity)).mpr (by exact_mod_cast (show 1<2*m by omega))
  42     have he : 0≤Real.exp (-lam^2*m/2):=(Real.exp_pos _).le
  43     simp only [Finset.sum_const,Finset.card_univ,Fintype.card_prod,Fintype.card_fin,hY,Nat.cast_mul,Nat.cast_ofNat,nsmul_eq_mul]
  44     dsimp [p]
  45     push_cast at hf
  46     nlinarith only [hf,hinv,mul_nonneg (Nat.cast_nonneg m) he]
  47   obtain ⟨P,_,hP⟩:=finiteLaw_common_event_exists μ Finset.univ Bad (fun _=>p) (fun z _=>htail z) hsum
  48   refine ⟨P,?_⟩
  49   intro y i
  50   have hh:=lt_of_not_ge (hP (y,i) (Finset.mem_univ _))
  51   change |((A y∩(fixedPartitionCell P i).val).card:ℝ)-(A y).card/2|<lam*m/2 at hh
  52   have hl:=(abs_lt.mp hh).1
  53   have hg:=hdeg y
  54   change 2*lam*m≤((A y).card:ℝ) at hg
  55   dsimp only [A] at hl
  56   linarith
  57 
  58 end Oberwolfach
