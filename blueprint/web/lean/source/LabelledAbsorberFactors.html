  28 theorem labelled_absorber_factors (R : C → Finset V) (S : VertexAbsorberTemplate c) (X : Finset V)
  29     (D : LabelledCycleRecord R S.val X) :
  30     Disjoint (labelledAbsorberFree R S X D) X ∧
  31     (∀ a∈labelledAbsorberZero R S X D,∀ b∈labelledAbsorberZero R S X D,a≠b → Disjoint (R a) (R b)) ∧
  32     (labelledAbsorberZero R S X D).biUnion R=labelledAbsorberFree R S X D ∧
  33     (∀ a∈labelledAbsorberOne R S X D,∀ b∈labelledAbsorberOne R S X D,a≠b → Disjoint (R a) (R b)) ∧
  34     (labelledAbsorberOne R S X D).biUnion R=labelledAbsorberFree R S X D∪X := by
  35   have h₀ := labelled_factor_image (fun i : Fin c => S.val (Sum.inl i))
  36     (Finset.univ.image Sum.inl) S.property.2.1 S.property.2.2.2.1 D.1.val D.1.property.1 R
  37     (fun i => (D.2 (Sum.inl i)).val) (fun i => (D.2 (Sum.inl i)).property)
  38   have h₁ := labelled_factor_image (fun i : Fin (c+1) => S.val (Sum.inr i))
  39     Finset.univ S.property.2.2.1 S.property.2.2.2.2.1 D.1.val D.1.property.1 R
  40     (fun i => (D.2 (Sum.inr i)).val) (fun i => (D.2 (Sum.inr i)).property)
  41   have hf : (Finset.univ.image Sum.inl).image D.1.val=labelledAbsorberFree R S X D := by
  42     simp [labelledAbsorberFree,Finset.image_image,Function.comp_def]
  43   have hu : Finset.univ.image D.1.val=labelledAbsorberFree R S X D∪X := by
  44     ext v
  45     constructor
  46     · intro hv
  47       obtain ⟨j,_,rfl⟩ := Finset.mem_image.mp hv
  48       rcases j with j | j
  49       · exact Finset.mem_union_left _ (Finset.mem_image.mpr ⟨j,Finset.mem_univ _,rfl⟩)
  50       · exact Finset.mem_union_right _ (rootedLabelMap_root_mem X D.1 j)
  51     · intro hv
  52       rcases Finset.mem_union.mp hv with hv | hv
  53       · obtain ⟨j,_,rfl⟩ := Finset.mem_image.mp hv
  54         exact Finset.mem_image.mpr ⟨Sum.inl j,Finset.mem_univ _,rfl⟩
  55       · have hx := D.1.property.2
  56         rw [← hx] at hv
  57         obtain ⟨j,_,rfl⟩ := Finset.mem_image.mp hv
  58         exact Finset.mem_image.mpr ⟨Sum.inr j,Finset.mem_univ _,rfl⟩
  59   refine ⟨?_,h₀.1,h₀.2.trans hf,h₁.1,h₁.2.trans hu⟩
  60   apply Finset.disjoint_left.mpr
  61   intro v hv hx
  62   obtain ⟨i,_,rfl⟩ := Finset.mem_image.mp hv
  63   have hroot := D.1.property.2
  64   have hx' : D.1.val (Sum.inl i)∈Finset.univ.image (fun j => D.1.val (Sum.inr j)) := by
  65     simpa only [hroot] using hx
  66   obtain ⟨j,_,he⟩ := Finset.mem_image.mp hx'
  67   have hh := D.1.property.1 he
  68   cases hh
