  22 def pairedTupleChoiceToRecord (D : PairedTupleChoice x Base Path) :
  23     PairedCycleRecord (c:=m+1) R (Finset.univ.image x) := by
  24   let y := D.1.val
  25   let g := D.2.val
  26   have hxy (i j : Fin (m+1)) : y i≠x j := by
  27     intro he
  28     exact D.1.property.2 i (Finset.mem_image.mpr ⟨j,Finset.mem_univ _,he.symm⟩)
  29   let f : RootedLabelMap V ((m+1)*(m+1)) (m+1) (Finset.univ.image x) :=
  30     ⟨pairedTupleLabels x y g,pairedTupleLabels_injective x y g hx (hbinj y D.1.property.1)
  31       hxy D.2.property.2.1 D.2.property.2.2,rfl⟩
  32   have hp (i : Fin (m+1)) := hpaths y D.1.property.1 D.1.property.2 i (g i) (D.2.property.1 i)
  33   refine ⟨f,⟨Classical.choose (hb y D.1.property.1),?_⟩,
  34     (fun i => ⟨Classical.choose (hp i).1,?_⟩),
  35     (fun i => ⟨Classical.choose (hp i).2,?_⟩)⟩
  36   · exact (Classical.choose_spec (hb y D.1.property.1)).trans (pairedTupleLabels_base x y g).symm
  37   · exact (Classical.choose_spec (hp i).1).trans (pairedTupleLabels_zero x y g i).symm
  38   · exact (Classical.choose_spec (hp i).2).trans (pairedTupleLabels_one x y g i).symm

  46   intro D E he
  47   have hf := congrArg (fun D : PairedCycleRecord (c:=m+1) R (Finset.univ.image x) => D.1.val) he
  48   rw [pairedTupleChoiceToRecord_labels,pairedTupleChoiceToRecord_labels] at hf

  50   rcases E with ⟨⟨z,hz⟩,⟨h,hh⟩⟩
  51   have hyz : y=z := by
  52     funext i
  53     have he := congrFun hf (pairedFreePosition i 0)
