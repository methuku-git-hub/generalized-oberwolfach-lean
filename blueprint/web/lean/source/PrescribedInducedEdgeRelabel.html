  10 theorem prescribed_induced_edge_mk (Q:SimpleGraph X) (A:Finset X) (x y:X):
  11     s(x,y)∈prescribedInducedSourceEdges Q A ↔ Q.Adj x y ∧ x∈A ∧ y∈A := by
  12   rw [prescribedInducedSourceEdges,Finset.mem_filter]
  13   constructor
  14   · rintro ⟨he,ha⟩
  15     have hxy:Q.Adj x y:=SimpleGraph.mem_edgeFinset.mp he
  16     have hx:x∈s(x,y).toFinset:=by simp [Sym2.toFinset_mk_eq]
  17     have hy:y∈s(x,y).toFinset:=by simp [Sym2.toFinset_mk_eq]
  18     exact ⟨hxy,ha x hx,ha y hy⟩
  19   · rintro ⟨hxy,hx,hy⟩
  20     refine ⟨SimpleGraph.mem_edgeFinset.mpr hxy,?_⟩
  21     intro z hz
  22     simp only [Sym2.toFinset_mk_eq,Finset.mem_insert,Finset.mem_singleton] at hz
  23     rcases hz with rfl|rfl <;> assumption

  25 theorem prescribed_induced_source_edges_relabel (Q:SimpleGraph X) (G:SimpleGraph Y) (f:X↪Y)
  26     (hAdj:∀ x y,G.Adj (f x) (f y) ↔ Q.Adj x y) (A:Finset X):
  27     prescribedInducedSourceEdges G (A.image f)=
  28       (prescribedInducedSourceEdges Q A).image (Sym2.map f) := by
  29   ext e
  30   induction e using Sym2.ind with
  31   | _ u v =>
  32     rw [prescribed_induced_edge_mk]
  33     constructor
  34     · rintro ⟨huv,hu,hv⟩
  35       obtain ⟨x,hx,rfl⟩:=Finset.mem_image.mp hu
  36       obtain ⟨y,hy,rfl⟩:=Finset.mem_image.mp hv
  37       apply Finset.mem_image.mpr
  38       refine ⟨s(x,y),?_,rfl⟩
  39       exact (prescribed_induced_edge_mk Q A x y).mpr ⟨(hAdj x y).mp huv,hx,hy⟩
  40     · intro he
  41       obtain ⟨e,he,heq⟩:=Finset.mem_image.mp he
  42       induction e using Sym2.ind with
  43       | _ x y =>
  44         have hh:Q.Adj x y ∧ x∈A ∧ y∈A:=(prescribed_induced_edge_mk Q A x y).mp he
  45         change s(f x,f y)=s(u,v) at heq
  46         rcases Sym2.eq_iff.mp heq with ⟨hx,hy⟩|⟨hx,hy⟩
  47         · subst u;subst v
  48           exact ⟨(hAdj x y).mpr hh.1,Finset.mem_image.mpr ⟨x,hh.2.1,rfl⟩,Finset.mem_image.mpr ⟨y,hh.2.2,rfl⟩⟩
  49         · subst u;subst v
  50           exact ⟨((hAdj x y).mpr hh.1).symm,Finset.mem_image.mpr ⟨y,hh.2.2,rfl⟩,Finset.mem_image.mpr ⟨x,hh.2.1,rfl⟩⟩
  51 end Oberwolfach
