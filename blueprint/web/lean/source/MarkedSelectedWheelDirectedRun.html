  12 theorem selected_marked_wheel_directed_run
  13     {V:Type*} (v:ℕ → V) (c n start:ℕ) (hne:∀ i,v i≠v (i+1))
  14     (C:ℕ → Fin c → V)
  15     (p:∀ i, start ≤ i  →  i < start+n  →  (SimpleGraph.completeGraph V).Walk (v i) (v (i+1)))
  16     (hp:∀ i (hi:start ≤ i) (hj:i < start+n),(p i hi hj).IsPath)
  17     (hlen:∀ i (hi:start ≤ i) (hj:i < start+n),(p i hi hj).length=c)
  18     (hsupport:∀ i (hi:start ≤ i) (hj:i < start+n),(p i hi hj).support.dropLast=List.ofFn (C i))
  19     (hdis:∀ i j, start ≤ i  →  i < start+n  →  start ≤ j  →  j < start+n  →  i ≠ j  →  ∀ a b:Fin c, C i a ≠ C j b)
  20     (hend:∀ i, start ≤ i  →  i < start+n  →  ∀ a:Fin c, v (start+n) ≠ C i a)
  21     (R:V → V → Prop)
  22     (hR:∀ i (hi:start ≤ i) (hj:i < start+n) k,k<(p i hi hj).length → R ((p i hi hj).getVert k) ((p i hi hj).getVert (k+1))):
  23     ∃q:(SimpleGraph.completeGraph V).Walk (v start) (v (start+n)),q.IsPath ∧ q.length=n*c ∧
  24       (∀x,x∈q.support ↔ x=v (start+n) ∨
  25         ∃i,start ≤ i ∧ i < start+n ∧ ∃a:Fin c,C i a=x) ∧
  26       (∀ j,j≤n → q.getVert (c*j)=v (start+j)) ∧
  27       (∀ k,k<q.length → R (q.getVert k) (q.getVert (k+1))):=by
  28   obtain ⟨p',hmatch⟩:=complete_graph_extend_interval_walks v start n hne p
  29   apply marked_wheel_directed_interval_from_rim_supports (SimpleGraph.completeGraph V) v c n start C p'
  30   · intro i hi hj;rw [hmatch i hi hj];exact hp i hi hj
  31   · intro i hi hj;rw [hmatch i hi hj];exact hlen i hi hj
