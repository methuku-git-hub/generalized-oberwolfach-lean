  15 theorem sampledColourDegrees_common_event
  16     {V : Type*} [Fintype V] [LinearOrder V] {k : ℕ} (G Z : SimpleGraph V)
  17     (ρ : Fin k → ℝ) (hρ : ∀ a, 0 ≤ ρ a) (htotal : ∑ a, ρ a = 1)
  18     {t : ℝ} (ht : 0 ≤ t) :
  19     1 - 6 * k * Fintype.card V * Real.exp (-2 * t ^ 2 / Fintype.card V) ≤
  20       (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.categoricalDirection ρ hρ htotal)).probability
  21         (fun σ => ∀ a x,
  22           |((∑ y, sampledColourOrientation G σ a x y : ℕ) : ℝ) - ρ a / 2 * G.degree x| < t ∧
  23           |((∑ y, sampledColourOrientation G σ a y x : ℕ) : ℝ) - ρ a / 2 * G.degree x| < t ∧
  24           |(sampledColourDegree Z (fun e => (σ e).1) a x : ℝ) - ρ a * Z.degree x| < t) := by
  25   let J := Fin k × (V × Fin 3)
  26   let X (j : J) (σ : Sym2 V → Fin k × Bool) : ℝ :=
  27     match j.2.2 with
  28     | 0 => ((∑ y, sampledColourOrientation G σ j.1 j.2.1 y : ℕ) : ℝ) - ρ j.1 / 2 * G.degree j.2.1
  29     | 1 => ((∑ y, sampledColourOrientation G σ j.1 y j.2.1 : ℕ) : ℝ) - ρ j.1 / 2 * G.degree j.2.1
  30     | _ => (sampledColourDegree Z (fun e => (σ e).1) j.1 j.2.1 : ℝ) - ρ j.1 * Z.degree j.2.1
  31   let b := 2 * Real.exp (-2 * t ^ 2 / Fintype.card V)
  32   have hbad (j : J) :
  33       (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.categoricalDirection ρ hρ htotal)).probability
  34         (fun σ => t ≤ |X j σ|) ≤ b := by
  35     obtain ⟨a, x, j⟩ := j
  36     fin_cases j
  37     · exact sampledColourOrientation_degree_tail G ρ hρ htotal a x ht
  38     · exact sampledColourOrientation_indegree_tail G ρ hρ htotal a x ht
  39     · exact sampledColourDegree_joint_tail Z ρ hρ htotal a x ht
  40   have hh := finiteLaw_common_event_lower
  41     (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.categoricalDirection ρ hρ htotal))
  42     Finset.univ (fun j σ => t ≤ |X j σ|) (fun _ => b) (fun j _ => hbad j)
  43   have hsum : (∑ _ : J, b) = 6 * k * Fintype.card V * Real.exp (-2 * t ^ 2 / Fintype.card V) := by
  44     simp only [Finset.sum_const, Finset.card_univ, J, Fintype.card_prod, Fintype.card_fin,
  45       Nat.cast_mul, Nat.cast_ofNat, nsmul_eq_mul, b]
  46     ring
  47   rw [hsum] at hh
  48   apply hh.trans
  49   apply finiteLaw_probability_mono
  50   intro σ hσ a x
  51   exact ⟨not_le.mp (hσ (a, x, 0) (Finset.mem_univ _)),
  52     not_le.mp (hσ (a, x, 1) (Finset.mem_univ _)),
  53     not_le.mp (hσ (a, x, 2) (Finset.mem_univ _))⟩

  55 theorem sampledColourDegrees_highProbability (k p : ℕ) :
  56     ∃ N₀ : ℕ, ∀ (V : Type u) [Fintype V] [LinearOrder V],
  57       let n := Fintype.card V
  58       N₀ ≤ n → ∀ (G Z : SimpleGraph V) (ρ : Fin k → ℝ)
  59         (hρ : ∀ a, 0 ≤ ρ a) (htotal : ∑ a, ρ a = 1),
  60       1 - 1 / (n : ℝ) ^ p ≤
  61         (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.categoricalDirection ρ hρ htotal)).probability
  62           (fun σ => ∀ a x,
  63             |((∑ y, sampledColourOrientation G σ a x y : ℕ) : ℝ) - ρ a / 2 * G.degree x| <
  64               (n : ℝ) ^ (3 / 5 : ℝ) ∧
  65             |((∑ y, sampledColourOrientation G σ a y x : ℕ) : ℝ) - ρ a / 2 * G.degree x| <
  66               (n : ℝ) ^ (3 / 5 : ℝ) ∧
  67             |(sampledColourDegree Z (fun e => (σ e).1) a x : ℝ) - ρ a * Z.degree x| <
  68               (n : ℝ) ^ (3 / 5 : ℝ)) := by
  69   obtain ⟨N, hN⟩ := eventually_stretchedExp_le_inv_pow (6 * k) 1
  70     (c := 2) (p := 1 / 5) (by norm_num) (by norm_num) p
  71   refine ⟨max N 1, fun V _ _ => ?_⟩
  72   intro n hn G Z ρ hρ htotal
  73   have hnpos : (0 : ℝ) < n := by exact_mod_cast (show 0 < n by omega)
  74   have hh := sampledColourDegrees_common_event G Z ρ hρ htotal
  75     (t := (n : ℝ) ^ (3 / 5 : ℝ)) (by positivity)
  76   change 1 - 6 * k * n * Real.exp (-2 * ((n : ℝ) ^ (3 / 5 : ℝ)) ^ 2 / n) ≤ _ at hh
  77   rw [threeFifths_tail_exponent hnpos] at hh
  78   have hb := hN n (le_trans (le_max_left _ _) hn)
  79   simp only [Real.rpow_one] at hb
  80   exact (sub_le_sub_left hb 1).trans hh
