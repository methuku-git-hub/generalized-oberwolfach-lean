  12 theorem linear_deviation_failure_exponential (C A ε : ℝ)
  13     (hA : 0<A) (hε : 0<ε) (K : ℕ) :
  14     ∀ n : ℕ,0<n → ∀ M D : ℝ,0≤M → M≤C*(n:ℝ)^K →
  15       0<D → D≤A*n →
  16       M*(2*Real.exp (-2*(ε*n)^2/D))≤2*C*(n:ℝ)^K*Real.exp (-(2*ε^2/A)*n) := by
  17   intro n hn M D hM hMb hD hDb
  18   have hn0 : (0:ℝ)<n := by exact_mod_cast (show 0<n by omega)
  19   have hex : -2*(ε*n)^2/D≤-(2*ε^2/A)*n := by
  20     apply (div_le_iff₀ hD).mpr
  21     have hmul := mul_le_mul_of_nonneg_left hDb (by positivity : 0≤2*ε^2/A*n)
  22     have heq : (2*ε^2/A*n)*(A*n)=2*(ε*n)^2 := by field_simp <;> ring
  23     rw [heq] at hmul
  24     nlinarith only [hmul]
  25   calc
  26     _ ≤ M*(2*Real.exp (-(2*ε^2/A)*n)) :=
  27       mul_le_mul_of_nonneg_left (mul_le_mul_of_nonneg_left (Real.exp_le_exp.mpr hex) (by norm_num)) hM
  28     _ ≤ (C*(n:ℝ)^K)*(2*Real.exp (-(2*ε^2/A)*n)) :=
  29       mul_le_mul_of_nonneg_right hMb (by positivity)
  30     _ = (2*C)*(n:ℝ)^K*Real.exp (-(2*ε^2/A)*n) := by ring
