   8 theorem ordinary_small_root_density (p pmin smin:ℝ) (s:U→Color→ℝ) (j:Color)
   9     (hpmin:0<pmin) (hp:pmin≤p) (hp1:p≤1)
  10     (hsmin:0<smin) (hsmin1:smin≤1) (hs:∀ u,smin≤s u j):
  11     ∀ (S:Finset V) (J:Finset U),S.card≤2→J.card≤2→
  12       pmin^2*smin^2≤p^S.card*(∏ u∈J,s u j) := by
  13   intro S J hS hJ
  14   have hpmin1:pmin≤1:=hp.trans hp1
  15   have hpowp:pmin^2≤p^S.card:=
  16     (pow_le_pow_of_le_one hpmin.le hpmin1 hS).trans (pow_le_pow_left₀ hpmin.le hp _)
  17   have hpowS:smin^2≤smin^J.card:=pow_le_pow_of_le_one hsmin.le hsmin1 hJ
  18   have hprod:smin^J.card≤∏ u∈J,s u j:=by
  19     rw [←Finset.prod_const]
  20     exact Finset.prod_le_prod₀ (fun _ _=>hsmin.le) (fun u _=>hs u)
  21   exact mul_le_mul hpowp (hpowS.trans hprod) (pow_nonneg hsmin.le _) (pow_nonneg (hpmin.le.trans hp) _)
  22 end Oberwolfach
