   9 theorem disjoint_deleted_path_endpoints_survive (F:SimpleGraph V)
  10     (s:Finset I) (u v:I→V) (P:∀ i,F.Walk (u i) (v i))
  11     (hP:(↑s:Set I).PairwiseDisjoint (fun i=>(P i).support.toFinset))
  12     (t:Finset J) (root:J→V) (C:∀ j,F.Walk (root j) (root j))
  13     (hcross:∀i∈s,∀j∈t,Disjoint (P i).support.toFinset (C j).support.toFinset):
  14     ∀i∈s,∀x∈({u i,v i}:Finset V),
  15       x∉(t.biUnion (fun j=>(C j).support.toFinset)) ∪
  16         (s.biUnion (fun i=>(P i).support.toFinset \ {u i,v i})) := by
  17   intro i hi x hx hdel
  18   have hxin:x∈(P i).support.toFinset := by
  19     simp only [Finset.mem_insert,Finset.mem_singleton] at hx
  20     rcases hx with rfl|rfl <;> simp
  21   rcases Finset.mem_union.mp hdel with hc|hp
  22   · obtain ⟨j,hj,hj'⟩:=Finset.mem_biUnion.mp hc
  23     exact Finset.disjoint_left.mp (hcross i hi j hj) hxin hj'
  24   · obtain ⟨j,hj,hj'⟩:=Finset.mem_biUnion.mp hp
  25     by_cases hij:i=j
  26     · subst j
  27       exact (Finset.mem_sdiff.mp hj').2 hx
  28     · exact Finset.disjoint_left.mp (hP hi hj hij) hxin (Finset.mem_sdiff.mp hj').1
  29 end Oberwolfach
