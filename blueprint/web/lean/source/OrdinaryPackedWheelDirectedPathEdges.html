  11 theorem ordinary_packed_wheel_directed_path_with_edges
  12     {V U:Type} [Fintype V] [LinearOrder V] [Fintype U] [DecidableEq U]
  13     (π:Equiv.Perm V) (b n:ℕ) (hn:2≤n) (W:OrderedOrdinaryWheel (U:=U) π b (n+1))
  14     (R:V→V→Prop) (hR:∀i:Fin (n+1),R (W.1.val i) (W.1.val (cycleNext i))):
  15     let start:=W.1.val (cycleNext 0)
  16     let finish:=W.1.val 0
  17     ∃q:(SimpleGraph.completeGraph V).Walk start finish,
  18       q.IsPath ∧ q.length=n ∧
  19       q.support.toFinset=Finset.univ.image W.1.val ∧
  20       R finish start ∧
  21       (∀k,k<q.length→∃i:Fin (n+1),q.getVert k=W.1.val i ∧ q.getVert (k+1)=W.1.val (cycleNext i)) ∧
  22       (∀k,k<q.length→R (q.getVert k) (q.getVert (k+1))):=by
  23   intro start finish
  24   let S:V→V→Prop:=fun x y=>R x y ∧ ∃i:Fin (n+1),x=W.1.val i ∧ y=W.1.val (cycleNext i)
  25   obtain ⟨q,hp,hl,hs,hclose,hedges⟩:=ordinary_packed_wheel_directed_path π b n hn W S
  26     (fun i=>⟨hR i,i,rfl,rfl⟩)
  27   exact ⟨q,hp,hl,hs,hclose.1,fun k hk=>(hedges k hk).2,fun k hk=>(hedges k hk).1⟩
  28 end Oberwolfach
