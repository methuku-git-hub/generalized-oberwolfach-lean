  11 theorem fibre_cycles {V K : Type*} [Fintype V] [DecidableEq K] (cell : V → K) :
  12     ∃ π : Equiv.Perm V, (∀ x, cell (π x) = cell x) ∧
  13       ∀ x k, (π ^ k) x = x ↔
  14         (Finset.univ.filter (fun y => cell y = cell x)).card ∣ k := by
  15   classical
  16   let A (c : K) : Finset V := Finset.univ.filter (fun x => cell x = c)
  17   have hex : ∀ c : K, ∃ p : Equiv.Perm V, p.IsCycleOn (A c) ∧ p.support ⊆ A c :=
  18     fun c => (A c).exists_cycleOn
  19   choose p hp using hex
  20   have hx (x : V) : x ∈ A (cell x) := by simp [A]
  21   have hforward (x : V) : cell (p (cell x) x) = cell x :=
  22     (Finset.mem_filter.mp ((hp (cell x)).1.1.mapsTo (hx x))).2
  23   have hbackward (x : V) : cell ((p (cell x)).symm x) = cell x :=
  24     (Finset.mem_filter.mp ((hp (cell x)).1.inv.1.mapsTo (hx x))).2
  25   let π : Equiv.Perm V := {
  26     toFun := fun x => p (cell x) x
  27     invFun := fun x => (p (cell x)).symm x
  28     left_inv := fun x => by
  29       dsimp only
  30       rw [hforward]
  31       exact (p (cell x)).symm_apply_apply x
  32     right_inv := fun x => by
  33       dsimp only
  34       rw [hbackward]
  35       exact (p (cell x)).apply_symm_apply x }
  36   have hpow (x : V) (k : ℕ) : (π ^ k) x = (p (cell x) ^ k) x := by
  37     induction k with
  38     | zero => rfl
  39     | succ k ih =>
  40       rw [pow_succ', pow_succ', Equiv.Perm.mul_apply, Equiv.Perm.mul_apply, ih]
  41       change p (cell ((p (cell x) ^ k) x)) ((p (cell x) ^ k) x) =
  42         p (cell x) ((p (cell x) ^ k) x)
  43       have hm := (hp (cell x)).1.1.mapsTo.perm_pow k (hx x)
  44       rw [(Finset.mem_filter.mp hm).2]
  45   refine ⟨π, hforward, ?_⟩
  46   intro x k
  47   rw [hpow]
  48   exact (hp (cell x)).1.pow_apply_eq (hx x)
