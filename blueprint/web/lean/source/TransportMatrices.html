  11 def transportMatrices {X Y : Type*} [Fintype X] [Fintype Y]
  12     (m : X → ℝ) (n : Y → ℝ) : Set (X → Y → ℝ) :=
  13   {Q | (∀ x y, 0 ≤ Q x y) ∧ (∀ x, ∑ y, Q x y = m x) ∧
  14     (∀ y, ∑ x, Q x y = n y)}

  16 theorem transportMatrices_entry_le {X Y : Type*} [Fintype X] [Fintype Y]
  17     {m : X → ℝ} {n : Y → ℝ} {Q : X → Y → ℝ}
  18     (hQ : Q ∈ transportMatrices m n) (x : X) (y : Y) :
  19     Q x y ≤ m x ∧ Q x y ≤ n y := by
  20   constructor
  21   · rw [← hQ.2.1 x]
  22     exact Finset.single_le_sum (fun z _ => hQ.1 x z) (Finset.mem_univ y)
  23   · rw [← hQ.2.2 y]
  24     exact Finset.single_le_sum (fun z _ => hQ.1 z y) (Finset.mem_univ x)

  26 theorem transportMatrices_nonempty {X Y : Type*} [Fintype X] [Fintype Y]
  27     (m : X → ℝ) (n : Y → ℝ) (B : ℝ)
  28     (hm : ∀ x, 0 ≤ m x) (hn : ∀ y, 0 ≤ n y)
  29     (hB : 0 < B) (hms : ∑ x, m x = B) (hns : ∑ y, n y = B) :
  30     (transportMatrices m n).Nonempty := by
  31   refine ⟨fun x y => m x * n y / B, ?_, ?_, ?_⟩
  32   · intro x y
  33     exact div_nonneg (mul_nonneg (hm x) (hn y)) hB.le
  34   · intro x
  35     rw [← Finset.sum_div, ← Finset.mul_sum, hns, mul_div_cancel_right₀ _ hB.ne']
  36   · intro y
  37     rw [← Finset.sum_div, ← Finset.sum_mul, hms, mul_div_cancel_left₀ _ hB.ne']

  39 theorem transportMatrices_convex {X Y : Type*} [Fintype X] [Fintype Y]
  40     (m : X → ℝ) (n : Y → ℝ) :
  41     Convex ℝ (transportMatrices m n) := by
  42   intro Q hQ R hR a b ha hb hab
  43   refine ⟨?_, ?_, ?_⟩
  44   · intro x y
  45     exact add_nonneg (mul_nonneg ha (hQ.1 x y)) (mul_nonneg hb (hR.1 x y))
  46   · intro x
  47     change (∑ y, (a * Q x y + b * R x y)) = m x
  48     rw [Finset.sum_add_distrib, ← Finset.mul_sum, ← Finset.mul_sum,
  49       hQ.2.1 x, hR.2.1 x, ← add_mul, hab, one_mul]
  50   · intro y
  51     change (∑ x, (a * Q x y + b * R x y)) = n y
  52     rw [Finset.sum_add_distrib, ← Finset.mul_sum, ← Finset.mul_sum,
  53       hQ.2.2 y, hR.2.2 y, ← add_mul, hab, one_mul]

  55 theorem transportMatrices_isClosed {X Y : Type*} [Fintype X] [Fintype Y]
  56     (m : X → ℝ) (n : Y → ℝ) :
  57     IsClosed (transportMatrices m n) := by
  58   unfold transportMatrices
  59   simp only [Set.ofPred_and, Set.ofPred_forall]
  60   apply IsClosed.inter
  61   · exact isClosed_iInter fun x => isClosed_iInter fun y =>
  62       isClosed_le continuous_const ((continuous_apply y).comp (continuous_apply x))
  63   · apply IsClosed.inter
  64     · exact isClosed_iInter fun x =>
  65         isClosed_eq (by fun_prop) continuous_const
  66     · exact isClosed_iInter fun y =>
  67         isClosed_eq (by fun_prop) continuous_const

  69 theorem transportMatrices_isCompact {X Y : Type*} [Fintype X] [Fintype Y]
  70     (m : X → ℝ) (n : Y → ℝ) :
  71     IsCompact (transportMatrices m n) := by
  72   refine (isCompact_Icc : IsCompact
  73     (Set.Icc (fun _ : X => fun _ : Y => (0 : ℝ)) (fun x _ => m x))).of_isClosed_subset
  74     (transportMatrices_isClosed m n) ?_
  75   intro Q hQ
  76   exact ⟨hQ.1, fun x y => (transportMatrices_entry_le hQ x y).1⟩

  78 def composeTransport {X Y Z : Type*} [Fintype Y]
  79     (m : Y → ℝ) (K : X → Y → ℝ) (L : Y → Z → ℝ) : X → Z → ℝ :=
  80   fun x z => ∑ y, K x y * L y z / m y

  82 theorem composeTransport_mem {X Y Z : Type*}
  83     [Fintype X] [Fintype Y] [Fintype Z]
  84     {a : X → ℝ} {m : Y → ℝ} {b : Z → ℝ}
  85     {K : X → Y → ℝ} {L : Y → Z → ℝ}
  86     (hK : K ∈ transportMatrices a m) (hL : L ∈ transportMatrices m b)
  87     (hm : ∀ y, 0 < m y) :
  88     composeTransport m K L ∈ transportMatrices a b := by
  89   refine ⟨?_, ?_, ?_⟩
  90   · intro x z
  91     exact Finset.sum_nonneg fun y _ =>
  92       div_nonneg (mul_nonneg (hK.1 x y) (hL.1 y z)) (hm y).le
  93   · intro x
  94     unfold composeTransport
  95     rw [Finset.sum_comm]
  96     simp_rw [← Finset.sum_div, ← Finset.mul_sum, hL.2.1,
  97       mul_div_cancel_right₀ _ (hm _).ne']
  98     exact hK.2.1 x
  99   · intro z
 100     unfold composeTransport
 101     rw [Finset.sum_comm]
 102     simp_rw [← Finset.sum_div, ← Finset.sum_mul, hK.2.2,
 103       mul_div_cancel_left₀ _ (hm _).ne']
 104     exact hL.2.2 z
