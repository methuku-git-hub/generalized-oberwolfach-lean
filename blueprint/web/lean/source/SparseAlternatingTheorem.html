  21         |bipTripleIntersection H u v t-(Fintype.card A:ℝ)*p u*p v*p t|≤δ*Fintype.card A) →

  25       ∀ z : A×B → ℝ,z∈zeroMarginalSpace H →
  26         ∃ t : BipartiteFourCycle H → ℝ,

  28           (∀ e,(∑ Q,if FourCycleIncident Q e then |t Q| else 0)≤C*‖z‖) := by
