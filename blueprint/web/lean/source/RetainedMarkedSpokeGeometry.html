  17 theorem retainedIntervalCovered_successor_blocked (π : Equiv.Perm V) (d : ℕ) (hd : 1≤d)
  18     (R : V→Γ→Prop) (v : V) (g : Γ) (hv : retainedIntervalCovered π d R v g) :
  19     retainedIntervalBlocked π d R (π v) := by
  20   obtain ⟨a,ha,hva⟩ := hv
  21   refine ⟨g,a,ha,?_⟩
  22   obtain ⟨i,hi⟩ := (permutationWindow_mem_iff π 0 (d-1) a v).mp hva
  23   apply (permutationWindow_mem_iff π 0 d a (π v)).mpr
  24   refine ⟨⟨i.val+1,by have hh:=i.isLt;omega⟩,?_⟩
  25   simp only [pow_zero,Equiv.Perm.one_apply] at hi ⊢
  26   rw [pow_succ',Equiv.Perm.mul_apply,hi]
