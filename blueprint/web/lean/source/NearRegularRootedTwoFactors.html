  13 theorem near_regular_host_rooted_two_factors (m:ℕ) (hm:2≤m) (α:ℝ)
  14     (hα:0<α) (hα1:α≤1) (s:ℕ) (hdivs:(m+1)∣s):
  15     ∃ η:ℝ,0<η ∧ ∃ k:ℕ,(m+1)∣k ∧ ∃ N:ℕ,
  16     ∀ (V:Type) [Fintype V] [DecidableEq V],N≤Fintype.card V→
  17     ∀ (G:SimpleGraph V) (T:Finset V),T.card=s→∀ θ:ℝ,
  18       cycleThreshold (m+1)+α≤θ→θ≤1→
  19       (∀ v,|((G.degree v):ℝ)-θ*Fintype.card V|≤η*Fintype.card V)→
  20       ∃ S:Finset V,S.card=k ∧ Disjoint S T ∧
  21         let R:=S∪T
  22         let H:=G.induce (R:Set V)
  23         let S':Finset (R:Set V):=S.subtype (fun v=>v∈R)
  24         ∃ M₁ M₂:Finset (OrdinaryCycle H (m+1)),
  25           (∀ C∈M₁,∀ E∈M₁,C≠E→Disjoint (ordinaryCycleVertices H (m+1) C) (ordinaryCycleVertices H (m+1) E)) ∧
  26           M₁.biUnion (ordinaryCycleVertices H (m+1))=Finset.univ ∧
  27           (∀ C∈M₂,∀ E∈M₂,C≠E→Disjoint (ordinaryCycleVertices H (m+1) C) (ordinaryCycleVertices H (m+1) E)) ∧
  28           M₂.biUnion (ordinaryCycleVertices H (m+1))=S' ∧
  29           (∀ C∈M₁,∀ E∈M₂,Disjoint C.val.val.edgeFinset E.val.val.edgeFinset) ∧
  30           (∀ C∈M₁,∀ v w,C.val.val.Adj v w→v∈S' ∨ w∈S'):=by
  31   obtain ⟨ηf,hηf,Kf,hfactor⟩:=two_near_regular_cycle_factors_at_hub m hm α hα hα1 s
  32   let ηs:=min ηf 1
  33   have hηs:0<ηs:=lt_min hηf (by norm_num)
  34   have hηsf:ηs≤ηf:=min_le_left _ _
  35   have hηs1:ηs≤1:=min_le_right _ _
  36   obtain ⟨Ks,hsample⟩:=rooted_near_regular_sample s ηs hηs hηs1
  37   let k:=(m+1)*max (max Ks Kf) 1
  38   have hk:Ks≤k ∧ Kf≤k:=by dsimp [k];constructor <;> nlinarith [Nat.le_max_left (max Ks Kf) 1,Nat.le_max_left Ks Kf,Nat.le_max_right Ks Kf]
  39   obtain ⟨N,hN⟩:=hsample k hk.1
  40   refine ⟨ηs/8,by positivity,k,dvd_mul_right _ _,N,?_⟩
  41   intro V _ _ hn G T hT θ hθ hθ1 hnear
  42   have hθ0:0≤θ:=by have hh:=cycleThreshold_nonneg (m+1);linarith
  43   obtain ⟨S,hSc,hdis,hdegree⟩:=hN V hn G T hT θ hθ0 hθ1 hnear
  44   let R:=S∪T
  45   let W:Type:={v:V // v∈R}
  46   let H:SimpleGraph W:=G.induce (R:Set V)
  47   let S':Finset W:=S.subtype (fun v=>v∈R)
  48   have hsub:S⊆R:=Finset.subset_union_left
  49   have hSc':S'.card=k:=by dsimp [S'];rw [Finset.card_subtype,Finset.filter_eq_self.mpr hsub,hSc]
  50   have hRc:R.card=k+s:=by dsimp [R];rw [Finset.card_union_of_disjoint hdis,hSc,hT]
  51   have hW:Fintype.card W=k+s:=by simpa [W] using hRc
  52   have hSmap:(inducedSetEmbedding R (⟨S',hSc'⟩:FixedSubset W k)).val=S:=by
  53     ext v
  54     simp only [inducedSetEmbedding,Finset.mem_map,Finset.mem_subtype]
  55     constructor
  56     · rintro ⟨w,hw,rfl⟩;exact Finset.mem_subtype.mp hw
  57     · intro hv;exact ⟨⟨v,hsub hv⟩,Finset.mem_subtype.mpr hv,rfl⟩
  58   have hcenter:(cycleThreshold (m+1)+α)*(S'.card:ℝ)≤θ*k:=by
  59     rw [hSc'];exact mul_le_mul_of_nonneg_right hθ (Nat.cast_nonneg _)
  60   have hcenterhi:θ*k≤S'.card:=by rw [hSc'];nlinarith [mul_le_mul_of_nonneg_right hθ1 (Nat.cast_nonneg (α:=ℝ) k)]
  61   have hnear':∀ v:W,|((H.neighborFinset v∩S').card:ℝ)-θ*k|≤ηf*S'.card:=by
  62     intro v
  63     have hh:=inducedSetEmbedding_degree G R (⟨S',hSc'⟩:FixedSubset W k) v
  64     change (H.neighborFinset v∩S').card=(G.neighborFinset v.val∩(inducedSetEmbedding R ⟨S',hSc'⟩).val).card at hh
  65     rw [hSmap] at hh
  66     rw [hh,hSc']
  67     exact (hdegree v.val v.property).trans (mul_le_mul_of_nonneg_right hηsf (Nat.cast_nonneg _))
  68   have hdivR:(m+1)∣Fintype.card W:=by rw [hW];exact dvd_add (dvd_mul_right _ _) hdivs
  69   have hq:Fintype.card W-S'.card≤s:=by rw [hW,hSc'];omega
  70   obtain ⟨M₁,M₂,hM₁,hcover₁,hM₂,hcover₂,hedge,havoid⟩:=hfactor W H S' (θ*k)
  71     (by rw [hSc'];exact hk.2) (by rw [hSc'];exact dvd_mul_right _ _) hdivR hq hcenter hcenterhi hnear'
  72   exact ⟨S,hSc,hdis,M₁,M₂,hM₁,hcover₁,hM₂,hcover₂,hedge,havoid⟩
  73 end Oberwolfach
