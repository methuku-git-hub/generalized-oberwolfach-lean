  17 theorem profileRootSet_sdiff_card_le {a : I → ℕ}
  18     (σ τ : ∀ i,FixedSubset (V i) (a i)) (i : I)
  19     (hagree : ∀ j,j≠i → σ j=τ j) :
  20     (profileRootSet τ\profileRootSet σ).card ≤ ((τ i).val\(σ i).val).card := by
  21   have hs : profileRootSet τ\profileRootSet σ ⊆
  22       ((τ i).val\(σ i).val).image (fun v => (⟨i,v⟩ : Σ j,V j)) := by
  23     rintro ⟨j,v⟩ hx
  24     have hx' := Finset.mem_sdiff.mp hx
  25     by_cases hji : j=i
  26     · subst j
  27       exact Finset.mem_image.mpr ⟨v,Finset.mem_sdiff.mpr
  28         ⟨(profileRootSet_mem τ i v).mp hx'.1,
  29           fun hv => hx'.2 ((profileRootSet_mem σ i v).mpr hv)⟩,rfl⟩
  30     · have he := hagree j hji
  31       exact False.elim (hx'.2 ((profileRootSet_mem σ j v).mpr
  32         (by rw [he];exact (profileRootSet_mem τ j v).mp hx'.1)))
  33   exact (Finset.card_le_card hs).trans Finset.card_image_le

  36     (σ τ : ∀ i,FixedSubset (V i) (a i)) (i : I)
  37     (hagree : ∀ j,j≠i → σ j=τ j) (u v : V i)
  38     (he : τ i=fixedSubsetMap (Equiv.swap u v) (σ i)) :
  39     (profileRootSet τ\profileRootSet σ).card ≤ 1 ∧
  40       (profileRootSet σ\profileRootSet τ).card ≤ 1 := by

  44     rw [he]

  52 theorem union_mapped_sdiff_card_le {W : Type*} [DecidableEq W]
  53     (e : (Σ i,V i) ↪ W) (Z : Finset W) (A D : Finset (Σ i,V i)) :
  54     ((Z∪D.map e)\(Z∪A.map e)).card ≤ (D\A).card := by
  55   have hs : (Z∪D.map e)\(Z∪A.map e) ⊆ (D\A).map e := by
  56     intro x hx
  57     have hx' := Finset.mem_sdiff.mp hx
  58     have hn : x∉Z := fun h => hx'.2 (Finset.mem_union_left _ h)
  59     have hd := (Finset.mem_union.mp hx'.1).resolve_left hn
  60     obtain ⟨y,hy,rfl⟩ := Finset.mem_map.mp hd
  61     exact Finset.mem_map.mpr ⟨y,Finset.mem_sdiff.mpr ⟨hy,
  62       fun ha => hx'.2 (Finset.mem_union_right _ (Finset.mem_map.mpr ⟨y,ha,rfl⟩))⟩,rfl⟩
  63   simpa only [Finset.card_map] using Finset.card_le_card hs

  66     (e : (Σ i,V i) ↪ W) (Z : Finset W) {a : I → ℕ}

  71     ((Z∪(profileRootSet τ).map e)\(Z∪(profileRootSet σ).map e)).card ≤ 1 ∧
  72     ((Z∪(profileRootSet σ).map e)∪(Z∪(profileRootSet τ).map e)).card ≤ Z.card+(∑ j,a j)+1 := by
