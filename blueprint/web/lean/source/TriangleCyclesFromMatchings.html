  16 theorem triangle_cycle_from_vertices (G : SimpleGraph V) (u v w : V)
  17     (huv : u≠v) (huw : u≠w) (hvw : v≠w)
  18     (huvG : G.Adj u v) (huwG : G.Adj u w) (hvwG : G.Adj v w) :
  19     ∃ C : OrdinaryCycle G 3,ordinaryCycleVertices G 3 C=Finset.univ.image (![u,v,w] : Fin 3→V) := by
  20   have hq1:=huvG.symm
  21   have hq2:=huwG.symm
  22   have hq3:=hvwG.symm
  23   have hi : Function.Injective (![u,v,w] : Fin 3→V) := by
  24     intro i j h;fin_cases i <;> fin_cases j <;> simp_all
  25   obtain ⟨C,hC⟩ := ordinary_cycle_from_embedding G ⊥ Finset.univ (by norm_num : 3≤3)
  26     ⟨![u,v,w],hi⟩ (fun _=>Finset.mem_univ _) (by simp)
  27     (by intro i j h;have hn:=h.ne;fin_cases i <;> fin_cases j <;> simp_all)
  28   exact ⟨C.val,hC⟩

  30 theorem triangle_factor_from_bijections (G : SimpleGraph V) (c : V→Fin 3)
  31     (σ : {v // c v=0}≃{v // c v=1}) (τ : {v // c v=0}≃{v // c v=2})
  32     (hG : ∀ x,G.Adj x.val (σ x).val ∧ G.Adj x.val (τ x).val ∧ G.Adj (σ x).val (τ x).val) :
  33     HasResourceDecomposition (ordinaryCycleVertices G 3) Finset.univ := by
  34   let X := {v // c v=0}
  35   let f : X→Fin 3→V := fun x=>![x.val,(σ x).val,(τ x).val]
  36   have hcolour (x : X) (i : Fin 3) : c (f x i)=i := by
  37     fin_cases i
  38     · exact x.property
  39     · exact (σ x).property
  40     · exact (τ x).property
  41   have hjoint : Function.Injective (fun p : X×Fin 3=>f p.1 p.2) := by
  42     rintro ⟨x,i⟩ ⟨y,j⟩ he
  43     have hij : i=j := (hcolour x i).symm.trans ((congrArg c he).trans (hcolour y j))
  44     subst j
  45     have hxy : x=y := by
  46       fin_cases i
  47       · exact Subtype.ext he
  48       · exact σ.injective (Subtype.ext he)
  49       · exact τ.injective (Subtype.ext he)
  50     exact Prod.ext hxy rfl
  51   have hsurj : Function.Surjective (fun p : X×Fin 3=>f p.1 p.2) := by
  52     intro v
  53     generalize hh : c v=i
  54     fin_cases i
  55     · exact ⟨(⟨v,hh⟩,0),rfl⟩
  56     · refine ⟨(σ.symm ⟨v,hh⟩,1),?_⟩
  57       change (σ (σ.symm ⟨v,hh⟩)).val=v
  58       rw [σ.apply_symm_apply]
  59     · refine ⟨(τ.symm ⟨v,hh⟩,2),?_⟩
  60       change (τ (τ.symm ⟨v,hh⟩)).val=v
  61       rw [τ.apply_symm_apply]
  62   have hex (x : X) : ∃ C : OrdinaryCycle G 3,ordinaryCycleVertices G 3 C=Finset.univ.image (f x) := by
  63     have hi : Function.Injective (f x) := fun i j h=>congrArg Prod.snd (hjoint (a₁:=(x,i)) (a₂:=(x,j)) h)
  64     have h01 : f x 0≠f x 1 := fun h=>by have hh:=hi h;norm_num at hh
  65     have h02 : f x 0≠f x 2 := fun h=>by have hh:=hi h;norm_num at hh
  66     have h12 : f x 1≠f x 2 := fun h=>by have hh:=hi h;norm_num at hh
  67     exact triangle_cycle_from_vertices G _ _ _ h01 h02 h12 (hG x).1 (hG x).2.1 (hG x).2.2
  68   let C : X→OrdinaryCycle G 3 := fun x=>Classical.choose (hex x)
  69   have hC (x) : ordinaryCycleVertices G 3 (C x)=Finset.univ.image (f x) := Classical.choose_spec (hex x)
  70   refine ⟨Finset.univ.image C,?_,?_⟩
  71   · intro D hD E hE hne
  72     obtain ⟨x,_,rfl⟩ := Finset.mem_image.mp hD
  73     obtain ⟨y,_,rfl⟩ := Finset.mem_image.mp hE
  74     apply Finset.disjoint_left.mpr
  75     intro v hv hw
  76     rw [hC] at hv hw
  77     obtain ⟨i,_,hi⟩ := Finset.mem_image.mp hv
  78     obtain ⟨j,_,hj⟩ := Finset.mem_image.mp hw
  79     have hxy := congrArg Prod.fst (hjoint (a₁:=(x,i)) (a₂:=(y,j)) (hi.trans hj.symm))
  80     exact hne (congrArg C hxy)
  81   · apply Finset.eq_univ_of_forall
  82     intro v
  83     obtain ⟨⟨x,i⟩,he⟩ := hsurj v
  84     apply Finset.mem_biUnion.mpr
  85     refine ⟨C x,Finset.mem_image.mpr ⟨x,Finset.mem_univ _,rfl⟩,?_⟩
  86     rw [hC]
  87     exact Finset.mem_image.mpr ⟨i,Finset.mem_univ _,he⟩
