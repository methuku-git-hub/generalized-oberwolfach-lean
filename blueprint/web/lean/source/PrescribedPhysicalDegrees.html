  21 theorem prescribed_physical_incidence_bound (Q:SimpleGraph V) (H:V→U→Prop)
  22     (E:Finset (Sym2 V⊕(V×U))) (D:ℕ) (hE:E⊆ordinaryHostResources Q H)
  23     (hr:∀ v,Q.degree v≤D)
  24     (hs:∀ v,(Finset.univ.filter (H v)).card≤D)
  25     (hc:∀ u,(Finset.univ.filter (fun v=>H v u)).card≤D):
  26     ∀ x,(E.filter (ordinaryResourceIncident x)).card≤2*D:=by
  27   intro x
  28   have hh:(E.filter (ordinaryResourceIncident x)).card≤(ordinaryPhysicalIncidentEdges Q H x).card:=by
  29     rw [←ordinaryHostResources_filter_incident]
  30     exact Finset.card_le_card (Finset.filter_subset_filter _ hE)
  31   apply hh.trans
  32   cases x with
  33   | inl v=>rw [ordinaryPhysicalIncidentEdges_rim_card];exact (Nat.add_le_add (hr v) (hs v)).trans_eq (two_mul D).symm
  34   | inr u=>rw [ordinaryPhysicalIncidentEdges_hub_card];exact (hc u).trans (by omega)
