  11 theorem buffered_outside_image_disjoint_endpoints
  12     {X V I:Type*} [Fintype X] [DecidableEq X] [DecidableEq V] [Fintype I]
  13     (n:I→ℕ) (inner:∀i,Fin (n i+1)↪X)
  14     (fC:{x // x∉bufferedPathStrictInterior n inner}↪V)
  15     (e:I→X) (heA:∀i,e i∈bufferedPathFullDomain n inner)
  16     (heD:∀i,e i∉bufferedPathStrictInterior n inner)
  17     (b:I→V) (hb:∀i,fC ⟨e i,heD i⟩=b i):
  18     Disjoint (outsideBufferedPathImage n inner fC) (Finset.univ.image b):=by
  19   classical
  20   apply Finset.disjoint_left.mpr
  21   intro v hvz hvb
  22   obtain ⟨z,hz,hzv⟩:=Finset.mem_image.mp hvz
  23   obtain ⟨i,_,hbv⟩:=Finset.mem_image.mp hvb
  24   have heq:fC z=fC ⟨e i,heD i⟩:=hzv.trans (hbv.symm.trans (hb i).symm)
  25   have heq':z.val=e i:=congrArg Subtype.val (fC.injective heq)
  26   have hzA:z.val∉bufferedPathFullDomain n inner:=(Finset.mem_filter.mp hz).2
  27   exact hzA (heq' ▸ heA i)
