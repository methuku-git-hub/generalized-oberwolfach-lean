   4 /-! Support and degree consequences of the exact hub-cover resource identity.
   5 Every wheel uses a prescribed hub; its rim vertices belong to that hub's

  34 theorem ordinary_hub_packing_wheel_support (M:Finset (OrdinaryWheel Q F H r))
  35     (hr:0<r) (u:Fin n→U) (hu:Function.Injective u) (T:Fin n→Finset V)
  36     (hunion:M.biUnion ordinaryWheelResources=(ordinaryWheelRimGraph M).edgeFinset.image Sum.inl ∪
  37       Finset.univ.biUnion (fun i=>(T i).image (fun v=>Sum.inr (v,u i))))
  38     (W:OrdinaryWheel Q F H r) (hW:W∈M):
  39     ∃ i,ordinaryWheelHub W=u i ∧ ∀ v∈(ordinaryWheelRim W).support,v∈T i:=by
  40   have hc:(ordinaryWheelRim W).support.toFinset.card=r:=W.val.1.val.property.support_card
  41   obtain ⟨v,hv⟩:=Finset.card_pos.mp (hc.symm ▸ hr)
  42   have hv':v∈(ordinaryWheelRim W).support:=by simpa only [Set.mem_toFinset] using hv
  43   obtain ⟨i,hi,hvi⟩:=(ordinary_hub_packing_spokes M u T hunion v (ordinaryWheelHub W)).mp ⟨W,hW,rfl,hv'⟩
  44   refine ⟨i,hi,?_⟩
  45   intro z hz
  46   obtain ⟨j,hj,hzj⟩:=(ordinary_hub_packing_spokes M u T hunion z (ordinaryWheelHub W)).mp ⟨W,hW,rfl,hz⟩
  47   have hji:j=i:=hu (hj.symm.trans hi)
  48   simpa only [hji] using hzj
