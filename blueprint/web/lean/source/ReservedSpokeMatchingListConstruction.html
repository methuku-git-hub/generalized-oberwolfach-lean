  10 def reservedSpokeMatchingList {V I:Type} [DecidableEq V] [DecidableEq I]

  14 theorem reserved_spoke_matching_lists {V I:Type} [Fintype I] [DecidableEq V] [DecidableEq I]
  15     (S:Sym2 V→Finset I) (A:I→Finset V) (b:ℕ)
  16     (hA:∀v,(Finset.univ.filter (fun i=>v∈A i)).card≤b):
  17     let L:=reservedSpokeMatchingList S A;
  18     (∀e,L e⊆S e) ∧ (∀e v,v∈e.toFinset→∀i∈L e,v∉A i) ∧
  19     (∀e,(S e).card≤(L e).card+2*b) :=by
  20   intro L
  21   have hsub:∀e,L e⊆S e:=by intro e i hi;exact (Finset.mem_filter.mp hi).1
  22   have hgood:∀e v,v∈e.toFinset→∀i∈L e,v∉A i:=by
  23     intro e v hv i hi
  24     exact (Finset.mem_filter.mp hi).2 v hv
  25   refine ⟨hsub,hgood,?_⟩
  26   intro e
  27   let Bad:=fun v:V=>(Finset.univ:Finset I).filter (fun i=>v∈A i)
  28   have hcover:S e⊆L e∪e.toFinset.biUnion Bad:=by
  29     intro i hi
  30     by_cases hg:∀v∈e.toFinset,v∉A i
  31     · exact Finset.mem_union_left _ (Finset.mem_filter.mpr ⟨hi,hg⟩)
  32     · push_neg at hg
  33       obtain ⟨v,hv,hvi⟩:=hg
  34       exact Finset.mem_union_right _ (Finset.mem_biUnion.mpr ⟨v,hv,Finset.mem_filter.mpr ⟨Finset.mem_univ _,hvi⟩⟩)
  35   have hEnds:e.toFinset.card≤2:=by rw [Sym2.card_toFinset];split_ifs <;> omega
  36   have hBad:(e.toFinset.biUnion Bad).card≤e.toFinset.card*b:=
  37     Finset.card_biUnion_le_card_mul _ _ _ (fun v _=>hA v)
  38   have hc:=(Finset.card_le_card hcover).trans (Finset.card_union_le _ _)
  39   nlinarith
  40 end Oberwolfach
