  10 theorem long_binary_pruned_parameters
  11     {I U:Type} [Fintype I] [DecidableEq I] [Fintype U] (ρclass:I→ℝ) (r ρ₀ s:ℝ)
  12     (i:I) (D:ℕ) (a b:U→ℝ) (hD:0<D) (heven:Even D)
  13     (hnonneg:∀j,0≤ρclass j) (htotal:∑j,ρclass j=1) (hlow:∀j,r≤ρclass j)
  14     (hclass:ρclass i/2*D=Fintype.card U) (hρlo:ρ₀≤ initialIndexDensity U D):
  15     let ρ:=initialIndexDensity U D
  16     let p:=ρ*(1/100)/2
  17     let k:=ρ*s/2
  18     let p₂:=(∑w,b w)/(D:ℝ)
  19     let k₂:=(∑w,a w)/(D:ℝ)
  20     let u:=2*(∑w,(1-1/100-s-a w-b w))/(D:ℝ)
  21     let m:=binaryClassMultiplicity D (Fintype.card U);
  22     (∑j,m j)=D/2 ∧
  23       (∀j,expandedBinaryProbability p k p₂ k₂ u j/2*D=(m j:ℝ)) ∧
  24       (ρ=1 ∨ ∀j,r≤expandedBinaryProbability p k p₂ k₂ u j):=by
  25   intro ρ p k p₂ k₂ u m
  26   have hDreal:(D:ℝ)≠0:=by exact_mod_cast (ne_of_gt hD)
  27   have heq:ρ=ρclass i:=by
  28     dsimp [ρ,initialIndexDensity]
  29     rw [←hclass]
  30     field_simp
  31   have hρhi:ρ≤1:=by
  32     rw [heq]
  33     have hh:=Finset.single_le_sum (fun j _=>hnonneg j) (Finset.mem_univ i)
  34     linarith [htotal]
  35   have hsize:ρ*D=2*Fintype.card U:=by
  36     dsimp [ρ,initialIndexDensity]
  37     field_simp
  38   obtain ⟨hs,hm,h0,h1⟩:=long_binary_partition_parameters ρ₀ ρ s D a b hD heven hρlo hρhi hsize
  39   refine ⟨hs,hm,?_⟩
  40   rcases selected_class_index_density_dichotomy ρclass r i D hD hnonneg htotal hlow hclass with hone|hboth
  41   · exact Or.inl hone
  42   · right
  43     intro j
  44     fin_cases j
  45     · change r≤expandedBinaryProbability p k p₂ k₂ u 0
  46       rw [h0]
  47       simpa [ρ] using hboth 0
  48     · change r≤expandedBinaryProbability p k p₂ k₂ u 1
  49       rw [h1]
  50       simpa [ρ] using hboth 1
