  12 theorem balanced_half_degrees_real {V : Type*} [Fintype V] (g : V → ℕ)
  13     (htotal : Even (∑ v,g v)) :
  14     ∃ gp gm : V → ℕ,(∀ v,gp v+gm v=g v) ∧ (∑ v,gp v=∑ v,gm v) ∧
  15       (∀ v,|(gp v : ℝ)-(g v : ℝ)/2|≤1/2) ∧
  16       (∀ v,|(gm v : ℝ)-(g v : ℝ)/2|≤1/2) := by
  17   obtain ⟨gp,gm,hs,ht,hd⟩ := balanced_integer_half_degrees g htotal
  18   refine ⟨gp,gm,hs,ht,?_,?_⟩
  19   all_goals
  20     intro v
  21     have hv : |(gp v : ℝ)-(gm v : ℝ)|≤1 := by exact_mod_cast hd v
  22     have hsum : (gp v : ℝ)+gm v=g v := by exact_mod_cast hs v
  23     rw [abs_le] at hv ⊢
  24     constructor <;> linarith [hv.1,hv.2]

  63     rw [← hgr v]
