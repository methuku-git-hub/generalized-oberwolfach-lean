  11 theorem endpoint_triangle_coefficient_bounds (a b κ p q s c C w : ℝ)
  12     (ha : 0<a) (hκ : 0<κ) (hp : 0<p) (hc : 0≤c) (hC : 0≤C)
  13     (hb : κ*a≤b) (hba : b≤a) (hq : p≤q) (hq1 : q≤1) (hs : p≤s) (hs1 : s≤1)
  14     (hw : c/a≤w ∧ w≤C/a) :
  15     c/a^2≤w/(b*q^2*s^3) ∧ w/(b*q^2*s^3)≤(C/(κ*p^5))/a^2 := by
  16   have hbpos : 0<b := (mul_pos hκ ha).trans_le hb
  17   have hqpos : 0<q := hp.trans_le hq
  18   have hspos : 0<s := hp.trans_le hs
  19   have hdpos : 0<b*q^2*s^3 := by positivity
  20   have hupper : b*q^2*s^3≤a := by
  21     have h1:=mul_le_mul_of_nonneg_left (pow_le_one₀ hqpos.le hq1 (n:=2)) hbpos.le
  22     have h2:=mul_le_mul_of_nonneg_left (pow_le_one₀ hspos.le hs1 (n:=3)) (by positivity : 0≤b*q^2)
  23     nlinarith only [h1,h2,hba]
  24   have hlower : κ*p^5*a≤b*q^2*s^3 := by
  25     have h1:=mul_le_mul hb (pow_le_pow_left₀ hp.le hq 2) (by positivity : 0≤p^2) hbpos.le
  26     have h2:=mul_le_mul h1 (pow_le_pow_left₀ hp.le hs 3) (by positivity : 0≤p^3) (by positivity : 0≤b*q^2)
  27     convert h2 using 1 <;> ring
  28   constructor
  29   · calc
  30       _ = (c/a)/a := by ring
  31       _ ≤ (c/a)/(b*q^2*s^3) := div_le_div_of_nonneg_left (div_nonneg hc ha.le) hdpos hupper
  32       _ ≤ _ := div_le_div_of_nonneg_right hw.1 hdpos.le
  33   · calc
  34       _ ≤ (C/a)/(b*q^2*s^3) := div_le_div_of_nonneg_right hw.2 hdpos.le
  35       _ ≤ (C/a)/(κ*p^5*a) := div_le_div_of_nonneg_left (div_nonneg hC ha.le) (by positivity) hlower
  36       _ = _ := by ring

  38 theorem endpoint_index_target_error (a b κ p q s C L m M δ : ℝ)
  39     (ha : 0<a) (hκ : 0<κ) (hp : 0<p) (hC : 0≤C) (hL : 0≤L)
  40     (hb : κ*a≤b) (hq : 0≤q) (hq1 : q≤1) (hs : p≤s)
  41     (hM : |M-m/2|≤(1+2*C)*L/2) (hbalance : |q*m/(2*b*s)-1|≤δ) :
  42     |q/(b*s)*M-1|≤δ+((1+2*C)/(2*κ*p))*L/a := by
  43   have hbpos : 0<b := (mul_pos hκ ha).trans_le hb
  44   have hspos : 0<s := hp.trans_le hs
  45   have hscale : q/(b*s)≤1/(κ*p*a) := by
  46     calc
  47       _ ≤ 1/(b*s) := div_le_div_of_nonneg_right hq1 (by positivity)
  48       _ ≤ 1/(κ*p*a) := by
  49         apply one_div_le_one_div_of_le (by positivity)
  50         have hh:=mul_le_mul hb hs hp.le hbpos.le
  51         nlinarith only [hh]
  52   have hmul:=mul_le_mul_of_nonneg_left hM (div_nonneg hq (by positivity : 0≤b*s))
  53   have hbound:=mul_le_mul_of_nonneg_right hscale (by positivity : 0≤(1+2*C)*L/2)
  54   have habs : |q/(b*s)*M-q*m/(2*b*s)|=q/(b*s)*|M-m/2| := by
  55     rw [show q/(b*s)*M-q*m/(2*b*s)=q/(b*s)*(M-m/2) by ring,abs_mul,
  56       abs_of_nonneg (div_nonneg hq (by positivity))]
  57   have htriangle:=abs_sub_le (q/(b*s)*M) (q*m/(2*b*s)) 1
  58   rw [habs] at htriangle
  59   have he : (1/(κ*p*a))*((1+2*C)*L/2)=((1+2*C)/(2*κ*p))*L/a := by ring
  60   rw [he] at hbound
  61   linarith only [htriangle,hmul,hbound,hbalance]
