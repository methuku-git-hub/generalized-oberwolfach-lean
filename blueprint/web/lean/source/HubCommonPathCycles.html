  25 theorem rootedCycleOrder_vertices (σ : Equiv.Perm (Fin (m+1))) (v : V) (z : Fin m → V) :
  26     Finset.univ.image (rootedCycleOrder σ v z)=insert v (Finset.univ.image z) := by
  27   ext u
  28   simp only [Finset.mem_image,Finset.mem_univ,true_and,Finset.mem_insert]
  29   constructor
  30   · rintro ⟨i,hi⟩
  31     rcases Fin.eq_zero_or_eq_succ (σ i) with he|⟨j,hj⟩
  32     · exact Or.inl (by simpa only [rootedCycleOrder,he,Fin.cases_zero] using hi.symm)
  33     · exact Or.inr ⟨j,by simpa only [rootedCycleOrder,hj,Fin.cases_succ] using hi⟩
  34   · rintro (rfl|⟨j,rfl⟩)
  35     · exact ⟨σ.symm 0,by simp only [rootedCycleOrder,Equiv.apply_symm_apply,Fin.cases_zero]⟩
  36     · exact ⟨σ.symm j.succ,by simp only [rootedCycleOrder,Equiv.apply_symm_apply,Fin.cases_succ]⟩

  38 theorem rootFirstCommonMarkedPath_hub_cycles (hm : 3<m+1)
  39     (w : U) (j : Bool) (v v' : V)
  40     (hv : if j then Hₖ v w else H₀ v w)
  41     (hv' : if j then Hₖ v' w else H₀ v' w)
  42     (z : RootFirstCommonMarkedPath tail head ktail khead m π b H₀ Hₖ j v v' w w) :
  43     (∃ C : HubMarkedCycle tail head ktail khead m π b H₀ Hₖ w,
  44       hubMarkedCycleVertices w C=insert v (Finset.univ.image z.val)) ∧
  45     (∃ C : HubMarkedCycle tail head ktail khead m π b H₀ Hₖ w,
  46       hubMarkedCycleVertices w C=insert v' (Finset.univ.image z.val)) := by
  47   have hr : (if markedPositionType ((markedRootOrder m j).symm 0) then Hₖ v w else H₀ v w) ∧
  48       (if markedPositionType ((markedRootOrder m j).symm 0) then Hₖ v w else H₀ v w) ∧
  49       (if markedPositionType ((markedRootOrder m j).symm 0) then Hₖ v' w else H₀ v' w) ∧
  50       (if markedPositionType ((markedRootOrder m j).symm 0) then Hₖ v' w else H₀ v' w) := by
  51     rw [markedRootOrder_type m hm j]
  52     exact ⟨hv,hv,hv',hv'⟩
  53   let Z := (rootFirstCommonMarkedPathEquiv j v v' w w) z
  54   let P := orderedCommonMarkedPath_to_pair hr Z
  55   constructor
  56   · refine ⟨⟨P.val.1.val,P.val.1.property,P.property.2.2.2.1⟩,?_⟩

  58     exact rootedCycleOrder_vertices _ _ _

  61     exact rootedCycleOrder_vertices _ _ _
