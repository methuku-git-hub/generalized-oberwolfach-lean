  12 theorem long_assigned_concrete_recipe_identity {n K m:ℕ} {U:Type} [Fintype U]
  13     (F:U→SimpleGraph (Fin n)) (hF:∀u,(F u).IsRegularOfDegree 2) (θ D:ℝ):
  14     let fraction:=fun (u:U) (j:Option (Fin K))=>
  15       ((Finset.univ.filter (fun v=>longPreparationCycleClass (F u) (hF u) K v=j)).card:ℝ)/n
  16     let retained:=fun (u:U)=>Finset.univ.filter (fun i=>θ/(100*(K:ℝ))≤fraction u (some i))
  17     let f:=fun (i:Fin K) u=>if i∈retained u then fraction u (some i) else 0
  18     let c:ℝ:=(m:ℝ)+1
  19     let t:ℝ:=(1/100)/(m:ℝ)
  20     let a:=fun u=>(1-θ)*fraction u none/c-t
  21     let b:=fun u=>(m:ℝ)*a u+(1-θ)*∑ i,f i u
  22     let q:=fun (u:U) (j:Bool)=>if j then a u else t
  23     let sco:=fun (u:U) (j:Bool)=>if j then b u else (1/100:ℝ);
  24     (∀u,a u=longAssignedApproxMarkedWeight F hF K m θ u) ∧
  25     (∀u,b u=longAssignedApproxOrdinaryWeight F hF K m θ u) ∧
  26     q=longAssignedIntervalProbability F hF K m θ ∧
  27     sco=longAssignedSpokeProbability F hF K m θ ∧
  28     (∑u,a u)/D=(∑u,longAssignedApproxMarkedWeight F hF K m θ u)/D ∧
  29     (∑u,b u)/D=(∑u,longAssignedApproxOrdinaryWeight F hF K m θ u)/D ∧
  30     2*(∑u,(1-(1/100:ℝ)-t-a u-b u))/D=
  31       2*(∑u,(1-(1/100:ℝ)-(1/100)/(m:ℝ)-longAssignedApproxMarkedWeight F hF K m θ u-
  32         longAssignedApproxOrdinaryWeight F hF K m θ u))/D :=by
  33   intro fraction retained f c t a b q sco
  34   have ha:∀u,a u=longAssignedApproxMarkedWeight F hF K m θ u:=by intro u;rfl
  35   have hf:∀i u,f i u=(if θ/(100*(K:ℝ))≤ordinaryAssignedFraction F hF K u (some i)
  36       then ordinaryAssignedFraction F hF K u (some i) else 0):=by
  37     intro i u
  38     simp only [f,retained,Finset.mem_filter,Finset.mem_univ,true_and,fraction,ordinaryAssignedFraction]
  39   have hb:∀u,b u=longAssignedApproxOrdinaryWeight F hF K m θ u:=by
  40     intro u
  41     dsimp only [b,longAssignedApproxOrdinaryWeight]
  42     rw [ha u]
  43     congr 1
  44     congr 1
  45     apply Finset.sum_congr rfl
  46     intro i _
  47     exact hf i u
  48   refine ⟨ha,hb,?_,?_,?_,?_,?_⟩
  49   · funext u j;cases j <;> simp only [q,t,longAssignedIntervalProbability,Bool.false_eq_true,↓reduceIte,ha]
  50   · funext u j;cases j <;> simp only [sco,longAssignedSpokeProbability,Bool.false_eq_true,↓reduceIte,hb]
  51   · rfl
  52   · congr 1;exact Finset.sum_congr rfl (fun u _=>hb u)
  53   · congr 1
  54     congr 1
  55     apply Finset.sum_congr rfl
  56     intro u _
  57     rw [ha u,hb u]
