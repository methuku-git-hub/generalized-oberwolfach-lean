   2 /-! The maps on disjoint actual components define one injective partial
   3 vertex map on their union, with no arbitrary extension outside that domain. -/

  10 theorem component_family_partial_embedding
  11     (F:SimpleGraph (Fin n)) (C:A→F.ConnectedComponent)
  12     (hC:Function.Injective C) (f:∀a,C a↪Fin n)
  13     (hdis:∀a b,a≠b→Disjoint (Finset.univ.image (f a)) (Finset.univ.image (f b))):
  14     ∃ψ:{v:Fin n // ∃a,v∈(C a).supp}↪Fin n,
  15       ∀a (x:C a),ψ ⟨x.val,⟨a,x.property⟩⟩=f a x :=by
  16   obtain ⟨src,dst,hsrc,hdst⟩:=component_family_embeddings_glue F C hC f hdis
  17   let ec:=Equiv.ofInjective src src.injective
  18   let intoRange:{v:Fin n // ∃a,v∈(C a).supp}↪Set.range src:=
  19     ⟨fun v=>⟨v.val,by
  20       obtain ⟨a,ha⟩:=v.property
  21       exact ⟨⟨a,⟨v.val,ha⟩⟩,hsrc _⟩⟩,by
  22       intro x y h
  23       apply Subtype.ext
  24       have hv:=congrArg (fun z:Set.range src=>z.val) h
  25       exact hv⟩
  26   let ψ: {v:Fin n // ∃a,v∈(C a).supp}↪Fin n:=
  27     (intoRange.trans ec.symm.toEmbedding).trans dst
  28   refine ⟨ψ,?_⟩
  29   intro a x
  30   have he:intoRange ⟨x.val,⟨a,x.property⟩⟩=ec ⟨a,x⟩:=by
  31     apply Subtype.ext
  32     exact (hsrc ⟨a,x⟩).symm
  33   change dst (ec.symm (intoRange ⟨x.val,⟨a,x.property⟩⟩))=f a x
  34   rw [he,Equiv.symm_apply_apply,hdst]
  35 end Oberwolfach
