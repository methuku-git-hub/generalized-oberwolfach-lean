  16 theorem supported_graph_full_free_degree (G:SimpleGraph V) (F R:Finset V)
  17     (hs:GraphSupportedOn (F∪R) G) (x:V) (hx:x∈F):
  18     G.degree x≤(spanningRestriction G F).degree x+R.card:=by
  19   have hsub:G.neighborFinset x⊆(spanningRestriction G F).neighborFinset x∪R:=by
  20     intro y hy
  21     have hxy:=(G.mem_neighborFinset x y).mp hy
  22     rcases Finset.mem_union.mp (hs x y hxy).2 with hyF|hyR
  23     · exact Finset.mem_union_left _ ((spanningRestriction G F).mem_neighborFinset x y |>.mpr
  24         ((spanningRestriction_adj G F x y).mpr ⟨hxy,hx,hyF⟩))
  25     · exact Finset.mem_union_right _ hyR
  26   have hc:=(Finset.card_le_card hsub).trans (Finset.card_union_le _ _)
  27   simpa only [SimpleGraph.card_neighborFinset_eq_degree] using hc

  29 theorem marked_packing_supported_on_rows_and_images
  30     (M:Finset (MarkedWheel tail head ktail khead m π b H₀ HK)) (F R:Finset V)
  31     (hF:∀ x,π x∈F↔x∈F) (hrows:∀ W∈M,markedWheelRows W⊆F∪R):
  32     GraphSupportedOn (F∪(R∪R.image π)) (markedPackingRimGraph M):=by
  33   intro x y hxy
  34   obtain ⟨W,hW,he⟩:=Finset.mem_biUnion.mp hxy.2
  35   have hs:=markedWheelRimGraph_supported W x y ⟨hxy.1,he⟩
  36   have hsub:markedWheelRows W∪(markedWheelRows W).image π⊆F∪(R∪R.image π):=by
  37     intro z hz
  38     rcases Finset.mem_union.mp hz with hz|hz
  39     · rcases Finset.mem_union.mp (hrows W hW hz) with hz|hz
  40       · exact Finset.mem_union_left _ hz
  41       · exact Finset.mem_union_right _ (Finset.mem_union_left _ hz)
  42     · obtain ⟨a,ha,rfl⟩:=Finset.mem_image.mp hz
  43       rcases Finset.mem_union.mp (hrows W hW ha) with ha|ha
  44       · exact Finset.mem_union_left _ ((hF a).mpr ha)
  45       · exact Finset.mem_union_right _ (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨a,ha,rfl⟩))
  46   exact ⟨hsub hs.1,hsub hs.2⟩

  48 theorem marked_hub_row_full_free_rim_degree
  49     (M:Finset (MarkedWheel tail head ktail khead m π b H₀ HK))
  50     (v:Fin n→V) (F:Finset V) (hF:∀ x,π x∈F↔x∈F)
  51     (hrows:∀ W∈M,markedWheelRows W⊆F∪Finset.univ.image v)
  52     (D:ℕ) (hdeg:∀ x,(spanningRestriction (markedPackingRimGraph M) F).degree x≤D):
  53     ∀ x∈F,(markedPackingRimGraph M).degree x≤D+2*n:=by
  54   let R:=Finset.univ.image v
  55   have hc:R.card≤n:=by exact (Finset.card_image_le).trans_eq (by simp)
  56   have hci:(R.image π).card≤n:=Finset.card_image_le.trans hc
  57   have htot:(R∪R.image π).card≤2*n:=by
  58     have hh:=(Finset.card_union_le R (R.image π)).trans (Nat.add_le_add hc hci)
  59     omega
  60   intro x hx
  61   have hh:=supported_graph_full_free_degree (markedPackingRimGraph M) F (R∪R.image π)
  62     (marked_packing_supported_on_rows_and_images M F R hF hrows) x hx
  63   exact hh.trans (Nat.add_le_add (hdeg x) htot)
