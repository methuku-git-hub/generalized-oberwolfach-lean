  25 theorem marked_active_inner_parent_degrees
  26     (f:VA↪V) (g:UA↪U) (P:SimpleGraph V)
  27     (I T:Finset V) (J W:Finset U) (X:Finset VA) (Z:Finset UA)
  28     (hX:X.image f=I\T) (hZ:Z.image g=J\W) (hW:W⊆J)
  29     (rho gamma err:ℝ) (hrho:0<rho) (hg:0≤gamma)
  30     (hT:(T.card:ℝ)≤(gamma/2)*I.card)
  31     (hparent:∀ v,|((P.neighborFinset v∩I).card:ℝ)-2*J.card/rho|≤err)
  32     (hmin:∀ v,(1/2+gamma)*I.card≤((P.neighborFinset v∩I).card:ℝ)):
  33     (∀ x,|(((P.comap f).neighborFinset x∩X).card:ℝ)-2*Z.card/rho|
  34       ≤err+T.card+2*W.card/rho)∧
  35     (∀ x,(1/2+gamma/2)*X.card≤((X∩(P.comap f).neighborFinset x).card:ℝ)):=by
  36   have hZcard:Z.card=(J\W).card:=by
  37     rw [←hZ,Finset.card_image_of_injective _ g.injective]
  38   have hJcard:(J.card:ℝ)=Z.card+W.card:=by
  39     exact_mod_cast (show J.card=Z.card+W.card by rw [hZcard];exact (Finset.card_sdiff_add_card_eq_card hW).symm)
  40   have hXle:(X.card:ℝ)≤I.card:=by
  41     have hh: X.card=(I\T).card:=by rw [←hX,Finset.card_image_of_injective _ f.injective]
  42     rw [hh];exact_mod_cast Finset.card_le_card (Finset.sdiff_subset : I\T⊆I)
  43   have hcounts (x:VA):
  44       (((P.comap f).neighborFinset x∩X).card:ℝ)≤(P.neighborFinset (f x)∩I).card∧
  45       ((P.neighborFinset (f x)∩I).card:ℝ)≤((P.comap f).neighborFinset x∩X).card+T.card:=by
  46     have he:P.neighborFinset (f x)∩(I\T)=(P.neighborFinset (f x)∩I)\T:=by ext y;simp;tauto
  47     rw [graph_neighbourhood_embedding_card f P X x,hX,he]
  48     constructor
  49     · exact_mod_cast Finset.card_le_card (Finset.sdiff_subset : (P.neighborFinset (f x)∩I)\T⊆P.neighborFinset (f x)∩I)
  50     · have hc:=Finset.card_sdiff_add_card_inter (P.neighborFinset (f x)∩I) T
  51       have hle:((P.neighborFinset (f x)∩I)∩T).card≤T.card:=Finset.card_le_card Finset.inter_subset_right
  52       exact_mod_cast (show (P.neighborFinset (f x)∩I).card≤((P.neighborFinset (f x)∩I)\T).card+T.card by omega)
  53   have htarget:2*(J.card:ℝ)/rho=2*Z.card/rho+2*W.card/rho:=by rw [hJcard];ring
  54   have hW0:0≤2*(W.card:ℝ)/rho:=by positivity
  55   constructor
  56   · intro x
  57     have hc:=hcounts x
  58     have hp:=abs_le.mp (hparent (f x))
  59     rw [htarget] at hp
  60     have ht0:0≤(T.card:ℝ):=Nat.cast_nonneg _
  61     apply abs_le.mpr
  62     constructor <;> linarith only [hc.1,hc.2,hp.1,hp.2,ht0,hW0]
  63   · intro x
  64     have hc:=(hcounts x).2
  65     have hp:=hmin (f x)
  66     have hh:=mul_le_mul_of_nonneg_left hXle (show 0≤1/2+gamma/2 by positivity)
  67     rw [Finset.inter_comm]
  68     nlinarith only [hc,hp,hT,hh]
