  62 theorem finiteLaw_positive_event_witness {Ω:Type} [Fintype Ω]
  63     (μ:FiniteLaw Ω) (E:Ω→Prop) (h:0<μ.probability E):
  64     ∃ω,0<μ.mass ω ∧ E ω:=by
  65   have hn:∀ω∈(Finset.univ:Finset Ω),0≤(if E ω then μ.mass ω else 0):=by
  66     intro ω _
  67     split_ifs <;> simp_all [μ.nonneg]
  68   have hp:0<∑ω:Ω,if E ω then μ.mass ω else 0:=by
  69     unfold FiniteLaw.probability at h
  70     convert h using 1
  71   obtain ⟨ω,_,hω⟩:=(Finset.sum_pos_iff_of_nonneg hn).mp hp
  72   by_cases he:E ω
  73   · exact ⟨ω,by simpa [he] using hω,he⟩
  74   · simp [he] at hω
