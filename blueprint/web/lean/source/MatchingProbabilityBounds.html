  21     μ*(Real.exp (η/4)-1)-(η/4)*(d+1)≤-η^2*h^4/8 := by

  52 theorem matching_normalization_error {x W D E η : ℝ}
  53     (hW : 0≤W) (hD : 0<D) (hη : 0≤η) (hDE : D≤E) (hED : E≤(1+8*η)*D)
  54     (hx : |x-W/E|<η*W/D+η*W/E) :
  55     |x-W/D|<10*η*W/D := by
  56   have hE : 0<E := hD.trans_le hDE
  57   have hdiv : W/E≤W/D := div_le_div_of_nonneg_left hW hD hDE
  58   have hdiff : W/D-W/E≤8*η*W/D := by
  59     apply (le_div_iff₀ hD).mpr
  60     have haux : E-D≤8*η*E := by nlinarith
  61     have hm := mul_le_mul_of_nonneg_left haux hW
  62     apply (mul_le_mul_iff_of_pos_right hE).mp
  63     field_simp
  64     nlinarith
  65   calc
  66     _ ≤ |x-W/E|+|W/E-W/D| := by convert abs_add_le (x-W/E) (W/E-W/D) using 1 <;> ring
  67     _ < (η*W/D+η*W/E)+8*η*W/D := add_lt_add_of_lt_of_le hx (by rw [abs_of_nonpos (by linarith)];linarith)
  68     _ ≤ 10*η*W/D := by

  70       simp only [← mul_div_assoc] at hm
