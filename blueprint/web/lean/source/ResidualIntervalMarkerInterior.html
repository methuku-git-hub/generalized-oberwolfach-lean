  10 theorem path_marker_in_interval_interior_iff {F:SimpleGraph V} {u v:V}
  11     (p:F.Walk u v) (hp:p.IsPath) (a b j:ℕ) (hab:a≤b)
  12     (hb:b≤p.length) (hj:j≤p.length):
  13     p.getVert j∈(walkInterval p a b hab).support.toFinset \ {p.getVert a,p.getVert b} ↔
  14       a<j ∧ j<b := by

  16   · intro hx
  17     have hx':=Finset.mem_sdiff.mp hx
  18     obtain ⟨k,hak,hkb,hk⟩:=(walk_interval_mem_support_iff p a b hab hb _).mp

  22     have hnotB:p.getVert j≠p.getVert b := by intro h;apply hx'.2;simp [h]
