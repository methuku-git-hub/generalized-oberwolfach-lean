  14 structure OrdinaryTransferConfiguration {V U : Type*} {r : ℕ} (hr : 3≤r)
  15     (Q F : SimpleGraph V) (H : V → U → Prop) (v : V) (u u' : U) where
  16   labels : (Fin r × Fin r) ↪ V
  17   omitted : v∉Set.range labels
  18   rim_edges : ∀ j : TransferCycleIndex r,transferCycleGraph (by omega) labels v omitted j≤Q
  19   spokes : ∀ z∈insert v (Set.range labels),H z u ∧ H z u'
  20   independent : ∀ x∈insert v (Set.range labels),∀ y∈insert v (Set.range labels),¬F.Adj x y

  32 theorem transferCycleGraph_support_subset {V : Type*} {r : ℕ} (hr : 3≤r)
  33     (f : (Fin r × Fin r) ↪ V) (v : V) (hv : v∉Set.range f)
  34     (j : TransferCycleIndex r) :
  35     (transferCycleGraph (by omega) f v hv j).support⊆insert v (Set.range f) := by
  36   rcases j with ⟨⟩|⟨i,b⟩
  37   · rw [transferCycleGraph,cycle_map_support hr]
  38     rintro z ⟨i,rfl⟩
  39     exact Set.mem_insert_of_mem _ ⟨(i,⟨0,by omega⟩),rfl⟩
  40   · cases b
  41     · rw [transferCycleGraph,cycle_map_support hr]
  42       rintro z ⟨j,rfl⟩
  43       exact Set.mem_insert_of_mem _ ⟨(i,j),rfl⟩
  44     · rw [transferCycleGraph,cycle_map_support hr]
  45       rintro z ⟨j,rfl⟩
  46       change (if j=⟨0,by omega⟩ then v else f (i,j))∈insert v (Set.range f)
  47       split_ifs
  48       · exact Set.mem_insert _ _
  49       · exact Set.mem_insert_of_mem _ ⟨(i,j),rfl⟩

  51 def OrdinaryTransferConfiguration.wheel {V U : Type*} {r : ℕ} {hr : 3≤r}
  52     {Q F : SimpleGraph V} {H : V → U → Prop} {v : V} {u u' : U}
  53     (T : OrdinaryTransferConfiguration hr Q F H v u u') (b : Bool)
  54     (j : TransferCycleIndex r) : OrdinaryWheel Q F H r :=
  55   ⟨(⟨⟨transferCycleGraph (by omega) T.labels v T.omitted j,
  56       transferCycleGraph_isCycle hr T.labels v T.omitted j⟩,T.rim_edges j⟩,
  57       if b then u else u'),by
  58     constructor
  59     · intro z hz
  60       have hh := T.spokes z (transferCycleGraph_support_subset hr T.labels v T.omitted j hz)
  61       cases b
  62       · exact hh.2
  63       · exact hh.1
  64     · intro x hx y hy
  65       exact T.independent x (transferCycleGraph_support_subset hr T.labels v T.omitted j hx)
  66         y (transferCycleGraph_support_subset hr T.labels v T.omitted j hy)⟩

  68 def OrdinaryTransferConfiguration.weight {V U : Type*} {r : ℕ} {hr : 3≤r}
  69     {Q F : SimpleGraph V} {H : V → U → Prop} {v : V} {u u' : U}
  70     (T : OrdinaryTransferConfiguration hr Q F H v u u') : OrdinaryWheel Q F H r → ℝ :=

  74 theorem OrdinaryTransferConfiguration.loads {V U : Type*} [Fintype V] [Fintype U]
  75     [DecidableEq V] [DecidableEq U] {r : ℕ} {hr : 3≤r}
  76     {Q F : SimpleGraph V} {H : V → U → Prop} {v : V} {u u' : U}
  77     (T : OrdinaryTransferConfiguration hr Q F H v u u') :
  78     (∀ x y,ordinaryRimLoad T.weight x y=0) ∧
  79     (∀ z h,ordinarySpokeLoad T.weight z h=
  80       (if z=v then (1:ℝ) else 0)*((if h=u then 1 else 0)-(if h=u' then 1 else 0))) := by
  81   apply indexed_wheel_transfer_loads (T.wheel true) (T.wheel false) transferCycleCoefficient v u u'
  82     (fun _ => rfl) (fun _ => rfl) (fun _ => rfl)
  83   intro z
  84   have hh := transfer_index_vertex_load hr T.labels v T.omitted z

  86   · apply Finset.sum_congr rfl
  87     intro j _
  88     dsimp only [ordinaryWheelRim,OrdinaryTransferConfiguration.wheel]
