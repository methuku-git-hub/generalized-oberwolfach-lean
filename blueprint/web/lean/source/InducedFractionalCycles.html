  12 def spanningRestriction {V : Type*} (G : SimpleGraph V) (S : Finset V) : SimpleGraph V :=
  13   (G.induce (S:Set V)).map (Function.Embedding.subtype _)

  15 theorem spanningRestriction_adj {V : Type*} (G : SimpleGraph V) (S : Finset V) (u v : V) :
  16     (spanningRestriction G S).Adj u v ↔ G.Adj u v ∧ u∈S ∧ v∈S := by
  17   rw [spanningRestriction,SimpleGraph.map_adj]
  18   constructor
  19   · rintro ⟨x,y,hxy,hx,hy⟩
  20     subst u
  21     subst v
  22     exact ⟨hxy,x.property,y.property⟩
  23   · rintro ⟨hG,hu,hv⟩
  24     exact ⟨⟨u,hu⟩,⟨v,hv⟩,hG,rfl,rfl⟩
  25 
  26 theorem spanningRestriction_le {V : Type*} (G : SimpleGraph V) (S : Finset V) :
  27     spanningRestriction G S≤G := fun u v h => ((spanningRestriction_adj G S u v).mp h).1

  29 theorem spanningRestriction_support {V : Type*} (G : SimpleGraph V) (S : Finset V) :
  30     (spanningRestriction G S).support⊆(S:Set V) := by
  31   intro u hu
  32   obtain ⟨v,huv⟩ := hu
  33   exact ((spanningRestriction_adj G S u v).mp huv).2.1

  35 theorem induced_degree_intersection {V : Type*} [Fintype V] [DecidableEq V]
  36     (G : SimpleGraph V) (S : Finset V) (v : (S:Set V)) :
  37     (G.induce (S:Set V)).degree v=(G.neighborFinset v.val∩S).card := by
  38   have hh := congrArg Finset.card (G.map_neighborFinset_induce (s:=(S:Set V)) v)
  39   simp only [Finset.card_map,SimpleGraph.card_neighborFinset_eq_degree,
  40     ← SimpleGraph.ncard_neighborSet] at hh ⊢
  41   simpa using hh

  43 theorem cycle_fractional_on_subset (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  44     ∃ N : ℕ,∀ (V : Type) [Fintype V] [DecidableEq V],∀ (G : SimpleGraph V) (S : Finset V),
  45       N≤S.card → (∀ v∈S,(cycleThreshold r+α)*S.card≤((G.neighborFinset v∩S).card:ℝ)) →
  46       HasFractionalDecomposition (spanningRestriction G S) (fun C => IsEmbeddedCycle C r) := by
  47   obtain ⟨N,hN⟩ := cycle_fractional_positive_margin_finite r hr α hα
  48   refine ⟨N,?_⟩
  49   intro V _ _ G S hn hdeg
  50   have hh : HasFractionalDecomposition (G.induce (S:Set V)) (fun C => IsEmbeddedCycle C r) := by
  51     apply hN (S:Set V) (by simpa using hn)
  52     intro v
  53     have hd := hdeg v.val v.property
  54     have he := induced_degree_intersection G S v
  55     simp only [← SimpleGraph.ncard_neighborSet] at he ⊢
  56     rw [he]
  57     simpa only [Fintype.card_coe,Finset.coe_sort_coe] using hd
  58   exact hh.map_embedded_cycles (Function.Embedding.subtype _)
