  16 theorem marked_resource_supported_inter
  17     (hp : Function.Injective (markedPhysicalEdge tail head ktail khead π))
  18     (A B : Finset V) (e : Finset V⊕(V×U))
  19     (hA : MarkedResourceSupported tail head ktail khead π A e)
  20     (hB : MarkedResourceSupported tail head ktail khead π B e) :
  21     MarkedResourceSupported tail head ktail khead π (A∩B) e := by
  22   cases e with
  23   | inl e=>
  24     obtain ⟨a,ha,hat,hah⟩ := hA
  25     obtain ⟨b,hb,hbt,hbh⟩ := hB
  26     have hab : a=b := hp (ha.trans hb.symm)
  27     subst b
  28     exact ⟨a,ha,Finset.mem_inter.mpr ⟨hat,hbt⟩,Finset.mem_inter.mpr ⟨hah,hbh⟩⟩
  29   | inr vu=>exact Finset.mem_inter.mpr ⟨hA,hB⟩

  31 theorem marked_wheel_resources_supported {m b : ℕ} {H₀ HK : V→U→Prop}
  32     (C : MarkedWheel tail head ktail khead m π b H₀ HK) (A : Finset V)
  33     (hA : markedWheelRows C⊆A) :
  34     ∀ e∈markedWheelPhysicalResources C,MarkedResourceSupported tail head ktail khead π A e := by
  35   intro e he
  36   have hh := marked_cycle_at_hub_supported C.1 C.2.val A
  37     (fun i=>hA (Finset.mem_image.mpr ⟨i,Finset.mem_univ _,rfl⟩)) e
  38   apply hh
  39   cases e with
  40   | inl e=>
  41     apply Finset.mem_union_left
  42     exact Finset.mem_image.mpr ⟨e,(markedWheelPhysicalResources_rim C e).mp he,rfl⟩
  43   | inr vu=>
  44     obtain ⟨hu,hv⟩ := (markedWheelPhysicalResources_spoke C vu.1 vu.2).mp he
  45     apply Finset.mem_union_right
  46     exact Finset.mem_image.mpr ⟨vu.1,hv,by rw [hu]⟩

  48 theorem marked_row_cover_preserves_other_root {m b : ℕ} {H₀ HK : V→U→Prop}
  49     (hp : Function.Injective (markedPhysicalEdge tail head ktail khead π))
  50     (C : MarkedWheel tail head ktail khead m π b H₀ HK)
  51     (F : Finset V) (v w : V) (hwv : w≠v) (hwF : w∉F)
  52     (hrows : markedWheelRows C⊆insert v F) :
  53     (∀ a : E⊕K,Sum.elim tail ktail a=w ∨ Sum.elim head khead a=w →
  54       Sum.inl (markedPhysicalEdge tail head ktail khead π a)∉markedWheelPhysicalResources C) ∧
  55     (∀ u,Sum.inr (w,u)∉markedWheelPhysicalResources C) := by
  56   have hw : w∉insert v F := by simpa using And.intro hwv hwF
  57   constructor
  58   · intro a ha he
  59     obtain ⟨b,hb,hbt,hbh⟩ := marked_wheel_resources_supported C (insert v F) hrows _ he
  60     have hab : b=a := hp hb
  61     subst b
  62     rcases ha with ha|ha
  63     · exact hw (ha ▸ hbt)
  64     · exact hw (ha ▸ hbh)
  65   · intro u he
  66     exact hw (marked_wheel_resources_supported C (insert v F) hrows _ he)

  72     (e : Finset V⊕(V×U)) (he : MarkedResourceSupported tail head ktail khead π F e) :
  73     markedFreeResource F e := by

  83 theorem marked_row_comparison_resources_disjoint
  84     (hp : Function.Injective (markedPhysicalEdge tail head ktail khead π))
  85     (T F : Finset V) (hF : ∀ z,π z∈F ↔ z∈F) (v : V) (hvT : v∉T)
  86     (hTF : Disjoint T F) (M C : Finset (Finset V⊕(V×U)))
  87     (hM : ∀ e∈M,MarkedResourceSupported tail head ktail khead π (T∪F) e)
  88     (hC : ∀ e∈C,MarkedResourceSupported tail head ktail khead π (insert v F) e)
  89     (hfree : Disjoint C (M.filter (markedFreeResource F))) : Disjoint C M := by
  90   have hi : (T∪F)∩insert v F⊆F := by
  91     intro z hz
  92     obtain ⟨hzT,hzV⟩ := Finset.mem_inter.mp hz
  93     rcases Finset.mem_union.mp hzT with hzT|hzF
  94     · rcases Finset.mem_insert.mp hzV with rfl|hzF
  95       · exact False.elim (hvT hzT)
  96       · exact False.elim (Finset.disjoint_left.mp hTF hzT hzF)
  97     · exact hzF
  98   apply Finset.disjoint_left.mpr
  99   intro e he hf
 100   have hsupport := marked_resource_supported_inter hp (T∪F) (insert v F) e (hM e hf) (hC e he)
 101   have hsmall := marked_resource_support_mono hi e hsupport
 102   exact Finset.disjoint_left.mp hfree he
 103     (Finset.mem_filter.mpr ⟨hf,marked_supported_resource_is_free F hF e hsmall⟩)
