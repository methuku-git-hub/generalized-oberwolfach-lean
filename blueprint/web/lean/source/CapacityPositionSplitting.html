  14 theorem capacity_position_splitting
  15     {V E : Type*} [Fintype V] [Fintype E] [DecidableEq V]
  16     (tail head : E → V) (k : ℕ) (hk : 0<k) (w : E → ℝ)
  17     (p q O I : V → ℝ) (M z err : ℝ) (herr : 0≤err)
  18     (hrow : ∀ v,(∑ e,if tail e=v then w e else 0)=O v)
  19     (hcol : ∀ v,(∑ e,if head e=v then w e else 0)=I v)
  20     (hbalance : ∀ v,O v-I v=p v-q v)
  21     (hp : ∑ v,p v=M) (hq : ∑ v,q v=M) (hO : ∑ v,O v=(k+1)*M)
  22     (hpclose : ∀ v,|p v-z|≤err) (hqclose : ∀ v,|q v-z|≤err)
  23     (hOclose : ∀ v,|O v-(k+1)*z|≤err) (hIclose : ∀ v,|I v-(k+1)*z|≤err)
  24     (κ s B δ ε h : ℝ) (hw : ∀ e,|w e-1|≤κ) (hs : 0<s) (hB : 1≤B)
  25     (hn : 0<Fintype.card V) (hδ : 0≤δ) (hδsmall : δ<1/4)
  26     (hdegp : ∀ v,|occurrenceDegree tail v-(Fintype.card E : ℝ)/Fintype.card V|≤ε)
  27     (hdegm : ∀ v,|occurrenceDegree head v-(Fintype.card E : ℝ)/Fintype.card V|≤ε)
  28     (hsmall : s*ε+3*err ≤ δ*s*((Fintype.card E : ℝ)/Fintype.card V))
  29     (hMC : (Fintype.card E : ℝ)/B ≤ h*Fintype.card V*Fintype.card V/2)
  30     (hlarge : ∀ (S T : Finset V),
  31       1-2*δ < (S.card : ℝ)/Fintype.card V+(T.card : ℝ)/Fintype.card V →
  32       h*Fintype.card V*Fintype.card V*
  33         min ((S.card : ℝ)/Fintype.card V) ((T.card : ℝ)/Fintype.card V) ≤
  34         ((Finset.univ.filter (fun e => tail e∈S ∧ head e∈T)).card : ℝ))
  35     (hbudget : κ/(k+1:ℝ)+k*(B*s) ≤ 1/(2*(k+1:ℝ))) :
  36     ∃ f : Fin (k+1) → E → ℝ,
  37       (∀ e,(∑ j,f j e)=w e) ∧
  38       (∀ j v,(∑ e,if tail e=v then f j e else 0)=capacityPositionDemands k (p v) (q v) (O v) j.castSucc) ∧
  39       (∀ j v,(∑ e,if head e=v then f j e else 0)=capacityPositionDemands k (p v) (q v) (O v) j.succ) ∧
  40       (∀ j e,1/(2*(k+1:ℝ)) ≤ f j e ∧ f j e ≤ 2/(k+1:ℝ)) := by
  41   classical
  42   let d (j : Fin (k+2)) (v : V) := capacityPositionDemands k (p v) (q v) (O v) j
  43   have hdtotal (j : Fin (k+2)) : (∑ v,d j v)=M := capacityPositionDemands_total k hk p q O M hp hq hO j
  44   have hI : (∑ v,I v)=(k+1)*M := by
  45     have he (v : V) : I v=O v-p v+q v := by linarith [hbalance v]
  46     simp only [he,Finset.sum_add_distrib,Finset.sum_sub_distrib,hO,hp,hq]
  47     ring
  48   have hderror (j : Fin (k+2)) (v : V) : |d j v-z|≤2*err :=
  49     capacityPositionDemands_error k hk (p v) (q v) (O v) z err herr (hpclose v) (hqclose v) (hOclose v) j
  50   let Δp (j : Fin k) (v : V) := d j.castSucc.castSucc v-O v/(k+1:ℝ)
  51   let Δm (j : Fin k) (v : V) := d j.castSucc.succ v-I v/(k+1:ℝ)
  52   have hΔp (j : Fin k) : (∑ v,Δp j v)=0 := equal_share_correction_total k _ O M (hdtotal _) hO
  53   have hΔm (j : Fin k) : (∑ v,Δm j v)=0 := equal_share_correction_total k _ I M (hdtotal _) hI
  54   have hΔpb (j : Fin k) (v : V) : |Δp j v|≤3*err := equal_share_correction_bound k _ (O v) z err herr (hderror _ v) (hOclose v)
  55   have hΔmb (j : Fin k) (v : V) : |Δm j v|≤3*err := equal_share_correction_bound k _ (I v) z err herr (hderror _ v) (hIclose v)
  56   have hex (j : Fin k) := bounded_signed_degree_correction tail head (Δp j) (Δm j)
  57     (hΔp j) (hΔm j) s B δ ε (3*err) h hs hB hn hδ hδsmall hdegp hdegm (hΔpb j) (hΔmb j) hsmall hMC hlarge
  58   choose u hu hut huh using hex
  59   let f := capacityPositionWeights k w u
  60   have hfrow : ∀ j v,(∑ e,if tail e=v then f j e else 0)=d j.castSucc v := by
  61     apply capacityPositionWeights_occurrence_marginals tail k w u (fun j => d j.castSucc)
  62     · intro v
  63       rw [hrow]
  64       exact capacityPositionDemands_outgoing k hk (p v) (q v) (O v)
  65     · intro j v
  66       simpa only [hrow] using hut j v
  67   have hfcol : ∀ j v,(∑ e,if head e=v then f j e else 0)=d j.succ v := by
  68     apply capacityPositionWeights_occurrence_marginals head k w u (fun j => d j.succ)
  69     · intro v
  70       rw [hcol]
  71       exact capacityPositionDemands_incoming k hk (p v) (q v) (O v) (I v) (hbalance v)
  72     · intro j v
  73       simpa only [hcol] using huh j v
  74   refine ⟨f,?_,hfrow,hfcol,?_⟩
  75   · intro e
  76     simpa only [Finset.sum_apply] using congrFun (capacityPositionWeights_total k w u) e
  77   · intro j e
  78     have hBs : 0≤B*s := mul_nonneg (by linarith) hs.le
  79     have hh := (capacityPositionWeights_bounds k hk w u κ (B*s) hBs hw hu j e).trans hbudget
  80     obtain ⟨hlo,hup⟩ := abs_le.mp hh
  81     have hid : 1/(k+1:ℝ)=2*(1/(2*(k+1:ℝ))) := by field_simp
  82     have hpos : 0≤1/(k+1:ℝ) := by positivity
  83     have hi2 : 2/(k+1:ℝ)=2*(1/(k+1:ℝ)) := by ring
  84     dsimp only [f]
  85     constructor <;> linarith
