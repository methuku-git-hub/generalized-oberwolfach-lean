   9 def complementDomainEmbedding {X:Type*} [DecidableEq X] {A D:Finset X} (hDA:D⊆A) :
  10     {x // x∉A}↪{x // x∉D} where
  11   toFun x:=⟨x.val,fun hx=>x.property (hDA hx)⟩
  12   inj' := by intro x y h;apply Subtype.ext;exact congrArg (fun z:{x // x∉D}=>z.val) h

  14 def endpointDomainEmbedding {X:Type*} [DecidableEq X] {E D:Finset X} (hED:Disjoint E D) :

  19 theorem residual_spoke_card_from_complement_embedding
  20     {X V:Type*} [Fintype X] [DecidableEq X] [Fintype V] [DecidableEq V]
  21     (A D E:Finset X) (hDA:D⊆A) (hEA:E⊆A) (hED:Disjoint E D)
  22     (fC:{x // x∉D}↪V) (hcard:Fintype.card X=Fintype.card V)
  23     (R Z B:Finset V)
  24     (hZ:Z=Finset.univ.image ((complementDomainEmbedding hDA).trans fC))
  25     (hB:B=Finset.univ.image ((endpointDomainEmbedding hED).trans fC))
  26     (hR:R=Finset.univ\(Z∪B)) : R.card=A.card-E.card := by
  27   classical
  28   have hZcard:Z.card=Fintype.card X-A.card:=by
  29     rw [hZ,Finset.card_image_of_injective _ ((complementDomainEmbedding hDA).trans fC).injective]
  30     simpa using Fintype.card_subtype_compl (fun x:X=>x∈A)
  31   have hBcard:B.card=E.card:=by
  32     rw [hB,Finset.card_image_of_injective _ ((endpointDomainEmbedding hED).trans fC).injective]
  33     simp
  34   have hZB:Disjoint Z B:=by
  35     rw [hZ,hB]
  36     apply Finset.disjoint_left.mpr
  37     intro v hvz hvb
  38     obtain ⟨x,_,hx⟩:=Finset.mem_image.mp hvz
  39     obtain ⟨y,_,hy⟩:=Finset.mem_image.mp hvb
  40     have heq:fC (complementDomainEmbedding hDA x)=fC (endpointDomainEmbedding hED y):=hx.trans hy.symm
  41     have hxy:complementDomainEmbedding hDA x=endpointDomainEmbedding hED y:=fC.injective heq
  42     have hval:x.val=y.val:=congrArg (fun z:{x // x∉D}=>z.val) hxy
  43     exact x.property (hval.symm ▸ hEA y.property)
  44   rw [hR,Finset.card_sdiff_of_subset (Finset.subset_univ _),Finset.card_univ,
  45     Finset.card_union_of_disjoint hZB,hZcard,hBcard,←hcard]
  46   have hA:A.card≤Fintype.card X:=Finset.card_le_univ A
  47   have hE:E.card≤A.card:=Finset.card_le_card hEA
  48   omega
  49 end Oberwolfach
