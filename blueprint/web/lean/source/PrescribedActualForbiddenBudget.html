  14 theorem prescribed_actual_forbidden_budget
  15     (Q:I→SimpleGraph X) (A A₀:I→Finset X) (f ψ:I→X→V) (T:Finset (I×X))
  16     (hinj:FamilyPartialInjections A f)
  17     (hdom:∀ i x,x∈A i ↔ x∈A₀ i ∨ (i,x)∈T)
  18     (hpres:∀ i x,x∈A₀ i→f i x=ψ i x) (hQ:∀ i x,(Q i).degree x≤2)
  19     (R:I→Finset (Sym2 X)) (E:I→Finset (Sym2 V))
  20     (hrepr:∀ i,E i=(R i).image (Sym2.map (f i)))
  21     (hsource:∀ i,R i⊆(Q i).edgeFinset)
  22     (hends:∀ i e,e∈R i→∀ x∈e.toFinset,x∈A i)
  23     (β δ n:ℝ) (hβ:0<β) (hβ1:β≤1) (hδ:0<δ) (hδbound:δ≤(β/100)^2)
  24     (hn:2/Real.sqrt δ≤n)
  25     (hold:∀ v,((Finset.univ.filter (fun i=>∃ x,x∈A₀ i ∧ ψ i x=v)).card:ℝ)≤δ*n)
  26     (hfree:∀ v,((T.filter (fun p=>f p.1 p.2=v)).card:ℝ)≤Real.sqrt δ*n+1)
  27     (J:SimpleGraph V) (hJ:∀ v,(J.degree v:ℝ)≤δ*n):
  28     ∃ F:SimpleGraph V,(∀ v,(F.degree v:ℝ)≤8*Real.sqrt δ*n) ∧
  29       J.edgeFinset⊆F.edgeFinset ∧ (∀ i,E i⊆F.edgeFinset) := by
  30   obtain ⟨F,hF,hJF,hHi⟩:=prescribed_actual_forbidden_graph Q A A₀ f ψ T hinj hdom hpres hQ δ n (Real.sqrt δ*n) hold hfree J hJ
  31   have hnum:=prescribed_embedding_margins_from_hypotheses β δ n hβ hβ1 hδ hδbound hn
  32   refine ⟨F,?_,hJF,?_⟩
  33   · intro v
  34     have hv:=hF v
  35     nlinarith [hnum.2.1]
  36   · intro i
  37     rw [hrepr i]
