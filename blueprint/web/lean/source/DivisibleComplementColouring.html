  16 theorem dense_divisible_complement_colouring (q h : ℕ) (hq : Even q) (hh : 0<h) (hhq : h∣q)
  17     (ε ξ : ℝ) (hε : 0<ε) (hξ : 0<ξ) (hξsmall : ξ≤1/(q+1:ℝ)) :
  18     ∃ N : ℕ,∀ (V : Type u) [Fintype V] [DecidableEq V],N≤Fintype.card V →
  19     ∀ G : SimpleGraph V,(∀ v,(1/2+ε)*Fintype.card V≤(G.degree v:ℝ)) →
  20     ∃ τ : Sym2 V → Fin (q+1),
  21       (∀ a,(∀ v,Even ((G\edgeColourGraph G τ a).degree v)) ∧
  22         h∣(G\edgeColourGraph G τ a).edgeFinset.card) ∧
  23       (∀ a v,((edgeColourGraph G τ a).degree v:ℝ)≤(G.degree v:ℝ)/(q+1:ℝ)+ξ*Fintype.card V) := by
  24   let ρ := ε/(q+1:ℝ)^2
  25   have hρ : 0<ρ := by dsimp [ρ];positivity
  26   obtain ⟨N₁,hN₁⟩ := divisibility_uniform_colour_sample (q+1) (by omega) ε ξ hε hξ
  27   obtain ⟨L,N₂,hL,hN₂⟩ := divisibility_repair_scales q h ρ ξ hρ hξ
  28   refine ⟨max N₁ N₂,?_⟩
  29   intro V instV instDec hn G hdeg
  30   obtain ⟨hnpos,hr,hlose,hpack,hchange⟩ := hN₂ (Fintype.card V) ((le_max_right _ _).trans hn)
  31   letI : Nonempty V := Fintype.card_pos_iff.mp hnpos
  32   have hnr : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr hnpos
  33   obtain ⟨σ,hσdeg,hσpath⟩ := hN₁ V ((le_max_left _ _).trans hn) G hdeg (Fin.last q)
  34   have hreserve : q*h≤(edgeColourGraph G σ (Fin.last q)).edgeFinset.card := by
  35     let v : V := Classical.choice inferInstance
  36     have hD : (Fintype.card V:ℝ)/2≤(G.degree v:ℝ) := by
  37       have hh := hdeg v
  38       have he := mul_nonneg hε.le hnr.le
  39       nlinarith
  40     have hdiv := div_le_div_of_nonneg_right hD (show (0:ℝ)≤q+1 by positivity)
  41     have herr := div_le_div_of_nonneg_right (mul_le_mul_of_nonneg_right hξsmall hnr.le)
  42       (show (0:ℝ)≤4 by norm_num)
  43     have hs := (abs_lt.mp (hσdeg (Fin.last q) v)).1
  44     have hqne : (q:ℝ)+1≠0 := by positivity
  45     have hc : (q:ℝ)*h≤((edgeColourGraph G σ (Fin.last q)).degree v:ℝ) := by
  46       simp only [Nat.cast_add,Nat.cast_one] at hs
  47       have he1 : ((Fintype.card V:ℝ)/2)/(q+1:ℝ)=(Fintype.card V:ℝ)/(4*(q+1:ℝ))*2 := by field_simp [hqne] <;> ring
  48       have he2 : (1/(q+1:ℝ)*Fintype.card V)/4=(Fintype.card V:ℝ)/(4*(q+1:ℝ)) := by field_simp [hqne] <;> ring
  49       rw [he1] at hdiv
  50       rw [he2] at herr
  51       linarith only [hr,hdiv,herr,hs]
  52     have hcNat : q*h≤(edgeColourGraph G σ (Fin.last q)).degree v := by exact_mod_cast hc
  53     exact hcNat.trans ((edgeColourGraph G σ (Fin.last q)).degree_le_card_edgeFinset v)
  54   have hpath (a : Fin q) (u v : V) (huv : u≠v) : ρ*Fintype.card V≤
  55       (colouredTwoPathCount G σ u v a.castSucc (Fin.last q):ℝ) := by
  56     simpa only [ρ,Nat.cast_add,Nat.cast_one] using hσpath a.castSucc u v huv
  57   obtain ⟨τ,hdiv,hchg⟩ := divisible_colour_repair hq hh hhq G σ ρ L hL hreserve hpath hlose hpack
  58   refine ⟨τ,hdiv,?_⟩
  59   intro a v
  60   have hc := (abs_le.mp (hchg a v)).2
  61   have hs := (abs_lt.mp (hσdeg a v)).2
  62   simp only [Nat.cast_add,Nat.cast_one] at hs
  63   have hpos := mul_nonneg hξ.le hnr.le
  64   linarith only [hc,hs,hchange,hpos]
