   9 theorem two_factor_component_ordered_cycle (F : SimpleGraph V)
  10     (hF : F.IsRegularOfDegree 2) (v : V) :
  11     ∃ p : F.Walk v v, p.IsCycle ∧ p.support.dropLast.Nodup ∧
  12       3 ≤ p.support.dropLast.length ∧
  13       p.support.dropLast ++ [v] = p.support ∧
  14       ∀ x, x ∈ p.support.dropLast ↔ x ∈ (F.connectedComponentMk v).supp := by
  15   obtain ⟨p,hp,hcomp⟩:=two_factor_component_cycle F hF v
  16   have hlen : p.support.dropLast.length = p.length := by
  17     simp [List.length_dropLast, p.length_support]
  18   have hlen3 : 3 ≤ p.support.dropLast.length := by
  19     rw [hlen]
  20     exact hp.isCircuit.three_le_length
  21   have hdrop : p.support.dropLast ≠ [] := by
  22     apply List.ne_nil_of_length_pos
  23     omega
  24   have hv : v ∈ p.support.dropLast := by
  25     have hh := List.head_mem hdrop
  26     simpa [List.head_dropLast, p.head_support] using hh
  27   have hclosed : p.support.dropLast ++ [v] = p.support := by
  28     simpa using List.dropLast_append_getLast p.support_ne_nil
  29   refine ⟨p,hp,hp.nodup_dropLast_support,hlen3,hclosed,?_⟩
  30   intro x
  31   have hmem : (x∈p.support.dropLast ∨ x=v) ↔ x∈p.support := by
  32     simpa only [List.mem_append,List.mem_singleton] using
  33       Iff.of_eq (congrArg (fun xs : List V => x∈xs) hclosed)
  34   rw [←hcomp, p.mem_verts_toSubgraph, ←hmem]
  35   constructor
  36   · exact Or.inl
  37   · rintro (hx|rfl)
  38     · exact hx
  39     · exact hv
  40 end Oberwolfach
