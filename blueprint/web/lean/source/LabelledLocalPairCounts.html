  17 theorem labelled_absorber_local_free_pair_count (R : C → Finset V)
  18     (S : T → VertexAbsorberTemplate c) (X Y : Finset V) (J : ℕ)
  19     (hJ : ∀ U : Finset V,Fintype.card {C₀ : C // R C₀=U} ≤ J)
  20     (x y : V) (hxy : x≠y) :
  21     Fintype.card {D : LabelledAbsorberFamily R S X //
  22       labelledAbsorberFree R (S D.1) X D.2⊆Y ∧
  23       x∈labelledAbsorberFree R (S D.1) X D.2 ∧ y∈labelledAbsorberFree R (S D.1) X D.2} ≤
  24       (c*c)^2*Fintype.card T*X.card^c*Y.card^(c*c-2)*J^(2*c+1) := by
  25   let F := LabelledAbsorberFamily R S X
  26   let Q (p : Fin (c*c) × Fin (c*c)) (D : F) :=
  27     (∀ i : Fin (c*c),D.2.1.val (Sum.inl i)∈Y) ∧
  28       D.2.1.val (Sum.inl p.1)=x ∧ D.2.1.val (Sum.inl p.2)=y
  29   have hc (p : Fin (c*c) × Fin (c*c)) : Fintype.card {D : F // Q p D} ≤
  30       Fintype.card T*(X.card^c*Y.card^(c*c-2)*J^(2*c+1)) := by
  31     by_cases hp : p.1=p.2
  32     · have hn (D : F) : ¬Q p D := by
  33         rintro ⟨_,hx,hy⟩
  34         exact hxy (hx.symm.trans (hp ▸ hy))
  35       simp only [Fintype.card_subtype,hn,Finset.filter_false,Finset.card_empty,Nat.zero_le]
  36     · have he : Fintype.card {D : F // Q p D}=
  37           ∑ t,Fintype.card {D : LabelledCycleRecord R (S t).val X //
  38             (∀ i : Fin (c*c),D.1.val (Sum.inl i)∈Y) ∧
  39               D.1.val (Sum.inl p.1)=x ∧ D.1.val (Sum.inl p.2)=y} := by
  40         convert finite_sigma_predicate_card (fun t (D : LabelledCycleRecord R (S t).val X) =>
  41           (∀ i : Fin (c*c),D.1.val (Sum.inl i)∈Y) ∧
  42             D.1.val (Sum.inl p.1)=x ∧ D.1.val (Sum.inl p.2)=y) using 1 <;> congr!
  43       rw [he]
  44       calc
  45         _ ≤ ∑ _t : T,X.card^c*Y.card^(c*c-2)*J^(2*c+1) := by
  46           apply Finset.sum_le_sum
  47           intro t _
  48           have hh := labelled_cycle_records_local_two_coordinate_count R (S t).val X Y J hJ p.1 p.2 hp x y
  49           simpa only [Fintype.card_sum,Fintype.card_fin,show c+(c+1)=2*c+1 by omega] using hh
  50         _ = _ := by simp
  51   have hp (D : F) :
  52       (labelledAbsorberFree R (S D.1) X D.2⊆Y ∧
  53         x∈labelledAbsorberFree R (S D.1) X D.2 ∧ y∈labelledAbsorberFree R (S D.1) X D.2) ↔ ∃ p,Q p D := by
  54     simp only [labelledAbsorberFree,Finset.image_subset_iff,Finset.mem_image,Finset.mem_univ,true_and]
  55     constructor
  56     · rintro ⟨h,⟨j,hj⟩,⟨k,hk⟩⟩
  57       exact ⟨⟨j,k⟩,fun i => h i trivial,hj,hk⟩
  58     · rintro ⟨⟨j,k⟩,h,hj,hk⟩
  59       exact ⟨fun i _ => h i,⟨j,hj⟩,⟨k,hk⟩⟩
  60   let A := Finset.univ.filter (fun D : F => labelledAbsorberFree R (S D.1) X D.2⊆Y ∧
  61     x∈labelledAbsorberFree R (S D.1) X D.2 ∧ y∈labelledAbsorberFree R (S D.1) X D.2)
  62   let B := fun p : Fin (c*c) × Fin (c*c) => Finset.univ.filter (Q p)
  63   have hsub : A⊆Finset.univ.biUnion B := by
  64     intro D hD
  65     obtain ⟨p,hpD⟩ := (hp D).mp (Finset.mem_filter.mp hD).2
  66     exact Finset.mem_biUnion.mpr ⟨p,Finset.mem_univ _,Finset.mem_filter.mpr ⟨Finset.mem_univ _,hpD⟩⟩
  67   simp only [Fintype.card_subtype]
  68   change A.card ≤ _
  69   calc
  70     _ ≤ (Finset.univ.biUnion B).card := Finset.card_le_card hsub
  71     _ ≤ ∑ p : Fin (c*c) × Fin (c*c),(B p).card := Finset.card_biUnion_le
  72     _ ≤ ∑ _p : Fin (c*c) × Fin (c*c),Fintype.card T*(X.card^c*Y.card^(c*c-2)*J^(2*c+1)) := by
  73       apply Finset.sum_le_sum
  74       intro p _
  75       simpa only [Fintype.card_subtype,B] using hc p
  76     _ = _ := by
  77       simp only [Finset.sum_const,Finset.card_univ,Fintype.card_prod,Fintype.card_fin,smul_eq_mul]
  78       ring
