  18 def ordinaryWheelToLarger (hQ : Q≤P) (hH : ∀ v u,H v u → Hbase v u)
  19     (W : OrdinaryWheel Q F H r) : OrdinaryWheel P F Hbase r :=
  20   ⟨(⟨W.val.1.val,W.val.1.property.trans hQ⟩,W.val.2),
  21     fun v hv=>hH v W.val.2 (W.property.1 v hv),W.property.2⟩

  34 def OrdinaryWheelSurvives (W : OrdinaryWheel P F Hbase r)
  35     (Q : SimpleGraph V) (H : V → U → Prop) : Prop :=
  36   ordinaryWheelRim W≤Q ∧ ∀ v∈(ordinaryWheelRim W).support,H v (ordinaryWheelHub W)

  38 def ordinaryWheelRestrictionEquiv (hQ : Q≤P) (hH : ∀ v u,H v u → Hbase v u) :
  39     {W : OrdinaryWheel P F Hbase r // OrdinaryWheelSurvives W Q H} ≃ OrdinaryWheel Q F H r where
  40   toFun W := ⟨(⟨W.val.val.1.val,W.property.1⟩,W.val.val.2),W.property.2,W.val.property.2⟩
  41   invFun W := ⟨ordinaryWheelToLarger hQ hH W,W.val.1.property,W.property.1⟩
  42   left_inv W := by rfl
  43   right_inv W := by rfl

  45 theorem ordinaryWheelSurvives_iff_resources (W : OrdinaryWheel P F Hbase r) :
  46     OrdinaryWheelSurvives W Q H ↔ ∀ e∈ordinaryWheelResources W,ordinaryResourceState Q H e=true := by
  47   constructor
  48   · intro h e he
  49     cases e with
  50     | inl e =>
  51       rw [ordinaryWheelResources_rim] at he
  52       have hh := SimpleGraph.edgeSet_mono h.1 he
  53       simpa only [ordinaryResourceState,Sum.elim_inl,decide_eq_true_eq] using hh
  54     | inr vu =>
  55       rw [ordinaryWheelResources_spoke] at he
  56       have hh := h.2 vu.1 he.2
  57       rw [he.1] at hh
  58       simpa only [ordinaryResourceState,Sum.elim_inr,decide_eq_true_eq] using hh
  59   · intro h
  60     constructor
  61     · intro x y hxy
  62       have he : Sum.inl s(x,y)∈ordinaryWheelResources W :=
  63         (ordinaryWheelResources_rim W s(x,y)).mpr hxy
  64       have hh := h _ he
  65       simpa only [ordinaryResourceState,Sum.elim_inl,decide_eq_true_eq,SimpleGraph.mem_edgeSet] using hh
  66     · intro v hv
  67       have he : Sum.inr (v,ordinaryWheelHub W)∈ordinaryWheelResources W :=
  68         (ordinaryWheelResources_spoke W v _).mpr ⟨rfl,hv⟩
  69       have hh := h _ he
  70       simpa only [ordinaryResourceState,Sum.elim_inr,decide_eq_true_eq] using hh

  72 theorem ordinary_wheel_restricted_resource_load (hQ : Q≤P) (hH : ∀ v u,H v u → Hbase v u)
  73     (w : OrdinaryWheel P F Hbase r → ℝ) (e : Sym2 V⊕(V×U)) :
  74     (∑ W : OrdinaryWheel Q F H r,if e∈ordinaryWheelResources W then w (ordinaryWheelToLarger hQ hH W) else 0)=
  75       ∑ W : OrdinaryWheel P F Hbase r,if e∈ordinaryWheelResources W ∧ OrdinaryWheelSurvives W Q H then w W else 0 := by
  76   let f (W : OrdinaryWheel P F Hbase r) := if e∈ordinaryWheelResources W ∧ OrdinaryWheelSurvives W Q H then w W else 0
  77   have he := finite_sum_restrict_of_zero (fun W=>OrdinaryWheelSurvives W Q H) f
  78     (by intro W hW;simp only [f,hW,and_false,ite_false])
  79   rw [he]
  80   have hh := (ordinaryWheelRestrictionEquiv (F:=F) (r:=r) hQ hH).sum_comp
  81     (fun W : OrdinaryWheel Q F H r=>if e∈ordinaryWheelResources W then w (ordinaryWheelToLarger hQ hH W) else 0)
  82   rw [← hh]
  83   apply Finset.sum_congr rfl
  84   intro W _
  85   have hw : ordinaryWheelToLarger hQ hH ((ordinaryWheelRestrictionEquiv hQ hH) W)=W.val := rfl
  86   have hr : ordinaryWheelResources ((ordinaryWheelRestrictionEquiv hQ hH) W)=ordinaryWheelResources W.val := rfl
  87   rw [hw,hr]
  88   dsimp only [f]
  89   simp only [W.property,and_true]

  92 /-- At a present root, the potential load is exactly the physical load of
  93 the restricted weighting. The coefficients are composed with the fixed inclusion. -/
  94 theorem ordinaryPotentialLoad_restriction (hQ : Q≤P) (hH : ∀ v u,H v u → Hbase v u)
  95     (w : OrdinaryWheel P F Hbase r → ℝ) (e : Sym2 V⊕(V×U))
  96     (he : ordinaryResourceState Q H e=true) :
  97     ordinaryPotentialLoad w e Q H=
  98       ∑ W : OrdinaryWheel Q F H r,if e∈ordinaryWheelResources W then w (ordinaryWheelToLarger hQ hH W) else 0 := by
  99   rw [ordinaryPotentialLoad_present w e Q H he]
 100   simp_rw [← ordinaryWheelSurvives_iff_resources]
 101   exact (ordinary_wheel_restricted_resource_load hQ hH w e).symm

 105     ordinaryPotentialLoad w e P Hbase=∑ W,if e∈ordinaryWheelResources W then w W else 0 := by

 167 /-- The original seed coefficients are restricted to surviving physical
 168 wheels; their rim loads change by the degree-controlled potential estimate. -/
 169 theorem ordinary_restricted_rim_load_stability
 170     (Q F D : SimpleGraph V) (H E : V → U → Prop) (r : ℕ) (hr : 2≤r)
 171     (hn : 0<Fintype.card V) (hu : Fintype.card U≤Fintype.card V)
 172     (w : OrdinaryWheel Q F H r → ℝ) (B d : ℝ) (hB : 0≤B) (hd : 0≤d)
 173     (hw : ∀ W,0≤w W ∧ w W≤B/(Fintype.card V : ℝ)^(r-1))
 174     (hD : ∀ v,(D.degree v : ℝ)≤d)
 175     (hEout : ∀ v,(∑ u,bipAdj E v u)≤d) (hEin : ∀ u,(∑ v,bipAdj E v u)≤d)
 176     (x y : V) (hxy : (Q\D).Adj x y) :
 177     |ordinaryRimLoad (fun W=>w (ordinaryWheelToLarger (sdiff_le : Q\D≤Q)
 178        (fun v u (h : H v u ∧ ¬E v u)=>h.1) W)) x y-ordinaryRimLoad w x y|≤
 179       8*(r : ℝ)^4*B*d/Fintype.card V := by
 180   have hh := ordinaryPotentialLoad_rim_degree_stability Q F H r hr hn hu w B hB hw
 181     Q (Q\D) H (fun v u=>H v u ∧ ¬E v u) d hd
 182     (graph_deletion_change_degree Q D d hD)
 183     (relation_deletion_change_row H E d hEout)
 184     (relation_deletion_change_column H E d hEin) x y hxy.ne
 185   rw [ordinaryPotentialLoad_restriction_rim (sdiff_le : Q\D≤Q) (fun v u (h : H v u ∧ ¬E v u)=>h.1) w x y hxy,
 186     ordinaryPotentialLoad_full,ordinaryWheelResources_load_rim] at hh
 187   exact hh
