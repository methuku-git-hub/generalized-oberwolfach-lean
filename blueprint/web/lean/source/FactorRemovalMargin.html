   8 theorem factor_small_removal_margin (D h n:ℕ) (δ α:ℝ)
   9     (hδ:0≤δ) (hα:0<α)
  10     (hdegree:(δ+α)*(n:ℝ)≤D) (hsmall:(h:ℝ)≤α*n/500):
  11     ∀ j≤h,2*j≤D ∧ (δ+α/2)*(n:ℝ)≤((D-2*j:ℕ):ℝ):=by
  12   intro j hj
  13   have hn:(0:ℝ)≤n:=by positivity
  14   have hjR:(j:ℝ)≤h:=by exact_mod_cast hj
  15   have hαn:0≤α*n:=mul_nonneg hα.le hn
  16   have hδn:0≤δ*n:=mul_nonneg hδ hn
  17   have hjbound:(2:ℝ)*j≤D:=by nlinarith
  18   have hjnat:2*j≤D:=by exact_mod_cast hjbound
  19   constructor
  20   · exact hjnat
