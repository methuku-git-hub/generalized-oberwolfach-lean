  10     {V U:Type} [Fintype V] [Fintype U] {G:SimpleGraph V}
  11     {D m:ℕ} {β ε:ℝ} (Q:Vortex U G D β ε m) (he:0<ε)
  12     {i:ℕ} (hi:1 ≤ i) (hL:i≤Q.length) (hpos:1≤(Q.extendedVertices i).card) :
  13     ((Q.extendedVertices (i-1)).card:ℝ)≤(2/ε)*(Q.extendedVertices i).card := by
  14   have hprev:i-1≤Q.length:=by omega
  15   have hsuc:i-1+1=i:=by omega
  16   have hf:=Nat.lt_floor_add_one (ε * (floorOrbit ε (Fintype.card V) (i-1):ℝ))
  17   rw [← floorOrbit_succ, hsuc, ← Q.sizes (i-1) hprev, ← Q.sizes i hL] at hf
  18   rw [Q.extendedVertices_eq hprev, Q.extendedVertices_eq hL] at *
  19   have hp:(1:ℝ)≤(Q.vertices i).card:=by exact_mod_cast hpos
  20   rw [div_mul_eq_mul_div]
  21   apply (le_div_iff₀ he).2
  22   nlinarith
