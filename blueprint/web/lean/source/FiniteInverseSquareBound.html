  10 theorem inverse_square_range_bound : ∀ n:ℕ, 1≤n →
  11     (∑ k∈Finset.range n, (1:ℝ)/((k:ℝ)+1)^2) ≤ 2-1/(n:ℝ) := by
  12   intro n hn
  13   induction n with
  14   | zero => omega
  15   | succ n ih =>
  16     by_cases h : n=0
  17     · subst n
  18       norm_num
  19     · have hnpos : (0:ℝ)<n := by exact_mod_cast (show 0<n by omega)
  20       have hp : (0:ℝ)<(n:ℝ)+1 := by positivity
  21       have hp2 : (0:ℝ)<((n:ℝ)+1)^2 := sq_pos_of_pos hp
  22       have hi:=ih (by omega)
  23       rw [Finset.sum_range_succ]
  24       push_cast
  25       have hstep : 1/((n:ℝ)+1)^2 ≤ 1/(n:ℝ)-1/((n:ℝ)+1) := by
  26         have hdiff : (1:ℝ)/(n:ℝ)-1/((n:ℝ)+1)=1/((n:ℝ)*((n:ℝ)+1)) := by
  27           field_simp
  28           <;> ring
  29         rw [hdiff]
  30         apply one_div_le_one_div_of_le (by positivity)
  31         nlinarith
  32       linarith
