  18 theorem freeAbsorberRecord_ext (R : C → Finset V) (S : VertexAbsorberTemplate c)
  19     (F G : FreeAbsorberRecord R S) (hf : ∀ j,F.1 j=G.1 j)
  20     (hi : ∀ i,(F.2 i).val=(G.2 i).val) : F=G := by
  21   rcases F with ⟨f,d⟩
  22   rcases G with ⟨g,e⟩
  23   have he : f=g := DFunLike.ext _ _ hf
  24   subst g
  25   have he' : d=e := funext (fun i => Subtype.ext (hi i))
  26   subst e;rfl

  28 theorem free_absorber_coordinate_count (R : C → Finset V) (S : VertexAbsorberTemplate c)
  29     (J : ℕ) (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J)
  30     (j : Fin (c*c)) (v : V) :
  31     Fintype.card {F : FreeAbsorberRecord R S // F.1 j=v} ≤ Fintype.card V^(c*c-1)*J^c := by
  32   obtain ⟨code,hcode⟩ := finite_fibre_codes R J hJ
  33   let A := {F : FreeAbsorberRecord R S // F.1 j=v}
  34   let B := {f : Fin (c*c) → V // f j=v}
  35   let enc (F : A) : B × (Fin c → Fin J) :=
  36     (⟨F.val.1,F.property⟩,fun i => code (F.val.2 i).val)
  37   have hinj : Function.Injective enc := by
  38     intro F G he
  39     have hf : ∀ k,F.val.1 k=G.val.1 k := fun k =>
  40       congrFun (congrArg (fun z : B × (Fin c → Fin J) => z.1.val) he) k
  41     apply Subtype.ext
  42     apply freeAbsorberRecord_ext R S F.val G.val hf
  43     intro i
  44     apply hcode
  45     · rw [(F.val.2 i).property,(G.val.2 i).property]
  46       exact congrArg (Finset.image · (freeLabelPositions (S.val (Sum.inl i)))) (funext hf)
  47     · exact congrFun (congrArg Prod.snd he) i
  48   have hh := Fintype.card_le_of_injective enc hinj
  49   have hb : Fintype.card B=Fintype.card V^(c*c-1) := by
  50     have hh := finite_function_fixed_value_card j v
  51     simp only [Fintype.card_fin] at hh
  52     convert hh using 1 <;> congr!
  53   simpa only [Fintype.card_prod,hb,Fintype.card_fun,Fintype.card_fin] using hh

  58 theorem free_absorber_family_vertex_count (R : C → Finset V) (S : T → VertexAbsorberTemplate c)
  59     (J : ℕ) (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J) (v : V) :
  60     Fintype.card {F : FreeAbsorberFamily R S // v∈freeFamilyVertices R S F} ≤
  61       (c*c)*Fintype.card T*Fintype.card V^(c*c-1)*J^c := by
  62   have hc (j : Fin (c*c)) : Fintype.card {F : FreeAbsorberFamily R S // F.2.1 j=v} ≤
  63       Fintype.card T*(Fintype.card V^(c*c-1)*J^c) := by
  64     have he : Fintype.card {F : FreeAbsorberFamily R S // F.2.1 j=v}=
  65         ∑ t,Fintype.card {F : FreeAbsorberRecord R (S t) // F.1 j=v} := by
  66       convert finite_sigma_predicate_card (fun t (F : FreeAbsorberRecord R (S t)) => F.1 j=v) using 1 <;> congr!
  67     rw [he]
  68     calc
  69       _ ≤ ∑ _t : T,Fintype.card V^(c*c-1)*J^c :=
  70         Finset.sum_le_sum (fun t _ => free_absorber_coordinate_count R (S t) J hJ j v)
  71       _ = _ := by simp
  72   have he : (fun F : FreeAbsorberFamily R S => v∈freeFamilyVertices R S F)=
  73       (fun F => ∃ j : Fin (c*c),F.2.1 j=v) := by
  74     funext F;apply propext;simp [freeFamilyVertices,freeAbsorberVertices]
  75   have hcard : Fintype.card {F : FreeAbsorberFamily R S // v∈freeFamilyVertices R S F}=
  76       Fintype.card {F : FreeAbsorberFamily R S // ∃ j : Fin (c*c),F.2.1 j=v} := by
  77     apply Fintype.card_congr
  78     exact Equiv.subtypeEquivRight (fun F => iff_of_eq (congrFun he F))
  79   rw [hcard]
  80   calc
  81     _ ≤ ∑ j : Fin (c*c),Fintype.card {F : FreeAbsorberFamily R S // F.2.1 j=v} := by
  82       convert finite_exists_card_le_sum (fun j (F : FreeAbsorberFamily R S) => F.2.1 j=v) using 1 <;> congr!
  83     _ ≤ ∑ _j : Fin (c*c),Fintype.card T*(Fintype.card V^(c*c-1)*J^c) := Finset.sum_le_sum (fun j _ => hc j)
  84     _ = _ := by simp only [Finset.sum_const,Finset.card_univ,Fintype.card_fin,smul_eq_mul];ring
