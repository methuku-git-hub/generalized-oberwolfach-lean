  13     (hclass:ρclass i/2*D=Fintype.card U) (hρlo:ρ₀ ≤ initialIndexDensity U D):
  14     let ρ:=initialIndexDensity U D
  15     let p:=ρ*(1/100)/2
  16     let k:=ρ*s/2
  17     let p₂:=(∑w,b w)/(D:ℝ)
