  13 theorem prescribed_position_greedy_singles (Q:SimpleGraph X) (hdeg:∀ x,Q.degree x≤2):
  14     ∀ z:ℕ,∀ S:Finset X,S.card+3*z≤Fintype.card X→
  15       ∃ vs:List X,vs.length=z ∧ (∀ v∈vs,v∉S) ∧ vs.Pairwise (prescribedSinglesApart Q) := by
  16   intro z
  17   induction z with
  18   | zero => intro S h;exact ⟨[],rfl,by simp,by simp⟩
  19   | succ z ih =>
  20     intro S h
  21     have hlt:S.card<Fintype.card X:=by omega
  22     have hnot:¬(Finset.univ:Finset X)⊆S:=by
  23       intro hs
  24       have hh:=Finset.card_le_card hs
  25       simp only [Finset.card_univ] at hh
  26       omega
  27     obtain ⟨x,_,hx⟩:=Finset.not_subset.mp hnot
  28     let T:=S∪prescribedClosedNeighbourhood Q x
  29     have hT:T.card≤S.card+3:=
  30       (Finset.card_union_le _ _).trans (Nat.add_le_add_left (prescribed_closed_neighbourhood_card Q hdeg x) _)
  31     have hbudget:T.card+3*z≤Fintype.card X:=by omega
  32     obtain ⟨rest,hlen,hrest,hpair⟩:=ih T hbudget
  33     refine ⟨x::rest,by simp [hlen],?_,?_⟩
  34     · intro y hy
  35       rcases List.mem_cons.mp hy with hy|hy
  36       · subst y;exact hx
  37       · intro hyS;exact hrest y hy (Finset.mem_union_left _ hyS)
  38     · apply List.pairwise_cons.mpr
  39       refine ⟨?_,hpair⟩
  40       intro y hy hclose
  41       exact hrest y hy (Finset.mem_union_right _ hclose)
