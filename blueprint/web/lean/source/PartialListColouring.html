  14 theorem extend_partial_list_colouring
  15     {V C : Type*} (G : SimpleGraph V) (L : V → Finset C)
  16     (p : V → Option C)
  17     (hlist : ∀ v c,p v=some c → c∈L v)
  18     (hproper : ∀ u v,G.Adj u v → ∀ c,p u=some c → p v≠some c)
  19     (L' : {v // p v=none} → Finset C)
  20     (hsub : ∀ v,L' v⊆L v.val)
  21     (havoid : ∀ v c,c∈L' v → ∀ u,G.Adj v.val u → p u≠some c)
  22     (f : {v // p v=none} → C) (hf : ∀ v,f v∈L' v)
  23     (hfproper : ∀ u v,G.Adj u.val v.val → f u≠f v) :
  24     ∃ g : V → C,(∀ v,g v∈L v) ∧ ∀ u v,G.Adj u v → g u≠g v := by
  25   classical
  26   let g (v : V) : C := if h : p v=none then f ⟨v,h⟩ else (p v).get (Option.isSome_iff_ne_none.mpr h)
  27   have hnone (v : V) (h : p v=none) : g v=f ⟨v,h⟩ := by simp [g,h]
  28   have hsome (v : V) (c : C) (h : p v=some c) : g v=c := by simp [g,h]
  29   refine ⟨g,?_,?_⟩
  30   · intro v
  31     cases hv : p v with
  32     | none => rw [hnone v hv];exact hsub ⟨v,hv⟩ (hf ⟨v,hv⟩)
  33     | some c => rw [hsome v c hv];exact hlist v c hv
  34   · intro u v huv he
  35     cases hu : p u with
  36     | none =>
  37       cases hv : p v with
  38       | none =>
  39         rw [hnone u hu,hnone v hv] at he
  40         exact hfproper ⟨u,hu⟩ ⟨v,hv⟩ huv he
  41       | some c =>
  42         rw [hnone u hu,hsome v c hv] at he
  43         exact havoid ⟨u,hu⟩ c (he ▸ hf ⟨u,hu⟩) v huv hv
  44     | some c =>
  45       cases hv : p v with
  46       | none =>
  47         rw [hsome u c hu,hnone v hv] at he
  48         exact havoid ⟨v,hv⟩ c (he.symm ▸ hf ⟨v,hv⟩) u huv.symm hu
  49       | some d =>
  50         rw [hsome u c hu,hsome v d hv] at he
  51         exact hproper u v huv c hu (hv.trans (congrArg some he.symm))

  53 theorem hypergraph_available_list_avoids_retained
  54     {V E C : Type*} [Fintype E]
  55     (R : E → Finset V) (L : E → Finset C) (σ : E → C×Bool)
  56     (e : E) (c : C)
  57     (hc : c∈nibbleAvailableColours (hypergraphLineGraph R) (R e)
  58       (hypergraphVertexClique R) (L e) σ)
  59     (f : E) (hef : (hypergraphLineGraph R).Adj e f) :
  60     coinRetainedColour (hypergraphLineGraph R) σ f≠some c := by
  61   have hh := (nibbleAvailableColours_mem_iff (hypergraphLineGraph R) (R e)
  62     (hypergraphVertexClique R) (fun v _=>hypergraphVertexClique_isClique R v) (L e) σ c).mp hc
  63   have ha := (clique_no_retained_iff_card (hypergraphLineGraph R) (R e)
  64     (hypergraphVertexClique R) (fun v _=>hypergraphVertexClique_isClique R v) σ c).mpr hh.2
  65   obtain ⟨v,hve,hvf⟩ := Finset.not_disjoint_iff.mp hef.2
  66   exact ha v hve f (Finset.mem_filter.mpr ⟨Finset.mem_univ _,hvf⟩)

  68 theorem hypergraph_extend_retained_colouring
  69     {V E C : Type*} [Fintype E]
  70     (R : E → Finset V) (L : E → Finset C) (σ : E → C×Bool)
  71     (hσ : ∀ e,(σ e).1∈L e)
  72     (f : {e // coinRetainedColour (hypergraphLineGraph R) σ e=none} → C)
  73     (hf : ∀ e,f e∈nibbleAvailableColours (hypergraphLineGraph R) (R e.val)
  74       (hypergraphVertexClique R) (L e.val) σ)
  75     (hfproper : ∀ e g,(hypergraphLineGraph R).Adj e.val g.val → f e≠f g) :
  76     ∃ g : E → C,(∀ e,g e∈L e) ∧
  77       ∀ e f,(hypergraphLineGraph R).Adj e f → g e≠g f := by
  78   apply extend_partial_list_colouring (hypergraphLineGraph R) L
  79     (coinRetainedColour (hypergraphLineGraph R) σ)
  80     (fun e c hc=>by rw [← ((coinRetainedColour_eq_some _ _ _ _).mp hc).1];exact hσ e)
  81     (fun e f hef c hc=>coinRetainedColour_proper _ _ hef c hc)
  82     (fun e=>nibbleAvailableColours (hypergraphLineGraph R) (R e.val)
  83       (hypergraphVertexClique R) (L e.val) σ)
  84     (fun e c hc=>by
  85       have hm : c∈L e.val ∧ ∀ j∈R e.val,∀ v∈hypergraphVertexClique R j,
