  10 def bufferedPathFullDomain {X I:Type*} [Fintype I] [DecidableEq X]
  11     (n:I→ℕ) (inner:∀i,Fin (n i+1)↪X) : Finset X :=
  12   Finset.univ.biUnion (fun i=>Finset.univ.image (inner i))

  14 def bufferedPathStrictInterior {X I:Type*} [Fintype I] [DecidableEq X]

  18 theorem buffered_strict_interior_subset_domain
  19     {X I:Type*} [Fintype I] [DecidableEq X]
  20     (n:I→ℕ) (inner:∀i,Fin (n i+1)↪X) :
  21     bufferedPathStrictInterior n inner ⊆ bufferedPathFullDomain n inner := by
  22   classical
  23   intro x hx
  24   rcases Finset.mem_biUnion.mp hx with ⟨i,_,hxi⟩
