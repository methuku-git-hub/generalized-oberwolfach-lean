  10 theorem factor_orientation_no_fixed_points {X:Type*} [Fintype X] [DecidableEq X]
  11     (G:SimpleGraph X) (hG:G.IsRegularOfDegree 2)
  12     (e:∀C:G.ConnectedComponent,SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph)
  13     (x:X):factorCycleOrientation G e x≠x:=by
  14   intro h
  15   have ha:=factor_cycle_orientation_adj G hG e x
  16   rw [h] at ha

  19 theorem factor_orientation_no_two_cycles {X:Type*} [Fintype X] [DecidableEq X]
  20     (G:SimpleGraph X) (hG:G.IsRegularOfDegree 2)
  21     (e:∀C:G.ConnectedComponent,SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph)
  22     (x:X):factorCycleOrientation G e (factorCycleOrientation G e x)≠x:=by
  23   classical
  24   intro htwo
  25   let σ:=factorCycleOrientation G e
  26   have hs:G.neighborFinset x⊆{σ x}:=by
  27     intro y hy
  28     have hxy:G.Adj x y:=(SimpleGraph.mem_neighborFinset G x y).mp hy
  29     rcases (factor_cycle_orientation_edge_iff G hG e x y).mp hxy with h | h
  30     · exact Finset.mem_singleton.mpr h.symm
  31     · have hyx:y=σ x:=σ.injective (h.trans htwo.symm)
  32       exact Finset.mem_singleton.mpr hyx
  33   have hc:=Finset.card_le_card hs
  34   rw [SimpleGraph.card_neighborFinset_eq_degree,hG x,Finset.card_singleton] at hc
  35   omega
  36 end Oberwolfach
