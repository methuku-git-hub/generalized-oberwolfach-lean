  13     (f : E → V) : (E → ℝ) →ₗ[ℝ] (V → ℝ) where

  37     (hE : 0≤E) (hd : |d-q|≤2*E) (hO : |O-(k+1)*q|≤E) :
  38     |d-O/(k+1:ℝ)|≤3*E := by
