  14  theorem marked_attachment_pairings {X Y S:Type*} [Fintype X] [Fintype Y] [Fintype S]
  15     (K:Finset X) (SK:Finset S)
  16     (hXY:Fintype.card X=Fintype.card Y) (hXS:Fintype.card X=Fintype.card S)
  17     (hK:K.card=SK.card) (hs:0<Fintype.card X)
  18     (hpad:¬(Fintype.card X=1∧K.card=1)):
  19     ∃ σ:Equiv.Perm (X⊕Y),∃ a:X≃S,∃ b:Y≃S,
  20       (∀ x,σ x≠x)∧
  21       (∀ x∈K,∃ y:X,σ (Sum.inl x)=Sum.inl y)∧
  22       (∀ x:X,a x∈SK↔σ.symm (Sum.inl x)∈K.image (Sum.inl (β:=Y)))∧
  23       ∀ y:Y,σ.symm (Sum.inr y)∉K.image (Sum.inl (β:=Y)):=by
  24   classical
  25   let J:Finset (X⊕Y):=K.image Sum.inl
  26   let A:Finset (X⊕Y):=Finset.univ.image (Sum.inl (β:=Y))
  27   have hJ:J.card=K.card:=Finset.card_image_of_injective _ Sum.inl_injective
  28   have hA:A.card=Fintype.card X:=by
  29     dsimp only [A]
  30     rw [Finset.card_image_of_injective _ Sum.inl_injective,Finset.card_univ]
  31   have hm:=marked_transformer_assignment_margins (Fintype.card X) K.card (Finset.card_le_univ K) hs hpad
  32   obtain ⟨σ,hσ,hmark⟩:=marked_transformer_attachment_assignment (Equiv.refl (X⊕Y)) J A
  33     (by rw [hJ,hA];exact Finset.card_le_univ K)
  34     (by rw [hJ,hA];exact hm.1)
  35     (by rw [Fintype.card_sum,← hXY,hJ];omega)
  36   have hx:∀ x:K,∃ y:X,σ (Sum.inl x.val)=Sum.inl y:=by
  37     intro x
  38     have hh:=hmark (Sum.inl x.val) (Finset.mem_image.mpr ⟨x.val,x.property,rfl⟩)
  39     obtain ⟨y,_,hy⟩:=Finset.mem_image.mp hh
  40     exact ⟨y,hy.symm⟩
  41   choose g hg using hx
  42   have hgi:Function.Injective g:=by
  43     intro x y he
  44     apply Subtype.ext
  45     exact Sum.inl.inj (σ.injective ((hg x).trans ((congrArg Sum.inl he).trans (hg y).symm)))
  46   let T:Finset X:=Finset.univ.image g
  47   have hT:T.card=K.card:=by
  48     dsimp only [T]
  49     rw [Finset.card_image_of_injective _ hgi,Finset.card_univ,Fintype.card_coe]
  50   obtain ⟨a,ha⟩:=finite_bijection_preserving_predicate (fun x:X=>x∈T) (fun s:S=>s∈SK) hXS
  51     (by simpa only [Fintype.card_coe] using hT.trans hK)
  52   let b:Y≃S:=Fintype.equivOfCardEq (hXY.symm.trans hXS)
  53   have ht (x:X):x∈T↔σ.symm (Sum.inl x)∈J:=by
  54     constructor
  55     · intro hh
  56       obtain ⟨z,_,hz⟩:=Finset.mem_image.mp hh
  57       have he:σ (Sum.inl z.val)=Sum.inl x:=(hg z).trans (congrArg Sum.inl hz)
  58       have he':σ.symm (Sum.inl x)=Sum.inl z.val:=by rw [← he,Equiv.symm_apply_apply]
  59       rw [he']
  60       exact Finset.mem_image.mpr ⟨z.val,z.property,rfl⟩
  61     · intro hh
  62       obtain ⟨z,hz,he⟩:=Finset.mem_image.mp hh
  63       refine Finset.mem_image.mpr ⟨⟨z,hz⟩,Finset.mem_univ _,?_⟩
  64       apply Sum.inl_injective (β:=Y)
  65       rw [← hg]
  66       have he':=congrArg σ he
  67       simpa only [Equiv.apply_symm_apply] using he'
  68   refine ⟨σ,a,b,hσ,?_,fun x=>(ha x).trans (ht x),?_⟩
  69   · intro x hx
  70     exact ⟨g ⟨x,hx⟩,hg ⟨x,hx⟩⟩
  71   · intro y hy
  72     have hh:=hmark (σ.symm (Sum.inr y)) hy
  73     rw [Equiv.apply_symm_apply] at hh
  74     obtain ⟨x,_,he⟩:=Finset.mem_image.mp hh
  75     exact Sum.inl_ne_inr he
