  12 theorem rooted_two_dense_graphs_exist (s:ℕ) (α:ℝ) (hα:0<α):
  13     ∃ K:ℕ,∀ k:ℕ,K≤k→∃ N:ℕ,
  14     ∀ (V:Type) [Fintype V] [DecidableEq V],N≤Fintype.card V→
  15     ∀ (P₁ P₂:SimpleGraph V) (T:Finset V),T.card=s→∀ δ₁ δ₂:ℝ,
  16     (∀ v,(δ₁+α)*Fintype.card V≤(P₁.degree v:ℝ))→
  17     (∀ v,(δ₂+α)*Fintype.card V≤(P₂.degree v:ℝ))→
  18     ∃ R:RootedSubset V T k,
  19       (∀ v∈R.val.val,(δ₁+α/2)*(s+k:ℝ)<((P₁.neighborFinset v∩R.val.val).card:ℝ)) ∧
  20       (∀ v∈R.val.val,(δ₂+α/2)*(s+k:ℝ)<((P₂.neighborFinset v∩R.val.val).card:ℝ)):=by
  21   obtain ⟨K,hK⟩:=rooted_dense_set_scales s α hα
  22   refine ⟨K,?_⟩
  23   intro k hk
  24   obtain ⟨hk2,hroots,hfail,hN⟩:=hK k hk
  25   obtain ⟨N,hN⟩:=hN 0 (le_refl _)
  26   refine ⟨N,?_⟩
  27   intro V _ _ hn P₁ P₂ T hT δ₁ δ₂ hP₁ hP₂
  28   have hsize:T.card+k≤Fintype.card V:=by
  29     have hh:=(hN (Fintype.card V) hn).2.1
  30     omega
  31   letI:Nonempty (RootedSubset V T k):=rootedSubset_nonempty T k hsize
  32   let μ:=FiniteLaw.uniform (RootedSubset V T k)
  33   let E₁:RootedSubset V T k→Prop:=fun R=>∀ v∈R.val.val,
  34     (δ₁+α/2)*(s+k:ℝ)<((P₁.neighborFinset v∩R.val.val).card:ℝ)
  35   let E₂:RootedSubset V T k→Prop:=fun R=>∀ v∈R.val.val,
  36     (δ₂+α/2)*(s+k:ℝ)<((P₂.neighborFinset v∩R.val.val).card:ℝ)
  37   have hroots':(T.card:ℝ)+1≤α*(T.card+k:ℝ)/4:=by simpa [hT] using hroots
  38   have hp₁:=rootedSubset_dense_conflict_free_probability P₁ ⊥ T k hsize hk2 δ₁ α 0 hα hP₁
  39     (by intro v;simp) (by intro x hx y hy;simp) hroots'
  40   have hp₂:=rootedSubset_dense_conflict_free_probability P₂ ⊥ T k hsize hk2 δ₂ α 0 hα hP₂
  41     (by intro v;simp) (by intro x hx y hy;simp) hroots'
  42   have hprob₁:(3/4:ℝ)≤μ.probability E₁:=by
  43     have hh:(1-2*(s+k:ℝ)*Real.exp (-α^2*(s+k:ℝ)/8))≤μ.probability E₁:=by
  44       simpa only [zero_mul,sub_zero,hT,SimpleGraph.bot_adj,not_false_eq_true,implies_true,and_true,μ,E₁] using hp₁
  45     linarith
  46   have hprob₂:(1-(1/4:ℝ))≤μ.probability E₂:=by
  47     have hh:(1-2*(s+k:ℝ)*Real.exp (-α^2*(s+k:ℝ)/8))≤μ.probability E₂:=by
  48       simpa only [zero_mul,sub_zero,hT,SimpleGraph.bot_adj,not_false_eq_true,implies_true,and_true,μ,E₂] using hp₂
  49     linarith
  50   have hi:=finiteLaw_good_intersection μ E₁ E₂ (1/4) hprob₂
  51   obtain ⟨R,_,hR⟩:=finiteLaw_positive_event_exists μ (fun R=>E₁ R∧E₂ R) (by linarith)
  52   exact ⟨R,hR.1,hR.2⟩
  53 end Oberwolfach
