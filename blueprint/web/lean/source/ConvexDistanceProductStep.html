  21 theorem finiteLaw_convexDistance_product_step
  22     (μ : Option I → FiniteLaw Ω)
  23     (ih : ∀ B : Finset (I → Ω),
  24       0<(FiniteLaw.pi (fun i=>μ (some i))).probability (fun x=>x∈B) →
  25       (FiniteLaw.pi (fun i=>μ (some i))).expectation (fun x=>Real.exp (convexDistanceSq B x/4))≤
  26         ((FiniteLaw.pi (fun i=>μ (some i))).probability (fun x=>x∈B))⁻¹)
  27     (A : Finset (Option I → Ω))
  28     (hA : 0<(FiniteLaw.pi μ).probability (fun x=>x∈A)) :
  29     (FiniteLaw.pi μ).expectation (fun x=>Real.exp (convexDistanceSq A x/4))≤
  30       ((FiniteLaw.pi μ).probability (fun x=>x∈A))⁻¹ := by
  31   classical
  32   let ν := FiniteLaw.pi (fun i=>μ (some i))
  33   let B := configurationProjection A
  34   let p := ν.probability (fun x=>x∈B)
  35   let q := (FiniteLaw.pi μ).probability (fun x=>x∈A)
  36   let f : Ω → ℝ := fun a=>ν.probability (fun x=>x∈configurationFibre A a)
  37   have hp : 0<p := hA.trans_le (finiteLaw_pi_projection_probability μ A)
  38   have hq : 0<q := hA
  39   have hAne := finiteLaw_positive_finset_nonempty (FiniteLaw.pi μ) A hA
  40   have hB := ih B hp
  41   have hf (a : Ω) : f a≤p := finiteLaw_probability_mono ν _ _
  42     (fun x hx=>configurationFibre_subset_projection A a hx)
  43   have hf0 (a : Ω) : 0≤f a := finiteLaw_probability_nonneg ν _
  44   have hfibre (a : Ω) :
  45       ν.expectation (fun x=>Real.exp (convexDistanceSq A (configurationCons a x)/4))≤
  46         (2-f a/p)/p := by
  47     by_cases ha : 0<f a
  48     · have hF := finiteLaw_positive_finset_nonempty ν (configurationFibre A a) ha
  49       apply finiteLaw_convexDistance_fibre_moment ν
  50         (fun x=>convexDistanceSq (configurationFibre A a) x/4)
  51         (fun x=>convexDistanceSq B x/4)
  52         (fun x=>convexDistanceSq A (configurationCons a x)/4)
  53         ha (hf a) (ih _ ha) hB
  54       intro θ hθ0 hθ1 x
  55       have hh := convexDistanceSq_fibre_mix A (configurationCons a x) hF hθ0 hθ1
  56       change convexDistanceSq A (configurationCons a x)≤
  57         θ*convexDistanceSq (configurationFibre A a) x+
  58           (1-θ)*convexDistanceSq B x+(1-θ)^2 at hh
  59       linarith
  60     · have hzero : f a=0 := le_antisymm (le_of_not_gt ha) (hf0 a)
  61       rw [hzero,zero_div,sub_zero]
  62       apply finiteLaw_convexDistance_empty_fibre_moment ν
  63         (fun x=>convexDistanceSq B x/4)
  64         (fun x=>convexDistanceSq A (configurationCons a x)/4) hp hB
  65       intro x
  66       have hh := convexDistanceSq_projection A hAne (configurationCons a x)
  67       change convexDistanceSq A (configurationCons a x)≤convexDistanceSq B x+1 at hh
  68       linarith
  69   have hfmean : (μ none).expectation f=q := (finiteLaw_pi_option_probability μ A).symm
  70   have hmean : (μ none).expectation (fun a=>(2-f a/p)/p)=(2-q/p)/p := by
  71     rw [← hfmean]
  72     unfold FiniteLaw.expectation
  73     have he (a : Ω) :
  74         (μ none).mass a*((2-f a/p)/p)=
  75           (μ none).mass a*(2/p)-((μ none).mass a*f a)/(p*p) := by ring
  76     simp_rw [he]
  77     rw [Finset.sum_sub_distrib,← Finset.sum_mul,← Finset.sum_div,(μ none).total,one_mul]
  78     ring
  79   have hfinal : (2-q/p)/p≤q⁻¹ := by
  80     rw [← one_div]
  81     apply (le_div_iff₀ hq).mpr
  82     have hp2 : 0<p*p := mul_pos hp hp
  83     have hs := sq_nonneg (p-q)
  84     have he : ((2-q/p)/p)*q=(2*p*q-q*q)/(p*p) := by field_simp <;> ring
  85     rw [he]
  86     apply (div_le_one hp2).mpr
  87     nlinarith
