  15 theorem ordinary_row_root_cover (lam:ℝ) (hlam:0<lam) :
  16     ∀ d:ℕ,∃ N:ℕ,∀ (V U:Type) [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U],
  17       ∀ (Q F:SimpleGraph V) (H:V→U→Prop) (r K:ℕ),4≤r → 0<K →
  18       ∀ (x:V) (A:Finset V),x∉A → Q.neighborFinset x⊆A →
  19       ∀ b:ℝ,
  20       let T:=Finset.univ.filter (H x)
  21       N≤T.card → Q.degree x=2*T.card →
  22       (∀ a,F.degree a≤d) →
  23       (∀ a,Q.Adj x a → ¬F.Adj x a) →
  24       (∀ a,Q.Adj x a → lam*T.card≤((T.filter (H a)).card:ℝ)) →
  25       (∀ u∈T,2*lam*T.card≤(((Q.neighborFinset x).filter (fun a=>H a u)).card:ℝ)) →
  26       (∀ S⊆Q.neighborFinset x,∀ J⊆T,lam/16*T.card≤(S.card:ℝ) → lam/16*T.card≤(J.card:ℝ) →
  27         lam*S.card*J.card≤(bipartiteMatrixCut (fun a u=>if H a u then 1 else 0) S J:ℝ)) →
  28       (∀ u∈T,∀ a,a≠x → ∀ c,c≠x →
  29         b≤(((A.filter (fun z=>H z u))∩Q.neighborFinset a∩Q.neighborFinset c).card:ℝ)) →
  30       (T.card*r:ℕ)/(K:ℝ)+((r:ℝ)+3)*(d+1)+4*(K+1)<b →
  31       ∃ C:Fin T.card→OrdinaryWheel Q F H r,
  32         (∀ i,∀ a∈(ordinaryWheelRim (C i)).support,a=x∨a∈A) ∧
  33         (∀ i,H x (ordinaryWheelHub (C i))) ∧
  34         (∀ i j,i≠j → Disjoint (ordinaryWheelResources (C i)) (ordinaryWheelResources (C j))) ∧
  35         (∀ a,(wheelFamilyRim C).Adj x a ↔ Q.Adj x a) ∧
  36         (∀ u,wheelFamilySpokes C x u ↔ H x u) ∧
  37         (∀ a,a≠x → (wheelFamilyRim C).degree a≤2*(K+1)) ∧
  38         (∀ a,a≠x → (Finset.univ.filter (fun u=>wheelFamilySpokes C a u)).card≤K+1) ∧
  39         (∀ u,(Finset.univ.filter (fun a=>wheelFamilySpokes C a u)).card≤r) := by
  40   intro d
  41   obtain ⟨N,hN⟩:=ordinary_neighbour_pairs_on_domains lam hlam d
  42   refine ⟨N,?_⟩
  43   intro V U _ _ _ _ Q F H r K hr hK x A hxA hSA b T hn hbal hF hFx hrow hcol hcut hcount hbudget
  44   let S:=Q.neighborFinset x
  45   obtain ⟨v,w,u,hu,huT,hucov,hpair,hvw,hdis,hcover⟩:=by
  46     exact hN V U S T hn
  47       (by simpa only [S,SimpleGraph.card_neighborFinset_eq_degree] using hbal) H F
  48       (fun a ha=>hrow a ((Q.mem_neighborFinset x a).mp ha)) hcol hF hcut
  49   have hxv (i:Fin T.card):Q.Adj x (v i):=(Q.mem_neighborFinset x _).mp (hpair i).1
  50   have hxw (i:Fin T.card):Q.Adj x (w i):=(Q.mem_neighborFinset x _).mp (hpair i).2.1
  51   have hroot (i:Fin T.card) : ∀ a∈({x,v i,w i}:Finset V),∀ c∈({x,v i,w i}:Finset V),¬F.Adj a c:=by
  52     intro a ha c hc
  53     simp only [Finset.mem_insert,Finset.mem_singleton] at ha hc
  54     rcases ha with rfl|rfl|rfl <;> rcases hc with rfl|rfl|rfl
  55     · exact F.loopless.irrefl _
  56     · exact hFx (v i) (hxv i)
  57     · exact hFx (w i) (hxw i)
  58     · exact fun h=>hFx (v i) (hxv i) h.symm
  59     · exact F.loopless.irrefl (v i)
  60     · exact (hpair i).2.2.2.2
  61     · exact fun h=>hFx (w i) (hxw i) h.symm
  62     · exact fun h=>(hpair i).2.2.2.2 h.symm
  63     · exact F.loopless.irrefl (w i)
  64   obtain ⟨C,hC,hCdis,hcap,hdeg,hCR,hCH,hrowdeg,hcoldeg⟩:=capped_ordinary_root_wheels Q F H
  65     r T.card K hr hK x v w u hu hvw hdis (fun i=>A.filter (fun a=>H a (u i)))
  66     (fun i hx=>hxA (Finset.mem_filter.mp hx).1) hxv hxw hroot
  67     (fun i=>⟨(Finset.mem_filter.mp (huT i)).2,(hpair i).2.2.1,(hpair i).2.2.2.1⟩)
  68     (fun i a ha=>(Finset.mem_filter.mp ha).2) d b (by positivity)
  69     (fun a=>Nat.cast_le.mpr (hF a)) (fun i a ha c hc=>hcount (u i) (huT i) a ha c hc) hbudget
  70   refine ⟨C,?_,?_,hCdis,?_,?_,hdeg,hrowdeg,hcoldeg⟩
  71   · intro i a ha
  72     rcases (hC i).2.2.2 a ha with ha|ha
  73     · simp only [Finset.mem_insert,Finset.mem_singleton] at ha
  74       rcases ha with rfl|rfl|rfl
  75       · exact Or.inl rfl
  76       · exact Or.inr (hSA (hpair i).1)
  77       · exact Or.inr (hSA (hpair i).2.1)
  78     · exact Or.inr (Finset.mem_filter.mp ha).1
  79   · intro i
  80     rw [(hC i).1]
  81     exact (Finset.mem_filter.mp (huT i)).2
  82   · intro a
  83     rw [hCR,←hcover]
  84     exact Q.mem_neighborFinset x a
  85   · intro z
  86     rw [hCH]
  87     constructor
  88     · rintro ⟨i,rfl⟩;exact (Finset.mem_filter.mp (huT i)).2
  89     · intro hz;exact hucov z (Finset.mem_filter.mpr ⟨Finset.mem_univ _,hz⟩)
