  13 theorem padding_old_hub_count (hub:S→U) (js:S→Bool) (m:ℕ) (u:U) (j:Bool):
  14     Fintype.card {s // paddingSpokeHub (I:=I) hub m s=Sum.inl u∧paddingSpokeType js m s=j}=
  15       Fintype.card {s:S // hub s=u∧js s=j}:=by
  16   rw [finite_card_sum_predicate (fun s=>paddingSpokeHub (I:=I) hub m s=Sum.inl u∧paddingSpokeType js m s=j)]
  17   simp only [paddingSpokeHub,paddingSpokeType,Sum.elim_inl,Sum.elim_inr,Sum.inl.injEq,
  18     Sum.inr_ne_inl,false_and]
  19   simp only [Fintype.card_eq_zero,add_zero]

  21 theorem padding_new_hub_count (hub:S→U) (js:S→Bool) (m:ℕ) (i:I) (j:Bool):
  22     Fintype.card {s // paddingSpokeHub hub m s=Sum.inr i∧paddingSpokeType js m s=j}=
  23       if j then 1 else m:=by
  24   rw [finite_card_sum_predicate (fun s=>paddingSpokeHub hub m s=Sum.inr i∧paddingSpokeType js m s=j)]
  25   simp only [paddingSpokeHub,paddingSpokeType,Sum.elim_inl,Sum.elim_inr,Sum.inl_ne_inr,
  26     Sum.inr.injEq,false_and,Fintype.card_eq_zero,zero_add]
  27   have he:Fintype.card {z:I×Fin (m+1) // z.1=i∧decide (z.2=0)=j}=
  28       Fintype.card {k:Fin (m+1) // decide (k=0)=j}:=by
  29     apply Fintype.card_congr
  30     exact {
  31       toFun:=fun z=>⟨z.val.2,z.property.2⟩
  32       invFun:=fun k=>⟨(i,k.val),rfl,k.property⟩
  33       left_inv:=fun z=>by apply Subtype.ext;exact Prod.ext z.property.1.symm rfl
  34       right_inv:=fun _=>rfl}
  35   rw [he]
  36   cases j
  37   · simp only [Bool.false_eq_true,ite_false,decide_eq_false_iff_not]
  38     rw [Fintype.card_subtype_compl,Fintype.card_subtype_eq,Fintype.card_fin]
  39     omega
  40   · simp only [ite_true,decide_eq_true_eq,Fintype.card_subtype_eq]

  42 theorem padding_hub_balance (hub:S→U) (js:S→Bool) (m:ℕ)
