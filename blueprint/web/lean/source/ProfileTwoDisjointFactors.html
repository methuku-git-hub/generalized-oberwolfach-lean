  17 theorem profile_two_disjoint_factors (a : I → ℕ) (c J : ℕ) (hc : 3 ≤ c)
  18     (ha : ∀ i,1 ≤ a i) (hsize : ∑ i,a i=c)
  19     (S : T → VertexAbsorberTemplate c) (a₀ b B : ℝ)
  20     (ha₀ : 0 < a₀) (hb : 0 < b) (hB : 0 ≤ B) :
  21     ∃ η : ℝ,0 < η ∧ ∃ k₀ : ℕ,∀ k : ℕ,k₀ ≤ k →
  22       ∀ (V : I → Type) [∀ i,Fintype (V i)] [∀ i,DecidableEq (V i)]
  23         (C : Type) [Fintype C] [DecidableEq C],
  24       (∀ i,Fintype.card (V i)=a i*k) →
  25       ∀ (R : C → Finset (Σ i,V i)) (w : C → ℝ),
  26       (∀ C₀ i,(Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈R C₀)).card=a i) →
  27       (∀ Y : Finset (Σ i,V i),Fintype.card {C₀ : C // R C₀=Y} ≤ J) →
  28       (∀ C₀,b/(k:ℝ)^(c-1) ≤ w C₀ ∧ w C₀ ≤ B/(k:ℝ)^(c-1)) →
  29       (∀ v,|1-hypergraphWeightedDegree R w v| ≤ η) →
  30       (∀ σ : ∀ i,FixedSubset (V i) (a i),
  31         a₀*(k:ℝ)^(c*c) ≤ Fintype.card (LabelledAbsorberFamily R S (profileRootSet σ))) →
  32       ∀ F : C → Finset ((Σ i,V i)×(Σ i,V i)),
  33       (∀ C₀,∀ e∈F C₀,e.1≠e.2 ∧ e.1∈R C₀ ∧ e.2∈R C₀) →
  34       (∀ C₀,(F C₀).card ≤ c) →
  35       (∀ C₀ v,((F C₀).filter (fun e => v=e.1 ∨ v=e.2)).card ≤ 2) →
  36       ∃ D₀ D₁ : Finset C,
  37         ((∀ x∈D₀,∀ y∈D₀,x≠y → Disjoint (R x) (R y)) ∧ D₀.biUnion R=Finset.univ) ∧
  38         ((∀ x∈D₁,∀ y∈D₁,x≠y → Disjoint (R x) (R y)) ∧ D₁.biUnion R=Finset.univ) ∧
  39         Disjoint (D₀.biUnion F) (D₁.biUnion F) := by
  40   obtain ⟨η₀,hη₀,k₀,h₀⟩ := profile_factor_extraction a c J hc ha hsize S a₀ b B ha₀ hb hB
  41   obtain ⟨η₁,hη₁,h₁⟩ := profile_factor_avoiding_pairs a c J hc ha hsize S a₀ b B ha₀ hb hB
  42   obtain ⟨k₁,h₁⟩ := h₁ 2
  43   refine ⟨min η₀ η₁,lt_min hη₀ hη₁,max k₀ k₁,?_⟩
  44   intro k hk V _ _ C _ _ hn R w hR hJ hw herr habs F hF hFsize hFd
  45   have hk₀ := (le_max_left k₀ k₁).trans hk
  46   have hk₁ := (le_max_right k₀ k₁).trans hk
  47   have herr₀ (v) := (herr v).trans (min_le_left η₀ η₁)
  48   have herr₁ (v) := (herr v).trans (min_le_right η₀ η₁)
  49   obtain ⟨D₀,hD₀⟩ := h₀ k hk₀ V C hn R w hR hJ hw herr₀ habs
  50   let E := D₀.biUnion F
  51   have hE (e) (he : e∈E) : e.1≠e.2 := by
  52     obtain ⟨C₀,_,heC⟩ := Finset.mem_biUnion.mp he
  53     exact (hF C₀ e heC).1
  54   have hRC (C₀ : C) : (R C₀).card=c := by rw [sigma_finset_card];simp only [hR];exact hsize
  55   have hnall : Fintype.card (Σ i,V i)=c*k := by
  56     simp only [Fintype.card_sigma,hn,← Finset.sum_mul,hsize]
  57   have hEsize : E.card ≤ c*k := by
  58     have hh := resource_factor_pairs_card R F D₀ hD₀.1 (fun C₀ _ => by simpa only [hRC] using hFsize C₀)
  59     rw [hD₀.2,Finset.card_univ,hnall] at hh
  60     exact hh
  61   have hEd (v : Σ i,V i) : (E.filter (fun e => v=e.1 ∨ v=e.2)).card ≤ 2 :=
  62     resource_factor_pairs_incidence R F D₀ hD₀.1 (fun C₀ _ e he => (hF C₀ e he).2)
  63       2 (fun C₀ _ v => hFd C₀ v) v
  64   obtain ⟨D₁,hD₁,havoid⟩ := h₁ k hk₁ V C hn R w hR hJ hw herr₁ habs E hE hEsize hEd
  65   exact ⟨D₀,D₁,hD₀,hD₁,resource_pair_avoidance_disjoint R F D₁ E
  66     (fun C₀ _ e he => (hF C₀ e he).2) havoid⟩
