  11 theorem simple_walk_interval_concatenation
  12     {V:Type*} (G:SimpleGraph V) (v:ℕ → V) (p:∀i:ℕ,G.Walk (v i) (v (i+1)))
  13     (c n start:ℕ)
  14     (hp:∀i,start ≤ i → i < start+n → (p i).IsPath)
  15     (hlen:∀i,start ≤ i → i < start+n → (p i).length=c)
  16     (hdis:∀i j,start ≤ i → i < start+n → start ≤ j → j < start+n → i ≠ j → 
  17       ∀x ∈ (p i).support.dropLast,x ∉ (p j).support.dropLast)
  18     (hend:∀i,start ≤ i → i < start+n → v (start+n) ∉ (p i).support.dropLast):
  19     ∃q:G.Walk (v start) (v (start+n)),q.IsPath ∧ q.length=n*c ∧
  20       ∀x,x ∈ q.support ↔ x=v (start+n) ∨
  21         ∃i,start ≤ i ∧ i < start+n ∧ x ∈ (p i).support.dropLast := by
  22   induction n generalizing start with
  23   | zero =>
  24     refine ⟨SimpleGraph.Walk.nil,SimpleGraph.Walk.IsPath.nil,by simp,?_⟩
  25     intro x
  26     simp
  27     omega
  28   | succ n ih =>
  29     have heq:start+1+n=start+(n+1):=by omega
  30     have hp':∀i,start+1 ≤ i → i < start+1+n → (p i).IsPath:=by
  31       intro i hi hj;apply hp i  <;> omega
  32     have hl':∀i,start+1 ≤ i → i < start+1+n → (p i).length=c:=by
  33       intro i hi hj;apply hlen i  <;> omega
  34     have hd':∀i j,start+1 ≤ i → i < start+1+n → start+1 ≤ j → j < start+1+n → i ≠ j → 
  35         ∀x ∈ (p i).support.dropLast,x ∉ (p j).support.dropLast:=by
  36       intro i j hi hj hk hl hij x hx
  37       exact hdis i j (by omega) (by omega) (by omega) (by omega) hij x hx
  38     have he':∀i,start+1 ≤ i → i < start+1+n → v (start+1+n) ∉ (p i).support.dropLast:=by
  39       intro i hi hj
  40       rw [heq]
  41       exact hend i (by omega) (by omega)
  42     obtain ⟨q,hq,hqlen,hqmem⟩:=ih (start+1) hp' hl' hd' he'
  43     have hfirst:(p start).IsPath:=hp start (by omega) (by omega)
  44     have hfirstlen:(p start).length=c:=hlen start (by omega) (by omega)
  45     have hsep:∀x ∈ (p start).support.dropLast,x ∉ q.support:=by
  46       intro x hx hq'
  47       rcases (hqmem x).mp hq' with hterminal | ⟨i,hi,hupper,hxi⟩
  48       · have hh:=hend start (by omega) (by omega)
  49         rw [←heq,←hterminal] at hh
  50         exact hh hx
  51       · exact hdis start i (by omega) (by omega) (by omega) (by omega) (by omega) x hx hxi
  52     obtain ⟨hsimple,hsum,hsupport⟩:=simple_walk_append_of_disjoint_initial_support (p start) q hfirst hq hsep
  53     have hresult:∃w:G.Walk (v start) (v (start+1+n)),w.IsPath ∧ w.length=(n+1)*c ∧
  54       ∀x,x ∈ w.support ↔ x=v (start+1+n) ∨
  55         ∃i,start ≤ i ∧ i < start+1+n ∧ x ∈ (p i).support.dropLast:=by
  56       refine ⟨(p start).append q,hsimple,?_,?_⟩
  57       · calc
  58           ((p start).append q).length=(p start).length+q.length:=hsum
  59           _=c+n*c:=by rw [hfirstlen,hqlen]
  60           _=(n+1)*c:=by nlinarith
  61       · intro x
  62         rw [hsupport,List.mem_append,hqmem]
  63         constructor
  64         · rintro (hx|hx|⟨i,hi,hupper,hxi⟩)
  65           · right;exact ⟨start,by omega,by omega,hx⟩
  66           · exact Or.inl hx
  67           · right;exact ⟨i,by omega,by omega,hxi⟩
  68         · rintro (hx|⟨i,hi,hupper,hxi⟩)
  69           · exact Or.inr (Or.inl hx)
  70           · by_cases h:i=start
  71             · subst i;left;exact hxi
  72             · right;right;exact ⟨i,by omega,by omega,hxi⟩
  73     rw [←heq]
  74     exact hresult
  75 end Oberwolfach
