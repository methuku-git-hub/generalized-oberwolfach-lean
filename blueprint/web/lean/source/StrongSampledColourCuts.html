  17 theorem sampledColourOrientation_large_cuts_strong
  18     (α pstar : ℝ) (hα : 0<α) (hα1 : α≤1/2) (hpstar : 0<pstar)
  19     (k q₀ : ℕ) :
  20     ∃ N : ℕ,∀ (V : Type) [Fintype V] [LinearOrder V],N≤Fintype.card V →
  21     ∀ (P : SimpleGraph V), (∀ v,(1/2+α)*Fintype.card V≤(P.degree v:ℝ)) →
  22     ∀ (ρ : Fin k→ℝ) (hρ : ∀ a,0≤ρ a) (hsum : ∑ a,ρ a=1),
  23     1-1/(Fintype.card V:ℝ)^q₀≤
  24       (FiniteLaw.pi (fun _ : Sym2 V=>FiniteLaw.categoricalDirection ρ hρ hsum)).probability
  25         (fun σ=>∀ a : Fin k,pstar≤ρ a → ∀ δ : ℝ,δ≤α/32 → δ≤ρ a/64 → ∀ S T : Finset V,
  26           (1-2*δ)*Fintype.card V<(S.card:ℝ)+T.card →
  27           (α*ρ a/4)*Fintype.card V*min (S.card:ℝ) T.card≤
  28             (matrixCut (sampledColourOrientation P σ a) S T:ℝ)) := by
  29   let ε : ℝ := pstar/32
  30   let q : ℝ := α*pstar^2/1024
  31   have hε : 0<ε := by dsimp [ε];positivity
  32   have hq : 0<q := by dsimp [q];positivity
  33   obtain ⟨N₁,hN₁⟩ := linear_deviation_failure_eventually_small (3*k) 1 ε
  34     (by norm_num) hε 1 (q₀+1)
  35   obtain ⟨N₂,hN₂⟩ := sampledColourCut_quadratic_event_pow k hq (q₀+1)
  36   refine ⟨max 2 (max N₁ N₂),?_⟩
  37   intro V _ _ hnN P hP ρ hρ hsum
  38   let n := Fintype.card V
  39   have hn : 0<n := by dsimp [n];omega
  40   have hn0 : (0:ℝ)<n := by exact_mod_cast hn
  41   let μ := FiniteLaw.pi (fun _ : Sym2 V=>FiniteLaw.categoricalDirection ρ hρ hsum)
  42   have hd := sampledColourDegrees_common_event P P ρ hρ hsum (t:=ε*n) (by positivity)
  43   have ht := hN₁ n (by dsimp [n] at *;omega) (3*k*n) n
  44     (by positivity) (by simp) hn0 (by simp)
  45   have htail : 6*k*n*Real.exp (-2*(ε*n)^2/n)≤1/(n:ℝ)^(q₀+1) := by
  46     convert ht using 1 <;> ring
  47   have hdegrees : 1-1/(n:ℝ)^(q₀+1)≤μ.probability
  48       (fun σ=>∀ a v,
  49         |((∑ y,sampledColourOrientation P σ a v y:ℕ):ℝ)-ρ a/2*P.degree v|<ε*n ∧
  50         |((∑ y,sampledColourOrientation P σ a y v:ℕ):ℝ)-ρ a/2*P.degree v|<ε*n ∧
  51         |(sampledColourDegree P (fun e=>(σ e).1) a v:ℝ)-ρ a*P.degree v|<ε*n) :=
  52     (sub_le_sub_left htail 1).trans hd
  53   have hcuts := hN₂ V (by dsimp [n] at *;omega) P ρ hρ hsum
  54   have hjoint := finiteLaw_good_intersection μ
  55     (fun σ=>∀ a v,
  56         |((∑ y,sampledColourOrientation P σ a v y:ℕ):ℝ)-ρ a/2*P.degree v|<ε*n ∧
  57         |((∑ y,sampledColourOrientation P σ a y v:ℕ):ℝ)-ρ a/2*P.degree v|<ε*n ∧
  58         |(sampledColourDegree P (fun e=>(σ e).1) a v:ℝ)-ρ a*P.degree v|<ε*n)
  59     _ (1/(n:ℝ)^(q₀+1)) hcuts
  60   have hbase := (sub_le_sub_right hdegrees (1/(n:ℝ)^(q₀+1))).trans hjoint
  61   have hsumerr := scaled_inverse_power_le n q₀ hn 2 (by exact_mod_cast (show 2≤n by dsimp [n];omega))
  62   have hbound : 1-1/(n:ℝ)^q₀≤(1-1/(n:ℝ)^(q₀+1))-1/(n:ℝ)^(q₀+1) := by
  63     convert sub_le_sub_left hsumerr 1 using 1 <;> ring
  64   apply (hbound.trans hbase).trans
  65   apply finiteLaw_probability_mono
  66   intro σ hσ a ha δ hδα hδρ S T hlarge
  67   have hrow (v : V) : ρ a*n/4-ε*n≤((∑ y,sampledColourOrientation P σ a v y:ℕ):ℝ) := by
  68     have hh := (abs_lt.mp (hσ.1 a v).1).1
  69     have hdeg := mul_le_mul_of_nonneg_left (hP v) (div_nonneg (hρ a) (by norm_num : (0:ℝ)≤2))
  70     have hpos := mul_nonneg (hρ a) (mul_nonneg hα.le hn0.le)
  71     nlinarith only [hh,hdeg,hpos]
  72   have hcol (v : V) : ρ a*n/4-ε*n≤((∑ y,sampledColourOrientation P σ a y v:ℕ):ℝ) := by
  73     have hh := (abs_lt.mp (hσ.1 a v).2.1).1
  74     have hdeg := mul_le_mul_of_nonneg_left (hP v) (div_nonneg (hρ a) (by norm_num : (0:ℝ)≤2))
  75     have hpos := mul_nonneg (hρ a) (mul_nonneg hα.le hn0.le)
  76     nlinarith only [hh,hdeg,hpos]
  77   have hh := matrixCut_large_of_degree_and_quadratic_strong P (sampledColourOrientation P σ a)
  78     hα.le hα1 (hρ a) hδα hδρ
  79     (by dsimp [ε,n];nlinarith only [mul_le_mul_of_nonneg_right ha hn0.le])
  80     (by
  81       have hs : pstar^2≤(ρ a)^2 := by nlinarith only [ha,hpstar,hρ a]
  82       dsimp [q]
  83       nlinarith only [mul_le_mul_of_nonneg_left hs hα.le])
  84     (by intro x y;unfold sampledColourOrientation;split_ifs <;> omega)
  85     hP hrow hcol (fun X Y=>hσ.2 a X Y) S T hlarge
  86   exact hh
