  13 theorem uniform_weighted_hypergraph_matching (k B : ℕ) (hk : 2≤k)
  14     (K a γ : ℝ) (hK : 0<K) (ha : 0<a) (hγ : 0<γ) :
  15     ∃ n₀ : ℕ, ∀ n D : ℕ,n₀≤n → n≤D → D≤n^B →
  16     ∀ (V E A : Type) [Fintype V] [Fintype E] [Fintype A] [DecidableEq V],
  17     ∀ (R : E → Finset V) (w : A → E → ℝ) (b W : A → ℝ),
  18     Fintype.card V≤n^B → Fintype.card A≤n^B →
  19     (∀ e,(R e).card=k) → (∀ v,0<Fintype.card (HypergraphEdgesAt R v)) →
  20     (∀ v,Fintype.card (HypergraphEdgesAt R v)≤D) →
  21     (∀ v z,v≠z → ((Finset.univ.filter (fun e=>v∈R e ∧ z∈R e)).card : ℝ)≤K*D/n) →
  22     (∀ i,0<b i) → (∀ i e,0≤w i e ∧ w i e≤b i) →
  23     (∀ i,∑ e,w i e=W i) → (∀ i,a*b i*D*n≤W i) →
  24     ∃ M : Finset E,(∀ e∈M,∀ f∈M,e≠f → Disjoint (R e) (R f)) ∧
  25       ∀ i,|(∑ e∈M,w i e)-W i/D|<γ*W i/D := by
  26   let η := min (γ/20) (1/100)
  27   have hη : 0<η := lt_min (by positivity) (by norm_num)
  28   have hη1 : η≤1/100 := min_le_right _ _
  29   have hηγ : 10*η≤γ := by have hh : η≤γ/20 := min_le_left _ _;linarith
  30   obtain ⟨c,hc,H₀,hH₀,hmatch⟩ := weighted_matching_large_scales k hk η K a hη hη1 hK ha
  31   let J := 128*k*B
  32   obtain ⟨N,hN⟩ := eventually_stretchedExp_le_inv_pow 10 ((16+3*J : ℕ) : ℝ) (c:=c) (p:=1) hc (by norm_num) 1
  33   let H := max H₀ N
  34   refine ⟨H^(64*k),?_⟩
  35   intro n D hn hDn hDB V E A _ _ _ _ R w b W hV hA hsize hpositive hdegree hcodeg hb hw htotal hweight
  36   obtain ⟨hH,hhpow,hnupper⟩ := matching_integer_root_scale (64*k) H n (by omega)
  37     (hH₀.trans (le_max_left _ _)) hn
  38   let h := Nat.nthRoot (64*k) n
  39   change H≤h at hH
  40   change h^(64*k)≤n at hhpow
  41   change n≤h^(2*(64*k)) at hnupper
  42   have hh : 2≤h := hH₀.trans ((le_max_left _ _).trans hH)
  43   have hhR : (1 : ℝ)<h := by exact_mod_cast hh
  44   have hnB : n^B≤h^J := by
  45     calc
  46       _ ≤ (h^(2*(64*k)))^B := Nat.pow_le_pow_left hnupper B
  47       _ = h^J := by rw [← pow_mul];congr 1;dsimp [J];ring
  48   have hsmall : 10*(h : ℝ)^(16+3*J)*Real.exp (-c*h)<1 := by
  49     have he := hN h ((le_max_right _ _).trans hH)
  50     simp only [Real.rpow_one,pow_one] at he
  51     have he' : 10*(h : ℝ)^(16+3*J)*Real.exp (-c*h)≤1/(h : ℝ) := by
  52       simpa only [Real.rpow_natCast] using he
  53     exact he'.trans_lt ((div_lt_one (by linarith : (0 : ℝ)<h)).mpr hhR)
  54   obtain ⟨M,hM,hMw⟩ := hmatch h n D J ((le_max_left _ _).trans hH) hhpow hDn
  55     (hDB.trans hnB) hsmall V E A R w b W (hV.trans hnB) (hA.trans hnB)
  56     hsize hpositive hdegree hcodeg hb hw htotal hweight
  57   refine ⟨M,hM,fun i=>(hMw i).trans_le ?_⟩
  58   have hn0 : 0<n := lt_of_lt_of_le (pow_pos (by omega : 0<h) _) hhpow
  59   have hD0 : 0<D := hn0.trans_le hDn
  60   have hW : 0<W i := (mul_pos (mul_pos (mul_pos ha (hb i)) (Nat.cast_pos.mpr hD0)) (Nat.cast_pos.mpr hn0)).trans_le (hweight i)
  61   exact div_le_div_of_nonneg_right (mul_le_mul_of_nonneg_right hηγ hW.le) (Nat.cast_nonneg _)
