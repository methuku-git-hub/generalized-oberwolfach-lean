  14 theorem select_endpoint_pairings (p:ℝ) (hp:0<p):
  15     ∃ N:ℕ,∀ (V J:Type) [Fintype V] [DecidableEq V] [Fintype J] [DecidableEq J],N≤Fintype.card V→
  16     ∀ (Q F:SimpleGraph V) (A B:J→Finset V),
  17     (∀ i j,i≠j→(((A i∪B i)∩(A j∪B j)).card:ℝ)≤p^2*Fintype.card V)→
  18     (∀ v,((Finset.univ.filter (fun j=>v∈A j∪B j)).card:ℝ)≤p*Fintype.card V)→
  19     (∀ j,∀ K:SimpleGraph V,K≤spanningRestriction Q (A j∪B j)→
  20       (∀ v,(K.degree v:ℝ)≤2*(p*Real.sqrt p*Fintype.card V)+
  21         2*(Nat.ceil (4*(p*Real.sqrt p*Fintype.card V)):ℝ))→
  22       ∃ P:EndpointPairing Q F (A j) (B j),Disjoint P.graph K)→
  23     ∃ P:∀ j,EndpointPairing Q F (A j) (B j),
  24       (∀ i j,i≠j→Disjoint (P i).graph (P j).graph) ∧
  25       ∀ v,((⨆ j,(P j).graph).degree v:ℝ)≤2*(p*Fintype.card V):=by
  26   obtain ⟨N,hN⟩:=bounded_degree_selection_indexed p hp
  27   refine ⟨N,?_⟩
  28   intro V J _ _ _ _ hn Q F A B hinter hmem hsingle
  29   let family:J→Set (SimpleGraph V):=fun j G=>∃ P:EndpointPairing Q F (A j) (B j),P.graph=G
  30   have hchoice (j:J) (K:SimpleGraph V) (hK:K≤spanningRestriction Q (A j∪B j))
  31       (hdeg:∀ v,(K.degree v:ℝ)≤(2:ℕ)*(p*Real.sqrt p*Fintype.card V)+
  32         (2:ℕ)*(Nat.ceil (4*(p*Real.sqrt p*Fintype.card V)):ℝ)):
  33       ∃ G:SimpleGraph V,G∈family j ∧ G≤spanningRestriction Q (A j∪B j) ∧
  34         (∀ v,G.degree v≤2) ∧ Disjoint G K:=by
  35     obtain ⟨P,hP⟩:=hsingle j K hK (by simpa only [Nat.cast_ofNat] using hdeg)
  36     exact ⟨P.graph,⟨P,rfl⟩,P.graph_le,P.graph_degree,hP⟩
  37   obtain ⟨G,hG,hdis,hdeg⟩:=hN V J hn 2 Q (fun j=>A j∪B j) family hinter hmem hchoice
  38   have hrep (j:J):∃ P:EndpointPairing Q F (A j) (B j),P.graph=G j:=(hG j).1
  39   choose P hP using hrep
  40   refine ⟨P,?_,?_⟩
  41   · intro i j hij;rw [hP,hP];exact hdis i j hij
  42   · intro v
  43     have he:(⨆ j,(P j).graph)=⨆ j,G j:=iSup_congr hP
  44     rw [he];simpa only [Nat.cast_ofNat] using hdeg v
