  14 theorem ordinary_rim_correction (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  15     ∃ N : ℕ,∀ (V U : Type) [Fintype V] [DecidableEq V] [Fintype U] [DecidableEq U],
  16       ∀ (P Q F : SimpleGraph V) (H : V → U → Prop) (k : ℕ),N≤k →
  17       ∀ θ : ℝ,cycleThreshold r≤θ → ∀ γ η : ℝ,0<γ → 0≤η →
  18       0<Fintype.card V → Q≤P →
  19       (∀ x y,Q.Adj x y → γ*(Fintype.card V:ℝ)^(k-1)≤
  20         (Fintype.card (OrdinaryIsolatorConfiguration P Q F H k θ α x y):ℝ)) →
  21       ∀ ψ : OrdinaryWheel Q F H r → ℝ,
  22       (∀ x y,|ordinaryRimResidual ψ x y|≤η) →
  23       ∃ Φ : OrdinaryWheel Q F H r → ℝ,
  24         (∀ x y,ordinaryRimLoad Φ x y=ordinaryRimResidual ψ x y) ∧
  25         (∀ D,|Φ D|≤η*(r+1)^2/(γ*(Fintype.card V:ℝ)^(r-1))) ∧
  26         (∀ v u,|ordinarySpokeLoad Φ v u|≤η/γ) := by
  27   obtain ⟨N,hN⟩ := ordinary_averaged_isolator_exists r hr α hα
  28   refine ⟨N,?_⟩
  29   intro V U _ _ _ _ P Q F H k hk θ hθ γ η hγ hη hn hQP hcount ψ herr
  30   have hex (x y : V) : ∃ I : OrdinaryWheel Q F H r → ℝ,
  31       (¬Q.Adj x y → ∀ D,I D=0) ∧
  32       (Q.Adj x y → ∀ a b,ordinaryRimLoad I a b=
  33         if (SimpleGraph.edge x y).Adj a b then 1 else 0) ∧
  34       (∀ D,|I D|≤1/(γ*(Fintype.card V:ℝ)^
  35         (((ordinaryWheelRim D).support.toFinset∪{x,y}).card-1))) ∧
  36       (∀ v u,|ordinarySpokeLoad I v u|≤
  37         (if v=x ∨ v=y then (1:ℝ) else 0)/(2*γ*Fintype.card V)) := by
  38     by_cases hxy : Q.Adj x y
  39     · obtain ⟨I,hR,hb,hs⟩ := hN V U P Q F H k hk θ hθ γ hγ hn x y (hQP hxy) (hcount x y hxy)
  40       exact ⟨I,(fun hn => False.elim (hn hxy)),(fun _ => hR),hb,hs⟩
  41     · refine ⟨fun _ => 0,(fun _ _ => rfl),(fun h => False.elim (hxy h)),?_,?_⟩
  42       · intro D;rw [abs_zero];positivity
  43       · intro v u
  44         simp only [ordinarySpokeLoad,ite_self,Finset.sum_const_zero,abs_zero]
  45         positivity
  46   choose I hzero hload hbound hspoke using hex
  47   let Φ (D : OrdinaryWheel Q F H r) : ℝ :=
  48     ∑ p : V×V,(ordinaryRimResidual ψ p.1 p.2/2)*I p.1 p.2 D
  49   have hb := ordinary_rim_correction_bounds hr ψ I γ η hγ hη hn herr
  50     (fun x D => hzero x x Q.irrefl D)
  51     (fun x y _ => hbound x y) hspoke
  52   exact ⟨Φ,ordinary_rim_correction_loads ψ I hload,hb.1,hb.2⟩
  53 
  54 end Oberwolfach
