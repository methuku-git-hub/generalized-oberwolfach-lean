  12 theorem marked_single_wheel_physical_path
  13     {V E K:Type*} (G:SimpleGraph V) (tail head:E→V) (ktail khead:K→V)
  14     (m b:ℕ) (π:Equiv.Perm V) (hb:1≤b)
  15     (C:MarkedCycleTuple tail head ktail khead m π b)
  16     (hroot:π (C.val 0)≠C.val 0)
  17     (hE:∀e:E,G.Adj (tail e) (head e))
  18     (hK:∀e:K,G.Adj (ktail e) (π (khead e))):
  19     ∃f:Fin ((m+1)+1)↪V,
  20       (∀i:Fin (m+1),f i.castSucc=C.val i) ∧ f (Fin.last (m+1))=π (C.val 0) ∧
  21       (∀j:Fin (m+1),G.Adj (f ⟨j.val,by omega⟩) (f ⟨j.val+1,by omega⟩)) := by
  22   have havoid:π (C.val 0)∉Set.range C.val:=by
  23     rintro ⟨i,hi⟩
  24     by_cases h:i=0
  25     · subst i
  26       exact hroot hi.symm
  27     · have hne:(0:Fin (m+1))≠i:=Ne.symm h
  28       exact (permutationSeparated_successor_avoid π b hb C.val C.property.1 0 i hne) hi.symm
  29   let f:Fin ((m+1)+1)↪V:=⟨Fin.snoc C.val (π (C.val 0)),
  30     Fin.snoc_injective_of_injective (permutationSeparated_injective π b C.val C.property.1) havoid⟩
  31   have hadj:∀j:Fin (m+1),G.Adj (f ⟨j.val,by omega⟩) (f ⟨j.val+1,by omega⟩):=by
  32     intro j
  33     change G.Adj (f j.castSucc) (f j.succ)
  34     refine Fin.lastCases ?_ (fun i=>?_) j
  35     · obtain ⟨e,he₁,he₂⟩:=C.property.2.2
  36       have h:=hK e
  37       rw [he₁,he₂] at h
  38       simpa [f] using h
  39     · obtain ⟨e,he₁,he₂⟩:=C.property.2.1 i
  40       have h:=hE e
  41       rw [he₁,he₂] at h
  42       have hi:i.castSucc.succ=i.succ.castSucc:=by apply Fin.ext; rfl
  43       simpa only [f, Function.Embedding.coeFn_mk, hi, Fin.snoc_castSucc] using h
  44   refine ⟨f,?_,?_,hadj⟩
  45   · intro i; simp only [f, Function.Embedding.coeFn_mk, Fin.snoc_castSucc]
  46   · simp only [f, Function.Embedding.coeFn_mk, Fin.snoc_last]
  47 end Oberwolfach
