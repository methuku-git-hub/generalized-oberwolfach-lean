  12 theorem prescribed_pairwise_rel_or_eq {A:Type} (R:A→A→Prop) (hsym:∀ a b,R a b→R b a):
  13     ∀ l:List A,l.Pairwise R→∀ a∈l,∀ b∈l,a=b ∨ R a b := by
  14   intro l
  15   induction l with
  16   | nil => intro h a ha;simp at ha
  17   | cons x xs ih =>
  18     intro h a ha b hb
  19     have hh:=List.pairwise_cons.mp h
  20     rcases List.mem_cons.mp ha with ha|ha
  21     · subst a
  22       rcases List.mem_cons.mp hb with hb|hb
  23       · subst b;exact Or.inl rfl
  24       · exact Or.inr (hh.1 b hb)
  25     · rcases List.mem_cons.mp hb with hb|hb
  26       · subst b;exact Or.inr (hsym x a (hh.1 a ha))
  27       · exact ih hh.2 a ha b hb

  36 theorem prescribed_source_family_unique_directions
  37     (Dsrc:I→X→X→Prop) (Fsrc:I→SimpleGraph X) (ps:List (I×PrescribedSourcePiece X))
  38     (hpieces:∀ p∈ps,p.2.Valid (Dsrc p.1) (Fsrc p.1)) (hapart:ps.Pairwise PrescribedPiecesApart)
  39     (hcoverA:∀ i x y,Dsrc i x y→∃ p∈ps,p.1=i ∧ (x,y)∈p.2.arcs):
  40     (∀ i x y z,Dsrc i x y→Dsrc i x z→y=z) ∧
  41     (∀ i x y z,Dsrc i y x→Dsrc i z x→y=z) := by

  48       · exact heq
