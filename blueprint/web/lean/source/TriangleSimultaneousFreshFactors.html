  11 def TriangleTransformerFactorPair {V:Type} [Fintype V] [DecidableEq V]
  12     (G:SimpleGraph V) (S Z:Finset V):Prop:=
  13   ∃ M₁ M₂:Finset (OrdinaryCycle G 3),
  14     (∀ C∈M₁,∀ D∈M₁,C≠D → Disjoint (ordinaryCycleVertices G 3 C) (ordinaryCycleVertices G 3 D)) ∧
  15     M₁.biUnion (ordinaryCycleVertices G 3)=S∪Z ∧
  16     (∀ C∈M₂,∀ D∈M₂,C≠D → Disjoint (ordinaryCycleVertices G 3 C) (ordinaryCycleVertices G 3 D)) ∧
  17     M₂.biUnion (ordinaryCycleVertices G 3)=S ∧
  18     (∀ C∈M₁,∀ D∈M₂,Disjoint C.val.val.edgeFinset D.val.val.edgeFinset) ∧
  19     (∀ C∈M₁,∀ v w,C.val.val.Adj v w → v∈S ∨ w∈S)

  21 theorem triangle_transformer_simultaneous_fresh_factors
  22     (K L:ℕ) (β:ℝ) (hβ:0<β) (hβ1:β≤1):
  23     ∃ s₀:ℕ,∀ s:ℕ,s₀≤s→3∣s→∃ N:ℕ,
  24     ∀ (V J:Type) [Fintype V] [DecidableEq V] [Fintype J] [DecidableEq J],
  25     N≤Fintype.card V→Fintype.card J≤K→
  26     ∀ (G:SimpleGraph V) (A Z:J→Finset V),
  27     (∀ v,(3/4+β)*(Fintype.card V:ℝ)≤(G.degree v:ℝ)) →
  28     (∀ i,(Z i).card≤L ∧ 3∣(Z i).card) →
  29     (∀ i,(((Finset.univ:Finset V)\A i).card:ℝ)≤(β/4)*(Fintype.card V:ℝ)) →
  30     ∃ S:J→Finset V,
  31       (∀ i,(S i).card=s ∧ S i⊆A i ∧ Disjoint (S i) (Z i) ∧ TriangleTransformerFactorPair G (S i) (Z i)) ∧
  32       (∀ i j,i≠j→Disjoint (S i) (S j)) :=by

  34   refine ⟨s₀,?_⟩

  37   obtain ⟨M,hM⟩:=exists_nat_gt (4*(K*s:ℝ)/β)
  38   refine ⟨max N₀ M,?_⟩
  39   intro V J _ _ _ _ hn hJ G A Z hdeg hZ hmiss
  40   have hn₀:N₀≤Fintype.card V:=(le_max_left _ _).trans hn
  41   have hnM:(M:ℝ)≤Fintype.card V:=Nat.cast_le.mpr ((le_max_right _ _).trans hn)
  42   have hbudget:(K*s:ℝ)≤(β/4)*(Fintype.card V:ℝ):=by
  43     have hm:4*(K*s:ℝ)<(M:ℝ)*β:=(div_lt_iff₀ hβ).mp hM
  44     nlinarith
  45   let P:J→Finset V→Prop:=fun i S=>S⊆A i ∧ Disjoint S (Z i) ∧ TriangleTransformerFactorPair G S (Z i)
  46   have hchoose:∀ i (B:Finset V),B.card≤Fintype.card J*s →
  47       ∃ S:Finset V,S.card=s ∧ P i S ∧ Disjoint S B:=by
  48     intro i B hB
  49     have hB':(B.card:ℝ)≤(K*s:ℝ):=by exact_mod_cast hB.trans (Nat.mul_le_mul_right s hJ)
  50     have hsub:(Finset.univ:Finset V)\(A i\B) ⊆ ((Finset.univ:Finset V)\A i)∪B:=by
  51       intro x hx
  52       simp only [Finset.mem_sdiff,Finset.mem_univ,true_and,Finset.mem_union] at hx ⊢
  53       tauto
  54     have hcard:(((Finset.univ:Finset V)\(A i\B)).card:ℝ)≤
  55         (((Finset.univ:Finset V)\A i).card:ℝ)+(B.card:ℝ):=by
  56       exact_mod_cast (Finset.card_le_card hsub).trans (Finset.card_union_le _ _)
  57     have hmiss':(((Finset.univ:Finset V)\(A i\B)).card:ℝ)≤(β/2)*(Fintype.card V:ℝ):=by
  58       have hi:=hmiss i
  59       linarith
  60     obtain ⟨S,hSc,hSA,hSZ,M₁,M₂,hp₁,hc₁,hp₂,hc₂,hcross,hold⟩:=
  61       hN₀ V hn₀ G (A i\B) (Z i) (hZ i).1 (hZ i).2 hdeg hmiss'
  62     refine ⟨S,hSc,⟨hSA.trans (Finset.sdiff_subset),hSZ,⟨M₁,M₂,hp₁,hc₁,hp₂,hc₂,hcross,hold⟩⟩,?_⟩
  63     exact Finset.disjoint_left.mpr (by intro x hxS hxB;exact (Finset.mem_sdiff.mp (hSA hxS)).2 hxB)
