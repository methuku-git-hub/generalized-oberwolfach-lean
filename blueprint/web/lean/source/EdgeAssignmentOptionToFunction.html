  11 theorem edge_assignment_option_to_function (S:Finset E) (f:E→Option W)
  12     (hfull:∀e∈S,∃w,f e=some w):
  13     ∃g:{e:E // e∈S}→W,
  14       (∀e,f e.1=some (g e)) ∧
  15       (∀w,(Finset.univ.filter (fun e:{e:E // e∈S}=>g e=w)).card=
  16         (S.filter (fun e=>f e=some w)).card) := by
  17   let g:{e:E // e∈S}→W:=fun e=>Classical.choose (hfull e.1 e.2)
  18   have hg:∀e,f e.1=some (g e):=fun e=>Classical.choose_spec (hfull e.1 e.2)
  19   refine ⟨g,hg,?_⟩
  20   intro w
  21   apply Finset.card_bij (fun e _=>e.1)
  22   · intro e he
  23     have he':g e=w:=(Finset.mem_filter.mp he).2
  24     apply Finset.mem_filter.mpr
  25     exact ⟨e.2,by rw [hg e,he']⟩
  26   · intro a ha b hb hab
  27     exact Subtype.ext hab
  28   · intro e he
  29     have hh:=Finset.mem_filter.mp he
  30     let x:{e:E // e∈S}:=⟨e,hh.1⟩
  31     have hx:g x=w:=Option.some.inj ((hg x).symm.trans hh.2)
  32     refine ⟨x,Finset.mem_filter.mpr ⟨Finset.mem_univ x,hx⟩,rfl⟩
