  15 theorem overlapping_clique_joint_retention
  16     {V C J : Type*} [Fintype V] [Fintype C] [Fintype J]
  17     (G : SimpleGraph V) (L : V → Finset C) (hL : ∀ v,(L v).Nonempty)
  18     {l K B₀ B₁ s A : ℕ} (hl : 1<l) (hcard : ∀ v,(L v).card=l)
  19     (T : J → Finset V) (hT : ∀ j,∀ u∈T j,∀ v∈T j,u≠v → G.Adj u v)
  20     (hinter : ∀ i j,i≠j → (T i∩T j).card≤B₀)
  21     (hsize : ∀ j,(T j).card≤s) (hsl : s≤l)
  22     (hcross : ∀ i j,i≠j → ∀ v∈disjointFamilyRestriction Finset.univ T i,
  23       ((disjointFamilyRestriction Finset.univ T j).filter (G.Adj v)).card≤A)
  24     (c : C) (hc : ∀ j,∀ v∈T j,c∈L v)
  25     (hbound : ∀ v,c∈L v → (colourNeighbourSet G L v c).card≤K)
  26     (hpair : ∀ u v,u≠v → ¬G.Adj u v →
  27       (colourNeighbourSet G L u c∩colourNeighbourSet G L v c).card≤B₁)
  28     (hsmall : 2*((Fintype.card J)^2*B₁)≤l) :
  29     let μ := FiniteLaw.pi (fun u=>(FiniteLaw.uniformOn (L u) (hL u)).withCoin
  30       (equalizationCoin G L l K u) (equalizationCoin_bounds G L (by omega) K u))
  31     let r := ((1-1/(l : ℝ))^K)/(l : ℝ)
  32     |μ.probability (fun σ=>∀ j,∃ v∈T j,coinRetainedColour G σ v=some c)-
  33       (∏ j,((T j).card : ℝ)*r)|≤
  34       (2*(((Fintype.card J)^2*B₁ : ℕ) : ℝ)+
  35         (((Fintype.card J)^2*A : ℕ) : ℝ)+2*(((Fintype.card J)^2*B₀ : ℕ) : ℝ))/(l : ℝ) := by
  36   classical
  37   dsimp only
  38   let μ := FiniteLaw.pi (fun u=>(FiniteLaw.uniformOn (L u) (hL u)).withCoin
  39     (equalizationCoin G L l K u) (equalizationCoin_bounds G L (by omega : 0<l) K u))
  40   let r : ℝ := ((1-1/(l : ℝ))^K)/(l : ℝ)
  41   let T' := disjointFamilyRestriction Finset.univ T
  42   let n := Fintype.card J
  43   have hrho := equalized_retention_rate_bounds (l:=l) (by omega) K
  44   have hlr : (0 : ℝ)<l := by exact_mod_cast (show 0<l by omega)
  45   have hr : 0≤r ∧ r≤1/(l : ℝ) :=
  46     ⟨div_nonneg hrho.1 hlr.le,div_le_div_of_nonneg_right hrho.2 hlr.le⟩
  47   have hsub (j : J) : T' j⊆T j := disjointFamilyRestriction_subset Finset.univ T j
  48   have hsmallT (j : J) : (T' j).card≤s := (Finset.card_le_card (hsub j)).trans (hsize j)
  49   have hclique (j : J) : ∀ u∈T' j,∀ v∈T' j,u≠v → G.Adj u v :=
  50     fun u hu v hv hne=>hT j u (hsub j hu) v (hsub j hv) hne
  51   have hdis (i j : J) (hij : i≠j) : Disjoint (T' i) (T' j) :=
  52     disjointFamilyRestriction_disjoint Finset.univ T (Finset.mem_univ i) hij
  53   have hloss (j : J) : ((T j)\T' j).card≤n*B₀ := by
  54     simpa only [Finset.card_univ] using disjointFamilyRestriction_loss Finset.univ T j
  55       (fun i _ hij=>hinter j i (Ne.symm hij))
  56   have hpoint (j : J) (v : V) (hv : v∈T j) :
  57       μ.probability (fun σ=>coinRetainedColour G σ v=some c)≤r := by
  58     exact (equalizedColour_probability_at G L hL (by omega) hcard v c (hc j v hv)
  59       (hbound v (hc j v hv))).le
  60   have hp := finiteLaw_clique_restriction_uniform_loss G μ T T' hsub c hr.1 hloss hpoint
  61   have hpbound : (n : ℝ)*((n*B₀ : ℕ) : ℝ)*r≤((n^2*B₀ : ℕ) : ℝ)/(l : ℝ) := by
  62     calc
  63       _ ≤ (n : ℝ)*((n*B₀ : ℕ) : ℝ)*(1/(l : ℝ)) :=
  64         mul_le_mul_of_nonneg_left hr.2 (by positivity)
  65       _ = _ := by push_cast;ring
  66   have hp' := hp.2.trans hpbound
  67   have hm (j : J) : (T j).card-(T' j).card≤n*B₀ := by
  68     rw [← Finset.card_sdiff_of_subset (hsub j)]
  69     exact hloss j
  70   have hprod := finiteFamily_restriction_product_loss
  71     (fun j=>(T j).card) (fun j=>(T' j).card) (by omega : 0<l)
  72     (fun j=>Finset.card_le_card (hsub j)) (fun j=>(hsize j).trans hsl) hm hr
  73   have hprodBound : (Fintype.card J : ℝ)*((n*B₀ : ℕ) : ℝ)/(l : ℝ)=
  74       ((n^2*B₀ : ℕ) : ℝ)/(l : ℝ) := by dsimp [n];push_cast;ring
  75   rw [hprodBound] at hprod
  76   have hraw := disjoint_clique_joint_retention_simple G L hL hl hcard T' hclique hdis
  77     hsmallT hsl hcross c (fun j v hv=>hc j v (hsub j hv)) hbound hpair hsmall
  78   dsimp only at hraw
  79   change |μ.probability _-(∏ j,((T' j).card : ℝ)*r)|≤_ at hraw
  80   change |μ.probability _-(∏ j,((T j).card : ℝ)*r)|≤_
  81   obtain ⟨hlo,hhi⟩ := abs_le.mp hraw
  82   have he : (2*((n^2*B₁ : ℕ) : ℝ)+((n^2*A : ℕ) : ℝ)+2*((n^2*B₀ : ℕ) : ℝ))/(l : ℝ)=
  83       2*((n^2*B₁ : ℕ) : ℝ)/(l : ℝ)+((n^2*A : ℕ) : ℝ)/(l : ℝ)+
  84         2*(((n^2*B₀ : ℕ) : ℝ)/(l : ℝ)) := by ring
  85   change _≤(2*((n^2*B₁ : ℕ) : ℝ)+((n^2*A : ℕ) : ℝ)+2*((n^2*B₀ : ℕ) : ℝ))/(l : ℝ)
  86   rw [he]
  87   apply abs_le.mpr
  88   constructor <;> linarith [hp.1,hprod.1,hprod.2]
