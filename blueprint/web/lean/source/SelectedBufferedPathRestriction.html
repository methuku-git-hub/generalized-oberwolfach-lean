  16 theorem buffered_selected_domain_subset
  17     {X I:Type*} [Fintype I] [DecidableEq I] [DecidableEq X]
  18     (S:Finset I) (n:I→ℕ) (inner:∀i,Fin (n i+1)↪X) :
  19     bufferedPathFullDomain (bufferedSelectedLengths S n) (bufferedSelectedPaths S n inner) ⊆
  20       bufferedPathFullDomain n inner := by
  21   classical
  22   intro x hx
  23   rcases Finset.mem_biUnion.mp hx with ⟨i,_,hi⟩
  24   rcases Finset.mem_image.mp hi with ⟨p,_,rfl⟩
