  11 theorem boundary_path_square_root_margin (n N D r:ℕ) (hn:0<n)
  12     (κ C β L d₀ e₀:ℝ) (hκ:0<κ) (hκ1:κ≤1) (hC:0≤C) (hβ:0<β)
  13     (hN:(N:ℝ)≤C*κ^2*(n:ℝ)^2) (hD:(D:ℝ)≤C*κ^2*n)
  14     (hd:d₀≤C*κ^2*n) (he:e₀≤C*κ^2*n)
  15     (hsmall:((2*(r:ℝ)+10)*C+4)*κ≤β/4)
  16     (horder:4*(((r:ℝ)+3)*(L+1)+4)≤β*n):
  17     0<Nat.ceil (κ*n) ∧
  18       (N*r:ℕ)/(Nat.ceil (κ*n):ℝ)+((r:ℝ)+3)*(L+1)+
  19         2*(d₀+2*(Nat.ceil (κ*n)+D:ℕ))+e₀+(D*r:ℕ)+3*D<β*n:=by
  20   have hnR:(0:ℝ)<n:=Nat.cast_pos.mpr hn
  21   have hK:0<Nat.ceil (κ*n):=Nat.ceil_pos.mpr (mul_pos hκ hnR)
  22   have hKlo:κ*n≤(Nat.ceil (κ*n):ℝ):=Nat.le_ceil _
  23   have hKhi:(Nat.ceil (κ*n):ℝ)≤κ*n+1:=(Nat.ceil_lt_add_one (by positivity : (0:ℝ)≤κ*n)).le
  24   have hNr:(N*r:ℕ)/(Nat.ceil (κ*n):ℝ)≤C*r*κ*n:=by
  25     apply (div_le_iff₀ (Nat.cast_pos.mpr hK)).mpr
  26     calc
  27       (N*r:ℕ)≤C*κ^2*(n:ℝ)^2*r:=by exact_mod_cast mul_le_mul_of_nonneg_right hN (Nat.cast_nonneg r)
  28       _=(C*r*κ*n)*(κ*n):=by ring
  29       _≤(C*r*κ*n)*(Nat.ceil (κ*n):ℝ):=
  30         mul_le_mul_of_nonneg_left hKlo (by positivity)
  31   have hκsq:κ^2≤κ:=by nlinarith only [hκ.le,hκ1]
  32   have hscale:C*κ^2*n≤C*κ*n:=by
  33     exact mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_left hκsq hC) hnR.le
  34   have hD':(D:ℝ)≤C*κ*n:=hD.trans hscale
  35   have hdr:d₀≤C*κ*n:=hd.trans hscale
  36   have her:e₀≤C*κ*n:=he.trans hscale
  37   have hDr:(D:ℝ)*r≤C*κ*n*r:=mul_le_mul_of_nonneg_right hD' (Nat.cast_nonneg r)
  38   have hmain:((2*(r:ℝ)+10)*C+4)*κ*n≤β*n/4:=by
  39     nlinarith only [mul_le_mul_of_nonneg_right hsmall hnR.le]
  40   refine ⟨hK,?_⟩
  41   push_cast
  42   have hNr':(N:ℝ)*r/(Nat.ceil (κ*n):ℝ)≤C*r*κ*n:=by exact_mod_cast hNr
  43   nlinarith only [hNr',hKhi,hD',hdr,her,hDr,hmain,horder,mul_pos hβ hnR]

  45 theorem boundary_path_square_root_degrees (n D r:ℕ) (κ C ν:ℝ)
  46     (hκ:0<κ) (hκ1:κ≤1) (hC:0≤C) (hscale:1≤κ*n)
  47     (hD:(D:ℝ)≤C*κ^2*n)
  48     (hrow:3*(C+2)*κ≤ν) (hcol:C*r*κ≤ν):
  49     (3*(Nat.ceil (κ*n)+D):ℕ)≤ν*n ∧ (D*r:ℕ)≤ν*n:=by
  50   have hn:(0:ℝ)≤n:=Nat.cast_nonneg _
  51   have hKhi:(Nat.ceil (κ*n):ℝ)≤κ*n+1:=(Nat.ceil_lt_add_one (by positivity : (0:ℝ)≤κ*n)).le
  52   have hκsq:κ^2≤κ:=by nlinarith only [hκ.le,hκ1]
  53   have hD':(D:ℝ)≤C*κ*n:=hD.trans
  54     (mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_left hκsq hC) hn)
