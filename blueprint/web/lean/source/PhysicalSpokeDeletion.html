  11 theorem physical_spoke_deleted_row (D : SimpleGraph (V⊕U)) (v : V) :
  12     (Finset.univ.filter (fun u=>D.Adj (Sum.inl v) (Sum.inr u))).card≤D.degree (Sum.inl v) := by
  13   let A:=Finset.univ.filter (fun u=>D.Adj (Sum.inl v) (Sum.inr u))
  14   have hs : A.image Sum.inr⊆D.neighborFinset (Sum.inl v) := by
  15     intro z hz
  16     obtain ⟨u,hu,rfl⟩:=Finset.mem_image.mp hz
  17     exact (SimpleGraph.mem_neighborFinset _ _ _).mpr (Finset.mem_filter.mp hu).2
  18   have he : (A.image (Sum.inr:U→V⊕U)).card=A.card := Finset.card_image_of_injective _ Sum.inr_injective
  19   have hh:=Finset.card_le_card hs
  20   rw [he,SimpleGraph.card_neighborFinset_eq_degree] at hh
  21   exact hh
