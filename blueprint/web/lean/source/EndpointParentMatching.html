  16 theorem endpoint_parent_matching_extension (P F : SimpleGraph V) (Px : Finset V) (x : V)
  17     (a c C : ℝ) (ha : 0<a) (hc : 0≤c) (hC : 0≤C)
  18     (weight : (spanningRestriction P Px).edgeSet→ℝ)
  19     (hweight : ∀ e,c/a≤weight e ∧ weight e≤C/a)
  20     (hsum : ∀ v∈Px,(∑ e:(spanningRestriction P Px).edgeSet,if v∈e.val then weight e else 0)=1) :
  21     let w:=graphEdgeWeightExtension (spanningRestriction P Px) weight
  22     (∀ v∈Px,∀ u∈Px,0≤w s(v,u) ∧ w s(v,u)≤C/a) ∧
  23     (∀ v∈Px,∑ u∈Px,w s(v,u)=1) ∧
  24     (∀ v∈Px,∀ u∈Px,¬P.Adj v u → w s(v,u)=0) ∧
  25     (∀ e∈(endpointEligibleGraph P F Px x).edgeSet,c/a≤w e ∧ w e≤C/a) := by
  26   dsimp only
  27   refine ⟨?_,?_,?_,?_⟩
  28   · intro v hv u hu
  29     by_cases he:s(v,u)∈(spanningRestriction P Px).edgeSet
  30     · rw [graphEdgeWeightExtension_edge _ _ _ he]
  31       exact ⟨(div_nonneg hc ha.le).trans (hweight ⟨_,he⟩).1,(hweight ⟨_,he⟩).2⟩
  32     · rw [graphEdgeWeightExtension_nonedge _ _ _ he]
  33       exact ⟨le_refl _,div_nonneg hC ha.le⟩
  34   · intro v hv
  35     calc
  36       _ = ∑ u,graphEdgeWeightExtension (spanningRestriction P Px) weight s(v,u) := by
  37         apply Finset.sum_subset (Finset.subset_univ _)
  38         intro u hu hnot
  39         apply graphEdgeWeightExtension_nonedge
  40         intro he
  41         exact hnot ((spanningRestriction_adj P Px v u).mp he).2.2
  42       _ = _ := (graphEdgeWeightExtension_incident _ weight v).trans (hsum v hv)
  43   · intro v hv u hu hnot
  44     apply graphEdgeWeightExtension_nonedge
  45     intro he
  46     exact hnot ((spanningRestriction_adj P Px v u).mp he).1
  47   · intro e he
  48     have he' : e∈(spanningRestriction P Px).edgeSet := by
  49       induction e using Sym2.ind with
  50       | _ v u => exact (spanningRestriction_adj P Px v u).mpr ⟨he.2.2.2.2.1,he.1,he.2.1⟩
  51     rw [graphEdgeWeightExtension_edge _ _ _ he']
  52     exact hweight ⟨_,he'⟩

  55     (hv : v∈Px) (hFx : ¬F.Adj x v) (weight : Sym2 V→ℝ)
  56     (hzero : ∀ u∈Px,¬P.Adj v u → weight s(v,u)=0) :

  58       ∑ e∈(endpointEligibleGraph P F Px x).edgeFinset,if v∈e then weight e else 0 := by
