  13 theorem prescribed_path_segment_infix (A:Finset X):
  14     ∀ (xs:List X) (left:Option X) (acc:List X)
  15       (seg:PrescribedPathSegment X),seg∈prescribedPathSegments A left acc xs→
  16       prescribedPathSegmentVertices seg <:+: (left.toList++acc.reverse++xs) := by
  17   intro xs
  18   induction xs with
  19   | nil =>
  20     intro left acc seg hs
  21     simp only [prescribedPathSegments,List.mem_singleton] at hs
  22     subst seg
  23     simp [prescribedPathSegmentVertices]
  24   | cons x xs ih =>
  25     intro left acc seg hs
  26     by_cases hx:x∈A
  27     · simp only [prescribedPathSegments,hx,ite_true,List.mem_cons] at hs
  28       rcases hs with hs|hs
  29       · subst seg
  30         refine ⟨[],xs,?_⟩
  31         simp [prescribedPathSegmentVertices,List.append_assoc]
  32       · have hi:=ih (some x) [] seg hs
  33         have hsuf:(x::xs) <:+: (left.toList++acc.reverse++x::xs):=by
  34           refine ⟨left.toList++acc.reverse,[],?_⟩
  35           simp [List.append_assoc]
  36         exact (by simpa using hi : prescribedPathSegmentVertices seg <:+: (x::xs)).trans hsuf
  37     · simp only [prescribedPathSegments,hx,ite_false] at hs
  38       have hi:=ih left (x::acc) seg hs
  39       simpa [List.reverse_cons,List.append_assoc] using hi
  40 end Oberwolfach
