  20     (hshort:∀j,Fintype.card (cycles j)<cutoff)

  24       (ext:∀i:A⊕B,Fin (c*d+5)↪X) (inner:∀i:A⊕B,Fin (c*d+1)↪X),
  25       (∀i,ext i=longReservedSigmaBlock F components c d (fun j=>E (components j)) (index i) d (le_refl _)) ∧
  26       (∀i k,inner i k=ext i ⟨k.val+2,by omega⟩) ∧
  27       (∀i j,i≠j→Disjoint (Finset.univ.image (ext i)) (Finset.univ.image (ext j))) ∧

  29       (∀i,(P i).IsPath) ∧ (∀i,(P i).length=c*d+4) ∧ (∀i,(P i).support=List.ofFn (ext i)) ∧
  30       (∀i j,i≠j→Disjoint (P i).support.toFinset (P j).support.toFinset) ∧

  33       (∀j,(C j).toSubgraph.verts=(cycles j).supp) ∧

  44     (fun i=>hlong (index i).1) hshort ℓ hsize ext hdis hcomponent _ _ P hp hsupp
