  27       (edgeColourGraph P (fun e=>(expandedBinaryClass (σ (Sum.inl e))).1) a).Adj x y))
  28     (hedit:∀a x,((Finset.univ.filter (fun y=>Xor ((H a).Adj x y)
  29       ((edgeColourGraph P (fun e=>(expandedBinaryClass (σ (Sum.inl e))).1) a).Adj x y))).card:ℝ)≤ξ*n):

  31       (∀x y,(Z⊔A).Adj x y→(S.Adj x y↔(longPhysicalStateGraph P σ {0,1,2,3,4}).Adj x y)) ∧
  32       (∀v,((changed ((longPhysicalStateGraph P σ {0,1,2,3,4}).neighborFinset v) (S.neighborFinset v)).card:ℝ)≤ξ*n):=by
