  13 theorem factor_cycle_orientation_reaches_component
  14     {X:Type*} [Fintype X] [DecidableEq X] (G:SimpleGraph X) (hG:G.IsRegularOfDegree 2)
  15     (e:∀C:G.ConnectedComponent,SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph)
  16     (C:G.ConnectedComponent) (a b:C):
  17     ∃n:ℕ,((factorCycleOrientation G e)^n) a.val=b.val:=by
  18   classical
  19   let N:=Fintype.card C
  20   let q:=finRotate N
  21   let ia:Fin N:=(e C).symm a
  22   let ib:Fin N:=(e C).symm b
  23   have h2:2≤N:=by have h3:3≤Fintype.card C:=(component_regular_two_cycle_iso G hG C).1;omega
  24   have hfix:∀i:Fin N,q i≠i:=by
  25     intro i hi
  26     have h:=factor_orientation_no_fixed_points G hG e (e C i).val
  27     apply h
  28     rw [factor_cycle_orientation_on_component G e C (e C i)]
  29     have hh:(finRotate N) ((e C).symm (e C i))=i:=by simpa [q] using hi
  30     simpa [N] using congrArg Subtype.val (congrArg (e C) hh)
  31   have hcy:q.IsCycleOn (↑(Finset.univ:Finset (Fin N)):Set (Fin N)):=by
  32     have hh: q.IsCycleOn {i:Fin N | q i≠i}:=(isCycle_finRotate_of_le h2).isCycleOn
  33     have heq:{i:Fin N | q i≠i}=Set.univ:=by ext i;simp [hfix i]
  34     simpa only [heq,Finset.coe_univ] using hh
  35   obtain ⟨m,_,hm⟩:=hcy.exists_pow_eq (Finset.mem_univ ia) (Finset.mem_univ ib)
  36   have hiter:∀k:ℕ,((factorCycleOrientation G e)^k) a.val=(e C ((q^k) ia)).val:=by
  37     intro k
  38     induction k with
  39     | zero => simp [ia]
  40     | succ k ih =>
  41       rw [pow_succ',Equiv.Perm.mul_apply,ih,factor_cycle_orientation_on_component G e C (e C ((q^k) ia))]
  42       simp only [RelIso.symm_apply_apply]
  43       simp only [pow_succ',Equiv.Perm.mul_apply,q,N]
  44   refine ⟨m,?_⟩
  45   rw [hiter m,hm]
  46   simp [ib]
  47 end Oberwolfach
