  11 def finiteWalkClamp {V:Type*} {n:ℕ} (v:Fin (n+1)→V) (i:ℕ):V :=
  12   v ⟨min i n,by omega⟩

  14 def finiteWalkExtension {V:Type*} {n:ℕ} (G:SimpleGraph V) (v:Fin (n+1)→V)
  15     (p:∀i:Fin n,G.Walk (v i.castSucc) (v i.succ)):
  16     ∀i:ℕ,G.Walk (finiteWalkClamp v i) (finiteWalkClamp v (i+1)) := by
  17   intro i
  18   by_cases hi:i<n
  19   · have h₁:v (⟨i,hi⟩:Fin n).castSucc=finiteWalkClamp v i:=by
  20       unfold finiteWalkClamp
  21       apply congrArg v
  22       apply Fin.ext
  23       simp only [Fin.val_castSucc,Fin.val_mk,Nat.min_eq_left (show i ≤ n by omega)]
  24     have h₂:v (⟨i,hi⟩:Fin n).succ=finiteWalkClamp v (i+1):=by
  25       unfold finiteWalkClamp
  26       apply congrArg v
  27       apply Fin.ext
  28       simp only [Fin.val_succ,Fin.val_mk,Nat.min_eq_left (show i+1 ≤ n by omega)]
  29     exact (p ⟨i,hi⟩).copy h₁ h₂
  30   · have hstay:finiteWalkClamp v i=finiteWalkClamp v (i+1):=by
  31       unfold finiteWalkClamp
  32       apply congrArg v
  33       apply Fin.ext
  34       change min i n=min (i+1) n
  35       rw [Nat.min_eq_right (show n ≤ i by omega),Nat.min_eq_right (show n ≤ i+1 by omega)]
  36     exact (SimpleGraph.Walk.nil : G.Walk (finiteWalkClamp v i) (finiteWalkClamp v i)).copy rfl hstay

  38 theorem finite_walk_extension_initial_properties
  39     {V:Type*} {n:ℕ} (G:SimpleGraph V) (v:Fin (n+1)→V)
  40     (p:∀i:Fin n,G.Walk (v i.castSucc) (v i.succ)) (i:Fin n):
  41     ((finiteWalkExtension G v p i).IsPath ↔ (p i).IsPath) ∧
  42       (finiteWalkExtension G v p i).length=(p i).length ∧
