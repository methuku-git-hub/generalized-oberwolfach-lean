  13 def finitePatternRequirements (G : SimpleGraph (Fin K)) (n : ℕ) (f : Fin n→V) : Finset V :=
  14   if h : n<K then (Finset.univ.filter (fun i : Fin n=>G.Adj ⟨i.val,lt_trans i.isLt h⟩ ⟨n,h⟩)).image f else ∅

  16 theorem finitePatternRequirements_card (G : SimpleGraph (Fin K)) (n : ℕ) (f : Fin n→V)
  17     (h : n<K) : (finitePatternRequirements G n f).card≤G.degree ⟨n,h⟩ := by
  18   rw [finitePatternRequirements,dif_pos h]
  19   apply Finset.card_image_le.trans
  20   let e : Fin n ↪ Fin K := ⟨fun i=>⟨i.val,lt_trans i.isLt h⟩,by intro i j he;exact Fin.ext (congrArg (fun i : Fin K=>i.val) he)⟩
  21   have hi : (Finset.univ.filter (fun i : Fin n=>G.Adj (e i) ⟨n,h⟩)).image e ⊆ G.neighborFinset ⟨n,h⟩ := by
  22     intro x hx
  23     obtain ⟨i,hi,rfl⟩ := Finset.mem_image.mp hx
  24     exact (G.mem_neighborFinset _ _).mpr (Finset.mem_filter.mp hi).2.symm
  25   have hh := Finset.card_le_card hi
  26   rw [Finset.card_image_of_injective _ e.injective,SimpleGraph.card_neighborFinset_eq_degree] at hh
  27   exact hh

  29 theorem finitePatternGood_adj (Q F : SimpleGraph V) (A W : Finset V)
  30     (G : SimpleGraph (Fin K)) (f : Fin K→V)
  31     (hf : SparseGreedyGood Q F A W (finitePatternRequirements G) f)
  32     (i j : Fin K) (hij : G.Adj i j) : Q.Adj (f i) (f j) := by
  33   have hordered (i j : Fin K) (hlt : i.val<j.val) (hadj : G.Adj i j) : Q.Adj (f i) (f j) := by
  34     apply hf.2.2.2.2 j
  35     rw [finitePatternRequirements,dif_pos j.isLt]
  36     refine Finset.mem_image.mpr ⟨⟨i.val,hlt⟩,?_,?_⟩
  37     · exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,hadj⟩
  38     · rfl
  39   rcases lt_or_gt_of_ne (show i.val≠j.val from fun h=>hij.ne (Fin.ext h)) with h|h
  40   · exact hordered i j h hij
  41   · exact (hordered j i h hij.symm).symm

  43 theorem finite_pattern_count (Q F : SimpleGraph V) (A W : Finset V)
  44     (G : SimpleGraph (Fin K)) (h : ℕ) (L b a : ℝ)
  45     (hL : 0≤L) (ha : 0≤a) (hF : ∀ v,(F.degree v:ℝ)≤L)
  46     (hbudget : a≤b-((W.card:ℝ)+K)*(L+1))
  47     (hdeg : ∀ i,G.degree i≤h)
  48     (hcommon : ∀ R : Finset V,R.card≤h → b≤((common A R (fun x=>Q.neighborFinset x)).card:ℝ)) :
  49     a^K≤(Fintype.card {f : Fin K→V // SparseGreedyGood Q F A W (finitePatternRequirements G) f}:ℝ) := by
