  15 def markedStarResource (T:Finset V) (W:Finset U):Finset V⊕(V×U)→Prop:=
  16   Sum.elim (fun e=>∃ v∈e,v∈T) (fun z=>z.1∈T∨z.2∈W)

  18 theorem marked_active_inner_avoids_star
  19     (I T:Finset V) (J W:Finset U) (e:Finset V⊕(V×U))
  20     (he:markedInnerResource (I\T) (J\W) e):¬markedStarResource T W e:=by
  21   cases e with
  22   | inl e=>
  23     rintro ⟨v,hv,hT⟩
  24     exact (Finset.mem_sdiff.mp (he hv)).2 hT
  25   | inr z=>
  26     rintro (ht|hw)
  27     · exact (Finset.mem_sdiff.mp he.1).2 ht
  28     · exact (Finset.mem_sdiff.mp he.2).2 hw

  30 theorem marked_protected_resource_restoration
  31     (FE FD FU E₀:Finset (Finset V⊕(V×U))) (I T:Finset V) (J W:Finset U)
  32     (hE:FE.filter (markedInnerResource I J)=E₀)
  33     (hD:FD.filter (markedInnerResource I J)=∅)
  34     (hU:FU.filter (markedInnerResource I J)=FU.filter (markedInnerResource (I\T) (J\W)))
  35     (hEU:Disjoint FE FU)
  36     (hstar:∀ e∈E₀,markedStarResource T W e)
  37     (hdeep:E₀.filter (markedInnerResource T W)=∅)
  38     (hpair:∀ e:Finset V,Sum.inl e∈FU→e.Nonempty):
  39     let A:=(FE∪FD∪FU).filter (markedInnerResource I J)
  40     A=E₀∪FU.filter (markedInnerResource (I\T) (J\W)) ∧
  41     Disjoint E₀ (FU.filter (markedInnerResource (I\T) (J\W))) ∧
  42     A.filter (markedStarResource T W)=E₀ ∧
  43     A.filter (markedInnerResource T W)=∅ ∧
  44     ∀ p:Finset V⊕(V×U)→Prop,
  45       (A.filter p).card≤(E₀.filter p).card+
  46         ((FU.filter (markedInnerResource (I\T) (J\W))).filter p).card:=by
  47   let B:=FU.filter (markedInnerResource (I\T) (J\W))
  48   have he:(FE∪FD∪FU).filter (markedInnerResource I J)=E₀∪B:=by
  49     rw [Finset.filter_union,Finset.filter_union,hE,hD,Finset.union_empty,hU]
  50   have hEB:Disjoint E₀ B:=by
  51     rw [←hE]
  52     exact hEU.mono (Finset.filter_subset _ _) (Finset.filter_subset _ _)
  53   have hBs:B.filter (markedStarResource T W)=∅:=by
  54     apply Finset.eq_empty_iff_forall_notMem.mpr
  55     intro e he
  56     obtain ⟨he,hs⟩:=Finset.mem_filter.mp he
  57     exact marked_active_inner_avoids_star I T J W e (Finset.mem_filter.mp he).2 hs
  58   have hBd:B.filter (markedInnerResource T W)=∅:=by
  59     apply Finset.eq_empty_iff_forall_notMem.mpr
  60     intro e he
  61     obtain ⟨he,hd⟩:=Finset.mem_filter.mp he
  62     obtain ⟨he,ha⟩:=Finset.mem_filter.mp he
  63     have hn:=marked_active_inner_avoids_star I T J W e ha
  64     cases e with
  65     | inl e=>
  66       obtain ⟨v,hv⟩:=hpair e he
  67       exact hn ⟨v,hv,hd hv⟩
  68     | inr z=>exact hn (Or.inl hd.1)
  69   refine ⟨he,hEB,?_,?_,?_⟩
  70   · rw [he,Finset.filter_union,hBs,Finset.union_empty]
  71     exact Finset.filter_eq_self.mpr hstar
  72   · rw [he,Finset.filter_union,hdeep,hBd,Finset.union_empty]
  73   · intro p
  74     rw [he,Finset.filter_union]
  75     exact Finset.card_union_le _ _
