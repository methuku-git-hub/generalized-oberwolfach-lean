  13     (P Bad Unused Unassigned:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ) (hd:1≤d) (I:Finset V)
  14     (α θ ε lam t C₀:ℝ) (hα:0<α) (hθ:0≤θ) (hθhi:θ≤1/100)
  15     (he:0≤ε) (hC:1≤C₀) (hehi:ε≤α/(1000*C₀))
  16     (hlam:0≤lam) (hlamhi:lam≤α/128) (ht:0≤t) (htbound:t≤lam*Fintype.card V)
  17     (hn:(1536*(d:ℝ))/α≤Fintype.card V)
  18     (hI:(I.card:ℝ)≤ε*Fintype.card V)
  19     (hA:∀v∈I,(Unassigned.degree v:ℝ)≤2*θ*I.card+t)
  20     (hB:∀v∈I,(Unused.degree v:ℝ)≤lam*Fintype.card V)
  21     (hSub:Bad≤(spanningRestriction Unassigned I ⊔ spanningRestriction Unused I) ⊔
  22       longPreparationIneligibleGraph P π d):
  23     (∀v,(Bad.degree v:ℝ)≤(α/16)*Fintype.card V) ∧
  24     (Bad.edgeFinset.card:ℝ)≤(θ*ε^2+lam*ε)*(Fintype.card V:ℝ)^2+24*d*Fintype.card V:=by

  26   let a:ℝ:=2*θ*I.card+t
  27   let b:ℝ:=lam*n
  28   have hn0:0≤n:=by dsimp [n];positivity
  29   have hI0:(0:ℝ)≤I.card:=by positivity
  30   have ha:0≤a:=by dsimp [a];positivity
