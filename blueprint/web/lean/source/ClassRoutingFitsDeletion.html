   8 theorem class_routing_fits_deletion (β δ:ℝ) (hβ:0<β) (hδ:0<δ):
   9     ∃q c:ℝ,0<q ∧ 0<c ∧ ∃N:ℕ,1≤N ∧ ∀n:ℕ,N≤n→∀a f b ζ:ℝ,
  10       0≤a→a≤c*n→f≤ζ*n→ζ≤β/16→b≤(δ/16)*n→
  11       ∃L:ℕ,0<L ∧ (n:ℝ)*a/L+2*a+4*L+2*f<β*n ∧ b+a+2*L≤δ*n:=by
  12   obtain ⟨q,c,hq,hc,hqδ,hcδ,hgap⟩:=class_routing_choose_scales_bounded β δ hβ hδ
  13   have hζgap:=hgap (β/16) (le_refl _)
  14   obtain ⟨N,hNpos,hN⟩:=class_routing_linear_margins c q (β/16) β hc.le hq hζgap
  15   refine ⟨q,c,hq,hc,max N ⌈4/δ⌉₊,le_trans hNpos (le_max_left _ _),?_⟩
  16   intro n hn a f b ζ ha hac hf hζ hb
  17   have hn0:0≤(n:ℝ):=Nat.cast_nonneg n
  18   have hnN:N≤n:=(le_max_left _ _).trans hn
  19   have hζn:=mul_le_mul_of_nonneg_right hζ hn0
  20   have hf':f≤(β/16)*n:=by linarith
  21   obtain ⟨hLpos,hrout,hL⟩:=hN n hnN a f ha hac hf'
  22   refine ⟨Nat.ceil (q*n),hLpos,hrout,?_⟩
  23   have hceil:(4/δ:ℝ)≤n:=(Nat.le_ceil _).trans (by exact_mod_cast ((le_max_right _ _).trans hn : ⌈4/δ⌉₊≤n))
  24   have hnlarge:4≤δ*n:=by exact (div_le_iff₀ hδ).mp hceil |>.trans_eq (mul_comm _ _)
  25   have hcN:=mul_le_mul_of_nonneg_right hcδ hn0
  26   have hqN:=mul_le_mul_of_nonneg_right hqδ hn0
