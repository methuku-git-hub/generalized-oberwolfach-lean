  17 theorem markedLocalBase_class_loss (P : SimpleGraph V) (π : Equiv.Perm V) (b m : ℕ)
  18     (A T₀ TK T₀' TK' : Finset V) (hT₀ : T₀⊆A) (hTK : TK⊆A)
  19     (hT₀' : T₀'⊆A) (hTK' : TK'⊆A) (δ : ℝ)
  20     (hD₀ : ((T₀\T₀').card:ℝ)≤δ) (hDK : ((TK\TK').card:ℝ)≤δ)
  21     (σ : Sym2 V→Option (Bool×Bool)) :
  22     (Fintype.card {C : DenseMarkedBaseFamily P π b m T₀ TK // MarkedBaseRims P π b m σ C.val}:ℝ)-
  23       Fintype.card {C : DenseMarkedBaseFamily P π b m T₀' TK' // MarkedBaseRims P π b m σ C.val}≤
  24       ((m:ℝ)+1)*δ*(A.card:ℝ)^m := by
  25   have hc (S₀ SK : Finset V) :
  26       Fintype.card {C : DenseMarkedBaseFamily P π b m S₀ SK // MarkedBaseRims P π b m σ C.val}=
  27       Fintype.card {f : Fin (m+1)→V // DenseMarkedBase P π b S₀ SK f ∧ MarkedBaseRims P π b m σ f} :=
  28     Fintype.card_congr (Equiv.subtypeSubtypeEquivSubtypeInter _ _)
  29   rw [hc,hc]
  30   let P₀ := fun f : Fin (m+1)→V=>DenseMarkedBase P π b T₀ TK f ∧ MarkedBaseRims P π b m σ f
  31   let P₁ := fun f : Fin (m+1)→V=>DenseMarkedBase P π b T₀' TK' f ∧ MarkedBaseRims P π b m σ f
  32   let D := fun i : Fin (m+1)=>if i=0 then TK\TK' else T₀\T₀'
  33   have h0 : ∀ f,P₀ f → ∀ i,f i∈A := by
  34     intro f hf i
  35     exact Fin.cases (hTK hf.1.2.1) (fun i=>hT₀ (hf.1.2.2.1 i)) i
  36   have h1 : ∀ f,P₁ f → ∀ i,f i∈A := by
  37     intro f hf i
  38     exact Fin.cases (hTK' hf.1.2.1) (fun i=>hT₀' (hf.1.2.2.1 i)) i
  39   have hD : ∀ i,((D i).card:ℝ)≤δ := by intro i;dsimp [D];split_ifs <;> assumption
  40   have hbad : ∀ f,P₀ f → ¬P₁ f → ∃ i,f i∈D i := by
  41     intro f hf hn
  42     by_contra hnone
  43     have hk : f 0∈TK' := by
  44       by_contra hnot
  45       apply hnone
  46       exact ⟨0,by simpa only [D,ite_true,Finset.mem_sdiff] using And.intro hf.1.2.1 hnot⟩
  47     have ho : ∀ i : Fin m,f i.succ∈T₀' := by
  48       intro i
  49       by_contra hnot
  50       apply hnone
  51       exact ⟨i.succ,by simpa only [D,Fin.succ_ne_zero,ite_false,Finset.mem_sdiff] using And.intro (hf.1.2.2.1 i) hnot⟩
  52     exact hn ⟨⟨hf.1.1,hk,ho,hf.1.2.2.2⟩,hf.2⟩
  53   have hh := fixedDomainTuple_predicate_loss A P₀ P₁ h0 h1 D δ hD hbad
  54   simp only [P₀,P₁,Fintype.card_fin,Nat.cast_add,Nat.cast_one,Nat.add_sub_cancel] at hh
  55   simpa only [Fintype.card_eq_nat_card] using hh

  57 theorem markedLocalPath_class_loss (P : SimpleGraph V) (π : Equiv.Perm V) (b m : ℕ)
  58     (j : Bool) (v v' : V) (A T₀ TK T₀' TK' : Finset V) (hT₀ : T₀⊆A) (hTK : TK⊆A)
  59     (hT₀' : T₀'⊆A) (hTK' : TK'⊆A) (δ : ℝ)
  60     (hD₀ : ((T₀\T₀').card:ℝ)≤δ) (hDK : ((TK\TK').card:ℝ)≤δ)
  61     (σ : Sym2 V→Option (Bool×Bool)) :
  62     (Fintype.card {C : DenseRootMarkedPathFamily P π b m j T₀ TK v v' // MarkedPathRims P π b m j v v' σ C.val}:ℝ)-
  63       Fintype.card {C : DenseRootMarkedPathFamily P π b m j T₀' TK' v v' // MarkedPathRims P π b m j v v' σ C.val}≤
  64       ((m:ℝ)+1)*δ*(A.card:ℝ)^m := by
  65   have hc (S₀ SK : Finset V) :
  66       Fintype.card {C : DenseRootMarkedPathFamily P π b m j S₀ SK v v' // MarkedPathRims P π b m j v v' σ C.val}=
  67       Fintype.card {f : Fin (m+1)→V // DenseRootMarkedPath P π b j S₀ SK v v' f ∧ MarkedPathRims P π b m j v v' σ f} :=
  68     Fintype.card_congr (Equiv.subtypeSubtypeEquivSubtypeInter _ _)
  69   rw [hc,hc]
  70   let P₀ := fun f : Fin (m+1)→V=>DenseRootMarkedPath P π b j T₀ TK v v' f ∧ MarkedPathRims P π b m j v v' σ f
  71   let P₁ := fun f : Fin (m+1)→V=>DenseRootMarkedPath P π b j T₀' TK' v v' f ∧ MarkedPathRims P π b m j v v' σ f
  72   let D := fun i : Fin (m+1)=>if j=false ∧ i.val=2 then TK\TK' else T₀\T₀'
  73   have h0 : ∀ f,P₀ f → ∀ i,f i∈A := by
  74     intro f hf i
  75     have hi := hf.1.2.2.1 i
  76     split_ifs at hi
  77     · exact hTK hi
  78     · exact hT₀ hi
  79   have h1 : ∀ f,P₁ f → ∀ i,f i∈A := by
  80     intro f hf i
  81     have hi := hf.1.2.2.1 i
  82     split_ifs at hi
  83     · exact hTK' hi
  84     · exact hT₀' hi
  85   have hD : ∀ i,((D i).card:ℝ)≤δ := by intro i;dsimp [D];split_ifs <;> assumption
  86   have hbad : ∀ f,P₀ f → ¬P₁ f → ∃ i,f i∈D i := by
  87     intro f hf hn
  88     by_contra hnone
  89     have hs : ∀ i,f i∈if j=false ∧ i.val=2 then TK' else T₀' := by
  90       intro i
  91       have hi := hf.1.2.2.1 i
  92       by_contra hnot
  93       apply hnone
  94       refine ⟨i,?_⟩
  95       dsimp only [D]
  96       split_ifs at hi hnot ⊢ <;> exact Finset.mem_sdiff.mpr ⟨hi,hnot⟩
  97     exact hn ⟨⟨hf.1.1,hf.1.2.1,hs,hf.1.2.2.2⟩,hf.2⟩
  98   have hh := fixedDomainTuple_predicate_loss A P₀ P₁ h0 h1 D δ hD hbad
  99   simp only [P₀,P₁,Fintype.card_fin,Nat.cast_add,Nat.cast_one,Nat.add_sub_cancel] at hh
