  11 theorem marked_single_wheel_directed_walk
  12     {V E K:Type*} (G:SimpleGraph V) (tail head:E→V) (ktail khead:K→V)
  13     (m b:ℕ) (π:Equiv.Perm V) (hb:1≤b)
  14     (C:MarkedCycleTuple tail head ktail khead m π b)
  15     (hroot:π (C.val 0)≠C.val 0)
  16     (hE:∀e:E,G.Adj (tail e) (head e))
  17     (hK:∀e:K,G.Adj (ktail e) (π (khead e)))
  18     (R:V→V→Prop) (hER:∀e:E,R (tail e) (head e))
  19     (hKR:∀e:K,R (ktail e) (π (khead e))):
  20     ∃p:G.Walk (C.val 0) (π (C.val 0)),p.IsPath ∧ p.length=m+1 ∧
  21       p.support.dropLast=List.ofFn C.val ∧
  22       ∀k,k<p.length→R (p.getVert k) (p.getVert (k+1)):=by
  23   obtain ⟨f,hprefix,hlast,p,hp,hlen,hsupp⟩:=marked_single_wheel_physical_walk G tail head ktail khead m b π hb C hroot hE hK
  24   have hdir:∀j:Fin (m+1),R (f j.castSucc) (f j.succ):=by
  25     intro j
  26     refine Fin.lastCases ?_ (fun i=>?_) j
  27     · obtain ⟨e,he₁,he₂⟩:=C.property.2.2
  28       have h:=hKR e
  29       rw [he₁,he₂] at h
  30       have hl:(Fin.last m).succ=Fin.last (m+1):=Fin.ext rfl
  31       rw [hprefix (Fin.last m),hl,hlast]
  32       exact h
  33     · obtain ⟨e,he₁,he₂⟩:=C.property.2.1 i
  34       have h:=hER e
  35       rw [he₁,he₂] at h
  36       have hi:i.castSucc.succ=i.succ.castSucc:=by apply Fin.ext;rfl
  37       rw [hprefix i.castSucc,hi,hprefix i.succ]
  38       exact h
  39   refine ⟨p,hp,hlen,?_,?_⟩
  40   · rw [hsupp]
  41     exact finite_sequence_dropLast_of_prefix m f C.val hprefix
  42   · intro k hk
  43     have hk':k<m+1:=by simpa [hlen] using hk
  44     have hv:∀j:Fin ((m+1)+1),p.getVert j.val=f j:=by
  45       intro j
  46       have hh:some (p.getVert j.val)=p.support[j.val]? :=p.getVert_eq_support_getElem? (by omega)
  47       rw [hsupp] at hh
  48       simpa only [List.getElem?_ofFn,dif_pos (show j.val<(m+1)+1 by omega),Option.some.injEq] using hh
  49     have hh:=hdir ⟨k,hk'⟩
  50     change R (f ⟨k,by omega⟩) (f ⟨k+1,by omega⟩) at hh
  51     simpa only [←hv ⟨k,by omega⟩,←hv ⟨k+1,by omega⟩] using hh
  52 end Oberwolfach
