  14     L*n^(d-1)≤β/2*n^d := by

  37     (hnbase : 2*(m+1:ℝ)^2≤β*Fintype.card V)

  43   have hbm := polynomial_forbidden_label_margin n β ((m+1:ℝ)^2) (m+1) hn (by omega) hnbase

  45   have hchoices := pairedTupleChoice_count x Base Path
