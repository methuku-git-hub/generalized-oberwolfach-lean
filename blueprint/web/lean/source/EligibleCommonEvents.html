  15 theorem finiteLaw_eligible_bad_probability {Ω J : Type*} [Fintype Ω]
  16     (μ : FiniteLaw Ω) (tests : Finset J) (Eligible Bad : J → Ω → Prop)
  17     (L ε : ℝ) (hε : 0 ≤ ε)
  18     (hL : ∀ ω,((tests.filter (fun j => Eligible j ω)).card:ℝ) ≤ L)
  19     (h : ∀ j∈tests,μ.probability (fun ω => Eligible j ω ∧ Bad j ω) ≤
  20       ε*μ.probability (Eligible j)) :
  21     μ.probability (fun ω => ∃ j∈tests,Eligible j ω ∧ Bad j ω) ≤ ε*L := by
  22   have hsum : (∑ j∈tests,μ.probability (Eligible j)) ≤ L := by
  23     rw [← finiteLaw_expectation_count]
  24     exact (finiteLaw_expectation_mono μ _ _ hL).trans_eq (finiteLaw_expectation_const μ L)
  25   calc
  26     _ ≤ ∑ j∈tests,μ.probability (fun ω => Eligible j ω ∧ Bad j ω) := finiteLaw_union_bound _ _ _
  27     _ ≤ ∑ j∈tests,ε*μ.probability (Eligible j) := Finset.sum_le_sum h
  28     _ = ε*(∑ j∈tests,μ.probability (Eligible j)) := (Finset.mul_sum ..).symm
  29     _ ≤ _ := mul_le_mul_of_nonneg_left hsum hε
