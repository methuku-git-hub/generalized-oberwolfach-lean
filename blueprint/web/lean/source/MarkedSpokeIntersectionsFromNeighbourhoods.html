  17 theorem markedRows_spoke_intersections
  18     (P:SimpleGraph V) (π:Equiv.Perm V) (d h:ℕ) (p k:ℝ)
  19     (q:U→Γ→ℝ) (s:U→Color→ℝ) (g:Γ) (j:Color) (err density:ℝ)
  20     (σ:RawMarkedOutcome V U Γ Color) (H:Bool→V→U→Prop) (mark:Bool)
  21     (hden:∀ u,(if mark then q u g else s u j)=density)
  22     (hr:MarkedRowsWithSpokes P π d h Finset.univ p k q s g j err σ H)
  23     (S:Finset U) (hS:S.card≤h):
  24     |((commonSpokeNeighbours (H mark) S).card:ℝ)-density^S.card*Fintype.card V|<err:=by
  25   let e:U↪(Bool×U):=⟨fun u=>(mark,u),fun _ _ he=>congrArg Prod.snd he⟩
  26   have hi:Set.InjOn (fun a:Bool×U=>a.2) (S.image e):=by
  27     intro a ha b hb he
  28     obtain ⟨a,ha,rfl⟩:=Finset.mem_image.mp ha
  29     obtain ⟨b,hb,rfl⟩:=Finset.mem_image.mp hb
  30     exact congrArg e he
  31   have hh:=hr ∅ ∅ (S.image e) (by simp) (by simp)
  32     (by simpa only [Finset.card_image_of_injective _ e.injective] using hS) hi
  33   have hdom:markedRowDomain P π (3*d) Finset.univ ∅ ∅=Finset.univ:=by
  34     simp [markedRowDomain,Set.InjOn]
  35   have hcount:markedRowCountWithSpokes P π d σ H Finset.univ ∅ (S.image e)=
  36       ((commonSpokeNeighbours (H mark) S).card:ℝ):=by
  37     simp only [markedRowCountWithSpokes,Finset.notMem_empty,false_implies,implies_true,forall_const,true_and,
  38       Finset.forall_mem_image,commonSpokeNeighbours,Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  39     rfl
  40   rw [hdom,hcount,Finset.prod_empty,one_mul,Finset.card_univ,
  41     Finset.prod_image (fun _ _ _ _ he=>e.injective he)] at hh
  42   have hprod:(∏ a∈S,if (e a).1 then q (e a).2 g else s (e a).2 j)=density^S.card:=by
  43     simp only [show ∀ a,(if (e a).1 then q (e a).2 g else s (e a).2 j)=density from hden,
  44       Finset.prod_const]
  45   rw [hprod] at hh
  46   exact hh

  48 theorem markedIndices_spoke_codegree
  49     (π:Equiv.Perm V) (d h:ℕ) (hh:2≤h)
  50     (q:U→Γ→ℝ) (s:U→Color→ℝ) (g:Γ) (j:Color) (err density:ℝ)
  51     (σ:RawMarkedOutcome V U Γ Color) (H:Bool→V→U→Prop) (mark:Bool)
  52     (hden:∀ u,(if mark then q u g else s u j)=density)
  53     (hi:MarkedIndicesWithSpokes π d h Finset.univ q s g j err σ H)
  54     (x y:V) (hxy:¬PermutationClose π (3*d) x y):
  55     |((Finset.univ.filter (fun u=>H mark x u∧H mark y u)).card:ℝ)-density^2*Fintype.card U|<err:=by
  56   have hne:x≠y:=fun he=>hxy (Or.inl he)
  57   have hpair:(mark,x)≠(mark,y):=fun he=>hne (congrArg Prod.snd he)
  58   have hsep:∀ a∈({(mark,x),(mark,y)}:Finset (Bool×V)),∀ b∈({(mark,x),(mark,y)}:Finset (Bool×V)),
  59       a≠b→¬PermutationClose π (3*d) a.2 b.2:=by
  60     intro a ha b hb hab
  61     simp only [Finset.mem_insert,Finset.mem_singleton] at ha hb
  62     rcases ha with rfl|rfl <;> rcases hb with rfl|rfl
  63     · exact False.elim (hab rfl)
  64     · exact hxy
  65     · exact fun hc=>hxy ((permutationClose_symm π (3*d) y x).mp hc)
  66     · exact False.elim (hab rfl)
  67   have hc:=hi {(mark,x),(mark,y)} (by simpa [hpair] using hh) hsep
  68   have hcount:markedIndexCountWithSpokes H Finset.univ {(mark,x),(mark,y)}=
  69       ((Finset.univ.filter (fun u=>H mark x u∧H mark y u)).card:ℝ):=by
  70     simp only [markedIndexCountWithSpokes,Finset.mem_insert,Finset.mem_singleton,
  71       forall_eq_or_imp,forall_eq,Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  72   have hp (u:U):(∏ a∈({(mark,x),(mark,y)}:Finset (Bool×V)),if a.1 then q u g else s u j)=density^2:=by
  73     rw [Finset.prod_insert (by simpa only [Finset.mem_singleton] using hpair),Finset.prod_singleton]
  74     rw [hden]
  75     ring
  76   rw [hcount] at hc
  77   simp only [hp,Finset.sum_const,Finset.card_univ,nsmul_eq_mul] at hc
  78   simpa only [mul_comm] using hc
