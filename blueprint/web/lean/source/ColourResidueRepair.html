  13 theorem colour_residue_repair {V : Type*} [Fintype V] [DecidableEq V] {q h : ℕ}
  14     (hh : 0<h) (hhq : h∣q) (G : SimpleGraph V) (σ : Sym2 V → Fin (q+1))
  15     (hreserve : q*h≤(edgeColourGraph G σ (Fin.last q)).edgeFinset.card) :
  16     ∃ τ : Sym2 V → Fin (q+1),
  17       (∀ a,((edgeColourGraph G τ a).edgeFinset.card:ZMod h)=(G.edgeFinset.card:ZMod h)) ∧
  18       (∀ a v,|((edgeColourGraph G τ a).degree v:ℝ)-((edgeColourGraph G σ a).degree v:ℝ)|≤q*h) ∧
  19       (∀ a u v,u≠v → |(colouredTwoPathCount G τ u v a (Fin.last q):ℝ)-
  20         (colouredTwoPathCount G σ u v a (Fin.last q):ℝ)|≤q*h) := by
  21   choose a ha using fun j : Fin q => residue_increment h G.edgeFinset.card
  22     (edgeColourGraph G σ j.castSucc).edgeFinset.card hh
  23   let I := (j : Fin q)×Fin (a j)
  24   have hI : Fintype.card I≤q*h := by
  25     simp only [I,Fintype.card_sigma,Fintype.card_fin]
  26     calc
  27       _ ≤ ∑ _ : Fin q,h := Finset.sum_le_sum (fun j _ => (ha j).1.le)
  28       _ = _ := by simp
  29   obtain ⟨ι,hι⟩ := Function.Embedding.exists_of_card_le_finset
  30     (α:=I) (s:=(edgeColourGraph G σ (Fin.last q)).edgeFinset) (hI.trans hreserve)
  31   have he (i : I) : ι i∈G.edgeFinset ∧ σ (ι i)=Fin.last q := by
  32     have hh := hι (Set.mem_range_self i)
  33     rw [edgeColourGraph_edges] at hh
  34     exact Finset.mem_filter.mp hh
  35   let c : I → Fin (q+1) := fun i => i.1.castSucc
  36   let τ := recolourCoordinates ι σ c
  37   have hcard (j : Fin q) : (Finset.univ.filter (fun i : I => c i=j.castSucc)).card=a j := by
  38     simp only [Finset.card_filter,I,Fintype.sum_sigma,c,Fin.castSucc_inj]
  39     rw [Finset.sum_eq_single j]
  40     · simp
  41     · intro l _ hlj
  42       simp [hlj]
  43     · simp
  44   have hres (j : Fin q) : ((edgeColourGraph G τ j.castSucc).edgeFinset.card:ZMod h)=
  45       (G.edgeFinset.card:ZMod h) := by
  46     have hc := recolour_nonreserve_card G ι σ c (Fin.last q) j.castSucc
  47       (fun i => (he i).1) (fun i => (he i).2) (Fin.castSucc_ne_last j)
  48     rw [hcard] at hc
  49     change ((edgeColourGraph G (recolourCoordinates ι σ c) j.castSucc).edgeFinset.card:ZMod h)=_
  50     rw [hc]
  51     exact (ZMod.natCast_eq_natCast_iff' _ _ h).mpr (ha j).2
  52   have hIr : (Fintype.card I:ℝ)≤(q:ℝ)*h := by exact_mod_cast hI
  53   refine ⟨τ,all_colour_residues_of_nonreserve G τ hhq hres,?_,?_⟩
  54   · intro j v
  55     exact (recolour_degree_stability G ι σ c j v).trans hIr
  56   · intro j u v huv
  57     exact (recolour_path_count_stability G ι σ c u v huv j (Fin.last q)).trans hIr
