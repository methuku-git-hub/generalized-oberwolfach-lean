  15 theorem labelIndicator_change_sum {X L : Type*} (A : Finset X) (f f' : X→L) (a : L) :
  16     (∑ x∈A,if ¬(f x=a ↔ f' x=a) then (1:ℝ) else 0)≤
  17       (A.filter (fun x=>f' x≠f x)).card := by
  18   simp only [Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  19   apply Finset.sum_le_sum
  20   intro x hx
  21   by_cases he : f' x=f x
  22   · simp [he]
  23   · split_ifs <;> norm_num

  34 theorem retainedMarkedSpokes_row_changes (π : Equiv.Perm V) (d : ℕ)
  35     (σ : RawIntervalFamilyOutcome V U Γ Color) (g₀ : Γ) (t : ℕ)
  36     (J : StartRetentionOutcome (rawStartHubSet π d σ g₀) t) (g : Γ) (j : Color)
  37     (a : Bool) (v : V) (T : Finset U) :
  38     (∑ u∈T,bipAdj (fun x u=>¬(rawMarkedSpokes π d σ g j a x u ↔
  39       retainedRawMarkedSpokes π d σ g₀ t J g j a x u)) v u)≤
  40       ((T.filter (fun u=>retainedRawFamilyState π d σ g₀ t J (v,u)≠
  41         rawIntervalFamilyState π d σ (v,u))).card:ℝ) := by
  42   have hh := labelIndicator_change_sum T (fun u=>rawIntervalFamilyState π d σ (v,u))
  43     (fun u=>retainedRawFamilyState π d σ g₀ t J (v,u))
  44     (if a then some (Sum.inl g) else some (Sum.inr j))
  45   convert hh using 1
  46   · apply Finset.sum_congr rfl
  47     intro u hu
  48     cases a <;> simp only [bipAdj,rawMarkedSpokes,retainedRawMarkedSpokes,Bool.false_eq_true,ite_false,ite_true] <;> split_ifs <;> rfl
  49   · apply congrArg (fun n : ℕ=>(n:ℝ))
  50     apply congrArg Finset.card
  51     ext u
