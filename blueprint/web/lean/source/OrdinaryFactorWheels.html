   6 /-! A local ordinary cycle factor gives actual wheels at the specified hub.
   7 Rims and spokes remain distinct resources; the deleted rim graph has degree
   8 at most two, and every prescribed spoke occurs exactly once. -/

  18 def localOrdinaryWheel (u : U) (hH : ∀ v∈T,H v u)
  19     (C : LocalOrdinaryCycle Q F T r) : OrdinaryWheel Q F H r :=
  20   ⟨(C.val,u),⟨fun v hv=>hH v (C.property.1 v hv),C.property.2⟩⟩

  23     Function.Injective (localOrdinaryWheel (Q:=Q) (F:=F) (r:=r) u hH) := by
  24   intro C D he

  40 theorem local_ordinary_wheel_resources_disjoint (u : U) (hH : ∀ v∈T,H v u)
  41     (C D : LocalOrdinaryCycle Q F T r)
  42     (hCD : Disjoint (localOrdinaryVertices C) (localOrdinaryVertices D)) :
  43     Disjoint (ordinaryWheelResources (localOrdinaryWheel u hH C))
  44       (ordinaryWheelResources (localOrdinaryWheel u hH D)) := by
  45   apply Finset.disjoint_left.mpr
  46   intro e he hf
  47   cases e with
  48   | inl e =>
  49     rw [ordinaryWheelResources_rim] at he hf
  50     induction e using Sym2.inductionOn with
  51     | _ x y =>
  52       have hC : C.val.val.val.Adj x y:=he
  53       have hD : D.val.val.val.Adj x y:=hf
  54       exact Finset.disjoint_left.mp hCD
  55         (Set.mem_toFinset.mpr hC.mem_support_left) (Set.mem_toFinset.mpr hD.mem_support_left)
  56   | inr vu =>
  57     rw [ordinaryWheelResources_spoke] at he hf
  58     exact Finset.disjoint_left.mp hCD
  59       (Set.mem_toFinset.mpr he.2) (Set.mem_toFinset.mpr hf.2)

  61 theorem local_ordinary_wheel_union (u : U) (hH : ∀ v∈T,H v u)
  62     (M : Finset (LocalOrdinaryCycle Q F T r)) :
  63     (M.image (localOrdinaryWheel u hH)).biUnion ordinaryWheelResources=
  64       (ordinaryWheelRimGraph (M.image (localOrdinaryWheel u hH))).edgeFinset.image Sum.inl ∪
  65       (M.biUnion localOrdinaryVertices).image (fun v=>Sum.inr (v,u)) := by
  66   ext e
  67   cases e with
  68   | inl e =>
  69     have hR := ordinaryWheelRimGraph_edges (M.image (localOrdinaryWheel u hH)) e
  70     simpa only [Finset.mem_biUnion,ordinaryWheelResources_rim,Finset.mem_union,
  71       Finset.mem_image,Sum.inl.injEq,Sum.inr_ne_inl,and_false,exists_false,or_false,
  72       SimpleGraph.mem_edgeFinset,exists_eq_right] using hR.symm
  73   | inr vu =>
  74     rcases vu with ⟨v,w⟩
  75     simp only [Finset.mem_biUnion,Finset.mem_image,ordinaryWheelResources_spoke,
  76       Finset.mem_union,Sum.inl_ne_inr,and_false,exists_false,false_or,
  77       Sum.inr.injEq,Prod.mk.injEq]
  78     constructor
  79     · rintro ⟨_,⟨C,hC,rfl⟩,hu,hv⟩
  80       exact ⟨v,⟨C,hC,Set.mem_toFinset.mpr hv⟩,rfl,hu⟩
  81     · rintro ⟨v',⟨C,hC,hv⟩,rfl,hu⟩
  82       exact ⟨localOrdinaryWheel u hH C,⟨C,hC,rfl⟩,hu,Set.mem_toFinset.mp hv⟩

  84 theorem local_ordinary_wheel_degree (u : U) (hH : ∀ v∈T,H v u)
  85     (M : Finset (LocalOrdinaryCycle Q F T r))
  86     (hM : ∀ C∈M,∀ D∈M,C≠D → Disjoint (localOrdinaryVertices C) (localOrdinaryVertices D))
  87     (v : V) : (ordinaryWheelRimGraph (M.image (localOrdinaryWheel u hH))).degree v≤2 := by
  88   let N : Finset (OrdinaryCycle Q r):=M.image Subtype.val
  89   have hN : ∀ C∈N,∀ D∈N,C≠D → Disjoint (ordinaryCycleVertices Q r C) (ordinaryCycleVertices Q r D) := by
  90     intro C hC D hD hne
  91     change C∈M.image Subtype.val at hC
  92     change D∈M.image Subtype.val at hD
  93     obtain ⟨C',hC',heC⟩ : ∃ C' : LocalOrdinaryCycle Q F T r,C'∈M ∧ C'.val=C := Finset.mem_image.mp hC
  94     obtain ⟨D',hD',heD⟩ : ∃ D' : LocalOrdinaryCycle Q F T r,D'∈M ∧ D'.val=D := Finset.mem_image.mp hD
  95     subst C;subst D
  96     exact hM C' hC' D' hD' (fun h=>hne (congrArg Subtype.val h))
  97   have he : ordinaryWheelRimGraph (M.image (localOrdinaryWheel u hH))=ordinaryFactorGraph N := by
  98     ext x y
  99     simp only [ordinaryWheelRimGraph,ordinaryFactorGraph,SimpleGraph.iSup_adj,N,Finset.mem_image]
 100     constructor
 101     · rintro ⟨_,⟨C,hC,rfl⟩,hxy⟩; exact ⟨C.val,⟨C,hC,rfl⟩,hxy⟩
 102     · rintro ⟨_,⟨C,hC,rfl⟩,hxy⟩; exact ⟨localOrdinaryWheel u hH C,⟨C,hC,rfl⟩,hxy⟩
 103   rw [he]
 104   exact ordinaryFactorGraph_degree_le_two N hN v

 106 theorem ordinary_factor_wheels (u : U) (hH : ∀ v∈T,H v u)
 107     (hfactor : HasResourceDecomposition (localOrdinaryVertices (Q:=Q) (F:=F) (T:=T) (r:=r)) T) :
 108     ∃ M : Finset (OrdinaryWheel Q F H r),
 109       (∀ W∈M,∀ Z∈M,W≠Z → Disjoint (ordinaryWheelResources W) (ordinaryWheelResources Z)) ∧
 110       M.biUnion ordinaryWheelResources=(ordinaryWheelRimGraph M).edgeFinset.image Sum.inl ∪
 111         T.image (fun v=>Sum.inr (v,u)) ∧
 112       (∀ v,(ordinaryWheelRimGraph M).degree v≤2) ∧
 113       (∀ W∈M,ordinaryWheelHub W=u) ∧
 114       (∀ v∈T,∃! W : M,Sum.inr (v,u)∈ordinaryWheelResources W.val) := by
 115   obtain ⟨N,hN,hcover⟩:=hfactor
 116   let M:=N.image (localOrdinaryWheel u hH)
 117   have hM : ∀ W∈M,∀ Z∈M,W≠Z → Disjoint (ordinaryWheelResources W) (ordinaryWheelResources Z) := by
 118     intro W hW Z hZ hne
 119     obtain ⟨C,hC,rfl⟩:=Finset.mem_image.mp hW
 120     obtain ⟨D,hD,rfl⟩:=Finset.mem_image.mp hZ
 121     apply local_ordinary_wheel_resources_disjoint
 122     exact hN C hC D hD (fun h=>hne (congrArg (localOrdinaryWheel u hH) h))
 123   have hunion : M.biUnion ordinaryWheelResources=(ordinaryWheelRimGraph M).edgeFinset.image Sum.inl ∪
 124       T.image (fun v=>Sum.inr (v,u)) := by
 125     simpa only [hcover] using local_ordinary_wheel_union u hH N
 126   refine ⟨M,hM,hunion,local_ordinary_wheel_degree u hH N hN,?_,?_⟩
 127   · intro W hW
 128     obtain ⟨C,hC,rfl⟩:=Finset.mem_image.mp hW
 129     rfl
 130   · intro v hv
 131     apply resource_factor_vertex_unique ordinaryWheelResources M hM
 132     rw [hunion]
 133     exact Finset.mem_union_right _ (Finset.mem_image.mpr ⟨v,hv,rfl⟩)
