  14 theorem finiteLaw_event_covariance_bound {Ω : Type*} [Fintype Ω]
  15     (μ : FiniteLaw Ω) (E A : Ω → Prop) :
  16     |μ.probability (fun x=>E x ∧ A x)-μ.probability E*μ.probability A|≤μ.probability E := by
  17   have hEA0 := finiteLaw_probability_nonneg μ (fun x=>E x ∧ A x)
  18   have hEA1 := finiteLaw_probability_mono μ (fun x=>E x ∧ A x) E (fun _ h=>h.1)
  19   have hE0 := finiteLaw_probability_nonneg μ E
  20   have hA0 := finiteLaw_probability_nonneg μ A
  21   have hA1 : μ.probability A≤1 := by
  22     have h := finiteLaw_probability_compl μ A
  23     have hn := finiteLaw_probability_nonneg μ (fun x=>¬A x)
  24     linarith
  25   have hprod0 := mul_nonneg hE0 hA0
  26   have hprod1 := mul_le_mul_of_nonneg_left hA1 hE0
  27   apply abs_le.mpr
  28   constructor <;> nlinarith
