  10 theorem marked_interval_root_card
  11     {V I:Type*} [Fintype V] [Fintype I] [DecidableEq V]
  12     (π:Equiv.Perm V) (k:I→ℕ) (a:I→V)
  13     (hinj:∀j,Function.Injective (fun i:Fin (k j+1)=>(π^i.val) (a j)))
  14     (hdis:∀j l:I,j≠l→Disjoint (permutationWindow π 0 (k j) (a j))
  15       (permutationWindow π 0 (k l) (a l))):
  16     (markedIntervalRootSet π k a).card=∑j,k j:=by
  17   classical
  18   have hsub (j:I):Finset.univ.image (fun i:Fin (k j)=>(π^i.val) (a j))⊆
  19       permutationWindow π 0 (k j) (a j):=by
  20     intro x hx
  21     obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hx
  22     unfold permutationWindow
  23     apply Finset.mem_image.mpr
  24     exact ⟨⟨i.val,by omega⟩,Finset.mem_univ _,by simp; rfl⟩
  25   have hinj' (j:I):Function.Injective (fun i:Fin (k j)=>(π^i.val) (a j)):=by
  26     intro x y h
  27     apply Fin.castSucc_injective
  28     apply hinj j
  29     exact h
  30   unfold markedIntervalRootSet
  31   rw [Finset.card_biUnion (by
  32     intro j hj l hl hjl
  33     exact (hdis j l hjl).mono (hsub j) (hsub l))]
  34   apply Finset.sum_congr rfl
  35   intro j hj
  36   rw [Finset.card_image_of_injective _ (hinj' j),Finset.card_univ,Fintype.card_fin]
