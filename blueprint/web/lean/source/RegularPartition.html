  14 /-- Splitting the directed incidence graph into equal-degree parts gives an
  15 edge partition of the original simple graph, with twice the directed degree.
  16 The pointwise hypothesis records both orientations of each physical edge
  17 exactly once, and rules out loops and opposite directed copies. -/
  18 theorem regularPartition_of_orientedMultiplicities
  19     {n k : ℕ} (G : SimpleGraph (Fin n))
  20     (Q : Fin k → Fin n → Fin n → ℕ) (m : Fin k → ℕ)
  21     (hphysical : ∀ x y, (∑ a, Q a x y) + (∑ a, Q a y x) =
  22       if G.Adj x y then 1 else 0)
  23     (hout : ∀ a x, ∑ y, Q a x y = m a)
  24     (hin : ∀ a y, ∑ x, Q a x y = m a) :
  25     ∃ H : Fin k → SimpleGraph (Fin n),
  26       EdgePartition G H ∧ (∀ a, (H a).IsRegularOfDegree (2 * m a)) ∧
  27       ∀ a x y, (H a).Adj x y ↔ 0 < Q a x y + Q a y x := by
  28   let W (a : Fin k) (x y : Fin n) := Q a x y + Q a y x
  29   have hWsum (x y : Fin n) : ∑ a, W a x y = if G.Adj x y then 1 else 0 := by
  30     simp only [W, Finset.sum_add_distrib]
  31     exact hphysical x y
  32   have hWle (a : Fin k) (x y : Fin n) : W a x y ≤ if G.Adj x y then 1 else 0 := by
  33     rw [← hWsum]
  34     exact Finset.single_le_sum (fun b _ => Nat.zero_le (W b x y)) (Finset.mem_univ a)
  35   let H : Fin k → SimpleGraph (Fin n) := fun a =>
  36     { Adj := fun x y => 0 < W a x y
  37       symm := by
  38         constructor
  39         intro x y h
  40         simpa only [W, Nat.add_comm] using h
  41       loopless := by
  42         constructor
  43         intro x h
  44         have hz : W a x x ≤ 0 := by simpa using hWle a x x
  45         omega }
  46   refine ⟨H, ?_, ?_, ?_⟩
  47   · intro x y
  48     constructor
  49     · change G.Adj x y ↔ ∃ a, 0 < W a x y
  50       have hp : (0 < ∑ a, W a x y) ↔ ∃ a, 0 < W a x y := by
  51         simp only [Finset.sum_pos_iff, Finset.mem_univ, true_and]
  52       rw [← hp, hWsum]
  53       split_ifs <;> simp_all
  54     · intro a b ha hb
  55       change 0 < W a x y at ha
  56       change 0 < W b x y at hb
  57       by_contra hab
  58       have hbmem : b ∈ Finset.univ.erase a :=
  59         Finset.mem_erase.mpr ⟨Ne.symm hab, Finset.mem_univ b⟩
  60       have hb' : W b x y ≤ ∑ c ∈ Finset.univ.erase a, W c x y :=
  61         Finset.single_le_sum (fun c _ => Nat.zero_le (W c x y)) hbmem
  62       have hs := Finset.sum_erase_add Finset.univ (fun c => W c x y)
  63         (Finset.mem_univ a)
  64       have ht : ∑ c, W c x y ≤ 1 := by
  65         rw [hWsum]
  66         split_ifs <;> omega
  67       omega
  68   · intro a x
  69     have hind (y : Fin n) : (if (H a).Adj x y then 1 else 0 : ℕ) = W a x y := by
  70       change (if 0 < W a x y then 1 else 0 : ℕ) = W a x y
  71       have hb : W a x y ≤ 1 := (hWle a x y).trans (by split_ifs <;> omega)
  72       split_ifs <;> omega
  73     rw [SimpleGraph.degree]
  74     calc
  75       _ = (Finset.univ.filter (fun y => (H a).Adj x y)).card := by
  76         congr 1
  77         ext y
  78         simp
  79       _ = ∑ y, W a x y := by rw [Finset.card_filter]; simp only [hind]
  80       _ = 2 * m a := by
  81         simp only [W, Finset.sum_add_distrib, hout, hin]
  82         omega
  83 
  84   · intro a x y
  85     rfl

  87 /-- Equal-degree grouping of an already constructed balanced orientation. -/
  88 theorem regularPartition_of_balancedOrientation
  89     {n k q : ℕ} (G : SimpleGraph (Fin n))
  90     (M : Fin n → Fin n → ℕ)
  91     (hphysical : ∀ x y, M x y + M y x = if G.Adj x y then 1 else 0)
  92     (hout : ∀ x, ∑ y, M x y = k * q)
  93     (hin : ∀ y, ∑ x, M x y = k * q) :
  94     ∃ H : Fin k → SimpleGraph (Fin n),
  95       EdgePartition G H ∧ ∀ a, (H a).IsRegularOfDegree (2 * q) := by
  96   obtain ⟨Q, hQsum, hQout, hQin⟩ :=
  97     regularBipartite_groupedDecomposition k q M hout hin
  98   obtain ⟨H, hH, hdegrees, _⟩ := regularPartition_of_orientedMultiplicities
  99     G Q (fun _ => q) (by simpa only [hQsum] using hphysical) hQout hQin
 100   exact ⟨H, hH, hdegrees⟩
