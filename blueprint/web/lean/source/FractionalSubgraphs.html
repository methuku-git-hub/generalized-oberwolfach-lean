  13 theorem HasFractionalDecomposition.map_embedding {V W : Type*}
  14     [Fintype V] [DecidableEq V] [Fintype W] [DecidableEq W]
  15     {G : SimpleGraph V} {P : SimpleGraph V → Prop}
  16     (h : HasFractionalDecomposition G P) (f : V ↪ W) :
  17     HasFractionalDecomposition (G.map f)
  18       (fun K => ∃ H,P H ∧ K=H.map f) := by
  19   obtain ⟨w,hw,hs,he⟩ := h
  20   apply fractionalDecomposition_of_indexed _ _ (fun H => H.map f) w hw
  21   · intro H hH
  22     exact ⟨SimpleGraph.map_monotone _ (hs H hH).1,H,(hs H hH).2,rfl⟩
  23   · intro u v
  24     by_cases hu : u∈Set.range f
  25     · obtain ⟨a,rfl⟩ := hu
  26       by_cases hv : v∈Set.range f
  27       · obtain ⟨b,rfl⟩ := hv
  28         simp_rw [SimpleGraph.map_adj_apply]
  29         exact he a b
  30       · have hn (H : SimpleGraph V) : ¬(H.map f).Adj (f a) v := by
  31           intro hadj
  32           rw [SimpleGraph.map_adj] at hadj
  33           obtain ⟨x,y,hxy,hx,hy⟩ := hadj
  34           exact hv ⟨y,hy⟩
  35         simp only [if_neg (hn _),Finset.sum_const_zero]
  36     · have hn (H : SimpleGraph V) : ¬(H.map f).Adj u v := by
  37         intro hadj
  38         rw [SimpleGraph.map_adj] at hadj
  39         obtain ⟨x,y,hxy,hx,hy⟩ := hadj
  40         exact hu ⟨x,hx⟩
  41       simp only [if_neg (hn _),Finset.sum_const_zero]

  43 theorem HasFractionalDecomposition.map_embedded_cycles {V W : Type*}
  44     [Fintype V] [DecidableEq V] [Fintype W] [DecidableEq W]
  45     {G : SimpleGraph V} {r : ℕ}
  46     (h : HasFractionalDecomposition G (fun H => IsEmbeddedCycle H r)) (f : V ↪ W) :
  47     HasFractionalDecomposition (G.map f) (fun H => IsEmbeddedCycle H r) :=
  48   (h.map_embedding f).mono_property (by
  49     rintro K ⟨H,hH,rfl⟩
  50     exact hH.map f)

  52 theorem cycle_fractional_positive_margin_finite (r : ℕ) (hr : 3≤r)
  53     (α : ℝ) (hα : 0<α) :
  54     ∃ N : ℕ,∀ (V : Type) [Fintype V] [DecidableEq V],N≤Fintype.card V →
  55       ∀ G : SimpleGraph V,
  56       (∀ v,(cycleThreshold r+α)*Fintype.card V≤(G.degree v:ℝ)) →
  57       HasFractionalDecomposition G (fun H => IsEmbeddedCycle H r) := by
  58   obtain ⟨N,hN⟩ := cycle_fractional_positive_margin r hr α hα
  59   refine ⟨N,?_⟩
  60   intro V _ _ hn G hdeg
  61   let e := Fintype.equivFin V
  62   have hg : HasFractionalDecomposition (G.map e.toEmbedding)
  63       (fun H => IsEmbeddedCycle H r) := by
  64     apply hN _ hn
  65     intro i
  66     obtain ⟨v,rfl⟩ := e.surjective i
  67     have hd := SimpleGraph.degree_map_apply G e.toEmbedding v
  68     have hh := hdeg v
  69     simp only [← SimpleGraph.ncard_neighborSet] at hd hh ⊢
  70     have hd' : ((G.map e.toEmbedding).neighborSet (e v)).ncard=(G.neighborSet v).ncard := by
  71       simpa only [Equiv.coe_toEmbedding] using hd
  72     rw [hd']
  73     exact hh
  74   have hh := hg.map_cycles e.symm
  75   have he : (G.map e.toEmbedding).map e.symm.toEmbedding=G := by
  76     rw [SimpleGraph.map_map]
  77     have hid : (⇑e.symm.toEmbedding ∘ ⇑e.toEmbedding)=id := by
  78       funext v
  79       exact e.symm_apply_apply v
  80     rw [hid,SimpleGraph.map_id]
  81   rwa [he] at hh
