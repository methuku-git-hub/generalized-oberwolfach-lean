  14 theorem extending_list_embed_cycle (D:V→V→Prop) (L:Finset V) (β:ℝ)
  15     (hext:IsExtendingList D L β) (F:SimpleGraph V) (M:ℝ) (hM:0≤M)
  16     (hdeg:∀ v,(F.degree v:ℝ)≤M) (B:Finset V) (ℓ:ℕ) (hℓ:3≤ℓ)
  17     (hbudget:(B.card:ℝ)+ℓ+2+2*M<β*Fintype.card V):
  18     ∃ xs:List V,xs.length=ℓ ∧ xs.Nodup ∧
  19       (∀ z∈xs,z∈L ∧ z∉B) ∧ DirectedCycleThrough D F xs := by
  20   have hfree:(B.card:ℝ)<β*Fintype.card V:=by nlinarith
  21   obtain ⟨x,hxL,hxB⟩:=extending_list_free_choice D L B β hext hfree
  22   let B₀:=insert x B
  23   have hB₀:(B₀.card:ℝ)≤B.card+1:=by exact_mod_cast Finset.card_insert_le x B
  24   have hchoice:(B₀.card:ℝ)+F.degree x<β*Fintype.card V:=by
  25     have hx:=hdeg x
  26     nlinarith
  27   obtain ⟨y,hyL,hyB,hdxy,hFxy⟩:=extending_list_one_arc_choice D L B₀ β hext F x true hchoice
  28   have hyB':y∉B:=fun h=>hyB (by simp [B₀,h])
  29   have hyx:y≠x:=by intro h;subst y;apply hyB;simp [B₀]
  30   let B':=insert x (insert y B)
  31   have hB':(B'.card:ℝ)≤B.card+2:=by
  32     have h₁:=Finset.card_insert_le x (insert y B)
  33     have h₂:=Finset.card_insert_le y B
  34     have hh:B'.card≤B.card+2:=by dsimp [B'];omega
  35     exact_mod_cast hh
  36   have hlen:ℓ-3+1=ℓ-2:=by omega
  37   have hbudget':(B'.card:ℝ)+((ℓ-3)+1:ℕ)+2+2*M<β*Fintype.card V:=by
  38     have hh:((ℓ-3)+1:ℕ)≤ℓ-2:=by omega
  39     have hhR:(((ℓ-3)+1:ℕ):ℝ)≤((ℓ-2:ℕ):ℝ):=by exact_mod_cast hh
  40     have hsub:((ℓ-2:ℕ):ℝ)+2=ℓ:=by exact_mod_cast Nat.sub_add_cancel (by omega:2≤ℓ)
  41     nlinarith
  42   obtain ⟨zs,hzslen,hzsnodup,hzs,hchain⟩:=extending_list_embed_internal_path D L β hext F M hM hdeg (ℓ-3) B' y x hyx hbudget'
  43   have hzx:x∉zs:=by intro hz;exact (hzs x hz).2.2.2 rfl
  44   have hzy:y∉zs:=by intro hz;exact (hzs y hz).2.2.1 rfl
  45   refine ⟨x::y::zs,?_,?_,?_,?_⟩

  47   · exact List.nodup_cons.mpr ⟨by simp [hyx.symm,hzx],List.nodup_cons.mpr ⟨hzy,hzsnodup⟩⟩
  48   · intro z hz
  49     rcases List.mem_cons.mp hz with rfl | hz
  50     · exact ⟨hxL,hxB⟩
  51     rcases List.mem_cons.mp hz with rfl | hz
  52     · exact ⟨hyL,hyB'⟩
  53     obtain ⟨hzL,hzB',_,_⟩:=hzs z hz
  54     exact ⟨hzL,fun h=>hzB' (by simp [B',h])⟩
  55   · exact ⟨hdxy,hFxy,hchain⟩
