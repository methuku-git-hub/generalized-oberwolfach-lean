  13 def twoPathArcs {V : Type*} [DecidableEq V] (x y z : V) : Finset (V × V) :=
  14   {(x, z), (z, y)}

  16 def packedArcs {I V : Type*} [DecidableEq V]
  17     (S : Finset I) (x y z : I → V) : Finset (V × V) :=
  18   S.biUnion (fun i => twoPathArcs (x i) (y i) (z i))

  20 def middleCount {I V : Type*} [DecidableEq V]
  21     (S : Finset I) (z : I → V) (v : V) : ℕ :=
  22   (S.filter (fun i => z i = v)).card

  24 theorem middleCount_total
  25     {I V : Type*} [Fintype V] [DecidableEq V] (S : Finset I) (z : I → V) :
  26     ∑ v, middleCount S z v = S.card := by
  27   classical
  28   unfold middleCount
  29   simpa using Finset.sum_card_fiberwise_eq_card_filter S Finset.univ z

  31 theorem saturatedMiddles_bound
  32     {I V : Type*} [Fintype V] [DecidableEq V]
  33     (S : Finset I) (z : I → V) (L : ℕ) :
  34     L * (Finset.univ.filter (fun v => L ≤ middleCount S z v)).card ≤ S.card := by
  35   calc
  36     _ = ∑ v ∈ Finset.univ.filter (fun v => L ≤ middleCount S z v), L := by simp [Nat.mul_comm]
  37     _ ≤ ∑ v ∈ Finset.univ.filter (fun v => L ≤ middleCount S z v), middleCount S z v :=
  38       Finset.sum_le_sum (fun v hv => (Finset.mem_filter.mp hv).2)
  39     _ ≤ ∑ v, middleCount S z v := Finset.sum_le_univ_sum_of_nonneg (fun _ => Nat.zero_le _)
  40     _ = S.card := middleCount_total S z

  42 theorem packedArcs_out_bound
  43     {I V : Type*} [Fintype V] [DecidableEq V]
  44     (S : Finset I) (x y z : I → V) (v : V) :
  45     (Finset.univ.filter (fun w => (v, w) ∈ packedArcs S x y z)).card ≤
  46       (S.filter (fun i => x i = v)).card + middleCount S z v := by
  47   classical
  48   have hsub : Finset.univ.filter (fun w => (v, w) ∈ packedArcs S x y z) ⊆
  49       ((S.filter (fun i => x i = v)).image z) ∪
  50         ((S.filter (fun i => z i = v)).image y) := by
  51     intro w hw
  52     obtain ⟨i, hi, hp⟩ := Finset.mem_biUnion.mp (Finset.mem_filter.mp hw).2
  53     simp only [twoPathArcs, Finset.mem_insert, Finset.mem_singleton,
  54       Prod.mk.injEq] at hp
  55     rcases hp with ⟨hx, hz⟩ | ⟨hz, hy⟩
  56     · exact Finset.mem_union_left _ (Finset.mem_image.mpr
  57         ⟨i, Finset.mem_filter.mpr ⟨hi, hx.symm⟩, hz.symm⟩)
  58     · exact Finset.mem_union_right _ (Finset.mem_image.mpr
  59         ⟨i, Finset.mem_filter.mpr ⟨hi, hz.symm⟩, hy.symm⟩)
  60   exact (Finset.card_le_card hsub).trans
  61     ((Finset.card_union_le _ _).trans (Nat.add_le_add (Finset.card_image_le) (Finset.card_image_le)))

  84 /-- One next path can be chosen from the fixed candidate set. Already used
  85 arcs, saturated middles and the two forbidden-neighbourhood sets are all
  86 removed by their realised cardinalities. -/
  87 theorem twoPathPacking_next
  88     {I V : Type*} [Fintype V] [DecidableEq V]
  89     (S : Finset I) (x y z : I → V) (u v : V) (C : Finset V)
  90     (Fu Fv : Finset V) (L : ℕ) (hL : 0 < L) {a f : ℝ}
  91     (hx : ((S.filter (fun i => x i = u)).card : ℝ) ≤ a)
  92     (hy : ((S.filter (fun i => y i = v)).card : ℝ) ≤ a)
  93     (hmiddle : ∀ w, middleCount S z w ≤ L)
  94     (hFu : (Fu.card : ℝ) ≤ f) (hFv : (Fv.card : ℝ) ≤ f)
  95     (hlarge : (S.card : ℝ) / L + 2 * a + 2 * L + 2 * f < C.card) :
  96     ∃ w ∈ C, middleCount S z w < L ∧
  97       (u, w) ∉ packedArcs S x y z ∧ (w, v) ∉ packedArcs S x y z ∧
  98       w ∉ Fu ∧ w ∉ Fv := by
  99   classical
 100   let Sat := Finset.univ.filter (fun w => L ≤ middleCount S z w)
 101   let Out := Finset.univ.filter (fun w => (u, w) ∈ packedArcs S x y z)
 102   let In := Finset.univ.filter (fun w => (w, v) ∈ packedArcs S x y z)
 103   let Bad := (((Sat ∪ Out) ∪ In) ∪ Fu) ∪ Fv
 104   have hSat : (Sat.card : ℝ) ≤ (S.card : ℝ) / L := by
 105     apply (le_div_iff₀ (Nat.cast_pos.mpr hL)).mpr
 106     have hh := saturatedMiddles_bound S z L
 107     exact_mod_cast (by simpa [Sat, Nat.mul_comm] using hh)
 108   have hOut : (Out.card : ℝ) ≤ a + L := by
 109     have hh := packedArcs_out_bound S x y z u
 110     have hm := hmiddle u
 111     have hr : (Out.card : ℝ) ≤
 112         ((S.filter (fun i => x i = u)).card : ℝ) + middleCount S z u := by
 113       exact_mod_cast hh
 114     have hmr : (middleCount S z u : ℝ) ≤ L := by exact_mod_cast hm
 115     linarith
 116   have hIn : (In.card : ℝ) ≤ a + L := by
 117     have hh := packedArcs_in_bound S x y z v
 118     have hm := hmiddle v
 119     have hr : (In.card : ℝ) ≤
 120         ((S.filter (fun i => y i = v)).card : ℝ) + middleCount S z v := by
 121       exact_mod_cast hh
 122     have hmr : (middleCount S z v : ℝ) ≤ L := by exact_mod_cast hm
 123     linarith
 124   have hBadNat : Bad.card ≤ Sat.card + Out.card + In.card + Fu.card + Fv.card := by
 125     exact (Finset.card_union_le _ _).trans (Nat.add_le_add_right
 126       ((Finset.card_union_le _ _).trans (Nat.add_le_add_right
 127         ((Finset.card_union_le _ _).trans (Nat.add_le_add_right
 128           (Finset.card_union_le _ _) _)) _)) _)
 129   have hBad : (Bad.card : ℝ) < C.card := by
 130     have hh : (Bad.card : ℝ) ≤
 131         (Sat.card : ℝ) + Out.card + In.card + Fu.card + Fv.card := by
 132       exact_mod_cast hBadNat
 133     linarith
 134   have hnot : ¬ C ⊆ Bad := by
 135     intro hsub
 136     have hh : (C.card : ℝ) ≤ Bad.card := by exact_mod_cast Finset.card_le_card hsub
 137     linarith
 138   obtain ⟨w, hw, hbad⟩ := Finset.not_subset.mp hnot
 139   have hh : (((middleCount S z w < L ∧ (u, w) ∉ packedArcs S x y z) ∧
 140       (w, v) ∉ packedArcs S x y z) ∧ w ∉ Fu) ∧ w ∉ Fv := by
 141     simpa only [Bad, Sat, Out, In, Finset.mem_union, Finset.mem_filter,
 142       Finset.mem_univ, true_and, not_or, not_le] using hbad
 143   exact ⟨w, hw, hh.1.1.1.1, hh.1.1.1.2, hh.1.1.2, hh.1.2, hh.2⟩

 147 theorem twoPathPacking_exists
 148     {I V : Type*} [Fintype I] [DecidableEq I]
 149     [Fintype V] [DecidableEq V] [Nonempty V]
 150     (x y : I → V) (C : I → Finset V) (F : V → Finset V)
 151     (L : ℕ) (hL : 0 < L) {a f : ℝ}
 152     (hx : ∀ v, ((Finset.univ.filter (fun i => x i = v)).card : ℝ) ≤ a)
 153     (hy : ∀ v, ((Finset.univ.filter (fun i => y i = v)).card : ℝ) ≤ a)
 154     (hF : ∀ v, ((F v).card : ℝ) ≤ f)
 155     (hlarge : ∀ i, (Fintype.card I : ℝ) / L + 2 * a + 2 * L + 2 * f < (C i).card) :
 156     ∃ z : I → V,
 157       (∀ i, z i ∈ C i ∧ z i ∉ F (x i) ∧ z i ∉ F (y i)) ∧
 158       (∀ i j, i ≠ j → Disjoint (twoPathArcs (x i) (y i) (z i))
 159         (twoPathArcs (x j) (y j) (z j))) ∧
 160       ∀ v, middleCount Finset.univ z v ≤ L := by
 161   classical
 162   have hall (S : Finset I) : ∃ z : I → V,
 163       (∀ i ∈ S, z i ∈ C i ∧ z i ∉ F (x i) ∧ z i ∉ F (y i)) ∧
 164       (∀ i ∈ S, ∀ j ∈ S, i ≠ j → Disjoint (twoPathArcs (x i) (y i) (z i))
 165         (twoPathArcs (x j) (y j) (z j))) ∧
 166       ∀ v, middleCount S z v ≤ L := by
 167     induction S using Finset.induction_on with
 168     | empty =>
 169       exact ⟨fun _ => Classical.choice inferInstance, by simp, by simp,
 170         fun v => by simp [middleCount]⟩
 171     | @insert i S hi ih =>
 172       obtain ⟨z, hz, hdisj, hmiddle⟩ := ih
 173       have hxp : ((S.filter (fun j => x j = x i)).card : ℝ) ≤ a := by
 174         apply le_trans _ (hx (x i))
 175         exact_mod_cast Finset.card_le_card
 176           (Finset.filter_subset_filter _ (Finset.subset_univ S))
 177       have hyp : ((S.filter (fun j => y j = y i)).card : ℝ) ≤ a := by
 178         apply le_trans _ (hy (y i))
 179         exact_mod_cast Finset.card_le_card
 180           (Finset.filter_subset_filter _ (Finset.subset_univ S))
 181       have hcp : (S.card : ℝ) / L + 2 * a + 2 * L + 2 * f < (C i).card := by
 182         have hc : (S.card : ℝ) ≤ Fintype.card I := by exact_mod_cast Finset.card_le_univ S
 183         have hd := div_le_div_of_nonneg_right hc (Nat.cast_nonneg L)
 184         linarith [hlarge i]
 185       obtain ⟨w, hw, hsat, hout, hin, hfu, hfv⟩ :=
 186         twoPathPacking_next S x y z (x i) (y i) (C i) (F (x i)) (F (y i))
 187           L hL hxp hyp hmiddle (hF (x i)) (hF (y i)) hcp
 188       let z' := Function.update z i w
 189       have heq (j : I) (hj : j ∈ S) : z' j = z j := by
 190         exact Function.update_of_ne (by intro hji; subst j; exact hi hj) _ _
 191       have hnew : z' i = w := Function.update_self _ _ _
 192       have hn (j : I) (hj : j ∈ S) :
 193           Disjoint (twoPathArcs (x i) (y i) w) (twoPathArcs (x j) (y j) (z j)) := by
 194         apply Finset.disjoint_left.mpr
 195         intro e he hej
 196         have hp : e ∈ packedArcs S x y z := Finset.mem_biUnion.mpr ⟨j, hj, hej⟩
 197         simp only [twoPathArcs, Finset.mem_insert, Finset.mem_singleton] at he
 198         rcases he with he | he
 199         · exact hout (he ▸ hp)
 200         · exact hin (he ▸ hp)
 201       have hcount (v : V) : middleCount (insert i S) z' v =
 202           (if w = v then 1 else 0) + middleCount S z v := by
 203         have hfilter : S.filter (fun j => z' j = v) = S.filter (fun j => z j = v) := by
 204           apply Finset.filter_congr
 205           intro j hj
 206           rw [heq j hj]
 207         unfold middleCount
 208         rw [Finset.filter_insert, hfilter, hnew]
 209         by_cases hwv : w = v <;> simp [hwv, hi, Nat.add_comm]
 210       refine ⟨z', ?_, ?_, ?_⟩
 211       · intro j hj
 212         rcases Finset.mem_insert.mp hj with rfl | hj
 213         · simpa only [hnew] using And.intro hw (And.intro hfu hfv)
 214         · simpa only [heq j hj] using hz j hj
 215       · intro j hj k hk hjk
 216         by_cases hji : j = i
 217         · subst j
 218           have hkS : k ∈ S := (Finset.mem_insert.mp hk).resolve_left (Ne.symm hjk)
 219           rw [hnew, heq k hkS]
 220           exact hn k hkS
 221         · have hjS : j ∈ S := (Finset.mem_insert.mp hj).resolve_left hji
 222           by_cases hki : k = i
 223           · subst k
 224             rw [heq j hjS, hnew]
 225             exact (hn j hjS).symm
 226           · have hkS : k ∈ S := (Finset.mem_insert.mp hk).resolve_left hki
 227             rw [heq j hjS, heq k hkS]
 228             exact hdisj j hjS k hkS hjk
 229       · intro v
 230         rw [hcount]
 231         by_cases hwv : w = v
 232         · subst v
 233           simp only [ite_true]
 234           omega
 235         · simp only [hwv, ite_false, zero_add]
 236           exact hmiddle v
 237   obtain ⟨z, hz, hdisj, hmiddle⟩ := hall Finset.univ
 238   exact ⟨z, fun i => hz i (Finset.mem_univ i),
 239     fun i j h => hdisj i (Finset.mem_univ i) j (Finset.mem_univ j) h, hmiddle⟩

 245 theorem twoPathMultiplicity_le_one
 246     {I V : Type*} [Fintype I] [DecidableEq V]
 247     (x y z : I → V)
 248     (hdisj : ∀ i j, i ≠ j → Disjoint (twoPathArcs (x i) (y i) (z i))
 249       (twoPathArcs (x j) (y j) (z j))) (u v : V) :
 250     twoPathMultiplicity x y z u v ≤ 1 := by
 251   classical

 254   by_contra hij
 255   exact Finset.disjoint_left.mp (hdisj i j hij)

 258 theorem twoPathMultiplicity_supported
 259     {I V : Type*} [Fintype I] [DecidableEq V]
 260     (x y z : I → V) (M : V → V → ℕ)
 261     (hdisj : ∀ i j, i ≠ j → Disjoint (twoPathArcs (x i) (y i) (z i))
 262       (twoPathArcs (x j) (y j) (z j)))
 263     (hpath : ∀ i, 0 < M (x i) (z i) ∧ 0 < M (z i) (y i)) :
 264     ∀ u v, twoPathMultiplicity x y z u v ≤ M u v := by
 265   intro u v
 266   by_cases ht : twoPathMultiplicity x y z u v = 0
 267   · simp [ht]
 268   · have hp : 0 < twoPathMultiplicity x y z u v := Nat.pos_of_ne_zero ht
 269     obtain ⟨i, hi⟩ := Finset.card_pos.mp hp
 270     have he := (Finset.mem_filter.mp hi).2
 271     simp only [twoPathArcs, Finset.mem_insert, Finset.mem_singleton, Prod.mk.injEq] at he
 272     have hm : 0 < M u v := by
 273       rcases he with ⟨rfl, rfl⟩ | ⟨rfl, rfl⟩
 274       · exact (hpath i).1
 275       · exact (hpath i).2
 276     exact (twoPathMultiplicity_le_one x y z hdisj u v).trans hm

 278 theorem twoPathMultiplicity_degrees
 279     {I V : Type*} [Fintype I] [Fintype V] [DecidableEq V]
 280     (x y z : I → V) (hxz : ∀ i, x i ≠ z i) (hzy : ∀ i, z i ≠ y i) :
 281     (∀ u, (∑ v, twoPathMultiplicity x y z u v) =
 282       (Finset.univ.filter (fun i => x i = u)).card + middleCount Finset.univ z u) ∧
 283     (∀ v, (∑ u, twoPathMultiplicity x y z u v) =
 284       (Finset.univ.filter (fun i => y i = v)).card + middleCount Finset.univ z v) := by
 285   classical
 286   have hind (i : I) (u v : V) :
 287       (if (u, v) ∈ twoPathArcs (x i) (y i) (z i) then 1 else 0 : ℕ) =
 288       (if u = x i ∧ v = z i then 1 else 0) +
 289       (if u = z i ∧ v = y i then 1 else 0) := by
 290     simp only [twoPathArcs, Finset.mem_insert, Finset.mem_singleton, Prod.mk.injEq]
 291     by_cases h₁ : u = x i ∧ v = z i
 292     · have h₂ : ¬ (u = z i ∧ v = y i) := by
 293         rintro ⟨hu, _⟩
 294         exact hxz i (h₁.1.symm.trans hu)
 295       simp [h₁, hxz i, hzy i]
 296     · simp [h₁]
 297   constructor
 298   · intro u
 299     simp only [twoPathMultiplicity, Finset.card_filter]
 300     rw [Finset.sum_comm]
 301     calc
 302       _ = ∑ i, ((if x i = u then 1 else 0) + (if z i = u then 1 else 0)) := by
 303         apply Finset.sum_congr rfl
 304         intro i _
 305         simp_rw [hind, Finset.sum_add_distrib]
 306         by_cases h₁ : u = x i <;> by_cases h₂ : u = z i <;>
 307           simp [h₁, h₂, eq_comm, hxz i]
 308       _ = _ := by simp only [middleCount, Finset.card_filter, Finset.sum_add_distrib]
 309   · intro v
 310     simp only [twoPathMultiplicity, Finset.card_filter]
 311     rw [Finset.sum_comm]
 312     calc
 313       _ = ∑ i, ((if y i = v then 1 else 0) + (if z i = v then 1 else 0)) := by
 314         apply Finset.sum_congr rfl
 315         intro i _
 316         simp_rw [hind, Finset.sum_add_distrib]
 317         by_cases h₁ : v = z i <;> by_cases h₂ : v = y i <;>
 318           simp [h₁, h₂, eq_comm, Ne.symm (hzy i)]
 319       _ = _ := by simp only [middleCount, Finset.card_filter, Finset.sum_add_distrib]
