  29 theorem ordinary_comparison_from_divisibility {V U : Type}
  30     [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U]
  31     (G : SimpleGraph V) (H : V → U → Prop) (r : ℕ) (hr : 3≤r)
  32     (hbal : ∀ v,G.degree v=2*(Finset.univ.filter (H v)).card)
  33     (hdiv : ∀ u,r∣(Finset.univ.filter (fun v => H v u)).card) :
  34     let k := G.edgeFinset.card/r
  35     let Q := ordinaryComparisonRim r k (by omega)
  36     let K := ordinaryComparisonSpokes r k (by omega)
  37     HasOrdinaryWheelDecomposition Q K r ∧
  38     (∀ a,Q.degree a=2*(Finset.univ.filter (K a)).card) ∧
  39     (∀ b,r∣(Finset.univ.filter (fun a => K a b)).card) ∧
  40     Q.edgeFinset.card=G.edgeFinset.card ∧
  41     Fintype.card {e:OrdinaryComparisonRows r k×OrdinaryComparisonHubs k // K e.1 e.2}=
  42       Fintype.card {e:V×U // H e.1 e.2} := by
  43   dsimp only
  44   let k := G.edgeFinset.card/r
  45   have hkr : k*r=G.edgeFinset.card := Nat.div_mul_cancel (ordinary_balanced_rim_total_divisible G H r hbal hdiv)
  46   have hcycle : IsEmbeddedCycle (SimpleGraph.cycleGraph r) r := by
  47     refine ⟨hr,Function.Embedding.refl _,?_⟩
  48     change SimpleGraph.cycleGraph r=(SimpleGraph.cycleGraph r).map id
  49     rw [SimpleGraph.map_id]
  50   have hfull : ∀ w:Fin r,w∈(SimpleGraph.cycleGraph r).support := by
  51     intro w
  52     rw [cycleGraph_support_eq_univ hr]
  53     exact Set.mem_univ _
  54   obtain ⟨hdec,hb,hd,he,hs⟩ := disjoint_wheel_comparison (I:=Fin k) (SimpleGraph.cycleGraph r) r hcycle
  55     ⟨0,by omega⟩ hfull (Fintype.card_fin r)
  56   have he' : (ordinaryComparisonRim r k (by omega)).edgeFinset.card=G.edgeFinset.card := by
  57     simpa only [ordinaryComparisonRim,OrdinaryComparisonRows,Fintype.card_fin,hkr] using he
  58   have hs' : Fintype.card {e:OrdinaryComparisonRows r k×OrdinaryComparisonHubs k //
  59       ordinaryComparisonSpokes r k (by omega) e.1 e.2}=Fintype.card {e:V×U // H e.1 e.2} := by
  60     calc
  61       _ = k*r := by simpa only [ordinaryComparisonSpokes,OrdinaryComparisonRows,OrdinaryComparisonHubs,Fintype.card_fin] using hs
  62       _ = G.edgeFinset.card := hkr
  63       _ = _ := ordinary_balanced_total_edges G H hbal
  64   exact ⟨hdec,hb,hd,he',hs'⟩
