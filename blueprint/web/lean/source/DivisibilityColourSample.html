  15 theorem divisibility_colour_common_event {V : Type*} [Fintype V] [DecidableEq V]
  16     {R : ℕ} (G : SimpleGraph V) (ρ : Fin R → ℝ) (hρ : ∀ a,0≤ρ a)
  17     (hsum : ∑ a,ρ a=1) (b : Fin R) (η : ℝ) (hη : 0≤η) (hn : 0<Fintype.card V) :
  18     1-4*R*(Fintype.card V:ℝ)^2*Real.exp (-η^2*Fintype.card V)≤
  19       (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.categorical ρ hρ hsum)).probability
  20         (fun σ => (∀ a v,|(sampledColourDegree G σ a v:ℝ)-ρ a*G.degree v|<η*Fintype.card V) ∧
  21           (∀ a u v,u≠v → |(colouredTwoPathCount G σ u v a b:ℝ)-
  22             ((G.neighborFinset u∩G.neighborFinset v).card:ℝ)*ρ a*ρ b|<η*Fintype.card V)) := by
  23   let n : ℝ := Fintype.card V
  24   have hnp : 0<n := Nat.cast_pos.mpr hn
  25   have hn1 : 1≤n := by
  26     dsimp only [n]
  27     exact_mod_cast (show 1≤Fintype.card V by omega)
  28   let μ := FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.categorical ρ hρ hsum)
  29   let J := (Fin R×V)⊕(Fin R×{p : V×V // p.1≠p.2})
  30   let X (j : J) (σ : Sym2 V → Fin R) : ℝ :=
  31     match j with
  32     | Sum.inl (a,v) => (sampledColourDegree G σ a v:ℝ)-ρ a*G.degree v
  33     | Sum.inr (a,p) => (colouredTwoPathCount G σ p.1.1 p.1.2 a b:ℝ)-
  34         ((G.neighborFinset p.1.1∩G.neighborFinset p.1.2).card:ℝ)*ρ a*ρ b
  35   let B := 2*Real.exp (-η^2*n)
  36   have hbad (j : J) : μ.probability (fun σ => η*n≤|X j σ|)≤B := by
  37     rcases j with ⟨a,v⟩ | ⟨a,p⟩
  38     · have hh := sampledColourDegree_tail G ρ hρ hsum a v (t:=η*n) (by positivity)
  39       have he : -2*(η*n)^2/n = -2*η^2*n := by field_simp <;> ring
  40       change μ.probability (fun σ => η*n≤|X (Sum.inl (a,v)) σ|)≤2*Real.exp (-2*(η*n)^2/n) at hh
  41       rw [he] at hh
  42       apply hh.trans
  43       apply mul_le_mul_of_nonneg_left _ (by norm_num)
  44       apply Real.exp_le_exp.mpr
  45       have hh0 : 0≤η^2*n := by positivity
  46       nlinarith
  47     · have hh := colouredTwoPathCount_tail G ρ hρ hsum p.1.1 p.1.2 p.2 a b (t:=η*n) (by positivity)
  48       have he : -(η*n)^2/n = -η^2*n := by field_simp <;> ring
  49       change μ.probability (fun σ => η*n≤|X (Sum.inr (a,p)) σ|)≤2*Real.exp (-(η*n)^2/n) at hh
  50       rw [he] at hh
  51       exact hh
  52   have hcard : (Fintype.card J:ℝ)≤2*R*n^2 := by
  53     have hpair : (Fintype.card {p : V×V // p.1≠p.2}:ℝ)≤n^2 := by
  54       have hh := Fintype.card_subtype_le (fun p : V×V => p.1≠p.2)
  55       simpa only [Fintype.card_prod,Nat.cast_mul,n,pow_two] using (Nat.cast_le (α:=ℝ)).mpr hh
  56     have hR : (0:ℝ)≤R := Nat.cast_nonneg _
  57     have hh := mul_le_mul_of_nonneg_left hpair hR
  58     simp only [J,Fintype.card_sum,Fintype.card_prod,Fintype.card_fin,Nat.cast_add,Nat.cast_mul]
  59     change (R:ℝ)*n+R*(Fintype.card {p : V×V // p.1≠p.2}:ℝ)≤2*R*n^2
  60     have hh2 := mul_le_mul_of_nonneg_left (show n≤n^2 by nlinarith) hR
  61     nlinarith
  62   have hbound : (∑ _ : J,B)≤4*R*n^2*Real.exp (-η^2*n) := by
  63     simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul]
  64     have hh := mul_le_mul_of_nonneg_right hcard (show 0≤B by dsimp [B];positivity)
  65     dsimp only [B] at *
  66     nlinarith
  67   have hh := finiteLaw_common_event_lower μ Finset.univ (fun j σ => η*n≤|X j σ|)
  68     (fun _ => B) (fun j _ => hbad j)
  69   apply (sub_le_sub_left hbound 1).trans (hh.trans _)
  70   apply finiteLaw_probability_mono
  71   intro σ hσ
  72   constructor
  73   · intro a v
  74     exact not_le.mp (hσ (Sum.inl (a,v)) (Finset.mem_univ _))
  75   · intro a u v huv
  76     exact not_le.mp (hσ (Sum.inr (a,⟨(u,v),huv⟩)) (Finset.mem_univ _))

  78 theorem divisibility_colour_sample (R : ℕ) (η : ℝ) (hη : 0<η) :
  79     ∃ N : ℕ,∀ (V : Type u) [Fintype V] [DecidableEq V],N≤Fintype.card V →
  80     ∀ (G : SimpleGraph V) (ρ : Fin R → ℝ) (hρ : ∀ a,0≤ρ a) (hsum : ∑ a,ρ a=1)
  81       (b : Fin R),∃ σ : Sym2 V → Fin R,
  82       (∀ a v,|(sampledColourDegree G σ a v:ℝ)-ρ a*G.degree v|<η*Fintype.card V) ∧
  83       (∀ a u v,u≠v → |(colouredTwoPathCount G σ u v a b:ℝ)-
  84         ((G.neighborFinset u∩G.neighborFinset v).card:ℝ)*ρ a*ρ b|<η*Fintype.card V) := by
  85   obtain ⟨N,hN⟩ := eventually_stretchedExp_le_inv_pow (4*R) 2 (c:=η^2) (p:=1)
  86     (sq_pos_of_pos hη) (by norm_num) 1
  87   refine ⟨max N 2,?_⟩
  88   intro V instV instDec hn G ρ hρ hsum b
  89   have hn2 : 2≤Fintype.card V := (le_max_right _ _).trans hn
  90   have hn2r : (2:ℝ)≤Fintype.card V := by exact_mod_cast hn2
  91   have hh := divisibility_colour_common_event G ρ hρ hsum b η hη.le (by omega)
  92   have hb := hN (Fintype.card V) ((le_max_left _ _).trans hn)
  93   simp only [Real.rpow_two,Real.rpow_one,pow_one] at hb
  94   have hp : 0<1-4*R*(Fintype.card V:ℝ)^2*Real.exp (-η^2*Fintype.card V) := by
  95     have hi : 1/(Fintype.card V:ℝ)<1 := (div_lt_one (by positivity)).mpr (by linarith)
  96     linarith
  97   have hprob := hp.trans_le hh
  98   by_contra hnone
  99   have hall (σ : Sym2 V → Fin R) : ¬((∀ a v,|(sampledColourDegree G σ a v:ℝ)-ρ a*G.degree v|<η*Fintype.card V) ∧
 100       (∀ a u v,u≠v → |(colouredTwoPathCount G σ u v a b:ℝ)-
 101         ((G.neighborFinset u∩G.neighborFinset v).card:ℝ)*ρ a*ρ b|<η*Fintype.card V)) :=
 102     fun h => hnone ⟨σ,h⟩
 103   simp only [FiniteLaw.probability,hall,ite_false,Finset.sum_const_zero] at hprob
 104   exact (lt_irrefl 0) hprob
