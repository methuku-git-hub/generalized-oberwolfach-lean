  17 theorem finiteLaw_supported_avoidance_independent
  18     (μ : ∀ e,FiniteLaw (State e)) (A : I → (∀ e,State e) → Prop)
  19     (S : I → Finset E)
  20     (hs : ∀ i σ τ,(∀ e∈S i,σ e=τ e) → (A i σ ↔ A i τ))
  21     (i : I) (T : Finset I) (hd : ∀ j∈T,Disjoint (S i) (S j)) :
  22     (FiniteLaw.pi μ).probability (fun σ=>A i σ ∧ avoidsEvents A T σ)=
  23       (FiniteLaw.pi μ).probability (A i)*
  24         (FiniteLaw.pi μ).probability (avoidsEvents A T) := by
  25   apply finiteLaw_disjoint_supported_events μ (S i) (T.biUnion S)
  26   · apply Finset.disjoint_left.mpr
  27     intro e he ht
  28     obtain ⟨j,hj,hej⟩ := Finset.mem_biUnion.mp ht
  29     exact Finset.disjoint_left.mp (hd j hj) he hej
  30   · exact hs i
  31   · intro σ τ he
  32     apply forall_congr'
  33     intro j
  34     apply forall_congr'
  35     intro hj
  36     exact not_congr (hs j σ τ (fun e hej=>he e (Finset.mem_biUnion.mpr ⟨j,hj,hej⟩)))

  38 theorem finiteLaw_coordinate_localLemma
  39     (μ : ∀ e,FiniteLaw (State e)) (A : I → (∀ e,State e) → Prop)
  40     (S : I → Finset E)
  41     (hs : ∀ i σ τ,(∀ e∈S i,σ e=τ e) → (A i σ ↔ A i τ))
  42     (N : I → Finset I)
  43     (hN : ∀ i j,i≠j → j∉N i → Disjoint (S i) (S j))
  44     (x : I → ℝ) (hx0 : ∀ i,0≤x i) (hx1 : ∀ i,x i<1)
  45     (hsmall : ∀ i,(FiniteLaw.pi μ).probability (A i)≤x i*∏ j∈N i,(1-x j)) :
  46     0<(FiniteLaw.pi μ).probability (fun σ=>∀ i,¬A i σ) := by
  47   have hind : ∀ i T,i∉T → Disjoint T (N i) →
  48       (FiniteLaw.pi μ).probability (fun σ=>A i σ ∧ avoidsEvents A T σ)=
  49         (FiniteLaw.pi μ).probability (A i)*
  50           (FiniteLaw.pi μ).probability (avoidsEvents A T) := by
  51     intro i T hi hdis
  52     apply finiteLaw_supported_avoidance_independent μ A S hs
  53     intro j hj
  54     exact hN i j (fun he=>hi (he ▸ hj))
  55       (fun hjN=>Finset.disjoint_left.mp hdis hj hjN)
  56   have he : avoidsEvents A Finset.univ=(fun σ=>∀ i,¬A i σ) := by
  57     funext σ
  58     exact propext (by simp [avoidsEvents])
  59   rw [← he]
  60   exact finiteLaw_localLemma (FiniteLaw.pi μ) A N x hx0 hx1 hind hsmall Finset.univ

  62 theorem finiteLaw_positive_event_exists
  63     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (A : Ω → Prop)
  64     (h : 0<μ.probability A) : ∃ ω,0<μ.mass ω ∧ A ω := by
  65   have hp : 0<∑ ω,if A ω then μ.mass ω else 0 := h
  66   obtain ⟨ω,_,hω⟩ := (Finset.sum_pos_iff_of_nonneg
  67     (fun ω (_ : ω∈Finset.univ)=>show 0≤(if A ω then μ.mass ω else 0) from
  68       by split_ifs;exact μ.nonneg ω;exact le_refl _)).mp hp
  69   by_cases ha : A ω
  70   · exact ⟨ω,by simpa only [ha,ite_true] using hω,ha⟩
  71   · simp only [ha,ite_false,lt_self_iff_false] at hω
