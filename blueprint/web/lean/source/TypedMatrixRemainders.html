  11 def TypedMatrixBalanced {V U : Type*} [Fintype V] [Fintype U]
  12     (q : ℕ) (R₀ Rₖ : V → V → ℕ) (H₀ Hₖ : V → U → ℕ) : Prop :=
  13   (∀ v,(∑ x,R₀ v x)+(∑ x,Rₖ v x)=(∑ u,H₀ v u)+(∑ u,Hₖ v u)) ∧
  14   (∀ v,(∑ x,R₀ x v)+(∑ x,Rₖ x v)=(∑ u,H₀ v u)+(∑ u,Hₖ v u)) ∧
  15   (∀ v,(∑ x,Rₖ x v)=(∑ u,Hₖ v u)) ∧
  16   (∀ u,(∑ v,H₀ v u)=q*(∑ v,Hₖ v u))

  18 theorem int_sum_tsub {I : Type*} [Fintype I] (f g : I → ℕ)
  19     (h : ∀ i,g i≤f i) :
  20     (∑ i,((f i-g i : ℕ):ℤ))=(∑ i,(f i : ℤ))-(∑ i,(g i : ℤ)) := by
  21   simp only [Int.ofNat_sub (h _),Finset.sum_sub_distrib]

  33 theorem typed_matrix_remainder_balanced {V U : Type*} [Fintype V] [Fintype U]
  34     (q : ℕ) (R₀ Rₖ J₀ Jₖ : V → V → ℕ) (H₀ Hₖ L₀ Lₖ : V → U → ℕ)
  35     (hJ0 : ∀ x y,J₀ x y≤R₀ x y) (hJK : ∀ x y,Jₖ x y≤Rₖ x y)
  36     (hL0 : ∀ v u,L₀ v u≤H₀ v u) (hLK : ∀ v u,Lₖ v u≤Hₖ v u)
  37     (hp : ∀ v,(∑ x,(J₀ v x : ℤ))+(∑ x,(Jₖ v x : ℤ))-
  38       (∑ u,(L₀ v u : ℤ))-(∑ u,(Lₖ v u : ℤ))=
  39       (∑ x,(R₀ v x : ℤ))+(∑ x,(Rₖ v x : ℤ))-
  40       (∑ u,(H₀ v u : ℤ))-(∑ u,(Hₖ v u : ℤ)))
  41     (hm : ∀ v,(∑ x,(J₀ x v : ℤ))+(∑ x,(Jₖ x v : ℤ))-
  42       (∑ u,(L₀ v u : ℤ))-(∑ u,(Lₖ v u : ℤ))=
  43       (∑ x,(R₀ x v : ℤ))+(∑ x,(Rₖ x v : ℤ))-
  44       (∑ u,(H₀ v u : ℤ))-(∑ u,(Hₖ v u : ℤ)))
  45     (hk : ∀ v,(∑ x,(Jₖ x v : ℤ))-(∑ u,(Lₖ v u : ℤ))=
  46       (∑ x,(Rₖ x v : ℤ))-(∑ u,(Hₖ v u : ℤ)))
  47     (hw : ∀ u,(∑ v,(L₀ v u : ℤ))-(q:ℤ)*(∑ v,(Lₖ v u : ℤ))=
  48       (∑ v,(H₀ v u : ℤ))-(q:ℤ)*(∑ v,(Hₖ v u : ℤ))) :
  49     TypedMatrixBalanced q (fun x y=>R₀ x y-J₀ x y) (fun x y=>Rₖ x y-Jₖ x y)
  50       (fun v u=>H₀ v u-L₀ v u) (fun v u=>Hₖ v u-Lₖ v u) := by
  51   refine ⟨?_,?_,?_,?_⟩
  52   · intro v
  53     have hh : (∑ x,((R₀ v x-J₀ v x : ℕ):ℤ))+(∑ x,((Rₖ v x-Jₖ v x : ℕ):ℤ))=
  54         (∑ u,((H₀ v u-L₀ v u : ℕ):ℤ))+(∑ u,((Hₖ v u-Lₖ v u : ℕ):ℤ)) := by
  55       rw [int_sum_tsub _ _ (hJ0 v),int_sum_tsub _ _ (hJK v),
  56         int_sum_tsub _ _ (hL0 v),int_sum_tsub _ _ (hLK v)]
  57       linarith only [hp v]
  58     exact_mod_cast hh
  59   · intro v
  60     have hh : (∑ x,((R₀ x v-J₀ x v : ℕ):ℤ))+(∑ x,((Rₖ x v-Jₖ x v : ℕ):ℤ))=
  61         (∑ u,((H₀ v u-L₀ v u : ℕ):ℤ))+(∑ u,((Hₖ v u-Lₖ v u : ℕ):ℤ)) := by
  62       rw [int_sum_tsub _ _ (fun x=>hJ0 x v),int_sum_tsub _ _ (fun x=>hJK x v),
  63         int_sum_tsub _ _ (hL0 v),int_sum_tsub _ _ (hLK v)]
  64       linarith only [hm v]
  65     exact_mod_cast hh
  66   · intro v
  67     have hh : (∑ x,((Rₖ x v-Jₖ x v : ℕ):ℤ))=(∑ u,((Hₖ v u-Lₖ v u : ℕ):ℤ)) := by
  68       rw [int_sum_tsub _ _ (fun x=>hJK x v),int_sum_tsub _ _ (hLK v)]
  69       linarith only [hk v]
  70     exact_mod_cast hh
  71   · intro u
  72     have hh : (∑ v,((H₀ v u-L₀ v u : ℕ):ℤ))=(q:ℤ)*(∑ v,((Hₖ v u-Lₖ v u : ℕ):ℤ)) := by
  73       rw [int_sum_tsub _ _ (fun v=>hL0 v u),int_sum_tsub _ _ (fun v=>hLK v u)]
  74       nlinarith only [hw u]
