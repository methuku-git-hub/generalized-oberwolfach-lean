  14 theorem finite_weight_discard_identity {I : Type*} [Fintype I]
  15     (weight : I → ℝ) (base good : I → Prop) :
  16     (∑ i,if base i then weight i else 0)-
  17       (∑ i,if base i ∧ good i then weight i else 0)=
  18       ∑ i,if base i ∧ ¬good i then weight i else 0 := by
  19   rw [← Finset.sum_sub_distrib]
  20   apply Finset.sum_congr rfl
  21   intro i _
  22   by_cases hb : base i <;> by_cases hg : good i <;> simp [hb,hg]

  32 theorem ordinary_admissible_cycle_edge_load (x y : V) (hxy : P.Adj x y) (hFxy : ¬F.Adj x y) :
  33     |(∑ C,if C.val.val.Adj x y ∧ (∀ v∈C.val.val.support,∀ z∈C.val.val.support,¬F.Adj v z)
  34       then weight C else 0)-1|≤5*(r:ℝ)^4*B*L/Fintype.card V := by
  35   let good := fun C : OrdinaryCycle P r=>∀ v∈C.val.val.support,∀ z∈C.val.val.support,¬F.Adj v z
  36   have hnot (C : OrdinaryCycle P r) : ¬good C↔∃ v∈C.val.val.support,∃ z∈C.val.val.support,F.Adj v z := by
  37     simp only [good,not_forall,not_imp,not_not,exists_prop]
  38   have hid := finite_weight_discard_identity weight (fun C=>C.val.val.Adj x y) good
  39   rw [he x y,if_pos hxy] at hid
  40   simp only [hnot] at hid
  41   have hlo : 0≤∑ C,if C.val.val.Adj x y ∧ (∃ v∈C.val.val.support,∃ z∈C.val.val.support,F.Adj v z)
  42       then weight C else 0 := Finset.sum_nonneg (fun C _=>by split_ifs;exact hweight C;exact le_refl _)
  43   have hhi := ordinary_cycle_rim_conflict_loss P r hr hn weight hweight B hB hb F L hdeg x y hxy.ne hFxy
  44   have hbound : |(∑ C,if C.val.val.Adj x y ∧ good C then weight C else 0)-1|≤
  45       5*(r:ℝ)^4*B*L/Fintype.card V := by
  46     rw [abs_sub_comm]
  47     erw [hid]
  48     rw [abs_of_nonneg hlo]
  49     convert hhi using 1 <;> congr!
