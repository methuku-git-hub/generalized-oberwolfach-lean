  27 theorem two_factor_subgraph_dependent_model (F H:SimpleGraph V)
  28     (hF:F.IsRegularOfDegree 2) (hHF:H≤F) (R:Finset V)
  29     (hisol:∀x,x∉R→∀y,¬H.Adj x y):
  30     ∃ (D:V→V→Prop) (ps:List (PrescribedSourcePiece V)),
  31       ps.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset) ∧
  32       (∀p∈ps,p.Shape ∧ p.OriginalValid D) ∧
  33       prescribedDependentVertices (fun _ : Unit=>ps) ()=R ∧
  34       prescribedDependentGraph (fun _ : Unit=>ps) ()=H ∧
  35       ∀x y,D x y→F.Adj x y := by
  36   obtain ⟨D,ps,hpair,hvalid,hverts,hedges,hD⟩:=two_factor_subgraph_ordered_source_pieces F H hF hHF R hisol
  37   refine ⟨D,ps,hpair,?_,?_,?_,hD⟩
  38   · intro p hp
  39     have hh:=hvalid p hp
  40     constructor
  41     · cases p with
  42       | path xs => exact hh.1
  43       | cycle a xs => exact ⟨hh.1,hh.2.1⟩
  44     · exact hh
  45   · rw [dependent_source_vertices_of_list,hverts]
  46   · exact dependent_source_graph_of_list ps H hedges
  47 end Oberwolfach
