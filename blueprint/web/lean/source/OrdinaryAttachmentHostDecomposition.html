  20 theorem attachment_embedded_cycle_separated (F : SimpleGraph V) (z : W) (ρ : I → V₀)
  21     (f : V₀⊕(I×W) ↪ V)
  22     (hcross : ∀ i w a,¬F.Adj (f (.inr (i,w))) (f (.inl a)))
  23     (hnew : ∀ i j w w',¬F.Adj (f (.inr (i,w))) (f (.inr (j,w'))))
  24     (i : I) (w w' : W) :
  25     ¬F.Adj (f (attachmentWheelEmbedding z ρ i w)) (f (attachmentWheelEmbedding z ρ i w')) := by
  26   have he (a : W) : attachmentWheelEmbedding z ρ i a=
  27       if a=z then Sum.inl (ρ i) else Sum.inr (i,a) := by
  28     by_cases ha : a=z
  29     · rw [if_pos ha]
  30       exact (attachmentWheelEmbedding_old z ρ i a (ρ i)).mpr ⟨ha,rfl⟩
  31     · rw [if_neg ha]
  32       exact (attachmentWheelEmbedding_private z ρ i i a a).mpr ⟨ha,rfl,rfl⟩
  33   rw [he,he]
  34   by_cases hw : w=z <;> by_cases hw' : w'=z
  35   · simp only [hw,hw',ite_true];exact fun h=>h.ne rfl
  36   · simp only [hw,hw',ite_true,ite_false]
  37     exact fun h=>hcross i w' (ρ i) h.symm
  38   · simp only [hw,hw',ite_true,ite_false];exact hcross i w (ρ i)
  39   · simp only [hw,hw',ite_false];exact hnew i i w w'

  41 theorem ordinary_attachment_extra_host_decomposition
  42     (Q F : SimpleGraph V) (H : V → U → Prop) (C : SimpleGraph W)
  43     (hC : IsEmbeddedCycle C r) (hfull : ∀ w,w∈C.support) (z : W) (ρ : I → V₀)
  44     (f : V₀⊕(I×W) ↪ V) (g : U₀⊕I ↪ U)
  45     (hQ : ∀ a b,(ordinaryAttachmentRim ⊥ C z ρ).Adj a b → Q.Adj (f a) (f b))
  46     (hH : ∀ a b,ordinaryAttachmentSpokes (fun (_ : V₀) (_ : U₀)=>False) z ρ a b → H (f a) (g b))
  47     (hcross : ∀ i w a,¬F.Adj (f (.inr (i,w))) (f (.inl a)))
  48     (hnew : ∀ i j w w',¬F.Adj (f (.inr (i,w))) (f (.inr (j,w')))) :
  49     HasResourceDecomposition (ordinaryWheelResources (P:=Q) (F:=F) (Hbase:=H) (r:=r))
  50       (ordinaryHostResources ((ordinaryAttachmentRim ⊥ C z ρ).map f)
  51         (ordinaryMappedSpokes (ordinaryAttachmentSpokes (fun (_ : V₀) (_ : U₀)=>False) z ρ) f g)) := by
  52   let W₀ := finiteAttachmentWheel (U:=U₀) C hC z ρ
  53   let W₁ := fun i=>universalOrdinaryWheelEmbedding f g (ordinaryWheelUniversal (W₀ i))
  54   have hWi (i : I) : ordinaryWheelResources (W₁ i)⊆ordinaryHostResources Q H := by
  55     apply (ordinaryWheelResources_subset_host _ Q H).mpr
  56     constructor
  57     · intro x y hxy
  58       obtain ⟨a,b,hab,rfl,rfl⟩ := (SimpleGraph.map_adj f _ _ _).mp hxy
  59       exact hQ a b ((W₀ i).val.1.property hab)
  60     · intro x hx
  61       change x∈((ordinaryWheelRim (W₀ i)).map f).support at hx
  62       rw [SimpleGraph.support_map] at hx
  63       obtain ⟨a,ha,rfl⟩ := hx
  64       exact hH a (.inr i) ((W₀ i).property.1 a ha)
  65   have hFi (i : I) : ∀ x∈(ordinaryWheelRim (W₁ i)).support,
  66       ∀ y∈(ordinaryWheelRim (W₁ i)).support,¬F.Adj x y := by
  67     intro x hx y hy
  68     change x∈((C.map (attachmentWheelEmbedding z ρ i)).map f).support at hx
  69     change y∈((C.map (attachmentWheelEmbedding z ρ i)).map f).support at hy
  70     rw [SimpleGraph.support_map] at hx hy
  71     obtain ⟨a,ha,rfl⟩ := hx
  72     obtain ⟨b,hb,rfl⟩ := hy
  73     rw [SimpleGraph.support_map] at ha hb
  74     obtain ⟨w,_,rfl⟩ := ha
  75     obtain ⟨w',_,rfl⟩ := hb
  76     exact attachment_embedded_cycle_separated F z ρ f hcross hnew i w w'
  77   let W₂ := fun i=>ordinaryWheelInHost Q F H (W₁ i) (hWi i) (hFi i)
  78   rw [←ordinaryHostResources_embedding_image]
  79   apply resource_decomposition_embedding (fun i=>ordinaryWheelResources (W₀ i))
  80     ordinaryWheelResources (ordinaryPairResourceEmbedding f g) W₂ (ordinaryPairResourceEmbedding f g).injective
  81     (fun i=>universalOrdinaryWheelEmbedding_resources f g (ordinaryWheelUniversal (W₀ i)))
  82   refine ⟨Finset.univ,(fun i _ j _ hij=>finiteAttachmentWheels_disjoint C hC z ρ i j hij),?_⟩
  83   rw [←wheelFamily_resources W₀]
  84   rw [finiteAttachmentWheel_family_rim,finiteAttachmentWheel_family_spokes C hC z ρ hfull]
