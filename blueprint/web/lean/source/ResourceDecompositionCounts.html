  16 theorem resource_decomposition_indicator (R : W → Finset E) (M : Finset W) (S : Finset E)
  17     (hM : ∀ a∈M, ∀ b∈M, a≠b → Disjoint (R a) (R b)) (hS : M.biUnion R=S) (e : E) :
  18     (∑ w∈M, if e∈R w then (1:ℝ) else 0)=if e∈S then 1 else 0 := by
  19   by_cases he : e∈S
  20   · rw [if_pos he]
  21     rw [← hS] at he
  22     obtain ⟨a,ha,hea⟩ := Finset.mem_biUnion.mp he
  23     have hh : (∑ w∈M, if e∈R w then (1:ℝ) else 0)=if e∈R a then 1 else 0 := by
  24       apply Finset.sum_eq_single_of_mem a ha
  25       intro b hb hba
  26       apply if_neg
  27       intro heb
  28       exact (Finset.disjoint_left.mp (hM b hb a ha hba)) heb hea
  29     simpa only [hea,ite_true] using hh
  30   · rw [if_neg he]
  31     apply Finset.sum_eq_zero
  32     intro w hw
  33     apply if_neg
  34     intro hew
  35     apply he
  36     rw [← hS]
  37     exact Finset.mem_biUnion.mpr ⟨w,hw,hew⟩

  39 theorem resource_decomposition_indicator_weight [Fintype W]
  40     (R : W → Finset E) (M : Finset W) (S : Finset E)
  41     (hM : ∀ a∈M, ∀ b∈M, a≠b → Disjoint (R a) (R b)) (hS : M.biUnion R=S) (e : E) :
  42     (∑ w, if e∈R w then (if w∈M then (1:ℝ) else 0) else 0)=if e∈S then 1 else 0 := by
  43   calc
  44     _ = ∑ w, if w∈M then (if e∈R w then (1:ℝ) else 0) else 0 := by
  45       apply Finset.sum_congr rfl
  46       intro w _
  47       by_cases hw : w∈M <;> by_cases he : e∈R w <;> simp [hw,he]
  48     _ = ∑ w∈M, if e∈R w then (1:ℝ) else 0 := by
  49       rw [← Finset.sum_filter]
  50       simp only [Finset.filter_mem_eq_inter,Finset.univ_inter]
  51     _ = _ := resource_decomposition_indicator R M S hM hS e
