  11 theorem finiteLaw_exp_centered_le_one_constant
  12     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X : Ω → ℝ)
  13     (h : μ.expectation (fun ω => Real.exp (X ω - μ.expectation X)) ≤ 1)
  14     (ω : Ω) (hω : 0 < μ.mass ω) : X ω = μ.expectation X := by
  15   let Z (x : Ω) := X x - μ.expectation X
  16   have hz : μ.expectation Z = 0 := by
  17     simp only [FiniteLaw.expectation, Z, mul_sub, Finset.sum_sub_distrib,
  18       ← Finset.sum_mul, μ.total, one_mul, sub_self]
  19   let Y (x : Ω) := Real.exp (Z x) - (Z x + 1)
  20   have hy (x : Ω) : 0 ≤ Y x := sub_nonneg.mpr (Real.add_one_le_exp _)
  21   have hsum : μ.expectation Y ≤ 0 := by
  22     have he : μ.expectation Y = μ.expectation (fun x => Real.exp (Z x)) - (μ.expectation Z + 1) := by
  23       simp only [FiniteLaw.expectation, Y, mul_sub, mul_add, mul_one,
  24         Finset.sum_sub_distrib, Finset.sum_add_distrib, μ.total]
  25     rw [he, hz]
  26     change μ.expectation (fun x => Real.exp (X x - μ.expectation X)) - (0 + 1) ≤ 0
  27     linarith
  28   have hs : μ.mass ω * Y ω ≤ 0 :=
  29     (Finset.single_le_sum (fun x _ => mul_nonneg (μ.nonneg x) (hy x)) (Finset.mem_univ ω)).trans hsum
  30   have hY : Y ω ≤ 0 := nonpos_of_mul_nonpos_right hs hω
  31   have heq : Z ω = 0 := by
  32     by_contra hn
  33     have hh := Real.add_one_lt_exp hn
  34     change Real.exp (Z ω) - (Z ω + 1) ≤ 0 at hY
  35     linarith
  36   exact sub_eq_zero.mp heq
