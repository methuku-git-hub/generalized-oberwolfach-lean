  13 theorem markedCycleIncidence_zero_one {V E K : Type*} [DecidableEq V]
  14     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (C : Fin (m+1) → V)
  15     (hC : Function.Injective C) (e : E⊕K) :
  16     markedCycleIncidence tail head ktail khead m C e=0 ∨
  17       markedCycleIncidence tail head ktail khead m C e=1 := by
  18   classical
  19   cases e with
  20   | inl e =>
  21     simp only [markedCycleIncidence,Sum.elim_inl]
  22     by_cases h : ∃ j : Fin m,tail e=C j.castSucc ∧ head e=C j.succ
  23     · obtain ⟨j,hj⟩ := h
  24       right
  25       rw [Finset.sum_eq_single j]
  26       · simp only [directedEdgeIndicator,hj.1,hj.2,and_self,ite_true]
  27       · intro l _ hlj
  28         have hh : ¬(tail e=C l.castSucc ∧ head e=C l.succ) := by
  29           intro hl
  30           have he := hC (hl.1.symm.trans hj.1)
  31           exact hlj (Fin.castSucc_injective m he)
  32         simp only [directedEdgeIndicator,hh,ite_false]
  33       · simp
  34     · left
  35       apply Finset.sum_eq_zero
  36       intro j _
  37       simp only [directedEdgeIndicator,show ¬(tail e=C j.castSucc ∧ head e=C j.succ) from fun hj => h ⟨j,hj⟩,ite_false]
  38   | inr e =>
  39     simp only [markedCycleIncidence,Sum.elim_inr,directedEdgeIndicator]
  40     split_ifs <;> simp

  42 def markedCycleTypedEdge {V E K : Type*} (tail head : E → V) (ktail khead : K → V)
  43     (m : ℕ) (π : Equiv.Perm V) (b : ℕ)
  44     (C : MarkedCycleTuple tail head ktail khead m π b) : Fin (m+1) → E⊕K :=
  45   Fin.snoc (fun j => Sum.inl (Classical.choose (C.property.2.1 j)))
  46     (Sum.inr (Classical.choose C.property.2.2))

  48 theorem markedCycleTypedEdge_endpoints {V E K : Type*}
  49     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (π : Equiv.Perm V) (b : ℕ)
  50     (C : MarkedCycleTuple tail head ktail khead m π b) (i : Fin (m+1)) :
  51     Sum.elim tail ktail (markedCycleTypedEdge tail head ktail khead m π b C i)=C.val i ∧
  52     Sum.elim head khead (markedCycleTypedEdge tail head ktail khead m π b C i)=C.val (finRotate (m+1) i) := by
  53   refine Fin.lastCases ?_ (fun j => ?_) i
  54   · simp only [markedCycleTypedEdge,Fin.snoc_last,Sum.elim_inr,finRotate_last]
  55     exact Classical.choose_spec C.property.2.2
  56   · simp only [markedCycleTypedEdge,Fin.snoc_castSucc,Sum.elim_inl,finRotate_castSucc_eq_succ]
  57     exact Classical.choose_spec (C.property.2.1 j)

  59 theorem markedCycleTypedEdge_injective {V E K : Type*}
  60     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (π : Equiv.Perm V) (b : ℕ)
  61     (C : MarkedCycleTuple tail head ktail khead m π b) :
  62     Function.Injective (markedCycleTypedEdge tail head ktail khead m π b C) := by
  63   intro i j hij
  64   apply permutationSeparated_injective π b C.val C.property.1
  65   rw [← (markedCycleTypedEdge_endpoints tail head ktail khead m π b C i).1,
  66       ← (markedCycleTypedEdge_endpoints tail head ktail khead m π b C j).1,hij]

  68 theorem markedCycle_physical_images_injective {V E K P : Type*}
  69     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (π : Equiv.Perm V) (b : ℕ)
  70     (C : MarkedCycleTuple tail head ktail khead m π b) (physical : E⊕K → P)
