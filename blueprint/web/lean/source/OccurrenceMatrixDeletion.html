  18 def occurrenceMultiplicity (tail head : E → V) (x y : V) : ℕ :=
  19   ∑ e,if tail e=x ∧ head e=y then 1 else 0

  47     (hinj : Function.Injective (fun e=>(tail e,head e))) (keep : E → Prop) (e : E) :
  48     occurrenceMultiplicity (fun f : {e // keep e}=>tail f.val) (fun f=>head f.val) (tail e) (head e)=
  49       if keep e then 1 else 0 := by
  50   have hi : Function.Injective (fun f : {e // keep e}=>(tail f.val,head f.val)) := by
  51     intro f g hfg;apply Subtype.ext;exact hinj hfg
  52   by_cases hk : keep e
  53   · rw [if_pos hk]
  54     exact occurrenceMultiplicity_at_edge _ _ hi ⟨e,hk⟩
  55   · rw [if_neg hk]

  57     rintro ⟨f,ht,hh⟩
  58     exact hk ((hinj (Prod.ext ht hh)) ▸ f.property)

  67     have hh := occurrenceMultiplicity_restricted_at_edge tail head hinj (fun f=>J (tail f) (head f)=0) e
