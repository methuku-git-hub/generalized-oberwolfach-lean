  10     (h':IsSelectedSegmentRun T a' b')
  11     (haj:a≤j) (hjb:j≤b) (ha'j:a'≤j) (hjb':j≤b'):
  12     a=a' ∧ b=b' := by
  13   have hab:=h.1
  14   have ha'b':=h'.1
  15   have haa:a=a' := by
  16     by_contra hne
  17     rcases lt_or_gt_of_ne hne with hlt|hgt
  18     · have hp:a'-1∈T := h.2.1 (a'-1) (by omega) (by omega)
  19       rcases h'.2.2.1 with hzero|hnot
  20       · omega
  21       · exact hnot hp

  32     (↑(selectedSegmentRuns d T):Set (ℕ×ℕ)).PairwiseDisjoint
  33       (fun p=>Finset.Icc p.1 p.2) := by
