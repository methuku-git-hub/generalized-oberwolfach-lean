  14 theorem transferCycleCoefficient_abs_sum {r : ℕ} (hr : 0<r) :
  15     (∑ j : TransferCycleIndex r,|transferCycleCoefficient j|)≤3 := by
  16   have hr' : (0:ℝ)<r := Nat.cast_pos.mpr hr
  17   have habs (j : TransferCycleIndex r) : |transferCycleCoefficient j|=1/(r:ℝ) := by
  18     rcases j with ⟨⟩|⟨i,b⟩
  19     · simp [transferCycleCoefficient,abs_of_nonneg (show 0≤1/(r:ℝ) by positivity)]
  20     · cases b <;> simp [transferCycleCoefficient,abs_of_nonneg (show 0≤1/(r:ℝ) by positivity)]
  21   simp only [habs,Finset.sum_const,Finset.card_univ,nsmul_eq_mul,
  22     Fintype.card_sum,Fintype.card_unique,Fintype.card_prod,Fintype.card_fin,Fintype.card_bool]
  23   push_cast
  24   have h1 : (1:ℝ)≤r := by exact_mod_cast hr
  25   rw [mul_one_div]
  26   apply (div_le_iff₀ hr').mpr
  27   linarith

  29 theorem indexedWheelWeight_abs_le {V U I : Type*} [Fintype I]
  30     {Q F : SimpleGraph V} {H : V → U → Prop} {r : ℕ}
  31     (W : I → OrdinaryWheel Q F H r) (a : I → ℝ) (D : OrdinaryWheel Q F H r) :
  32     |indexedWheelWeight W a D|≤∑ i,|a i| := by
  33   unfold indexedWheelWeight
  34   apply (Finset.abs_sum_le_sum_abs (fun i => if W i=D then a i else 0) Finset.univ).trans
  35   apply Finset.sum_le_sum
  36   intro i _
  37   split_ifs
  38   · rfl
  39   · simpa only [abs_zero] using abs_nonneg (a i)

  41 theorem indexedWheelWeight_zero_of_no_image {V U I : Type*} [Fintype I]
  42     {Q F : SimpleGraph V} {H : V → U → Prop} {r : ℕ}
  43     (W : I → OrdinaryWheel Q F H r) (a : I → ℝ) (D : OrdinaryWheel Q F H r)
  44     (hD : ∀ i,W i≠D) : indexedWheelWeight W a D=0 := by
  45   apply Finset.sum_eq_zero
  46   intro i _
  47   exact if_neg (hD i)

  49 theorem OrdinaryTransferConfiguration.coefficient_bound {V U : Type*} {r : ℕ} {hr : 3≤r}
  50     {Q F : SimpleGraph V} {H : V → U → Prop} {v : V} {u u' : U}
  51     (T : OrdinaryTransferConfiguration hr Q F H v u u') (D : OrdinaryWheel Q F H r) :
  52     |T.weight D|≤6 := by
  53   have h₁ := (indexedWheelWeight_abs_le (T.wheel true) transferCycleCoefficient D).trans
  54     (transferCycleCoefficient_abs_sum (by omega))
  55   have h₂ := (indexedWheelWeight_abs_le (T.wheel false) transferCycleCoefficient D).trans
  56     (transferCycleCoefficient_abs_sum (by omega))
  57   have hh := abs_add_le (indexedWheelWeight (T.wheel true) transferCycleCoefficient D)
  58     (-(indexedWheelWeight (T.wheel false) transferCycleCoefficient D))
  59   simp only [abs_neg,← sub_eq_add_neg] at hh
  60   change |indexedWheelWeight (T.wheel true) transferCycleCoefficient D-
  61     indexedWheelWeight (T.wheel false) transferCycleCoefficient D|≤6
  62   linarith

  64 theorem OrdinaryTransferConfiguration.weight_support {V U : Type*} {r : ℕ} {hr : 3≤r}
  65     {Q F : SimpleGraph V} {H : V → U → Prop} {v : V} {u u' : U}
  66     (T : OrdinaryTransferConfiguration hr Q F H v u u') (D : OrdinaryWheel Q F H r)
  67     (hD : T.weight D≠0) :
  68     (ordinaryWheelHub D=u ∨ ordinaryWheelHub D=u') ∧
  69       (ordinaryWheelRim D).support⊆insert v (Set.range T.labels) := by
  70   have hex : ∃ b : Bool,∃ j : TransferCycleIndex r,T.wheel b j=D := by
  71     by_contra hn
  72     have hzero (b : Bool) : indexedWheelWeight (T.wheel b) transferCycleCoefficient D=0 := by
  73       apply indexedWheelWeight_zero_of_no_image
  74       intro j hj
  75       exact hn ⟨b,j,hj⟩
  76     apply hD
  77     change indexedWheelWeight (T.wheel true) transferCycleCoefficient D-
  78       indexedWheelWeight (T.wheel false) transferCycleCoefficient D=0
  79     rw [hzero,hzero,sub_self]
  80   obtain ⟨b,j,rfl⟩ := hex
  81   constructor
  82   · cases b
  83     · exact Or.inr rfl
  84     · exact Or.inl rfl
  85   · exact transferCycleGraph_support_subset hr T.labels v T.omitted j
