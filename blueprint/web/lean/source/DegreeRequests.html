   9 theorem indexPrescribedCounts
  10     {V : Type*} [Fintype V] [DecidableEq V]
  11     (f : V → ℕ) (m : ℕ) (hm : ∑ v, f v = m) :
  12     ∃ x : Fin m → V, ∀ v, (Finset.univ.filter (fun i => x i = v)).card = f v := by
  13   classical
  14   let A := Σ v, Fin (f v)
  15   have hcard : Fintype.card A = m := by simpa [A, Fintype.card_sigma] using hm
  16   let σ : Fin m ≃ A := (Fintype.equivFinOfCardEq hcard).symm
  17   refine ⟨fun i => (σ i).1, ?_⟩
  18   intro v
  19   rw [Finset.card_filter, ← σ.symm.sum_comp]
  20   simp only [Equiv.apply_symm_apply]
  21   change (∑ u : (Σ w, Fin (f w)), if u.1 = v then 1 else 0) = f v
  22   rw [Fintype.sum_sigma]
  23   have hh (w : V) : (∑ _ : Fin (f w), if w = v then 1 else 0) =
  24       if w = v then f w else 0 := by
  25     by_cases hw : w = v <;> simp [hw]
  26   simp_rw [hh]
  27   simp

  29 /-- Positive and negative discrepancies give lists of equal length.
  30 The two endpoint classes are disjoint, and their combined use at a vertex
  31 is bounded by its original absolute discrepancy. -/
  32 theorem degreeDiscrepancy_requests
  33     {V : Type*} [Fintype V] [DecidableEq V]
  34     (d t : V → ℕ) (htotal : ∑ v, d v = ∑ v, t v)
  35     {a : ℝ} (_ha : 0 ≤ a)
  36     (herr : ∀ v, |(d v : ℝ) - t v| ≤ a) :
  37     ∃ m : ℕ, ∃ x y : Fin m → V,
  38       ((m : ℝ) ≤ Fintype.card V * a) ∧
  39       (∀ v, (Finset.univ.filter (fun i => x i = v)).card = d v - t v) ∧
  40       (∀ v, (Finset.univ.filter (fun i => y i = v)).card = t v - d v) ∧
  41       (∀ v, ((Finset.univ.filter (fun i => x i = v)).card : ℝ) +
  42         ((Finset.univ.filter (fun i => y i = v)).card : ℝ) ≤ a) ∧
  43       ∀ i j, x i ≠ y j := by
  44   classical
  45   let p (v : V) := d v - t v
  46   let q (v : V) := t v - d v
  47   have hpoint (v : V) : d v + q v = t v + p v := by dsimp [p, q]; omega
  48   have hequal : ∑ v, p v = ∑ v, q v := by
  49     have hh := Finset.sum_congr (s₁ := Finset.univ) rfl (fun v _ => hpoint v)
  50     simp only [Finset.sum_add_distrib, htotal] at hh
  51     omega
  52   let m := ∑ v, p v
  53   obtain ⟨x, hx⟩ := indexPrescribedCounts p m rfl
  54   obtain ⟨y, hy⟩ := indexPrescribedCounts q m hequal.symm
  55   have hbound (v : V) : (p v : ℝ) + (q v : ℝ) ≤ a := by
  56     by_cases hd : t v ≤ d v
  57     · have hq : q v = 0 := Nat.sub_eq_zero_of_le hd
  58       simp only [p, Nat.cast_sub hd, hq, Nat.cast_zero, add_zero]
  59       exact (le_abs_self _).trans (herr v)
  60     · have hp : p v = 0 := Nat.sub_eq_zero_of_le (le_of_not_ge hd)
  61       have htd : d v ≤ t v := le_of_lt (lt_of_not_ge hd)
  62       simp only [q, Nat.cast_sub htd, hp, Nat.cast_zero, zero_add]
  63       have hh := (neg_le_abs ((d v : ℝ) - t v)).trans (herr v)
  64       linarith
  65   have hsize : (m : ℝ) ≤ Fintype.card V * a := by
  66     calc
  67       (m : ℝ) = ∑ v, (p v : ℝ) := by simp [m]
  68       _ ≤ ∑ _ : V, a := Finset.sum_le_sum (fun v _ => by
  69         have hp : (0 : ℝ) ≤ q v := Nat.cast_nonneg (q v)
  70         linarith [hbound v])
  71       _ = _ := by simp
  72   refine ⟨m, x, y, hsize, hx, hy, ?_, ?_⟩
  73   · intro v
  74     rw [hx, hy]
  75     exact hbound v
  76   · intro i j hij
  77     have hxp : 0 < p (x i) := by
  78       rw [← hx]
  79       exact Finset.card_pos.mpr ⟨i, by simp⟩
  80     have hyq : 0 < q (y j) := by
  81       rw [← hy]
  82       exact Finset.card_pos.mpr ⟨j, by simp⟩
  83     rw [← hij] at hyq
  84     dsimp [p] at hxp
  85     dsimp [q] at hyq
  86     omega
  87
