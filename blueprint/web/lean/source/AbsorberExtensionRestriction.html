  17 theorem labelled_absorber_extension_occurrence (R : C → Finset V)
  18     (S : VertexAbsorberTemplate c) (X A Z : Finset V) (hAZ : Disjoint A Z)
  19     (D : LabelledCycleRecord R S.val X)
  20     (hfree : labelledAbsorberFree R S X D⊆A) (o : Fin c ⊕ Fin (c+1)) :
  21     (R (D.2 o).val∩Z).card ≤ 1 := by
  22   apply Finset.card_le_one.mpr
  23   intro x hx y hy
  24   obtain ⟨hx,hxZ⟩ := Finset.mem_inter.mp hx
  25   obtain ⟨hy,hyZ⟩ := Finset.mem_inter.mp hy
  26   have hxX : x∈X := by
  27     have hh := Finset.mem_union.mp (labelled_absorber_occurrence_subset R S X D o hx)
  28     exact hh.resolve_left (fun hf => Finset.disjoint_left.mp hAZ (hfree hf) hxZ)
  29   have hyX : y∈X := by
  30     have hh := Finset.mem_union.mp (labelled_absorber_occurrence_subset R S X D o hy)
  31     exact hh.resolve_left (fun hf => Finset.disjoint_left.mp hAZ (hfree hf) hyZ)
  32   exact labelled_absorber_occurrence_root_unique R S X D o x y hx hy hxX hyX

  35     (S : T → VertexAbsorberTemplate c) (X A Z : Finset V) (hAZ : Disjoint A Z) :
  36     absorberGroundCount R S X A ≤

  42   let f : D₀ → D₁ := fun D => ⟨D.val,fun o =>
