  21 theorem marked_typed_rim_subhost_deletion
  22     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ)
  23     (σ τ:RawMarkedOutcome V U Γ Color)

  25       markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) mark x y=1):
  26     ∃ D:SimpleGraph V,D≤P ∧
  27       (∀ x,D.degree x=(Finset.univ.filter (fun y=>x≠y ∧ markedRemovedPhysicalRim P π d σ τ s(x,y))).card) ∧
  28       ∀ mark x y,markedPhysicalRim P π (3*d) (fun e=>τ (Sum.inl e)) mark x y=
  29         markedPhysicalRim P π (3*d) (fun e=>rawMarkedDeleteRims σ D.edgeFinset (Sum.inl e)) mark x y := by

  31   let D:SimpleGraph V:=SimpleGraph.fromEdgeSet {e | Rem e}
  32   letI hinst:Fintype D.edgeSet:=D.fintypeEdgeSet
  33   have hfin:@SimpleGraph.edgeFinset V D hinst=@SimpleGraph.edgeFinset V D D.fintypeEdgeSet:=by
  34     ext e
  35     simp only [SimpleGraph.mem_edgeFinset]
  36   have hmem (e:Sym2 V) (he:¬e.IsDiag):e∈D.edgeFinset↔Rem e:=by
  37     rw [SimpleGraph.mem_edgeFinset,SimpleGraph.edgeSet_fromEdgeSet]
  38     simp only [Set.mem_sdiff,Set.mem_ofPred_eq,Sym2.mem_diagSet,he,not_false_eq_true,and_true]
  39   have hadj (e:Sym2 V) (he:Rem e):e∈P.edgeSet:=by
  40     obtain ⟨mark,x,y,hσ,_,rfl⟩:=he
  41     exact ((markedPhysicalRim_eq_one_iff P π (3*d) (fun e=>σ (Sum.inl e)) mark x y).mp hσ).2.1
  42   refine ⟨D,?_,?_,?_⟩
  43   · intro x y h
  44     have hrem:Rem s(x,y):=((SimpleGraph.fromEdgeSet_adj _).mp h).1
  45     exact hadj _ hrem
  46   · intro x
  47     rw [←SimpleGraph.card_neighborFinset_eq_degree]
  48     congr 1
  49     ext y
  50     simp only [SimpleGraph.mem_neighborFinset,D,SimpleGraph.fromEdgeSet_adj,Set.mem_ofPred_eq,
  51       Finset.mem_filter,Finset.mem_univ,true_and,Rem]
  52     tauto
  53   · intro mark x y
  54     let e:Sym2 V:=s(x,if mark then π y else y)
  55     have hσ1 : markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) mark x y=1 → ¬e.IsDiag:=by
  56       intro hh
  57       have hp:=((markedPhysicalRim_eq_one_iff P π (3*d) (fun e=>σ (Sum.inl e)) mark x y).mp hh).2.1
  58       dsimp [e];rw [Sym2.mk_isDiag_iff];exact hp.ne
  59     have hkeep : markedPhysicalRim P π (3*d) (fun e=>τ (Sum.inl e)) mark x y=1 → e∉D.edgeFinset:=by
  60       intro ht hd
  61       have hs:=hsub mark x y ht
  62       have hr:Rem e:=(hmem e (hσ1 hs)).mp hd
  63       obtain ⟨m',x',y',hs',ht',heq⟩:=hr
  64       have hu:=markedPhysicalRim_physical_unique P π (3*d) (fun e=>σ (Sum.inl e)) mark m' x y x' y' hs hs' heq
  65       obtain ⟨rfl,rfl,rfl⟩:=hu
  66       exact ht' ht
  67     have hlost : markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) mark x y=1 →
  68         markedPhysicalRim P π (3*d) (fun e=>τ (Sum.inl e)) mark x y≠1 → e∈D.edgeFinset:=by
  69       intro hs ht
  70       exact (hmem e (hσ1 hs)).mpr ⟨mark,x,y,hs,ht,rfl⟩
  71     have hiff : markedPhysicalRim P π (3*d) (fun e=>τ (Sum.inl e)) mark x y=1 ↔
  72         (markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) mark x y=1 ∧ e∉D.edgeFinset):=by

  79         (markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) mark x y=1 ∧ e∉D.edgeFinset):=by
