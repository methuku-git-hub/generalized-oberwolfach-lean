  16 def permutedNeighbourChoices {V : Type*} [Fintype V]
  17     (P : SimpleGraph V) (π τ : Equiv.Perm V) (b : ℕ) (A R S : Finset V) : Finset V :=
  18   A.filter (fun v=>(∀ x∈R,P.Adj x (τ v)) ∧ ∀ x∈S,¬PermutationClose π b x v)

  20 theorem permutationClose_union_card {V : Type*} [Fintype V] [DecidableEq V]
  21     (π : Equiv.Perm V) (b : ℕ) (S : Finset V) :
  22     (S.biUnion (fun x=>Finset.univ.filter (fun v=>PermutationClose π b x v))).card≤S.card*(2*b+1) := by
  23   calc
  24     _ ≤ ∑ x∈S,(Finset.univ.filter (fun v=>PermutationClose π b x v)).card := Finset.card_biUnion_le
  25     _ ≤ ∑ _x∈S,(2*b+1) := by
  26       apply Finset.sum_le_sum
  27       intro x _
  28       simpa only [Fintype.card_subtype] using permutationClose_neighbour_card π b x
  29     _ = _ := by simp

  31 theorem permutedNeighbourChoices_card_lower {V : Type*} [Fintype V] [DecidableEq V]
  32     (P : SimpleGraph V) (π τ : Equiv.Perm V) (b : ℕ) (A R S : Finset V)
  33     (α : ℝ) (hα1 : α≤1) (hR : R.card≤2)
  34     (hdegree : ∀ x∈R,(1/2+α/2)*A.card≤((P.neighborFinset x∩A.image τ).card:ℝ)) :
  35     α*A.card-(S.card:ℝ)*(2*(b:ℝ)+1)≤(permutedNeighbourChoices P π τ b A R S).card := by
  36   let N := fun x=>Finset.univ.filter (fun v=>P.Adj x (τ v))
  37   have hdeg (x : V) (hx : x∈R) : (1/2+α/2)*A.card≤((A∩N x).card:ℝ) := by
  38     have he : A∩N x=A.filter (fun v=>P.Adj x (τ v)) := by ext v;simp [N]
  39     rw [he,permutation_neighbour_filter_card]
  40     exact hdegree x hx
  41   have hc := common_neighbour_card_lower A R N (1/2+α/2) hdeg
  42   have hR' : (R.card:ℝ)≤2 := by exact_mod_cast hR
  43   have hco : α≤1-(R.card:ℝ)*(1-(1/2+α/2)) := by
  44     have hh := mul_le_mul_of_nonneg_right hR' (show 0≤1-(1/2+α/2) by linarith)
  45     nlinarith only [hh]
  46   let D := S.biUnion (fun x=>Finset.univ.filter (fun v=>PermutationClose π b x v))
  47   have hD : (D.card:ℝ)≤(S.card:ℝ)*(2*(b:ℝ)+1) := by
  48     exact_mod_cast permutationClose_union_card π b S
  49   have he : permutedNeighbourChoices P π τ b A R S=common A R N\D := by
  50     ext v
  51     simp only [permutedNeighbourChoices,common,N,D,Finset.mem_filter,Finset.mem_sdiff,
  52       Finset.mem_univ,true_and,Finset.mem_biUnion,not_exists,not_and]
  53     tauto
  54   rw [he]
  55   have hl : ((common A R N).card:ℝ)≤(common A R N\D).card+D.card := by
  56     exact_mod_cast (Finset.card_le_card_sdiff_add_card (s:=common A R N) (t:=D))
  57   have hm := mul_le_mul_of_nonneg_right hco (Nat.cast_nonneg A.card : (0:ℝ)≤A.card)
  58   linarith only [hc,hD,hl,hm]
