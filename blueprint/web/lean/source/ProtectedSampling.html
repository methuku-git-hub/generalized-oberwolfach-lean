  16 theorem finiteLaw_inter_lower {Ω : Type*} [Fintype Ω]
  17     (μ : FiniteLaw Ω) (A B : Ω → Prop) :
  18     μ.probability A + μ.probability B - 1 ≤ μ.probability (fun ω => A ω ∧ B ω) := by
  19   rw [sub_le_iff_le_add, ← μ.total]
  20   unfold FiniteLaw.probability
  21   rw [← Finset.sum_add_distrib, ← Finset.sum_add_distrib]
  22   apply Finset.sum_le_sum
  23   intro ω _
  24   by_cases ha : A ω <;> by_cases hb : B ω <;> simp [ha, hb, μ.nonneg ω]

  26 structure ProtectedSampleGood {V : Type*} [Fintype V] [LinearOrder V] {k : ℕ}
  27     (G Z : SimpleGraph V) (ρ : Fin k → ℝ) (α q : ℝ)
  28     (σ : Sym2 V → Fin k × Bool) : Prop where
  29   degrees : ∀ a x,
  30     |((∑ y, sampledColourOrientation G σ a x y : ℕ) : ℝ) - ρ a / 2 * G.degree x| <
  31       (Fintype.card V : ℝ) ^ (3 / 5 : ℝ) ∧
  32     |((∑ y, sampledColourOrientation G σ a y x : ℕ) : ℝ) - ρ a / 2 * G.degree x| <
  33       (Fintype.card V : ℝ) ^ (3 / 5 : ℝ) ∧
  34     |(sampledColourDegree Z (fun e => (σ e).1) a x : ℝ) - ρ a * Z.degree x| <
  35       (Fintype.card V : ℝ) ^ (3 / 5 : ℝ)
  36   cuts : ∀ a S T,
  37     |((∑ x ∈ S, ∑ y ∈ T, sampledColourOrientation G σ a x y : ℕ) : ℝ) -
  38       ρ a / 2 * graphCut G S T| < q * (Fintype.card V : ℝ) ^ 2
  39   balance : ∀ F : SimpleGraph V, (∀ v, (F.degree v : ℝ) ≤ α * Fintype.card V / 100) →
  40     ∃ M : V → V → ℕ,
  41       (∀ x y, M x y + M y x = if G.Adj x y then 1 else 0) ∧
  42       (∀ v, (∑ w, M v w) = G.degree v / 2 ∧ (∑ w, M w v) = G.degree v / 2) ∧
  43       (∀ x y, M x y ≠ sampledOrientation G (fun e => (σ e).2) x y ∨
  44         M y x ≠ sampledOrientation G (fun e => (σ e).2) y x → ¬ F.Adj x y) ∧
  45       ∀ x, ((Finset.univ.filter (fun y => M x y ≠ sampledOrientation G (fun e => (σ e).2) x y ∨
  46         M y x ≠ sampledOrientation G (fun e => (σ e).2) y x)).card : ℝ) ≤
  47           4 * (Fintype.card V : ℝ) ^ (4 / 5 : ℝ)
