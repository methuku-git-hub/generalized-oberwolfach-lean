   9 theorem cycle_sampling_almost_regular_parameters
  10     (B θ c:ℝ) (hB:1≤B) (hθ:0<θ) (hθsmall:θ≤1/20) (hc:0≤c):
  11     ∃ N:ℕ,∀ n:ℕ,N≤n→∀ d:ℝ,(n:ℝ)/2≤d→d≤n→
  12       let scale:ℝ:=n/(2*B);
  13       ∃ D:ℕ,n≤D ∧ D≤n^2 ∧ c*n≤(8*B*c)*D/n ∧
  14         ∀ deg:ℝ,|deg-d|≤θ*(n:ℝ)/4→
  15           (1-10*θ)*D≤(1-θ)*(scale*deg/2) ∧
  16           (1+θ)*(scale*deg/2)≤D := by
  17   have hBpos:0<B := by linarith
  18   let N:=Nat.ceil (8*B*(1+1/θ)+1)
  19   refine ⟨N,?_⟩
  20   intro n hn d hdlo hdhi scale
  21   let T:ℝ:=scale*d/2
  22   have hN : 8*B*(1+1/θ)+1≤(n:ℝ) :=
  23     (Nat.le_ceil _).trans (by exact_mod_cast hn)
  24   have hid : 8*B*(1+1/θ) = 8*B+8*B/θ := by ring
  25   rw [hid] at hN
  26   have hn1:1≤(n:ℝ) := by
  27     have hh:0≤8*B+8*B/θ := by positivity
  28     linarith
  29   have hnpos:0<(n:ℝ) := by linarith
  30   have hnB:8*B≤(n:ℝ) := by
  31     have hh:0≤8*B/θ := by positivity
  32     nlinarith
  33   have hnTheta:8*B/θ≤(n:ℝ) := by
  34     have hh:0≤8*B := by positivity
  35     nlinarith
  36   have hscale:0≤scale := by dsimp [scale];positivity
  37   have hdpos:0<d := by linarith
  38   have hT:0≤T := by dsimp [T]; positivity
  39   have hlo:n^2/(8*B)≤T := by
  40     have hh:=mul_le_mul_of_nonneg_left hdlo hscale
  41     have hid:scale*((n:ℝ)/2)/2=(n:ℝ)^2/(8*B) := by
  42       dsimp [scale];field_simp;ring
  43     rw [←hid]
  44     dsimp [T]
  45     linarith only [hh]
  46   have hhi:T≤(n:ℝ)^2/(4*B) := by
  47     have hh:=mul_le_mul_of_nonneg_left hdhi hscale
  48     have hid:scale*(n:ℝ)/2=(n:ℝ)^2/(4*B) := by
  49       dsimp [scale];field_simp;ring
  50     rw [←hid]
  51     dsimp [T]
  52     linarith only [hh]
  53   have hlarge:1≤θ*T := by
  54     have hdiv: (n:ℝ)/(8*B)≥1/θ := (le_div_iff₀ (by positivity)).2 (by
  55       have heq:(1/θ)*(8*B)=8*B/θ := by ring
  56       rw [heq]
  57       exact hnTheta)
  58     have hquad:(n:ℝ)/(8*B)≤(n:ℝ)^2/(8*B) := by
  59       apply div_le_div_of_nonneg_right _ (by positivity)
  60       nlinarith
  61     have hbase:1/θ≤T := hdiv.trans (hquad.trans hlo)
  62     have hh':=mul_le_mul_of_nonneg_left hbase hθ.le
  63     have : θ*(1/θ)=1 := by field_simp
  64     linarith
  65   obtain ⟨D,hDlo,hDhi,hrows⟩:=cycle_sampling_integer_degree θ T hθ.le (by linarith) hT hlarge
  66   have hDreal:(n:ℝ)≤D := by
  67     have hdiv:(1:ℝ)≤(n:ℝ)/(8*B):=(le_div_iff₀ (by positivity)).2 (by linarith)
  68     have hh:=mul_le_mul_of_nonneg_left hdiv hnpos.le
  69     have hid:(n:ℝ)*((n:ℝ)/(8*B))=(n:ℝ)^2/(8*B) := by ring
  70     have hh' : (n:ℝ)≤(n:ℝ)^2/(8*B) := by rw [←hid];linarith
  71     have hTlo:T≤(1+2*θ)*T := by nlinarith
  72     exact hh'.trans (hlo.trans (hTlo.trans hDlo))
  73   have hDn:n≤D := by exact_mod_cast hDreal
  74   have hDsq:(D:ℝ)≤(n:ℝ)^2 := by
  75     have ht:T≤(n:ℝ)^2/4 := by
  76       have hBden:4≤4*B:=by linarith
  77       exact hhi.trans (div_le_div_of_nonneg_left (sq_nonneg _) (by positivity) hBden)
  78     have hm:=mul_le_mul_of_nonneg_left ht (show 0≤1+3*θ by positivity)
  79     nlinarith [sq_nonneg (n:ℝ)]
  80   have hDnat:D≤n^2 := by exact_mod_cast hDsq
  81   have hcode:c*n≤(8*B*c)*D/n := by
  82     have hDquad:(n:ℝ)^2/(8*B)≤D := hlo.trans (by
  83       have hh:T≤(1+2*θ)*T:=by nlinarith
  84       exact hh.trans hDlo)
  85     have hh:=mul_le_mul_of_nonneg_left hDquad (show 0≤8*B*c by positivity)
  86     have hd: (8*B*c)*((n:ℝ)^2/(8*B))/(n:ℝ)=c*n := by
  87       field_simp
  88     rw [←hd]
  89     exact div_le_div_of_nonneg_right hh hnpos.le
  90   refine ⟨D,hDn,hDnat,hcode,?_⟩
  91   intro deg hdeg
  92   obtain ⟨hdeglo,hdeghi⟩:=abs_le.mp hdeg
  93   have hdtheta:θ*(n:ℝ)/4≤θ*d/2 := by
  94     have hh:=mul_le_mul_of_nonneg_left hdlo hθ.le
  95     nlinarith
  96   have hdeglo':(1-θ/2)*d≤deg := by nlinarith
  97   have hdeghi':deg≤(1+θ/2)*d := by nlinarith
  98   apply hrows (scale*deg/2)
  99   · have hh:=mul_le_mul_of_nonneg_left hdeglo' hscale
 100     dsimp [T];nlinarith only [hh]
 101   · have hh:=mul_le_mul_of_nonneg_left hdeghi' hscale
 102     dsimp [T];nlinarith only [hh]
 103 end Oberwolfach
