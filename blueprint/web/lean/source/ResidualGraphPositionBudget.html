   9 theorem two_factor_residual_position_edge_budget
  10     (F:SimpleGraph V) (hF:F.IsRegularOfDegree 2)
  11     (s:Finset I) (u v:I→V) (P:∀ i,F.Walk (u i) (v i))
  12     (hp:∀i∈s,(P i).IsPath) (hne:∀i∈s,u i≠v i)
  13     (hP:(↑s:Set I).PairwiseDisjoint (fun i=>(P i).support.toFinset))
  14     (t:Finset J) (root:J→V) (C:∀ j,F.Walk (root j) (root j))
  15     (hc:∀j∈t,(C j).IsCycle)
  16     (hC:(↑t:Set J).PairwiseDisjoint (fun j=>(C j).support.toFinset))
  17     (hcross:∀i∈s,∀j∈t,Disjoint (P i).support.toFinset (C j).support.toFinset)
  18     (A:Finset V) (hAu:∀i∈s,u i∈A) (hAv:∀i∈s,v i∈A):
  19     let deletedV := (t.biUnion (fun j=>(C j).support.toFinset)) ∪
  20       (s.biUnion (fun i=>(P i).support.toFinset \ {u i,v i}))
  21     let deletedE := (t.biUnion (fun j=>(C j).edges.toFinset)) ∪
  22       (s.biUnion (fun i=>(P i).edges.toFinset))
  23     let Q:=F.deleteEdges (deletedE:Set (Sym2 V));
  24     (Finset.univ\deletedV).card≤Q.edgeFinset.card+A.card ∧
  25       ∀ x∈deletedV,∀ y,¬Q.Adj x y := by
  26   have hr:=two_factor_residual_graph_after_disjoint_deletions F hF s u v P hp hne hP t root C hc hC hcross
  27   have ha:=disjoint_walk_endpoints_card_le F s u v P hne hP A hAu hAv
  28   dsimp only at hr ⊢
  29   exact ⟨by omega,hr.2⟩
  30 end Oberwolfach
