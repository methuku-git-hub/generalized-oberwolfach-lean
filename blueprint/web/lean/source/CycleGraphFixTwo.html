  11 theorem cycleGraph_injective_hom_fix_zero_one {r:ℕ} (hr:3≤r) (σ:Fin r→Fin r)
  12     (hinj:Function.Injective σ)
  13     (hmap:∀ i j,(SimpleGraph.cycleGraph r).Adj i j→(SimpleGraph.cycleGraph r).Adj (σ i) (σ j))
  14     (h0:σ ⟨0,by omega⟩=⟨0,by omega⟩) (h1:σ ⟨1,by omega⟩=⟨1,by omega⟩):∀ i,σ i=i:=by
  15   intro i
  16   induction hi:i.val using Nat.strong_induction_on generalizing i with
  17   | h n IH =>
  18     by_cases hz:n=0
  19     · have eq0:i=⟨0,by omega⟩:=Fin.ext (by simpa [hi] using hz)
  20       simpa [eq0] using h0
  21     by_cases ho:n=1
  22     · have eq1:i=⟨1,by omega⟩:=Fin.ext (by simpa [hi] using ho)
  23       simpa [eq1] using h1
  24     have hn:2≤n:=by omega
  25     let j:Fin r:=⟨n-1,by have:=i.isLt;omega⟩
  26     let p:Fin r:=⟨n-2,by have:=i.isLt;omega⟩
  27     have hj:σ j=j:=IH (n-1) (by omega) j rfl
  28     have hp:σ p=p:=IH (n-2) (by omega) p rfl
  29     have hjnext:j.val+1<r:=by dsimp [j];have:=i.isLt;omega
  30     have hsource:(SimpleGraph.cycleGraph r).Adj j i:=by
  31       rw [cycleGraph_modular_adj (by omega)]
  32       left
  33       rw [Nat.mod_eq_of_lt hjnext]
  34       dsimp [j]
  35       omega
  36     have htarget:=hmap j i hsource
  37     rw [hj,cycleGraph_modular_adj (by omega)] at htarget
  38     rcases htarget with hfor|hback
  39     · apply Fin.ext
  40       rw [Nat.mod_eq_of_lt hjnext] at hfor
  41       dsimp [j] at hfor
  42       omega
  43     · have hlt:(σ i).val+1<r:=by
  44         by_contra hnot
  45         have hEq:(σ i).val+1=r:=by have:=(σ i).isLt;omega
  46         rw [hEq,Nat.mod_self] at hback
  47         dsimp [j] at hback
  48         omega
  49       rw [Nat.mod_eq_of_lt hlt] at hback
  50       have he:σ i=σ p:=by
  51         rw [hp]
  52         apply Fin.ext
  53         dsimp [j,p] at *
  54         omega
  55       have :=hinj he
  56       have hv:=congrArg Fin.val this
  57       dsimp [p] at hv
  58       omega
  59 end Oberwolfach
