   3 /-! The physical vertex set of an ordinary wheel includes its rim vertices
   4 in the first summand and its hub in the second. The deep cover has exactly
   5 one prescribed root per wheel. -/

  11 attribute [local instance] Classical.propDecidable
  12 variable {V U:Type} [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U]
  13   {Q F:SimpleGraph V} {H:V→U→Prop} {r nh nr:ℕ}

  15 def ordinaryWheelVertices (W:OrdinaryWheel Q F H r):Finset (V⊕U):=
  16   (ordinaryWheelRim W).support.toFinset.image Sum.inl∪{Sum.inr (ordinaryWheelHub W)}

  18 theorem ordinaryWheelVertices_rim (W:OrdinaryWheel Q F H r) (v:V):
  19     Sum.inl v∈ordinaryWheelVertices W↔v∈(ordinaryWheelRim W).support:=by
  20   simp only [ordinaryWheelVertices,Finset.mem_union,Finset.mem_image,Set.mem_toFinset,
  21     Finset.mem_singleton,Sum.inl_ne_inr,or_false,Sum.inl.injEq,exists_eq_right]
  22 
  23 theorem ordinaryWheelVertices_hub (W:OrdinaryWheel Q F H r) (a:U):
  24     Sum.inr a∈ordinaryWheelVertices W↔a=ordinaryWheelHub W:=by
  25   simp only [ordinaryWheelVertices,Finset.mem_union,Finset.mem_image,Sum.inl_ne_inr,and_false,
  26     exists_false,false_or,Finset.mem_singleton,Sum.inr.injEq]

  28 theorem ordinary_deep_wheel_unique_root (W:OrdinaryWheel Q F H r)
  29     (I:Finset V) (J:Finset U) (u:Fin nh→U) (x:Fin nr→V)
  30     (hu:∀ i,u i∈J) (hx:∀ i,x i∈I)
  31     (hsupport:
  32       (∃ i,ordinaryWheelHub W=u i ∧ ∀ v∈(ordinaryWheelRim W).support,v∉I) ∨
  33       (∃ i,x i∈(ordinaryWheelRim W).support ∧ ordinaryWheelHub W∉J ∧
  34         ∀ v∈(ordinaryWheelRim W).support,v=x i∨v∉I)):
  35     (ordinaryWheelVertices W∩((Finset.univ.image (fun i=>Sum.inl (x i)))∪
  36       (Finset.univ.image (fun i=>Sum.inr (u i))))).card=1:=by
  37   let Roots:Finset (V⊕U):=Finset.univ.image (fun i=>Sum.inl (x i))∪Finset.univ.image (fun i=>Sum.inr (u i))
  38   have hR (v:V):Sum.inl v∈Roots↔∃ i,x i=v:=by
  39     simp only [Roots,Finset.mem_union,Finset.mem_image,Finset.mem_univ,true_and,
  40       Sum.inl.injEq,Sum.inr_ne_inl,exists_false,or_false]
  41   have hU (a:U):Sum.inr a∈Roots↔∃ i,u i=a:=by
  42     simp only [Roots,Finset.mem_union,Finset.mem_image,Finset.mem_univ,true_and,
  43       Sum.inr.injEq,Sum.inl_ne_inr,exists_false,false_or]
  44   change (ordinaryWheelVertices W∩Roots).card=1
  45   rcases hsupport with ⟨i,hi,hs⟩|⟨i,hxi,hi,hs⟩
  46   · have he:ordinaryWheelVertices W∩Roots={Sum.inr (u i)}:=by
  47       ext z
  48       cases z with
  49       | inl v=>
  50         simp only [Finset.mem_inter,ordinaryWheelVertices_rim,hR,Finset.mem_singleton,Sum.inl_ne_inr,iff_false,not_and]
  51         intro hv hroot
  52         obtain ⟨k,rfl⟩:=hroot
  53         exact hs (x k) hv (hx k)
  54       | inr a=>
  55         simp only [Finset.mem_inter,ordinaryWheelVertices_hub,hU,Finset.mem_singleton,Sum.inr.injEq,hi]
  56         constructor
  57         · exact fun h=>h.1
  58         · intro ha;exact ⟨ha,⟨i,ha.symm⟩⟩
  59     rw [he,Finset.card_singleton]
  60   · have he:ordinaryWheelVertices W∩Roots={Sum.inl (x i)}:=by
  61       ext z
  62       cases z with
  63       | inl v=>
  64         simp only [Finset.mem_inter,ordinaryWheelVertices_rim,hR,Finset.mem_singleton,Sum.inl.injEq]
  65         constructor
  66         · rintro ⟨hv,k,hk⟩
  67           exact (hs v hv).resolve_right (fun h=>h (hk ▸ hx k))
  68         · rintro rfl;exact ⟨hxi,i,rfl⟩
  69       | inr a=>
  70         simp only [Finset.mem_inter,ordinaryWheelVertices_hub,hU,Finset.mem_singleton,Sum.inr_ne_inl,iff_false,not_and]
  71         intro ha hroot
  72         obtain ⟨k,hk⟩:=hroot
  73         exact hi (ha ▸ hk ▸ hu k)
  74     rw [he,Finset.card_singleton]
