  14 def ordinaryNeighbourTransfer {V U : Type*} [Fintype V]
  15     {r : ℕ} (hr : 3≤r) (Q F : SimpleGraph V) (H : V → U → Prop)
  16     (v : V) (i j : {u // H v u}) (D : OrdinaryWheel Q F H r) : ℝ :=
  17   if i.val=j.val then 0 else ordinaryAverageTransfer hr Q F H v i.val j.val D

  19 theorem ordinaryNeighbourTransfer_loads {V U : Type*} [Fintype V] [Fintype U]
  20     [DecidableEq V] [DecidableEq U]
  21     {r : ℕ} (hr : 3≤r) (Q F : SimpleGraph V) (H : V → U → Prop)
  22     (v : V) (i j : {u // H v u})
  23     (hT : ∀ i j : {u // H v u},i.val≠j.val →
  24       0<Fintype.card (OrdinaryTransferConfiguration hr Q F H v i.val j.val)) :
  25     (∀ x y,ordinaryRimLoad (ordinaryNeighbourTransfer hr Q F H v i j) x y=0) ∧
  26     (∀ z h,ordinarySpokeLoad (ordinaryNeighbourTransfer hr Q F H v i j) z h=
  27       (if z=v then (1:ℝ) else 0)*((if h=i.val then 1 else 0)-(if h=j.val then 1 else 0))) := by
  28   by_cases hij : i.val=j.val
  29   · constructor
  30     · intro x y
  31       simp [ordinaryNeighbourTransfer,hij,ordinaryRimLoad]
  32     · intro z h
  33       simp [ordinaryNeighbourTransfer,hij,ordinarySpokeLoad]
  34   · have he : ordinaryNeighbourTransfer hr Q F H v i j =
  35         ordinaryAverageTransfer hr Q F H v i.val j.val := by
  36       funext D
  37       exact if_neg hij
  38     rw [he]
  39     exact ordinaryAverageTransfer_loads hr Q F H v i.val j.val (hT i j hij)

  41 theorem ordinaryNeighbourTransfer_bound {V U : Type*} [Fintype V] [DecidableEq V]
  42     {r : ℕ} (hr : 3≤r) (Q F : SimpleGraph V) (H : V → U → Prop)
  43     (v : V) (i j : {u // H v u}) (γ : ℝ) (hγ : 0<γ) (hn : 0<Fintype.card V)
  44     (hT : ∀ i j : {u // H v u},i.val≠j.val →
  45       γ*(Fintype.card V:ℝ)^(r^2)≤
  46         (Fintype.card (OrdinaryTransferConfiguration hr Q F H v i.val j.val):ℝ))
  47     (D : OrdinaryWheel Q F H r) :
  48     |ordinaryNeighbourTransfer hr Q F H v i j D|≤
  49       if v∈(ordinaryWheelRim D).support then
  50         (6*((r:ℝ)^2)^r/γ)/(Fintype.card V:ℝ)^(r-1)
  51       else (6*((r:ℝ)^2)^r/γ)/(Fintype.card V:ℝ)^r := by
  52   by_cases hij : i.val=j.val
  53   · simp only [ordinaryNeighbourTransfer,if_pos hij,abs_zero]
  54     split_ifs <;> positivity
  55   · simpa only [ordinaryNeighbourTransfer,if_neg hij] using
  56       ordinaryAverageTransfer_coefficient_bound hr Q F H v i.val j.val γ hγ hn (hT i j hij) D
