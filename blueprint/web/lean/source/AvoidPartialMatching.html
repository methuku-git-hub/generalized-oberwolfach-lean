  13  theorem injection_avoiding_partial_matching {X Y:Type*} [Fintype X] [Fintype Y]
  14     (Bad:X→Y→Prop)
  15     (hrow:∀ x,(Finset.univ.filter (Bad x)).card≤1)
  16     (hcol:∀ y,(Finset.univ.filter (fun x=>Bad x y)).card≤1)
  17     (hsize:Fintype.card X≤Fintype.card Y)
  18     (hnotone:2≤Fintype.card Y ∨ Fintype.card X=0):
  19     ∃ f:X↪Y,∀ x,¬Bad x (f x):=by
  20   classical
  21   let C:X→Finset Y:=fun x=>Finset.univ.filter (fun y=>¬Bad x y)
  22   have hhall (S:Finset X):S.card≤(S.biUnion C).card:=by
  23     rcases S.eq_empty_or_nonempty with rfl|hS
  24     · simp
  25     obtain ⟨x,hx⟩:=hS
  26     have hY:2≤Fintype.card Y:=by
  27       rcases hnotone with h|h
  28       · exact h
  29       · have hp:=Fintype.card_pos_iff.mpr (show Nonempty X from ⟨x⟩)
  30         omega
  31     by_cases hSone:S.card=1
  32     · have hc:1≤(C x).card:=by
  33         have hh:=Finset.card_filter_add_card_filter_not (s:=Finset.univ) (p:=Bad x)
  34         have hr:=hrow x
  35         simp only [Finset.card_univ] at hh
  36         dsimp [C]
  37         omega
  38       have hs:C x⊆S.biUnion C:=fun y hy=>Finset.mem_biUnion.mpr ⟨x,hx,hy⟩
  39       rw [hSone]
  40       exact hc.trans (Finset.card_le_card hs)
  41     · have htwo:2≤S.card:=by have hp:=Finset.card_pos.mpr ⟨x,hx⟩;omega
  42       have hfull:S.biUnion C=Finset.univ:=by
  43         apply Finset.eq_univ_of_forall
  44         intro y
  45         by_contra hy
  46         have hs:S⊆Finset.univ.filter (fun x=>Bad x y):=by
  47           intro z hz
  48           apply Finset.mem_filter.mpr
  49           refine ⟨Finset.mem_univ _,?_⟩
  50           by_contra hbad
  51           exact hy (Finset.mem_biUnion.mpr ⟨z,hz,Finset.mem_filter.mpr ⟨Finset.mem_univ _,hbad⟩⟩)
  52         have hh:=Finset.card_le_card hs
  53         have hc:=hcol y
  54         omega
  55       rw [hfull,Finset.card_univ]
  56       exact (Finset.card_le_univ S).trans hsize
  57   obtain ⟨f,hf,hfC⟩:=(Finset.all_card_le_biUnion_card_iff_existsInjective' C).mp hhall
  58   exact ⟨⟨f,hf⟩,fun x=>(Finset.mem_filter.mp (hfC x)).2⟩

  60  theorem bijection_avoiding_partial_matching {X Y:Type*} [Fintype X] [Fintype Y]
  61     (Bad:X→Y→Prop)
  62     (hrow:∀ x,(Finset.univ.filter (Bad x)).card≤1)
  63     (hcol:∀ y,(Finset.univ.filter (fun x=>Bad x y)).card≤1)
  64     (hsize:Fintype.card X=Fintype.card Y)
  65     (hnotone:Fintype.card Y≠1):
  66     ∃ f:X≃Y,∀ x,¬Bad x (f x):=by
  67   obtain ⟨f,hf⟩:=injection_avoiding_partial_matching Bad hrow hcol (Nat.le_of_eq hsize)
  68     (by omega)
  69   exact ⟨Equiv.ofBijective f ((Fintype.bijective_iff_injective_and_card f).mpr ⟨f.injective,hsize⟩),hf⟩
