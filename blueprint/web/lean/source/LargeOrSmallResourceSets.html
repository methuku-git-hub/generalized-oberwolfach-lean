   9 theorem all_resource_sets_from_large_sets
  10     (S:V→Finset W) (covered:Finset W) (n a γ B:ℝ)
  11     (hn:0≤n) (hγ:0≤γ) (hB:0≤B)
  12     (hsize:∀v,((S v).card:ℝ)≤B*n)
  13     (hlarge:∀v,a*n≤(S v).card → (((S v)\covered).card:ℝ)≤γ*(S v).card):
  14     ∀v,(((S v)\covered).card:ℝ)≤max a (γ*B)*n := by
  15   intro v
  16   have hsubset:((S v\covered).card:ℝ)≤(S v).card := by
  17     exact_mod_cast (Finset.card_le_card (Finset.sdiff_subset : S v\covered ⊆ S v))
  18   have hamax:a≤max a (γ*B):=le_max_left _ _
  19   have hbmax:γ*B≤max a (γ*B):=le_max_right _ _
  20   by_cases hv:a*n≤(S v).card
  21   · have hh:=hlarge v hv
  22     have hh' : γ*((S v).card:ℝ)≤γ*(B*n) := mul_le_mul_of_nonneg_left (hsize v) hγ
  23     have hh'' : γ*B*n≤max a (γ*B)*n := mul_le_mul_of_nonneg_right hbmax hn
  24     nlinarith
  25   · have hh : (S v).card≤a*n := le_of_lt (lt_of_not_ge hv)
  26     have hh' := mul_le_mul_of_nonneg_right hamax hn
  27     linarith
  28 end Oberwolfach
