  11 theorem prescribed_selected_induced_arcs (Q:SimpleGraph X) (D:X→X→Prop) (A:Finset X)
  12     (hDQ:∀x y,D x y→Q.Adj x y) (horiented:∀x y,D x y→¬D y x)
  13     (hA:∀x∈A,∀y∈A,¬Q.Adj x y)
  14     (es:List (X×X)) (vs:List X)
  15     (hE:∀e∈es,D e.1 e.2 ∧ e.1∉prescribedClosedNeighbourhoods Q A ∧ e.2∉prescribedClosedNeighbourhoods Q A)
  16     (hpair:es.Pairwise (prescribedPairsApart Q))
  17     (hvs:∀v∈vs,v∉prescribedClosedNeighbourhoods Q A)
  18     (hcross:∀ e∈es,∀v∈vs,v∉prescribedPairClosed Q e)
  19     (hsingle:vs.Pairwise (prescribedSinglesApart Q)):
  20     ∀ x y,D x y→x∈A∪(es.flatMap prescribedPairVertices++vs).toFinset→
  21       y∈A∪(es.flatMap prescribedPairVertices++vs).toFinset→(x,y)∈es := by
  22   intro x y hxy hx hy
  23   have hx':x∈A ∨ x∈es.flatMap prescribedPairVertices++vs:=by
  24     rcases Finset.mem_union.mp hx with hx|hx
  25     · exact Or.inl hx
  26     · exact Or.inr (List.mem_toFinset.mp hx)
  27   have hy':y∈A ∨ y∈es.flatMap prescribedPairVertices++vs:=by
  28     rcases Finset.mem_union.mp hy with hy|hy
  29     · exact Or.inl hy
  30     · exact Or.inr (List.mem_toFinset.mp hy)
  31   have hout:∀v∈es.flatMap prescribedPairVertices++vs,v∉prescribedClosedNeighbourhoods Q A:=by
  32     intro v hv
  33     rcases List.mem_append.mp hv with hv|hv
  34     · obtain ⟨e,he,hv⟩:=List.mem_flatMap.mp hv
  35       simp only [prescribedPairVertices,List.mem_cons,List.not_mem_nil,or_false] at hv
  36       rcases hv with hv|hv
  37       · subst v;exact (hE e he).2.1
  38       · subst v;exact (hE e he).2.2
  39     · exact hvs v hv
  40   rcases hx' with hx|hx <;> rcases hy' with hy|hy
  41   · exact (hA x hx y hy (hDQ x y hxy)).elim
  42   · exact ((prescribed_outside_closed_neighbourhood Q A y (hout y hy)).2 x hx (hDQ x y hxy)).elim
  43   · exact ((prescribed_outside_closed_neighbourhood Q A x (hout x hx)).2 y hy (hDQ x y hxy).symm).elim
