  16 theorem count_error_after_comparing_target (a b target p err : ℝ)
  17     (hp : 0≤p) (hp1 : p≤1) (herr : 0≤err)
  18     (ha : |a-p*target|≤err) (hb : |b-target|≤err) : |a-p*b|≤2*err := by
  19   obtain ⟨ha₁,ha₂⟩ := abs_le.mp ha
  20   obtain ⟨hb₁,hb₂⟩ := abs_le.mp hb
  21   have h₁ := mul_le_mul_of_nonneg_left hb₁ hp
  22   have h₂ := mul_le_mul_of_nonneg_left hb₂ hp
  23   have h₃ := mul_le_mul_of_nonneg_right hp1 herr
  24   apply abs_le.mpr
  25   constructor <;> nlinarith only [ha₁,ha₂,h₁,h₂,h₃]

  27 theorem marked_endpoint_index_estimates
  28     (π : Equiv.Perm V) (d h : ℕ) (hh : 3≤h) (B : Finset U)
  29     (q : U→Γ→ℝ) (s : U→Color→ℝ) (g : Γ) (j : Color) (err : ℝ) (herr : 0≤err)
  30     (σ : RawMarkedOutcome V U Γ Color) (H : Bool→V→U→Prop)
  31     (hi : MarkedIndicesWithSpokes π d h B q s g j err σ H)
  32     (w : Bool→ℝ) (hq : ∀ u∈B,q u g=w true) (hs : ∀ u∈B,s u j=w false)
  33     (hw : ∀ t,0≤w t ∧ w t≤1) (r t : Bool) (x : V) :
  34     let Y := B.filter (H r x)
  35     (|((Y.card:ℝ)-w r*B.card)|≤err) ∧
  36     (∀ z,¬PermutationClose π (3*d) x z →
  37       |((Y.filter (H t z)).card:ℝ)-w t*Y.card|≤2*err) ∧
  38     (∀ z z',¬PermutationClose π (3*d) x z → ¬PermutationClose π (3*d) x z' →
  39       ¬PermutationClose π (3*d) z z' →
  40       ((Y.filter (fun u=>H t z u ∧ H t z' u)).card:ℝ)≤(w t)^2*Y.card+2*err) := by
  41   let Y := B.filter (H r x)
  42   have hsingle : |(Y.card:ℝ)-w r*B.card|≤err := by
  43     have he := marked_index_tuple_estimate π d h B q s g j err σ H hi w hq hs
  44       (fun _ : Fin 1=>x) (fun _=>r) (by intro a b _;exact Subsingleton.elim _ _)
  45       (by simpa using (show 1≤h by omega)) (by intro a b hne;exact (hne (Subsingleton.elim _ _)).elim)
  46     simpa [Y] using he.le
  47   refine ⟨hsingle,?_,?_⟩
  48   · intro z hxz
  49     have hne : x≠z := fun he=>hxz (Or.inl he)
  50     have he := marked_index_tuple_estimate π d h B q s g j err σ H hi w hq hs
  51       ![x,z] ![r,t] (by intro a b he;fin_cases a <;> fin_cases b <;> simp_all)
  52       (by simpa using (show 2≤h by omega)) (by
  53         intro a b hab
  54         fin_cases a <;> fin_cases b <;> simp_all [permutationClose_symm])
  55     have he' : |((Y.filter (H t z)).card:ℝ)-w t*(w r*B.card)|≤err := by
  56       simpa [Y,Finset.filter_filter,Fin.forall_fin_two,mul_left_comm,mul_assoc] using he.le
  57     exact count_error_after_comparing_target _ _ _ _ err (hw t).1 (hw t).2 herr he' hsingle
  58   · intro z z' hxz hxz' hzz'
  59     have hne : x≠z := fun he=>hxz (Or.inl he)
  60     have hne' : x≠z' := fun he=>hxz' (Or.inl he)
  61     have hzz : z≠z' := fun he=>hzz' (Or.inl he)
  62     have he := marked_index_tuple_estimate π d h B q s g j err σ H hi w hq hs
  63       ![x,z,z'] ![r,t,t] (by intro a b he;fin_cases a <;> fin_cases b <;> simp_all)
  64       (by simpa using hh) (by
  65         intro a b hab
  66         fin_cases a <;> fin_cases b <;> simp_all [permutationClose_symm])
  67     have he' : |((Y.filter (fun u=>H t z u ∧ H t z' u)).card:ℝ)-(w t)^2*(w r*B.card)|≤err := by
  68       have heq : (w r*(w t*w t))*B.card=(w t)^2*(w r*B.card) := by ring
  69       simpa [Y,Finset.filter_filter,Fin.forall_fin_succ,Fin.prod_univ_succ,pow_two,mul_assoc,mul_comm,mul_left_comm] using he.le
  70     have hp2 : (w t)^2≤1 := by nlinarith [(hw t).1,(hw t).2]
  71     have hc := count_error_after_comparing_target _ _ _ _ err (sq_nonneg _) hp2 herr he' hsingle
  72     exact (by linarith only [(abs_le.mp hc).2])
