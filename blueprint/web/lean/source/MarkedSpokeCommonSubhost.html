  12 theorem marked_spoke_common_subhost_deletion
  13     (H H':Bool→V→U→Prop) (I:Finset V) (J:Finset U)
  14     (htypes:∀ v u,¬(H false v u∧H true v u))
  15     (hinner:∀ b v u,v∈I→u∈J→(H b v u↔H' b v u)):
  16     ∃ D:SimpleGraph (V⊕U),
  17       (∀ b v u,markedDeletePhysicalSpokes H D b v u↔H b v u∧H' b v u) ∧
  18       (∀ v∈I,∀ u∈J,¬D.Adj (Sum.inl v) (Sum.inr u)) ∧
  19       (∀ v,D.degree (Sum.inl v)=(Finset.univ.filter (fun u=>∃ b,H b v u∧¬H' b v u)).card) ∧
  20       (∀ u,D.degree (Sum.inr u)=(Finset.univ.filter (fun v=>∃ b,H b v u∧¬H' b v u)).card) :=by
  21   let S:Finset (V×U):=Finset.univ.filter (fun z=>∃ b,H b z.1 z.2∧¬H' b z.1 z.2)
  22   let D:=markedFiniteSpokeGraph S
  23   refine ⟨D,?_,?_,?_,?_⟩
  24   · intro b v u
  25     simp only [markedDeletePhysicalSpokes,D,markedFiniteSpokeGraph_adj,S,Finset.mem_filter,Finset.mem_univ,true_and]

  28   · intro v hv u hu
