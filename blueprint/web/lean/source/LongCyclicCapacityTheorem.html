  20     ∃ b₀ b₁ : ℝ,0<b₀ ∧ 0<b₁ ∧
  21     ∀ n : ℕ,1≤n →
  22     ∀ (X : Type u) [Fintype X],Fintype.card X=n →
  23     ∀ (a : Fin c → X → X → ℝ) (d : Fin c → X → ℝ),
  24     (∀ i x y,0≤a i x y) →
  25     (∀ i x,∑ y,a i x y=d i x) →
  26     (∀ i y,∑ x,a i x y=d (finRotate c i) y) →
  27     (∀ i x y,0<a i x y → a₀ ≤ a i x y ∧ a i x y ≤ a₁) →
  28     (∀ i x,d₀*n ≤ d i x ∧ d i x ≤ d₁*n) →
  29     ∀ J : Finset (Fin c),J.card≤1 →
  30     (∀ i,i∉J → ∀ x y,cyclicCapacityTransition a d i x y ≤
  31       B*(d (finRotate c i) y/(∑ z,d ⟨0,by omega⟩ z))) →
  32     (∀ i,i∉J → finRotate c i∉J → finRotate c (finRotate c i)∉J →
  33       ∀ x z,κ*(d (finRotate c (finRotate c (finRotate c i))) z/(∑ z,d ⟨0,by omega⟩ z)) ≤
  34         ∑ y,∑ w,cyclicCapacityTransition a d i x y*cyclicCapacityTransition a d (finRotate c i) y w*
  35           cyclicCapacityTransition a d (finRotate c (finRotate c i)) w z) →
  36     ∃ wC : DirectedTransversalCycle a → ℝ,

  38       (∀ i x y,(∑ C,if C.val i=x ∧ C.val (finRotate c i)=y then wC C else 0)=a i x y) ∧
  39       (∀ C,b₀/(n:ℝ)^(c-2) ≤ wC C ∧ wC C ≤ b₁/(n:ℝ)^(c-2)) := by
