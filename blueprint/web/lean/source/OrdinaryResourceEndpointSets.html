  13     (hr:∀ v w,Sum.inl s(v,w)∈K→v∈N∧w∈N)
  14     (hs:∀ v u,Sum.inr (v,u)∈K→v∈N) : ordinaryResourceRows K⊆N := by
  15   intro v hv
  16   obtain ⟨e,he,hv⟩:=Finset.mem_biUnion.mp hv
  17   cases e with
  18   | inl e=>
  19     induction e using Sym2.ind with
  20     | _ a b=>
  21       have hv':v=a∨v=b:=by simpa using hv
  22       rcases hv' with h|h
  23       · rw [h];exact (hr a b he).1
  24       · rw [h];exact (hr a b he).2
  25   | inr p=>
  26     have hv':v=p.1:=by simpa using hv
  27     rw [hv'];exact hs p.1 p.2 he
