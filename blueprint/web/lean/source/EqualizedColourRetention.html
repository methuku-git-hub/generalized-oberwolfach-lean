  13 def FiniteLaw.uniformOn {C : Type*} [Fintype C] (L : Finset C)
  14     (hL : L.Nonempty) : FiniteLaw C where
  15   mass c := if c∈L then 1/(L.card : ℝ) else 0
  16   nonneg c := by split_ifs <;> positivity

  21     have hc : (L.card : ℝ)≠0 := by exact_mod_cast hL.card_pos.ne'

  47 theorem equalizationCoin_bounds
  48     {V C : Type*} [Fintype V] (G : SimpleGraph V)
  49     (L : V → Finset C) {l : ℕ} (hl : 0<l) (K : ℕ) (v : V) (c : C) :
  50     0≤equalizationCoin G L l K v c ∧ equalizationCoin G L l K v c≤1 := by
  51   have hlr : (1 : ℝ)≤l := by exact_mod_cast hl
  52   have hp : (1 : ℝ)/(l : ℝ)≤1 := (div_le_one (by positivity)).mpr hlr
  53   have hq : 0≤1-1/(l : ℝ) := sub_nonneg.mpr hp
  54   have hq1 : 1-1/(l : ℝ)≤1 := sub_le_self _ (by positivity)
  55   exact ⟨pow_nonneg hq _,pow_le_one₀ hq hq1⟩
