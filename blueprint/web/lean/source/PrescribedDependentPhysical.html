   1 import Oberwolfach.PrescribedDependentSourceModel
   2 /-! Relabelling separately typed source pieces preserves their physical
   3 undirected edges and source-graph adjacency. -/
   4 set_option autoImplicit false
   5 noncomputable section
   6 namespace Oberwolfach
   7 attribute [local instance] Classical.propDecidable
   8 variable {I:Type} [Fintype I] [DecidableEq I] {X:I→Type} [∀ i,Fintype (X i)] [∀ i,DecidableEq (X i)]

  10 def prescribedDependentPhysical (pieces:∀ i,List (PrescribedSourcePiece (X i))) (i:I):Set (Sym2 (X i)):=
  11   {e | ∃ p∈pieces i,e∈p.edges}
  12 
  13 def prescribedDependentGraph (pieces:∀ i,List (PrescribedSourcePiece (X i))) (i:I):SimpleGraph (X i):=
  14   SimpleGraph.fromEdgeSet (prescribedDependentPhysical pieces i)

  16 theorem prescribed_lifted_physical_membership (pieces:∀ i,List (PrescribedSourcePiece (X i)))
  17     (i:I) (e:Sym2 (X i)):
  18     Sym2.map (Sigma.mk i) e∈prescribedSourcePhysical (prescribedLiftDependentPieces Finset.univ.toList pieces) i ↔
  19       e∈prescribedDependentPhysical pieces i := by
  20   constructor
  21   · rintro ⟨p,hp,hpi,he⟩
  22     obtain ⟨j,hj,hp⟩:=List.mem_flatMap.mp hp
  23     obtain ⟨q,hq,rfl⟩:=List.mem_map.mp hp
  24     simp only at hpi
  25     subst j
  26     rw [prescribed_piece_map_edges] at he
  27     obtain ⟨e',he',heq⟩:=Finset.mem_image.mp he
  28     have heq':e'=e:=Sym2.map.injective sigma_mk_injective heq
  29     subst e'
  30     exact ⟨q,hq,he'⟩
  31   · rintro ⟨q,hq,he⟩
  32     refine ⟨(i,q.map (Sigma.mk i)),?_,rfl,?_⟩
  33     · apply List.mem_flatMap.mpr
  34       exact ⟨i,by simp,List.mem_map.mpr ⟨q,hq,rfl⟩⟩
  35     · rw [prescribed_piece_map_edges]
  36       exact Finset.mem_image.mpr ⟨e,he,rfl⟩
  37 
  38 theorem prescribed_lifted_source_adjacency (pieces:∀ i,List (PrescribedSourcePiece (X i)))
  39     (i:I) (x y:X i):
  40     (prescribedSourceGraph (prescribedLiftDependentPieces Finset.univ.toList pieces) i).Adj (Sigma.mk i x) (Sigma.mk i y) ↔
  41       (prescribedDependentGraph pieces i).Adj x y := by
  42   change (SimpleGraph.fromEdgeSet _).Adj (Sigma.mk i x) (Sigma.mk i y) ↔ (SimpleGraph.fromEdgeSet _).Adj x y
  43   rw [SimpleGraph.fromEdgeSet_adj,SimpleGraph.fromEdgeSet_adj]
  44   have he:s(Sigma.mk i x,Sigma.mk i y)=Sym2.map (Sigma.mk i) s(x,y):=rfl
  45   rw [he,prescribed_lifted_physical_membership]
  46   constructor
  47   · rintro ⟨he,hne⟩
  48     exact ⟨he,fun h=>hne (by simpa [h])⟩
  49   · rintro ⟨he,hne⟩
  50     exact ⟨he,fun h=>hne (sigma_mk_injective h)⟩
  51 end Oberwolfach
