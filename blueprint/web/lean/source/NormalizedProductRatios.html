  13 theorem finite_product_deficit {J : Type*} (S : Finset J) (x : J → ℝ)
  14     (hx : ∀ j∈S,0 ≤ x j ∧ x j ≤ 1) :
  15     0 ≤ 1-(∏ j∈S,(1-x j)) ∧ 1-(∏ j∈S,(1-x j)) ≤ ∑ j∈S,x j := by
  16   have hh := finite_product_perturbation S (fun _ => (1:ℝ)) (fun j => 1-x j)
  17     (fun j hj => ⟨by linarith [(hx j hj).2],by linarith [(hx j hj).1],le_rfl⟩)
  18   simpa only [Finset.prod_const_one,sub_sub_cancel] using hh

  20 theorem normalized_product_ratio_error {J : Type*} (S : Finset J) (x y : J → ℝ)
  21     (hx : ∀ j∈S,0 ≤ x j ∧ x j ≤ 1) (hy : ∀ j∈S,0 ≤ y j ∧ y j ≤ 1)
  22     (hsmall : (∑ j∈S,y j) ≤ 1/2) :
  23     |(∏ j∈S,(1-x j))/(∏ j∈S,(1-y j))-1| ≤ 2*((∑ j∈S,x j)+(∑ j∈S,y j)) := by
  24   have h₁ := finite_product_deficit S x hx
  25   have h₂ := finite_product_deficit S y hy
  26   have hB : (0:ℝ)<∏ j∈S,(1-y j) := by linarith
  27   have hdiff : |(∏ j∈S,(1-x j))-(∏ j∈S,(1-y j))| ≤
  28       (∑ j∈S,x j)+(∑ j∈S,y j) := by
  29     apply abs_le.mpr
  30     constructor <;> linarith
  31   have hsum : 0 ≤ (∑ j∈S,x j)+(∑ j∈S,y j) :=
