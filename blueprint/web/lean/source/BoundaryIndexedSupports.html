  15 def innerBoundarySupport (G:SimpleGraph V) (A:Finset V) (x:V):Finset A:=
  16   Finset.univ.filter (fun v=>G.Adj x v.val)

  18 theorem innerBoundarySupport_image (G:SimpleGraph V) (A:Finset V) (x:V):
  19     (innerBoundarySupport G A x).image Subtype.val=A∩G.neighborFinset x:=by
  20   ext v
  21   simp only [innerBoundarySupport,Finset.mem_image,Finset.mem_filter,Finset.mem_univ,true_and,
  22     Finset.mem_inter,SimpleGraph.mem_neighborFinset]
  23   constructor
  24   · rintro ⟨w,hw,rfl⟩;exact ⟨w.property,hw⟩
  25   · rintro ⟨hv,hw⟩;exact ⟨⟨v,hv⟩,hw,rfl⟩

  27 theorem innerBoundarySupport_card (G:SimpleGraph V) (A:Finset V) (x:V):
  28     (innerBoundarySupport G A x).card=(A∩G.neighborFinset x).card:=by
  29   rw [←innerBoundarySupport_image]
  30   exact (Finset.card_image_of_injective _ Subtype.val_injective).symm

  32 theorem innerBoundarySupport_inter_card (G:SimpleGraph V) (A:Finset V) (x y:V):
  33     (innerBoundarySupport G A x∩innerBoundarySupport G A y).card=
  34       ((A∩G.neighborFinset x)∩(A∩G.neighborFinset y)).card:=by
  35   rw [←innerBoundarySupport_image G A x,←innerBoundarySupport_image G A y,
  36     ←Finset.image_inter _ _ Subtype.val_injective]

  39 theorem innerBoundarySupport_membership_bound (G:SimpleGraph V) (A Z:Finset V)
  40     (x:J↪V) (hx:∀ j,x j∈Z) (v:A):
  41     (Finset.univ.filter (fun j=>v∈innerBoundarySupport G A (x j))).card≤
  42       (Z.filter (fun z=>v.val∈A∩G.neighborFinset z)).card:=by
  43   have hs:((Finset.univ.filter (fun j=>v∈innerBoundarySupport G A (x j))).image x)⊆
  44       Z.filter (fun z=>v.val∈A∩G.neighborFinset z):=by
  45     intro z hz
  46     obtain ⟨j,hj,rfl⟩:=Finset.mem_image.mp hz
  47     have hh:G.Adj (x j) v.val:=by simpa only [Finset.mem_filter,Finset.mem_univ,true_and,innerBoundarySupport] using hj
  48     exact Finset.mem_filter.mpr ⟨hx j,Finset.mem_inter.mpr ⟨v.property,(SimpleGraph.mem_neighborFinset _ _ _).mpr hh⟩⟩
  49   have hh:=Finset.card_le_card hs
  50   simpa only [Finset.card_image_of_injective _ x.injective] using hh
