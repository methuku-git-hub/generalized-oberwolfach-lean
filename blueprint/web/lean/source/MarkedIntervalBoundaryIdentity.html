  12 theorem marked_interval_boundary_identity
  13     {V I:Type*} [Fintype V] [Fintype I] [DecidableEq V]
  14     (π:Equiv.Perm V) (k:I→ℕ) (a:I→V) (hk:∀j,0<k j):
  15     markedIntervalRootSet π k a ∪ Finset.univ.image (fun j=>(π^(k j)) (a j))=
  16       (markedIntervalRootSet π k a).image π ∪ Finset.univ.image a:=by
  17   classical
  18   have hpow (j:I) (n:ℕ):π ((π^n) (a j))=(π^(n+1)) (a j):=by
  19     rw [pow_succ',Equiv.Perm.mul_apply]
  20   have hroot (j:I) (i:Fin (k j)):(π^i.val) (a j)∈markedIntervalRootSet π k a:=by
  21     exact Finset.mem_biUnion.mpr ⟨j,Finset.mem_univ _,Finset.mem_image.mpr ⟨i,Finset.mem_univ _,rfl⟩⟩
  22   apply Finset.Subset.antisymm
  23   · intro x hx
  24     rcases Finset.mem_union.mp hx with hy | hb
  25     · obtain ⟨j,_,hj⟩:=Finset.mem_biUnion.mp hy
  26       obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hj
  27       by_cases hi:i.val=0
  28       · apply Finset.mem_union_right
  29         apply Finset.mem_image.mpr
  30         refine ⟨j,Finset.mem_univ _,?_⟩
  31         simp [hi]
  32       · apply Finset.mem_union_left
  33         let q:Fin (k j):=⟨i.val-1,by omega⟩
  34         refine Finset.mem_image.mpr ⟨(π^q.val) (a j),hroot j q,?_⟩
  35         have hs:i.val-1+1=i.val:=by omega
  36         simpa [q,hs] using hpow j (i.val-1)
  37     · obtain ⟨j,_,rfl⟩:=Finset.mem_image.mp hb
  38       apply Finset.mem_union_left
  39       let q:Fin (k j):=⟨k j-1,by have:=hk j;omega⟩
  40       refine Finset.mem_image.mpr ⟨(π^q.val) (a j),hroot j q,?_⟩
  41       have hs:k j-1+1=k j:=by have:=hk j;omega
  42       simpa [q,hs] using hpow j (k j-1)
  43   · intro x hx
  44     rcases Finset.mem_union.mp hx with hy | ht
  45     · exact marked_interval_successor_subset π k a hy
  46     · obtain ⟨j,_,rfl⟩:=Finset.mem_image.mp ht
  47       apply Finset.mem_union_left
  48       simpa using hroot j ⟨0,hk j⟩

  50 theorem marked_shifted_roots_disjoint_starts
  51     {V I:Type*} [Fintype V] [Fintype I] [DecidableEq V]
  52     (π:Equiv.Perm V) (k:I→ℕ) (a:I→V)
  53     (hinj:∀j,Function.Injective (fun i:Fin (k j+1)=>(π^i.val) (a j)))
  54     (hdis:∀j l:I,j≠l→Disjoint (permutationWindow π 0 (k j) (a j))
  55       (permutationWindow π 0 (k l) (a l))):
  56     Disjoint ((markedIntervalRootSet π k a).image π) (Finset.univ.image a):=by
  57   classical
  58   apply Finset.disjoint_left.mpr
  59   intro x hx ht
  60   obtain ⟨y,hy,hyx⟩:=Finset.mem_image.mp hx
  61   obtain ⟨j,_,hj⟩:=Finset.mem_biUnion.mp hy
  62   obtain ⟨i,_,hi⟩:=Finset.mem_image.mp hj
  63   obtain ⟨l,_,hl⟩:=Finset.mem_image.mp ht
  64   have hp:π ((π^i.val) (a j))=(π^(i.val+1)) (a j):=by
  65     rw [pow_succ',Equiv.Perm.mul_apply]
  66   have hxj:(π^(i.val+1)) (a j)=x:=by rw [←hp,hi,hyx]
  67   have hwin:(π^(i.val+1)) (a j)∈permutationWindow π 0 (k j) (a j):=by
  68     unfold permutationWindow
  69     exact Finset.mem_image.mpr ⟨⟨i.val+1,by omega⟩,Finset.mem_univ _,by simp; rfl⟩
  70   have hstart:a l∈permutationWindow π 0 (k l) (a l):=permutationWindow_self π 0 (k l) (a l)
  71   by_cases hjl:j=l
  72   · subst l
  73     have heq:(π^(⟨i.val+1,by omega⟩:Fin (k j+1)).val) (a j)=
  74         (π^(0:Fin (k j+1)).val) (a j):=by simpa using hxj.trans hl.symm
  75     have he:=hinj j heq
  76     have hv:=congrArg Fin.val he
  77     simp at hv
  78   · apply Finset.disjoint_left.mp (hdis j l hjl) hwin
  79     have he:(π^(i.val+1)) (a j)=a l:=hxj.trans hl.symm
  80     exact he.symm ▸ hstart
