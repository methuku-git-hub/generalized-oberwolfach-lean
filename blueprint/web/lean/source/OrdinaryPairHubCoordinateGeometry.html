  43     (ha:∀ e∈ordinaryPairHubCoordinates π d u u' a,σ e=τ e):

  45       ordinaryResourceState (rawWheelGraph P τ) (ordinaryPairHubSpokes π d τ j u u') a := by
  46   have h₁:=rawWheelResourceState_supported P π d j (ordinaryPairHubEmbedding u a) σ τ

  49     (fun e he=>ha e (Finset.mem_union_right _ he))

  52   | inr a=>

  57       simpa [ordinaryPairHubEmbedding,ordinaryResourceState] using h₂
  58     have hh₁:(rawWheelSpokes π d σ j v u↔rawWheelSpokes π d τ j v u):=by simpa only [decide_eq_decide] using he₁
