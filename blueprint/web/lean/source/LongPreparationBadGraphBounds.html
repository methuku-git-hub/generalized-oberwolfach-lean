  15 theorem long_preparation_bad_graph_bounds
  16     (P Bad Unused Unassigned:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ) (I:Finset V)
  17     (a b:ℝ) (ha:0≤a) (hb:0≤b)
  18     (hA:∀v∈I,(Unassigned.degree v:ℝ)≤a)
  19     (hB:∀v∈I,(Unused.degree v:ℝ)≤b)
  20     (hSub:Bad≤(spanningRestriction Unassigned I ⊔ spanningRestriction Unused I) ⊔
  21       longPreparationIneligibleGraph P π d):
  22     (∀v,(Bad.degree v:ℝ)≤a+b+(6*d+2:ℕ)) ∧
  23     2*(Bad.edgeFinset.card:ℝ)≤I.card*(a+b)+(6*d+2:ℕ)*Fintype.card V:=by
  24   let A:=spanningRestriction Unassigned I
  25   let B:=spanningRestriction Unused I
  26   let F:=longPreparationIneligibleGraph P π d
  27   have hA' (v:V):(A.degree v:ℝ)≤ if v∈I then a else 0:=by
  28     by_cases hv:v∈I
  29     · simp only [hv,ite_true]
  30       exact spanningRestriction_degree_bound Unassigned I a ha hA v
  31     · simp only [hv,ite_false]
  32       have hz:A.degree v=0:=by
  33         rw [← SimpleGraph.card_neighborFinset_eq_degree,Finset.card_eq_zero]
  34         apply Finset.eq_empty_iff_forall_notMem.mpr
  35         intro w hw
  36         have hh:=(spanningRestriction_adj Unassigned I v w).mp (((spanningRestriction Unassigned I).mem_neighborFinset v w).mp hw)
  37         exact hv hh.2.1
  38       simp [hz]
  39   have hB' (v:V):(B.degree v:ℝ)≤ if v∈I then b else 0:=by
  40     by_cases hv:v∈I
  41     · simp only [hv,ite_true]
  42       exact spanningRestriction_degree_bound Unused I b hb hB v
  43     · simp only [hv,ite_false]
  44       have hz:B.degree v=0:=by
  45         rw [← SimpleGraph.card_neighborFinset_eq_degree,Finset.card_eq_zero]
  46         apply Finset.eq_empty_iff_forall_notMem.mpr
  47         intro w hw
  48         have hh:=(spanningRestriction_adj Unused I v w).mp (((spanningRestriction Unused I).mem_neighborFinset v w).mp hw)
  49         exact hv hh.2.1
  50       simp [hz]
  51   have hf (v:V):(F.degree v:ℝ)≤(6*d+2:ℕ):=by exact_mod_cast long_preparation_ineligible_degree P π d v
  52   have hv (v:V):(Bad.degree v:ℝ)≤(if v∈I then a+b else 0)+(6*d+2:ℕ):=by
  53     have hdeg:(Bad.degree v:ℝ)≤((A⊔B⊔F).degree v:ℝ):=by exact_mod_cast SimpleGraph.degree_le_of_le hSub
  54     have hs1:(((A⊔B⊔F).degree v):ℝ)≤((A⊔B).degree v:ℝ)+F.degree v:=by
  55       have hn: (A⊔B⊔F).degree v≤(A⊔B).degree v+F.degree v:=by
  56         convert graph_degree_sup_le (A⊔B) F v using 1 <;> congr!
  57       exact_mod_cast hn
  58     have hs2:(((A⊔B).degree v):ℝ)≤(A.degree v:ℝ)+B.degree v:=by
  59       have hn: (A⊔B).degree v≤A.degree v+B.degree v:=by
  60         convert graph_degree_sup_le A B v using 1 <;> congr!
  61       exact_mod_cast hn
  62     have hAv:=hA' v
  63     have hBv:=hB' v
  64     by_cases h:v∈I <;> simp only [h,ite_true,ite_false] at hAv hBv ⊢ <;> linarith [hf v]
  65   constructor
  66   · intro v
  67     have hh:=hv v
  68     by_cases h:v∈I <;> simp only [h,ite_true,ite_false] at hh <;> linarith
  69   · have hsum:∑v:V,(Bad.degree v:ℝ)≤∑v:V,((if v∈I then a+b else 0)+(6*d+2:ℕ)):=
  70       Finset.sum_le_sum (fun v _=>hv v)
  71     have he:∑v:V,(Bad.degree v:ℝ)=2*(Bad.edgeFinset.card:ℝ):=by
  72       rw [← Nat.cast_sum,Bad.sum_degrees_eq_twice_card_edges]
  73       push_cast
  74       ring
  75     rw [he] at hsum
  76     simp only [Finset.sum_add_distrib,Finset.sum_ite_mem,Finset.univ_inter,Finset.sum_const,
  77       Finset.card_univ,nsmul_eq_mul] at hsum
  78     nlinarith [hsum]
  79 end Oberwolfach
