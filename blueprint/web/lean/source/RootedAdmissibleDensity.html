  15 theorem rootedSubset_dense_conflict_free_probability {V : Type*} [Fintype V] [DecidableEq V]
  16     (P F : SimpleGraph V) (T : Finset V) (k : ℕ) (hk : T.card+k≤Fintype.card V) (hk2 : 2≤k)
  17     [Nonempty (RootedSubset V T k)] (δ α L : ℝ) (hα : 0<α)
  18     (hP : ∀ v,(δ+α)*Fintype.card V≤(P.degree v:ℝ))
  19     (hF : ∀ v,(F.degree v:ℝ)≤L)
  20     (hT : ∀ x∈T,∀ y∈T,¬F.Adj x y)
  21     (hroots : (T.card:ℝ)+1≤α*(T.card+k:ℝ)/4) :
  22     let q : ℝ := (k:ℝ)/(Fintype.card V-T.card:ℕ)
  23     1-2*(T.card+k:ℝ)*Real.exp (-α^2*(T.card+k:ℝ)/8)-
  24         L*(2*(T.card:ℝ)*q+Fintype.card V*q^2)≤
  25       (FiniteLaw.uniform (RootedSubset V T k)).probability (fun S=>
  26         (∀ v∈S.val.val,(δ+α/2)*(T.card+k:ℝ)<((P.neighborFinset v∩S.val.val).card:ℝ)) ∧
  27         (∀ x∈S.val.val,∀ y∈S.val.val,¬F.Adj x y)) := by
  28   intro q
  29   letI : Nonempty (FixedSubset V (T.card+k)) := fixedSubset_nonempty hk
  30   let μ := FiniteLaw.uniform (RootedSubset V T k)
  31   let Dense : RootedSubset V T k → Prop := fun S=>
  32     ∀ v∈S.val.val,(δ+α/2)*(T.card+k:ℝ)<((P.neighborFinset v∩S.val.val).card:ℝ)
  33   let Good : RootedSubset V T k → Prop := fun S=>∀ x∈S.val.val,∀ y∈S.val.val,¬F.Adj x y
  34   let loss : ℝ := L*(2*(T.card:ℝ)*q+Fintype.card V*q^2)
  35   have hbad := rootedSubset_conflict_probability F T k hk L hF hT
  36   have hgood : 1-loss≤μ.probability Good := by
  37     have hc := finiteLaw_probability_compl μ
  38       (fun S=>∃ x∈S.val.val,∃ y∈S.val.val,F.Adj x y)
  39     have he : (fun S : RootedSubset V T k=>¬(∃ x∈S.val.val,∃ y∈S.val.val,F.Adj x y))=Good := by
  40       funext S
  41       apply propext
  42       simp only [Good,not_exists,not_and]
  43     rw [he] at hc
  44     change μ.probability (fun S=>∃ x∈S.val.val,∃ y∈S.val.val,F.Adj x y)≤loss at hbad
  45     linarith only [hc,hbad]
  46   have hd : 1-2*(T.card+k:ℝ)*Real.exp (-α^2*(T.card+k:ℝ)/8)≤μ.probability Dense := by
  47     have hh := fixedSubset_rooted_dense_probability T (T.card+k) hk (by omega)
  48       (fun v=>P.neighborFinset v) δ α hα
  49       (by simpa only [SimpleGraph.card_neighborFinset_eq_degree] using hP)
  50       (by simpa only [Nat.cast_add] using hroots)
  51     simpa only [Nat.cast_add,μ,Dense] using hh
  52   have hh := finiteLaw_good_intersection μ Dense Good loss hgood
  53   exact (sub_le_sub_right hd loss).trans hh
