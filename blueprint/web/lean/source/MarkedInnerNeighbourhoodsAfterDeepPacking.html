  27     (hSrow:∀ x∈A,∀ u,¬DS.Adj (Sum.inl x) (Sum.inr u))
  28     (hScol:∀ u∈B,∀ x,¬DS.Adj (Sum.inl x) (Sum.inr u)):

  32       (rawMarkedDeleteRims σ DR.edgeFinset) (markedDeletePhysicalSpokes H DS):=by
  33   have hphys (a:Bool×Bool×V) (x:V) (hx:x∈A):
  34       markedPairPhysical π (markedRowRim a x)∉DR.edgeFinset:=by
  35     rcases a with ⟨mark,incoming,v⟩
  36     cases mark <;> cases incoming <;>
  37       simp only [markedPairPhysical,markedRowRim,Bool.false_eq_true,ite_false,ite_true,Sum.elim_inl,Sum.elim_inr,
  38         SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet]
  39     · exact hR v x hx
  40     · exact fun he=>hR v x hx he.symm
  41     · exact hR v (π x) ((hA x).mpr hx)
  42     · exact fun he=>hR (π v) x hx he.symm
  43   have hrim:=markedRows_rim_changes P π d h A p k q s g j errR 0 (by norm_num) σ
  44     (rawMarkedDeleteRims σ DR.edgeFinset) H hr (by
  45       intro a
  46       have he:∀ x∈A,(if ¬(rawMarkedPairRims P π d σ (markedRowRim a x)↔
  47           rawMarkedPairRims P π d (rawMarkedDeleteRims σ DR.edgeFinset) (markedRowRim a x)) then (1:ℝ) else 0)=0:=by

  53     (rawMarkedDeleteRims σ DR.edgeFinset) H (markedDeletePhysicalSpokes H DS) hrim (by
  54       intro mark u
  55       have he:∀ x∈A,(if ¬(H mark x u↔markedDeletePhysicalSpokes H DS mark x u) then (1:ℝ) else 0)=0:=by
  56         intro x hx;simp [markedDeletePhysicalSpokes,hSrow x hx u]
  57       simp only [Finset.sum_congr rfl he,Finset.sum_const_zero,le_refl])
  58   have hidx:=markedIndices_spoke_relation_changes π d h B q s g j errI 0 (by norm_num)

  61       intro mark x

  63         intro u hu;simp [markedDeletePhysicalSpokes,hScol u hu x]
  64       simp only [Finset.sum_congr rfl he,Finset.sum_const_zero,le_refl])
  65   simpa only [mul_zero,add_zero] using And.intro hrows hidx
