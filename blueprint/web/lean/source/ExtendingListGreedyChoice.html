  12 theorem extending_list_two_arc_choice (D:V→V→Prop) (L B:Finset V) (β:ℝ)
  13     (hext:IsExtendingList D L β) (F:SimpleGraph V) (x y:V) (hxy:x≠y)
  14     (hbudget:(B.card:ℝ)+F.degree x+F.degree y<β*Fintype.card V):
  15     ∃ z∈L,z∉B ∧ D x z ∧ D z y ∧ ¬F.Adj x z ∧ ¬F.Adj y z := by
  16   let C:=directedOutNeighbours D x∩directedInNeighbours D y∩L
  17   let T:=B∪F.neighborFinset x∪F.neighborFinset y
  18   have hcard:(T.card:ℝ)≤(B.card:ℝ)+F.degree x+F.degree y:=by
  19     have h₁:=Finset.card_union_le (B∪F.neighborFinset x) (F.neighborFinset y)
  20     have h₂:=Finset.card_union_le B (F.neighborFinset x)
  21     have hh:T.card≤B.card+F.degree x+F.degree y:=by
  22       dsimp [T]
  23       simpa only [SimpleGraph.card_neighborFinset_eq_degree] using (by omega :
  24         ((B∪F.neighborFinset x)∪F.neighborFinset y).card≤
  25           B.card+(F.neighborFinset x).card+(F.neighborFinset y).card)
  26     exact_mod_cast hh
  27   have hnot:¬C⊆T:=by
  28     intro hsub
  29     have hh:C.card≤T.card:=Finset.card_le_card hsub
  30     have hhR:(C.card:ℝ)≤T.card:=by exact_mod_cast hh
  31     have hlo:β*Fintype.card V≤(C.card:ℝ):=hext.2.2.2 x y hxy
  32     linarith
  33   obtain ⟨z,hzC,hzT⟩:=Finset.not_subset.mp hnot
  34   have hz:=Finset.mem_inter.mp hzC
  35   have hzpair:=Finset.mem_inter.mp hz.1
  36   have hzout:D x z:=(Finset.mem_filter.mp hzpair.1).2
  37   have hzin:D z y:=(Finset.mem_filter.mp hzpair.2).2
  38   have hzB:z∉B:=fun h=>hzT (by simp [T,h])
  39   have hzFx:¬F.Adj x z:=fun h=>hzT (by simp [T,h])
  40   have hzFy:¬F.Adj y z:=fun h=>hzT (by simp [T,h])
  41   exact ⟨z,hz.2,hzB,hzout,hzin,hzFx,hzFy⟩
  42 end Oberwolfach
