  10 theorem prescribed_embeddings_glue (A B:Finset X) (ψ φ:X→V)
  11     (hψ:Set.InjOn ψ (↑A:Set X)) (hφ:Set.InjOn φ (↑B:Set X))
  12     (hag:∀x∈A∩B,ψ x=φ x)
  13     (havoid:∀x∈B\A,∀y∈A,φ x≠ψ y):
  14     ∃g:X→V,Set.InjOn g (↑(A∪B):Set X) ∧ (∀x∈A,g x=ψ x) ∧ (∀x∈B,g x=φ x) := by
  15   let g:X→V:=fun x=>if x∈A then ψ x else φ x
  16   have hgA:∀x∈A,g x=ψ x:=by intro x hx;simp [g,hx]
  17   have hgB:∀x∈B,g x=φ x:=by
  18     intro x hx
  19     by_cases hA:x∈A
  20     · rw [hgA x hA,hag x (Finset.mem_inter.mpr ⟨hA,hx⟩)]
  21     · simp [g,hA]
  22   refine ⟨g,?_,hgA,hgB⟩
  23   intro x hx y hy heq
  24   rcases Finset.mem_union.mp hx with hxA|hxB <;> rcases Finset.mem_union.mp hy with hyA|hyB
  25   · exact hψ hxA hyA (by simpa [hgA x hxA,hgA y hyA] using heq)
  26   · by_cases hyA:y∈A
  27     · exact hψ hxA hyA (by simpa [hgA x hxA,hgA y hyA] using heq)
  28     · have hh:φ y≠ψ x:=havoid y (Finset.mem_sdiff.mpr ⟨hyB,hyA⟩) x hxA
  29       exact (hh (by simpa [hgA x hxA,hgB y hyB] using heq.symm)).elim
  30   · by_cases hxA:x∈A
  31     · exact hψ hxA hyA (by simpa [hgA x hxA,hgA y hyA] using heq)
  32     · have hh:φ x≠ψ y:=havoid x (Finset.mem_sdiff.mpr ⟨hxB,hxA⟩) y hyA
  33       exact (hh (by simpa [hgB x hxB,hgA y hyA] using heq)).elim
  34   · exact hφ hxB hyB (by simpa [hgB x hxB,hgB y hyB] using heq)
  35 end Oberwolfach
