  14 theorem finiteLaw_weighted_indicators {Ω V : Type*} [Fintype Ω] [Fintype V]
  15     (μ : FiniteLaw Ω) (P : V → Ω → Prop) (y : V → ℝ) :
  16     μ.expectation (fun ω => ∑ v,if P v ω then y v else 0)=
  17       ∑ v,y v*μ.probability (P v) := by
  18   unfold FiniteLaw.expectation FiniteLaw.probability
  19   simp_rw [Finset.mul_sum]
  20   rw [Finset.sum_comm]
  21   apply Finset.sum_congr rfl
  22   intro v _
  23   apply Finset.sum_congr rfl
  24   intro ω _
  25   by_cases h : P v ω <;> simp [h,mul_comm]

  29 theorem fixedSubset_weighted_mean (hr : r ≤ Fintype.card V) (y : V → ℝ) :
  30     (FiniteLaw.uniform (FixedSubset V r)).expectation (fun S => ∑ v∈S.val,y v)=
  31       (r:ℝ)/Fintype.card V*(∑ v,y v) := by
  32   have heq (S : FixedSubset V r) : (∑ v∈S.val,y v)=∑ v,if v∈S.val then y v else 0 := by simp
  33   simp_rw [heq]
  34   calc
  35     _ = ∑ v,y v*(FiniteLaw.uniform (FixedSubset V r)).probability (fun S => v∈S.val) := by
  36       convert finiteLaw_weighted_indicators (FiniteLaw.uniform (FixedSubset V r)) (fun v S => v∈S.val) y using 1 <;> congr!
  37     _ = _ := by
  38       simp_rw [fixedSubset_vertex_probability hr]
  39       rw [← Finset.sum_mul,mul_comm]

  41 theorem fixedSubset_distinct_pair_probability_formula (hr : r ≤ Fintype.card V) (u v : V) (huv : u≠v) :
  42     (FiniteLaw.uniform (FixedSubset V r)).probability (fun S => u∈S.val ∧ v∈S.val)=
  43       (r:ℝ)*(r-1)/((Fintype.card V:ℝ)*(Fintype.card V-1)) := by

  50 theorem fixedSubset_centred_moment (hr : r ≤ Fintype.card V) (hn : 2 ≤ Fintype.card V)
  51     (y : V → ℝ) (hy : ∑ v,y v=0) (v : V) :
  52     (FiniteLaw.uniform (FixedSubset V r)).expectation
  53       (fun S => if v∈S.val then (∑ u∈S.val,y u) else 0)=
  54       (r:ℝ)*(Fintype.card V-r)/((Fintype.card V:ℝ)*(Fintype.card V-1))*y v := by
  55   let μ := FiniteLaw.uniform (FixedSubset V r)
  56   let a : ℝ := r/Fintype.card V
  57   let b : ℝ := (r:ℝ)*(r-1)/((Fintype.card V:ℝ)*(Fintype.card V-1))
  58   have heq (S : FixedSubset V r) :
  59       (if v∈S.val then (∑ u∈S.val,y u) else 0)=
  60       ∑ u,if v∈S.val ∧ u∈S.val then y u else 0 := by
  61     by_cases hv : v∈S.val <;> simp [hv]
  62   simp_rw [heq]
  63   calc
  64     _ = ∑ u,y u*μ.probability (fun S => v∈S.val ∧ u∈S.val) := by
  65       convert finiteLaw_weighted_indicators μ (fun u S => v∈S.val ∧ u∈S.val) y using 1 <;> congr!
  66     _ = _ := by
  67       have hi (u : V) : y u*μ.probability (fun S => v∈S.val ∧ u∈S.val)=
  68           (if u=v then (a-b)*y v else 0)+b*y u := by
  69         by_cases huv : u=v
  70         · subst u
  71           simp only [and_self,fixedSubset_vertex_probability hr,ite_true]
  72           dsimp [a,μ]
  73           rw [fixedSubset_vertex_probability hr]
  74           ring
  75         · rw [fixedSubset_distinct_pair_probability_formula hr v u (Ne.symm huv)]
  76           simp only [huv,ite_false,zero_add]
  77           dsimp [b]
  78           ring
  79       change (∑ u,y u*μ.probability (fun S => v∈S.val ∧ u∈S.val))=_
  80       simp_rw [hi,Finset.sum_add_distrib,← Finset.mul_sum,hy,mul_zero,add_zero]
  81       simp only [Finset.sum_ite_eq',Finset.mem_univ,ite_true]
  82       have hnR : (1:ℝ)<Fintype.card V := by exact_mod_cast (show 1<Fintype.card V by omega)
  83       have hn0 : (Fintype.card V:ℝ)≠0 := by linarith
  84       have hn1 : (Fintype.card V:ℝ)-1≠0 := by linarith
  85       dsimp [a,b]
  86       field_simp
  87       <;> ring
