   3 /-! Rescale fixed coefficients from ambient order n to the class size k.
   4 The weights themselves are unchanged. -/

   7  theorem marked_class_weight_rescale (m:ℕ) (a n k lo hi w:ℝ)
   8     (ha:0<a) (hn:0<n) (hak:a*n≤k) (hkn:k≤n) (hlo:0≤lo) (hhi:0≤hi)
   9     (hw:lo/n^m≤w∧w≤hi/n^m):
  10     lo*a^m/k^m≤w∧w≤hi/k^m:=by
  11   have hk:0<k:=lt_of_lt_of_le (mul_pos ha hn) hak
  12   have hnk:n^m≤(k/a)^m:=pow_le_pow_left₀ hn.le ((le_div_iff₀ ha).mpr (by nlinarith only [hak])) m
  13   have hakpow:(a*n)^m≤k^m:=pow_le_pow_left₀ (by positivity) hak m
  14   have hknpow:k^m≤n^m:=pow_le_pow_left₀ hk.le hkn m
  15   constructor
  16   · apply le_trans _ hw.1
  17     apply (div_le_div_iff₀ (pow_pos hk m) (pow_pos hn m)).mpr
  18     have hh:=mul_le_mul_of_nonneg_left hakpow hlo
  19     rw [mul_pow] at hh
  20     nlinarith only [hh]
  21   · apply hw.2.trans
  22     exact div_le_div_of_nonneg_left hhi (pow_pos hk m) hknpow
