  15 def finiteAttachmentWheel (C : SimpleGraph W) (hC : IsEmbeddedCycle C r) (z : W) (ρ : I → V) (i : I) :
  16     OrdinaryWheel (ordinaryAttachmentRim ⊥ C z ρ) ⊥
  17       (ordinaryAttachmentSpokes (fun (_ : V) (_ : U) => False) z ρ) r := by
  18   let D := C.map (attachmentWheelEmbedding z ρ i)
  19   have hD : IsEmbeddedCycle D r := hC.map _
  20   have hDQ : D≤ordinaryAttachmentRim ⊥ C z ρ := by
  21     rw [ordinaryAttachmentRim_eq]
  22     exact le_sup_of_le_right (le_iSup (fun j => C.map (attachmentWheelEmbedding z ρ j)) i)
  23   refine ⟨(⟨⟨D,hD⟩,hDQ⟩,Sum.inr i),?_,?_⟩
  24   · intro a ha
  25     change a∈(C.map (attachmentWheelEmbedding z ρ i)).support at ha
  26     rw [SimpleGraph.support_map] at ha
  27     obtain ⟨w,hw,rfl⟩ := ha
  28     by_cases hwz : w=z <;> simp [attachmentWheelEmbedding,ordinaryAttachmentSpokes,hwz]
  29   · intro a b ha hb
  30     exact not_false

  38 theorem finiteAttachmentWheels_disjoint (C : SimpleGraph W) (hC : IsEmbeddedCycle C r)
  39     (z : W) (ρ : I → V) (i j : I) (hij : i≠j) :
  40     Disjoint (ordinaryWheelResources (finiteAttachmentWheel (U:=U) C hC z ρ i))
  41       (ordinaryWheelResources (finiteAttachmentWheel C hC z ρ j)) := by
  42   apply ordinaryWheels_disjoint_at_one_root _ _ (Sum.inl (ρ i))
  43   · intro a hi hj
  44     rw [finiteAttachmentWheel_rim,SimpleGraph.support_map] at hi hj
  45     obtain ⟨w,hw,rfl⟩ := hi
  46     obtain ⟨w',hw',heq⟩ := hj
  47     by_cases hwz : w=z
  48     · simp [attachmentWheelEmbedding,hwz]
  49     · have he : attachmentWheelEmbedding z ρ j w'=Sum.inr (i,w) := by
  50         simpa [attachmentWheelEmbedding,hwz] using heq
  51       have hji := ((attachmentWheelEmbedding_private z ρ j i w' w).mp he).2.1
  52       exact (hij hji.symm).elim
  53   · simp only [finiteAttachmentWheel_hub,ne_eq,Sum.inr.injEq]
  54     exact hij

  59       ordinaryAttachmentSpokes (fun (_ : V) (_ : U) => False) z ρ a (.inr i) := by
