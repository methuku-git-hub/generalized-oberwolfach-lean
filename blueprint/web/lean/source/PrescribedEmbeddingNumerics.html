   8 theorem prescribed_embedding_numerical_margins (β δ n:ℝ)
   9     (hβ:0<β) (hβ1:β≤1) (hδ:0<δ) (hroot:Real.sqrt δ≤β/100)
  10     (hn:2/Real.sqrt δ≤n):
  11     let t:=Real.sqrt δ*n;
  12     2≤t ∧ 3*δ*n+2*t+2≤8*t ∧ 2*δ*n+2*t+2≤5*t ∧
  13     0<β*n-t-δ*n-16*t := by
  14   intro t
  15   have hr0:0<Real.sqrt δ:=Real.sqrt_pos.2 hδ
  16   have hr1:Real.sqrt δ≤1:=by linarith
  17   have hsq:(Real.sqrt δ)^2=δ:=Real.sq_sqrt hδ.le
  18   have hn0:0<n:=by
  19     have hh:(0:ℝ)<2/Real.sqrt δ:=by positivity
  20     linarith
  21   have ht:2≤t:=by
  22     dsimp [t]
  23     have hh:2≤n*Real.sqrt δ:=(div_le_iff₀ hr0).mp hn
  24     nlinarith
  25   have hδroot:δ≤Real.sqrt δ:=by nlinarith
  26   have hδn:δ*n≤t:=by
  27     dsimp [t]
  28     exact mul_le_mul_of_nonneg_right hδroot hn0.le
  29   have hsmall:17*Real.sqrt δ+δ<β:=by nlinarith [hδroot]
  30   have hsmalln:17*(Real.sqrt δ*n)+δ*n<β*n:=by
  31     have hh:=mul_lt_mul_of_pos_right hsmall hn0
  32     nlinarith
  33   refine ⟨ht,?_,?_,?_⟩ <;> dsimp [t] at * <;> nlinarith
  34 theorem prescribed_embedding_margins_from_hypotheses (β δ n:ℝ)
  35     (hβ:0<β) (hβ1:β≤1) (hδ:0<δ) (hδbound:δ≤(β/100)^2)
  36     (hn:2/Real.sqrt δ≤n):
  37     let t:=Real.sqrt δ*n;
  38     2≤t ∧ 3*δ*n+2*t+2≤8*t ∧ 2*δ*n+2*t+2≤5*t ∧
  39     0<β*n-t-δ*n-16*t := by

  44     (hβ:0<β) (hβ1:β≤1) (hδ:0<δ) (hδbound:δ≤(β/100)^2)
  45     (hnβ:100/β≤n):

  54     linarith
