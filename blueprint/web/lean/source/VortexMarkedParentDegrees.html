  12 theorem vortex_degree_margins (n D a u deg β α ρ : ℝ)
  13     (hn : 0< n) (ha : 0≤a) (hD : (1/2+α)*n≤D) (hDu : D≤n)
  14     (hα : 0<α) (hβ : 0≤β) (hβα : β≤α/8) (hβ1 : β≤1/4)
  15     (hc : |2*u/ρ-(D/n)*a|≤β*a) (hd : |deg-2*u/ρ|≤β*a) :
  16     a/2≤2*u/ρ ∧ 2*u/ρ≤2*a ∧ (1/2+α/2)*a≤deg := by
  17   have hDlo : 1/2+α≤D/n := (le_div_iff₀ hn).mpr hD
  18   have hDhi : D/n≤1 := (div_le_iff₀ hn).mpr (by simpa using hDu)
  19   have hlo := mul_le_mul_of_nonneg_right hDlo ha
  20   have hhi := mul_le_mul_of_nonneg_right hDhi ha
  21   have hbα := mul_le_mul_of_nonneg_right hβα ha
  22   have hb1 := mul_le_mul_of_nonneg_right hβ1 ha
  23   have hcp := (abs_le.mp hc).1
  24   have hcm := (abs_le.mp hc).2
  25   have hdp := (abs_le.mp hd).1
  26   constructor
  27   · nlinarith only [hlo,hcp,hbα,mul_nonneg hα.le ha]
  28   constructor
  29   · nlinarith only [hhi,hcm,hb1,ha]
  30   · nlinarith only [hlo,hcp,hdp,hbα,mul_nonneg hα.le ha]

  32 theorem Vortex.marked_parent_degrees
  33     {V U : Type*} [Fintype V] [Fintype U] {G : SimpleGraph V}
  34     {D m : ℕ} {β ε α : ℝ} (Q : Vortex U G D β ε m)
  35     (hreg : G.IsRegularOfDegree D) (hn : 0< Fintype.card V)
  36     (hD : (1/2+α)*Fintype.card V≤(D:ℝ)) (hα : 0<α)
  37     (hρ : 0< initialIndexDensity U D) (hb : 0≤β) (hbα : β≤α/8) (hb1 : β≤1/4)
  38     {i : ℕ} (hi : i≤Q.length) (x : V) (hx : x∈Q.vertices (i-2)) :
  39     (Q.vertices i).card/2≤2*(Q.indices i).card/initialIndexDensity U D ∧
  40     2*(Q.indices i).card/initialIndexDensity U D≤2*(Q.vertices i).card ∧
  41     (1/2+α/2)*(Q.vertices i).card≤((G.neighborFinset x∩Q.vertices i).card:ℝ) ∧
  42     |((G.neighborFinset x∩Q.vertices i).card:ℝ)-2*(Q.indices i).card/initialIndexDensity U D|≤β*(Q.vertices i).card := by
  43   have he := Q.estimates_including_zero hreg hn hρ hb i hi
  44   have hn' : (0:ℝ)< Fintype.card V := by exact_mod_cast hn
  45   have hDu : (D:ℝ)≤Fintype.card V := by exact_mod_cast regular_degree_le_card hreg hn
  46   obtain ⟨ha,hb',hc⟩ := vortex_degree_margins (Fintype.card V) D (Q.vertices i).card (Q.indices i).card
  47     (G.neighborFinset x∩Q.vertices i).card β α (initialIndexDensity U D) hn' (Nat.cast_nonneg _)
  48     hD hDu hα hb hbα hb1 he.1 (he.2 x hx)
  49   exact ⟨ha,hb',hc,he.2 x hx⟩

  51 theorem Vortex.marked_difference_parent_degrees
  52     {V U : Type*} [Fintype V] [Fintype U] {G : SimpleGraph V}
  53     {D m : ℕ} {β ε α : ℝ} (Q : Vortex U G D β ε m)
  54     (hreg : G.IsRegularOfDegree D) (hn : 0< Fintype.card V)
  55     (hD : (1/2+α)*Fintype.card V≤(D:ℝ)) (hα : 0<α)
  56     (hρ : 0< initialIndexDensity U D) (hb : 0≤β) (hbα : β≤3*α/40) (hb1 : β≤3/20)
  57     (he : 0≤ε) (he1 : ε≤1/4) {j k : ℕ} (hjk : j< k)
  58     (x : V) (hx : x∈Q.extendedVertices (k-2)) :
  59     let A:=Q.extendedVertices j\Q.extendedVertices k
  60     let B:=Q.extendedIndices j\Q.extendedIndices k;
  61     A.card/2≤2*B.card/initialIndexDensity U D ∧
  62     2*B.card/initialIndexDensity U D≤2*A.card ∧
  63     (1/2+α/2)*A.card≤((G.neighborFinset x∩A).card:ℝ) ∧
  64     |((G.neighborFinset x∩A).card:ℝ)-2*B.card/initialIndexDensity U D|≤(5/3)*β*A.card := by
  65   intro A B
  66   have hest := Q.extended_difference_degree_estimates hreg hn hρ hb he he1 hjk x hx
  67   have hn' : (0:ℝ)< Fintype.card V := by exact_mod_cast hn
  68   have hDu : (D:ℝ)≤Fintype.card V := by exact_mod_cast regular_degree_le_card hreg hn
  69   obtain ⟨ha,hb',hc⟩ := vortex_degree_margins (Fintype.card V) D A.card B.card
  70     (G.neighborFinset x∩A).card ((5/3)*β) α (initialIndexDensity U D) hn' (Nat.cast_nonneg _)
  71     hD hDu hα (by positivity) (by linarith only [hbα]) (by linarith only [hb1]) hest.1 hest.2
  72   exact ⟨ha,hb',hc,hest.2⟩

  74 theorem Vortex.cover_down_parent_orders
  75     {V U : Type*} [Fintype V] [Fintype U] {G : SimpleGraph V}
  76     {D m : ℕ} {β ε : ℝ} (Q : Vortex U G D β ε m)
  77     (he : 0<ε) (hm : 0< m) {i : ℕ} (hi : i< Q.length) :
  78     (m:ℝ)/2≤(Q.vertices i).card := by
  79   have hl := Q.vertices_lower (by omega : i+1≤Q.length)
  80   have hu := (Q.consecutive_size_bounds he hm hi).2
  81   nlinarith only [hl,hu,he]
