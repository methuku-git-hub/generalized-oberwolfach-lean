  16 def markedRowEligible (P : SimpleGraph V) (π : Equiv.Perm V) (b : ℕ)
  17     (a : Bool×Bool×V) (z : V) : Prop :=
  18   ¬PermutationClose π b a.2.2 z ∧
  19     Sum.elim (fun xy=>P.Adj xy.1 xy.2) (fun xy=>P.Adj xy.1 (π xy.2)) (markedRowRim a z)

  21 def markedRowDomain (P : SimpleGraph V) (π : Equiv.Perm V) (b : ℕ)
  22     (A : Finset V) (S T : Finset (Bool×Bool×V)) : Finset V :=
  23   A.filter (fun z=>(∀ a∈S,markedRowEligible P π b a z) ∧
  24     Set.InjOn (fun a=>markedPairPhysical π (markedRowRim a z)) S ∧
  25     (∀ a∈T,Sum.elim (fun xy=>P.Adj xy.1 xy.2) (fun xy=>P.Adj xy.1 (π xy.2)) (markedRowRim a z)))
