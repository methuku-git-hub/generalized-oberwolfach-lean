  12 theorem profile_sample_order_union_margin (c q N₀ : ℕ) (hc : 1 ≤ c)
  13     (κv κr : ℝ) (hκv : 0 < κv) (hκr : 0 < κr) :
  14     ∃ t : ℕ,N₀ ≤ t ∧ q ≤ t ∧ 4 ≤ t ∧
  15       (2*Real.exp (-κv*t))*(c*(t+q):ℕ)+
  16         (2*Real.exp (-κr*t))*((c*(t+q):ℕ):ℝ)^c ≤ 1/4 := by
  17   let κ := min κv κr
  18   have hκ : 0 < κ := lt_min hκv hκr
  19   let C := 4*((2:ℝ)*c)^c
  20   obtain ⟨N,hN⟩ := eventually_stretchedExp_le_inv_pow C (c:ℝ) (c:=κ) (p:=1) hκ (by norm_num) 1
  21   let t := max (max N₀ q) (max N 4)
  22   have ht₀ : N₀ ≤ t := (le_max_left _ _).trans (le_max_left _ _)
  23   have htq : q ≤ t := (le_max_right _ _).trans (le_max_left _ _)
  24   have htN : N ≤ t := (le_max_left _ _).trans (le_max_right _ _)
  25   have ht4 : 4 ≤ t := (le_max_right _ _).trans (le_max_right _ _)
  26   have htR : (4:ℝ) ≤ t := by exact_mod_cast ht4
  27   have htpos : (0:ℝ) < t := by linarith
  28   have htqR : (q:ℝ) ≤ t := by exact_mod_cast htq
  29   have hcR : (1:ℝ) ≤ c := by exact_mod_cast hc
  30   let M := ((c*(t+q):ℕ):ℝ)
  31   have hM : 1 ≤ M := by dsimp [M];push_cast;nlinarith [show (0:ℝ) ≤ q by positivity]
  32   have hMt : M ≤ (2:ℝ)*c*t := by dsimp [M];push_cast;nlinarith
  33   have hMc : M ≤ M^c := by
  34     have hh : M^1 ≤ M^c := pow_le_pow_right₀ hM hc
  35     simpa only [pow_one] using hh
  36   have hv : Real.exp (-κv*t) ≤ Real.exp (-κ*t) := by
  37     apply Real.exp_le_exp.mpr
  38     have hh := min_le_left κv κr
  39     dsimp [κ]
  40     nlinarith
  41   have hr : Real.exp (-κr*t) ≤ Real.exp (-κ*t) := by
  42     apply Real.exp_le_exp.mpr
  43     have hh := min_le_right κv κr
  44     dsimp [κ]
  45     nlinarith
  46   have htail : C*(t:ℝ)^c*Real.exp (-κ*t) ≤ 1/t := by
  47     simpa only [Real.rpow_natCast,Real.rpow_one,pow_one] using hN t htN
  48   refine ⟨t,ht₀,htq,ht4,?_⟩
  49   change (2*Real.exp (-κv*t))*M+(2*Real.exp (-κr*t))*M^c ≤ _
  50   calc
  51     _ ≤ 4*Real.exp (-κ*t)*M^c := by
  52       have hv' := mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_left hv (show (0:ℝ) ≤ 2 by norm_num)) (show 0 ≤ M by linarith)
  53       have hr' := mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_left hr (show (0:ℝ) ≤ 2 by norm_num)) (pow_nonneg (show 0 ≤ M by linarith) c)
  54       have hm' := mul_le_mul_of_nonneg_left hMc (show 0 ≤ 2*Real.exp (-κ*t) by positivity)
  55       linarith
  56     _ ≤ 4*Real.exp (-κ*t)*(((2:ℝ)*c*t)^c) := mul_le_mul_of_nonneg_left
  57       (pow_le_pow_left₀ (le_trans (by norm_num) hM) hMt c) (by positivity)
  58     _ = C*(t:ℝ)^c*Real.exp (-κ*t) := by dsimp [C];rw [mul_pow];ring
  59     _ ≤ 1/t := htail
  60     _ ≤ 1/4 := (div_le_div_of_nonneg_left (by norm_num) (by norm_num) htR)
