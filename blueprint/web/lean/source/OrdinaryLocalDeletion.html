  16 theorem ordinaryLocalPotential_sdiff_loss (P Q F D : SimpleGraph V) (T : Finset V)
  17     (φ : OrdinaryCycle P r→ℝ) (hφ : ∀ C,0≤φ C) (v : V) :
  18     |ordinaryLocalPotential P (Q\D) F T φ v-ordinaryLocalPotential P Q F T φ v|≤
  19       ∑ C,if v∈C.val.val.support ∧
  20         (∃ x∈C.val.val.support,∃ y∈C.val.val.support,D.Adj x y) then φ C else 0 := by
  21   let a : OrdinaryCycle P r→Prop:=fun C=>v∈C.val.val.support ∧ C.val.val≤Q ∧
  22     (∀ x∈C.val.val.support,∀ y∈C.val.val.support,¬F.Adj x y) ∧
  23     (∀ z∈C.val.val.support,z≠v → z∈T)
  24   let b : OrdinaryCycle P r→Prop:=fun C=>v∈C.val.val.support ∧ C.val.val≤Q\D ∧
  25     (∀ x∈C.val.val.support,∀ y∈C.val.val.support,¬F.Adj x y) ∧
  26     (∀ z∈C.val.val.support,z≠v → z∈T)
  27   have hba (C) : b C→a C := fun h=>⟨h.1,h.2.1.trans sdiff_le,h.2.2⟩
  28   have hab (C) (ha : a C) (hb : ¬b C) :
  29       ∃ x∈C.val.val.support,∃ y∈C.val.val.support,D.Adj x y := by
  30     by_contra hn
  31     have hno : ∀ x∈C.val.val.support,∀ y∈C.val.val.support,¬D.Adj x y := by
  32       simpa only [not_exists,not_and] using hn
  33     apply hb
  34     refine ⟨ha.1,?_,ha.2.2⟩
  35     intro x y hxy
  36     exact ⟨ha.2.1 hxy,hno x hxy.mem_support_left y hxy.mem_support_right⟩
  37   have hnonneg : 0≤(∑ C,if a C then φ C else 0)-(∑ C,if b C then φ C else 0) := by
  38     apply sub_nonneg.mpr
  39     apply Finset.sum_le_sum
  40     intro C _
  41     by_cases hb : b C
  42     · simp only [if_pos hb,if_pos (hba C hb),le_refl]
  43     · rw [if_neg hb];split_ifs;exact hφ C;exact le_refl _
  44   have haeq : ordinaryLocalPotential P Q F T φ v=∑ C,if a C then φ C else 0 := by
  45     unfold ordinaryLocalPotential;congr!
  46   have hbeq : ordinaryLocalPotential P (Q\D) F T φ v=∑ C,if b C then φ C else 0 := by
  47     unfold ordinaryLocalPotential;congr!
  48   rw [abs_sub_comm,haeq,hbeq]
  49   rw [abs_of_nonneg hnonneg,←Finset.sum_sub_distrib]
  50   apply Finset.sum_le_sum
  51   intro C _
  52   by_cases hb : b C
  53   · simp only [if_pos hb,if_pos (hba C hb),sub_self]
  54     split_ifs;exact hφ C;exact le_refl _
  55   · rw [if_neg hb,sub_zero]
  56     by_cases ha : a C
  57     · rw [if_pos ha,if_pos ⟨ha.1,hab C ha hb⟩]
  58     · rw [if_neg ha];split_ifs;exact hφ C;exact le_refl _

  60 theorem ordinaryLocalPotential_sdiff_bound (P Q F D : SimpleGraph V) (T : Finset V)
  61     (hr : 2≤r) (hn : 0<Fintype.card V) (φ : OrdinaryCycle P r→ℝ)
  62     (hφ : ∀ C,0≤φ C) (B L : ℝ) (hB : 0≤B)
  63     (hφB : ∀ C,φ C≤B/(Fintype.card V:ℝ)^(r-1))
  64     (hD : ∀ v,(D.degree v:ℝ)≤L) (v : V) :
  65     |ordinaryLocalPotential P (Q\D) F T φ v-ordinaryLocalPotential P Q F T φ v|≤
  66       3*(r:ℝ)^4*B*L/Fintype.card V := by
  67   have hnR : (0:ℝ)<Fintype.card V:=Nat.cast_pos.mpr hn
  68   have hcoef : ∀ C,φ C≤(B/Fintype.card V)/(Fintype.card V:ℝ)^(r-2) := by
  69     intro C
  70     have he : (B/Fintype.card V)/(Fintype.card V:ℝ)^(r-2)=B/(Fintype.card V:ℝ)^(r-1) := by
  71       rw [div_div,←pow_succ',show r-2+1=r-1 by omega]
  72     rw [he];exact hφB C
  73   have hh:=ordinary_cycle_vertex_conflict_loss P r hr hn φ hφ (B/Fintype.card V)
  74     (div_nonneg hB hnR.le) hcoef D L hD v
  75   have h:= (ordinaryLocalPotential_sdiff_loss P Q F D T φ hφ v).trans hh
  76   convert h using 1 <;> ring

  78 theorem ordinary_common_sdiff_bound (Q D : SimpleGraph V) (T S : Finset V) (L : ℝ)
  79     (hD : ∀ v,(D.degree v:ℝ)≤L) :
  80     |((common T S (fun v=>(Q\D).neighborFinset v)).card:ℝ)-
  81       (common T S (fun v=>Q.neighborFinset v)).card|≤S.card*L := by
  82   apply neighbourhood_stability
  83   intro v _
  84   have hsub : changed (Q.neighborFinset v) ((Q\D).neighborFinset v)⊆D.neighborFinset v := by
  85     intro w hw
  86     simp only [changed,Finset.mem_union,Finset.mem_sdiff,SimpleGraph.mem_neighborFinset,
  87       SimpleGraph.sdiff_adj] at hw
  88     rw [SimpleGraph.mem_neighborFinset]
  89     rcases hw with h|h
  90     · exact not_not.mp (fun hn=>h.2 ⟨h.1,hn⟩)
  91     · exact (h.2 h.1.1).elim
  92   have hc : ((changed (Q.neighborFinset v) ((Q\D).neighborFinset v)).card:ℝ)≤D.degree v := by
  93     exact_mod_cast (Finset.card_le_card hsub).trans_eq (SimpleGraph.card_neighborFinset_eq_degree _ _)
  94   exact hc.trans (hD v)
