  17  theorem canonical_edge_endpoint_sum (G:SimpleGraph V) (f:Sym2 V→V→ℤ):
  18     (∑ e∈G.edgeFinset,(f e e.out.1+f e e.out.2))=
  19       ∑ v,∑ w:G.neighborSet v,f s(v,w.val) v:=by
  20   let E:(Σ v,G.neighborSet v)≃G.Dart:=
  21     { toFun:=fun p=>⟨(p.1,p.2.val),p.2.property⟩
  22       invFun:=fun d=>⟨d.fst,⟨d.snd,d.adj⟩⟩
  23       left_inv:=by intro p;cases p;rfl
  24       right_inv:=by intro d;cases d;rfl }
  25   have hs:(∑ d:G.Dart,f d.edge d.fst)=∑ v,∑ w:G.neighborSet v,f s(v,w.val) v:=by
  26     rw [←E.sum_comp (fun d=>f d.edge d.fst),Fintype.sum_sigma]
  27     rfl
  28   rw [←hs]
  29   have hf:=Finset.sum_fiberwise_of_maps_to (s:=Finset.univ) (t:=G.edgeFinset)
  30     (g:=SimpleGraph.Dart.edge) (fun d _=>by rw [SimpleGraph.mem_edgeFinset];exact d.edge_mem)
  31     (fun d=>f d.edge d.fst)
  32   rw [←hf]
  33   apply Finset.sum_congr rfl
  34   intro e he
  35   have heq:s(e.out.1,e.out.2)=e:=by rw [Sym2.mk];exact e.out_eq
  36   have hadj:G.Adj e.out.1 e.out.2:=SimpleGraph.mem_edgeFinset.mp (heq.symm ▸ he)
  37   let d:G.Dart:=⟨(e.out.1,e.out.2),hadj⟩
  38   have hde:d.edge=e:=heq
  39   have hfilter:(Finset.univ.filter (fun a:G.Dart=>a.edge=e))={d,d.symm}:=by
  40     rw [←hde];exact d.edge_fiber
  41   rw [hfilter,Finset.sum_pair (Ne.symm d.symm_ne)]
  42   simp only [SimpleGraph.Dart.edge_symm,hde]
  43   rfl

  45  theorem canonical_edge_endpoint_sum_subtype (G:SimpleGraph V) (f:(e:G.edgeFinset)→V→ℤ):
  46     (∑ e:G.edgeFinset,(f e e.val.out.1+f e e.val.out.2))=
  47       ∑ v,∑ w:G.neighborSet v,f ⟨s(v,w.val),SimpleGraph.mem_edgeFinset.mpr w.property⟩ v:=by
  48   let f':Sym2 V→V→ℤ:=fun e v=>if he:e∈G.edgeFinset then f ⟨e,he⟩ v else 0
  49   have hh:=canonical_edge_endpoint_sum G f'
  50   have hleft:(∑ e:G.edgeFinset,(f e e.val.out.1+f e e.val.out.2))=
  51       ∑ e∈G.edgeFinset,(f' e e.out.1+f' e e.out.2):=by
  52     calc
  53       _ = ∑ e:G.edgeFinset,(f' e.val e.val.out.1+f' e.val e.val.out.2):=by
  54         apply Finset.sum_congr rfl
  55         intro e _
  56         simp only [f',dite_eq_left e.property]
  57       _ = _:=by simpa only [] using (Finset.sum_coe_sort G.edgeFinset (fun e=>f' e e.out.1+f' e e.out.2))
  58   rw [hleft,hh]
  59   apply Finset.sum_congr rfl
  60   intro v _
  61   apply Finset.sum_congr rfl
  62   intro w _
  63   have hw:s(v,w.val)∈G.edgeFinset:=SimpleGraph.mem_edgeFinset.mpr w.property
  64   simp only [f',dite_eq_left hw]
