  15  theorem marked_attachment_endpoint_ownership (n:ℕ)
  16     (src dst:A→V) (row:S→V)
  17     (a:∀ v,{e:A // dst e=v}≃{s:S // row s=v})
  18     (d:∀ v,{e:A // src e=v}≃{s:S // row s=v})
  19     (pr:A→(Fin (n+1)↪V)) (ps:S→(Fin (n+2)↪V))
  20     (pv:(Σ v,DirectedIncidence src dst v)→Fin (n+2)→V)
  21     (hr:∀ e f,e≠f→Disjoint (Finset.univ.image (pr e)) (Finset.univ.image (pr f)))
  22     (hrs:∀ e s,Disjoint (Finset.univ.image (pr e)) (Finset.univ.image (ps s)))
  23     (hvin:∀ v e,pv ⟨v,Sum.inl e⟩ 0=pr e.val 0∧
  24       pv ⟨v,Sum.inl e⟩ (Fin.last (n+1))=ps (a v e).val (Fin.last (n+1)))
  25     (hvout:∀ v e,pv ⟨v,Sum.inr e⟩ 0=ps (d v e).val 0∧
  26       pv ⟨v,Sum.inr e⟩ (Fin.last (n+1))=pr e.val (Fin.last n)):
  27     ∀ z:Σ v,DirectedIncidence src dst v,∀ e:A,
  28       pv z 0∈Finset.univ.image (pr e)∨pv z (Fin.last (n+1))∈Finset.univ.image (pr e)→
  29       e=directedIncidenceEdge z.2:=by
  30   have hinj (e f:A) (k:Fin (n+1)) (h:pr f k∈Finset.univ.image (pr e)):e=f:=by
  31     by_contra hn
  32     exact Finset.disjoint_left.mp (hr e f hn) h (Finset.mem_image.mpr ⟨k,Finset.mem_univ _,rfl⟩)
  33   rintro ⟨v,z⟩ e hz
  34   cases z with
  35   | inl z=>
  36     rcases hz with hz|hz
  37     · rw [(hvin v z).1] at hz
  38       exact hinj e z.val 0 hz
  39     · rw [(hvin v z).2] at hz
  40       exact False.elim (Finset.disjoint_left.mp (hrs e (a v z).val) hz
  41         (Finset.mem_image.mpr ⟨Fin.last _,Finset.mem_univ _,rfl⟩))
  42   | inr z=>
  43     rcases hz with hz|hz
  44     · rw [(hvout v z).1] at hz
  45       exact False.elim (Finset.disjoint_left.mp (hrs e (d v z).val) hz
  46         (Finset.mem_image.mpr ⟨0,Finset.mem_univ _,rfl⟩))
  47     · rw [(hvout v z).2] at hz
  48       exact hinj e z.val (Fin.last n) hz
