  15 theorem ordinary_triangle_wheel (Q F:SimpleGraph V) (H:V→U→Prop)
  16     (x a b:V) (u:U) (hxa:Q.Adj x a) (hxb:Q.Adj x b) (hab:Q.Adj a b)
  17     (hH:H x u∧H a u∧H b u)
  18     (hF:∀ v∈({x,a,b}:Finset V),∀ w∈({x,a,b}:Finset V),¬F.Adj v w):
  19     ∃ C:OrdinaryWheel Q F H 3,ordinaryWheelHub C=u ∧
  20       (∀ v,v∈(ordinaryWheelRim C).support↔v∈({x,a,b}:Finset V)) ∧
  21       (ordinaryWheelRim C).Adj x a ∧ (ordinaryWheelRim C).Adj x b ∧
  22       (ordinaryWheelRim C).Adj a b:=by
  23   obtain ⟨C,hC⟩:=triangle_cycle_from_vertices_support Q x a b hxa.ne hxb.ne hab.ne hxa hxb hab
  24   have hs (v:V):v∈C.val.val.support↔v∈({x,a,b}:Finset V):=by
  25     rw [←Set.mem_toFinset,hC]
  26   have hh:∀ v∈C.val.val.support,H v u:=by
  27     intro v hv
  28     rcases (by simpa only [Finset.mem_insert,Finset.mem_singleton] using (hs v).mp hv : v=x∨v=a∨v=b) with rfl|rfl|rfl
  29     · exact hH.1
  30     · exact hH.2.1
  31     · exact hH.2.2
  32   let W:OrdinaryWheel Q F H 3:=⟨(C,u),hh,fun v hv w hw=>hF v ((hs v).mp hv) w ((hs w).mp hw)⟩
  33   have he (v w:V):C.val.val.Adj v w↔v∈({x,a,b}:Finset V)∧w∈({x,a,b}:Finset V)∧v≠w:=by
  34     rw [C.val.property.triangle_adj,hs,hs]
  35   exact ⟨W,rfl,hs,(he x a).mpr ⟨by simp,by simp,hxa.ne⟩,
  36     (he x b).mpr ⟨by simp,by simp,hxb.ne⟩,(he a b).mpr ⟨by simp,by simp,hab.ne⟩⟩

  38 theorem triangular_vertex_wheel_pair (Q F:SimpleGraph V) (H:V→U→Prop)
  39     (x y a b:V) (u:U) (hxy:x≠y)
  40     (hxa:Q.Adj x a) (hxb:Q.Adj x b) (hya:Q.Adj y a) (hyb:Q.Adj y b) (hab:Q.Adj a b)
  41     (hH:H x u∧H y u∧H a u∧H b u)
  42     (hF:∀ v∈({x,a,b}:Finset V),∀ w∈({x,a,b}:Finset V),¬F.Adj v w)
  43     (hFy:¬F.Adj y a∧¬F.Adj y b):
  44     ∃ D:OrdinaryVertexWheelPair Q F H 3 x y a b u,
  45       (∀ v,v∈(ordinaryWheelRim D.first).support↔v∈({x,a,b}:Finset V)) ∧
  46       (∀ v,v∈(ordinaryWheelRim D.second).support↔v∈({y,a,b}:Finset V)) ∧
  47       rootWheelInternal D.first x a b=∅:=by
  48   obtain ⟨C,hCu,hCs,hCxa,hCxb,_⟩:=ordinary_triangle_wheel Q F H x a b u hxa hxb hab
  49     ⟨hH.1,hH.2.2.1,hH.2.2.2⟩ hF
  50   have hy:y∉(ordinaryWheelRim C).support:=by
  51     rw [hCs]
  52     simp only [Finset.mem_insert,Finset.mem_singleton,not_or]
  53     exact ⟨hxy.symm,hya.ne,hyb.ne⟩
  54   have hneighbors (v:V) (hv:(ordinaryWheelRim C).Adj x v):v=a∨v=b:=by
  55     have hm: v∈(ordinaryWheelRim C).neighborFinset x:=((ordinaryWheelRim C).mem_neighborFinset x v).mpr hv
  56     rw [ordinaryWheel_root_neighbors C x a b hab.ne hCxa hCxb] at hm
  57     simpa only [Finset.mem_insert,Finset.mem_singleton] using hm
  58   have hremaining (v:V) (hv:v∈(ordinaryWheelRim C).support) (hvx:v≠x):v=a∨v=b:=by
  59     have hm: v=x∨v=a∨v=b:=by simpa only [Finset.mem_insert,Finset.mem_singleton] using (hCs v).mp hv
  60     exact hm.resolve_left hvx
  61   obtain ⟨C',hmap,hC'u,hs,hrim,hsp⟩:=ordinary_wheel_root_replacement C x y u hCxa.mem_support_left hy
  62     (fun v hv=>(hneighbors v hv).elim (fun h=>h ▸ hya) (fun h=>h ▸ hyb))
  63     (fun v hv hvx=>(hremaining v hv hvx).elim (fun h=>h ▸ hFy.1) (fun h=>h ▸ hFy.2)) hH.2.1
  64     (fun v hv hvx=>(hremaining v hv hvx).elim (fun h=>h ▸ hH.2.2.1) (fun h=>h ▸ hH.2.2.2))
  65   have hnew (v:V) (hv:(ordinaryWheelRim C).Adj x v):(ordinaryWheelRim C').Adj y v:=by
  66     have hvy:v≠y:=fun he=>hy (he ▸ hv.mem_support_right)
  67     rw [hmap,graph_swap_adj,Equiv.swap_apply_right,Equiv.swap_apply_of_ne_of_ne hv.ne.symm hvy]
  68     exact hv
  69   have hedge (z:V):graphRootEdges (ordinaryWheelRim C) x z={s(z,a),s(z,b)}:=by
  70     unfold graphRootEdges
  71     rw [ordinaryWheel_root_neighbors C x a b hab.ne hCxa hCxb]
  72     simp
  73   let D:OrdinaryVertexWheelPair Q F H 3 x y a b u:=
  74     ⟨C,C',hCu,hC'u,hCxa,hCxb,hnew a hCxa,hnew b hCxb,hmap,hs,
  75       fun e=>by simpa only [hedge] using hrim e,
  76       fun v z=>by have hh:=hsp v z;rw [hCu] at hh;omega⟩
  77   refine ⟨D,hCs,?_,?_⟩
  78   · intro v
  79     change v∈(ordinaryWheelRim C').support↔v∈({y,a,b}:Finset V)
  80     rw [hs,hCs]
  81     simp only [Finset.mem_insert,Finset.mem_singleton]
  82     constructor
  83     · rintro (he|⟨he|he|he,hne⟩)
  84       · exact Or.inl he
  85       · exact (hne he).elim
  86       · exact Or.inr (Or.inl he)
  87       · exact Or.inr (Or.inr he)
  88     · rintro (rfl|rfl|rfl)
  89       · exact Or.inl rfl
  90       · exact Or.inr ⟨Or.inr (Or.inl rfl),hxa.ne.symm⟩

  92   · ext v

  94     exact fun hv=>hv.2 ((hCs v).mp hv.1)
