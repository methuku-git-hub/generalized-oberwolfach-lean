  11 theorem triangle_manuscript_absorber (β:ℝ) (hβ:0<β):
  12   ∃ ε₀:ℝ,0<ε₀ ∧ ∀ m:ℕ,∃ M₀ N₀:ℕ,
  13   ∀ (V₀ U₀ V U:Type) [Fintype V₀] [Fintype U₀] [DecidableEq V₀] [DecidableEq U₀]
  14     [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U],
  15     N₀≤Fintype.card V→M₀≤Fintype.card U→Fintype.card V₀+Fintype.card U₀≤m→
  16   ∀ (G:SimpleGraph V₀) (Hs:V₀→U₀→Prop) (Q:SimpleGraph V) (H:V→U→Prop)
  17     (φ:V₀↪V) (ψ:U₀↪U),
  18     (∀ v,G.degree v=2*(Finset.univ.filter (Hs v)).card)→
  19     (∀ u,3∣(Finset.univ.filter (fun v=>Hs v u)).card)→
  20     (∀ v w,G.Adj v w→Q.Adj (φ v) (φ w))→
  21     (∀ v u,Hs v u→H (φ v) (ψ u))→
  22     (∀ v,(3/4+β)*(Fintype.card V:ℝ)≤(Q.degree v:ℝ))→
  23   ∀ ε:ℝ,0≤ε→ε≤ε₀→
  24     (∀ u,(((Finset.univ:Finset V).filter (fun v=>¬H v u)).card:ℝ)≤ε*(Fintype.card V:ℝ))→
  25     (∀ v,(((Finset.univ:Finset U).filter (fun u=>¬H v u)).card:ℝ)≤ε*(Fintype.card U:ℝ))→
  26   ∃ (R:SimpleGraph V) (K:V→U→Prop),
  27     R≤Q ∧ (∀ v u,K v u→H v u) ∧ Disjoint R (G.map φ) ∧
  28     (∀ v u,¬(K v u∧ordinaryMappedSpokes Hs φ ψ v u)) ∧
  29     (ordinaryPhysicalGraph R K).support.toFinset.card≤M₀ ∧
  30     (∀ v∈Finset.univ.image φ,∀ w∈Finset.univ.image φ,¬R.Adj v w) ∧
  31     (∀ v∈Finset.univ.image φ,∀ u∈Finset.univ.image ψ,¬K v u) ∧
  32     HasResourceDecomposition (ordinaryWheelResources (P:=Q) (F:=⊥) (Hbase:=H) (r:=3)) (ordinaryHostResources R K) ∧
  33     HasResourceDecomposition (ordinaryWheelResources (P:=Q) (F:=⊥) (Hbase:=H) (r:=3))
  34       (ordinaryHostResources (R⊔G.map φ) (fun v u=>K v u∨ordinaryMappedSpokes Hs φ ψ v u)) :=by
  35   obtain ⟨ε₀,hε₀,hbase⟩:=triangle_dense_host_absorber_graph β hβ
  36   refine ⟨ε₀,hε₀,?_⟩
  37   intro m
  38   obtain ⟨N,K,habs⟩:=hbase m m
  39   let M:=max K (2*(5*ordinaryDiagramSize 3 m+2*m+4*K))
  40   refine ⟨M,N,?_⟩
  41   intro V₀ U₀ V U _ _ _ _ _ _ _ _ hn hU hm G Hs Q H φ ψ hbal hdiv hG hHs hdeg ε hε hεsmall hnearV hnearU
  42   have hUK:K≤Fintype.card U:=(le_max_left _ _).trans hU
  43   have hB:(Finset.univ.image ψ).card≤m:=by
  44     rw [Finset.card_image_of_injective _ ψ.injective,Finset.card_univ]
  45     omega
  46   obtain ⟨R,Ks,hQ,hH,hdis,hsp,hcard,hR,hK,hdec,hdec'⟩:=habs V₀ U₀ V U hn hUK hm G Hs Q H ∅ (Finset.univ.image ψ) φ ψ
  47     hbal hdiv hG hHs (by intro u;exact Finset.mem_image.mpr ⟨u,Finset.mem_univ _,rfl⟩)
  48     (by simp) hB hdeg ε hε hεsmall hnearV hnearU
  49   refine ⟨R,Ks,hQ,hH,hdis,hsp,?_,?_,?_,hdec,hdec'⟩
  50   · exact hcard.trans (by dsimp [M];simpa only [max_self] using (le_max_right K (2*(5*ordinaryDiagramSize 3 m+2*m+4*K))))
  51   · intro v hv w hw
  52     exact hR v (Finset.mem_union_left _ hv) w (Finset.mem_union_left _ hw)
  53   · intro v hv u hu
  54     exact hK v (Finset.mem_union_left _ hv) u hu
