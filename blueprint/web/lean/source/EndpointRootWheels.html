   6 /-! Complete an actual endpoint triangle to a physical three-wheel by
   7 adjoining the external root. The index becomes the hub; the two row
   8 vertices and the root form the rim triangle. -/

  18 theorem endpoint_triangle_root_wheel (Q H F : SimpleGraph V) (A B S T : Finset V)
  19     (hAB : Disjoint A B) (hSA : S⊆A) (hTB : T⊆B) (x : V)
  20     (hS : ∀ v∈S,Q.Adj x v ∧ ¬F.Adj x v) (hT : ∀ u∈T,H.Adj x u)
  21     (C : EndpointTriangle (endpointHostGraph Q H A B) F S T) :
  22     ∃ W : OrdinaryWheel Q F H.Adj 3,
  23       ordinaryWheelHub W=endpointTriangleIndex C ∧
  24       (ordinaryWheelRim W).support.toFinset=insert x (C.val∩S) := by
  25   obtain ⟨v,w,hvw,hrows⟩:=Finset.card_eq_two.mp C.property.2.1
  26   have hv : v∈C.val ∧ v∈S:=Finset.mem_inter.mp (by rw [hrows];simp)
  27   have hw : w∈C.val ∧ w∈S:=Finset.mem_inter.mp (by rw [hrows];simp)
  28   let u:=endpointTriangleIndex C
  29   have hu : u∈C.val ∧ u∈T:=endpointTriangleIndex_mem C
  30   have huv : u≠v:=fun he=>Finset.disjoint_left.mp hAB (he ▸ hSA hv.2) (hTB hu.2)
  31   have huw : u≠w:=fun he=>Finset.disjoint_left.mp hAB (he ▸ hSA hw.2) (hTB hu.2)
  32   have hvwQ : Q.Adj v w:=(endpointHostGraph_row Q H A B hAB (hSA hv.2) (hSA hw.2)).mp
  33     (C.property.2.2.2.1 v hv.1 w hw.1 hvw)
  34   have hvuH : H.Adj v u:=(endpointHostGraph_spoke Q H A B hAB (hSA hv.2) (hTB hu.2)).mp
  35     (C.property.2.2.2.1 v hv.1 u hu.1 huv.symm)
  36   have hwuH : H.Adj w u:=(endpointHostGraph_spoke Q H A B hAB (hSA hw.2) (hTB hu.2)).mp
  37     (C.property.2.2.2.1 w hw.1 u hu.1 huw.symm)
  38   have hxvQ:=(hS v hv.2).1
  39   have hxwQ:=(hS w hw.2).1
  40   have hxvF:=(hS v hv.2).2
  41   have hxwF:=(hS w hw.2).2
  42   have hvwF:=C.property.2.2.2.2 v hv.1 w hw.1
  43   obtain ⟨D,hD⟩:=triangle_cycle_from_vertices_support Q x v w hxvQ.ne hxwQ.ne hvw hxvQ hxwQ hvwQ
  44   have hsupp (z : V) : z∈D.val.val.support ↔ z=x ∨ z=v ∨ z=w := by
  45     rw [←Set.mem_toFinset,hD];simp
  46   have hH : ∀ z∈D.val.val.support,H.Adj z u := by
  47     intro z hz
  48     rcases (hsupp z).mp hz with rfl|rfl|rfl
  49     · exact hT u hu.2
  50     · exact hvuH
  51     · exact hwuH
  52   have hF : ∀ a∈D.val.val.support,∀ b∈D.val.val.support,¬F.Adj a b := by
  53     intro a ha b hb
  54     have hvxF : ¬F.Adj v x:=fun h=>hxvF h.symm
  55     have hwxF : ¬F.Adj w x:=fun h=>hxwF h.symm
  56     have hwvF : ¬F.Adj w v:=fun h=>hvwF h.symm
  57     rcases (hsupp a).mp ha with rfl|rfl|rfl <;>
  58       rcases (hsupp b).mp hb with rfl|rfl|rfl <;> simp_all
  59   refine ⟨⟨(D,u),hH,hF⟩,rfl,?_⟩
  60   change D.val.val.support.toFinset=insert x (C.val∩S)
  61   rw [hD,hrows]
  62
