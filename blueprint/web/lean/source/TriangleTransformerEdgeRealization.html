  48  theorem triangular_transformer_edge_realization
  49     (G Q F:SimpleGraph V) (H₀ H:V→U→Prop) (φ:V→V) (ψ:U→U)
  50     (A R W:Finset V) (B:Finset U)
  51     (hbalance:∀ v,G.degree v=2*((Finset.univ:Finset U).filter (H₀ v)).card)
  52     (hroot:∀ v∈G.support,v∈R∧φ v∈R)
  53     (hcross:∀ v∈G.support,∀ w∈G.support,φ v≠w)
  54     (hQ:G≤Q) (hφQ:∀ v w,G.Adj v w→Q.Adj (φ v) (φ w))
  55     (hφedge:Function.Injective (fun e:G.edgeFinset=>s(φ e.val.out.1,φ e.val.out.2)))
  56     (hψedge:Function.Injective (fun e:{e:V×U // H₀ e.1 e.2}=>(φ e.val.1,ψ e.val.2)))
  57     (hH:∀ v u,H₀ v u→H v u∧H (φ v) (ψ u))
  58     (hFroots:∀ v w,G.Adj v w→∀ a∈({v,w,φ v,φ w}:Finset V),
  59       ∀ c∈({v,w,φ v,φ w}:Finset V),¬F.Adj a c)
  60     (d c cH:ℝ) (hd:0≤d) (hF:∀ v,(F.degree v:ℝ)≤d)
  61     (hfour:∀ w x y z,c≤((A.filter (fun a=>Q.Adj w a∧Q.Adj x a∧Q.Adj y a∧Q.Adj z a)).card:ℝ))
  62     (hspfour:∀ u u' w x y z,c≤((A.filter (fun a=>H a u∧H a u'∧Q.Adj w a∧Q.Adj x a∧Q.Adj y a∧Q.Adj z a)).card:ℝ))
  63     (hbudget:W.card+((R.card:ℝ)+G.edgeFinset.card+2*Fintype.card {e:V×U // H₀ e.1 e.2}+1)*(d+1)<c)
  64     (hhubs:∀ T:Finset V,T.card≤9→(∀ v∈T,∀ w∈T,¬F.Adj v w)→
  65       cH≤((common (Finset.univ:Finset U) T (fun v=>Finset.univ.filter (H v))).card:ℝ))
  66     (hhbudget:B.card+G.edgeFinset.card<cH):
  67     Nonempty (TriangleTransformerEdgeData G Q F H₀ H φ ψ A R W B):=by
  68   obtain ⟨σ,b,hσ,_⟩:=transformer_endpoint_assignments G H₀ hbalance
  69   have hspRoot (i:{e:V×U // H₀ e.1 e.2}):i.val.1∈R∧φ i.val.1∈R:=
  70     hroot _ (balanced_spoke_root_mem_support G H₀ hbalance i)
  71   have hspF (i:{e:V×U // H₀ e.1 e.2}):¬F.Adj i.val.1 (φ i.val.1):=by
  72     apply balanced_spoke_root_separation G F H₀ φ hbalance _ _ _ i.property
  73     intro v w hvw
  74     exact hFroots v w hvw v (by simp) (φ v) (by simp)
  75   have heRoots (i:G.edgeFinset):i.val.out.1∈R∧i.val.out.2∈R∧φ i.val.out.1∈R∧φ i.val.out.2∈R:=
  76     ⟨(hroot _ (edge_out_adj G i).mem_support_left).1,(hroot _ (edge_out_adj G i).mem_support_right).1,
  77       (hroot _ (edge_out_adj G i).mem_support_left).2,(hroot _ (edge_out_adj G i).mem_support_right).2⟩
  78   obtain ⟨p,hp,hrim,hsp,hclass,hdis,hFF,hends,hspends,hspH,hab⟩:=
  79     triangular_transformer_paths G Q F H₀ H φ ψ b A W R d c hd hF hfour hspfour hbudget
  80   have hFR:∀ i,∀ v∈(p i).2.2.support,∀ w∈R,¬F.Adj v w:=fun i v hv=>(hclass i v hv).2.2.2
  81   obtain ⟨E⟩:=ordinary_actual_rim_wheels G Q F H₀ H φ b σ p 3 (by omega) R B hroot hQ hφQ hcross hφedge
  82     (fun i=>⟨hp _,hrim i⟩) hends (fun i v hv=>(hclass i v hv).2.2.1)
  83     hdis hFF hFR (fun i=>hFroots _ _ (edge_out_adj G i)) cH (by simpa using hhubs) hhbudget
  84   have hFsp (i:{e:V×U // H₀ e.1 e.2}):
  85       ∀ v∈insert i.val.1 (insert (φ i.val.1) (p (.inr i)).2.2.support.toFinset),
  86       ∀ w∈insert i.val.1 (insert (φ i.val.1) (p (.inr i)).2.2.support.toFinset),¬F.Adj v w:=by
  87     intro v hv w hw
  88     simp only [Finset.mem_insert,List.mem_toFinset] at hv hw
  89     rcases hv with rfl|rfl|hv <;> rcases hw with rfl|rfl|hw
  90     · exact F.loopless.irrefl _
  91     · exact hspF i
  92     · exact fun h=>hFR _ _ hw _ (hspRoot i).1 h.symm
  93     · exact fun h=>hspF i h.symm
  94     · exact F.loopless.irrefl _
  95     · exact fun h=>hFR _ _ hw _ (hspRoot i).2 h.symm
  96     · exact hFR _ _ hv _ (hspRoot i).1
  97     · exact hFR _ _ hv _ (hspRoot i).2
  98     · exact hFF _ _ _ hv _ hw
  99   obtain ⟨S⟩:=ordinary_actual_spoke_wheels G Q F H₀ H φ ψ 3 (by omega) p R
 100     (fun i=>⟨(hspRoot i).1,(hspRoot i).2,(hcross _ (balanced_spoke_root_mem_support G H₀ hbalance i)
 101       _ (balanced_spoke_root_mem_support G H₀ hbalance i)).symm⟩)
 102     (fun i=>⟨hp _,hsp i⟩) hspends
 103     (fun i v hv=>(hclass i v hv).2.2.1) hdis hFsp (fun i=>hH _ _ i.property) hspH hψedge
 104   let W':=W∪Finset.univ.biUnion (fun i=>(p i).2.2.support.toFinset)
 105   have hcard (i):(p i).2.2.support.toFinset.card≤2:=by
 106     have hh:=List.toFinset_card_le (p i).2.2.support
 107     rw [(p i).2.2.length_support] at hh
 108     rcases i with i|i
 109     · rw [hrim i] at hh;omega
 110     · rw [hsp i] at hh;omega
 111   have hW'card:W'.card≤W.card+(G.edgeFinset.card+Fintype.card {e:V×U // H₀ e.1 e.2})*2:=by
 112     have hc:=Finset.card_biUnion_le (s:=(Finset.univ:Finset (G.edgeFinset⊕{e:V×U // H₀ e.1 e.2})))
 113       (t:=fun i=>(p i).2.2.support.toFinset)
 114     have hc':(∑ i,(p i).2.2.support.toFinset.card)≤
 115         (G.edgeFinset.card+Fintype.card {e:V×U // H₀ e.1 e.2})*2:=by
 116       calc
 117         _≤∑ _i:(G.edgeFinset⊕{e:V×U // H₀ e.1 e.2}),(2:ℕ):=Finset.sum_le_sum (fun i _=>hcard i)
 118         _= _:=by simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul,Fintype.card_sum,Fintype.card_coe,Nat.cast_id]
 119     exact (Finset.card_union_le _ _).trans (Nat.add_le_add_left (hc.trans hc') _)
 120   refine ⟨⟨b,σ,hσ,p,hp,hrim,hab,hsp,
 121     (fun i v hv=>⟨(hclass i v hv).1,(hclass i v hv).2.1,(hclass i v hv).2.2.1⟩),hdis,hFF,hFR,
 122     hends,hspends,hspH,W',Finset.subset_union_left,?_,hW'card,E,S⟩⟩
 123   intro i v hv
 124   exact Finset.mem_union_right _ (Finset.mem_biUnion.mpr ⟨i,Finset.mem_univ _,List.mem_toFinset.mpr hv⟩)
