  29     (f:V↪V') (g:U↪U') (hcomm:∀ x,f (ρ x)=π (f x))
  30     (e:E→E') (k:K→K')
  31     (het:∀ a,tail' (e a)=f (tail a)) (heh:∀ a,head' (e a)=f (head a))
  32     (hkt:∀ a,ktail' (k a)=f (ktail a)) (hkh:∀ a,khead' (k a)=f (khead a))
  33     (hH₀:∀ x u,H₀ x u→H₀' (f x) (g u)) (hHK:∀ x u,HK x u→HK' (f x) (g u))
  34     (hsurj:Function.Surjective (markedWheelVertexEmbedding (m:=m) (b:=b) f g hcomm e k het heh hkt hkh hH₀ hHK))
  35     (w:MarkedWheel tail' head' ktail' khead' m π b H₀' HK'→ℝ):
  36     let F:=markedWheelVertexEmbedding (m:=m) (b:=b) f g hcomm e k het heh hkt hkh hH₀ hHK
  37     (∀ a:E⊕K,markedWheelRimLoad (fun W=>w (F W)) a=markedWheelRimLoad w (Sum.map e k a))∧
  38     (∀ j v u,markedWheelSpokeLoad (fun W=>w (F W)) j v u=markedWheelSpokeLoad w j (f v) (g u)):=by

  52   · intro j v u
