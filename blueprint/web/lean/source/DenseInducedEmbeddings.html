  16 theorem induced_degree_delete_bound (G : SimpleGraph V) (A : Finset V) (v : (A:Set V)) :
  17     G.degree v.val ≤ (G.induce (A:Set V)).degree v+(Finset.univ\A).card := by
  18   rw [induced_degree_intersection]
  19   have he := Finset.card_inter_add_card_sdiff (G.neighborFinset v.val) A
  20   have hsub : G.neighborFinset v.val\A ⊆ Finset.univ\A := Finset.sdiff_subset_sdiff (Finset.subset_univ _) le_rfl
  21   have hh := Finset.card_le_card hsub
  22   rw [SimpleGraph.card_neighborFinset_eq_degree] at he
  23   omega

  25 theorem induced_dense_degree (G : SimpleGraph V) (A : Finset V) (n : ℕ) (β : ℝ)
  26     (hdeg : ∀ v,(1/2+β)*(n:ℝ) ≤ G.degree v)
  27     (hA : ((Finset.univ\A).card:ℝ) ≤ (β/2)*n) (v : (A:Set V)) :
  28     (1/2+β/2)*(n:ℝ) ≤ (G.induce (A:Set V)).degree v := by
  29   have hh : (G.degree v.val:ℝ) ≤ (G.induce (A:Set V)).degree v+((Finset.univ\A).card:ℝ) := by
  30     exact_mod_cast induced_degree_delete_bound G A v
  31   linarith [hdeg v.val]

  33 theorem fixed_cycle_inside_large_subset (β : ℝ) (hβ : 0 < β) (r b : ℕ) (hr : 3 ≤ r) :
  34     ∃ n₀ : ℕ,∀ n : ℕ,n₀ ≤ n →
  35     ∀ (V : Type*) [Fintype V] [DecidableEq V],Fintype.card V ≤ n →
  36     ∀ (G : SimpleGraph V),(∀ v,(1/2+β)*(n:ℝ) ≤ G.degree v) →
  37     ∀ (A S : Finset V),((Finset.univ\A).card:ℝ) ≤ (β/2)*n → S.card ≤ b →
  38     ∀ x : V,x∈A → x∉S →
  39     ∃ C : SimpleGraph V,IsEmbeddedCycle C r ∧ C ≤ G ∧ x∈C.support ∧
  40       C.support ⊆ (A:Set V) ∧ ∀ v∈C.support,v∉S := by
  41   obtain ⟨n₀,hembed⟩ := fixed_cycle_through_vertex (β/2) (by positivity) r b hr
  42   refine ⟨n₀,?_⟩
  43   intro n hn V _ _ hV G hdeg A S hA hS x hxA hxS
  44   let S' : Finset (A:Set V) := Finset.univ.filter (fun v => v.val∈S)
  45   have hS' : S'.card ≤ b := by
  46     let f : S' → S := fun v => ⟨v.val.val,(Finset.mem_filter.mp v.property).2⟩
  47     have hf : Function.Injective f := by
  48       intro v w he
  49       apply Subtype.ext
  50       apply Subtype.ext
  51       exact congrArg (fun z : S => z.val) he
  52     have hh := Fintype.card_le_of_injective f hf
  53     have hh' : S'.card ≤ S.card := by simpa only [Fintype.card_coe] using hh
  54     exact hh'.trans hS
  55   have hAcard : Fintype.card (A:Set V) ≤ n := by
  56     simpa using (Finset.card_le_univ A).trans hV
  57   obtain ⟨C,hC,hCG,hxC,havoid⟩ := hembed n hn (A:Set V) hAcard (G.induce (A:Set V))
  58     (induced_dense_degree G A n β hdeg hA) ⟨x,hxA⟩ S' hS' (by simpa [S'] using hxS)
  59   let f : (A:Set V) ↪ V := Function.Embedding.subtype _
  60   let D := C.map f
  61   have hDG : D ≤ G := by
  62     have hh := SimpleGraph.map_monotone f hCG
  63     exact hh.trans (spanningRestriction_le G A)
  64   refine ⟨D,hC.map f,hDG,?_,?_,?_⟩
  65   · rw [show D=C.map f from rfl,C.support_map]
  66     exact ⟨⟨x,hxA⟩,hxC,rfl⟩
  67   · intro v hv
  68     rw [show D=C.map f from rfl,C.support_map] at hv
  69     obtain ⟨w,hw,rfl⟩ := hv
  70     exact w.property
  71   · intro v hv
  72     rw [show D=C.map f from rfl,C.support_map] at hv
  73     obtain ⟨w,hw,rfl⟩ := hv
  74     have hh := havoid w hw
  75     change w.val ∉ S
  76     simpa [S'] using hh

  78 theorem fixed_hub_wheel_embedding (β : ℝ) (hβ : 0 < β) (r b : ℕ) (hr : 3 ≤ r) :
  79     ∃ n₀ : ℕ,∀ n : ℕ,n₀ ≤ n →
  80     ∀ (V U : Type*) [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U],Fintype.card V ≤ n →
  81     ∀ (G : SimpleGraph V) (H : V → U → Prop),
  82     (∀ v,(1/2+β)*(n:ℝ) ≤ G.degree v) →
  83     ∀ (u : U),((Finset.univ.filter (fun v => ¬H v u)).card:ℝ) ≤ (β/2)*n →
  84     ∀ (x : V) (S : Finset V),H x u → S.card ≤ b → x∉S →
  85     ∃ W : OrdinaryWheel G ⊥ H r,ordinaryWheelHub W=u ∧ x∈(ordinaryWheelRim W).support ∧
  86       ∀ v∈(ordinaryWheelRim W).support,v∉S := by
  87   obtain ⟨n₀,hn₀⟩ := fixed_cycle_inside_large_subset β hβ r b hr
  88   refine ⟨n₀,?_⟩
  89   intro n hn V U _ _ _ _ hV G H hdeg u hu x S hxu hS hx
  90   let A := Finset.univ.filter (fun v => H v u)
  91   have hAc : ((Finset.univ\A).card:ℝ) ≤ (β/2)*n := by
  92     have he : Finset.univ\A=Finset.univ.filter (fun v => ¬H v u) := by ext v;simp [A]
  93     rwa [he]
  94   obtain ⟨C,hC,hCG,hxC,hCA,havoid⟩ := hn₀ n hn V hV G hdeg A S hAc hS x (by simp [A,hxu]) hx
  95   let W : OrdinaryWheel G ⊥ H r := ⟨(⟨⟨C,hC⟩,hCG⟩,u),
  96     (fun v hv => (Finset.mem_filter.mp (hCA hv)).2),fun _ _ _ _ => not_false⟩
  97   exact ⟨W,rfl,hxC,havoid⟩
