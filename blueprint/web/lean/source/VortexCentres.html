   9 theorem rounded_vortex_centre_error {ε ρ ρ₀ D : ℝ}
  10     (he : 0 ≤ ε) (heq : ε < 1 / 4) (hp : 0 < ρ₀) (hρ : ρ₀ ≤ ρ)
  11     {n u : ℕ} (hn : 0 < n)
  12     (hu : (u : ℝ) ≤ (n : ℝ) / 2) (hcentre : 2 * (u : ℝ) = ρ * D) (i : ℕ) :
  13     |2 * (floorOrbit ε u i : ℝ) / ρ -
  14       (D / n) * floorOrbit ε n i| < 3 / ρ₀ := by
  15   have hnreal : (0 : ℝ) < n := by exact_mod_cast hn
  16   have hρpos : 0 < ρ := hp.trans_le hρ
  17   let a := (u : ℝ) / n
  18   have ha : 0 ≤ a := by dsimp [a]; positivity
  19   have ha1 : a ≤ 1 := by
  20     dsimp [a]
  21     apply (div_le_iff₀ hnreal).mpr
  22     linarith
  23   have hinit : (u : ℝ) = a * n := by dsimp [a]; field_simp
  24   have hr := floorOrbit_rounding_error he (by linarith) ha ha1 hinit i
  25   have hden : 0 < 1 - ε := by linarith
  26   have hone : 1 / (1 - ε) < (3 / 2 : ℝ) := by
  27     apply (div_lt_iff₀ hden).mpr
  28     linarith
  29   have herr : |(floorOrbit ε u i : ℝ) - a * floorOrbit ε n i| < 3 / 2 := hr.trans hone
  30   have hcoef : (2 / ρ) * a = D / n := by
  31     dsimp [a]
  32     field_simp
  33     nlinarith [hcentre]
  34   have heq' : 2 * (floorOrbit ε u i : ℝ) / ρ - (D / n) * floorOrbit ε n i =
  35       (2 / ρ) * ((floorOrbit ε u i : ℝ) - a * floorOrbit ε n i) := by
  36     rw [← hcoef]
  37     ring
  38   rw [heq', abs_mul, abs_of_nonneg (by positivity : 0 ≤ 2 / ρ)]
  39   calc
  40     _ < (2 / ρ) * (3 / 2) := mul_lt_mul_of_pos_left herr (by positivity)
  41     _ = 3 / ρ := by ring
  42     _ ≤ 3 / ρ₀ := div_le_div_of_nonneg_left (by norm_num) hp hρ
