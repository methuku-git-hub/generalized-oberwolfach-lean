  10 theorem mapped_edge_finset_mem_physical_set
  11     {X Y:Type*} [Fintype X] [DecidableEq X] [DecidableEq Y]
  12     (F:SimpleGraph X) (f:X→Y) (e:Sym2 Y)
  13     (he:e∈F.edgeFinset.image (Sym2.map f)) : e∈partialFactorPhysicalEdgesPoly X F f := by
  14   classical
  15   obtain ⟨a,ha,heq⟩:=Finset.mem_image.mp he
  16   induction a using Sym2.inductionOn with
  17   | _ x y =>
  18     have hxy:F.Adj x y:=by simpa using ha
  19     refine ⟨x,y,hxy,?_⟩
  20     simpa using heq.symm

  22 theorem completed_two_factor_set_packing_gives_decomposition {n D:ℕ}
  23     (hEven:Even D) (G:SimpleGraph (Fin n)) (hG:G.IsRegularOfDegree D)
  24     (F:Fin (D/2)→SimpleGraph (Fin n)) (hF:∀i,(F i).IsRegularOfDegree 2)
  25     (f:Fin (D/2)→Fin n↪Fin n)
  26     (hgraph:∀i x y,(F i).Adj x y→G.Adj (f i x) (f i y))
  27     (hdis:∀i j,i≠j→Disjoint (partialFactorPhysicalEdgesPoly (Fin n) (F i) (f i))
  28       (partialFactorPhysicalEdgesPoly (Fin n) (F j) (f j))) :
  29     HasFactorDecomposition G F := by
  30   classical
  31   apply even_regular_two_factor_embeddings_give_decomposition hEven G hG F hF (fun i=>f i) (fun i=>(f i).injective)
  32   · intro i e he
  33     obtain ⟨x,y,hxy,heq⟩:=mapped_edge_finset_mem_physical_set (F i) (f i) e he
  34     rw [heq]
