  13     (c n start:ℕ)
  14     (hp:∀i,start ≤ i → i < start+n → (p i).IsPath)
  15     (hlen:∀i,start ≤ i → i < start+n → (p i).length=c)
  16     (hdis:∀i j,start ≤ i → i < start+n → start ≤ j → j < start+n → i ≠ j → 
  17       ∀x ∈ (p i).support.dropLast,x ∉ (p j).support.dropLast)
  18     (hend:∀i,start ≤ i → i < start+n → v (start+n) ∉ (p i).support.dropLast):
  19     ∃q:G.Walk (v start) (v (start+n)),q.IsPath ∧ q.length=n*c ∧
  20       (∀x,x ∈ q.support ↔ x=v (start+n) ∨
  21         ∃i,start ≤ i ∧ i < start+n ∧ x ∈ (p i).support.dropLast) ∧
  22       (∀e,e ∈ q.darts ↔ ∃i,start ≤ i ∧ i < start+n ∧ e ∈ (p i).darts) := by
  23   induction n generalizing start with
  24   | zero =>
  25     refine ⟨SimpleGraph.Walk.nil,SimpleGraph.Walk.IsPath.nil,by simp,?_,?_⟩
  26     · intro x
  27       simp
  28       omega
  29     · intro e
  30       simp
  31       omega
  32   | succ n ih =>
  33     have heq:start+1+n=start+(n+1):=by omega
  34     have hp':∀i,start+1 ≤ i → i < start+1+n → (p i).IsPath:=by
  35       intro i hi hj;apply hp i  <;> omega
  36     have hl':∀i,start+1 ≤ i → i < start+1+n → (p i).length=c:=by
  37       intro i hi hj;apply hlen i  <;> omega
  38     have hd':∀i j,start+1 ≤ i → i < start+1+n → start+1 ≤ j → j < start+1+n → i ≠ j → 
  39         ∀x ∈ (p i).support.dropLast,x ∉ (p j).support.dropLast:=by
  40       intro i j hi hj hk hl hij x hx
  41       exact hdis i j (by omega) (by omega) (by omega) (by omega) hij x hx
  42     have he':∀i,start+1 ≤ i → i < start+1+n → v (start+1+n) ∉ (p i).support.dropLast:=by
  43       intro i hi hj
  44       rw [heq]
  45       exact hend i (by omega) (by omega)
  46     obtain ⟨q,hq,hqlen,hqmem,hqedges⟩:=ih (start+1) hp' hl' hd' he'
  47     have hfirst:(p start).IsPath:=hp start (by omega) (by omega)
  48     have hfirstlen:(p start).length=c:=hlen start (by omega) (by omega)
  49     have hsep:∀x ∈ (p start).support.dropLast,x ∉ q.support:=by
  50       intro x hx hq'
  51       rcases (hqmem x).mp hq' with hterminal | ⟨i,hi,hupper,hxi⟩
  52       · have hh:=hend start (by omega) (by omega)
  53         rw [←heq,←hterminal] at hh
  54         exact hh hx
  55       · exact hdis start i (by omega) (by omega) (by omega) (by omega) (by omega) x hx hxi
  56     obtain ⟨hsimple,hsum,hsupport⟩:=simple_walk_append_of_disjoint_initial_support (p start) q hfirst hq hsep
  57     have hresult:∃w:G.Walk (v start) (v (start+1+n)),w.IsPath ∧ w.length=(n+1)*c ∧
  58       (∀x,x ∈ w.support ↔ x=v (start+1+n) ∨
  59         ∃i,start ≤ i ∧ i < start+1+n ∧ x ∈ (p i).support.dropLast) ∧
  60       (∀e,e ∈ w.darts ↔ ∃i,start ≤ i ∧ i < start+1+n ∧ e ∈ (p i).darts):=by
  61       refine ⟨(p start).append q,hsimple,?_,?_,?_⟩
  62       · calc
  63           ((p start).append q).length=(p start).length+q.length:=hsum
  64           _=c+n*c:=by rw [hfirstlen,hqlen]
  65           _=(n+1)*c:=by nlinarith
  66       · intro x
  67         rw [hsupport,List.mem_append,hqmem]
