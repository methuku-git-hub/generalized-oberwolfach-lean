  10 theorem long_approximate_spoke_coefficients_sum_one {U I:Type} [Fintype I]
  11     (m:ℕ) (a b:U→ℝ) (f:I→U→ℝ)
  12     (hb:∀u,0<b u) (hbalance:∀u,b u=(m:ℝ)*a u+∑i,f i u):
  13     ∀u,(m:ℝ)*a u/b u+∑i,f i u/b u=1:=by

  18 theorem long_approximate_ordinary_rim_coefficients_sum_one {U I:Type} [Fintype U] [Fintype I]
  19     (m:ℕ) (D p k:ℝ) (hD:0<D) (hp:0<p)
  20     (a b:U→ℝ) (f:I→U→ℝ)
  21     (hbalance:∀u,b u=(m:ℝ)*a u+∑i,f i u)
  22     (hpdef:p=(∑u,b u)/D) (hkdef:k=(∑u,a u)/D):
  23     (m:ℝ)*k/p+∑i,(∑u,f i u)/(D*p)=1:=by
  24   have hsum:(∑u,b u)=(m:ℝ)*(∑u,a u)+∑i,∑u,f i u:=by
  25     simp_rw [hbalance]
  26     rw [Finset.sum_add_distrib,←Finset.mul_sum,Finset.sum_comm]
  27   rw [←Finset.sum_div]
  28   have hpD:D*p=∑u,b u:=by rw [hpdef];field_simp
  29   have hkD:D*k=∑u,a u:=by rw [hkdef];field_simp
  30   have hfirst:(m:ℝ)*k/p=((m:ℝ)*(D*k))/(D*p):=by field_simp
  31   rw [hfirst,←add_div]
  32   have hden:0<D*p:=mul_pos hD hp
  33   apply (div_eq_one_iff_eq (ne_of_gt hden)).mpr
  34   nlinarith [hsum,hpD,hkD]
  35 end Oberwolfach
