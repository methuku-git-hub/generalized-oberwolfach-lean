  13 theorem finite_sum_good_approximation {X : Type*} [Fintype X]
  14     (P : X → Prop) (f : X → ℝ) {b ε : ℝ} (hb : 0≤b) (hε : 0≤ε)
  15     (hgood : ∀ x,¬P x → b≤f x ∧ f x≤(1+ε)*b)
  16     (hbad : ∀ x,P x → f x=0) :
  17     |(∑ x,f x)-(Fintype.card X : ℝ)*b|≤
  18       (Fintype.card X : ℝ)*ε*b+
  19         ((Finset.univ.filter P).card : ℝ)*b := by
  20   classical
  21   have hpoint (x : X) : |f x-b|≤ε*b+(if P x then b else 0) := by
  22     by_cases hx : P x
  23     · rw [hbad x hx,if_pos hx,zero_sub,abs_neg,abs_of_nonneg hb]
  24       nlinarith
  25     · rw [if_neg hx,add_zero,abs_of_nonneg (sub_nonneg.mpr (hgood x hx).1)]
  26       linarith [(hgood x hx).2]
  27   calc
  28     _ = |∑ x,(f x-b)| := by rw [Finset.sum_sub_distrib];simp
  29     _ ≤ ∑ x,|f x-b| := Finset.abs_sum_le_sum_abs _ _
  30     _ ≤ ∑ x,(ε*b+(if P x then b else 0)) := Finset.sum_le_sum (fun x _=>hpoint x)
  31     _ = _ := by
  32       rw [Finset.sum_add_distrib]
  33       simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul]
  34       have he : (∑ x,if P x then b else 0)=((Finset.univ.filter P).card : ℝ)*b := by
  35         rw [← Finset.sum_filter]
  36         simp
  37       rw [he]
  38       ring
