  14 theorem cyclicInterfaceGraph_fractional_cycles_bounded {k : ℕ} {X Y : Type*}
  15     [Fintype X] [Fintype Y] [DecidableEq X] [DecidableEq Y] [Nonempty X]
  16     (L : Fin (k + 2) → X → Y → Prop) (R : Fin (k + 2) → Y → X → Prop)
  17     (δ α : ℝ) (hn : 0 < Fintype.card Y) (hα : 0 < α)
  18     (hcard : Fintype.card X = Fintype.card Y)
  19     (hthreshold : ((k + 2) + (2 * (k + 2) - 1) * α) * Fintype.card Y ≤
  20       (2 * (k + 2) - 1) * δ)
  21     (hdL : ∀ i x, δ ≤ (Finset.univ.filter (L i x)).card)
  22     (hdY : ∀ i y, δ ≤ (Finset.univ.filter (fun x => L i x y)).card)
  23     (hY : ∀ i y, (Finset.univ.filter (fun x => L i x y)).card =
  24       (Finset.univ.filter (R i y)).card)
  25     (hX : ∀ i z, (Finset.univ.filter (fun y => R i y z)).card =
  26       (Finset.univ.filter (L (finRotate (k + 2) i) z)).card) :
  27     ∃ w : SimpleGraph ((Fin (k + 2) × X) ⊕ (Fin (k + 2) × Y)) → ℝ,
  28       (∀ H, 0 ≤ w H) ∧
  29       (∀ H, 0 < w H → H ≤ cyclicInterfaceGraph L R ∧
  30         IsEmbeddedCycle H ((k + 2) * 2) ∧
  31           ∃ p : Fin (k + 2) → X × Y × X, H = alternatingTupleGraph p) ∧
  32       (∀ u v, (∑ H, if H.Adj u v then w H else 0) =
  33         if (cyclicInterfaceGraph L R).Adj u v then 1 else 0) ∧
  34       (∀ H, H ≤ cyclicInterfaceGraph L R →
  35         (∃ p : Fin (k + 2) → X × Y × X, H = alternatingTupleGraph p) →
  36         ((k + 2 : ℝ) * cyclicSectorLower k α) / (Fintype.card Y : ℝ) ^ (2 * k + 2) ≤ w H ∧
  37         w H ≤ ((k + 2 : ℝ) * cyclicSectorUpper k α) /
  38           (Fintype.card Y : ℝ) ^ (2 * k + 2)) := by
  39   obtain ⟨ω, hω, _, hl, hr, hc, hb⟩ := cyclic_interfaces_fractional_weighting_bounded
  40     L R δ α hn hα hcard hthreshold hdL hdY hY hX
  41   obtain ⟨hs, he⟩ := cyclic_weighting_graph_equations L R ω hω hl hr hc
  42   have hprops := indexed_fractional_weight_properties (cyclicInterfaceGraph L R)
  43     (fun H => IsEmbeddedCycle H ((k + 2) * 2) ∧
  44       ∃ p : Fin (k + 2) → X × Y × X, H = alternatingTupleGraph p)
  45     alternatingTupleGraph ω hω
  46     (fun p hp => ⟨hs p hp, ⟨⟨by omega, alternatingCycleEmbedding p, rfl⟩, p, rfl⟩⟩) he
  47   refine ⟨indexedFractionalWeight alternatingTupleGraph ω, hprops.1, hprops.2.1,
  48     hprops.2.2, ?_⟩
  49   intro H hsub hpart
  50   obtain ⟨p, rfl⟩ := hpart
  51   let v := cyclicTupleVertices p
  52   have hcanon : alternatingTupleGraph (cyclicTupleOfVertices v) = alternatingTupleGraph p := rfl
  53   have hactual := (canonicalCycleGraph_le_iff L R v).mp (hcanon ▸ hsub)
  54   have hweight : indexedFractionalWeight alternatingTupleGraph ω (alternatingTupleGraph p) =
  55       ω (cyclicTupleOfVertices v) := by
  56     rw [← hcanon]
  57     exact cyclic_weight_of_canonical_graph ω hω hc v
  58   rw [hweight]
  59   exact hb (cyclicTupleOfVertices v) (fun _ => rfl) hactual
