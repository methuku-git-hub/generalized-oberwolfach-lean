  14 theorem triangular_path_endpoint_geometry (G Q:SimpleGraph V) (H₀:V→U→Prop)
  15     (φ:V→V) (hφ:∀ v w,G.Adj v w→φ v≠φ w)
  16     (b:∀ v,G.neighborSet v≃({u:U // H₀ v u}×Fin 2))
  17     (σ:∀ v,Equiv.Perm (G.neighborSet v)) (hσ:∀ v w,σ v w≠w) (h:G.edgeFinset↪U)
  18     (p:(G.edgeFinset⊕{e:V×U // H₀ e.1 e.2})→(Σ a:V,Σ c:V,Q.Walk a c))
  19     (hrim:∀ e:G.edgeFinset,(p (.inl e)).2.2.length=0)
  20     (hsp:∀ e:{e:V×U // H₀ e.1 e.2},(p (.inr e)).2.2.IsPath∧0<(p (.inr e)).2.2.length)
  21     (hdis:∀ i j,i≠j→Disjoint (p i).2.2.support.toFinset (p j).2.2.support.toFinset):
  22     let a:=transformerRimEndpoint G Q H₀ p
  23     let z:=transformerSpokeEndpoint G Q H₀ b p
  24     let u:=transformerAssignedHub G σ h
  25     Function.Injective z ∧ (∀ d e,a d≠z e) ∧
  26       Function.Injective (fun d:Σ v,G.neighborSet v=>(φ d.1,a d)) ∧
  27       Function.Injective (fun d=>(a d,u d)):=by
  28   dsimp only
  29   let pe:G.edgeFinset↪V:=⟨fun e=>(p (.inl e)).1,by
  30     intro e f he
  31     change (p (.inl e)).1=(p (.inl f)).1 at he
  32     by_contra hne
  33     exact Finset.disjoint_left.mp (hdis (.inl e) (.inl f) (fun hh=>hne (Sum.inl.inj hh)))
  34       (List.mem_toFinset.mpr (p (.inl e)).2.2.start_mem_support)
  35       (he.symm ▸ List.mem_toFinset.mpr (p (.inl f)).2.2.start_mem_support)⟩
  36   have ha (d:Σ v,G.neighborSet v):transformerRimEndpoint G Q H₀ p d=pe (rimIncidenceEquiv G d).1:=by
  37     have hend:=(p (.inl (rimIncidenceEquiv G d).1)).2.2.eq_of_length_eq_zero (hrim _)
  38     simp only [transformerRimEndpoint,graphPathEndpoint,←hend,ite_self]
  39     rfl
  40   have hzinj:Function.Injective (transformerSpokeEndpoint G Q H₀ b p):=by
  41     have hp:=disjoint_path_endpoint_injective (fun e:{e:V×U // H₀ e.1 e.2}=>p (.inr e))
  42       (fun e=>(hsp e).1) (fun e=>(hsp e).2)
  43       (fun e f hef=>hdis (.inr e) (.inr f) (fun hh=>hef (Sum.inr.inj hh)))
  44     intro d e he
  45     exact (spokeEndpointEquiv G H₀ b).injective (hp he)
  46   refine ⟨hzinj,?_,?_,?_⟩
  47   · intro d e he
  48     have ha:=graphPathEndpoint_mem (p (.inl (rimIncidenceEquiv G d).1)) (rimIncidenceEquiv G d).2
  49     have hz:=graphPathEndpoint_mem (p (.inr (spokeEndpointEquiv G H₀ b e).1)) (spokeEndpointEquiv G H₀ b e).2
  50     change transformerRimEndpoint G Q H₀ p d∈_ at ha
  51     change transformerSpokeEndpoint G Q H₀ b p e∈_ at hz
  52     exact Finset.disjoint_left.mp (hdis _ _ Sum.inl_ne_inr) (List.mem_toFinset.mpr ha)
  53       (he.symm ▸ List.mem_toFinset.mpr hz)
  54   · simpa only [ha] using triangle_root_shared_endpoint_injective G φ pe hφ
  55   · simpa only [ha] using triangle_shared_endpoint_hub_injective G σ hσ pe h
