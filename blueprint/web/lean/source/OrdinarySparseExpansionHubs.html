  15 theorem common_hub_from_count (H : V → U → Prop) (T : Finset V) (B : Finset U) (c : ℝ)
  16     (hcount : c≤((common Finset.univ T (fun v => Finset.univ.filter (H v))).card:ℝ))
  17     (hbudget : (B.card:ℝ)<c) : ∃ u,u∉B ∧ ∀ v∈T,H v u := by
  18   have hsize : B.card<(common Finset.univ T (fun v => Finset.univ.filter (H v))).card := by
  19     exact_mod_cast hbudget.trans_le hcount
  20   obtain ⟨u,hu,huB⟩ := Finset.exists_mem_notMem_of_card_lt_card hsize
  21   exact ⟨u,huB,by simpa only [common,Finset.mem_filter,Finset.mem_univ,true_and] using hu⟩

  23 theorem sparse_rim_expansion_embedding (Q F : SimpleGraph V) (H : V → U → Prop)
  24     (A W R : Finset V) (B : Finset U) (x y : V) (r : ℕ) (hr : 3≤r) (d c cH : ℝ) (hd : 0≤d)
  25     (hF : ∀ v,(F.degree v:ℝ)≤d)
  26     (hcount : ∀ v w,c≤((A∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  27     (hbudget : W.card+((R.card:ℝ)+r+6)*(d+1)<c)
  28     (hhub : ∀ T:Finset V,T.card≤r → (∀ v∈T,∀ w∈T,¬F.Adj v w) →
  29       cH≤((common Finset.univ T (fun v => Finset.univ.filter (H v))).card:ℝ))
  30     (hB : (B.card:ℝ)<cH) :
  31     ∃ (f : Fin r ↪ V) (u : U),
  32       (∀ i j,(SimpleGraph.cycleGraph r).Adj i j → Q.Adj (f i) (f j)) ∧
  33       (∀ i,H (f i) u) ∧ u∉B ∧
  34       Q.Adj y (f ⟨0,by omega⟩) ∧ Q.Adj x (f ⟨r-1,by omega⟩) ∧
  35       (∀ i j,¬F.Adj (f i) (f j)) ∧
  36       ∀ i,f i∈A ∧ f i∉W ∧ f i∉R ∧ ∀ v∈R,¬F.Adj (f i) v := by
  37   obtain ⟨f,hf,hy,hx,hind,havoid⟩ := sparse_rim_expansion_cycle Q F A W R x y r hr d c hd hF hcount hbudget
  38   let T := Finset.univ.image f
  39   have hTc : T.card≤r := Finset.card_image_le.trans (by simp)
  40   have hTF : ∀ v∈T,∀ w∈T,¬F.Adj v w := by
  41     intro v hv w hw
  42     obtain ⟨i,_,rfl⟩ := Finset.mem_image.mp hv
  43     obtain ⟨j,_,rfl⟩ := Finset.mem_image.mp hw
  44     exact hind i j
  45   obtain ⟨u,hu,hHu⟩ := common_hub_from_count H T B cH (hhub T hTc hTF) hB
  46   exact ⟨f,u,hf,(fun i => hHu _ (Finset.mem_image.mpr ⟨i,Finset.mem_univ _,rfl⟩)),hu,hy,hx,hind,havoid⟩

  48 theorem sparse_spoke_expansion_embedding (Q F : SimpleGraph V) (H : V → U → Prop)

  55     (hB : (B.card:ℝ)<cH) :

  62     intro v hv w hw
  63     rw [Finset.mem_singleton.mp hv,Finset.mem_singleton.mp hw]
  64     exact F.loopless.irrefl x
