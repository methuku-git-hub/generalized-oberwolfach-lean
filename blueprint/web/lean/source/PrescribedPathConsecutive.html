  10 theorem prescribed_path_consecutive (D:V→V→Prop) (F:SimpleGraph V) (forward:Bool):
  11     ∀ xs:List V,∀ x:V,DirectedPathFrom D F forward x xs→
  12     ∀ i:ℕ,∀ hi:i+1<(x::xs).length,
  13       (if forward then D ((x::xs)[i]'(by omega)) ((x::xs)[i+1]'(by omega))
  14        else D ((x::xs)[i+1]'(by omega)) ((x::xs)[i]'(by omega))) ∧
  15       ¬F.Adj ((x::xs)[i]'(by omega)) ((x::xs)[i+1]'(by omega)) := by
  16   intro xs
  17   induction xs with
  18   | nil =>
  19     intro x hpath i hi
  20     simp at hi
  21   | cons z zs ih =>
  22     intro x hpath i hi
  23     cases i with
  24     | zero => simpa using And.intro hpath.1 hpath.2.1
  25     | succ i =>
  26       have hi':i+1<(z::zs).length:=by simpa using hi
  27       have hh:=ih z hpath.2.2 i hi'
  28       simpa using hh
  29 end Oberwolfach
