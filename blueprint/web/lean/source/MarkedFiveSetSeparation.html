  11 theorem marked_five_sets_separated (π:Equiv.Perm V) (b:ℕ)
  12     (L R:Finset V) (T:Fin 3→Finset V) (O:Finset V)
  13     (hL:∀ v∈L\O,∀ w∈L\O,v≠w→¬PermutationClose π b v w)
  14     (hR:∀ v∈R,∀ w∈R,v≠w→¬PermutationClose π b v w)
  15     (hT:∀ i,∀ v∈T i,∀ w∈T i,v≠w→¬PermutationClose π b v w)
  16     (hLR:∀ v∈L,∀ w∈R,¬PermutationClose π b v w)
  17     (hLT:∀ i,∀ v∈L,∀ w∈T i,¬PermutationClose π b v w)
  18     (hRT:∀ i,∀ v∈R,∀ w∈T i,¬PermutationClose π b v w)
  19     (hTT:∀ i j,i≠j→∀ v∈T i,∀ w∈T j,¬PermutationClose π b v w):
  20     ∀ v∈ordinaryFiveUnion L R T\O,∀ w∈ordinaryFiveUnion L R T\O,v≠w→¬PermutationClose π b v w:=by
  21   intro v hv w hw hne
  22   obtain ⟨hv,hvO⟩:=Finset.mem_sdiff.mp hv
  23   obtain ⟨hw,hwO⟩:=Finset.mem_sdiff.mp hw
  24   have cases_mem:∀ x∈ordinaryFiveUnion L R T,x∈L∨x∈R∨∃ i,x∈T i:=by
  25     intro x hx
  26     rcases Finset.mem_union.mp hx with hx|hx
  27     · rcases Finset.mem_union.mp hx with hx|hx
  28       · exact Or.inl hx
  29       · exact Or.inr (Or.inl hx)
  30     · obtain ⟨i,_,hx⟩:=Finset.mem_biUnion.mp hx
  31       exact Or.inr (Or.inr ⟨i,hx⟩)
  32   rcases cases_mem v hv with hv|hv|⟨i,hv⟩ <;> rcases cases_mem w hw with hw|hw|⟨j,hw⟩
  33   · exact hL v (Finset.mem_sdiff.mpr ⟨hv,hvO⟩) w (Finset.mem_sdiff.mpr ⟨hw,hwO⟩) hne
  34   · exact hLR v hv w hw
  35   · exact hLT j v hv w hw
  36   · intro hc;exact hLR w hw v hv ((permutationClose_symm π b _ _).mp hc)
  37   · exact hR v hv w hw hne
  38   · exact hRT j v hv w hw
  39   · intro hc;exact hLT i w hw v hv ((permutationClose_symm π b _ _).mp hc)
  40   · intro hc;exact hRT i w hw v hv ((permutationClose_symm π b _ _).mp hc)
  41   · by_cases hij:i=j
  42     · subst j;exact hT i v hv w hw hne
  43     · exact hTT i j hij v hv w hw
  44 
  45 end Oberwolfach
