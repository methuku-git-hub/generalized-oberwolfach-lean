  13 theorem ordinary_actual_diagram_orders (G:SimpleGraph V₀) (Hs:V₀→U₀→Prop)
  14     (r m:ℕ) (hr:3≤r) (hm:Fintype.card V₀+Fintype.card U₀≤m) :
  15     let k:=G.edgeFinset.card/r
  16     let Gc:=ordinaryComparisonRim r k (by omega)
  17     let Hc:=ordinaryComparisonSpokes r k (by omega)
  18     (Fintype.card (ActualExpansionRow G Hs (Fin r))+Fintype.card (ActualExpansionHub G Hs)≤ordinaryDiagramSize r m)∧
  19     (Fintype.card (ActualExpansionRow Gc Hc (Fin r))+Fintype.card (ActualExpansionHub Gc Hc)≤ordinaryDiagramSize r m)∧
  20     (∀ i:Fin 3,Fintype.card (ExpansionRow (ordinaryDiagramOldRows G r i) (ordinaryDiagramEdges G r hr i)
  21         (ordinaryDiagramSpokes G Hs r hr i) (Fin r))+
  22       Fintype.card (ExpansionHub (ordinaryDiagramOldHubs (U₀:=U₀) G r i) (ordinaryDiagramEdges G r hr i)
  23         (ordinaryDiagramSpokes G Hs r hr i))≤ordinaryDiagramSize r m) := by
  24   dsimp only
  25   refine ⟨(actual_expansion_order_bound G Hs r m hm).trans ?_,?_,?_⟩
  26   · dsimp [ordinaryDiagramSize];omega
  27   · have hJ:=ordinary_comparison_order_bound G r m (by omega)
  28     exact (actual_expansion_order_bound _ _ r (m^2*(r+2)) hJ).trans (by dsimp [ordinaryDiagramSize];omega)
  29   · intro i
  30     rw [ordinaryExpansion_order]
  31     simpa only [Fintype.card_fin] using ordinary_diagram_model_size G Hs r m hr hm i
