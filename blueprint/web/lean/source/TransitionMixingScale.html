  10 theorem geometric_decay_after_logarithmic_steps (κ C ε : ℝ) (q : ℕ)
  11     (hκ : 0 < κ) (hκ1 : κ < 1) (hC : 0 < C) (hε : 0 < ε)
  12     (hq : Real.log (C/ε) / (-Real.log (1-κ)) ≤ (q : ℝ)) :
  13     (1-κ)^q*C ≤ ε := by
  14   have hp : 0 < 1-κ := sub_pos.mpr hκ1
  15   have hlog : Real.log (1-κ) < 0 := Real.log_neg hp (by linarith)
  16   have hd : 0 < -Real.log (1-κ) := neg_pos.mpr hlog
  17   have hq' := (div_le_iff₀ hd).mp hq
  18   rw [Real.log_div hC.ne' hε.ne'] at hq'
  19   apply (Real.log_le_log_iff (mul_pos (pow_pos hp _) hC) hε).mp
  20   rw [Real.log_mul (pow_ne_zero _ hp.ne') hC.ne',Real.log_pow]
  21   nlinarith

  23 theorem long_capacity_mixing_steps (κ B : ℝ) (hκ : 0 < κ) (hκ1 : κ < 1)
  24     (hB : 1 ≤ B) :
  25     (1-κ)^(⌈Real.log (10*(B+1)) / (-Real.log (1-κ))⌉₊)*(B+1) ≤ 1/10 := by
  26   apply geometric_decay_after_logarithmic_steps κ (B+1) (1/10) _ hκ hκ1 (by linarith) (by norm_num)
  27   convert Nat.le_ceil (Real.log (10*(B+1)) / (-Real.log (1-κ))) using 1
  28   congr 2 <;> ring

  34 theorem transition_segment_mixing {X : Type*} [Fintype X]
  35     (R : ℕ → X → X → ℝ) (π : ℕ → X → ℝ)
  36     (hR : ∀ i, IsTransition (R i)) (hπ0 : ∀ i x, 0 ≤ π i x)
  37     (hπsum : ∀ i, ∑ x, π i x = 1)
  38     (hπ : ∀ i, transitionApply (π i) (R i) = π (i+1))
  39     (i L q : ℕ) (κ B ε : ℝ) (hB : 0 ≤ B) (hL : 1+q*3 ≤ L)
  40     (hupper : ∀ x y, R i x y ≤ B*π (i+1) y)
  41     (hminor : ∀ j < q, ∀ x y, κ*π (i+1+(j+1)*3) y ≤ transitionWalk R (i+1+j*3) 3 x y)
  42     (hdecay : (1-κ)^q*(B+1) ≤ ε) :
  43     ∀ x, RelativeError (transitionWalk R i L x) (π (i+L)) ε := by
  44   intro x
  45   have hinit := transition_initial_relative_error (R i x) (π (i+1)) B hB
  46     ((hR i).1 x) (hπ0 (i+1)) (hupper x)
  47   have hcontract := relative_error_block_contraction R π hR hπsum hπ (R i x)
  48     ((hR i).2 x) (i+1) 3 q κ (B+1) hinit hminor
  49   have hrow : transitionApply (R i x) (transitionWalk R (i+1) (q*3)) =
  50       transitionWalk R i (1+q*3) x := by
  51     rw [transitionWalk_add,transitionWalk_one]
  52     rfl
  53   rw [hrow] at hcontract
  54   have hcontract' : RelativeError (transitionWalk R i (1+q*3) x) (π (i+(1+q*3))) ε := by
  55     convert relative_error_mono _ _ _ _ (hπ0 _) hdecay hcontract using 1 <;> congr 1 <;> omega
  56   have htail := relative_error_propagated_walk R π hR hπ (i+(1+q*3)) (L-(1+q*3))
  57     (transitionWalk R i (1+q*3) x) ε hcontract'
  58   have hcompose : transitionApply (transitionWalk R i (1+q*3) x)
  59       (transitionWalk R (i+(1+q*3)) (L-(1+q*3))) = transitionWalk R i L x := by
  60     change transitionCompose _ _ x = _
  61     rw [← transitionWalk_add, Nat.add_sub_of_le hL]
  62   rw [hcompose] at htail
  63   convert htail using 1 <;> congr 1 <;> omega
