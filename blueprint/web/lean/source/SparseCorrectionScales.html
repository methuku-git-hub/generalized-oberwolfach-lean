   9 theorem sparse_correction_numeric {r n : ℕ} {δ ω ξ η h : ℝ}
  10     (hr : 0<r) (hδ : 0<δ) (hδ1 : δ≤1) (hω : 0<ω) (hω1 : ω≤1)
  11     (hξ : 0<ξ) (hξh : ξ≤h/100) (hη : 0≤η) (hηsmall : η≤δ*ω*ξ/1000)
  12     (hn : 100*((r : ℝ)+2)≤δ*ω*ξ*n) :
  13     let base := Nat.floor (ξ*n)
  14     (r : ℝ)≤δ*base ∧ 1≤δ*ω*base ∧
  15     η*n+2≤2*ω*base ∧
  16     3/2+η*n≤δ*(ω*base-η*n/2) ∧
  17     (base : ℝ)+r+η*n/2≤h*n/2 ∧
  18     2*((base : ℝ)+r+1)+η*n≤4*ξ*n ∧
  19     (base : ℝ)+r+1≤2*ξ*n := by
  20   intro base
  21   have hn0 : (0 : ℝ)≤n := Nat.cast_nonneg n
  22   have hr1 : (1 : ℝ)≤r := by exact_mod_cast hr
  23   have ha : 0≤ξ*n := mul_nonneg hξ.le hn0
  24   have hbl : ξ*n-1≤(base : ℝ) := by
  25     have hh := Nat.lt_floor_add_one (ξ*n)
  26     change ξ*n<(base : ℝ)+1 at hh
  27     linarith
  28   have hbu : (base : ℝ)≤ξ*n := Nat.floor_le ha
  29   have hc0 : 0≤δ*ω := mul_nonneg hδ.le hω.le
  30   have hc1 : δ*ω≤1 := by nlinarith [mul_le_mul_of_nonneg_left hω1 hδ.le]
  31   have hpA : δ*ω*(ξ*n)≤ξ*n := by nlinarith [mul_le_mul_of_nonneg_right hc1 ha]
  32   have hpW : δ*ω*(ξ*n)≤ω*(ξ*n) := by
  33     have hh := mul_le_mul_of_nonneg_right hδ1 (mul_nonneg hω.le ha)
  34     nlinarith
  35   have he0 : 0≤η*n := mul_nonneg hη hn0
  36   have hesmall : η*n≤δ*ω*(ξ*n)/1000 := by
  37     have hh := mul_le_mul_of_nonneg_right hηsmall hn0
  38     nlinarith
  39   have hδbl : δ*(ξ*n)-δ≤δ*base := by nlinarith [mul_le_mul_of_nonneg_left hbl hδ.le]
  40   have hcbl : δ*ω*(ξ*n)-1≤δ*ω*base := by
  41     have hh := mul_le_mul_of_nonneg_left hbl hc0
  42     nlinarith
  43   have hωbl : ω*(ξ*n)-1≤ω*base := by
  44     have hh := mul_le_mul_of_nonneg_left hbl hω.le
  45     nlinarith
  46   have hpδ : δ*ω*(ξ*n)≤δ*(ξ*n) := by
  47     have hh := mul_le_mul_of_nonneg_left hω1 (mul_nonneg hδ.le ha)
  48     nlinarith
  49   have hδe : δ*(η*n)≤η*n := by nlinarith [mul_le_mul_of_nonneg_right hδ1 he0]
  50   have hcap : 2*(ξ*n)≤h*n/2 := by
  51     have hh := mul_le_mul_of_nonneg_right hξh hn0
  52     nlinarith
  53   refine ⟨?_,?_,?_,?_,?_,?_,?_⟩ <;> nlinarith

  55 /-- The order threshold is chosen before the vertex classes, graphs,
  56 imbalance function or forbidden edges. -/
  57 theorem sparse_correction_order_threshold (r : ℕ) {δ ω ξ : ℝ}
  58     (hδ : 0<δ) (hω : 0<ω) (hξ : 0<ξ) :
  59     ∃ n₀ : ℕ,1≤n₀ ∧ ∀ n : ℕ,n₀≤n →
  60       100*((r : ℝ)+2)≤δ*ω*ξ*n := by
  61   obtain ⟨N,hN⟩ := exists_nat_ge (100*((r : ℝ)+2)/(δ*ω*ξ))
  62   refine ⟨max N 1,le_max_right _ _,?_⟩
  63   intro n hn
  64   have hh : (N : ℝ)≤n := by exact_mod_cast (le_max_left _ _).trans hn
  65   have hp : 0<δ*ω*ξ := by positivity
  66   have h := (div_le_iff₀ hp).mp (hN.trans hh)
  67   nlinarith
