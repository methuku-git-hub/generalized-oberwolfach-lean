  10 theorem capacity_mass_bounds {X : Type*} [Fintype X]
  11     (d : X → ℝ) (n M d₀ d₁ : ℝ) (hn : (Fintype.card X : ℝ)=n)
  12     (hm : ∑ x, d x=M) (hd : ∀ x, d₀*n ≤ d x ∧ d x ≤ d₁*n) :
  13     d₀*n^2 ≤ M ∧ M ≤ d₁*n^2 := by
  14   have hlo := Finset.sum_le_sum (fun x (_ : x ∈ Finset.univ) => (hd x).1)
  15   have hup := Finset.sum_le_sum (fun x (_ : x ∈ Finset.univ) => (hd x).2)
  16   simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul,hn,hm] at hlo hup
  17   constructor <;> nlinarith

  19 theorem capacity_profile_scale {X : Type*} [Fintype X]
  20     (d : X → ℝ) (n M d₀ d₁ : ℝ) (hn : (Fintype.card X : ℝ)=n)
  21     (hnpos : 0 < n) (hd₀ : 0 < d₀) (hd₁ : 0 < d₁)
  22     (hm : ∑ x, d x=M) (hd : ∀ x, d₀*n ≤ d x ∧ d x ≤ d₁*n) (x : X) :
  23     (d₀/d₁)/n ≤ d x/M ∧ d x/M ≤ (d₁/d₀)/n := by
  24   obtain ⟨hMlo,hMup⟩ := capacity_mass_bounds d n M d₀ d₁ hn hm hd
  25   have hM : 0 < M := (mul_pos hd₀ (sq_pos_of_pos hnpos)).trans_le hMlo
  26   have hlo := div_le_div₀ ((mul_pos hd₀ hnpos).le.trans (hd x).1) (hd x).1 hM hMup
  27   have hup := div_le_div₀ (mul_pos hd₁ hnpos).le (hd x).2
  28     (mul_pos hd₀ (sq_pos_of_pos hnpos)) hMlo
  29   constructor
  30   · convert hlo using 1; field_simp [hd₁.ne',hnpos.ne']
  31   · convert hup using 1; field_simp [hd₀.ne',hnpos.ne']

  33 theorem path_probability_bounds {X : Type*}
  34     (R : ℕ → X → X → ℝ) (i k : ℕ) (p : Fin (k+1) → X)
  35     (a b : ℝ) (ha : 0 ≤ a)
  36     (hR : ∀ j : Fin k, a ≤ R (i+j.val) (p j.castSucc) (p j.succ) ∧
  37       R (i+j.val) (p j.castSucc) (p j.succ) ≤ b) :
  38     a^k ≤ transitionPathWeight R i k p ∧ transitionPathWeight R i k p ≤ b^k := by
  39   rw [transitionPathWeight_product]
  40   constructor

  42       Finset.prod_le_prod₀ (s := Finset.univ) (fun _ _ => ha) (fun j _ => (hR j).1)
