  12 theorem marked_present_occurrences_physical_injective
  13     [Fintype A] (src dst:A→V) (jr:A→Bool)
  14     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  15     (ha:∀ a,markedArcPresent tail head ktail khead (jr a) (src a) (dst a))
  16     (hi:Function.Injective (fun a=>(src a,dst a))):
  17     Function.Injective (fun a=>if jr a then ({src a,π (dst a)}:Finset V) else {src a,dst a}):=by
  18   intro a a' he
  19   obtain ⟨e,hp₁,ht₁,hh₁⟩:=marked_root_rim_occurrence (π:=π) src dst jr ha a
  20   obtain ⟨e',hp₂,ht₂,hh₂⟩:=marked_root_rim_occurrence (π:=π) src dst jr ha a'
  21   have hee:e=e':=hp (hp₁.trans (he.trans hp₂.symm))
  22   apply hi
  23   apply Prod.ext
  24   · exact ht₁.symm.trans ((congrArg (Sum.elim tail ktail) hee).trans ht₂)
  25   · exact hh₁.symm.trans ((congrArg (Sum.elim head khead) hee).trans hh₂)
