  13 theorem finRotate_invariant_constant {k : ℕ} {T : Type*}
  14     (f : Fin (k + 2) → T) (h : ∀ i, f (finRotate (k + 2) i) = f i)
  15     (i : Fin (k + 2)) : f i = f 0 := by
  16   induction i using Fin.induction with
  17   | zero => rfl
  18   | succ i ih =>
  19     have hi := h i.castSucc
  20     rw [finRotate_apply, Fin.coeSucc_eq_succ] at hi
  21     exact hi.trans ih

  23 def cyclicInterfaceDegree {I X Y : Type*} [Fintype Y]
  24     (L : I → X → Y → Prop) (i : I) (x : X) : ℝ :=
  25   (Finset.univ.filter (L i x)).card

  27 theorem cyclic_interface_mass {k : ℕ} {X Y : Type*} [Fintype X] [Fintype Y]
  28     (L : Fin (k + 2) → X → Y → Prop) (R : Fin (k + 2) → Y → X → Prop)
  29     (hY : ∀ i y, (Finset.univ.filter (fun x => L i x y)).card =
  30       (Finset.univ.filter (R i y)).card)
  31     (hX : ∀ i z, (Finset.univ.filter (fun y => R i y z)).card =
  32       (Finset.univ.filter (L (finRotate (k + 2) i) z)).card)
  33     (i : Fin (k + 2)) :
  34     (∑ x, cyclicInterfaceDegree L i x) = ∑ x, cyclicInterfaceDegree L 0 x := by
  35   refine finRotate_invariant_constant (fun j => ∑ x, cyclicInterfaceDegree L j x) ?_ i
  36   intro j
  37   symm
  38   calc
  39     (∑ x, cyclicInterfaceDegree L j x) =
  40         ∑ x, ∑ y, interfaceCapacity (L j) x y := by
  41       simp only [interfaceCapacity_row, cyclicInterfaceDegree]
  42     _ = ∑ y, ∑ x, interfaceCapacity (L j) x y := Finset.sum_comm
  43     _ = ∑ y, ∑ z, interfaceCapacity (R j) y z := by
  44       simp_rw [interfaceCapacity_column, interfaceCapacity_row, hY]
  45     _ = ∑ z, ∑ y, interfaceCapacity (R j) y z := Finset.sum_comm
  46     _ = ∑ z, cyclicInterfaceDegree L (finRotate (k + 2) j) z := by
  47       simp only [interfaceCapacity_column, hX, cyclicInterfaceDegree]

  49 theorem cyclic_interface_marginals {k : ℕ} {X Y : Type*}
  50     [Fintype X] [Fintype Y] [Nonempty X]
  51     (L : Fin (k + 2) → X → Y → Prop) (R : Fin (k + 2) → Y → X → Prop)
  52     (δ : ℝ) (hn : 0 < Fintype.card Y)
  53     (hthreshold : (k + 2 : ℝ) * Fintype.card Y ≤ (2 * (k + 2) - 1) * δ)
  54     (hdL : ∀ i x, δ ≤ (Finset.univ.filter (L i x)).card)
  55     (hY : ∀ i y, (Finset.univ.filter (fun x => L i x y)).card =
  56       (Finset.univ.filter (R i y)).card)
  57     (hX : ∀ i z, (Finset.univ.filter (fun y => R i y z)).card =
  58       (Finset.univ.filter (L (finRotate (k + 2) i) z)).card) :
  59     let m := fun i x => cyclicInterfaceDegree L i x / (k + 2 : ℝ)
  60     let B := ∑ x, m 0 x
  61     0 < B ∧ (∀ i x, 0 < m i x) ∧ (∀ i, ∑ x, m i x = B) := by
  62   dsimp only
  63   have hk : (0 : ℝ) < k + 2 := by positivity
  64   have hn' : (0 : ℝ) < Fintype.card Y := by exact_mod_cast hn
  65   have hδ : 0 < δ := by
  66     have hprod := mul_pos hk hn'
  67     have hfac : (0 : ℝ) < 2 * (k + 2) - 1 := by linarith [Nat.cast_nonneg (α := ℝ) k]
  68     exact pos_of_mul_pos_right (lt_of_lt_of_le hprod hthreshold) hfac.le
  69   have hm (i : Fin (k + 2)) (x : X) :
  70       0 < cyclicInterfaceDegree L i x / (k + 2 : ℝ) :=
  71     div_pos (lt_of_lt_of_le hδ (hdL i x)) hk
  72   refine ⟨Finset.sum_pos (fun x _ => hm 0 x) Finset.univ_nonempty, hm, ?_⟩
  73   intro i
  74   rw [← Finset.sum_div, ← Finset.sum_div, cyclic_interface_mass L R hY hX i]
