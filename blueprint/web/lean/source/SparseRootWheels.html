  16 theorem ordinary_root_wheel_from_common_counts_within (Q F : SimpleGraph V) (H : V→U→Prop)
  17     (A W Ω : Finset V) (hAΩ : A⊆Ω) (x v w : V) (hvΩ : v∈Ω) (hwΩ : w∈Ω) (u:U) (hvw : v≠w)
  18     (hxv : Q.Adj x v) (hxw : Q.Adj x w)
  19     (hFroot : ∀ a∈({x,v,w}:Finset V),∀ b∈({x,v,w}:Finset V),¬F.Adj a b)
  20     (hHroot : H x u∧H v u∧H w u) (hHA : ∀ a∈A,H a u)
  21     (r : ℕ) (hr : 4≤r) (L b : ℝ) (hL : 0≤L) (hF : ∀ a,(F.degree a:ℝ)≤L)
  22     (hcount : ∀ a∈Ω,∀ b'∈Ω,b≤((A∩Q.neighborFinset a∩Q.neighborFinset b').card:ℝ))
  23     (hbudget : W.card+((r:ℝ)+3)*(L+1)<b) :
  24     ∃ C : OrdinaryWheel Q F H r,ordinaryWheelHub C=u ∧
  25       (ordinaryWheelRim C).Adj x v ∧ (ordinaryWheelRim C).Adj x w ∧
  26       (∀ a∈(ordinaryWheelRim C).support,a∈({x,v,w}:Finset V) ∨ a∈A∧a∉W) := by
  27   let Z:Finset V:={x,v,w}
  28   have hz : (Z.card:ℝ)≤3:=by exact_mod_cast (Finset.card_le_three (a:=x) (b:=v) (c:=w))
  29   have hk : ((r-2:ℕ):ℝ)≤r:=by exact_mod_cast Nat.sub_le r 2
  30   have hbudget' : W.card+((Z.card:ℝ)+(r-2:ℕ))*(L+1)<b:=by
  31     have hh:=mul_le_mul_of_nonneg_right (show (Z.card:ℝ)+(r-2:ℕ)≤(r:ℝ)+3 by linarith)
  32       (show 0≤L+1 by linarith)
  33     linarith only [hh,hbudget]
  34   obtain ⟨p,hp,hlen,hclass,hind,hroot⟩:=sparse_fixed_length_forbidden_path_within Q F A W Z Ω hAΩ v w hvΩ hwΩ hvw
  35     (by simp [Z]) (by simp [Z]) hFroot (r-2) (by omega) L b hL hF hcount hbudget'
  36   have hxp : x∉p.support:=by
  37     intro hx
  38     rcases hclass x hx with he|he|hc
  39     · exact hxv.ne he
  40     · exact hxw.ne he
  41     · exact hc.2.2 (by simp [Z])
  42   let q : Q.Walk x w:=p.cons hxv
  43   have hq : q.IsPath:=by
  44     exact SimpleGraph.Walk.IsPath.cons hp hxp
  45   have hqlen : q.length=r-1:=by simp only [q,SimpleGraph.Walk.length_cons,hlen];omega
  46   obtain ⟨C,hC,hCQ,hCxw,hCs,hsteps⟩:=embedded_cycle_from_path_with_support Q r (by omega) x w q hq hqlen hxw
  47   have hCxv : C.Adj x v:=by
  48     have hh:=hsteps 0 (by rw [hqlen];omega)
  49     simpa [q] using hh
  50   have hHs (a:V) (ha:a∈C.support) : H a u:=by
  51     have hh:= (hCs a).mp ha
  52     change a∈(p.cons hxv).support at hh
  53     rw [SimpleGraph.Walk.support_cons] at hh
  54     rcases List.mem_cons.mp hh with he|hp'
  55     · rw [he];exact hHroot.1
  56     · rcases hclass a hp' with he|he|hc
  57       · rw [he];exact hHroot.2.1
  58       · rw [he];exact hHroot.2.2
  59       · exact hHA a hc.1
  60   have hFs : ∀ a∈C.support,∀ b∈C.support,¬F.Adj a b:=by
  61     intro a ha b hb
  62     have ha':=(hCs a).mp ha
  63     have hb':=(hCs b).mp hb
  64     change a∈(p.cons hxv).support at ha'
  65     change b∈(p.cons hxv).support at hb'
  66     simp only [SimpleGraph.Walk.support_cons,List.mem_cons] at ha' hb'
  67     rcases ha' with he|ha' <;> rcases hb' with he'|hb'
  68     · rw [he,he'];exact F.loopless.irrefl x
  69     · rw [he];exact fun h=>hroot b hb' x (by simp [Z]) h.symm
  70     · rw [he'];exact hroot a ha' x (by simp [Z])
  71     · exact hind a ha' b hb'
  72   let wheel : OrdinaryWheel Q F H r:=⟨(⟨⟨C,hC⟩,hCQ⟩,u),hHs,hFs⟩
  73   refine ⟨wheel,rfl,hCxv,hCxw,?_⟩
  74   intro a ha
  75   have hh:=(hCs a).mp ha
  76   change a∈(p.cons hxv).support at hh
  77   simp only [SimpleGraph.Walk.support_cons,List.mem_cons] at hh
  78   rcases hh with he|hp'
  79   · exact Or.inl (by simp [he])
  80   · rcases hclass a hp' with he|he|hc
  81     · exact Or.inl (by simp [he])
  82     · exact Or.inl (by simp [he])
  83     · exact Or.inr ⟨hc.1,hc.2.1⟩
