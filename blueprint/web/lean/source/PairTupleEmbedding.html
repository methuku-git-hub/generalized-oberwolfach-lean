  13 def pairTupleEmbedding {I X V : Type*} (f g : I → X ↪ V) :
  14     (I → X × X) ↪ (I → V × V) where
  15   toFun s i := (f i (s i).1, g i (s i).2)
  16   inj' := by
  17     intro s t h
  18     funext i
  19     exact Prod.ext ((f i).injective (congrArg Prod.fst (congrFun h i)))
  20       ((g i).injective (congrArg Prod.snd (congrFun h i)))

  22 theorem pairTupleEmbedding_range {I X V : Type*} (f g : I → X ↪ V) (t : I → V × V) :
  23     t ∈ Set.range (pairTupleEmbedding f g) ↔
  24       ∀ i, (t i).1 ∈ Set.range (f i) ∧ (t i).2 ∈ Set.range (g i) := by
  25   constructor
  26   · rintro ⟨s,rfl⟩ i
  27     exact ⟨⟨(s i).1,rfl⟩,⟨(s i).2,rfl⟩⟩
  28   · intro h
  29     choose a ha using fun i => (h i).1
  30     choose b hb using fun i => (h i).2
  31     refine ⟨fun i => (a i,b i), ?_⟩
  32     funext i
  33     exact Prod.ext (ha i) (hb i)

  35 theorem pairTuple_weight_left_edge {I X V : Type*} [Fintype I] [DecidableEq I] [Fintype X] [DecidableEq X] [Fintype V]
  36     (f g : I → X ↪ V) (w : (I → X × X) → ℝ) (i : I) (x y : X) :
  37     (∑ t, if t i = (f i x,g i y) then embeddedWeight (pairTupleEmbedding f g) w t else 0) =
  38       ∑ s, if s i = (x,y) then w s else 0 := by
  39   have h := embeddedWeight_incidence (pairTupleEmbedding f g) w
  40     (fun t => t i = (f i x,g i y))
  41   convert h using 1
  42   · apply Finset.sum_congr rfl
  43     intro t _
  44     split_ifs <;> rfl
  45   · apply Finset.sum_congr rfl
  46     intro s _
  47     have he : pairTupleEmbedding f g s i = (f i x,g i y) ↔ s i = (x,y) := by
  48       change (f i (s i).1,g i (s i).2) = (f i x,g i y) ↔ _
  49       rw [Prod.mk.injEq, (f i).injective.eq_iff, (g i).injective.eq_iff]
  50       constructor
  51       · rintro ⟨h₁,h₂⟩; exact Prod.ext h₁ h₂
  52       · intro h; exact ⟨congrArg Prod.fst h, congrArg Prod.snd h⟩
  53     simp only [he]

  71 theorem pairTuple_weight_positions {I X V : Type*} [Fintype I] [DecidableEq I] [Fintype X] [DecidableEq X] [Fintype V]
  72     (f g : I → X ↪ V) (w : (I → X × X) → ℝ) (i : I) (x : X) :
  73     (∑ t, if (t i).1 = f i x then embeddedWeight (pairTupleEmbedding f g) w t else 0) =
  74       (∑ s, if (s i).1 = x then w s else 0) ∧
  75     (∑ t, if (t i).2 = g i x then embeddedWeight (pairTupleEmbedding f g) w t else 0) =
  76       (∑ s, if (s i).2 = x then w s else 0) := by
  77   constructor
  78   · have h := embeddedWeight_incidence (pairTupleEmbedding f g) w (fun t => (t i).1 = f i x)
  79     simpa only [pairTupleEmbedding, Function.Embedding.coeFn_mk, (f i).injective.eq_iff] using h
  80   · have h := embeddedWeight_incidence (pairTupleEmbedding f g) w (fun t => (t i).2 = g i x)
  81     simpa only [pairTupleEmbedding, Function.Embedding.coeFn_mk, (g i).injective.eq_iff] using h

  84     [Fintype I] [DecidableEq I] [Fintype X] [DecidableEq X] [Fintype V]
  85     (f g : I → X ↪ V) (w : (I → X × X) → ℝ) (i : I) (v : V)
