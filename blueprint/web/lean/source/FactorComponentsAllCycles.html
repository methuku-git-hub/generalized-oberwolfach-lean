   9 theorem two_factor_all_components_enumerated (F:SimpleGraph V)
  10     (hF:F.IsRegularOfDegree 2):
  11     ∃ (root:F.ConnectedComponent→V) (tail:F.ConnectedComponent→List V),
  12       (∀ c,(root c::tail c).Nodup ∧ 3≤(root c::tail c).length) ∧
  13       (∀ c x,x∈root c::tail c ↔ x∈c.supp) ∧
  14       (∀ c c' x,x∈root c::tail c→x∈root c'::tail c'→c=c') ∧
  15       (∀ x,∃ c,x∈root c::tail c) ∧
  16       (∀ x y,F.Adj x y ↔ ∃ c,x∈root c::tail c ∧
  17         s(x,y)∈prescribedOrderedPathEdges ((root c::tail c)++[root c])) := by
  18   choose root tail hnd hlen hvertices hedges using
  19     fun c : F.ConnectedComponent => two_factor_component_enumeration F hF c
  20   refine ⟨root,tail,?_,hvertices,?_,?_,?_⟩
  21   · intro c
  22     exact ⟨hnd c,hlen c⟩
  23   · intro c c' x hx hx'
  24     have ha:= (hvertices c x).1 hx
  25     have hb:= (hvertices c' x).1 hx'
  26     rw [SimpleGraph.ConnectedComponent.mem_supp_iff] at ha hb
  27     exact ha.symm.trans hb
  28   · intro x
  29     refine ⟨F.connectedComponentMk x, (hvertices _ _).2 ?_⟩
  30     rfl
  31   · intro x y

  33     · intro h
