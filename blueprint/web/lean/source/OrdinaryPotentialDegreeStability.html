  25 theorem ordinaryPotentialLoad_rim_degree_stability (x y : V) (hxy : x≠y) :
  26     |ordinaryPotentialLoad w (Sum.inl s(x,y)) Q' H'-ordinaryPotentialLoad w (Sum.inl s(x,y)) Q H|≤
  27       8*(r:ℝ)^4*B*d/Fintype.card V := by
  28   let Dq := fun x y=>¬(Q.Adj x y↔Q'.Adj x y)
  29   let Dh := fun x u=>¬(H x u↔H' x u)
  30   let n : ℝ := Fintype.card V
  31   let K : ℝ := (r:ℝ)^4*B
  32   let M := K/n
  33   let N := K/n^2
  34   have hn0 : 0<n := Nat.cast_pos.mpr hn
  35   have hM : 0≤M := by dsimp [M,K];positivity
  36   have hN : 0≤N := by dsimp [N,K];positivity
  37   have hQi (y : V) : (∑ x,bipAdj Dq x y)≤d := by
  38     have he : (∑ x,bipAdj Dq x y)=∑ x,bipAdj Dq y x := by
  39       apply Finset.sum_congr rfl
  40       intro x _
  41       simp only [bipAdj,Dq,Q.adj_comm,Q'.adj_comm]
  42       all_goals split_ifs <;> rfl
  43     rw [he];exact hQ y
  44   have hR := relation_root_incident_cost Dq d hQ hQi {x,y} {x,y} M N hM hN
  45   have hS := relation_root_incident_cost Dh d hHout hHin {x,y} ∅ M N hM hN
  46   have hc : ({x,y}:Finset V).card=2 := by simp [hxy]
  47   simp only [hc,Nat.cast_ofNat,Finset.card_empty,Nat.cast_zero,Finset.notMem_empty,or_false,add_zero] at hR hS
  48   have hstart := ordinaryPotentialLoad_stability_ordered w (fun W=>(hw W).1) (Sum.inl s(x,y)) Q Q' H H'
  49   apply hstart.trans
  50   calc
  51     _ ≤ (∑ z,∑ t,if Dq z t then (if z∈({x,y}:Finset V) ∨ t∈({x,y}:Finset V) then M else N) else 0)+
  52         (∑ z,∑ u,if Dh z u then (if z∈({x,y}:Finset V) then M else N) else 0) := by
  53       apply add_le_add
  54       · apply Finset.sum_le_sum
  55         intro z _
  56         apply Finset.sum_le_sum
  57         intro t _
  58         by_cases h : Dq z t
  59         · dsimp only [Dq,Dh] at h ⊢
  60           simp only [h,ite_true,ite_false]
  61           exact ordinaryOmittedPairLoad_rim_cost P F Hbase r hr hn hu w B hB hw x y z t hxy
  62         · dsimp only [Dq,Dh] at h ⊢
  63           simp only [h,ite_true,ite_false,le_refl]
  64       · apply Finset.sum_le_sum
  65         intro z _
  66         apply Finset.sum_le_sum
  67         intro u _
  68         by_cases h : Dh z u
  69         · dsimp only [Dq,Dh] at h ⊢
  70           simp only [h,ite_true,ite_false]
  71           exact ordinaryOmittedPairLoad_rim_spoke_cost P F Hbase r hr hn hu w B hB hw x y z u hxy
  72         · dsimp only [Dq,Dh] at h ⊢
  73           simp only [h,ite_true,ite_false,le_refl]
  74     _ ≤ d*((2+2)*M+n*N)+d*(2*M+n*N) := by
  75       convert (add_le_add hR hS) using 1
  76       all_goals congr 1 <;> (apply Finset.sum_congr rfl; intro a _; apply Finset.sum_congr rfl; intro b _; split_ifs <;> rfl)
