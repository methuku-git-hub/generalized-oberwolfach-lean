  25 theorem marked_supported_sets_disjoint_of_free
  26     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  27     (C D:MarkedSupportedEdgeSet tail head ktail khead π U) (Z:Finset V) (B:Finset U)
  28     (hrows:Disjoint (C.rows\Z) (D.rows\Z)) (hhubs:Disjoint (C.hubs\B) (D.hubs\B))
  29     (hC:∀ e∈C.edges,¬MarkedOldResource tail head ktail khead π Z B e):Disjoint C.edges D.edges:=by
  30   have hr:∀ v∈C.rows,v∈D.rows→v∈Z:=by
  31     intro v hc hd;by_contra hz
  32     exact Finset.disjoint_left.mp hrows (Finset.mem_sdiff.mpr ⟨hc,hz⟩) (Finset.mem_sdiff.mpr ⟨hd,hz⟩)
  33   have hh:∀ u∈C.hubs,u∈D.hubs→u∈B:=by
  34     intro u hc hd;by_contra hb
  35     exact Finset.disjoint_left.mp hhubs (Finset.mem_sdiff.mpr ⟨hc,hb⟩) (Finset.mem_sdiff.mpr ⟨hd,hb⟩)
  36   apply Finset.disjoint_left.mpr
  37   intro e hc hd
  38   apply hC e hc
  39   cases e with
  40   | inl e=>
  41     obtain ⟨a,ha,hat,hah⟩:=C.supported _ hc
  42     obtain ⟨a',ha',hat',hah'⟩:=D.supported _ hd
  43     have he:a=a':=hp (ha.trans ha'.symm)
  44     subst a'
  45     exact ⟨a,ha,hr _ hat hat',hr _ hah hah'⟩
  46   | inr vu=>exact ⟨hr _ (C.supported _ hc).1 (D.supported _ hd).1,hh _ (C.supported _ hc).2 (D.supported _ hd).2⟩
