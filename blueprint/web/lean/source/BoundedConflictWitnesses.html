  14 theorem bounded_monochromatic_conflict_witnesses
  15     {V C : Type*} [Fintype V] (G : SimpleGraph V) (D : Finset V)
  16     (σ : V → C×Bool) (c : C) {k : ℕ}
  17     (hlocal : ∀ (w : V) (S : Finset V),
  18       (∀ v∈S,G.Adj w v) →
  19       (∀ u∈S,∀ v∈S,u≠v → ¬G.Adj u v) → S.card≤k)
  20     (hcolour : ∀ v∈D,(σ v).1=c)
  21     (hbad : ∀ v∈D,¬coinColourSurvives G σ v) :
  22     ∃ f : V → V,
  23       (∀ v∈D,(σ (f v)).1=c ∧
  24         ((f v=v ∧ (σ v).2≠true) ∨ G.Adj v (f v))) ∧
  25       (∀ w,(D.filter (fun v=>f v=w)).card≤k+2) := by
  26   classical
  27   obtain ⟨m,hm,hind⟩ := finite_maximal_pairing G D
  28   have hconf (v : V) (hv : v∈D) (hcoin : (σ v).2=true) :
  29       ∃ u,G.Adj v u ∧ (σ u).1=(σ v).1 := by
  30     by_contra hn
  31     apply hbad v hv
  32     refine ⟨hcoin,?_⟩
  33     intro u hu he
  34     exact hn ⟨u,hu,he⟩
  35   let f : V → V := fun v=>match m v with
  36     | some u=>u
  37     | none=>if h : v∈D ∧ (σ v).2=true then
  38         Classical.choose (hconf v h.1 h.2) else v
  39   have hsome (v u : V) (h : m v=some u) : f v=u := by simp [f,h]
  40   have hspec (v : V) (hv : v∈D) :
  41       (σ (f v)).1=c ∧ ((f v=v ∧ (σ v).2≠true) ∨ G.Adj v (f v)) := by
  42     cases hmv : m v with
  43     | some u =>
  44       rw [hsome v u hmv]
  45       exact ⟨hcolour u (hm v u hmv).2.1,Or.inr (hm v u hmv).2.2.1⟩
  46     | none =>
  47       by_cases hc : (σ v).2=true
  48       · have h : v∈D ∧ (σ v).2=true := ⟨hv,hc⟩
  49         have he : f v=Classical.choose (hconf v h.1 h.2) := by simp [f,hmv,h]
  50         rw [he]
  51         have hw := Classical.choose_spec (hconf v h.1 h.2)
  52         exact ⟨hw.2.trans (hcolour v hv),Or.inr hw.1⟩
  53       · have he : f v=v := by simp [f,hmv,hc]
  54         rw [he]
  55         exact ⟨hcolour v hv,Or.inl ⟨rfl,hc⟩⟩
  56   refine ⟨f,hspec,?_⟩
  57   intro w
  58   let F := D.filter (fun v=>f v=w)
  59   let P := F.filter (fun v=>m v≠none)
  60   let U := F.filter (fun v=>m v=none ∧ v≠w)
  61   have hP : P.card≤1 := by
  62     apply Finset.card_le_one.mpr
  63     intro u hu v hv
  64     have huF := Finset.mem_filter.mp hu
  65     have hvF := Finset.mem_filter.mp hv
  66     have huf : f u=w := (Finset.mem_filter.mp huF.1).2
  67     have hvf : f v=w := (Finset.mem_filter.mp hvF.1).2
  68     have hum : m u=some w := by
  69       cases hmu : m u with
  70       | none => exact False.elim (huF.2 hmu)
  71       | some z =>
  72         have hzw : z=w := (hsome u z hmu).symm.trans huf
  73         simpa only [hzw] using hmu
  74     have hvm : m v=some w := by
  75       cases hmv : m v with
  76       | none => exact False.elim (hvF.2 hmv)
  77       | some z =>
  78         have hzw : z=w := (hsome v z hmv).symm.trans hvf
  79         simpa only [hzw] using hmv
  80     exact Option.some.inj ((hm u w hum).2.2.2.symm.trans (hm v w hvm).2.2.2)
  81   have hU : U.card≤k := by
  82     apply hlocal w U
  83     · intro v hv
  84       obtain ⟨hvF,hmv,hvw⟩ := Finset.mem_filter.mp hv
  85       obtain ⟨hvD,hfv⟩ := Finset.mem_filter.mp hvF
  86       rcases (hspec v hvD).2 with hself|hadj
  87       · exact False.elim (hvw (hself.1.symm.trans hfv))
  88       · exact (hfv ▸ hadj).symm
  89     · intro u hu v hv _hne
  90       obtain ⟨huF,hmu,_⟩ := Finset.mem_filter.mp hu
  91       obtain ⟨hvF,hmv,_⟩ := Finset.mem_filter.mp hv
  92       exact hind u (Finset.mem_filter.mp huF).1 v (Finset.mem_filter.mp hvF).1 hmu hmv
  93   have hcover : F⊆(P∪U)∪{w} := by
  94     intro v hv
  95     by_cases hmnone : m v=none
  96     · by_cases he : v=w
  97       · exact Finset.mem_union_right _ (Finset.mem_singleton.mpr he)
  98       · exact Finset.mem_union_left _ (Finset.mem_union_right _
  99           (Finset.mem_filter.mpr ⟨hv,hmnone,he⟩))
 100     · exact Finset.mem_union_left _ (Finset.mem_union_left _
 101         (Finset.mem_filter.mpr ⟨hv,hmnone⟩))
 102   have hcard : F.card≤P.card+U.card+1 :=
 103     (Finset.card_le_card hcover).trans
 104       ((Finset.card_union_le _ _).trans (by
 105         simpa only [Finset.card_singleton] using
 106           Nat.add_le_add_right (Finset.card_union_le P U) 1))
 107   change F.card≤_
 108   omega
