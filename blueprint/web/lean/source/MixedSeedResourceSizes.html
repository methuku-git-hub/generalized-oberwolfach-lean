  16 theorem typed_marked_seed_resource_size
  17     (W:MarkedWheel tail head ktail khead m π b H₀ Hₖ):
  18     2≤(typedMarkedOccurrenceResources W).card ∧
  19       (typedMarkedOccurrenceResources W).card≤2*(m+1) := by
  20   refine ⟨?_,typedMarkedOccurrenceResources_card W⟩
  21   let a:(E⊕K)⊕(Bool×V×U):=Sum.inl (markedCycleTypedEdge tail head ktail khead m π b W.1 0)
  22   let c:(E⊕K)⊕(Bool×V×U):=Sum.inr (true,W.1.val 0,W.2.val)
  23   have ha:a∈typedMarkedOccurrenceResources W:=by
  24     apply (typedMarkedOccurrenceResources_rim W _).mpr
  25     exact ⟨0,rfl⟩
  26   have hc:c∈typedMarkedOccurrenceResources W:=by
  27     apply (typedMarkedOccurrenceResources_spoke W _).mpr
  28     exact ⟨0,by simp [c]⟩
  29   have hne:a≠c:=by simp [a,c]
  30   have hsub:({a,c}:Finset ((E⊕K)⊕(Bool×V×U)))⊆typedMarkedOccurrenceResources W:=by
  31     intro z hz
  32     simp only [Finset.mem_insert,Finset.mem_singleton] at hz
  33     rcases hz with rfl|rfl <;> assumption
  34   have hcard:=Finset.card_le_card hsub
