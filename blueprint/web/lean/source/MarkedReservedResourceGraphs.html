  14 def markedReservedSpokePairs (C:Finset (Finset V⊕(V×U))):Finset (V×U):=
  15   Finset.univ.filter (fun z=>Sum.inr z∈C)

  17 theorem marked_reserved_spoke_pairs_card (C:Finset (Finset V⊕(V×U))):
  18     (markedReservedSpokePairs C).card≤C.card :=by
  19   apply Finset.card_le_card_of_injOn Sum.inr
  20   · intro z hz
  21     exact (Finset.mem_filter.mp hz).2
  22   · intro a _ b _ h
  23     exact Sum.inr.inj h

  25 theorem marked_reserved_resource_graphs (C:Finset (Finset V⊕(V×U))):
  26     let R:=physicalEdgeGraph (markedReservedRimPairs C)
  27     let S:=markedFiniteSpokeGraph (markedReservedSpokePairs C);
  28     (∀ x y,R.Adj x y ↔ x≠y∧Sum.inl ({x,y}:Finset V)∈C)∧
  29     (∀ v u,S.Adj (Sum.inl v) (Sum.inr u) ↔ Sum.inr (v,u)∈C)∧
  30     (∀ x,R.degree x≤C.card)∧(∀ z,S.degree z≤C.card):=by
  31   intro R S
  32   refine ⟨?_,?_,?_,?_⟩
  33   · intro x y
  34     change x≠y∧({x,y}:Finset V)∈markedReservedRimPairs C ↔ _
  35     rw [marked_reserved_rim_pairs_membership]
  36   · intro v u
  37     rw [markedFiniteSpokeGraph_adj]
  38     simp [markedReservedSpokePairs]
  39   · intro x
  40     exact (physicalEdgeGraph_degree_le_incident _ x).trans
  41       ((Finset.card_filter_le _ _).trans (marked_reserved_rim_pairs_card C))
  42   · intro z
  43     obtain ⟨hrow,hcol⟩:=finite_spoke_graph_degree_counts (markedReservedSpokePairs C)
  44     cases z with
  45     | inl x=>exact (hrow x).trans ((Finset.card_filter_le _ _).trans (marked_reserved_spoke_pairs_card C))
  46     | inr u=>exact (hcol u).trans ((Finset.card_filter_le _ _).trans (marked_reserved_spoke_pairs_card C))

  48 /-- Resources internal to a protected pair avoid the reserved-resource graphs
  49 when the reserved set itself avoids that pair. -/
  50 theorem marked_reserved_graphs_avoid_internal_resources
  51     (C:Finset (Finset V⊕(V×U))) (I:Finset V) (J:Finset U)
  52     (hC:∀ e∈C,¬markedInnerResource I J e):
  53     let R:=physicalEdgeGraph (markedReservedRimPairs C)
  54     let S:=markedFiniteSpokeGraph (markedReservedSpokePairs C);
  55     (∀ x∈I,∀ y∈I,¬R.Adj x y) ∧
  56     (∀ v∈I,∀ u∈J,¬S.Adj (Sum.inl v) (Sum.inr u)):=by
  57   intro R S
  58   have hg:=marked_reserved_resource_graphs C
  59   constructor
  60   · intro x hx y hy hxy
  61     have he:Sum.inl ({x,y}:Finset V)∈C:=(hg.1 x y).mp hxy |>.2
  62     apply hC _ he
  63     change ({x,y}:Finset V)⊆I
  64     simpa only [Finset.insert_subset_iff,Finset.singleton_subset_iff] using And.intro hx hy
  65   · intro v hv u hu huv
  66     have he:Sum.inr (v,u)∈C:=(hg.2.1 v u).mp huv
  67     exact hC _ he ⟨hv,hu⟩

  69 /-- In particular, prescribed typed arcs and spokes inside the pair remain
  70 available after the reserved-resource graphs are removed. -/
  71 theorem marked_reserved_graphs_preserve_prescribed_resources
  72     {E K:Type} (tail head:E→V) (ktail khead:K→V) (π:Equiv.Perm V)
  73     (C:Finset (Finset V⊕(V×U))) (I:Finset V) (J:Finset U)
  74     (hI:∀ x,π x∈I↔x∈I) (hC:∀ e∈C,¬markedInnerResource I J e):
  75     let R:=physicalEdgeGraph (markedReservedRimPairs C)
  76     let S:=markedFiniteSpokeGraph (markedReservedSpokePairs C);
  77     (∀ a:E⊕K,Sum.elim tail ktail a∈I→Sum.elim head khead a∈I→
  78       Sum.elim (fun e=>s(tail e,head e)) (fun e=>s(ktail e,π (khead e))) a∉R.edgeFinset)∧
  79     (∀ v∈I,∀ u∈J,¬S.Adj (Sum.inl v) (Sum.inr u)):=by
  80   intro R S
  81   obtain ⟨hR,hS⟩:=marked_reserved_graphs_avoid_internal_resources C I J hC
  82   refine ⟨?_,hS⟩
  83   intro a ha hb
  84   cases a with
  85   | inl e=>
  86     simpa only [Sum.elim_inl,SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using hR (tail e) ha (head e) hb
  87   | inr e=>
  88     simpa only [Sum.elim_inr,SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using hR (ktail e) ha (π (khead e)) ((hI _).mpr hb)

  90 /-- A packing avoiding the two reserved graphs avoids every reserved physical
  91 resource, including its marked rim and spoke types. -/
  92 theorem marked_packing_avoiding_reserved_graphs
  93     {E K:Type} {tail head:E→V} {ktail khead:K→V} {m b:ℕ}
  94     {π:Equiv.Perm V} {H₀ HK:V→U→Prop} (hm:1≤m) (hb:1≤b)
  95     (C:Finset (Finset V⊕(V×U)))
  96     (M:Finset (MarkedWheel tail head ktail khead m π b H₀ HK))
  97     (hR:Disjoint (markedPackingRimGraph M) (physicalEdgeGraph (markedReservedRimPairs C)))
  98     (hS:∀ x u,markedPackingSpokes M x u→¬(markedFiniteSpokeGraph (markedReservedSpokePairs C)).Adj (Sum.inl x) (Sum.inr u)):
  99     Disjoint C (M.biUnion markedWheelPhysicalResources):=by
 100   apply Finset.disjoint_left.mpr
 101   intro e hC hM
 102   cases e with
 103   | inr z=>
 104     have hh:=(marked_reserved_resource_graphs C).2.1 z.1 z.2
 105     exact hS z.1 z.2 ((markedPackingResources_spoke M z.1 z.2).mp hM) (hh.mpr hC)
 106   | inl e=>
 107     have he:=(markedPackingResources_rim M e).mp hM
 108     obtain ⟨W,hW,hWe⟩:=Finset.mem_biUnion.mp he
 109     have hcard:e.card=2:=by
 110       obtain ⟨i,_,hi⟩:=Finset.mem_image.mp hWe
 111       rw [←hi]
 112       exact markedCycle_physical_edge_card tail head ktail khead m π b hm hb W.1 i
 113     obtain ⟨x,y,hxy,rfl⟩:=Finset.card_eq_two.mp hcard
 114     have hMR:(markedPackingRimGraph M).Adj x y:=⟨hxy,Finset.mem_biUnion.mpr ⟨W,hW,hWe⟩⟩
 115     have hCR:(physicalEdgeGraph (markedReservedRimPairs C)).Adj x y:=
 116       ((marked_reserved_resource_graphs C).1 x y).mpr ⟨hxy,hC⟩
 117     exact disjoint_graphs_no_common_edge _ _ hR x y hMR hCR
