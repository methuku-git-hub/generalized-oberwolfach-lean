  14 /-- A positive demand deficit forces a cut into the range of the
  15 large-cut hypothesis. Only upper bounds on individual demands are needed
  16 for this implication. -/
  17 theorem normalizedDemand_cut
  18     {δ δ₀ M C s t Fₛ Fₜ e : ℝ}
  19     (hδ : 0 ≤ δ) (hδ₀ : δ ≤ δ₀) (hδsmall : δ₀ < 1 / 4)
  20     (hM : 0 ≤ M) (hMC : 2 * M ≤ C)
  21     (hs : 0 ≤ s) (ht : 0 ≤ t)
  22     (hFs : Fₛ ≤ M) (hFt : Fₜ ≤ M)
  23     (hFsbound : Fₛ ≤ (1 + δ) * M * s)
  24     (hFtbound : Fₜ ≤ (1 + δ) * M * t)
  25     (he : 0 ≤ e)
  26     (hlarge : 1 - 2 * δ₀ < s + t → C * min s t ≤ e) :
  27     Fₛ + Fₜ - M ≤ e := by
  28   by_cases hpos : 0 < Fₛ + Fₜ - M
  29   · have hMpos : 0 < M := by linarith
  30     have hlarge' : 1 - 2 * δ₀ < s + t := by
  31       by_contra h
  32       have hst : s + t ≤ 1 - 2 * δ₀ := le_of_not_gt h
  33       have hcoef : (1 + δ) * (1 - 2 * δ₀) ≤ 1 := by
  34         nlinarith [mul_nonneg hδ (le_trans hδ hδ₀)]
  35       have hscale := mul_le_mul_of_nonneg_left hst
  36         (mul_nonneg (by linarith : 0 ≤ 1 + δ) hM)
  37       have hscale' := mul_le_mul_of_nonneg_left hcoef hM
  38       nlinarith
  39     have hcoeff : (1 + δ) * M ≤ C := by nlinarith
  40     have hmin : Fₛ + Fₜ - M ≤ (1 + δ) * M * min s t := by
  41       rcases le_total s t with h | h
  42       · rw [min_eq_left h]; linarith
  43       · rw [min_eq_right h]; linarith
  44     have hbound := mul_le_mul_of_nonneg_right hcoeff (le_min hs ht)
  45     exact hmin.trans (hbound.trans (hlarge hlarge'))
  46   · exact (le_of_not_gt hpos).trans he

  48 /-- Cut inequalities under the manuscript's normalized cut hypothesis.
  49 The two part sizes are positive and the demands have the same fixed total.
  50 This is the integral conclusion used by the deletion-subgraph callers. -/
  51 theorem nearUniform_cutCondition
  52     {X Y E : Type*} [Fintype X] [Fintype Y] [Fintype E]
  53     [DecidableEq X] [DecidableEq Y]
  54     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ)
  55     {δ δ₀ M h : ℝ}
  56     (ha : 0 < Fintype.card X) (hb : 0 < Fintype.card Y)
  57     (hδ : 0 ≤ δ) (hδ₀ : δ ≤ δ₀) (hδsmall : δ₀ < 1 / 4)
  58     (hM : 0 ≤ M)
  59     (hMC : M ≤ h * Fintype.card X * Fintype.card Y / 2)
  60     (hftotal : ∑ x, (f x : ℝ) = M) (hgtotal : ∑ y, (g y : ℝ) = M)
  61     (hf : ∀ x, (f x : ℝ) ≤ (1 + δ) * M / Fintype.card X)
  62     (hg : ∀ y, (g y : ℝ) ≤ (1 + δ) * M / Fintype.card Y)
  63     (hlarge : ∀ (S : Finset X) (T : Finset Y),
  64       1 - 2 * δ₀ < (S.card : ℝ) / Fintype.card X + (T.card : ℝ) / Fintype.card Y →
  65       h * Fintype.card X * Fintype.card Y *
  66           min ((S.card : ℝ) / Fintype.card X) ((T.card : ℝ) / Fintype.card Y) ≤
  67         ((Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∈ T)).card : ℝ)) :
  68     ∀ (S : Finset X) (T : Finset Y),
  69       ∑ x ∈ S, f x ≤ (∑ y ∈ T, g y) +
  70         (Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∉ T)).card := by
  71   classical
  72   intro S T
  73   let U := Tᶜ
  74   have hFs : ∑ x ∈ S, (f x : ℝ) ≤ M := by
  75     rw [← hftotal]
  76     exact Finset.sum_le_univ_sum_of_nonneg (fun x => Nat.cast_nonneg (f x))
  77   have hFt : ∑ y ∈ U, (g y : ℝ) ≤ M := by
  78     rw [← hgtotal]
  79     exact Finset.sum_le_univ_sum_of_nonneg (fun y => Nat.cast_nonneg (g y))
  80   have hFsbound : ∑ x ∈ S, (f x : ℝ) ≤
  81       (1 + δ) * M * ((S.card : ℝ) / Fintype.card X) := by
  82     calc
  83       _ ≤ ∑ _ ∈ S, (1 + δ) * M / Fintype.card X :=
  84         Finset.sum_le_sum (fun x _ => hf x)
  85       _ = _ := by simp only [Finset.sum_const, nsmul_eq_mul]; ring
  86   have hFtbound : ∑ y ∈ U, (g y : ℝ) ≤
  87       (1 + δ) * M * ((U.card : ℝ) / Fintype.card Y) := by
  88     calc
  89       _ ≤ ∑ _ ∈ U, (1 + δ) * M / Fintype.card Y :=
  90         Finset.sum_le_sum (fun y _ => hg y)
  91       _ = _ := by simp only [Finset.sum_const, nsmul_eq_mul]; ring
  92   have hc := normalizedDemand_cut hδ hδ₀ hδsmall hM
  93     (show 2 * M ≤ h * Fintype.card X * Fintype.card Y by linarith)
  94     (div_nonneg (Nat.cast_nonneg S.card) (Nat.cast_pos.mpr ha).le)
  95     (div_nonneg (Nat.cast_nonneg U.card) (Nat.cast_pos.mpr hb).le)
  96     hFs hFt hFsbound hFtbound
  97     (Nat.cast_nonneg (Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∈ U)).card)
  98     (hlarge S U)
  99   have hsum : (∑ y ∈ T, (g y : ℝ)) + (∑ y ∈ U, (g y : ℝ)) = M := by
 100     rw [show U = Tᶜ from rfl, Finset.sum_add_sum_compl, hgtotal]
 101   have he : Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∈ U) =
 102       Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∉ T) := by
 103     ext e
 104     simp [U]
 105   rw [he] at hc
 106   have hreal : (∑ x ∈ S, (f x : ℝ)) ≤ (∑ y ∈ T, (g y : ℝ)) +
 107       ((Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∉ T)).card : ℝ) := by
 108     linarith
 109   exact_mod_cast hreal

 111 /-- The integral existence conclusion of the normalized-demand lemma. -/
 112 theorem bipartite_nearUniformDegrees
 113     {X Y E : Type*} [Fintype X] [Fintype Y] [Fintype E]
 114     [DecidableEq X] [DecidableEq Y]
 115     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ)
 116     {δ δ₀ M h : ℝ}
 117     (ha : 0 < Fintype.card X) (hb : 0 < Fintype.card Y)
 118     (hδ : 0 ≤ δ) (hδ₀ : δ ≤ δ₀) (hδsmall : δ₀ < 1 / 4)
 119     (hM : 0 ≤ M)
 120     (hMC : M ≤ h * Fintype.card X * Fintype.card Y / 2)
 121     (hftotal : ∑ x, (f x : ℝ) = M) (hgtotal : ∑ y, (g y : ℝ) = M)
 122     (hf : ∀ x, (f x : ℝ) ≤ (1 + δ) * M / Fintype.card X)
 123     (hg : ∀ y, (g y : ℝ) ≤ (1 + δ) * M / Fintype.card Y)
 124     (hlarge : ∀ (S : Finset X) (T : Finset Y),
 125       1 - 2 * δ₀ < (S.card : ℝ) / Fintype.card X + (T.card : ℝ) / Fintype.card Y →
 126       h * Fintype.card X * Fintype.card Y *
 127           min ((S.card : ℝ) / Fintype.card X) ((T.card : ℝ) / Fintype.card Y) ≤
 128         ((Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∈ T)).card : ℝ)) :
 129     ∃ H : Finset E,
 130       (∀ x, (H.filter (fun e => tail e = x)).card = f x) ∧
 131       (∀ y, (H.filter (fun e => head e = y)).card = g y) := by
 132   classical
 133   have htotal : ∑ x, f x = ∑ y, g y := by
 134     exact_mod_cast hftotal.trans hgtotal.symm
 135   exact bipartite_prescribedDegrees tail head f g htotal
 136     (nearUniform_cutCondition tail head f g ha hb hδ hδ₀ hδsmall hM hMC
 137       hftotal hgtotal hf hg hlarge)

 139 /-- The same normalized cut argument for a matrix of available edge
 140 multiplicities. This is the form used after deleting protected arcs. -/
 141 theorem nearUniformMultiplicity_cut
 142     {V : Type*} [Fintype V] [DecidableEq V]
 143     (L : V → V → ℕ) (f g : V → ℕ) {δ M h : ℝ}
 144     (hn : 0 < Fintype.card V) (hδ : 0 ≤ δ) (hδsmall : δ < 1 / 4)
 145     (hM : 0 ≤ M) (hMC : M ≤ h * Fintype.card V * Fintype.card V / 2)
 146     (hftotal : ∑ x, (f x : ℝ) = M) (hgtotal : ∑ y, (g y : ℝ) = M)
 147     (hf : ∀ x, (f x : ℝ) ≤ (1 + δ) * M / Fintype.card V)
 148     (hg : ∀ y, (g y : ℝ) ≤ (1 + δ) * M / Fintype.card V)
 149     (hlarge : ∀ (S T : Finset V),
 150       1 - 2 * δ < (S.card : ℝ) / Fintype.card V + (T.card : ℝ) / Fintype.card V →
 151       h * Fintype.card V * Fintype.card V *
 152           min ((S.card : ℝ) / Fintype.card V) ((T.card : ℝ) / Fintype.card V) ≤
 153         ((∑ x, ∑ y, if x ∈ S ∧ y ∈ T then L x y else 0 : ℕ) : ℝ)) :
 154     ∀ (S T : Finset V), ∑ x ∈ S, f x ≤ (∑ y ∈ T, g y) +
 155       ∑ x, ∑ y, if x ∈ S ∧ y ∉ T then L x y else 0 := by
 156   classical
 157   have hc := nearUniform_cutCondition
 158     (fun e : BipartiteEdges L => e.1) (fun e => e.2.1) f g
 159     hn hn hδ (le_refl δ) hδsmall hM hMC hftotal hgtotal hf hg
 160     (fun S T ht => by
 161       rw [bipartiteEdges_filter_card L (fun x y => x ∈ S ∧ y ∈ T)]
 162       exact hlarge S T ht)
 163   intro S T
 164   have hs := hc S T
 165   rw [bipartiteEdges_filter_card L (fun x y => x ∈ S ∧ y ∉ T)] at hs
 166   exact hs

 168 /-- Demands in `[q-e,q+e]` satisfy the normalized cut criterion with
 169 relative error `2e/(q-e)`. Here `e` is real, as is the error `5 n^(4/5)`
 170 in the protected-partition proof; it is not rounded or assumed integral. -/
 171 theorem intervalDemands_cut
 172     {V : Type*} [Fintype V] [DecidableEq V]
 173     (L : V → V → ℕ) (f g : V → ℕ) {q e δ h : ℝ}
 174     (hn : 0 < Fintype.card V) (_he : 0 ≤ e) (heq : e < q)
 175     (hδ : 0 ≤ δ) (hδsmall : δ < 1 / 4)
 176     (hmargin : 2 * e / (q - e) ≤ δ)
 177     (hcapacity : q + e ≤ h * Fintype.card V / 2)
 178     (htotal : ∑ x, f x = ∑ y, g y)
 179     (hf : ∀ x, q - e ≤ (f x : ℝ) ∧ (f x : ℝ) ≤ q + e)
 180     (hg : ∀ y, q - e ≤ (g y : ℝ) ∧ (g y : ℝ) ≤ q + e)
 181     (hlarge : ∀ (S T : Finset V),
 182       1 - 2 * δ < (S.card : ℝ) / Fintype.card V + (T.card : ℝ) / Fintype.card V →
 183       h * Fintype.card V * Fintype.card V *
 184           min ((S.card : ℝ) / Fintype.card V) ((T.card : ℝ) / Fintype.card V) ≤
 185         ((∑ x, ∑ y, if x ∈ S ∧ y ∈ T then L x y else 0 : ℕ) : ℝ)) :
 186     ∀ (S T : Finset V), ∑ x ∈ S, f x ≤ (∑ y ∈ T, g y) +
 187       ∑ x, ∑ y, if x ∈ S ∧ y ∉ T then L x y else 0 := by
 188   classical
 189   let M : ℝ := ∑ x, (f x : ℝ)
 190   have hn' : (0 : ℝ) < Fintype.card V := Nat.cast_pos.mpr hn
 191   have hM : 0 ≤ M := Finset.sum_nonneg (fun x _ => Nat.cast_nonneg (f x))
 192   have htotal' : ∑ y, (g y : ℝ) = M := by
 193     dsimp [M]
 194     exact_mod_cast htotal.symm
 195   have hMlow : Fintype.card V * (q - e) ≤ M := by
 196     calc
 197       _ = ∑ _ : V, (q - e) := by simp; ring
 198       _ ≤ _ := Finset.sum_le_sum (fun x _ => (hf x).1)
 199   have hMhigh : M ≤ Fintype.card V * (q + e) := by
 200     calc
 201       _ ≤ ∑ _ : V, (q + e) := Finset.sum_le_sum (fun x _ => (hf x).2)
 202       _ = _ := by simp; ring
 203   have hMC : M ≤ h * Fintype.card V * Fintype.card V / 2 := by
 204     have hs := mul_le_mul_of_nonneg_left hcapacity hn'.le
 205     nlinarith
 206   have hrelative : Fintype.card V * (q + e) ≤ (1 + δ) * M := by
 207     have hb := (div_le_iff₀ (sub_pos.mpr heq)).mp hmargin
 208     have hb' := mul_le_mul_of_nonneg_left hb hn'.le
 209     have hmδ := mul_le_mul_of_nonneg_left hMlow hδ
 210     nlinarith
 211   have hupper (z : ℝ) (hz : z ≤ q + e) :
 212       z ≤ (1 + δ) * M / Fintype.card V := by
 213     apply (le_div_iff₀ hn').mpr
 214     have hz' := mul_le_mul_of_nonneg_left hz hn'.le
 215     nlinarith
 216   exact nearUniformMultiplicity_cut L f g hn hδ hδsmall hM hMC
 217     rfl htotal' (fun x => hupper _ (hf x).2) (fun y => hupper _ (hg y).2) hlarge
