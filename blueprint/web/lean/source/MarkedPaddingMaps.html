  16 theorem padding_row_map_extension (m:ℕ) (φ:V→V) (p p':I→(Fin m↪V)) (Z:Finset V)
  17     (hdis:∀ i j,i≠j→Disjoint (Finset.univ.image (p i)) (Finset.univ.image (p j)))
  18     (hav:∀ i k,p i k∉Z):
  19     ∃ φ':V→V,(∀ v∈Z,φ' v=φ v)∧∀ i k,φ' (p i k)=p' i k:=by
  20   let f:I×Fin m→V:=fun z=>p z.1 z.2
  21   let g:I×Fin m→V:=fun z=>p' z.1 z.2
  22   have hf:Function.Injective f:=disjoint_paths_joint_injective m p hdis
  23   refine ⟨Function.extend f g φ,?_,fun i k=>hf.extend_apply g φ (i,k)⟩
  24   intro v hv
  25   apply Function.extend_apply'
  26   rintro ⟨⟨i,k⟩,he⟩
  27   have he':p i k=v:=he
  28   exact hav i k (he'.symm ▸ hv)

  40     (Z:Finset V) (hv:∀ i,v i∈Z)
  41     (hdis:∀ i j,i≠j→Disjoint (Finset.univ.image (p i)) (Finset.univ.image (p j)))
  42     (hav:∀ i k,p i k∉Z):

  44       φ' (Fin.cons (α:=fun _=>V) (v i) (p i) k)=Fin.cons (α:=fun _=>V) (φ (v i)) (p' i) k:=by
  45   obtain ⟨φ',hφ,hp⟩:=padding_row_map_extension m φ p p' Z hdis hav
