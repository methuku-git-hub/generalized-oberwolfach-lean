  18     ∀C∈M,∃x:{x // x∈Y},select x=C := by

  20   have hx:C.val 0∈Y:=(hmarked (C.val 0)).1 C.property.2.1

  25   have hroot:(select x).val 0=C.val 0:=by simpa [x] using (hselected x).2

  28     rw [hroot]
