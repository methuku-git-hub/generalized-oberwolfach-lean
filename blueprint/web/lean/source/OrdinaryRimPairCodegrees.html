  13 theorem uniform_wheel_pair_scale (n R B : ℝ) (hn : 1≤n) (hR : 1≤R) (hB : 0≤B)
  14     (s a : ℕ) (hs : s≤4) (ha : 1≤a) : R^s*B/n^a≤R^4*B/n := by
  15   have hn0 : 0<n := lt_of_lt_of_le zero_lt_one hn
  16   have hpow : R^s≤R^4 := pow_le_pow_right₀ hR hs
  17   have hnp : n≤n^a := by simpa only [pow_one] using pow_le_pow_right₀ hn ha
  18   calc
  19     _ ≤ R^4*B/n^a := div_le_div_of_nonneg_right (mul_le_mul_of_nonneg_right hpow hB) (by positivity)
  20     _ ≤ _ := div_le_div_of_nonneg_left (by positivity) hn0 hnp

  22 theorem distinct_pair_union_card {V : Type*} [DecidableEq V]
  23     (x y z t : V) (hxy : x≠y) (hzt : z≠t) (he : s(x,y)≠s(z,t)) :
  24     3≤({x,y,z,t}:Finset V).card ∧ ({x,y,z,t}:Finset V).card≤4 := by
  25   constructor
  26   · have hthree (v : V) (hxv : x≠v) (hyv : y≠v) (hv : v∈({z,t}:Finset V)) :
  27         3≤({x,y,z,t}:Finset V).card := by
  28       have hs : ({x,y,v}:Finset V)⊆{x,y,z,t} := by
  29         intro a ha
  30         simp only [Finset.mem_insert,Finset.mem_singleton] at ha hv ⊢
  31         rcases ha with rfl | rfl | rfl <;> tauto
  32       have hc : ({x,y,v}:Finset V).card=3 := by simp [hxy,hxv,hyv]
  33       rw [← hc]
  34       exact Finset.card_le_card hs
  35     by_cases hz : z=x ∨ z=y
  36     · by_cases ht : t=x ∨ t=y
  37       · exfalso
  38         rcases hz with rfl|rfl <;> rcases ht with rfl|rfl
  39         · exact hzt rfl
  40         · exact he rfl
  41         · exact he (Sym2.eq_swap)
  42         · exact hzt rfl
  43       · apply hthree t (by tauto) (by tauto) (by simp)
  44     · apply hthree z (by tauto) (by tauto) (by simp)
  45   · have h1 := Finset.card_insert_le x ({y,z,t}:Finset V)
  46     have h2 := Finset.card_insert_le y ({z,t}:Finset V)
  47     have h3 := Finset.card_insert_le z ({t}:Finset V)
  48     have h4 : ({t}:Finset V).card=1 := Finset.card_singleton _
  49     omega
