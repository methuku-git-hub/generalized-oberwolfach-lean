  11 theorem buffered_path_follows_permutation
  12     {X I:Type*} [Fintype I] [DecidableEq X]
  13     (n:I→ℕ) (inner:∀i,Fin (n i+1)↪X) (σ:Equiv.Perm X)
  14     (hstep:∀i (p:Fin (n i)),σ (inner i p.castSucc)=inner i p.succ):
  15     ∀i (p:Fin (n i+1)),inner i p=(σ^p.val) (inner i 0):=by
  16   intro i p
  17   induction p using Fin.induction with
  18   | zero => simp
  19   | succ p ih =>
  20     calc
  21       inner i p.succ = σ (inner i p.castSucc):=(hstep i p).symm
  22       _ = σ ((σ^p.val) (inner i 0)):=by rw [ih]; rfl
  23       _ = (σ^(p.val+1)) (inner i 0):=by rw [pow_succ',Equiv.Perm.mul_apply]
