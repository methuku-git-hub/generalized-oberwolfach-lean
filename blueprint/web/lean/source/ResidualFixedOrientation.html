  10 theorem two_factor_fixed_orientation_for_all_subgraphs (F:SimpleGraph V)
  11     (hF:F.IsRegularOfDegree 2):
  12     ∃ (D:V→V→Prop) (orig:List (PrescribedSourcePiece V)),
  13       orig.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset) ∧
  14       (∀p∈orig,∃a xs,p=.cycle a xs ∧ p.OriginalValid D) ∧
  15       sourcePieceVerticesUnion orig=Finset.univ ∧ sourcePieceEdgesUnion orig=F.edgeFinset ∧
  16       (∀x y,D x y ↔ (x,y)∈sourcePieceArcsUnion orig) ∧
  17       ∀ (H:SimpleGraph V) (R:Finset V),H≤F→(∀x,x∉R→∀y,¬H.Adj x y)→
  18         ∃ ps:List (PrescribedSourcePiece V),
  19           ps.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset) ∧
  20           (∀p∈ps,p.Shape ∧ p.OriginalValid D) ∧
  21           prescribedDependentVertices (fun _ : Unit=>ps) ()=R ∧
  22           prescribedDependentGraph (fun _ : Unit=>ps) ()=H ∧
  23           ∀x y,prescribedDependentDigraph (fun _ : Unit=>ps) () x y ↔ D x y ∧ H.Adj x y := by
  24   obtain ⟨D₀,orig,hpair,hcycles,hvertices,hedges,hD₀⟩:=two_factor_directed_source_cycles F hF
  25   let D:V→V→Prop:=fun x y=>(x,y)∈sourcePieceArcsUnion orig
  26   have hcycles':∀p∈orig,∃a xs,p=.cycle a xs ∧ p.OriginalValid D := by
  27     intro p hp
  28     obtain ⟨a,xs,rfl,hvalid⟩:=hcycles p hp
  29     refine ⟨a,xs,rfl,hvalid.1,hvalid.2.1,?_⟩
  30     apply (prescribed_directed_edges_chain D ((a::xs)++[a])).2
  31     intro e he
  32     exact Finset.mem_biUnion.mpr ⟨.cycle a xs,List.mem_toFinset.mpr hp,he⟩
  33   refine ⟨D,orig,hpair,hcycles',hvertices,hedges,by intro x y;rfl,?_⟩
  34   intro H R hHF hisol
  35   obtain ⟨ps,hps,hvalid,hv,he,ha⟩:=retained_cycle_family_source_pieces_with_arcs H R hisol D orig hpair hcycles'
  36   have hpsV:sourcePieceVerticesUnion ps=R := by
  37     ext x
  38     rw [hv x,hvertices]
  39     simp
  40   have hpsE:sourcePieceEdgesUnion ps=H.edgeFinset := by
  41     rw [he,hedges]
  42     ext e
  43     simp only [Finset.mem_filter]
  44     have hsub:H.edgeFinset⊆F.edgeFinset := SimpleGraph.edgeFinset_mono hHF
  45     exact ⟨And.right,fun hx=>⟨hsub hx,hx⟩⟩
  46   refine ⟨ps,hps,?_,?_,?_,?_⟩
  47   · intro p hp
  48     have hh:=hvalid p hp
  49     constructor
  50     · cases p with
  51       | path xs => exact hh.1
  52       | cycle a xs => exact ⟨hh.1,hh.2.1⟩
  53     · exact hh
  54   · rw [dependent_source_vertices_of_list,hpsV]
  55   · exact dependent_source_graph_of_list ps H hpsE
  56   · intro x y
  57     rw [dependent_source_arcs_of_list,ha,Finset.mem_filter]
  58 end Oberwolfach
