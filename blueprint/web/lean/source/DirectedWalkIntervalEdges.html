  11 theorem simple_directed_walk_interval_concatenation_with_edges
  12     {V:Type*} (G:SimpleGraph V) (v:ℕ → V) (p:∀i:ℕ,G.Walk (v i) (v (i+1)))
  13     (c n start:ℕ)
  14     (hp:∀i,start ≤ i → i < start+n → (p i).IsPath)
  15     (hlen:∀i,start ≤ i → i < start+n → (p i).length=c)
  16     (hdis:∀i j,start ≤ i → i < start+n → start ≤ j → j < start+n → i ≠ j → 
  17       ∀x ∈ (p i).support.dropLast,x ∉ (p j).support.dropLast)
  18     (hend:∀i,start ≤ i → i < start+n → v (start+n) ∉ (p i).support.dropLast)
  19     (R:V→V→Prop)
  20     (hR:∀ i, start ≤ i → i < start+n → ∀ k, k < (p i).length → R ((p i).getVert k) ((p i).getVert (k+1))):
  21     ∃q:G.Walk (v start) (v (start+n)),q.IsPath ∧ q.length=n*c ∧
  22       (∀x,x ∈ q.support ↔ x=v (start+n) ∨
  23         ∃i,start ≤ i ∧ i < start+n ∧ x ∈ (p i).support.dropLast) ∧
  24       (∀j,j≤n→q.getVert (c*j)=v (start+j)) ∧
  25       (∀e:Sym2 V,e∈q.edges ↔ ∃i,start ≤ i ∧ i < start+n ∧ e∈(p i).edges) ∧
  26       (∀ k, k < q.length → R (q.getVert k) (q.getVert (k+1))) := by
  27   induction n generalizing start with
  28   | zero =>
  29     refine ⟨SimpleGraph.Walk.nil,SimpleGraph.Walk.IsPath.nil,by simp,?_,?_,?_,?_⟩
  30     · intro x
  31       simp
  32       omega
  33     · intro j hj
  34       have :j=0:=by omega
  35       subst j
  36       simp
  37     · intro e
  38       simp
  39       omega
  40     · intro k hk
  41       simp at hk
  42   | succ n ih =>
  43     have heq:start+1+n=start+(n+1):=by omega
  44     have hp':∀i,start+1 ≤ i → i < start+1+n → (p i).IsPath:=by
  45       intro i hi hj;apply hp i  <;> omega
  46     have hl':∀i,start+1 ≤ i → i < start+1+n → (p i).length=c:=by
  47       intro i hi hj;apply hlen i  <;> omega
  48     have hd':∀i j,start+1 ≤ i → i < start+1+n → start+1 ≤ j → j < start+1+n → i ≠ j → 
  49         ∀x ∈ (p i).support.dropLast,x ∉ (p j).support.dropLast:=by
  50       intro i j hi hj hk hl hij x hx
  51       exact hdis i j (by omega) (by omega) (by omega) (by omega) hij x hx
  52     have he':∀i,start+1 ≤ i → i < start+1+n → v (start+1+n) ∉ (p i).support.dropLast:=by
  53       intro i hi hj
  54       rw [heq]
  55       exact hend i (by omega) (by omega)
  56     have hR':∀ i, start+1 ≤ i → i < start+1+n → ∀ k, k < (p i).length → R ((p i).getVert k) ((p i).getVert (k+1)):=by
  57       intro i hi hj k hk
  58       exact hR i (by omega) (by omega) k hk
  59     obtain ⟨q,hq,hqlen,hqmem,hqmark,hqedges,hqR⟩:=ih (start+1) hp' hl' hd' he' hR' 
  60     have hfirst:(p start).IsPath:=hp start (by omega) (by omega)
  61     have hfirstlen:(p start).length=c:=hlen start (by omega) (by omega)
  62     have hsep:∀x ∈ (p start).support.dropLast,x ∉ q.support:=by
  63       intro x hx hq'
  64       rcases (hqmem x).mp hq' with hterminal | ⟨i,hi,hupper,hxi⟩
  65       · have hh:=hend start (by omega) (by omega)
  66         rw [←heq,←hterminal] at hh
  67         exact hh hx
  68       · exact hdis start i (by omega) (by omega) (by omega) (by omega) (by omega) x hx hxi
  69     obtain ⟨hsimple,hsum,hsupport⟩:=simple_walk_append_of_disjoint_initial_support (p start) q hfirst hq hsep
  70     have hresult:∃w:G.Walk (v start) (v (start+1+n)),w.IsPath ∧ w.length=(n+1)*c ∧
  71       (∀x,x ∈ w.support ↔ x=v (start+1+n) ∨
  72         ∃i,start ≤ i ∧ i < start+1+n ∧ x ∈ (p i).support.dropLast) ∧
  73       (∀j,j≤n+1→w.getVert (c*j)=v (start+j)) ∧
  74       (∀e:Sym2 V,e∈w.edges ↔ ∃i,start ≤ i ∧ i < start+1+n ∧ e∈(p i).edges) ∧
  75       (∀ k, k < w.length → R (w.getVert k) (w.getVert (k+1))):=by
  76       refine ⟨(p start).append q,hsimple,?_,?_,?_,?_,?_⟩
  77       · calc
  78           ((p start).append q).length=(p start).length+q.length:=hsum
  79           _=c+n*c:=by rw [hfirstlen,hqlen]
  80           _=(n+1)*c:=by nlinarith
  81       · intro x
  82         rw [hsupport,List.mem_append,hqmem]
  83         constructor
  84         · rintro (hx|hx|⟨i,hi,hupper,hxi⟩)
  85           · right;exact ⟨start,by omega,by omega,hx⟩
  86           · exact Or.inl hx
  87           · right;exact ⟨i,by omega,by omega,hxi⟩
  88         · rintro (hx|⟨i,hi,hupper,hxi⟩)
  89           · exact Or.inr (Or.inl hx)
  90           · by_cases h:i=start
  91             · subst i;left;exact hxi
  92             · right;right;exact ⟨i,by omega,by omega,hxi⟩
  93       · intro j hj
  94         cases j with
  95         | zero => simp
  96         | succ j =>
  97           have hj':j≤n:=by omega
  98           have hk:c*(j+1)≥(p start).length:=by rw [hfirstlen];nlinarith
  99           rw [SimpleGraph.Walk.getVert_append]
 100           simp only [if_neg (show ¬c*(j+1)<(p start).length by omega)]
 101           have he:c*(j+1)-(p start).length=c*j:=by rw [hfirstlen,Nat.mul_add,Nat.mul_one,Nat.add_sub_cancel_right]
 102           rw [he,hqmark j hj']
 103           congr 1 <;> omega
 104       · intro e
 105         rw [SimpleGraph.Walk.edges_append,List.mem_append,hqedges]
 106         constructor
 107         · rintro (h|⟨i,hi,hj,he⟩)
 108           · exact ⟨start,by omega,by omega,h⟩
 109           · exact ⟨i,by omega,by omega,he⟩
 110         · rintro ⟨i,hi,hj,he⟩
 111           by_cases h:i=start
 112           · subst i;exact Or.inl he
 113           · exact Or.inr ⟨i,by omega,by omega,he⟩
 114       · exact walk_append_preserves_successor_relation (p start) q R (hR start (by omega) (by omega)) hqR
 115     rw [←heq]
 116     exact hresult
