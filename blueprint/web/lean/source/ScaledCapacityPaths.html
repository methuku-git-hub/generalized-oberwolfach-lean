  11 def conditionalPathLower (a₀ d₀ d₁ : ℝ) (k : ℕ) : ℝ :=
  12   (a₀/d₁)^k / ((11/10)*(d₁/d₀))
  13 def conditionalPathUpper (a₁ d₀ d₁ : ℝ) (k : ℕ) : ℝ :=
  14   (a₁/d₀)^k / ((9/10)*(d₀/d₁))
  15 def halfResidualLower (a₀ d₀ d₁ : ℝ) (k : ℕ) : ℝ :=
  16   (d₀/4)*(a₀/(2*d₁))^k
  17 def halfResidualUpper (a₁ d₀ d₁ : ℝ) (k : ℕ) : ℝ :=
  18   (d₁/2)*(2*a₁/d₀)^k

  36 theorem conditional_path_scale_bounds {X : Type*}
  37     (R : ℕ → X → X → ℝ) (i k : ℕ) (p : Fin (k+1) → X)
  38     (n a₀ a₁ d₀ d₁ T : ℝ) (hn : 0<n) (hk : 0<k)
  39     (ha₀ : 0<a₀) (ha₁ : 0<a₁) (hd₀ : 0<d₀) (hd₁ : 0<d₁)
  40     (hR : ∀ j : Fin k,
  41       a₀/(d₁*n) ≤ R (i+j.val) (p j.castSucc) (p j.succ) ∧
  42       R (i+j.val) (p j.castSucc) (p j.succ) ≤ a₁/(d₀*n))
  43     (hT : ((9/10)*(d₀/d₁))/n ≤ T ∧ T ≤ ((11/10)*(d₁/d₀))/n) :
  44     conditionalPathLower a₀ d₀ d₁ k/n^(k-1) ≤ transitionPathWeight R i k p/T ∧
  45       transitionPathWeight R i k p/T ≤ conditionalPathUpper a₁ d₀ d₁ k/n^(k-1) := by
  46   obtain ⟨hplo,hpup⟩ := path_probability_bounds R i k p _ _ (by positivity) hR
  47   have hTp : 0<T := (by positivity : 0<((9/10)*(d₀/d₁))/n).trans_le hT.1
  48   have hpp : 0 ≤ transitionPathWeight R i k p := (by positivity : 0 ≤ (a₀/(d₁*n))^k).trans hplo
  49   have hlo := div_le_div₀ hpp hplo hTp hT.2
  50   have hup := div_le_div₀ (by positivity : 0 ≤ (a₁/(d₀*n))^k) hpup
  51     (by positivity : 0<((9/10)*(d₀/d₁))/n) hT.1
  52   rw [normalized_path_scale a₀ d₁ _ n hn k hk] at hlo
  53   rw [normalized_path_scale a₁ d₀ _ n hn k hk] at hup
  54   exact ⟨hlo,hup⟩

  56 theorem half_residual_scale_bounds {X : Type*} [Fintype X] {k : ℕ} (hk : 0<k)
  57     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ)
  58     (ρ : (Fin (k+1) → X) → ℝ)
  59     (ha : ∀ j x y, 0 ≤ a j x y) (hd : ∀ j x, 0<d j x)
  60     (hrow : ∀ j x, ∑ y, a j x y=d j.castSucc x)
  61     (hcol : ∀ j y, ∑ x, a j x y=d j.succ y)
  62     (hρ : ∀ p, 0 ≤ ρ p)
  63     (hreserve : ∀ j x y, pathEdgeLoad ρ j x y ≤ (11/27)*a j x y)
  64     (p : Fin (k+1) → X) (n a₀ a₁ d₀ d₁ : ℝ)
  65     (hn : 0<n) (ha₀ : 0<a₀) (hd₀ : 0<d₀)
  66     (hap : ∀ j, a₀ ≤ a j (p j.castSucc) (p j.succ) ∧ a j (p j.castSucc) (p j.succ) ≤ a₁)
  67     (hdp : ∀ j, d₀*n ≤ d j (p j) ∧ d j (p j) ≤ d₁*n) :
  68     halfResidualLower a₀ d₀ d₁ k/n^(k-1) ≤ halfResidualPathWeight a d ρ p ∧
  69       halfResidualPathWeight a d ρ p ≤ halfResidualUpper a₁ d₀ d₁ k/n^(k-1) := by
  70   obtain ⟨hlo,hup⟩ := halfResidualPathWeight_bounds hk a d ρ ha hd hrow hcol hρ hreserve
  71     p a₀ a₁ (d₀*n) (d₁*n) ha₀ (mul_pos hd₀ hn) hap hdp
  72   have elo : ((d₀*n/2)*((a₀/2)/(d₁*n))^k)/2 = halfResidualLower a₀ d₀ d₁ k/n^(k-1) := by
  73     rw [show d₀*n/2=(d₀/2)*n by ring, degree_times_path_scale (a₀/2) d₁ (d₀/2) n hn k hk]
  74     unfold halfResidualLower
  75     rw [show (a₀/2)/d₁=a₀/(2*d₁) by ring]
  76     ring
  77   have eup : ((d₁*n)*(a₁/(d₀*n/2))^k)/2 = halfResidualUpper a₁ d₀ d₁ k/n^(k-1) := by
  78     rw [show d₀*n/2=(d₀/2)*n by ring, degree_times_path_scale a₁ (d₀/2) d₁ n hn k hk]
  79     unfold halfResidualUpper
  80     rw [show a₁/(d₀/2)=2*a₁/d₀ by ring]
  81     ring
  82   exact ⟨elo ▸ hlo,eup ▸ hup⟩
