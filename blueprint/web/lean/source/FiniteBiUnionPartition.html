   8 theorem finite_biUnion_partition
   9     {I R E:Type*} [Fintype I] [Fintype R] (blocks:I→Finset R) (edges:R→Finset E)
  10     (hcover:Finset.univ.biUnion blocks=Finset.univ)
  11     (hblocks:∀i j:I,i≠j→Disjoint (blocks i) (blocks j))
  12     (hedges:∀r s:R,r≠s→Disjoint (edges r) (edges s)):
  13     (∀i j:I,i≠j→Disjoint ((blocks i).biUnion edges) ((blocks j).biUnion edges)) ∧
  14     Finset.univ.biUnion (fun i=>(blocks i).biUnion edges)=Finset.univ.biUnion edges := by
  15   constructor
  16   · intro i j hij
  17     apply Finset.disjoint_left.mpr
  18     intro x hxi hxj
  19     obtain ⟨r,hri,hxr⟩:=Finset.mem_biUnion.mp hxi
  20     obtain ⟨s,hsj,hxs⟩:=Finset.mem_biUnion.mp hxj
  21     have hrs:r≠s:=by
  22       intro h
  23       subst s
  24       exact Finset.disjoint_left.mp (hblocks i j hij) hri hsj
  25     exact Finset.disjoint_left.mp (hedges r s hrs) hxr hxs
  26   · ext x
  27     simp only [Finset.mem_biUnion,Finset.mem_univ,true_and]
  28     constructor
  29     · rintro ⟨i,r,hr,hx⟩
  30       exact ⟨r,hx⟩
  31     · rintro ⟨r,hx⟩
  32       have hr:r∈Finset.univ.biUnion blocks:=by rw [hcover];exact Finset.mem_univ r
  33       obtain ⟨i,_,hri⟩:=Finset.mem_biUnion.mp hr
  34       exact ⟨i,r,hri,hx⟩
  35 end Oberwolfach
