  11 def cycleNext {r : ℕ} (i : Fin r) : Fin r :=
  12   ⟨(i.val + 1) % r, Nat.mod_lt _ (Nat.zero_lt_of_lt i.isLt)⟩
  13 
  14 theorem cycleNext_val {r : ℕ} (i : Fin r) :
  15     (cycleNext i).val = if i.val + 1 < r then i.val + 1 else 0 := by
  16   dsimp only [cycleNext]
  17   split_ifs with h
  18   · exact Nat.mod_eq_of_lt h
  19   · have hi : i.val + 1 = r := by omega
  20     rw [hi, Nat.mod_self]
  21 
  22 theorem cycleNext_two_ne {r : ℕ} (hr : 3 ≤ r) (i : Fin r) :
  23     cycleNext (cycleNext i) ≠ i := by
  24   intro heq
  25   have hv := congrArg Fin.val heq
  26   rw [cycleNext_val, cycleNext_val] at hv
  27   split_ifs at hv <;> omega

  29 theorem cycle_edge_injective {V : Type*} {r : ℕ} (hr : 3 ≤ r)
  30     (f : Fin r ↪ V) : Function.Injective (fun i => s(f i, f (cycleNext i))) := by
  31   intro i j hij
  32   rcases Sym2.eq_iff.mp hij with h | h
  33   · exact f.injective h.1
  34   · have hi := f.injective h.1
  35     have hj := f.injective h.2
  36     exfalso
  37     apply cycleNext_two_ne hr i
  38     rw [hj, ← hi]

  40 theorem cycle_copy_edgeFinset {n r : ℕ} {H : SimpleGraph (Fin n)}
  41     (_hr : 3 ≤ r) (f : Fin r ↪ Fin n)
  42     (hf : ∀ u v, H.Adj u v ↔ ∃ i j : Fin r, f i = u ∧ f j = v ∧
  43       ((i.val + 1) % r = j.val ∨ (j.val + 1) % r = i.val)) :
  44     H.edgeFinset = Finset.univ.image (fun i => s(f i, f (cycleNext i))) := by
  45   ext e
  46   induction e using Sym2.ind with
  47   | _ u v =>
  48     rw [SimpleGraph.mem_edgeFinset, SimpleGraph.mem_edgeSet]
  49     simp only [Finset.mem_image, Finset.mem_univ, true_and]
  50     constructor
  51     · intro huv
  52       obtain ⟨i, j, hi, hj, hstep | hstep⟩ := (hf u v).mp huv
  53       · have hnext : cycleNext i = j := Fin.ext hstep
  54         exact ⟨i, by rw [hnext, hi, hj]⟩
  55       · have hnext : cycleNext j = i := Fin.ext hstep
  56         exact ⟨j, by rw [hnext, hi, hj]; exact Sym2.eq_swap⟩
  57     · rintro ⟨i, hi⟩
  58       have hedge : H.Adj (f i) (f (cycleNext i)) :=
  59         (hf _ _).mpr ⟨i, cycleNext i, rfl, rfl, Or.inl rfl⟩
  60       have hmem : s(f i, f (cycleNext i)) ∈ H.edgeFinset :=
  61         SimpleGraph.mem_edgeFinset.mpr hedge
  62       rw [hi] at hmem
  63       exact SimpleGraph.mem_edgeFinset.mp hmem

  65 theorem cycle_copy_filtered_edge_count {n r : ℕ} {H : SimpleGraph (Fin n)}
  66     (hr : 3 ≤ r) (f : Fin r ↪ Fin n)
  67     (hf : ∀ u v, H.Adj u v ↔ ∃ i j : Fin r, f i = u ∧ f j = v ∧
  68       ((i.val + 1) % r = j.val ∨ (j.val + 1) % r = i.val))
  69     (P : Sym2 (Fin n) → Prop) [DecidablePred P] :
  70     (H.edgeFinset.filter P).card =
  71       ∑ i : Fin r, if P s(f i, f (cycleNext i)) then 1 else 0 := by
  72   rw [cycle_copy_edgeFinset hr f hf, Finset.card_filter, Finset.sum_image]
  73   exact fun i hi j hj hij => cycle_edge_injective hr f hij
