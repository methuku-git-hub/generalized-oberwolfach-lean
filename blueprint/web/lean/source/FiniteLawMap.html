  13 def map {Ω Γ : Type*} [Fintype Ω] [Fintype Γ] (μ : FiniteLaw Ω) (f : Ω → Γ) : FiniteLaw Γ where
  14   mass γ := ∑ ω, if f ω = γ then μ.mass ω else 0
  15   nonneg γ := Finset.sum_nonneg (fun ω _ => by split_ifs; exact μ.nonneg ω; exact le_refl _)
  16   total := by
  17     rw [Finset.sum_comm]
  18     simp only [Finset.sum_ite_eq, Finset.mem_univ, ite_true]
  19     exact μ.total

  29 theorem finiteLaw_ext
  30     {Ω : Type*} [Fintype Ω] {μ ν : FiniteLaw Ω} (h : ∀ ω, μ.mass ω = ν.mass ω) : μ = ν := by
  31   cases μ with
  32   | mk m hm ht =>
  33     cases ν with
  34     | mk n hn hs =>
  35       have heq : m = n := funext h
  36       subst n
  37       rfl

  39 theorem finiteLaw_map_probability
  40     {Ω Γ : Type*} [Fintype Ω] [Fintype Γ] (μ : FiniteLaw Ω) (f : Ω → Γ) (A : Γ → Prop) :
  41     (μ.map f).probability A = μ.probability (fun ω => A (f ω)) := by
  42   unfold FiniteLaw.probability
  43   change (∑ γ, if A γ then (∑ ω, if f ω = γ then μ.mass ω else 0) else 0) = _
  44   have hh (γ : Γ) : (if A γ then (∑ ω, if f ω = γ then μ.mass ω else 0) else 0) =
  45       ∑ ω, if f ω = γ then (if A (f ω) then μ.mass ω else 0) else 0 := by
  46     by_cases ha : A γ
  47     · simp only [ha, ite_true]
  48       apply Finset.sum_congr rfl
  49       intro ω _
  50       by_cases hf : f ω = γ <;> simp [hf, ha]
  51     · simp only [ha, ite_false]
  52       symm
  53       apply Finset.sum_eq_zero
  54       intro ω _
  55       by_cases hf : f ω = γ <;> simp [hf, ha]
  56   simp_rw [hh]
  57   rw [Finset.sum_comm]
  58   simp only [Finset.sum_ite_eq, Finset.mem_univ, ite_true]

  69 theorem finiteLaw_pi_map
  70     {E : Type*} [Fintype E] [DecidableEq E]
  71     {State Out : E → Type*} [∀ e, Fintype (State e)] [∀ e, Fintype (Out e)]
  72     (μ : ∀ e, FiniteLaw (State e)) (f : ∀ e, State e → Out e) :
  73     (FiniteLaw.pi μ).map (fun σ e => f e (σ e)) =
  74       FiniteLaw.pi (fun e => (μ e).map (f e)) := by
  75   apply finiteLaw_ext
  76   intro τ
  77   dsimp only [FiniteLaw.map, FiniteLaw.pi]
  78   rw [Fintype.prod_sum]
  79   apply Finset.sum_congr rfl
  80   intro σ _
  81   by_cases h : (fun e => f e (σ e)) = τ
  82   · have he (e : E) : f e (σ e) = τ e := congrFun h e
  83     simp [he]
  84   · have he : ∃ e, f e (σ e) ≠ τ e := by
  85       by_contra hn
  86       apply h
  87       funext e
  88       by_contra hf
  89       exact hn ⟨e, hf⟩
  90     obtain ⟨e, he⟩ := he
  91     rw [ite_eq_right h]
  92     symm
  93     apply Finset.prod_eq_zero (Finset.mem_univ e)
  94     simp [he]
