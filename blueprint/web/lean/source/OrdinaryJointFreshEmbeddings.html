   4 /-! Joint embeddings of finitely many different finite graph pairs. The
   5 single-pair existence input is used after adding every earlier row and hub
   6 label to the forbidden sets. Thus all cross-separation conclusions hold on
   7 one family of embeddings, rather than on separately chosen witnesses. -/

  17 theorem ordinary_joint_fresh_embeddings (l M:ℕ)
  18     (Vd Ud:Fin l→Type) [∀ i,Fintype (Vd i)] [∀ i,Fintype (Ud i)]
  19     [∀ i,DecidableEq (Vd i)] [∀ i,DecidableEq (Ud i)]
  20     (G:∀ i,SimpleGraph (Vd i)) (Hs:∀ i,Vd i→Ud i→Prop)
  21     (Q F:SimpleGraph V) (H:V→U→Prop) (A W R:Finset V) (B:Finset U)
  22     (hV:∀ i,Fintype.card (Vd i)≤M) (hU:∀ i,Fintype.card (Ud i)≤M)
  23     (hchoose:∀ (R':Finset V) (B':Finset U),R⊆R'→B⊆B'→
  24       R'.card≤R.card+l*M→B'.card≤B.card+l*M→∀ i:Fin l,
  25       ∃ e:OrdinaryPairHostEmbedding (G i) (Hs i) Q H,e.IsFresh F A W R' B') :
  26     ∃ e:∀ i,OrdinaryPairHostEmbedding (G i) (Hs i) Q H,
  27       (∀ i,(e i).IsFresh F A W R B)∧
  28       (∀ i j,i≠j→∀ v:Vd i,∀ w:Vd j,(e i).row v≠(e j).row w)∧
  29       (∀ i j,i≠j→∀ u:Ud i,∀ u':Ud j,(e i).hub u≠(e j).hub u')∧
  30       (∀ i j,i≠j→∀ v:Vd i,∀ w:Vd j,¬F.Adj ((e i).row v) ((e j).row w)) := by
  31   let C:=Option (Σ i:Fin l,OrdinaryPairHostEmbedding (G i) (Hs i) Q H)
  32   let rows:C→Finset V:=fun c=>c.elim ∅ (fun p=>Finset.univ.image p.2.row)
  33   let hubs:C→Finset U:=fun c=>c.elim ∅ (fun p=>Finset.univ.image p.2.hub)
  34   let allowed:=fun (i:Fin l) (c:C)=>∃ e:OrdinaryPairHostEmbedding (G i) (Hs i) Q H,
  35     c=some ⟨i,e⟩∧e.IsFresh F A W R B
  36   let compatible:=fun (c c':C)=>Disjoint (rows c) (rows c')∧Disjoint (hubs c) (hubs c')∧
  37     ∀ v∈rows c,∀ w∈rows c',¬F.Adj v w
  38   obtain ⟨chosen,hchosen,hcomp⟩:=sequential_select_prefix l allowed compatible none (by
  39     intro i chosen hp
  40     let s:=Finset.univ.filter (fun j:Fin l=>j.val < i.val)
  41     let R':=R∪s.biUnion (fun j=>rows (chosen j))
  42     let B':=B∪s.biUnion (fun j=>hubs (chosen j))
  43     have hsc:s.card≤l:=by simpa using Finset.card_le_univ s
  44     have hrc:(s.biUnion (fun j=>rows (chosen j))).card≤l*M:=by
  45       apply Finset.card_biUnion_le.trans
  46       calc
  47         _ ≤ ∑ _j∈s,M := by
  48           apply Finset.sum_le_sum
  49           intro j hj
  50           obtain ⟨e,he,hfresh⟩:=hp.2.1 j (Finset.mem_filter.mp hj).2
  51           rw [he]
  52           exact Finset.card_image_le.trans ((Finset.card_univ).trans_le (hV j))
  53         _ = s.card*M := by simp
  54         _ ≤ l*M := Nat.mul_le_mul_right _ hsc
  55     have hbc:(s.biUnion (fun j=>hubs (chosen j))).card≤l*M:=by
  56       apply Finset.card_biUnion_le.trans
  57       calc
  58         _ ≤ ∑ _j∈s,M := by
  59           apply Finset.sum_le_sum
  60           intro j hj
  61           obtain ⟨e,he,hfresh⟩:=hp.2.1 j (Finset.mem_filter.mp hj).2
  62           rw [he]
  63           exact Finset.card_image_le.trans ((Finset.card_univ).trans_le (hU j))
  64         _ = s.card*M := by simp
  65         _ ≤ l*M := Nat.mul_le_mul_right _ hsc
  66     obtain ⟨e,hfresh⟩:=hchoose R' B' Finset.subset_union_left Finset.subset_union_left
  67       ((Finset.card_union_le _ _).trans (Nat.add_le_add_left hrc _))
  68       ((Finset.card_union_le _ _).trans (Nat.add_le_add_left hbc _)) i
  69     refine ⟨some ⟨i,e⟩,⟨e,rfl,?_,(fun a h=>hfresh.2.1 a (Finset.mem_union_left _ h)),hfresh.2.2⟩,?_⟩
  70     · intro a
  71       exact ⟨(hfresh.1 a).1,(hfresh.1 a).2.1,
  72         fun h=>(hfresh.1 a).2.2.1 (Finset.mem_union_left _ h),
  73         fun z hz=>(hfresh.1 a).2.2.2 z (Finset.mem_union_left _ hz)⟩
  74     · intro j hj
  75       have hjS:j∈s:=Finset.mem_filter.mpr ⟨Finset.mem_univ _,hj⟩
  76       have hrj:rows (chosen j)⊆R':=fun _ h=>Finset.mem_union_right _ (Finset.mem_biUnion.mpr ⟨j,hjS,h⟩)
  77       have hbj:hubs (chosen j)⊆B':=fun _ h=>Finset.mem_union_right _ (Finset.mem_biUnion.mpr ⟨j,hjS,h⟩)
  78       refine ⟨?_,?_,?_⟩
  79       · apply Finset.disjoint_left.mpr
  80         intro v hv hv'
  81         obtain ⟨a,ha,rfl⟩:=Finset.mem_image.mp hv'
  82         exact (hfresh.1 a).2.2.1 (hrj hv)
  83       · apply Finset.disjoint_left.mpr
  84         intro u hu hu'
  85         obtain ⟨a,ha,rfl⟩:=Finset.mem_image.mp hu'
  86         exact hfresh.2.1 a (hbj hu)
  87       · intro v hv w hw
  88         obtain ⟨a,ha,rfl⟩:=Finset.mem_image.mp hw
  89         exact fun h=>(hfresh.1 a).2.2.2 v (hrj hv) h.symm)
  90   choose e he hfresh using hchosen
  91   refine ⟨e,hfresh,?_,?_,?_⟩
  92   · intro i j hij v w heq
  93     have hsymm (i j:Fin l) (hij:i.val < j.val):Disjoint (rows (chosen i)) (rows (chosen j)):=(hcomp i j hij).1
  94     have hd:Disjoint (rows (chosen i)) (rows (chosen j)):=by
  95       rcases lt_or_gt_of_ne hij with h|h
  96       · exact hsymm i j h
  97       · exact (hsymm j i h).symm
