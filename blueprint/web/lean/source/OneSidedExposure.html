  11 theorem finiteLaw_mgf_one_sided_tails
  12     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X : Ω → ℝ)
  13     {v : ℝ} (hv : 0 ≤ v)
  14     (hmgf : ∀ u, μ.expectation (fun ω => Real.exp (u * (X ω - μ.expectation X))) ≤
  15       Real.exp (u ^ 2 * v / 8)) {t : ℝ} (ht : 0 ≤ t) :
  16     μ.probability (fun ω => t ≤ X ω - μ.expectation X) ≤ Real.exp (-2 * t ^ 2 / v) ∧
  17     μ.probability (fun ω => t ≤ μ.expectation X - X ω) ≤ Real.exp (-2 * t ^ 2 / v) := by
  18   let : MeasurableSpace Ω := ⊤
  19   let q : ℝ≥0 := ⟨v / 4, by positivity⟩
  20   have hq : (q : ℝ) = v / 4 := rfl
  21   have hsub : HasSubgaussianMGF (fun ω => X ω - μ.expectation X) q μ.toPMF.toMeasure := by
  22     refine ⟨fun _ => Integrable.of_finite, ?_⟩
  23     intro u
  24     simp only [mgf, finiteLaw_integral]
  25     convert hmgf u using 1
  26     rw [hq]
  27     congr 1
  28     ring
  29   have he : -t ^ 2 / (2 * (q : ℝ)) = -2 * t ^ 2 / v := by rw [hq]; ring
  30   constructor
  31   · have hh := hsub.measure_ge_le ht
  32     rwa [finiteLaw_probability_measure, he] at hh
  33   · have hh := hsub.neg.measure_ge_le ht
  34     simpa only [Pi.neg_apply, neg_sub, finiteLaw_probability_measure, he] using hh
