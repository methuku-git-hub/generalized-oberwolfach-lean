  15 theorem marked_hub_cover_support
  16     (M : Finset (MarkedWheel tail head ktail khead m π b H₀ HK)) (u : Fin n→U)
  17     (hcover : M.biUnion markedWheelPhysicalResources=(markedPackingRims M).image Sum.inl∪
  18       Finset.univ.biUnion (fun i=>(Finset.univ.filter (fun v=>H₀ v (u i)∨HK v (u i))).image (fun v=>Sum.inr (v,u i))))
  19     (F : Finset V) (hF : ∀ z,π z∈F ↔ z∈F)
  20     (hneighbours : ∀ i x,H₀ x (u i)∨HK x (u i)→x∈F) :
  21     (∀ W∈M,W.2.val∈Finset.univ.image u) ∧
  22     (∀ W∈M,markedWheelRows W⊆F) ∧ GraphSupportedOn F (markedPackingRimGraph M) := by
  23   have hhub (W) (hW : W∈M) : W.2.val∈Finset.univ.image u := by
  24     have he : Sum.inr (W.1.val 0,W.2.val)∈M.biUnion markedWheelPhysicalResources :=
  25       Finset.mem_biUnion.mpr ⟨W,hW,(markedWheelPhysicalResources_spoke W _ _).mpr
  26         ⟨rfl,Finset.mem_image.mpr ⟨0,Finset.mem_univ _,rfl⟩⟩⟩
  27     rw [hcover] at he
  28     rcases Finset.mem_union.mp he with he|he
  29     · obtain ⟨e,_,he⟩ := Finset.mem_image.mp he
  30       cases he
  31     · obtain ⟨i,_,hi⟩ := Finset.mem_biUnion.mp he
  32       obtain ⟨x,hx,he⟩ := Finset.mem_image.mp hi
  33       exact Finset.mem_image.mpr ⟨i,Finset.mem_univ _,(Prod.mk.inj (Sum.inr.inj he)).2⟩
  34   have hrows (W) (hW : W∈M) : markedWheelRows W⊆F := by
  35     obtain ⟨i,_,hi⟩ := Finset.mem_image.mp (hhub W hW)
  36     intro x hx
  37     have he := markedWheelPhysicalResources_spoke_present W x W.2.val
  38       ((markedWheelPhysicalResources_spoke W x W.2.val).mpr ⟨rfl,hx⟩)
  39     rw [←hi] at he
  40     exact hneighbours i x he
  41   refine ⟨hhub,hrows,?_⟩
  42   intro x y hxy
  43   obtain ⟨W,hW,he⟩ := Finset.mem_biUnion.mp hxy.2
  44   have hs := markedWheelRimGraph_supported W x y ⟨hxy.1,he⟩
  45   have hsub : markedWheelRows W∪(markedWheelRows W).image π⊆F := by
  46     intro z hz
  47     rcases Finset.mem_union.mp hz with hz|hz
  48     · exact hrows W hW hz
  49     · obtain ⟨a,ha,rfl⟩ := Finset.mem_image.mp hz
  50       exact (hF a).mpr (hrows W hW ha)
  51   exact ⟨hsub hs.1,hsub hs.2⟩
