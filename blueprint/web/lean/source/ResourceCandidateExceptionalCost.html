  13 theorem finite_weighted_exception_union {I Z:Type*} [Fintype I]
  14     (T:Finset Z) (w:I→ℝ) (hw:∀ i,0≤w i) (P:I→Prop) (Q:Z→I→Prop):
  15     (∑ i,if P i∧(∃ z∈T,Q z i) then w i else 0)≤
  16       ∑ z∈T,∑ i,if P i∧Q z i then w i else 0:=by
  17   rw [Finset.sum_comm]
  18   apply Finset.sum_le_sum
  19   intro i _
  20   have hnon:∀ z∈T,0≤(if P i∧Q z i then w i else 0):=by
  21     intro z _;split_ifs <;> first | exact hw i | rfl
  22   by_cases he:P i∧(∃ z∈T,Q z i)
  23   · obtain ⟨z,hz,hQ⟩:=he.2
  24     rw [if_pos he]
  25     have hh:=Finset.single_le_sum hnon hz
  26     rw [if_pos (show P i∧Q z i from ⟨he.1,hQ⟩)] at hh
  27     exact hh
  28   · rw [if_neg he]
  29     exact Finset.sum_nonneg hnon

  31 theorem resourceRootOmittedLoad_exceptional_candidates {I A:Type*}
  32     [Fintype I] [Fintype A] [DecidableEq A]
  33     (R:I→Finset A) (w:I→ℝ) (hw:∀ i,0≤w i)
  34     (keep keep':A→Prop) (Good:I→Prop)
  35     (hsame:∀ i,Good i→∀ a∈R i,keep a↔keep' a) (e:A):
  36     |resourceRootOmittedLoad R w keep' e-resourceRootOmittedLoad R w keep e|≤
  37       ∑ i,if e∈R i∧¬Good i then w i else 0:=by
  38   unfold resourceRootOmittedLoad
  39   rw [←Finset.sum_sub_distrib]
  40   apply (Finset.abs_sum_le_sum_abs _ _).trans
  41   apply Finset.sum_le_sum
  42   intro i _
  43   by_cases he:e∈R i
  44   · by_cases hg:Good i
  45     · have hh:(∀ a∈R i,a≠e→keep a)↔(∀ a∈R i,a≠e→keep' a):=by
  46         constructor
  47         · intro h a ha hne;exact (hsame i hg a ha).mp (h a ha hne)
  48         · intro h a ha hne;exact (hsame i hg a ha).mpr (h a ha hne)
  49       have hz:(if e∈R i∧(∀ a∈R i,a≠e→keep' a) then w i else 0)=
  50           (if e∈R i∧(∀ a∈R i,a≠e→keep a) then w i else 0):=by
  51         by_cases h:∀ a∈R i,a≠e→keep a
  52         · rw [if_pos ⟨he,hh.mp h⟩,if_pos ⟨he,h⟩]
  53         · rw [if_neg (fun h'=>h (hh.mpr h'.2)),if_neg (fun h'=>h h'.2)]
  54       rw [hz,sub_self,abs_zero,if_neg (fun h=>h.2 hg)]
  55     · simp only [if_pos (show e∈R i∧¬Good i from ⟨he,hg⟩)]
  56       split_ifs <;> simp only [sub_self,abs_zero,sub_zero,zero_sub,abs_neg,abs_of_nonneg (hw i)] <;> first | rfl | exact hw i
  57   · simp only [he,false_and,ite_false,sub_self,abs_zero,le_refl]
