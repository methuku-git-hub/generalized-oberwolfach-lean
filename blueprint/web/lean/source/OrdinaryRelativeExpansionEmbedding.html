  16 theorem ordinary_relative_expansion_embedding (Q F : SimpleGraph V) (H : V → U → Prop)
  17     (A W R : Finset V) (B : Finset U) (φ : V₀ ↪ V) (ψ : U₀ ↪ U)
  18     (x y : E → V₀) (v : J → V₀) (u : J → U₀)
  19     (hφ : ∀ a,φ a∈R) (hψ : ∀ b,ψ b∈B)
  20     (r : ℕ) (hr : 3≤r) (d c cH : ℝ) (hd : 0≤d)
  21     (hF : ∀ v,(F.degree v:ℝ)≤d)
  22     (hcount : ∀ v w,c≤((A∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  23     (hpair : ∀ u u' v w,c≤(((A.filter (fun a => H a u ∧ H a u'))∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  24     (hhub : ∀ T:Finset V,T.card≤r → (∀ v∈T,∀ w∈T,¬F.Adj v w) →
  25       cH≤((common Finset.univ T (fun v => Finset.univ.filter (H v))).card:ℝ))
  26     (hbudget : W.card+((R.card:ℝ)+Fintype.card (E⊕J)*r+r+6)*(d+1)<c)
  27     (hB : (B.card:ℝ)+Fintype.card (E⊕J)<cH) :
  28     ∃ (f : ExpansionRow V₀ E J (Fin r) ↪ V) (g : ExpansionHub U₀ E J ↪ U),
  29       (∀ a,f (.inl a)=φ a) ∧ (∀ b,g (.inl b)=ψ b) ∧
  30       (∀ i w,f (.inr (i,w))∈A ∧ f (.inr (i,w))∉W ∧ f (.inr (i,w))∉R ∧
  31         ∀ z∈R,¬F.Adj (f (.inr (i,w))) z) ∧
  32       (∀ i,g (.inr i)∉B) ∧
  33       (∀ i j w w',¬F.Adj (f (.inr (i,w))) (f (.inr (j,w')))) ∧
  34       (∀ a b,(ordinaryExpansionRim (J:=J) (SimpleGraph.cycleGraph r)
  35         ⟨0,by omega⟩ ⟨r-1,by omega⟩ x y).Adj a b → Q.Adj (f a) (f b)) ∧
  36       ∀ a b,ordinaryExpansionSpokes (E:=E) ⟨0,by omega⟩ v u a b → H (f a) (g b) := by
  37   obtain ⟨f,a,hf,hH,hfR,haB,hind,hsep,hrim,hspoke⟩ := ordinary_relative_expansion_family Q F H A W R B
  38     (φ∘x) (φ∘y) (φ∘v) (ψ∘u) r hr d c cH hd hF hcount hpair hhub hbudget hB
  39   have hfroot : ∀ i w b,f i w≠φ b := by intro i w b he;exact (hfR i w).2.2.1 (he ▸ hφ b)
  40   have haroot : ∀ i b,a i≠ψ b := by intro i b he;exact haB i (he ▸ hψ b)
  41   have hainj : Function.Injective a := by intro i j he;by_contra hne;exact (hsep i j hne).1 he
  42   have hfsep : ∀ i j,i≠j → ∀ w w',f i w≠f j w' := fun i j hij w w' => ((hsep i j hij).2 w w').1
  43   let f' := expansionRowEmbedding φ f hfroot hfsep
  44   let g' := expansionHubEmbedding ψ a haroot hainj
  45   refine ⟨f',g',(fun _ => rfl),(fun _ => rfl),hfR,haB,?_,
  46     expansionRowEmbedding_hom φ f hfroot hfsep (SimpleGraph.cycleGraph r)
  47       ⟨0,by omega⟩ ⟨r-1,by omega⟩ x y Q hf hrim,
  48     expansionEmbedding_spokes φ f hfroot hfsep ψ a haroot hainj ⟨0,by omega⟩ v u H hH hspoke⟩
  49   intro i j w w'
  50   by_cases hij : i=j
  51   · subst j;exact hind i w w'
  52   · exact ((hsep i j hij).2 w w').2
