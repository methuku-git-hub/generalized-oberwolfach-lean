  12 def directedEdgeIndicator {V E : Type*} [DecidableEq V]
  13     (tail head : E → V) (x y : V) (e : E) : ℝ :=
  14   if tail e=x ∧ head e=y then 1 else 0

  16 theorem directedEdgeIndicator_single {V E : Type*} [DecidableEq V]
  17     (tail head : E → V) (hinj : Function.Injective (fun e => (tail e,head e)))
  18     (e f : E) : directedEdgeIndicator tail head (tail e) (head e) f=
  19       if f=e then 1 else 0 := by
  20   classical
  21   have hi : tail f=tail e ∧ head f=head e ↔ f=e :=
  22     ⟨fun h => hinj (Prod.ext h.1 h.2),fun h => by subst f;exact ⟨rfl,rfl⟩⟩
  23   simp only [directedEdgeIndicator,hi]
