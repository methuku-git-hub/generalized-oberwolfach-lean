  15 theorem regularBipartite_card_eq
  16     {X Y : Type*} [Fintype X] [Fintype Y]
  17     (M : X → Y → ℕ) {d : ℕ} (hd : 0 < d)
  18     (hrow : ∀ x, ∑ y, M x y = d)
  19     (hcol : ∀ y, ∑ x, M x y = d) :
  20     Fintype.card X = Fintype.card Y := by
  21   have h : Fintype.card X * d = Fintype.card Y * d := by
  22     calc
  23       _ = ∑ x, ∑ y, M x y := by simp [hrow]
  24       _ = ∑ y, ∑ x, M x y := Finset.sum_comm
  25       _ = _ := by simp [hcol]
  26   exact Nat.eq_of_mul_eq_mul_right hd h

  28 theorem regularBipartite_hall
  29     {X Y : Type*} [Fintype X] [Fintype Y] [DecidableEq Y]
  30     (M : X → Y → ℕ) {d : ℕ} (hd : 0 < d)
  31     (hrow : ∀ x, ∑ y, M x y = d)
  32     (hcol : ∀ y, ∑ x, M x y = d) (S : Finset X) :
  33     S.card ≤ (S.biUnion (fun x => Finset.univ.filter (fun y => 0 < M x y))).card := by
  34   classical
  35   let N := S.biUnion (fun x => Finset.univ.filter (fun y => 0 < M x y))
  36   have hzero (x : X) (hx : x ∈ S) (y : Y) (hy : y ∉ N) : M x y = 0 := by
  37     by_contra h
  38     apply hy
  39     exact Finset.mem_biUnion.mpr ⟨x, hx, Finset.mem_filter.mpr
  40       ⟨Finset.mem_univ _, Nat.pos_of_ne_zero h⟩⟩
  41   have hleft : ∑ x ∈ S, ∑ y ∈ N, M x y = S.card * d := by
  42     calc
  43       _ = ∑ x ∈ S, ∑ y, M x y := by
  44         apply Finset.sum_congr rfl
  45         intro x hx
  46         apply Finset.sum_subset (Finset.subset_univ _) 
  47         intro y _ hy
  48         exact hzero x hx y hy
  49       _ = _ := by simp [hrow]
  50   have hright : ∑ y ∈ N, ∑ x ∈ S, M x y ≤ N.card * d := by
  51     calc
  52       _ ≤ ∑ y ∈ N, ∑ x, M x y := by
  53         apply Finset.sum_le_sum
  54         intro y _
  55         exact Finset.sum_le_univ_sum_of_nonneg (fun x => Nat.zero_le (M x y))
  56       _ = _ := by simp [hcol]
  57   rw [Finset.sum_comm, hleft] at hright
  58   exact Nat.le_of_mul_le_mul_right hright hd

  60 /-- A positive regular bipartite multigraph contains one perfect matching. -/
  61 theorem regularBipartite_perfectMatching
  62     {X Y : Type*} [Fintype X] [Fintype Y]
  63     (M : X → Y → ℕ) {d : ℕ} (hd : 0 < d)
  64     (hrow : ∀ x, ∑ y, M x y = d)
  65     (hcol : ∀ y, ∑ x, M x y = d) :
  66     ∃ σ : X ≃ Y, ∀ x, 0 < M x (σ x) := by
  67   classical
  68   obtain ⟨f, hf, hpos⟩ :=
  69     (Finset.all_card_le_biUnion_card_iff_existsInjective'
  70       (fun x => Finset.univ.filter (fun y => 0 < M x y))).mp
  71       (regularBipartite_hall M hd hrow hcol)
  72   let σ := Equiv.ofBijective f
  73     ((Fintype.bijective_iff_injective_and_card f).mpr
  74       ⟨hf, regularBipartite_card_eq M hd hrow hcol⟩)
  75   exact ⟨σ, fun x => (Finset.mem_filter.mp (hpos x)).2⟩

  77 /-- Decomposition into perfect matchings, with exact edge multiplicities. -/
  78 theorem regularBipartite_decomposition
  79     {X Y : Type*} [Fintype X] [Fintype Y] [DecidableEq Y]
  80     (d : ℕ) (M : X → Y → ℕ)
  81     (hrow : ∀ x, ∑ y, M x y = d)
  82     (hcol : ∀ y, ∑ x, M x y = d) :
  83     ∃ σ : Fin d → X ≃ Y,
  84       ∀ x y, M x y = ∑ c, if σ c x = y then 1 else 0 := by
  85   classical
  86   induction d generalizing M with
  87   | zero =>
  88     refine ⟨Fin.elim0, ?_⟩
  89     intro x y
  90     have hle : M x y ≤ ∑ z, M x z :=
  91       Finset.single_le_sum (fun z _ => Nat.zero_le (M x z)) (Finset.mem_univ y)
  92     simpa [hrow x] using hle
  93   | succ d ih =>
  94     obtain ⟨σ, hσ⟩ := regularBipartite_perfectMatching M (Nat.succ_pos d) hrow hcol
  95     let I : X → Y → ℕ := fun x y => if σ x = y then 1 else 0
  96     have hI (x : X) (y : Y) : I x y ≤ M x y := by
  97       dsimp [I]
  98       split_ifs with h
  99       · subst y
 100         exact hσ x
 101       · exact Nat.zero_le _
 102     have hIrow (x : X) : ∑ y, I x y = 1 := by simp [I]
 103     have hIcol (y : Y) : ∑ x, I x y = 1 := by
 104       simp only [I, ← Equiv.eq_symm_apply]
 105       simp
 106     let N : X → Y → ℕ := fun x y => M x y - I x y
 107     have hNrow (x : X) : ∑ y, N x y = d := by
 108       dsimp [N]
 109       rw [Finset.sum_tsub_distrib _ (fun y _ => hI x y), hrow x, hIrow x]
 110       omega
 111     have hNcol (y : Y) : ∑ x, N x y = d := by
 112       dsimp [N]
 113       rw [Finset.sum_tsub_distrib _ (fun x _ => hI x y), hcol y, hIcol y]
 114       omega
 115     obtain ⟨τ, hτ⟩ := ih N hNrow hNcol
 116     refine ⟨Fin.cons σ τ, ?_⟩
 117     intro x y
 118     rw [Fin.sum_univ_succ]
 119     simp only [Fin.cons_zero, Fin.cons_succ]
 120     rw [← hτ x y]
 121     exact (Nat.add_sub_of_le (hI x y)).symm

 123 /-- Kőnig's theorem for a finite bipartite multigraph. `R c x y` means that
 124 colour `c` contains one of the edges from `x` to `y`. The cardinality identity
 125 retains every parallel edge, and the last two conclusions say each colour is
 126 a matching. -/
 127 theorem bipartiteMultigraph_edgeColouring
 128     {X Y : Type*} [Fintype X] [Fintype Y]
 129     (d : ℕ) (M : X → Y → ℕ)
 130     (hrow : ∀ x, ∑ y, M x y ≤ d)
 131     (hcol : ∀ y, ∑ x, M x y ≤ d) :
 132     ∃ R : Fin d → X → Y → Prop,
 133       (∀ x y, Nat.card {c : Fin d // R c x y} = M x y) ∧
 134       (∀ c x y z, R c x y → R c x z → y = z) ∧
 135       (∀ c x z y, R c x y → R c z y → x = z) := by
 136   classical
 137   let N : X ⊕ Y → Y ⊕ X → ℕ
 138     | .inl x, .inl y => M x y
 139     | .inl x, .inr z => if x = z then d - ∑ y, M x y else 0
 140     | .inr y, .inl z => if y = z then d - ∑ x, M x y else 0
 141     | .inr y, .inr x => M x y
 142   have hNrow : ∀ u, ∑ v, N u v = d := by
 143     intro u
 144     cases u with
 145     | inl x => simp [Fintype.sum_sum_type, N, Nat.add_sub_of_le (hrow x)]
 146     | inr y => simp [Fintype.sum_sum_type, N, Nat.sub_add_cancel (hcol y)]
 147   have hNcol : ∀ v, ∑ u, N u v = d := by
 148     intro v
 149     cases v with
 150     | inl y => simp [Fintype.sum_sum_type, N, Nat.add_sub_of_le (hcol y)]
 151     | inr x => simp [Fintype.sum_sum_type, N, Nat.sub_add_cancel (hrow x)]
 152   obtain ⟨σ, hσ⟩ := regularBipartite_decomposition d N hNrow hNcol
 153   refine ⟨fun c x y => σ c (.inl x) = .inl y, ?_, ?_, ?_⟩
 154   · intro x y
 155     rw [Nat.card_eq_fintype_card, Fintype.card_subtype, Finset.card_filter]
 156     exact (hσ (.inl x) (.inl y)).symm
 157   · intro c x y z hy hz
 158     exact Sum.inl_injective (hy.symm.trans hz)
 159   · intro c x z y hx hz
 160     exact Sum.inl_injective ((σ c).injective (hx.trans hz.symm))

 162 /-- The grouping used in the protected regular partition: a `k*q`-regular
 163 bipartite multigraph splits into `k` multigraphs of degree `q` on both sides. -/
 164 theorem regularBipartite_groupedDecomposition
 165     {X Y : Type*} [Fintype X] [Fintype Y]
 166     (k q : ℕ) (M : X → Y → ℕ)
 167     (hrow : ∀ x, ∑ y, M x y = k * q)
 168     (hcol : ∀ y, ∑ x, M x y = k * q) :
 169     ∃ Q : Fin k → X → Y → ℕ,
 170       (∀ x y, ∑ a, Q a x y = M x y) ∧
 171       (∀ a x, ∑ y, Q a x y = q) ∧
 172       (∀ a y, ∑ x, Q a x y = q) := by
 173   classical
 174   obtain ⟨σ, hσ⟩ := regularBipartite_decomposition (k * q) M hrow hcol
 175   let τ (a : Fin k) (b : Fin q) := σ (finProdFinEquiv (a, b))
 176   refine ⟨fun a x y => ∑ b, if τ a b x = y then 1 else 0, ?_, ?_, ?_⟩
 177   · intro x y
 178     calc
 179       _ = ∑ p : Fin k × Fin q, if τ p.1 p.2 x = y then 1 else 0 :=
 180         (Fintype.sum_prod_type _).symm
 181       _ = ∑ c, if σ c x = y then 1 else 0 :=
 182         finProdFinEquiv.sum_comp (fun c => if σ c x = y then 1 else 0)
 183       _ = M x y := (hσ x y).symm
 184   · intro a x
 185     rw [Finset.sum_comm]
 186     simp
 187   · intro a y
 188     rw [Finset.sum_comm]
 189     simp only [← Equiv.eq_symm_apply]
 190     simp

 192 /-- Distinct parallel copies are separate edges. -/
 193 abbrev BipartiteEdges {X Y : Type*} (M : X → Y → ℕ) := Σ x, Σ y, Fin (M x y)

 195 /-- A proper edge colouring of the actual multigraph edge set. Equality of
 196 colours and a common endpoint force equality of the edges, including the
 197 parallel-copy index. -/
 198 theorem bipartiteMultigraph_properEdgeColouring
 199     {X Y : Type*} [Fintype X] [Fintype Y]
 200     (d : ℕ) (M : X → Y → ℕ)
 201     (hrow : ∀ x, ∑ y, M x y ≤ d)
 202     (hcol : ∀ y, ∑ x, M x y ≤ d) :
 203     ∃ colour : BipartiteEdges M → Fin d,
 204       ∀ e f, colour e = colour f →
 205         (e.1 = f.1 ∨ e.2.1 = f.2.1) → e = f := by
 206   classical
 207   obtain ⟨R, hR, hright, hleft⟩ := bipartiteMultigraph_edgeColouring d M hrow hcol
 208   have hcard (x : X) (y : Y) :
 209       Fintype.card {c : Fin d // R c x y} = M x y := by
 210     simpa only [Nat.card_eq_fintype_card] using hR x y
 211   let E (x : X) (y : Y) : Fin (M x y) ≃ {c : Fin d // R c x y} :=
 212     (finCongr (hcard x y).symm).trans (Fintype.equivFin _).symm
 213   let colour : BipartiteEdges M → Fin d :=
 214     fun e => (E e.1 e.2.1 e.2.2).val
 215   refine ⟨colour, ?_⟩
 216   intro e f hc he
 217   rcases e with ⟨x, y, i⟩
 218   rcases f with ⟨z, w, j⟩
 219   change (E x y i).val = (E z w j).val at hc
 220   have hx : x = z := by
 221     rcases he with h | h
 222     · exact h
 223     · change y = w at h
 224       subst w
 225       exact hleft _ x z y (E x y i).property (hc.symm ▸ (E z y j).property)
 226   subst z
 227   have hy : y = w :=
 228     hright _ x y w (E x y i).property (hc.symm ▸ (E x w j).property)
 229   subst w
 230   have hij : i = j := (E x y).injective (Subtype.ext hc)
 231   subst j
 232   rfl
