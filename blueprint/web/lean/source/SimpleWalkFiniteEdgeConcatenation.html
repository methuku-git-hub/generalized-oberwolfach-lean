  11 theorem simple_walk_finite_concatenation_edges
  12     {V:Type*} {n:ℕ} (G:SimpleGraph V) (v:Fin (n+1) → V)
  13     (p:∀i:Fin n,G.Walk (v i.castSucc) (v i.succ)) (c:ℕ)
  14     (hp:∀i,(p i).IsPath) (hlen:∀i,(p i).length=c)
  15     (hdis:∀i j:Fin n,i ≠ j → ∀x ∈ (p i).support.dropLast,x ∉ (p j).support.dropLast)
  16     (hend:∀i:Fin n,v (Fin.last n) ∉ (p i).support.dropLast):
  17     ∃q:G.Walk (v 0) (v (Fin.last n)),q.IsPath ∧ q.length=n*c ∧
  18       (∀x,x ∈ q.support ↔ x=v (Fin.last n) ∨ ∃i:Fin n,x ∈ (p i).support.dropLast) ∧
  19       (∀e,e ∈ q.edges ↔ ∃i:Fin n,e ∈ (p i).edges) := by
  20   let v':ℕ → V:=finiteWalkClamp v
  21   let p':∀i:ℕ,G.Walk (v' i) (v' (i+1)):=finiteWalkExtension G v p
  22   have h0:v' 0=v 0:=by rfl
  23   have hn:v' (0+n)=v (Fin.last n):=by
  24     dsimp [v',finiteWalkClamp]
  25     congr 1
  26     apply Fin.ext
  27     simp
  28   have hp':∀i,0 ≤ i → i < 0+n → (p' i).IsPath:=by
  29     intro i _ hi
  30     have hh:=finite_walk_extension_initial_properties G v p ⟨i,by omega⟩
  31     exact hh.1.mpr (hp ⟨i,by omega⟩)
  32   have hl':∀i,0 ≤ i → i < 0+n → (p' i).length=c:=by
  33     intro i _ hi
  34     have hh:=finite_walk_extension_initial_properties G v p ⟨i,by omega⟩
  35     exact hh.2.1.trans (hlen ⟨i,by omega⟩)
  36   have hd':∀i j,0 ≤ i → i < 0+n → 0 ≤ j → j < 0+n → i ≠ j → 
  37       ∀x ∈ (p' i).support.dropLast,x ∉ (p' j).support.dropLast:=by
  38     intro i j _ hi _ hj hij x hx hy
  39     have hsi:=(finite_walk_extension_initial_properties G v p ⟨i,by omega⟩).2.2
  40     have hsj:=(finite_walk_extension_initial_properties G v p ⟨j,by omega⟩).2.2
  41     rw [hsi] at hx
  42     rw [hsj] at hy
  43     have hne:(⟨i,by omega⟩:Fin n) ≠ ⟨j,by omega⟩:=fun h=>hij (congrArg Fin.val h)
  44     exact hdis ⟨i,by omega⟩ ⟨j,by omega⟩ hne x hx hy
  45   have he':∀i,0 ≤ i → i < 0+n → v' (0+n) ∉ (p' i).support.dropLast:=by
  46     intro i _ hi hx
  47     have hsi:=(finite_walk_extension_initial_properties G v p ⟨i,by omega⟩).2.2
  48     rw [hsi,hn] at hx
  49     exact hend ⟨i,by omega⟩ hx
  50   obtain ⟨q,hq,hqlen,hqmem,hqedges⟩:=simple_walk_interval_concatenation_edges G v' p' c n 0 hp' hl' hd' he'
  51   let w:G.Walk (v 0) (v (Fin.last n)):=q.copy h0 hn
  52   have hwpath:w.IsPath:=by simpa [w] using hq
  53   have hwlen:w.length=n*c:=by simpa [w] using hqlen
  54   refine ⟨w,hwpath,hwlen,?_,?_⟩
  55   · intro x
  56     have hwmem:x ∈ w.support ↔ x ∈ q.support:=by simp [w]
  57     rw [hwmem,hqmem,hn]
  58     constructor
  59     · rintro (h|⟨i,_,hi,hx⟩)
  60       · exact Or.inl h
  61       · right
  62         have hh:=(finite_walk_extension_initial_properties G v p ⟨i,by omega⟩).2.2
  63         rw [hh] at hx
  64         exact ⟨⟨i,by omega⟩,hx⟩
  65     · rintro (h|⟨i,hx⟩)
  66       · exact Or.inl h
  67       · right;refine ⟨i.val,by omega,by simpa using i.isLt,?_⟩
  68         have hh:=(finite_walk_extension_initial_properties G v p i).2.2
  69         rw [hh]
  70         exact hx
  71   · intro e
  72     have hwedge:e ∈ w.edges ↔ e ∈ q.edges:=by simp [w]
  73     rw [hwedge,hqedges]
  74     constructor
  75     · rintro ⟨i,_,hi,he⟩
  76       have hh:=finite_walk_extension_initial_edges G v p ⟨i,by omega⟩
  77       rw [hh] at he
  78       exact ⟨⟨i,by omega⟩,he⟩
  79     · rintro ⟨i,he⟩
  80       refine ⟨i.val,by omega,by simpa using i.isLt,?_⟩
  81       have hh:=finite_walk_extension_initial_edges G v p i
  82       rw [hh]
  83       exact he
