  11 theorem marked_inner_packing_numeric_budgets
  12     (L q k dr ds nr ns a j sep B:ℝ) (hL:0≤L) (hq:0<q) (hB:0<B)
  13     (hklo:q≤k) (hkhi:k≤2*q) (hdr:dr≤q) (hds:ds≤q)
  14     (hnr:nr≤q^2) (hns:ns≤q^2) (ha:a≤B/10) (hj:j≤B/10)
  15     (hsep:sep≤B/10) (hqB:100*(L+20)*q≤B):
  16     (a+nr*(L+5)/k+sep+8*(k+dr)<B)∧
  17     (j+nr/k+(L+5)*(k+dr)<B)∧
  18     (a+ns*(L+5)/k+ds*(L+5)+sep+8*(k+ds)+8*(k+dr)+2*(k*(L+5))<B):=by
  19   have hk:0<k:=hq.trans_le hklo
  20   have hLf:0≤L+5:=by linarith
  21   have hrdiv:nr/k≤q:=by
  22     apply (div_le_iff₀ hk).mpr
  23     nlinarith only [hnr,mul_le_mul_of_nonneg_left hklo hq.le]
  24   have hsdiv:ns/k≤q:=by
  25     apply (div_le_iff₀ hk).mpr
  26     nlinarith only [hns,mul_le_mul_of_nonneg_left hklo hq.le]
  27   have hrdiv':nr*(L+5)/k≤q*(L+5):=by
  28     convert mul_le_mul_of_nonneg_right hrdiv hLf using 1 <;> ring
  29   have hsdiv':ns*(L+5)/k≤q*(L+5):=by
  30     convert mul_le_mul_of_nonneg_right hsdiv hLf using 1 <;> ring
  31   have hdsm:ds*(L+5)≤q*(L+5):=mul_le_mul_of_nonneg_right hds hLf
  32   have hkm:k*(L+5)≤2*q*(L+5):=mul_le_mul_of_nonneg_right hkhi hLf
  33   have hkrm:(L+5)*(k+dr)≤3*q*(L+5):=by
  34     have hh:=mul_le_mul_of_nonneg_left (add_le_add hkhi hdr) hLf
  35     nlinarith only [hh]
  36   have hlq:0≤L*q:=mul_nonneg hL hq.le

  38   · nlinarith only [ha,hrdiv',hsep,hkhi,hdr,hqB,hB,hq,hlq]

  43 theorem marked_inner_packing_thresholds (l:ℕ) (β:ℝ) (hβ:0<β):
  44     ∃ η₀:ℝ,0<η₀∧∀ η:ℝ,0<η→η≤η₀→∀ b:ℕ,∃ n₀:ℕ,∀ n:ℕ,n₀≤n→
  45       (0<n)∧(1≤η*n)∧(η≤1)∧
  46       (100*((l:ℝ)+20)*(η*n)≤β*n)∧(((l:ℝ)+6)*(2*b+1)≤β*n/10)∧
  47       (η*n≤(Nat.ceil (η*n):ℝ))∧((Nat.ceil (η*n):ℝ)≤2*(η*n)):=by
  48   let η₀:=min 1 (β/(100*((l:ℝ)+20)))
  49   have hη₀:0<η₀:=lt_min (by norm_num) (div_pos hβ (by positivity))
  50   refine ⟨η₀,hη₀,?_⟩
  51   intro η hη hsmall b
  52   obtain ⟨n₀,hn₀⟩:=exists_nat_gt (max (1/η) (10*(((l:ℝ)+6)*(2*b+1))/β))
  53   refine ⟨n₀,?_⟩
  54   intro n hn
  55   have hnreal:(n₀:ℝ)≤n:=by exact_mod_cast hn
  56   have ha:1/η<(n:ℝ):=(le_max_left _ _).trans_lt (hn₀.trans_le hnreal)
  57   have hb:10*(((l:ℝ)+6)*(2*b+1))/β<(n:ℝ):=(le_max_right _ _).trans_lt (hn₀.trans_le hnreal)
  58   have hηn:1<η*n:=by have hh:1<(n:ℝ)*η:=(div_lt_iff₀ hη).mp ha;nlinarith only [hh]
  59   have hnn:(0:ℝ)<n:=lt_trans (by positivity : (0:ℝ)<1/η) ha
  60   have hs:η≤β/(100*((l:ℝ)+20)):=hsmall.trans (min_le_right _ _)
  61   have hs':100*((l:ℝ)+20)*η≤β:=by
  62     have hh:η*(100*((l:ℝ)+20))≤β:=(le_div_iff₀ (by positivity)).mp hs
  63     nlinarith only [hh]
  64   have hceil:η*n≤(Nat.ceil (η*n):ℝ):=Nat.le_ceil _
  65   have hceil':(Nat.ceil (η*n):ℝ)<η*n+1:=Nat.ceil_lt_add_one (by positivity)
  66   refine ⟨by exact_mod_cast hnn,hηn.le,hsmall.trans (min_le_left _ _),?_,?_,hceil,?_⟩
  67   · nlinarith only [mul_le_mul_of_nonneg_right hs' hnn.le]
  68   · have hh:10*(((l:ℝ)+6)*(2*b+1))<(n:ℝ)*β:=(div_lt_iff₀ hβ).mp hb
  69     nlinarith only [hh]
  70   · linarith only [hceil',hηn]
