  15 def optionLabelledRows {r : ℕ} (o : Option ((Fin r ↪ V)×U)) : Finset V :=
  16   o.elim ∅ (fun p => Finset.univ.image p.1)

  18 def optionLabelledHubs {r : ℕ} (o : Option ((Fin r ↪ V)×U)) : Finset U :=
  19   o.elim ∅ (fun p => {p.2})

  21 theorem optionLabelledRows_card {r : ℕ} (o : Option ((Fin r ↪ V)×U)) :
  22     (optionLabelledRows o).card≤r := by
  23   cases o with
  24   | none => simp [optionLabelledRows]
  25   | some p => exact Finset.card_image_le.trans (by simp)

  30 theorem disjoint_labelled_gadgets (r N : ℕ) (R : Finset V) (B : Finset U)
  31     (P : Fin N → (Fin r ↪ V) → U → Prop)
  32     (hlocal : ∀ i (S : Finset V) (T : Finset U),R⊆S → B⊆T →
  33       S.card≤R.card+N*r → T.card≤B.card+N →
  34       ∃ (f : Fin r ↪ V) (u : U),P i f u ∧ (∀ a,f a∉S) ∧ u∉T) :
  35     ∃ (f : Fin N → Fin r ↪ V) (u : Fin N → U),
  36       (∀ i,P i (f i) (u i) ∧ (∀ a,f i a∉R) ∧ u i∉B) ∧
  37       (∀ i j,i≠j → u i≠u j ∧ ∀ a b,f i a≠f j b) := by
  38   let allowed := fun i (o : Option ((Fin r ↪ V)×U)) =>
  39     ∃ f u,o=some (f,u) ∧ P i f u ∧ (∀ a,f a∉R) ∧ u∉B
  40   let compatible := fun (o p : Option ((Fin r ↪ V)×U)) =>
  41     ∀ f u g v,o=some (f,u) → p=some (g,v) → u≠v ∧ ∀ a b,f a≠g b
  42   obtain ⟨F,hF,hcomp⟩ := sequential_select_prefix N allowed compatible none (by
  43     intro i F _hprefix
  44     let J := Finset.univ.filter (fun j : Fin N => j.val < i.val)
  45     let S := R∪J.biUnion (fun j => optionLabelledRows (F j))
  46     let T := B∪J.biUnion (fun j => optionLabelledHubs (F j))
  47     have hJ : J.card≤N := (Finset.card_le_univ _).trans_eq (Fintype.card_fin N)
  48     have hS : S.card≤R.card+N*r := by
  49       have hsum : (J.biUnion (fun j => optionLabelledRows (F j))).card≤J.card*r := by
  50         calc
  51           _ ≤ ∑ j∈J,(optionLabelledRows (F j)).card := Finset.card_biUnion_le
  52           _ ≤ ∑ _j∈J,r := Finset.sum_le_sum (fun _ _ => optionLabelledRows_card _)
  53           _ = _ := by simp
  54       have hh := Finset.card_union_le R (J.biUnion (fun j => optionLabelledRows (F j)))
  55       dsimp [S]
  56       nlinarith
  57     have hT : T.card≤B.card+N := by
  58       have hsum : (J.biUnion (fun j => optionLabelledHubs (F j))).card≤J.card := by
  59         calc
  60           _ ≤ ∑ j∈J,(optionLabelledHubs (F j)).card := Finset.card_biUnion_le
  61           _ ≤ ∑ _j∈J,1 := Finset.sum_le_sum (fun _ _ => optionLabelledHubs_card _)
  62           _ = _ := by simp
  63       have hh := Finset.card_union_le B (J.biUnion (fun j => optionLabelledHubs (F j)))
  64       dsimp [T]
  65       omega
  66     obtain ⟨f,u,hP,hf,hu⟩ := hlocal i S T Finset.subset_union_left Finset.subset_union_left hS hT
  67     refine ⟨some (f,u),⟨f,u,rfl,hP,?_,?_⟩,?_⟩
  68     · exact fun a h => hf a (Finset.mem_union_left _ h)
  69     · exact fun h => hu (Finset.mem_union_left _ h)
  70     · intro j hj g v f' u' hg hfu
  71       have he : (f,u)=(f',u') := Option.some.inj hfu
  72       cases he
  73       have hjJ : j∈J := Finset.mem_filter.mpr ⟨Finset.mem_univ _,hj⟩
  74       refine ⟨?_,?_⟩
  75       · intro he
  76         apply hu
  77         apply Finset.mem_union_right
  78         apply Finset.mem_biUnion.mpr
  79         refine ⟨j,hjJ,?_⟩
  80         rw [hg]
  81         simpa [optionLabelledHubs] using he.symm
  82       · intro a b he
  83         apply hf b
  84         apply Finset.mem_union_right
  85         apply Finset.mem_biUnion.mpr
  86         refine ⟨j,hjJ,?_⟩
  87         rw [hg]
  88         exact Finset.mem_image.mpr ⟨a,Finset.mem_univ _,he⟩)
  89   choose f u he hP hf hu using hF
  90   refine ⟨f,u,(fun i => ⟨hP i,hf i,hu i⟩),?_⟩
  91   intro i j hij
  92   rcases lt_or_gt_of_ne hij with h|h
  93   · exact hcomp i j h (f i) (u i) (f j) (u j) (he i) (he j)
  94   · have hh := hcomp j i h (f j) (u j) (f i) (u i) (he j) (he i)
  95     exact ⟨Ne.symm hh.1,fun a b => Ne.symm (hh.2 b a)⟩

  97 theorem disjoint_labelled_gadgets_finite {I : Type*} [Fintype I]
  98     (r : ℕ) (R : Finset V) (B : Finset U) (P : I → (Fin r ↪ V) → U → Prop)
  99     (hlocal : ∀ i (S : Finset V) (T : Finset U),R⊆S → B⊆T →
 100       S.card≤R.card+Fintype.card I*r → T.card≤B.card+Fintype.card I →
 101       ∃ (f : Fin r ↪ V) (u : U),P i f u ∧ (∀ a,f a∉S) ∧ u∉T) :
 102     ∃ (f : I → Fin r ↪ V) (u : I → U),
 103       (∀ i,P i (f i) (u i) ∧ (∀ a,f i a∉R) ∧ u i∉B) ∧
 104       (∀ i j,i≠j → u i≠u j ∧ ∀ a b,f i a≠f j b) := by
 105   let e := Fintype.equivFin I
 106   obtain ⟨f,u,hP,hsep⟩ := disjoint_labelled_gadgets r (Fintype.card I) R B
 107     (fun i => P (e.symm i)) (fun i => hlocal (e.symm i))
 108   refine ⟨fun i => f (e i),fun i => u (e i),?_,?_⟩
 109   · intro i
 110     simpa only [e.symm_apply_apply] using hP (e i)
 111   · intro i j hij
 112     exact hsep (e i) (e j) (fun he => hij (e.injective he))
