  14 theorem finite_domain_indicator_sum (B:Finset U) (P:U→Prop) :
  15     (∑ u:B,if P u.val then (1:ℝ) else 0)=(B.filter P).card := by
  16   rw [←finite_domain_predicate_card B P]
  17   simp only [Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]

  19 theorem finite_domain_codegree_sum (B:Finset U) (P Q:U→Prop) :
  20     (∑ u:B,(if P u.val then (1:ℝ) else 0)*(if Q u.val then 1 else 0))=
  21       (B.filter (fun u=>P u∧Q u)).card := by
  22   trans (∑ u:B,if P u.val∧Q u.val then (1:ℝ) else 0)
  23   · apply Finset.sum_congr rfl
  24     intro u _
  25     by_cases hp:P u.val <;> by_cases hq:Q u.val <;> simp [hp,hq]
  26   · convert finite_domain_indicator_sum B (fun u=>P u∧Q u) using 1 <;> congr!

  28 theorem degree_codegree_cuts_on_domains (H:V→U→Prop) (F:SimpleGraph V)
  29     (A:Finset V) (B:Finset U) (p ε η:ℝ) (d:ℕ)
  30     (hp:0<p) (hp1:p≤1) (hε:0≤ε) (hη:0<η) (hb:0<B.card)
  31     (hF:∀ v,F.degree v≤d)
  32     (hrow:∀ x∈A,(p-ε)*B.card≤((B.filter (H x)).card:ℝ))
  33     (hcodeg:∀ x∈A,∀ y∈A,x≠y → ¬F.Adj x y →
  34       ((B.filter (fun u=>H x u∧H y u)).card:ℝ)≤(p^2+ε)*B.card)
  35     (hsmall:ε≤p^2*η/24) (hlarge:(d+1:ℝ)≤p^2*η^2*B.card/8)
  36     (S:Finset V) (hSA:S⊆A) (T:Finset U) (hTB:T⊆B)
  37     (hS:η*B.card≤(S.card:ℝ)) (hT:η*B.card≤(T.card:ℝ)) :
  38     p/2*S.card*T.card≤(bipartiteMatrixCut (fun x u=>if H x u then 1 else 0) S T:ℝ) := by
  39   let e:A↪V:=⟨Subtype.val,Subtype.val_injective⟩
  40   let f:B↪U:=⟨Subtype.val,Subtype.val_injective⟩
  41   let H':A→B→Prop:=fun x u=>H x.val u.val
  42   let F':SimpleGraph A:=F.comap e
  43   let E:A→Finset A:=fun x=>insert x (F'.neighborFinset x)
  44   have hE (x:A):((E x).card:ℝ)≤d+1:=by
  45     have hh:(E x).card≤d+1:=by
  46       have h:=Finset.card_insert_le x (F'.neighborFinset x)
  47       rw [SimpleGraph.card_neighborFinset_eq_degree] at h
  48       exact h.trans (Nat.add_le_add_right ((embedding_comap_degree_le e F x).trans (hF x.val)) 1)
  49     exact_mod_cast hh
  50   have hrow' (x:A):(p-ε)*Fintype.card B≤∑ u,bipAdj H' x u:=by
  51     have he:(∑ u:B,bipAdj H' x u)=(B.filter (H x.val)).card:=finite_domain_indicator_sum B (H x.val)
  52     rw [he,Fintype.card_coe]
  53     exact hrow x.val x.property
  54   have hcodeg' (x y:A) (hy:y∉E x):
  55       (∑ u,bipAdj H' x u*bipAdj H' y u)≤(p^2+ε)*Fintype.card B:=by
  56     have hne:x.val≠y.val:=by
  57       intro he;have he':x=y:=Subtype.ext he;apply hy;simp [E,he']
  58     have hxy:¬F.Adj x.val y.val:=by
  59       intro ha;apply hy
  60       exact Finset.mem_insert_of_mem ((F'.mem_neighborFinset x y).mpr ha)
  61     have he:(∑ u:B,bipAdj H' x u*bipAdj H' y u)=(B.filter (fun u=>H x.val u∧H y.val u)).card:=
  62       finite_domain_codegree_sum B (H x.val) (H y.val)
  63     rw [he,Fintype.card_coe]
  64     exact hcodeg x.val x.property y.val y.property hne hxy
  65   let S':=embeddingPullFinset e S
  66   let T':=embeddingPullFinset f T
  67   have hSc:S'.card=S.card:=embeddingPullFinset_card e S (fun x hx=>⟨⟨x,hSA hx⟩,rfl⟩)
  68   have hTc:T'.card=T.card:=embeddingPullFinset_card f T (fun u hu=>⟨⟨u,hTB hu⟩,rfl⟩)
  69   have hSi:S'.image e=S:=embeddingPullFinset_image e S (fun x hx=>⟨⟨x,hSA hx⟩,rfl⟩)
  70   have hTi:T'.image f=T:=embeddingPullFinset_image f T (fun u hu=>⟨⟨u,hTB hu⟩,rfl⟩)
  71   have hh:=bipartite_cuts_from_degrees_and_codegrees H' p ε (d+1) η hp hp1 hε hη
  72     (by simpa only [Fintype.card_coe] using hb) E hE hrow' hcodeg' hsmall
  73     (by simpa only [Fintype.card_coe] using hlarge) S' T'
  74     (by simpa only [Fintype.card_coe,hSc] using hS) (by simpa only [Fintype.card_coe,hTc] using hT)
  75   have hcut:bipartiteMatrixCut (fun x u=>if H' x u then 1 else 0) S' T'=
  76       bipartiteMatrixCut (fun x u=>if H x u then 1 else 0) S T:=by
  77     rw [←hSi,←hTi]
  78     simp only [bipartiteMatrixCut]
  79     rw [Finset.sum_image (by intros a ha b hb he;exact e.injective he)]
  80     apply Finset.sum_congr rfl
  81     intro a ha
  82     rw [Finset.sum_image (by intros u hu v hv he;exact f.injective he)]
  83     rfl
  84   rw [hSc,hTc,hcut] at hh
  85   exact hh
