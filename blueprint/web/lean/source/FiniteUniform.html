  19 def uniform (Ω : Type*) [Fintype Ω] [Nonempty Ω] : FiniteLaw Ω where
  20   mass _ := 1 / Fintype.card Ω
  21   nonneg _ := by positivity
  22   total := by
  23     simp only [Finset.sum_const, Finset.card_univ, nsmul_eq_mul]
  24     have hn : (Fintype.card Ω : ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  25     field_simp
  26 
  27 def sigma {Ω : Type*} [Fintype Ω] {Γ : Ω → Type*} [∀ ω, Fintype (Γ ω)]
  28     (μ : FiniteLaw Ω) (ν : ∀ ω, FiniteLaw (Γ ω)) : FiniteLaw (Σ ω, Γ ω) where
  29   mass p := μ.mass p.1 * (ν p.1).mass p.2
  30   nonneg p := mul_nonneg (μ.nonneg _) ((ν p.1).nonneg _)
  31   total := by
  32     rw [Fintype.sum_sigma]
  33     simp only [← Finset.mul_sum, (ν _).total, mul_one]
  34     exact μ.total
  35 
  36 end FiniteLaw

  54 theorem finiteLaw_sigma_expectation
  55     {Ω : Type*} [Fintype Ω] {Γ : Ω → Type*} [∀ ω, Fintype (Γ ω)]
  56     (μ : FiniteLaw Ω) (ν : ∀ ω, FiniteLaw (Γ ω)) (X : (Σ ω, Γ ω) → ℝ) :
  57     (μ.sigma ν).expectation X =
  58       μ.expectation (fun ω => (ν ω).expectation (fun γ => X ⟨ω, γ⟩)) := by
  59   unfold FiniteLaw.expectation
  60   rw [Fintype.sum_sigma]
  61   apply Finset.sum_congr rfl
  62   intro ω _
  63   rw [Finset.mul_sum]
  64   apply Finset.sum_congr rfl
  65   intro γ _
  66   change (μ.mass ω * (ν ω).mass γ) * X ⟨ω, γ⟩ = μ.mass ω * ((ν ω).mass γ * X ⟨ω, γ⟩)
  67   ring

  82 theorem finiteLaw_map_expectation
  83     {Ω Γ : Type*} [Fintype Ω] [Fintype Γ]
  84     (μ : FiniteLaw Ω) (f : Ω → Γ) (X : Γ → ℝ) :
  85     (μ.map f).expectation X = μ.expectation (fun ω => X (f ω)) := by
  86   unfold FiniteLaw.expectation
  87   dsimp only [FiniteLaw.map]
  88   simp_rw [Finset.sum_mul]
  89   rw [Finset.sum_comm]
  90   apply Finset.sum_congr rfl
  91   intro ω _
  92   have hh (γ : Γ) : (if f ω = γ then μ.mass ω else 0) * X γ =
  93       if f ω = γ then μ.mass ω * X (f ω) else 0 := by
  94     by_cases h : f ω = γ <;> simp [h]
  95   simp_rw [hh]
