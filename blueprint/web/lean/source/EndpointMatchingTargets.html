  17     (weight : V→V→ℝ) : ℝ :=

  20 theorem endpoint_matching_row_target (F : SimpleGraph V) (P : Finset V) (x v : V)
  21     (weight : V→V→ℝ) (a C L : ℝ) (ha : 0<a) (hC : 0≤C)
  22     (hF : ∀ u,(F.degree u:ℝ)≤L)
  23     (hweight : ∀ w∈P,0≤weight v w ∧ weight v w≤C/a)
  24     (hsum : ∑ w∈P,weight v w=1) :
  25     |endpointEligibleRowWeight F P x v weight-1|≤2*C*L/a := by
  26   let Bad:=P.filter (fun w=>F.Adj x w ∨ F.Adj v w)
  27   have hBad : Bad⊆F.neighborFinset x∪F.neighborFinset v := by
  28     intro w hw
  29     rcases (Finset.mem_filter.mp hw).2 with h|h
  30     · exact Finset.mem_union_left _ (by simpa using h)
  31     · exact Finset.mem_union_right _ (by simpa using h)
  32   have hcard:=(Finset.card_le_card hBad).trans (Finset.card_union_le _ _)
  33   rw [SimpleGraph.card_neighborFinset_eq_degree,SimpleGraph.card_neighborFinset_eq_degree] at hcard
  34   have hcardR : (Bad.card:ℝ)≤2*L := by
  35     have hh : (Bad.card:ℝ)≤(F.degree x:ℝ)+(F.degree v:ℝ) := by exact_mod_cast hcard
  36     linarith only [hh,hF x,hF v]
  37   have hsplit : endpointEligibleRowWeight F P x v weight+(∑ w∈Bad,weight v w)=1 := by
  38     rw [←hsum,Finset.sum_filter]
  39     unfold endpointEligibleRowWeight
  40     rw [←Finset.sum_add_distrib]
  41     apply Finset.sum_congr rfl
  42     intro w hw
  43     by_cases hx:F.Adj x w <;> by_cases hv:F.Adj v w <;> simp [hx,hv]
  44   have hnonneg : 0≤∑ w∈Bad,weight v w := Finset.sum_nonneg
  45     (fun w hw=>(hweight w (Finset.mem_filter.mp hw).1).1)
  46   have hupper : (∑ w∈Bad,weight v w)≤2*C*L/a := by
  47     calc
  48       _ ≤ ∑ _w∈Bad,C/a := Finset.sum_le_sum (fun w hw=>(hweight w (Finset.mem_filter.mp hw).1).2)
  49       _ = (Bad.card:ℝ)*(C/a) := by simp
  50       _ ≤ (2*L)*(C/a) := mul_le_mul_of_nonneg_right hcardR (div_nonneg hC ha.le)
  51       _ = _ := by ring
  52   have he : endpointEligibleRowWeight F P x v weight-1=-(∑ w∈Bad,weight v w) := by linarith only [hsplit]
  53   rw [he,abs_neg,abs_of_nonneg hnonneg]
  54   exact hupper

  60 theorem endpoint_matching_total_target (F : SimpleGraph V) (P : Finset V) (x : V)
  61     (weight : V→V→ℝ) (a C L : ℝ) (ha : 0<a) (hC : 0≤C)
  62     (hPa : (P.card:ℝ)≤a) (hF : ∀ u,(F.degree u:ℝ)≤L)
  63     (hweight : ∀ v∈P,∀ w∈P,0≤weight v w ∧ weight v w≤C/a)
  64     (hsum : ∀ v∈P,∑ w∈P,weight v w=1) :
  65     |endpointEligibleOrderedMass F P x weight/2-(P.card:ℝ)/2|≤(1+2*C)*L/2 := by
  66   have hL : 0≤L := (Nat.cast_nonneg (F.degree x)).trans (hF x)
  67   let Good:=P.filter (fun v=>¬F.Adj x v)
  68   let Bad:=P.filter (fun v=>F.Adj x v)
  69   have hcount : Good.card+Bad.card=P.card := by
  70     dsimp [Good,Bad]
  71     rw [Finset.card_filter,Finset.card_filter,←Finset.sum_add_distrib]
  72     calc
  73       _ = ∑ _v∈P,(1:ℕ) := by
  74         apply Finset.sum_congr rfl
  75         intro v hv
  76         by_cases h:F.Adj x v <;> simp [h]
  77       _ = _ := by simp
  78   have hBad : (Bad.card:ℝ)≤L := by
  79     have hh : Bad⊆F.neighborFinset x := by intro v hv;simpa using (Finset.mem_filter.mp hv).2
  80     exact (by exact_mod_cast Finset.card_le_card hh : (Bad.card:ℝ)≤(F.degree x:ℝ)).trans (hF x)
  81   have hGood : (Good.card:ℝ)≤a := (by exact_mod_cast Finset.card_le_card (Finset.filter_subset _ _) :
  82     (Good.card:ℝ)≤P.card).trans hPa
  83   have hrow (v : V) (hv:v∈Good) :
  84       |endpointEligibleRowWeight F P x v weight-1|≤2*C*L/a :=
  85     endpoint_matching_row_target F P x v weight a C L ha hC hF
  86       (hweight v (Finset.mem_filter.mp hv).1) (hsum v (Finset.mem_filter.mp hv).1)
  87   have he : endpointEligibleOrderedMass F P x weight=∑ v∈Good,endpointEligibleRowWeight F P x v weight := by
  88     dsimp [endpointEligibleOrderedMass,Good];rw [Finset.sum_filter]
  89   have htotal : |endpointEligibleOrderedMass F P x weight-(Good.card:ℝ)|≤2*C*L := by
  90     rw [he]
  91     have hh : (∑ v∈Good,endpointEligibleRowWeight F P x v weight)-(Good.card:ℝ)=
  92         ∑ v∈Good,(endpointEligibleRowWeight F P x v weight-1) := by rw [Finset.sum_sub_distrib];simp
  93     rw [hh]
  94     calc
  95       _ ≤ ∑ v∈Good,|endpointEligibleRowWeight F P x v weight-1| := Finset.abs_sum_le_sum_abs _ _
  96       _ ≤ ∑ _v∈Good,2*C*L/a := Finset.sum_le_sum hrow
  97       _ = (Good.card:ℝ)*(2*C*L/a) := by simp
  98       _ ≤ a*(2*C*L/a) := mul_le_mul_of_nonneg_right hGood (by positivity)
  99       _ = 2*C*L := by field_simp
 100   have hcountR : (P.card:ℝ)=Good.card+(Bad.card:ℝ) := by exact_mod_cast hcount.symm
 101   have hd : |endpointEligibleOrderedMass F P x weight-(P.card:ℝ)|≤(1+2*C)*L := by
 102     rw [hcountR]
 103     have hh:=abs_sub_le (endpointEligibleOrderedMass F P x weight) (Good.card:ℝ)
 104       ((Good.card:ℝ)+Bad.card)
 105     have he' : |(Good.card:ℝ)-((Good.card:ℝ)+Bad.card)|=Bad.card := by
 106       rw [show (Good.card:ℝ)-((Good.card:ℝ)+Bad.card)=-(Bad.card:ℝ) by ring,abs_neg,abs_of_nonneg (Nat.cast_nonneg _)]
 107     rw [he'] at hh
 108     nlinarith only [hh,htotal,hBad]
 109   rw [←sub_div,abs_div,abs_of_pos (by norm_num : (0:ℝ)<2)]
 110   exact div_le_div_of_nonneg_right hd (by norm_num)
