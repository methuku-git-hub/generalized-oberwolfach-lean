  12     (a : Fin k → X → X → ℝ) (ρ : (Fin (k+1) → X) → ℝ) (j : Fin k) (x y : X) : ℝ :=
  13   a j x y-pathEdgeLoad ρ j x y

  15 def residualIntervalDegree {X : Type*} [Fintype X] {k : ℕ}

  19 theorem pathVertexLoad_nonneg {X : Type*} [Fintype X] {k : ℕ}
  20     (ρ : (Fin (k+1) → X) → ℝ) (hρ : ∀ p, 0 ≤ ρ p)
  21     (j : Fin (k+1)) (x : X) : 0 ≤ pathVertexLoad ρ j x := by
  22   apply Finset.sum_nonneg
  23   intro p _
  24   split_ifs <;> first | exact hρ p | exact le_refl _

  26 theorem residualIntervalDegree_bounds {X : Type*} [Fintype X] {k : ℕ} (hk : 0 < k)
  27     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ)
  28     (ρ : (Fin (k+1) → X) → ℝ) (hρ : ∀ p, 0 ≤ ρ p)
  29     (hd : ∀ j x, 0 < d j x)
  30     (hrow : ∀ j x, ∑ y, a j x y=d j.castSucc x)
  31     (hcol : ∀ j y, ∑ x, a j x y=d j.succ y)
  32     (hreserve : ∀ j x y, pathEdgeLoad ρ j x y ≤ (11/27)*a j x y) :
  33     ∀ j x, d j x/2 ≤ residualIntervalDegree d ρ j x ∧ residualIntervalDegree d ρ j x ≤ d j x := by
  34   intro j x
  35   have hv : pathVertexLoad ρ j x ≤ (11/27)*d j x := by
  36     by_cases hj : j.val < k
  37     · let e : Fin k := ⟨j.val,hj⟩
  38       have he : e.castSucc=j := by ext; rfl
  39       have hh := Finset.sum_le_sum (fun y (_ : y ∈ Finset.univ) => hreserve e x y)
  40       rw [pathEdgeLoad_row,← Finset.mul_sum,hrow,he] at hh
  41       exact hh
  42     · let e : Fin k := ⟨k-1,by omega⟩
  43       have he : e.succ=j := by ext; dsimp [e]; omega
  44       have hh := Finset.sum_le_sum (fun y (_ : y ∈ Finset.univ) => hreserve e y x)
  45       rw [pathEdgeLoad_column,← Finset.mul_sum,hcol,he] at hh
  46       exact hh
  47   have h0 := pathVertexLoad_nonneg ρ hρ j x
  48   have h1 := hd j x
  49   unfold residualIntervalDegree
  50   constructor <;> nlinarith
