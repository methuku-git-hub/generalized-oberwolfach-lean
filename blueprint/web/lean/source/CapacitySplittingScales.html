   8 theorem marked_position_signed_scale (ε ρ δ d A n mean : ℝ)
   9     (hρ : 0<ρ) (hδ : 0<δ) (hd : 0<d) (hA : 24/(δ*d)≤A) (hn : 0<n)
  10     (hε : ε≤δ*d/16) (hmean : d*n/2≤mean) :
  11     (A*ρ)*(2*ε*n)+3*(ρ*n) ≤ δ*(A*ρ)*mean := by
  12   have hAp : 0<A := (div_pos (by norm_num) (mul_pos hδ hd)).trans_le hA
  13   have hAmul : 24≤A*(δ*d) := (div_le_iff₀ (mul_pos hδ hd)).mp hA
  14   have heps := mul_le_mul_of_nonneg_left hε hAp.le
  15   have hbase : A*(2*ε)+3≤δ*A*d/2 := by nlinarith
  16   calc
  17     _ = (A*(2*ε)+3)*(ρ*n) := by ring
  18     _ ≤ (δ*A*d/2)*(ρ*n) := mul_le_mul_of_nonneg_right hbase (mul_pos hρ hn).le
  19     _ = (δ*(A*ρ))*(d*n/2) := by ring
  20     _ ≤ _ := mul_le_mul_of_nonneg_left hmean (mul_pos hδ (mul_pos hAp hρ)).le

  30 theorem exists_capacity_splitting_parameters (k : ℕ) (δ d q A B : ℝ)
  31     (hδ : 0<δ) (hd : 0<d) (hq : 0<q) (hA : 0<A) (hB : 0<B) :
  32     ∃ ε κ : ℝ,0<ε ∧ 0<κ ∧ ε≤δ*d/16 ∧ κ≤1/2 ∧
  33       2*(ε+κ)≤q/10 ∧ κ/(k+1:ℝ)+k*(B*A*(ε+κ))≤1/(2*(k+1:ℝ)) := by
  34   let C : ℝ := 1/(k+1:ℝ)+2*k*B*A
  35   have hC : 0<C := by unfold C;positivity
  36   let η := min (δ*d/16) (min (1/2) (min (q/40) (1/(2*(k+1:ℝ)*C))))/2
  37   have hη : 0<η := by unfold η;positivity
  38   have hle : η≤min (δ*d/16) (min (1/2) (min (q/40) (1/(2*(k+1:ℝ)*C)))) := by
  39     dsimp only [η]
  40     have hp : 0<min (δ*d/16) (min (1/2) (min (q/40) (1/(2*(k+1:ℝ)*C)))) := by positivity
  41     linarith
  42   have he := hle.trans (min_le_left _ _)
  43   have hk := hle.trans ((min_le_right _ _).trans (min_le_left _ _))
  44   have hq' := hle.trans ((min_le_right _ _).trans ((min_le_right _ _).trans (min_le_left _ _)))
  45   have hc := hle.trans ((min_le_right _ _).trans ((min_le_right _ _).trans (min_le_right _ _)))
  46   refine ⟨η,η,hη,hη,he,hk,by linarith,?_⟩
  47   have hm := mul_le_mul_of_nonneg_right hc hC.le
  48   have hid : (1/(2*(k+1:ℝ)*C))*C=1/(2*(k+1:ℝ)) := by field_simp
