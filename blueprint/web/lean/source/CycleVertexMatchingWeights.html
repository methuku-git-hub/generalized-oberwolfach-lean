  23 theorem ordinary_cycle_vertex_weight_identity
  24     {V : Type*} [Fintype V] [DecidableEq V]
  25     (G : SimpleGraph V) (r : ℕ) (a : OrdinaryCycle G r → ℝ)
  26     (he : ∀ x y,(∑ C,if C.val.val.Adj x y then a C else 0)=if G.Adj x y then 1 else 0)
  27     (v : V) :
  28     2*(∑ C,if v∈ordinaryCycleVertices G r C then a C else 0)=(G.degree v : ℝ) := by
  29   have hrow (C : OrdinaryCycle G r) :
  30       (∑ u,if C.val.val.Adj v u then a C else 0)=
  31         2*(if v∈ordinaryCycleVertices G r C then a C else 0) := by
  32     have hs : (∑ u,if C.val.val.Adj v u then a C else 0)=
  33         (C.val.val.degree v : ℝ)*a C := by
  34       calc
  35         _ = (∑ u,if C.val.val.Adj v u then (1 : ℝ) else 0)*a C := by
  36           rw [Finset.sum_mul]
  37           apply Finset.sum_congr rfl
  38           intro u _
  39           split_ifs <;> simp
  40         _ = _ := by rw [graph_adjacency_row_sum]
  41     rw [hs]
  42     rcases C.val.property.degree_zero_or_two v with hz|ht
  43     · have hn : v∉ordinaryCycleVertices G r C := by
  44         simpa only [ordinaryCycleVertices,Set.mem_toFinset] using
  45           (C.val.val.degree_eq_zero_iff_notMem_support v).mp hz
  46       simp [hz,hn]
  47     · have hm : v∈ordinaryCycleVertices G r C := by
  48         apply Set.mem_toFinset.mpr
  49         exact (C.val.val.degree_pos_iff_mem_support v).mp (by omega)
  50       simp [ht,hm]
  51   calc
  52     _ = ∑ C : OrdinaryCycle G r,∑ u,if C.val.val.Adj v u then a C else 0 := by
  53       rw [Finset.mul_sum]
  54       exact Finset.sum_congr rfl (fun C _=>(hrow C).symm)
  55     _ = ∑ u,∑ C : OrdinaryCycle G r,if C.val.val.Adj v u then a C else 0 := Finset.sum_comm
  56     _ = ∑ u,if G.Adj v u then (1 : ℝ) else 0 := Finset.sum_congr rfl (fun u _=>he v u)
  57     _ = _ := graph_adjacency_row_sum G v

  59 theorem ordinary_cycle_rescaled_vertex_load
  60     {V : Type*} [Fintype V] [DecidableEq V]
  61     (G : SimpleGraph V) (r : ℕ) (a : OrdinaryCycle G r → ℝ)
  62     (he : ∀ x y,(∑ C,if C.val.val.Adj x y then a C else 0)=if G.Adj x y then 1 else 0)
  63     (d : ℝ) (v : V) :
  64     (∑ C,if v∈ordinaryCycleVertices G r C then 2*a C/d else 0)=
  65       (G.degree v : ℝ)/d := by
  66   calc
  67     _ = 2*(∑ C,if v∈ordinaryCycleVertices G r C then a C else 0)/d := by
  68       rw [Finset.mul_sum,Finset.sum_div]
  69       exact Finset.sum_congr rfl (fun C _=>by split_ifs <;> simp)
  70     _ = _ := by rw [ordinary_cycle_vertex_weight_identity G r a he v]

 114 theorem ordinary_cycle_vertex_matching_weights
 115     {V : Type*} [Fintype V] [DecidableEq V]
 116     (G : SimpleGraph V) (r : ℕ) (hr : 0<r) (hn : 0<Fintype.card V)
 117     (a : OrdinaryCycle G r → ℝ) (ha : ∀ C,0≤a C)
 118     (B : ℝ) (hB : 0≤B) (hb : ∀ C,a C≤B/(Fintype.card V : ℝ)^(r-2))
 119     (he : ∀ x y,(∑ C,if C.val.val.Adj x y then a C else 0)=if G.Adj x y then 1 else 0)
 120     {d η : ℝ} (hd : (Fintype.card V : ℝ)/2≤d)
 121     (hdeg : ∀ v,(1-η)*d≤(G.degree v : ℝ) ∧ (G.degree v : ℝ)≤d) :
 122     ∃ b : OrdinaryCycle G r → ℝ,(∀ C,0≤b C) ∧
 123       (∀ v,1-η≤(∑ C,if v∈ordinaryCycleVertices G r C then b C else 0) ∧
 124         (∑ C,if v∈ordinaryCycleVertices G r C then b C else 0)≤1) ∧
 125       (∀ x y,x≠y →
 126         (∑ C,if x∈ordinaryCycleVertices G r C ∧ y∈ordinaryCycleVertices G r C then b C else 0)≤
 127           4*(r : ℝ)^2*B/(Fintype.card V : ℝ)) := by
 128   have hnR : (0 : ℝ)<Fintype.card V := Nat.cast_pos.mpr hn
 129   have hdpos : 0<d := by linarith
 130   refine ⟨fun C=>2*a C/d,fun C=>div_nonneg (mul_nonneg (by norm_num) (ha C)) hdpos.le,?_,?_⟩
 131   · intro v
 132     rw [ordinary_cycle_rescaled_vertex_load G r a he]
 133     exact ⟨(le_div_iff₀ hdpos).mpr (hdeg v).1,(div_le_one hdpos).mpr (hdeg v).2⟩
 134   · intro x y hxy
 135     apply (ordinary_cycle_rescaled_vertex_pair_load G r hr hn a ha B hB hb hdpos x y hxy).trans
 136     apply (div_le_div_iff₀ hdpos hnR).mpr
 137     have hp := mul_nonneg (mul_nonneg (sq_nonneg (r : ℝ)) hB)
 138       (show 0≤2*d-(Fintype.card V : ℝ) by linarith)
