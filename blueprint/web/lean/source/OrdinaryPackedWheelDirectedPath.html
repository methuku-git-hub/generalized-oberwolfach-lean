  19 theorem ordinary_packed_wheel_directed_path
  20     {V U:Type} [Fintype V] [LinearOrder V] [Fintype U] [DecidableEq U]
  21     (π:Equiv.Perm V) (b n:ℕ) (hn:2≤n) (W:OrderedOrdinaryWheel (U:=U) π b (n+1))
  22     (R:V→V→Prop) (hR:∀i:Fin (n+1),R (W.1.val i) (W.1.val (cycleNext i))):
  23     let start:=W.1.val (cycleNext 0)
  24     let finish:=W.1.val 0
  25     ∃q:(SimpleGraph.completeGraph V).Walk start finish,
  26       q.IsPath ∧ q.length=n ∧
  27       q.support.toFinset=Finset.univ.image W.1.val ∧
  28       R finish start ∧
  29       (∀k,k<q.length→R (q.getVert k) (q.getVert (k+1))):=by
  30   classical
  31   intro start finish
  32   let f:Fin (n+1)↪V:=⟨fun i=>W.1.val (cycleNext i),
  33     (permutationSeparated_injective π b W.1.val W.1.property).comp cycleNext_injective⟩
  34   have hnext:∀i:Fin n,cycleNext (cycleNext i.castSucc)=cycleNext i.succ:=by
  35     intro i
  36     apply Fin.ext
  37     simp only [cycleNext_val,Fin.val_castSucc,Fin.val_succ]
  38     have hi:=i.isLt
  39     split_ifs <;> omega
  40   have hf0:f ⟨0,by omega⟩=start:=rfl
  41   have hfn:f ⟨n,by omega⟩=finish:=by
  42     dsimp [f,finish]
  43     have hh:cycleNext (⟨n,by omega⟩:Fin (n+1))=0:=by
  44       apply Fin.ext
  45       rw [cycleNext_val]
  46       simp
  47     rw [hh]
  48   have hadj:∀j:Fin n,(SimpleGraph.completeGraph V).Adj (f ⟨j.val,by omega⟩) (f ⟨j.val+1,by omega⟩):=by
  49     intro j
  50     simp only [SimpleGraph.completeGraph_eq_top,SimpleGraph.top_adj]
  51     intro he
  52     have hh:=f.injective he
  53     have hh':j.val=j.val+1:=congrArg Fin.val hh
  54     omega
  55   obtain ⟨p,hp,hlen,hsupport⟩:=finite_sequence_simple_walk (SimpleGraph.completeGraph V) n f hadj
  56   have hdir:∀k,k<p.length→R (p.getVert k) (p.getVert (k+1)):=by
  57     intro k hk
  58     have hk':k<n:=by simpa [hlen] using hk
  59     have hv:∀j:Fin (n+1),p.getVert j.val=f j:=by
  60       intro j
  61       have hh:some (p.getVert j.val)=p.support[j.val]? :=p.getVert_eq_support_getElem? (by omega)
  62       rw [hsupport] at hh
  63       simpa only [List.getElem?_ofFn,dif_pos (show j.val<n+1 by omega),Option.some.injEq] using hh
  64     rw [hv ⟨k,by omega⟩,hv ⟨k+1,by omega⟩]
  65     have hh:=hR (cycleNext ⟨k,by omega⟩)
  66     have hnex:=hnext ⟨k,hk'⟩
  67     change cycleNext (cycleNext (⟨k,by omega⟩:Fin (n+1)))=cycleNext (⟨k+1,by omega⟩:Fin (n+1)) at hnex
  68     rw [hnex] at hh
  69     exact hh
  70   have hmem:p.support.toFinset=Finset.univ.image W.1.val:=by
  71     rw [hsupport]
  72     ext x
  73     simp only [List.mem_toFinset,List.mem_ofFn,Finset.mem_image,Finset.mem_univ,true_and]
  74     constructor
  75     · rintro ⟨i,hi⟩
  76       exact ⟨cycleNext i,hi⟩
  77     · rintro ⟨i,hi⟩
  78       have hsurj:Function.Surjective (@cycleNext (n+1)):=Finite.injective_iff_surjective.mp cycleNext_injective
  79       obtain ⟨j,hj⟩:=hsurj i
  80       exact ⟨j,by simpa [f,hj] using hi⟩
  81   have hclose:R finish start:=hR 0
  82   rw [←hf0,←hfn]
  83   refine ⟨p,hp,hlen,?_,?_,hdir⟩
  84   · ext x
  85     have hh:=congrArg (fun s:Finset V=>x∈s) hmem
  86     exact Iff.of_eq (by simpa only [List.mem_toFinset,Finset.mem_image,Finset.mem_univ,true_and] using hh)
  87   · simpa only [hf0,hfn] using hclose
