  18     (p:I→(Σ a:V,Σ b:V,Q.Walk a b))
  19     (hroot:∀ i,x i∈R∧y i∈R) (hxy:∀ i,x i≠y i)

  22       Q.Adj (y i) (p i).1∧Q.Adj (y i) (p i).2.1)

  25     (hF:∀ i,∀ v∈insert (x i) (insert (y i) (p i).2.2.support.toFinset),
  26       ∀ w∈insert (x i) (insert (y i) (p i).2.2.support.toFinset),¬F.Adj v w)
  27     (hH:∀ i,H (x i) (u i)∧H (y i) (u' i))
  28     (hHp:∀ i,∀ v∈(p i).2.2.support,H v (u i)∧H v (u' i))
  29     (hinj:Function.Injective (fun i=>(x i,u i))) (hinj':Function.Injective (fun i=>(y i,u' i))):
  30     ∃ W W':I→OrdinaryWheel Q F H r,
  31       (∀ i,ordinaryWheelHub (W i)=u i∧ordinaryWheelHub (W' i)=u' i) ∧
  32       (∀ i,∀ v,v∈(ordinaryWheelRim (W i)).support↔v=x i∨v∈(p i).2.2.support) ∧
  33       (∀ i,∀ v,v∈(ordinaryWheelRim (W' i)).support↔v=y i∨v∈(p i).2.2.support) ∧
  34       (∀ i j,i≠j→Disjoint (ordinaryWheelResources (W i)) (ordinaryWheelResources (W j))) ∧
  35       (∀ i j,i≠j→Disjoint (ordinaryWheelResources (W' i)) (ordinaryWheelResources (W' j))) ∧
  36       (∀ i e,
  37         (if Sum.inl e∈ordinaryWheelResources (W i) then (1:ℤ) else 0)-
  38           (if Sum.inl e∈ordinaryWheelResources (W' i) then 1 else 0)=
  39         (if e=s(x i,(p i).1) then 1 else 0)+(if e=s(x i,(p i).2.1) then 1 else 0)-
  40           (if e=s(y i,(p i).1) then 1 else 0)-(if e=s(y i,(p i).2.1) then 1 else 0)) ∧
  41       (∀ i v z,
  42         (if Sum.inr (v,z)∈ordinaryWheelResources (W i) then (1:ℤ) else 0)-
  43           (if Sum.inr (v,z)∈ordinaryWheelResources (W' i) then 1 else 0)=
  44         (if v=x i∧z=u i then 1 else 0)-(if v=y i∧z=u' i then 1 else 0)+
  45         (if v∈(p i).2.2.support∧z=u i then 1 else 0)-
  46           (if v∈(p i).2.2.support∧z=u' i then 1 else 0)):=by

  48     (p i).1 (p i).2.1 (u i) (u' i) (p i).2.2 (hpath i).1 (hpath i).2 (hxy i)
  49     (hends i).1 (hends i).2.1 (hends i).2.2.1 (hends i).2.2.2

  51       fun he=>havoid i v hv (he.symm ▸ (hroot i).2)⟩) (hF i) (hH i) (hHp i)
