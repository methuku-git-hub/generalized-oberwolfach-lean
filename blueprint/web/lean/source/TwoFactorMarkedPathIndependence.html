  10 theorem two_factor_spaced_path_vertices_independent
  11     (F:SimpleGraph V) (hF:F.IsRegularOfDegree 2)
  12     (s:Finset I) (u v:I→V) (P:∀ i,F.Walk (u i) (v i))
  13     (hp:∀i∈s,(P i).IsPath)
  14     (hP:(↑s:Set I).PairwiseDisjoint (fun i=>(P i).support.toFinset))
  15     (T:I→Finset ℕ)
  16     (hint:∀i∈s,∀a∈T i,0<a ∧ a<(P i).length)
  17     (hspace:∀i∈s,∀a∈T i,∀b∈T i,a≠b→a+1≠b ∧ b+1≠a):
  18     ∀ x∈s.biUnion (fun i=>(T i).image (P i).getVert),
  19       ∀ y∈s.biUnion (fun i=>(T i).image (P i).getVert),¬F.Adj x y := by
  20   intro x hx y hy hxy
  21   obtain ⟨i,hi,hx⟩:=Finset.mem_biUnion.mp hx
  22   obtain ⟨a,ha,rfl⟩:=Finset.mem_image.mp hx
  23   obtain ⟨j,hj,hy⟩:=Finset.mem_biUnion.mp hy
  24   obtain ⟨b,hb,rfl⟩:=Finset.mem_image.mp hy
  25   have hai:=hint i hi a ha
  26   have hbj:=hint j hj b hb
  27   by_cases hij:i=j
  28   · subst j
  29     have hab:a≠b := by
  30       intro h;subst b
  31       exact F.irrefl hxy
  32     have hc:=two_factor_path_adjacent_indices F hF (P i) (hp i hi) a b
  33       (by omega) hai.2 (Nat.le_of_lt hbj.2) hxy
  34     have hs:=hspace i hi a ha b hb hab
  35     rcases hc with hc|hc
  36     · exact hs.2 hc
  37     · exact hs.1 hc
  38   · have hys : (P j).getVert b∈(P i).support.toFinset :=
  39       two_factor_internal_neighbors_on_path F hF (P i) (hp i hi) a (by omega) hai.2 _ hxy
  40     have hyj : (P j).getVert b∈(P j).support.toFinset := by
  41       apply List.mem_toFinset.mpr
  42       exact (P j).getVert_mem_support b
  43     exact Finset.disjoint_left.mp (hP hi hj hij) hys hyj
  44 end Oberwolfach
