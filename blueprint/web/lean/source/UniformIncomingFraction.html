   2 /-! A head distributes its prescribed integer demand uniformly across its

  12 def eligibleIncoming (head : E → Y) (eligible : E → Prop) (y : Y) : Finset E :=
  13   Finset.univ.filter (fun e => eligible e ∧ head e = y)

  15 def uniformIncomingWeight (head : E → Y) (eligible : E → Prop) (t : Y → ℕ) (e : E) : ℝ :=
  16   if eligible e then (t (head e):ℝ)/(eligibleIncoming head eligible (head e)).card else 0

  18 theorem uniform_incoming_weight_bounds (head : E → Y) (eligible : E → Prop) (t : Y → ℕ)
  19     (ht : ∀ y, t y ≤ (eligibleIncoming head eligible y).card) (e : E) :
  20     0 ≤ uniformIncomingWeight head eligible t e ∧ uniformIncomingWeight head eligible t e ≤ 1 := by
  21   unfold uniformIncomingWeight
  22   split_ifs with he
  23   · have hle : (t (head e):ℝ) ≤ (eligibleIncoming head eligible (head e)).card := by exact_mod_cast ht (head e)
  24     have hn : 0 ≤ ((eligibleIncoming head eligible (head e)).card:ℝ) := by positivity
  25     constructor
  26     · exact div_nonneg (by positivity) hn
  27     · by_cases hzero : (eligibleIncoming head eligible (head e)).card = 0
  28       · simp [hzero]
  29       · have hpos : 0 < ((eligibleIncoming head eligible (head e)).card:ℝ) := by exact_mod_cast Nat.pos_of_ne_zero hzero
  30         exact (div_le_one₀ hpos).2 hle
  31   · norm_num

  33 theorem uniform_incoming_weight_columns (head : E → Y) (eligible : E → Prop) (t : Y → ℕ)
  34     (ht : ∀ y, t y ≤ (eligibleIncoming head eligible y).card) (y : Y) :
  35     (∑ e, if head e=y then uniformIncomingWeight head eligible t e else 0) = t y := by
  36   let S := eligibleIncoming head eligible y
  37   have heq : (∑ e, if head e=y then uniformIncomingWeight head eligible t e else 0) =
  38       ∑ e∈S, (t y:ℝ)/(S.card:ℝ) := by
  39     simp only [S,eligibleIncoming,Finset.sum_filter]
  40     apply Finset.sum_congr rfl
  41     intro e _
  42     by_cases he : eligible e <;> by_cases hy : head e=y <;>
  43       simp [S,eligibleIncoming,uniformIncomingWeight,he,hy]
  44   rw [heq,Finset.sum_const, nsmul_eq_mul]
  45   by_cases hS : S.card = 0
  46   · have ht0 : t y = 0 := by have hh:=ht y; change t y ≤ S.card at hh; omega
  47     simp [hS,ht0]
  48   · have hpos : (S.card:ℝ) ≠ 0 := by exact_mod_cast hS
  49     field_simp
  50 end Oberwolfach
