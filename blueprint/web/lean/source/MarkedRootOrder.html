   4 /-! The two root-first orders used in the manuscript. For an ordinary
   5 root the unique marked arc is from the second to the third free label. -/
   6 set_option autoImplicit false
   7 namespace Oberwolfach

   9 def markedRootOrder (m : ℕ) (j : Bool) : Equiv.Perm (Fin (m+1)) :=
  10   if j then Equiv.refl _ else Equiv.addRight (3:Fin (m+1))
  11 
  12 theorem markedRootOrder_successor (m : ℕ) (j : Bool) (k : Fin (m+1)) :
  13     markedRootOrder m j (finRotate (m+1) k)=finRotate (m+1) (markedRootOrder m j k) := by
  14   cases j
  15   · change finRotate (m+1) k+3=finRotate (m+1) (k+3)
  16     simp only [finRotate_apply]
  17     abel
  18   · rfl
  19 
  20 theorem markedRootOrder_type (m : ℕ) (hm : 3<m+1) (j : Bool) :
  21     markedPositionType ((markedRootOrder m j).symm 0)=j := by
  22   cases j
  23   · have hn : ((Equiv.addRight (3:Fin (m+1))).symm 0)≠0 := by
  24       intro he
  25       have hh := congrArg (Equiv.addRight (3:Fin (m+1))) he
  26       simp only [Equiv.apply_symm_apply] at hh
  27       change (0:Fin (m+1))=0+3 at hh
  28       rw [zero_add] at hh
  29       have hv := congrArg Fin.val hh
  30       change 0 = 3 % (m+1) at hv
  31       rw [Nat.mod_eq_of_lt hm] at hv
  32       omega
  33     simpa only [markedRootOrder,Bool.false_eq_true,ite_false,markedPositionType,decide_eq_false_iff_not] using hn
  34   · simp [markedRootOrder,markedPositionType]
  35 
  36 theorem markedRootOrder_marked_head (m : ℕ) : markedRootOrder m false 0=(3:Fin (m+1)) := by
  37   change (0:Fin (m+1))+3=3
  38   exact zero_add _
  39 
  40 theorem markedRootOrder_marked_tail (m : ℕ) : markedRootOrder m false (Fin.last m)=(2:Fin (m+1)) := by
  41   have hh := markedRootOrder_successor m false (Fin.last m)
  42   rw [finRotate_last,markedRootOrder_marked_head,finRotate_apply] at hh
  43   have he : (2:Fin (m+1))+1=3 := by
  44     apply Fin.ext
  45     change (2 % (m+1)+1 % (m+1)) % (m+1) = 3 % (m+1)
  46     rw [← Nat.add_mod]
  47   exact add_right_cancel (hh.symm.trans he.symm)
