  24 theorem cycleWeightsOfIntervals_edge {X : Type*} [Fintype X]
  25     (L : Fin 3 → ℕ) (hL : ∀ i,0<L i)
  26     (ω : (∀ i,Fin (L i+1) → X) → ℝ)
  27     (hω : ∀ p,0≤ω p)
  28     (hsupp : ∀ p,0<ω p → ∀ i,p i (Fin.last (L i))=p (finRotate 3 i) 0)
  29     (a : Fin (∑ i,L i) → X → X → ℝ)
  30     (hedge : ∀ (i : Fin 3) (j : Fin (L i)) (x y : X),
  31       (∑ p,if p i j.castSucc=x ∧ p i j.succ=y then ω p else 0)=a (finSigmaFinEquiv ⟨i,j⟩) x y)
  32     (k : Fin (∑ i,L i)) (x y : X) :
  33     (∑ v,if v k=x ∧ v (finRotate (∑ i,L i) k)=y then cycleWeightsOfIntervals L hL ω v else 0)=a k x y := by
  34   let q : Σ i,Fin (L i) := finSigmaFinEquiv.symm k
  35   have hq : finSigmaFinEquiv q=k := Equiv.apply_symm_apply _ _
  36   let f (p : ∀ i,Fin (L i+1) → X) := if p q.1 q.2.castSucc=x ∧ p q.1 q.2.succ=y then ω p else 0
  37   have hf : ∀ p,f p≠0 → ∀ i,p i (Fin.last (L i))=p (finRotate 3 i) 0 := by
  38     intro p hp
  39     have hz : ω p≠0 := by intro hz;simp [f,hz] at hp
  40     exact hsupp p (lt_of_le_of_ne (hω p) (Ne.symm hz))
  41   have hh := threePathsOfCycle_sum L hL f hf
  42   have heval (v : Fin (∑ i,L i) → X) :
  43       f (threePathsOfCycle L hL v)=
  44         if v k=x ∧ v (finRotate (∑ i,L i) k)=y then cycleWeightsOfIntervals L hL ω v else 0 := by
  45     have hb := threePathsOfCycle_edge L hL v q.1 q.2
  46     simp only [f,hb.1,hb.2,hq,cycleWeightsOfIntervals]
  47   simp only [heval] at hh
  48   rw [show (∑ p,f p)=a k x y by simpa only [hq] using hedge q.1 q.2 x y] at hh
  49   exact hh
