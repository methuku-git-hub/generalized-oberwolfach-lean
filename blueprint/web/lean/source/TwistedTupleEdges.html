  13 def PermutationSeparated {I V : Type*} (π : Equiv.Perm V) (b : ℕ) (f : I → V) : Prop :=
  14   ∀ i j, i ≠ j → ¬ PermutationClose π b (f i) (f j)

  16 theorem permutationSeparated_injective {I V : Type*}
  17     (π : Equiv.Perm V) (b : ℕ) (f : I → V) (h : PermutationSeparated π b f) :
  18     Function.Injective f := by
  19   intro i j he
  20   by_contra hij
  21   exact h i j hij (Or.inl he)
  22 
  23 theorem permutationSeparated_successor_avoid {I V : Type*}
  24     (π : Equiv.Perm V) (b : ℕ) (hb : 1 ≤ b) (f : I → V)
  25     (h : PermutationSeparated π b f) (i j : I) (hij : i ≠ j) :
  26     π (f i) ≠ f j := by
  27   intro he
  28   apply h i j hij
  29   exact Or.inr ⟨⟨0, by omega⟩, Or.inl (by simpa using he)⟩

  31 def twistedTupleEdge {c : ℕ} {V : Type*} (f : Fin (c + 1) → V) (z : V) :
  32     Fin c ⊕ Unit → Sym2 V
  33   | Sum.inl i => s(f i.castSucc, f i.succ)
  34   | Sum.inr _ => s(f (Fin.last c), z)
  35 
  36 theorem twistedTupleEdge_injective {c : ℕ} (hc : 2 ≤ c) {V : Type*}
  37     (f : Fin (c + 1) → V) (hf : Function.Injective f) (z : V)
  38     (hz : ∀ i : Fin (c + 1), i ≠ 0 → z ≠ f i) :
  39     Function.Injective (twistedTupleEdge f z) := by
  40   have hpath (i j : Fin c) (h : s(f i.castSucc, f i.succ) = s(f j.castSucc, f j.succ)) : i = j := by
  41     rcases Sym2.eq_iff.mp h with ⟨h1, h2⟩ | ⟨h1, h2⟩
  42     · apply Fin.ext
  43       have hi := congrArg Fin.val (hf h1)
  44       simpa only [Fin.val_castSucc] using hi
  45     · have hi := congrArg Fin.val (hf h1)
  46       have hj := congrArg Fin.val (hf h2)
  47       simp only [Fin.val_castSucc, Fin.val_succ] at hi hj
  48       omega
  49   have hlast (i : Fin c) : s(f i.castSucc, f i.succ) ≠ s(f (Fin.last c), z) := by
  50     intro h
  51     rcases Sym2.eq_iff.mp h with ⟨h1, h2⟩ | ⟨h1, h2⟩
  52     · have hi := congrArg Fin.val (hf h1)
  53       simp only [Fin.val_castSucc, Fin.val_last] at hi
  54       omega
  55     · have hi : i.castSucc = 0 := by
  56         by_contra hne
  57         exact hz i.castSucc hne h1.symm
  58       have hj := congrArg Fin.val (hf h2)
  59       have hi' := congrArg Fin.val hi
  60       simp only [Fin.val_castSucc, Fin.val_zero, Fin.val_succ, Fin.val_last] at hi' hj
  61       omega
  62   intro x y h
  63   cases x with
  64   | inl i =>
  65     cases y with
  66     | inl j => exact congrArg Sum.inl (hpath i j h)
  67     | inr u => exact (hlast i h).elim
  68   | inr u =>
  69     cases y with
  70     | inl j => exact (hlast j h.symm).elim
  71     | inr v => congr

  73 theorem separated_successor_edges_injective {c : ℕ} (hc : 2 ≤ c) {V : Type*}
  74     (π : Equiv.Perm V) (b : ℕ) (hb : 1 ≤ b) (f : Fin (c + 1) → V)
  75     (h : PermutationSeparated π b f) :
  76     Function.Injective (twistedTupleEdge f (π (f 0))) :=
  77   twistedTupleEdge_injective hc f (permutationSeparated_injective π b f h) _
  78     (fun i hi => permutationSeparated_successor_avoid π b hb f h 0 i hi.symm)
