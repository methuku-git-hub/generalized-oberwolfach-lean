  14 theorem typed_endpoint_triangle (Q F:SimpleGraph V) (H:V→U→Prop) (A:Finset V) (B:Finset U)
  15     (C:EndpointTriangle (ordinaryPhysicalGraph Q H) (ordinaryPhysicalGraph F (fun _ _=>False))
  16       (A.image Sum.inl) (B.image Sum.inr)):
  17     ∃ v w:V,∃ u:U,v∈A∧w∈A∧u∈B∧v≠w∧
  18       C.val={Sum.inl v,Sum.inl w,Sum.inr u}∧Q.Adj v w∧H v u∧H w u∧¬F.Adj v w:=by
  19   obtain ⟨v',w',hne,hvw⟩:=Finset.card_eq_two.mp C.property.2.1
  20   have hv':v'∈C.val∩A.image Sum.inl:=by rw [hvw];simp
  21   have hw':w'∈C.val∩A.image Sum.inl:=by rw [hvw];simp
  22   obtain ⟨v,hv,rfl⟩:=Finset.mem_image.mp (Finset.mem_inter.mp hv').2
  23   obtain ⟨w,hw,rfl⟩:=Finset.mem_image.mp (Finset.mem_inter.mp hw').2
  24   obtain ⟨u,hu,huid⟩:=Finset.mem_image.mp (endpointTriangleIndex_mem C).2
  25   have hC:C.val={Sum.inl v,Sum.inl w,Sum.inr u}:=by
  26     have he:=endpointTriangle_vertices_coordinates C
  27     rw [(endpointTriangleRowEdge_spec C).2,hvw,←huid] at he
  28     exact he.trans (by ext a;simp only [Finset.mem_insert,Finset.mem_singleton];tauto)
  29   have hvC:Sum.inl v∈C.val:=(Finset.mem_inter.mp hv').1
  30   have hwC:Sum.inl w∈C.val:=(Finset.mem_inter.mp hw').1
  31   have huC:Sum.inr u∈C.val:=huid ▸ (endpointTriangleIndex_mem C).1
  32   refine ⟨v,w,u,hv,hw,hu,fun he=>hne (congrArg Sum.inl he),hC,
  33     C.property.2.2.2.1 _ hvC _ hwC hne,
  34     C.property.2.2.2.1 _ hvC _ huC (by simp),
  35     C.property.2.2.2.1 _ hwC _ huC (by simp),
  36     C.property.2.2.2.2 _ hvC _ hwC⟩

  38 theorem endpoint_triangle_complete_at_row (Q F:SimpleGraph V) (H:V→U→Prop)
  39     (A:Finset V) (B:Finset U) (x:V) (hx:x∉A)
  40     (hQ:∀ v∈A,Q.Adj x v) (hH:∀ u∈B,H x u) (hF:∀ v∈A,¬F.Adj x v)
  41     (C:EndpointTriangle (ordinaryPhysicalGraph Q H) (ordinaryPhysicalGraph F (fun _ _=>False))
  42       (A.image Sum.inl) (B.image Sum.inr)):
  43     ∃ W:OrdinaryWheel Q F H 3,
  44       ordinaryWheelPhysicalGraph W=spanningRestriction ⊤ (insert (Sum.inl x) C.val) ∧
  45       (∀ a∈(ordinaryWheelRim W).support,a=x∨a∈A) ∧ ordinaryWheelHub W∈B:=by
  46   obtain ⟨v,w,u,hv,hw,hu,hvw,hC,hQvw,hHvu,hHwu,hFvw⟩:=typed_endpoint_triangle Q F H A B C
  47   have hxv:x≠v:=fun he=>hx (he.symm ▸ hv)
  48   have hxw:x≠w:=fun he=>hx (he.symm ▸ hw)
  49   have hcard:({x,v,w}:Finset V).card=3:=by simp [hxv,hxw,hvw]
  50   have hrim:∀ a∈({x,v,w}:Finset V),∀ b∈({x,v,w}:Finset V),a≠b→Q.Adj a b:=by
  51     intro a ha b hb hab
  52     simp only [Finset.mem_insert,Finset.mem_singleton] at ha hb
  53     rcases ha with rfl|rfl|rfl <;> rcases hb with rfl|rfl|rfl
  54     all_goals first | exact (hab rfl).elim | exact hQ _ hv | exact hQ _ hw | exact (hQ _ hv).symm | exact (hQ _ hw).symm | exact hQvw | exact hQvw.symm
  55   have hspoke:∀ a∈({x,v,w}:Finset V),H a u:=by
  56     intro a ha
  57     simp only [Finset.mem_insert,Finset.mem_singleton] at ha
  58     rcases ha with rfl|rfl|rfl
  59     · exact hH u hu
  60     · exact hHvu
  61     · exact hHwu
  62   have hconf:∀ a∈({x,v,w}:Finset V),∀ b∈({x,v,w}:Finset V),¬F.Adj a b:=by
  63     intro a ha b hb
  64     simp only [Finset.mem_insert,Finset.mem_singleton] at ha hb
  65     rcases ha with rfl|rfl|rfl <;> rcases hb with rfl|rfl|rfl
  66     all_goals first | exact fun h=>h.ne rfl | exact hF _ hv | exact hF _ hw | exact fun h=>hF _ hv h.symm | exact fun h=>hF _ hw h.symm | exact hFvw | exact fun h=>hFvw h.symm
  67   have hclique:∀ a∈({x,v,w}:Finset V).image Sum.inl∪{Sum.inr u},
  68       ∀ b∈({x,v,w}:Finset V).image Sum.inl∪{Sum.inr u},a≠b→(ordinaryPhysicalGraph Q H).Adj a b:=by
  69     intro a ha b hb hab
  70     cases a with
  71     | inl a=>
  72       have ha':a∈({x,v,w}:Finset V):=by simpa using ha
  73       cases b with
  74       | inl b=>
  75         have hb':b∈({x,v,w}:Finset V):=by simpa using hb
  76         exact hrim a ha' b hb' (fun h=>hab (congrArg Sum.inl h))
  77       | inr b=>
  78         have hb':b=u:=by simpa using hb
  79         subst b;exact hspoke a ha'
  80     | inr a=>
  81       have ha':a=u:=by simpa using ha
  82       subst a
  83       cases b with
  84       | inl b=>
  85         have hb':b∈({x,v,w}:Finset V):=by simpa using hb
  86         exact hspoke b hb'
  87       | inr b=>
  88         have hb':b=u:=by simpa using hb
  89         exact (hab (congrArg Sum.inr hb'.symm)).elim
  90   obtain ⟨W,hWr,hWu,hW⟩:=physical_triangle_wheel Q F H {x,v,w} u hcard hclique hconf
  91   refine ⟨W,?_,?_,hWu.symm ▸ hu⟩
  92   · rw [hW,hC]
  93     congr 1
  94     ext a
  95     simp only [Finset.mem_union,Finset.mem_image,Finset.mem_insert,Finset.mem_singleton]
  96     constructor
  97     · rintro (⟨a,(rfl|rfl|rfl),rfl⟩|rfl) <;> tauto
  98     · rintro (rfl|rfl|rfl|rfl)
  99       · exact Or.inl ⟨x,Or.inl rfl,rfl⟩
 100       · exact Or.inl ⟨v,Or.inr (Or.inl rfl),rfl⟩
 101       · exact Or.inl ⟨w,Or.inr (Or.inr rfl),rfl⟩
 102       · exact Or.inr rfl
