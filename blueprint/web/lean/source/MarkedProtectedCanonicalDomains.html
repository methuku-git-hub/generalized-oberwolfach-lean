  14 theorem marked_protected_canonical_domains

  16     (hI:∀ x,π x∈I↔x∈I) (hT:∀ x,π x∈T↔x∈T):
  17     let A:Type:={x:V//x∉T}
  18     let B:Type:={u:U//u∉W}
  19     let C:Type:={x:V//x∈I\T};

  21     (∀ x,f (ρA x)=π (f x))∧(∀ x,i (ρ x)=ρA (i x))∧

  27     (∀ x,ρA x∈X↔x∈X)∧
  28     (∀ x,(i.comp f) x=x.val)∧
