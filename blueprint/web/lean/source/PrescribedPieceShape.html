  14 theorem prescribed_path_no_loops:
  15     ∀ xs:List X,xs.Nodup→∀ x,(x,x)∉prescribedDirectedPathEdges xs := by
  16   intro xs
  17   induction xs with
  18   | nil => intro hn x;simp [prescribedDirectedPathEdges]
  19   | cons a xs ih =>
  20     cases xs with
  21     | nil => intro hn x;simp [prescribedDirectedPathEdges]
  22     | cons b bs =>
  23       intro hn x he
  24       have hh:=List.nodup_cons.mp hn
  25       simp only [prescribedDirectedPathEdges,Finset.mem_insert,Prod.mk.injEq] at he
  26       rcases he with ⟨ha,hb⟩|he
  27       · have hab:a=b:=ha.symm.trans hb
  28         exact hh.1 (by simp [hab])
  29       · exact ih hh.2 x he

  31 theorem prescribed_path_append_fresh_no_loops (a:X):
  32     ∀ xs:List X,xs.Nodup→a∉xs→∀ x,(x,x)∉prescribedDirectedPathEdges (xs++[a]) := by
  33   intro xs
  34   induction xs with
  35   | nil => intro hn ha x;simp [prescribedDirectedPathEdges]
  36   | cons b xs ih =>
  37     intro hn ha x he
  38     have hb:=List.nodup_cons.mp hn
  39     have ha':a∉xs:=by intro h;exact ha (by simp [h])
  40     cases xs with
  41     | nil =>
  42       change (x,x)∈prescribedDirectedPathEdges [b,a] at he
  43       simp only [prescribedDirectedPathEdges,Finset.mem_insert,Finset.notMem_empty,or_false,Prod.mk.injEq] at he

  55     ∀ x,(x,x)∉p.arcs := by
