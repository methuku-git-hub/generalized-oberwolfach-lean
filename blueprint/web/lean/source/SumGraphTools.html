  12 theorem sum_degree_inl {V W : Type*} [Fintype V] [Fintype W] [DecidableEq V] [DecidableEq W]
  13     (G : SimpleGraph V) (H : SimpleGraph W) (v : V) :
  14     (G.sum H).degree (Sum.inl v) = G.degree v := by
  15   have hs : (G.sum H).neighborSet (Sum.inl v) = Sum.inl '' G.neighborSet v := by
  16     ext x
  17     cases x <;> simp [SimpleGraph.mem_neighborSet, SimpleGraph.sum_adj]
  18   rw [← SimpleGraph.ncard_neighborSet, hs,
  19     Set.ncard_image_of_injective _ Sum.inl_injective, SimpleGraph.ncard_neighborSet]

  21 theorem sum_degree_inr {V W : Type*} [Fintype V] [Fintype W] [DecidableEq V] [DecidableEq W]
  22     (G : SimpleGraph V) (H : SimpleGraph W) (v : W) :
  23     (G.sum H).degree (Sum.inr v) = H.degree v := by
  24   have hs : (G.sum H).neighborSet (Sum.inr v) = Sum.inr '' H.neighborSet v := by
  25     ext x
  26     cases x <;> simp [SimpleGraph.mem_neighborSet, SimpleGraph.sum_adj]
  27   rw [← SimpleGraph.ncard_neighborSet, hs,
  28     Set.ncard_image_of_injective _ Sum.inr_injective, SimpleGraph.ncard_neighborSet]

  30 def disjointSumOnFin {a b : ℕ} (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b)) :
  31     SimpleGraph (Fin (a + b)) :=
  32   (G.sum H).map (finSumFinEquiv : Fin a ⊕ Fin b ≃ Fin (a + b)).toEmbedding
  33 
  34 def sumLeftEmbedding (a b : ℕ) : Fin a ↪ Fin (a + b) :=
  35   Function.Embedding.inl.trans
  36     (finSumFinEquiv : Fin a ⊕ Fin b ≃ Fin (a + b)).toEmbedding
  37 
  38 def sumLeftVertices (a b : ℕ) : Finset (Fin (a + b)) :=
  39   Finset.univ.map (sumLeftEmbedding a b)

  41 theorem mem_sumLeftVertices_inl {a b : ℕ} (v : Fin a) :
  42     (finSumFinEquiv : Fin a ⊕ Fin b ≃ Fin (a + b)) (Sum.inl v) ∈ sumLeftVertices a b := by
  43   exact Finset.mem_map.mpr ⟨v, Finset.mem_univ v, rfl⟩
  44 
  45 theorem not_mem_sumLeftVertices_inr {a b : ℕ} (v : Fin b) :
  46     (finSumFinEquiv : Fin a ⊕ Fin b ≃ Fin (a + b)) (Sum.inr v) ∉ sumLeftVertices a b := by
  47   intro hv
  48   obtain ⟨x, hx, hxeq⟩ := Finset.mem_map.mp hv
  49   change finSumFinEquiv (Sum.inl x) = finSumFinEquiv (Sum.inr v) at hxeq
  50   cases (finSumFinEquiv.injective hxeq)

  52 theorem disjointSumOnFin_adj {a b : ℕ} (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b))
  53     (u v : Fin a ⊕ Fin b) :
  54     (disjointSumOnFin G H).Adj (finSumFinEquiv u) (finSumFinEquiv v) ↔
  55       (G.sum H).Adj u v := by
  56   exact SimpleGraph.map_adj_apply
  57 
  58 theorem disjointSumOnFin_closed {a b : ℕ} (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b))
  59     (u v : Fin (a + b)) (huv : (disjointSumOnFin G H).Adj u v) :
  60     u ∈ sumLeftVertices a b ↔ v ∈ sumLeftVertices a b := by
  61   obtain ⟨u, rfl⟩ := (finSumFinEquiv : Fin a ⊕ Fin b ≃ Fin (a + b)).surjective u
  62   obtain ⟨v, rfl⟩ := (finSumFinEquiv : Fin a ⊕ Fin b ≃ Fin (a + b)).surjective v
  63   rw [disjointSumOnFin_adj] at huv
  64   cases u with
  65   | inl x =>
  66     cases v with
  67     | inl y => exact iff_of_true (mem_sumLeftVertices_inl x) (mem_sumLeftVertices_inl y)
  68     | inr y => cases huv
  69   | inr x =>
  70     cases v with
  71     | inl y => cases huv
  72     | inr y => exact iff_of_false (not_mem_sumLeftVertices_inr x) (not_mem_sumLeftVertices_inr y)

  74 theorem disjointSumOnFin_regular {a b d : ℕ} (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b))
  75     (hG : G.IsRegularOfDegree d) (hH : H.IsRegularOfDegree d) :
  76     (disjointSumOnFin G H).IsRegularOfDegree d := by
  77   intro x
  78   let e : Fin a ⊕ Fin b ≃ Fin (a + b) := finSumFinEquiv
  79   obtain ⟨y, rfl⟩ := e.surjective x
  80   have hh := SimpleGraph.degree_map_apply (G.sum H) e.toEmbedding y
  81   have hdeg : (G.sum H).degree y = d := by
  82     cases y with
  83     | inl v => exact (sum_degree_inl G H v).trans (hG v)
  84     | inr v => exact (sum_degree_inr G H v).trans (hH v)
  85   rw [hdeg] at hh
  86   simpa only [disjointSumOnFin, e, SimpleGraph.Iso.map_apply,
  87     Equiv.toEmbedding_apply, ← SimpleGraph.ncard_neighborSet] using hh

  89 theorem disjointSumOnFin_internal_left {a b : ℕ}
  90     (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b)) :
  91     internalEdges (disjointSumOnFin G H) (sumLeftVertices a b) =
  92       (G.map (sumLeftEmbedding a b)).edgeFinset := by
  93   classical
  94   have hemb (x : Fin a) (u : Fin a ⊕ Fin b) :
  95       sumLeftEmbedding a b x = finSumFinEquiv u ↔ Sum.inl x = u := by
  96     change finSumFinEquiv (Sum.inl x) = finSumFinEquiv u ↔ _
  97     exact finSumFinEquiv.injective.eq_iff
  98   ext e
  99   induction e using Sym2.ind with
 100   | _ u v =>
 101     obtain ⟨u, rfl⟩ := (finSumFinEquiv : Fin a ⊕ Fin b ≃ Fin (a + b)).surjective u
 102     obtain ⟨v, rfl⟩ := (finSumFinEquiv : Fin a ⊕ Fin b ≃ Fin (a + b)).surjective v
 103     rw [mem_internalEdges, disjointSumOnFin_adj, SimpleGraph.mem_edgeFinset,
 104       SimpleGraph.mem_edgeSet,
 105       SimpleGraph.map_adj]
 106     cases u <;> cases v <;>
 107       simp only [SimpleGraph.sum_adj, mem_sumLeftVertices_inl, not_mem_sumLeftVertices_inr,
 108         hemb, Sum.inl.injEq, Sum.inl_ne_inr] <;> aesop

 110 theorem disjointSumOnFin_internal_left_card {a b : ℕ}
 111     (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b)) :
 112     (internalEdges (disjointSumOnFin G H) (sumLeftVertices a b)).card = G.edgeFinset.card := by
 113   rw [disjointSumOnFin_internal_left]
 114   simpa only [← Set.ncard_coe_finset, SimpleGraph.coe_edgeFinset]
 115     using SimpleGraph.card_edgeFinset_map (sumLeftEmbedding a b) G
