  22     (T : ∀ i,Finset (ProfileRemainingClass F i)) (v : Σ i,V i) :
  23     (∃ w,((profileRootFreeEmbedding T).trans (profileComplementEmbedding F)) w=v) ↔
  24       v∉F∪(Finset.univ.sigma T).map (profileComplementEmbedding F) := by
  25   let e := profileComplementEmbedding F
  26   constructor
  27   · rintro ⟨w,rfl⟩ h
  28     rcases Finset.mem_union.mp h with hF | hT
  29     · exact (profileComplementEmbedding_range F _).mp ⟨profileRootFreeEmbedding T w,rfl⟩ hF
  30     · obtain ⟨u,hu,heu⟩ := Finset.mem_map.mp hT
  31       have heq : u=profileRootFreeEmbedding T w := e.injective heu
  32       subst u
  33       exact (profileRootFreeEmbedding_range T _).mp ⟨w,rfl⟩ hu
  34   · intro hv

  41     exact ⟨w,rfl⟩
