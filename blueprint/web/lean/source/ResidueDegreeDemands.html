  12 theorem finite_average_interval {I : Type*} [Fintype I]
  13     (hI : 0<Fintype.card I) (f : I → ℝ) {a b : ℝ}
  14     (hf : ∀ i,a≤f i ∧ f i≤b) :
  15     a≤(∑ i,f i)/Fintype.card I ∧ (∑ i,f i)/Fintype.card I≤b := by
  16   have hn : (0 : ℝ)<Fintype.card I := by exact_mod_cast hI
  17   constructor
  18   · apply (le_div_iff₀ hn).mpr
  19     calc
  20       _ = ∑ _ : I,a := by simp;ring
  21       _ ≤ _ := Finset.sum_le_sum (fun i _=>(hf i).1)
  22   · apply (div_le_iff₀ hn).mpr
  23     calc
  24       _ ≤ ∑ _ : I,b := Finset.sum_le_sum (fun i _=>(hf i).2)
  25       _ = _ := by simp;ring

  27 theorem residue_degree_demands {V U : Type*} [Fintype V] [Fintype U]
  28     (hn : 0<Fintype.card V) (hs : 0<Fintype.card U)
  29     (hsn : Fintype.card U≤Fintype.card V)
  30     (d : U → ℕ) (base r : ℕ) (hr : 0<r)
  31     {ω δ : ℝ} (hω : 0<ω) (hωn : ω*Fintype.card V≤Fintype.card U)
  32     (hδ : 0≤δ) (hrδ : (r : ℝ)≤δ*base) (hunit : 1≤δ*ω*base) :
  33     ∃ (k : U → ℕ) (f : V → ℕ) (T : ℕ),
  34       (∀ u,base≤k u ∧ k u<base+r ∧ k u%r=d u%r) ∧
  35       (∑ u,k u=T) ∧ (∑ v,f v=T) ∧
  36       (ω*base≤(T : ℝ)/Fintype.card V) ∧
  37       ((T : ℝ)/Fintype.card V≤base+r) ∧
  38       (∀ v,|(f v : ℝ)-(T : ℝ)/Fintype.card V|≤1) ∧
  39       (∀ u,(k u : ℝ)≤(1+δ)*(T : ℝ)/Fintype.card U) ∧
  40       (∀ v,(f v : ℝ)≤(1+δ)*(T : ℝ)/Fintype.card V) := by
  41   classical
  42   choose k hk using (fun u=>residue_in_integer_interval base (d u) r hr)
  43   let T := ∑ u,k u
  44   obtain ⟨f,hf,he⟩ := near_uniform_integer_demands (V:=V) hn T
  45   have hnR : (0 : ℝ)<Fintype.card V := by exact_mod_cast hn
  46   have hsR : (0 : ℝ)<Fintype.card U := by exact_mod_cast hs
  47   have hsNR : (Fintype.card U : ℝ)≤Fintype.card V := by exact_mod_cast hsn
  48   have hklo (u : U) : (base : ℝ)≤k u := by exact_mod_cast (hk u).1
  49   have hkhi (u : U) : (k u : ℝ)≤base+r := by exact_mod_cast (hk u).2.1.le
  50   have hTlo : (Fintype.card U : ℝ)*base≤T := by
  51     calc
  52       _ = ∑ _ : U,(base : ℝ) := by simp
  53       _ ≤ ∑ u,(k u : ℝ) := Finset.sum_le_sum (fun u _=>hklo u)
  54       _ = _ := by simp [T]
  55   have hThi : (T : ℝ)≤Fintype.card U*(base+r) := by
  56     calc
  57       _ = ∑ u,(k u : ℝ) := by simp [T]
  58       _ ≤ ∑ _ : U,((base : ℝ)+r) := Finset.sum_le_sum (fun u _=>hkhi u)
  59       _ = _ := by simp;ring
  60   have hmeanlo : ω*base≤(T : ℝ)/Fintype.card V := by
  61     apply (le_div_iff₀ hnR).mpr
  62     have := mul_le_mul_of_nonneg_right hωn (Nat.cast_nonneg base : (0 : ℝ)≤base)
  63     nlinarith
  64   have hmeanhi : (T : ℝ)/Fintype.card V≤base+r := by
  65     apply (div_le_iff₀ hnR).mpr
  66     have := mul_le_mul_of_nonneg_right hsNR (show (0 : ℝ)≤base+r by positivity)
  67     nlinarith
  68   have hublo : (base : ℝ)≤(T : ℝ)/Fintype.card U := by
  69     apply (le_div_iff₀ hsR).mpr
  70     nlinarith
  71   refine ⟨k,f,T,hk,rfl,hf,hmeanlo,hmeanhi,he,?_,?_⟩
  72   · intro u
  73     have hm := mul_le_mul_of_nonneg_left hublo hδ
  74     calc
  75       (k u : ℝ) ≤ base+r := hkhi u
  76       _ ≤ (1+δ)*((T : ℝ)/Fintype.card U) := by nlinarith
  77       _ = _ := by ring
  78   · intro v
  79     have hm := mul_le_mul_of_nonneg_left hmeanlo hδ
  80     have hh := (abs_le.mp (he v)).2
  81     calc
  82       (f v : ℝ) ≤ (1+δ)*((T : ℝ)/Fintype.card V) := by nlinarith
  83       _ = _ := by ring
