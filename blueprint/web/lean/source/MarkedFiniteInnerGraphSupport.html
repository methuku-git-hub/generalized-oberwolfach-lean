  14 theorem marked_finite_inner_graph_support
  15     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ) (σ:RawMarkedOutcome V U Γ Color)
  16     (I:Finset V) (J:Finset U) (hI:∀ x,π x∈I↔x∈I):
  17     let E:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) false
  18     let K:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) true;
  19     ∀ (A:Finset (E⊕K)) (S:Finset (V×U)),
  20     (∀ a∈A,Sum.elim (fun e=>e.val.1) (fun e=>e.val.1) a∈I∧
  21       Sum.elim (fun e=>e.val.2) (fun e=>e.val.2) a∈I)→
  22     (∀ z∈S,z.1∈I∧z.2∈J)→
  23     (∀ x y,(markedSampleArcSetGraph P π d σ A).Adj x y→x∈I∧y∈I)∧
  24     (∀ x u,(markedFiniteSpokeGraph S).Adj (Sum.inl x) (Sum.inr u)→x∈I∧u∈J):=by
  25   intro E K A S hA hS
  26   constructor
  27   · intro x y hxy
  28     obtain ⟨a,ha,he⟩:=Finset.mem_image.mp hxy.2
  29     have heI:∀ z∈markedPhysicalEdge (fun e:E=>e.val.1) (fun e=>e.val.2)
  30         (fun e:K=>e.val.1) (fun e=>e.val.2) π a,z∈I:=by
  31       have hh:=hA a ha
  32       cases a with
  33       | inl a=>
  34         intro z hz
  35         have hz':z=a.val.1∨z=a.val.2:=by simpa only [markedPhysicalEdge,Sum.elim_inl,Finset.mem_insert,Finset.mem_singleton] using hz
  36         rcases hz' with rfl|rfl
  37         · exact hh.1
  38         · exact hh.2
  39       | inr a=>
  40         intro z hz
  41         have hz':z=a.val.1∨z=π a.val.2:=by simpa only [markedPhysicalEdge,Sum.elim_inr,Finset.mem_insert,Finset.mem_singleton] using hz
  42         rcases hz' with rfl|rfl
  43         · exact hh.1
  44         · exact (hI _).mpr hh.2
  45     rw [he] at heI
  46     exact ⟨heI x (by simp),heI y (by simp)⟩
  47   · intro x u h
  48     exact hS (x,u) ((markedFiniteSpokeGraph_adj S x u).mp h)
