  13 theorem long_preparation_factor_family_seed_coefficients {U:Type} [Fintype U] {n:ℕ} (hn:0<n)
  14     (F:U→SimpleGraph (Fin n)) (hF:∀ u,(F u).IsRegularOfDegree 2)
  15     (K:ℕ) (hK:0<K) (m d:ℕ) (hm:7≤m) (hd:2≤d) (θ ρ D:ℝ)
  16     (hθ0:0≤θ) (hθ:θ≤1/100) (hρ:0<ρ) (hρ1:ρ≤1) (hD:0<D)
  17     (hsize:ρ*D=2*Fintype.card U) (hcut:(1/((m:ℝ)+1))/(d:ℝ)≤θ)
  18     (hlong:∀ u,(1/2:ℝ)≤
  19       ((Finset.univ.filter (fun v=>K<Fintype.card ((F u).connectedComponentMk v))).card:ℝ)/n):
  20     let fraction:=fun (u:U) (j:Option (Fin K))=>
  21       ((Finset.univ.filter (fun v=>longPreparationCycleClass (F u) (hF u) K v=j)).card:ℝ)/n
  22     let retained:=fun (u:U)=>Finset.univ.filter (fun i=>θ/(100*(K:ℝ))≤fraction u (some i))
  23     let f:=fun (i:Fin K) u=>if i∈retained u then fraction u (some i) else 0
  24     let c:ℝ:=(m:ℝ)+1
  25     let t:ℝ:=(1/100)/(m:ℝ)
  26     let a:=fun u=>(1-θ)*fraction u none/c-t
  27     let b:=fun u=>(m:ℝ)*a u+(1-θ)*∑ i,f i u
  28     let η:=fun i u=>(1-θ)*f i u/b u
  29     let p:=(∑ u,b u)/D
  30     let k:=(∑ u,a u)/D
  31     let pstar:=min (1/4:ℝ) (min (ρ/8) (ρ/(6*c)));
  32     0<pstar ∧ pstar≤p ∧ pstar≤k ∧ (∀ u,pstar≤b u) ∧
  33     (∀ i u,0≤η i u ∧ η i u≤1/b u) ∧
  34     (∀ u,(m:ℝ)*a u+(∑ i,η i u)*b u=b u) ∧
  35     (∑ u,a u)=D*k ∧ (∑ u,b u)=D*p:=by
  36   intro fraction retained f c t a b η p k pstar
  37   have hc:0<c:=by dsimp [c];positivity
  38   have hm0:(0:ℝ)≤m:=by positivity
  39   have hpstar0:0<pstar:=by dsimp [pstar];positivity
  40   have hpstarb:pstar≤1/4:=min_le_left _ _
  41   have hpstarp:pstar≤ρ/8:=le_trans (min_le_right _ _) (min_le_left _ _)
  42   have hpstark:pstar≤ρ/(6*c):=le_trans (min_le_right _ _) (min_le_right _ _)
  43   have hsize':ρ*D/2≤(Fintype.card U:ℝ):=by nlinarith [hsize]
  44   have hfamily:=long_preparation_family_fractions hn F hF K hK θ hθ0 hlong
  45   have hf:∀ i u,0≤f i u:=by simpa [fraction,retained,f] using hfamily.2.2.1
  46   have hmarg:=long_preparation_factor_family_actual_marginals hn F hF K hK m d hm hd θ ρ D hθ0 hθ hρ hρ1 hD hsize hcut hlong
  47   have hbounds:∀ u,1/(3*c)≤a u ∧ 1/4≤b u ∧ b u≤1:=by
  48     intro u
  49     have hh:=hmarg.1 u
  50     exact ⟨hh.1,hh.2.1,hh.2.2.1⟩
  51   have ha:∀ u,1/(3*c)≤a u:=fun u=>(hbounds u).1
  52   have hb:∀ u,1/4≤b u ∧ b u≤1:=fun u=>(hbounds u).2
  53   have hdef:∀ u,b u=(m:ℝ)*a u+(1-θ)*∑ i,f i u:=by intro u;rfl
  54   have hh:=long_preparation_concrete_marginals D ρ c (m:ℝ) θ pstar hD hρ hc hm0 (by linarith)
  55     hpstar0 hpstarb hpstarp hpstark a b f ha hb hf hsize' hdef
  56   exact ⟨hpstar0,by simpa [η,p,k] using hh⟩
