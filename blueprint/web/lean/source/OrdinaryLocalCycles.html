  17 abbrev LocalOrdinaryCycle (Q F : SimpleGraph V) (T : Finset V) (r : ℕ) :=
  18   {C : OrdinaryCycle Q r // (∀ v∈C.val.val.support,v∈T) ∧
  19     ∀ v∈C.val.val.support,∀ w∈C.val.val.support,¬F.Adj v w}

  27 theorem ordinary_cycle_from_embedding (Q F : SimpleGraph V) (T : Finset V) (hr : 3≤r)
  28     (f : Fin r ↪ V) (hT : ∀ i,f i∈T) (hF : ∀ i j,¬F.Adj (f i) (f j))
  29     (hQ : ∀ i j,(SimpleGraph.cycleGraph r).Adj i j → Q.Adj (f i) (f j)) :
  30     ∃ C : LocalOrdinaryCycle Q F T r,localOrdinaryVertices C=Finset.univ.image f := by
  31   let G := (SimpleGraph.cycleGraph r).map f
  32   have hG : G≤Q := by
  33     intro x y h
  34     obtain ⟨i,j,hij,rfl,rfl⟩ := (SimpleGraph.map_adj f _ _ _).mp h
  35     exact hQ i j hij
  36   have hs : G.support=Set.range f := cycle_map_support hr f
  37   let C : LocalOrdinaryCycle Q F T r := ⟨⟨⟨G,⟨hr,⟨f,rfl⟩⟩⟩,hG⟩,by
  38     constructor
  39     · intro v hv
  40       obtain ⟨i,rfl⟩ := hs ▸ hv
  41       exact hT i
  42     · intro v hv w hw
  43       obtain ⟨i,rfl⟩ := hs ▸ hv
  44       obtain ⟨j,rfl⟩ := hs ▸ hw
  45       exact hF i j⟩
  46   refine ⟨C,?_⟩
  47   ext v
  48   simp only [localOrdinaryVertices,C,Set.mem_toFinset,hs,Set.mem_range,Finset.mem_image,Finset.mem_univ,true_and]
