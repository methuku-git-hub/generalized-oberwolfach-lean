  16 theorem marked_packings_disjoint_of_rims_and_hubs (hm : 1≤m) (hb : 1≤b)
  17     (M N : Finset (MarkedWheel tail head ktail khead m π b H₀ HK))
  18     (hrim : Disjoint (markedPackingRimGraph M) (markedPackingRimGraph N))
  19     (hhubs : ∀ W∈M,∀ Z∈N,W.2.val≠Z.2.val) :
  20     Disjoint (M.biUnion markedWheelPhysicalResources) (N.biUnion markedWheelPhysicalResources) := by
  21   apply Finset.disjoint_left.mpr
  22   intro e he hf
  23   obtain ⟨W,hW,he⟩ := Finset.mem_biUnion.mp he
  24   obtain ⟨Z,hZ,hf⟩ := Finset.mem_biUnion.mp hf
  25   cases e with
  26   | inr p=>
  27     exact hhubs W hW Z hZ (((markedWheelPhysicalResources_spoke W p.1 p.2).mp he).1.trans
  28       ((markedWheelPhysicalResources_spoke Z p.1 p.2).mp hf).1.symm)
  29   | inl e=>
  30     have he' := (markedWheelPhysicalResources_rim W e).mp he
  31     have hf' := (markedWheelPhysicalResources_rim Z e).mp hf
  32     have hcard : e.card=2 := by
  33       obtain ⟨a,_,ha⟩ := Finset.mem_image.mp he'
  34       rw [←ha]
  35       exact markedCycle_physical_edge_card tail head ktail khead m π b hm hb W.1 a
  36     obtain ⟨x,y,hxy,rfl⟩ := Finset.card_eq_two.mp hcard
  37     exact disjoint_graphs_no_common_edge _ _ hrim x y
  38       ⟨hxy,Finset.mem_biUnion.mpr ⟨W,hW,he'⟩⟩ ⟨hxy,Finset.mem_biUnion.mpr ⟨Z,hZ,hf'⟩⟩

  40 theorem marked_hub_row_stages_combine (hm : 1≤m) (hb : 1≤b)
  41     (M : Finset (MarkedWheel tail head ktail khead m π b H₀ HK)) (Z : Finset U)
  42     (hM : ∀ W∈M,∀ X∈M,W≠X→Disjoint (markedWheelPhysicalResources W) (markedWheelPhysicalResources X))
  43     (hMhub : ∀ W∈M,W.2.val∈Z)
  44     (C : Fin n→Finset (MarkedWheel tail head ktail khead m π b H₀ HK))
  45     (v : Fin n→V) (F : Finset V) (required : Fin n→Finset (Finset V⊕(V×U))) (Y : Fin n→Finset U)
  46     (hC : ∀ i,MarkedRowCover (C i) (v i) F (required i) (Y i) K)
  47     (hcross : ∀ i j,i≠j→Disjoint ((C i).biUnion markedWheelPhysicalResources) ((C j).biUnion markedWheelPhysicalResources))
  48     (hR : ∀ i,Disjoint (markedPackingRimGraph (C i)) (markedPackingRimGraph M))
  49     (hY : ∀ i,Disjoint (Y i) Z) :
  50     let N := M∪Finset.univ.biUnion C
  51     (∀ W∈N,∀ X∈N,W≠X→Disjoint (markedWheelPhysicalResources W) (markedWheelPhysicalResources X)) ∧
  52     M.biUnion markedWheelPhysicalResources ∪ Finset.univ.biUnion required⊆N.biUnion markedWheelPhysicalResources := by
  53   let N := M∪Finset.univ.biUnion C
  54   have hMC (i) : Disjoint ((C i).biUnion markedWheelPhysicalResources) (M.biUnion markedWheelPhysicalResources) :=
  55     marked_packings_disjoint_of_rims_and_hubs hm hb (C i) M (hR i)
  56       (fun W hW X hX he=>Finset.disjoint_left.mp (hY i) ((hC i).hubs W hW) (he ▸ hMhub X hX))
  57   refine ⟨?_,?_⟩
  58   · intro W hW X hX hne
  59     rcases Finset.mem_union.mp hW with hW|hW <;> rcases Finset.mem_union.mp hX with hX|hX
  60     · exact hM W hW X hX hne
  61     · obtain ⟨j,_,hX⟩ := Finset.mem_biUnion.mp hX
  62       exact (hMC j).symm.mono (resource_subset_union _ M W hW) (resource_subset_union _ (C j) X hX)
  63     · obtain ⟨i,_,hW⟩ := Finset.mem_biUnion.mp hW
  64       exact (hMC i).mono (resource_subset_union _ (C i) W hW) (resource_subset_union _ M X hX)
  65     · obtain ⟨i,_,hW⟩ := Finset.mem_biUnion.mp hW
  66       obtain ⟨j,_,hX⟩ := Finset.mem_biUnion.mp hX
  67       by_cases hij : i=j
  68       · subst j
  69         exact (hC i).disjoint W hW X hX hne
  70       · exact (hcross i j hij).mono (resource_subset_union _ (C i) W hW) (resource_subset_union _ (C j) X hX)
  71   · intro e he
  72     rcases Finset.mem_union.mp he with he|he
  73     · obtain ⟨W,hW,he⟩ := Finset.mem_biUnion.mp he
  74       exact Finset.mem_biUnion.mpr ⟨W,Finset.mem_union_left _ hW,he⟩
  75     · obtain ⟨i,_,hi⟩ := Finset.mem_biUnion.mp he
  76       obtain ⟨W,hW,he⟩ := Finset.mem_biUnion.mp ((hC i).covers hi)
  77       exact Finset.mem_biUnion.mpr ⟨W,Finset.mem_union_right _ (Finset.mem_biUnion.mpr ⟨i,Finset.mem_univ _,hW⟩),he⟩
