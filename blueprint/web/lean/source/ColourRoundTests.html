  15 def realColourClass {V E C : Type*}
  16     (R : E → Finset V) (L : E → Finset C) (H : Finset E) (v : V) (c : C) : Finset E :=
  17   H.filter (fun e=>v∈R e ∧ c∈L e)

  19 abbrev ColourRoundTest (V C : Type*) {E : Type*} (H : Finset E) :=
  20   {e // e∈H} ⊕ (V×C)
  21 
  22 def colourRoundBase {V E C : Type*} (R : E → Finset V) (L : E → Finset C) (H : Finset E) :
  23     ColourRoundTest V C H → Finset E
  24   | .inl e => {e.val}
  25   | .inr p => realColourClass R L H p.1 p.2

  27 theorem realColourClass_subset {V E C : Type*} [Fintype E]
  28     (R : E → Finset V) (L : E → Finset C) (H : Finset E) (v : V) (c : C) :
  29     realColourClass R L H v c⊆hypergraphColourClique R L v c := by
  30   intro e he
  31   exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,(Finset.mem_filter.mp he).2⟩

  33 theorem colourRoundBase_card {V E C : Type*} [Fintype E]
  34     (R : E → Finset V) (L : E → Finset C) (H : Finset E) {s : ℕ} (hs : 1≤s)
  35     (hdegree : ∀ v c,(hypergraphColourClique R L v c).card≤s)
  36     (i : ColourRoundTest V C H) : (colourRoundBase R L H i).card≤s := by
  37   cases i with
  38   | inl e => simpa only [colourRoundBase,Finset.card_singleton] using hs
  39   | inr p => exact (Finset.card_le_card (realColourClass_subset R L H p.1 p.2)).trans (hdegree p.1 p.2)

  41 theorem colourRoundBase_incidence
  42     {V E C : Type*} [Fintype V] [Fintype C]
  43     (R : E → Finset V) (L : E → Finset C) (H : Finset E) {k l : ℕ}
  44     (hsize : ∀ e,(R e).card≤k) (hcard : ∀ e,(L e).card=l) (e : E) :
  45     (Finset.univ.filter (fun i : ColourRoundTest V C H=>e∈colourRoundBase R L H i)).card≤1+k*l := by
  46   classical
  47   have hleft : (∑ f : {f // f∈H},if e=f.val then (1 : ℕ) else 0)≤1 := by
  48     by_cases he : e∈H
  49     · have hh (f : {f // f∈H}) : e=f.val ↔ f=⟨e,he⟩ := by
  50         constructor
  51         · intro h;exact Subtype.ext h.symm
  52         · intro h;exact (congrArg Subtype.val h).symm
  53       simp_rw [hh]
  54       simp
  55     · have hh (f : {f // f∈H}) : e≠f.val := fun h=>he (h.symm ▸ f.property)
  56       simp only [hh,ite_false,Finset.sum_const_zero,Nat.zero_le]
  57   have hright : (∑ p : V×C,if e∈realColourClass R L H p.1 p.2 then (1 : ℕ) else 0)≤k*l := by
  58     rw [Fintype.sum_prod_type]
  59     by_cases he : e∈H
  60     · simp only [realColourClass,Finset.mem_filter,he,true_and,ite_and]
  61       simp_rw [Finset.sum_ite_irrel,Finset.sum_const_zero]
  62       have hcount : (∑ v : V,if v∈R e then ∑ c : C,if c∈L e then (1 : ℕ) else 0 else 0)=
  63           (R e).card*(L e).card := by
  64         simp
  65       rw [hcount,hcard]
  66       exact Nat.mul_le_mul_right _ (hsize e)
  67     · simp only [realColourClass,Finset.mem_filter,he,false_and,ite_false,Finset.sum_const_zero]
  68       exact Nat.zero_le _
  69   simp only [Finset.card_eq_sum_ones,Finset.sum_filter]
  70   rw [Fintype.sum_sum_type]
  71   simp only [colourRoundBase,Finset.mem_singleton]
  72   exact add_le_add hleft hright
