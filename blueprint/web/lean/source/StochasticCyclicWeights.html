  14 theorem stochastic_cyclic_weights {Ω X : Type*} [Fintype Ω] [Fintype X] [Nonempty X]
  15     {k : ℕ} (μ : FiniteLaw Ω)
  16     (M : (Fin (k + 2) ⊕ Fin (k + 2)) → X → X → ℕ)
  17     (hM : ∀ i x y, M i x y ≤ 1)
  18     (R : Ω → (Fin (k + 2) ⊕ Fin (k + 2)) → X → X → ℕ)
  19     (hsub : ∀ ω i x y, R ω i x y ≤ M i x y)
  20     (q : ℕ) (hrow : ∀ ω i x, ∑ y, R ω i x y = q)
  21     (hcol : ∀ ω i y, ∑ x, R ω i x y = q)
  22     (p α : ℝ) (hα : 0 < α)
  23     (hprob : ∀ i x y, 0 < M i x y → μ.probability (fun ω => R ω i x y = 0) ≤ p)
  24     (hthreshold : ((k + 2) + (2 * (k + 2) - 1) * α) * Fintype.card X ≤
  25       (2 * (k + 2) - 1) * q) :
  26     ∃ w : (Fin (k + 2) → X × X) → ℝ,
  27       (∀ v, 0 ≤ w v) ∧
  28       (∀ v, 0 < w v → CyclicMatrixTupleValid M v) ∧
  29       (∀ i x y, (1 - p) * M (Sum.inl i) x y ≤
  30         (∑ v, if v i = (x,y) then w v else 0) ∧
  31         (∑ v, if v i = (x,y) then w v else 0) ≤ M (Sum.inl i) x y) ∧
  32       (∀ i y z, (1 - p) * M (Sum.inr i) y z ≤
  33         (∑ v, if ((v i).2, (v (finRotate (k + 2) i)).1) = (y,z) then w v else 0) ∧
  34         (∑ v, if ((v i).2, (v (finRotate (k + 2) i)).1) = (y,z) then w v else 0) ≤
  35           M (Sum.inr i) y z) ∧
  36       (∀ i x, (∑ v, if (v i).1 = x then w v else 0) = q ∧
  37         (∑ v, if (v i).2 = x then w v else 0) = q) ∧
  38       (∀ v, w v ≤ ((k + 2 : ℝ) * cyclicSectorUpper k α) /
  39         (Fintype.card X : ℝ) ^ (2 * k + 2)) ∧
  40       (∀ v, CyclicMatrixTupleValid M v →
  41         (1 - (2 * (k + 2 : ℝ)) * p) *
  42           (((k + 2 : ℝ) * cyclicSectorLower k α) /
  43             (Fintype.card X : ℝ) ^ (2 * k + 2)) ≤ w v) := by
  44   have hex (ω : Ω) := regular_cyclic_matrix_weights (R ω)
  45     (fun i x y => (hsub ω i x y).trans (hM i x y)) q (hrow ω) (hcol ω) α hα hthreshold
  46   choose W hW hs hl hr hv hb hlo using hex
  47   let w := averagedWeight μ W 1
  48   have hnonneg (v) : 0 ≤ w v := averagedWeight_nonneg μ W (by norm_num) hW v
  49   have hsupport (v) (hp : 0 < w v) : CyclicMatrixTupleValid M v := by
  50     by_contra hn
  51     have hz (ω) : W ω v = 0 := by
  52       by_contra hne
  53       have hvalid := hs ω v (lt_of_le_of_ne (hW ω v) (Ne.symm hne))
  54       apply hn
  55       intro i
  56       exact ⟨(hvalid i).1.trans_le (hsub ω _ _ _), (hvalid i).2.trans_le (hsub ω _ _ _)⟩
  57     have hzero : w v = 0 := by simp [w, averagedWeight, hz, finiteLaw_expectation_const]
  58     linarith
  59   refine ⟨w, hnonneg, hsupport, ?_, ?_, ?_, ?_, ?_⟩
  60   · intro i x y
  61     have he := averagedWeight_incidence μ W 1 (fun v => v i = (x,y))
  62     simp only [div_one] at he
  63     have he' : (∑ v, if v i = (x,y) then w v else 0) =
  64         μ.expectation (fun ω => (R ω (Sum.inl i) x y : ℝ)) := by
  65       convert he using 1
  66       · apply Finset.sum_congr rfl
  67         intro v _
  68         split_ifs <;> rfl
  69       · congr 1
  70         funext ω
  71         symm
  72         convert hl ω i x y using 1
  73         apply Finset.sum_congr rfl
  74         intro v _
  75         split_ifs <;> rfl
  76     rw [he']
  77     exact finiteLaw_retained_binary_mean μ _ (hM _ _ _) _
  78       (fun ω => hsub ω _ _ _) p (hprob _ _ _)
  79   · intro i y z

  81       (fun v => ((v i).2, (v (finRotate (k + 2) i)).1) = (y,z))
  82     simp only [div_one] at he
  83     have he' : (∑ v, if ((v i).2, (v (finRotate (k + 2) i)).1) = (y,z) then w v else 0) =
  84         μ.expectation (fun ω => (R ω (Sum.inr i) y z : ℝ)) := by
  85       convert he using 1

 104       simpa only [fun ω => (hv ω i x).2, finiteLaw_expectation_const, div_one] using he
