  15 theorem ordinary_expansion_host_family (β : ℝ) (hβ : 0<β) (r q N : ℕ) (hr : 3≤r) :
  16     ∃ n₀ : ℕ,∀ n : ℕ,n₀≤n →
  17     ∀ (V U E J : Type*) [Fintype V] [Fintype U] [Fintype E] [Fintype J]
  18       [DecidableEq V] [DecidableEq U],Fintype.card V=n → Fintype.card (E⊕J)≤N →
  19     ∀ (G : SimpleGraph V) (H : V → U → Prop),
  20     (∀ a,(1/2+β)*(n:ℝ)≤G.degree a) →
  21     (∀ a,(1-1/(2*(r:ℝ)))*(Fintype.card U:ℝ)≤(Finset.univ.filter (H a)).card) →
  22     (∀ u,((Finset.univ.filter (fun v => ¬H v u)).card:ℝ)≤(β/2)*n) →
  23     (∀ u,((Finset.univ.filter (fun v => ¬H v u)).card:ℝ)≤(n:ℝ)/4) →
  24     ∀ (R : Finset V) (B : Finset U) (x y : E → V) (v : J → V) (u : J → U),
  25       R.card≤q → (∀ e,x e∈R ∧ y e∈R) → (∀ j,v j∈R ∧ u j∈B) →
  26       2*(B.card+N)<Fintype.card U →
  27     ∃ (f : E⊕J → Fin r ↪ V) (a : E⊕J → U),
  28       (∀ i j k,(SimpleGraph.cycleGraph r).Adj j k → G.Adj (f i j) (f i k)) ∧
  29       (∀ i j,H (f i j) (a i)) ∧ (∀ i j,f i j∉R) ∧ (∀ i,a i∉B) ∧
  30       (∀ i i',i≠i' → a i≠a i' ∧ ∀ j k,f i j≠f i' k) ∧
  31       (∀ e,G.Adj (y e) (f (.inl e) ⟨0,by omega⟩) ∧ G.Adj (x e) (f (.inl e) ⟨r-1,by omega⟩)) ∧
  32       ∀ j,H (v j) (a (.inr j)) ∧ H (f (.inr j) ⟨0,by omega⟩) (u j) := by
  33   obtain ⟨n₁,hrim⟩ := ordinary_rim_expansion_embedding β hβ r (q+N*r) hr
  34   obtain ⟨n₂,hspoke⟩ := ordinary_spoke_expansion_embedding β hβ r (q+N*r) hr
  35   refine ⟨max n₁ n₂,?_⟩
  36   intro n hn V U E J _ _ _ _ _ _ hV hN G H hdeg hrow hcol hquarter R B x y v u hR hxy hvu hB
  37   have hrow' : ∀ a,(1/2:ℝ)*(Fintype.card U:ℝ)≤(Finset.univ.filter (H a)).card := by
  38     intro a
  39     have hrR : (1:ℝ)≤r := by exact_mod_cast (show 1≤r by omega)
  40     have hfrac : 1/(2*(r:ℝ))≤1/2 := by apply (div_le_iff₀ (by positivity)).mpr;nlinarith
  41     have hh := mul_le_mul_of_nonneg_right (show (1/2:ℝ)≤1-1/(2*(r:ℝ)) by linarith)
  42       (Nat.cast_nonneg (Fintype.card U))
  43     exact hh.trans (hrow a)
  44   let P : E⊕J → (Fin r ↪ V) → U → Prop := fun i f a =>
  45     (∀ j k,(SimpleGraph.cycleGraph r).Adj j k → G.Adj (f j) (f k)) ∧
  46     (∀ j,H (f j) a) ∧
  47     Sum.elim (fun e => G.Adj (y e) (f ⟨0,by omega⟩) ∧ G.Adj (x e) (f ⟨r-1,by omega⟩))
  48       (fun j => H (v j) a ∧ H (f ⟨0,by omega⟩) (u j)) i
  49   obtain ⟨f,a,hf,hsep⟩ := disjoint_labelled_gadgets_finite r R B P (by
  50     intro i S T hRS hBT hS hT
  51     have hSq : S.card≤q+N*r := by nlinarith
  52     have hTb : 2*T.card<Fintype.card U := by omega
  53     cases i with
  54     | inl e =>
  55       obtain ⟨g,b,hg,hH,havoid,hb,hy,hx,_⟩ := hrim n ((le_max_left _ _).trans hn)
  56         V U hV.le G H hdeg hrow S T (x e) (y e) hSq (hRS (hxy e).1) (hRS (hxy e).2) hTb
  57       exact ⟨g,b,⟨hg,hH,hy,hx⟩,havoid,hb⟩
  58     | inr j =>
  59       obtain ⟨g,b,hg,hH,havoid,hb,hv,hu⟩ := hspoke n ((le_max_right _ _).trans hn)
  60         V U hV G H hdeg hrow' hcol hquarter S T (v j) (u j) hSq
  61         (hRS (hvu j).1) (hBT (hvu j).2) hTb
  62       exact ⟨g,b,⟨hg,hH,hv,hu⟩,havoid,hb⟩)
  63   exact ⟨f,a,(fun i => (hf i).1.1),(fun i => (hf i).1.2.1),
  64     (fun i => (hf i).2.1),(fun i => (hf i).2.2),hsep,
  65     (fun e => (hf (.inl e)).1.2.2),(fun j => (hf (.inr j)).1.2.2)⟩
