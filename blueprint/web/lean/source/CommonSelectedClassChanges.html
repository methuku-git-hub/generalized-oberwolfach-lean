  52 theorem commonSelectedClass_overlap_change (π : Equiv.Perm V)
  53     (H H' : Bool→V→U→Prop) (κ : V→Bool) (u u' : U) (δ : ℝ)
  54     (hc : ∀ b z,(∑ x,if ¬(H b x z↔H' b x z) then (1:ℝ) else 0)≤δ)
  55     (hdis : Disjoint (commonSelectedClass H κ u u' false) ((commonSelectedClass H κ u u' true).image π)) :
  56     (((commonSelectedClass H' κ u u' false)∩(commonSelectedClass H' κ u u' true).image π).card:ℝ)≤4*δ := by
  57   let T := commonSelectedClass H κ u u'
  58   let T' := commonSelectedClass H' κ u u'
  59   have hsub : T' false∩(T' true).image π⊆changed (T false) (T' false)∪(changed (T true) (T' true)).image π := by
  60     intro x hx
  61     obtain ⟨h₀,hh⟩ := Finset.mem_inter.mp hx
  62     obtain ⟨y,hy,rfl⟩ := Finset.mem_image.mp hh
  63     by_cases hOld₀ : π y∈T false
  64     · by_cases hOldK : y∈T true
  65       · exact False.elim (Finset.disjoint_left.mp hdis hOld₀ (Finset.mem_image.mpr ⟨y,hOldK,rfl⟩))
  66       · apply Finset.mem_union_right
  67         apply Finset.mem_image.mpr
  68         exact ⟨y,by simp [changed,hy,hOldK],rfl⟩
  69     · apply Finset.mem_union_left
  70       simp [changed,h₀,hOld₀]
  71   have hb := (Finset.card_le_card hsub).trans (Finset.card_union_le _ _)
  72   have him := Finset.card_image_le (s:=changed (T true) (T' true)) (f:=π)
  73   have hbR : ((T' false∩(T' true).image π).card:ℝ)≤(changed (T false) (T' false)).card+(changed (T true) (T' true)).card := by
  74     exact_mod_cast hb.trans (Nat.add_le_add_left him _)
