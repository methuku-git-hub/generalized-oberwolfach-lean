  14 theorem uniform_resources_between_count (R : C → Finset V) (c J : ℕ)
  15     (hR : ∀ C₀,(R C₀).card=c) (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J)
  16     (S Y : Finset V) (hSY : S⊆Y) (hS : S.card ≤ c) :
  17     Fintype.card {C₀ : C // S⊆R C₀ ∧ R C₀⊆Y} ≤ J*Y.card^(c-S.card) := by
  18   obtain ⟨code,hcode⟩ := finite_fibre_codes R J hJ
  19   let B := {U : Finset V // U∈(Y.powersetCard c).filter (S⊆·)}
  20   let enc (D : {C₀ : C // S⊆R C₀ ∧ R C₀⊆Y}) : B×Fin J :=
  21     (⟨R D.val,Finset.mem_filter.mpr ⟨Finset.mem_powersetCard.mpr ⟨D.property.2,hR D.val⟩,D.property.1⟩⟩,
  22       code D.val)
  23   have hi : Function.Injective enc := by
  24     intro D E he
  25     apply Subtype.ext
  26     exact hcode D.val E.val (congrArg (fun z : B×Fin J => z.1.val) he) (congrArg Prod.snd he)
  27   have hb : Fintype.card B ≤ Y.card^(c-S.card) := by
  28     rw [Fintype.card_coe,Finset.card_filter_powersetCard_subset S Y c hSY hS]
  29     exact (Nat.choose_le_pow _ _).trans (Nat.pow_le_pow_left (Nat.sub_le _ _) _)
  30   have hh := Fintype.card_le_of_injective enc hi
  31   rw [Fintype.card_prod,Fintype.card_fin] at hh
  32   exact hh.trans (by simpa only [Nat.mul_comm] using Nat.mul_le_mul_right J hb)

  34 theorem uniform_resources_local_pair_count (R : C → Finset V) (c J : ℕ) (hc : 2 ≤ c)
  35     (hR : ∀ C₀,(R C₀).card=c) (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J)
  36     (Y : Finset V) (u v : V) (hu : u∈Y) (hv : v∈Y) (huv : u≠v) :
  37     Fintype.card {C₀ : C // u∈R C₀ ∧ v∈R C₀ ∧ R C₀⊆Y} ≤ J*Y.card^(c-2) := by
  38   have hcard : ({u,v}:Finset V).card=2 := by simp [huv]
  39   have hh := uniform_resources_between_count R c J hR hJ {u,v} Y
  40     (by
  41       intro x hx
  42       rcases Finset.mem_insert.mp hx with rfl | hx
  43       · exact hu
  44       · have he := Finset.mem_singleton.mp hx
  45         subst x
  46         exact hv) (by simpa only [hcard] using hc)
  47   simpa only [hcard,Finset.insert_subset_iff,Finset.singleton_subset_iff,and_assoc] using hh
