   5 /-! One simultaneous tentative-colouring round. A vertex keeps its colour
   6 exactly when none of its neighbours chose that colour. -/

  16 def tentativeColourSurvives (G : SimpleGraph V) (σ : V → C) (v : V) : Prop :=
  17   ∀ u,G.Adj v u → σ u≠σ v
  18 
  19 def retainedColour (G : SimpleGraph V) (σ : V → C) (v : V) : Option C :=
  20   if tentativeColourSurvives G σ v then some (σ v) else none
  21 
  22 theorem retainedColour_eq_some (G : SimpleGraph V) (σ : V → C) (v : V) (c : C) :
  23     retainedColour G σ v=some c ↔ σ v=c ∧ tentativeColourSurvives G σ v := by
  24   unfold retainedColour
  25   split_ifs <;> simp_all
  26 
  27 theorem retainedColour_none (G : SimpleGraph V) (σ : V → C) (v : V) :
  28     retainedColour G σ v=none ↔ ∃ u,G.Adj v u ∧ σ u=σ v := by
  29   simp [retainedColour,tentativeColourSurvives]

  31 theorem retainedColour_proper (G : SimpleGraph V) (σ : V → C)
  32     {u v : V} (h : G.Adj u v) (c : C)
  33     (hu : retainedColour G σ u=some c) : retainedColour G σ v≠some c := by
  34   intro hv
  35   obtain ⟨huc,hus⟩ := (retainedColour_eq_some G σ u c).mp hu
  36   obtain ⟨hvc,_⟩ := (retainedColour_eq_some G σ v c).mp hv
  37   exact hus v h (hvc.trans huc.symm)

  39 theorem tentativeColour_probability [Fintype V] [DecidableEq V] [Fintype C]
  40     (G : SimpleGraph V) (μ : V → FiniteLaw C) (v : V) (c : C) :
  41     (FiniteLaw.pi μ).probability
  42       (fun σ=>σ v=c ∧ tentativeColourSurvives G σ v)=
  43       (μ v).mass c*∏ u∈Finset.univ.filter (fun u=>G.Adj v u),(1-(μ u).mass c) := by
  44   classical
  45   let A : V → C → Prop := fun u a=>
  46     if u=v then a=c else if G.Adj v u then a≠c else True
  47   have hevent (σ : V → C) :
  48       (σ v=c ∧ tentativeColourSurvives G σ v) ↔ ∀ u,A u (σ u) := by
  49     constructor
  50     · rintro ⟨hroot,hkeep⟩ u
  51       by_cases hu : u=v
  52       · simp [A,hu,hroot]
  53       · by_cases ha : G.Adj v u
  54         · simpa [A,hu,ha,hroot] using hkeep u ha
  55         · simp [A,hu,ha]
  56     · intro h
  57       have hroot : σ v=c := by simpa [A] using h v
  58       refine ⟨hroot,?_⟩
  59       intro u hu
  60       have huv : u≠v := fun he=>G.irrefl (he ▸ hu)
  61       simpa [A,huv,hu,hroot] using h u
  62   rw [finiteLaw_probability_congr (FiniteLaw.pi μ) _ _ hevent,
  63     finiteLaw_pi_probability_forall μ A]
  64   have hcoord (u : V) :
  65       (μ u).probability (A u)=
  66         if u=v then (μ v).mass c else if G.Adj v u then 1-(μ u).mass c else 1 := by
  67     by_cases hu : u=v
  68     · subst u
  69       simp only [A,ite_true]
  70       exact finiteLaw_probability_point _ _
  71     · by_cases ha : G.Adj v u
  72       · simp only [A,hu,ite_false,ha,ite_true]
  73         exact finiteLaw_probability_not_point _ _
  74       · simp only [A,hu,ite_false,ha]
  75         exact finiteLaw_probability_true _
  76   simp_rw [hcoord]
  77   rw [← Finset.mul_prod_erase Finset.univ
  78     (fun u=>if u=v then (μ v).mass c else if G.Adj v u then 1-(μ u).mass c else 1)
  79     (Finset.mem_univ v)]
  80   simp only [ite_true]
  81   congr 1
  82   have he : (Finset.univ.erase v).filter (fun u=>G.Adj v u)=
  83       Finset.univ.filter (fun u=>G.Adj v u) := by
  84     ext u
  85     simp only [Finset.mem_filter,Finset.mem_erase,Finset.mem_univ,and_true,true_and]
  86     constructor
  87     · exact fun h=>h.2
  88     · intro h
  89       exact ⟨fun he=>G.irrefl (he ▸ h),h⟩
  90   calc
  91     _ = ∏ u∈Finset.univ.erase v,if G.Adj v u then 1-(μ u).mass c else 1 := by
  92       apply Finset.prod_congr rfl
  93       intro u hu
  94       simp only [(Finset.mem_erase.mp hu).1,ite_false]
  95     _ = ∏ u∈(Finset.univ.erase v).filter (fun u=>G.Adj v u),(1-(μ u).mass c) := by
  96       rw [Finset.prod_filter]
  97     _ = _ := by rw [he]
