  11  theorem triangle_root_spokes_bounded {V U:Type} [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U]
  12     (H:V→U→Prop) (R:Finset V) (B:Finset U)
  13     (h:∀ v u,H v u→v∈R∧u∈B):
  14     Fintype.card {e:V×U // H e.1 e.2}≤R.card*B.card:=by
  15   let f:{e:V×U // H e.1 e.2}→{v:V // v∈R}×{u:U // u∈B}:=fun e=>
  16     (⟨e.val.1,(h e.val.1 e.val.2 e.property).1⟩,⟨e.val.2,(h e.val.1 e.val.2 e.property).2⟩)
  17   have hf:Function.Injective f:=by
  18     intro a b hab
  19     apply Subtype.ext
  20     exact Prod.ext (congrArg (fun z:({v:V // v∈R}×{u:U // u∈B})=>z.1.val) hab)
  21       (congrArg (fun z:({v:V // v∈R}×{u:U // u∈B})=>z.2.val) hab)
  22   have hh:=Fintype.card_le_of_injective f hf
  23   simpa only [Fintype.card_prod,Fintype.card_coe] using hh

  25  theorem triangle_root_complexity_bound {V U:Type} [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U]
  26     (G:SimpleGraph V) (H:V→U→Prop) (R₀:Finset V) (B₀:Finset U) (q:ℕ)
  27     (hbalance:∀ v,G.degree v=2*((Finset.univ:Finset U).filter (H v)).card)
  28     (hsupport:∀ v u,H v u→v∈R₀∧u∈B₀) (hR:R₀.card≤q) (hB:B₀.card≤q):
  29     G.edgeFinset.card≤q*q ∧ Fintype.card {e:V×U // H e.1 e.2}≤q*q ∧
  30     Fintype.card (Σ v,G.neighborSet v)≤2*(q*q):=by
  31   have ht:Fintype.card {e:V×U // H e.1 e.2}≤q*q:=
  32     (triangle_root_spokes_bounded H R₀ B₀ hsupport).trans (Nat.mul_le_mul hR hB)
  33   have hq:∀ u,((Finset.univ:Finset V).filter (fun v=>H v u)).card≤R₀.card:=by
  34     intro u
  35     apply Finset.card_le_card
  36     intro v hv
  37     exact (hsupport v u (Finset.mem_filter.mp hv).2).1
  38   have hn:Fintype.card {u:U // ∃v,H v u}≤Fintype.card U:=Fintype.card_subtype_le _
  39   obtain ⟨he,hd,_⟩:=balanced_root_edge_counts G H hbalance R₀.card (Fintype.card U) hq hn
  40   exact ⟨he.le.trans ht,ht,by rw [hd,he];exact Nat.mul_le_mul_left 2 ht⟩
  41 end Oberwolfach
