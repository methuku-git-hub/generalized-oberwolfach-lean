  22     (σ : RawMarkedOutcome V U Γ Color) : Prop :=
  23   let R := fun mark incoming x=>(markedRestrictedDegree P π (3*d) (fun e=>σ (Sum.inl e)) mark incoming x A:ℝ)
  24   let H := fun b x=>rawMarkedRestrictedSpokeRow π d (fun e=>σ (Sum.inr e)) g j b x B
