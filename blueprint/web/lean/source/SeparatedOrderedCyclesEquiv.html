  12 abbrev SeparatedHostCycle (P:SimpleGraph V) (π:Equiv.Perm V) (b r:ℕ):=
  13   {H:SimpleGraph V // IsEmbeddedCycle H r ∧ H≤P ∧
  14     ∀ x∈H.support,∀ y∈H.support,x≠y→¬PermutationClose π b x y}

  16 abbrev SeparatedHostCycleTuple (P:SimpleGraph V) (π:Equiv.Perm V) (b r:ℕ):=
  17   {C:Fin r→V // PermutationSeparated π b C ∧ ∀ i,P.Adj (C i) (C (cycleNext i))}
