  10 theorem successor_regularized_degree_lower {n m D s q β ε : ℝ}
  11     (hn : 0 < n) (hm : 0 < m)
  12     (hD : (q + β - ε) * n ≤ D)
  13     (hmargin : ε + s + 1 / m ≤ β / 2) :
  14     (q + β / 2) * m ≤ (⌊D * m / n - s * m⌋₊ : ℝ) := by
  15   have hd : (q + β - ε) * m ≤ D * m / n := by
  16     apply (le_div_iff₀ hn).mpr
  17     nlinarith [mul_le_mul_of_nonneg_right hD hm.le]
  18   have herr := mul_le_mul_of_nonneg_right hmargin hm.le
  19   have hcancel : (1 / m : ℝ) * m = 1 := by field_simp
  20   have hf := Nat.lt_floor_add_one (D * m / n - s * m)
  21   rw [add_mul, add_mul, hcancel] at herr
  22   linarith

  24 theorem successor_regularized_positional_load {n m D s : ℝ}
  25     (hn : 1 ≤ n) (hm : 0 < m) (hD : 0 ≤ D) (hDupper : D ≤ 2 * n)
  26     (hs : 0 ≤ s) (harg : 0 ≤ D * m / n - s * m) :
  27     D - s * n - n / m - 2 ≤ (⌊D * m / n - s * m⌋₊ : ℝ) * (n - 1) / m ∧
  28       (⌊D * m / n - s * m⌋₊ : ℝ) * (n - 1) / m ≤ D := by
  29   have hn0 : 0 < n := by linarith
  30   have hscale : 0 ≤ (n - 1) / m := div_nonneg (by linarith) hm.le
  31   have hlo := (Nat.lt_floor_add_one (D * m / n - s * m)).le
  32   have hhi := Nat.floor_le harg
  33   have hlow : (D * m / n - s * m - 1) * ((n - 1) / m) ≤
  34       (⌊D * m / n - s * m⌋₊ : ℝ) * ((n - 1) / m) :=
  35     mul_le_mul_of_nonneg_right (by linarith) hscale
  36   have hupper : (⌊D * m / n - s * m⌋₊ : ℝ) * ((n - 1) / m) ≤
  37       (D * m / n) * ((n - 1) / m) :=
  38     mul_le_mul_of_nonneg_right (by nlinarith [mul_nonneg hs hm.le]) hscale
  39   have hloweq : (D * m / n - s * m - 1) * ((n - 1) / m) =
  40       D - D / n - s * n + s - n / m + 1 / m := by field_simp; ring
  41   have hupeq : (D * m / n) * ((n - 1) / m) = D - D / n := by field_simp
  42   rw [hloweq, ← mul_div_assoc] at hlow
  43   rw [hupeq, ← mul_div_assoc] at hupper
  44   have hD2 : D / n ≤ 2 := (div_le_iff₀ hn0).mpr hDupper
  45   have hD0 : 0 ≤ D / n := div_nonneg hD hn0.le
  46   have hmInv : 0 ≤ (1 : ℝ) / m := by positivity
  47   constructor <;> linarith
