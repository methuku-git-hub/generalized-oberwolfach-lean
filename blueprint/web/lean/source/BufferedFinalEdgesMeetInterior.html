   9 theorem buffered_final_edges_meet_strict_interior
  10     {X I:Type} [Fintype X] [DecidableEq X] [Fintype I]
  11     (G:SimpleGraph X) (n:I→ℕ) (inner:∀i,Fin (n i+1)↪X)
  12     (u v:I→X) (p:∀i,G.Walk (u i) (v i))
  13     (hsize:∀i,2≤n i) (hsupport:∀i,(p i).support=List.ofFn (inner i)):
  14     ∀e∈Finset.univ.biUnion (fun i=>(p i).edges.toFinset),
  15       ∃x∈e.toFinset,x∈bufferedPathStrictInterior n inner :=by
  16   intro e he
  17   obtain ⟨i,_,he⟩:=Finset.mem_biUnion.mp he
  18   induction e using Sym2.ind with
  19   | _ x y =>
  20     have hlen:(p i).length=n i:=by
  21       have hh:=congrArg List.length (hsupport i)
  22       simp only [SimpleGraph.Walk.length_support,List.length_ofFn] at hh
  23       omega
  24     have hmem:s(x,y)∈(p i).edges:=by simpa using he
  25     obtain ⟨j,hj,hedge⟩:=((p i).mk_mem_edges_iff_exists).mp hmem
  26     have hjn:j<n i:=by simpa [hlen] using hj
  27     have hq:∀k:ℕ,(hk:k≤n i)→(p i).getVert k=inner i ⟨k,by omega⟩:=by
  28       intro k hk
  29       have hh:some ((p i).getVert k)=(p i).support[k]? :=
  30         (p i).getVert_eq_support_getElem? (by simpa [hlen] using hk)
  31       rw [hsupport i] at hh
  32       simpa only [List.getElem?_ofFn,dif_pos (show k<n i+1 by omega),Option.some.injEq] using hh
  33     by_cases hj0:j=0
  34     · let q:Fin (n i+1):=⟨j+1,by omega⟩
  35       refine ⟨inner i q,?_,?_⟩
  36       · have hv:(p i).getVert (j+1)∈s(x,y).toFinset:=by rw [←hedge];simp
  37         simpa only [hq (j+1) (by omega)] using hv
  38       · apply Finset.mem_biUnion.mpr
  39         refine ⟨i,Finset.mem_univ _,Finset.mem_image.mpr ⟨q,?_,rfl⟩⟩
  40         simp only [Finset.mem_filter,Finset.mem_univ,true_and,q]
  41         have :=hsize i
  42         omega
  43     · let q:Fin (n i+1):=⟨j,by omega⟩
  44       refine ⟨inner i q,?_,?_⟩
  45       · have hv:(p i).getVert j∈s(x,y).toFinset:=by rw [←hedge];simp
  46         simpa only [hq j (by omega)] using hv
  47       · apply Finset.mem_biUnion.mpr
  48         refine ⟨i,Finset.mem_univ _,Finset.mem_image.mpr ⟨q,?_,rfl⟩⟩
  49         simp only [Finset.mem_filter,Finset.mem_univ,true_and,q]
