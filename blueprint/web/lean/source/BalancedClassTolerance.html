  11  theorem balanced_class_tolerances (m:ℕ) (q β η K:ℝ)
  12     (hq:0<q) (hβ:0<β) (hη:0<η) (hK:0≤K):
  13     ∃ ε δ:ℝ,0<ε∧0<δ∧ε+δ≤q/2∧ε+K*δ≤η∧(m+1)*δ≤β/2∧
  14       ∀ f:ℝ,0≤f→∃ N:ℕ,1≤N∧∀ n:ℕ,N≤n→
  15         4*(m+1)*ε*n+2*(m+1)*f+(m+1)≤δ*n:=by
  16   let c:ℝ:=m+1
  17   have hc:0<c:=by dsimp [c];positivity
  18   let δ:ℝ:=min (q/4) (min (η/(2*(K+1))) (β/(2*c)))
  19   have hδ:0<δ:=by dsimp [δ];positivity
  20   have hδq:δ≤q/4:=min_le_left _ _
  21   have hδη:δ≤η/(2*(K+1)):=(min_le_right _ _).trans (min_le_left _ _)
  22   have hδβ:δ≤β/(2*c):=(min_le_right _ _).trans (min_le_right _ _)
  23   let ε:ℝ:=min (δ/(8*c)) (min (q/4) (η/2))
  24   have hε:0<ε:=by dsimp [ε];positivity
  25   have hεδ:ε≤δ/(8*c):=min_le_left _ _
  26   have hεq:ε≤q/4:=(min_le_right _ _).trans (min_le_left _ _)
  27   have hεη:ε≤η/2:=(min_le_right _ _).trans (min_le_right _ _)
  28   have hεc:4*c*ε≤δ/2:=by nlinarith only [(le_div_iff₀ (by positivity:0<8*c)).mp hεδ]
  29   have hKδ:K*δ≤η/2:=by
  30     have hh:δ*(2*(K+1))≤η:=(le_div_iff₀ (by positivity)).mp hδη
  31     nlinarith only [hh,hδ.le]
  32   refine ⟨ε,δ,hε,hδ,by linarith,by linarith,?_,?_⟩
  33   · have hh:δ*(2*c)≤β:=(le_div_iff₀ (by positivity)).mp hδβ
  34     dsimp [c] at hh
  35     nlinarith only [hh]
  36   · intro f hf
  37     obtain ⟨N,hN⟩:=exists_nat_gt (max 1 (2*(2*c*f+c)/δ))
  38     have hN1:1≤N:=by have hh: (1:ℝ)<N:=(le_max_left _ _).trans_lt hN;exact_mod_cast hh.le
  39     refine ⟨N,hN1,?_⟩
