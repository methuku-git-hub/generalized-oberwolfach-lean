  18       (∀ v,(allowedThreshold Allowed+α)*Fintype.card V≤(G.degree v:ℝ))→

  21       (∀ i x y,(∑ C,if C.val.val.Adj x y then w i C else 0)=if G.Adj x y then 1 else 0) ∧
  22       (∀ i:{i:Fin K // i∈S},5*((n i.val+2:ℕ):ℝ)^4*B i.val*(2*(3*d))/Fintype.card V≤τ ∧
