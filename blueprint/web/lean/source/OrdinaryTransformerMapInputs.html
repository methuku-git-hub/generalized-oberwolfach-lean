  14 theorem ordinary_resource_map_rim_injective (G : SimpleGraph V) [DecidableRel G.Adj] (H : V → U → Prop) (f : V → V) (g : U → U)
  15     (h : Set.InjOn (ordinaryPairResourceMap f g) (ordinaryHostResources G H)) :
  16     Function.Injective (fun i:G.edgeFinset => s(f i.val.out.1,f i.val.out.2)) := by
  17   intro i j he
  18   have hmap (a : Sym2 V) : Sym2.map f a=s(f a.out.1,f a.out.2) := by
  19     calc
  20       _ = Sym2.map f s(a.out.1,a.out.2) := congrArg (Sym2.map f) a.out_eq.symm
  21       _ = _ := rfl
  22   have hi : Sum.inl i.val∈ordinaryHostResources G H := (ordinaryHostResources_rim _ _ _).mpr
  23     (by simpa only [SimpleGraph.mem_edgeFinset] using i.property)
  24   have hj : Sum.inl j.val∈ordinaryHostResources G H := (ordinaryHostResources_rim _ _ _).mpr
  25     (by simpa only [SimpleGraph.mem_edgeFinset] using j.property)
  26   have he' : ordinaryPairResourceMap f g (.inl i.val)=ordinaryPairResourceMap f g (.inl j.val) := by
  27     change Sum.inl (Sym2.map f i.val)=Sum.inl (Sym2.map f j.val)
  28     rw [hmap,hmap]
  29     exact congrArg Sum.inl he
  30   exact Subtype.ext (Sum.inl.inj (h hi hj he'))

  32 theorem ordinary_resource_map_spoke_injective (G : SimpleGraph V) (H : V → U → Prop) (f : V → V) (g : U → U)
  33     (h : Set.InjOn (ordinaryPairResourceMap f g) (ordinaryHostResources G H)) :
  34     Function.Injective (fun i:{e:V×U // H e.1 e.2} => (f i.val.1,g i.val.2)) := by
  35   intro i j he
  36   have hi := (ordinaryHostResources_spoke G H i.val.1 i.val.2).mpr i.property
  37   have hj := (ordinaryHostResources_spoke G H j.val.1 j.val.2).mpr j.property
  38   exact Subtype.ext (Sum.inr.inj (h hi hj (congrArg Sum.inr he)))
