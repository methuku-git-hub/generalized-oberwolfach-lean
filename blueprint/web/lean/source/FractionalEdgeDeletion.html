  13 theorem graph_map_delete_edge {V W : Type*} (G : SimpleGraph V) (f : V ↪ W) (u v : V) :
  14     (G\SimpleGraph.edge u v).map f=G.map f\SimpleGraph.edge (f u) (f v) := by
  15   ext x y
  16   rw [SimpleGraph.map_adj,SimpleGraph.sdiff_adj]
  17   constructor
  18   · rintro ⟨a,b,hab,rfl,rfl⟩
  19     refine ⟨SimpleGraph.map_adj_apply.mpr hab.1,?_⟩
  20     simpa only [SimpleGraph.edge_adj,ne_eq,f.injective.eq_iff] using hab.2
  21   · rintro ⟨hab,hnot⟩
  22     rw [SimpleGraph.map_adj] at hab
  23     obtain ⟨a,b,hab,rfl,rfl⟩ := hab
  24     refine ⟨a,b,⟨hab,?_⟩,rfl,rfl⟩
  25     simpa only [SimpleGraph.edge_adj,ne_eq,f.injective.eq_iff] using hnot

  27 theorem fractional_cycle_edge_deletion (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  28     ∃ N : ℕ,∀ (V : Type) [Fintype V] [DecidableEq V],N≤Fintype.card V →
  29       ∀ G : SimpleGraph V,
  30       (∀ x,(cycleThreshold r+α)*Fintype.card V≤(G.degree x:ℝ)) →
  31       HasFractionalDecomposition G (fun C => IsEmbeddedCycle C r) ∧
  32       ∀ u v,G.Adj u v →
  33         HasFractionalDecomposition (G\SimpleGraph.edge u v) (fun C => IsEmbeddedCycle C r) := by
  34   obtain ⟨N₁,hN₁⟩ := cycle_fractional_positive_margin_finite r hr (α/2) (by positivity)
  35   obtain ⟨N₂,hN₂⟩ := eventually_constant_le_linear 1 (α/2) (by positivity)
  36   refine ⟨max N₁ N₂,?_⟩
  37   intro V _ _ hn G hdeg
  38   constructor
  39   · apply hN₁ V ((le_max_left _ _).trans hn) G
  40     intro x
  41     have ha : 0≤α*(Fintype.card V:ℝ) := mul_nonneg hα.le (Nat.cast_nonneg _)
  42     nlinarith [hdeg x]
  43   · intro u v huv
  44     apply hN₁ V ((le_max_left _ _).trans hn) (G\SimpleGraph.edge u v)
  45     intro x
  46     have hc := congrArg (fun a : ℕ => (a:ℝ)) (single_edge_deletion_degree G u v huv x)
  47     simp only [Nat.cast_add] at hc
  48     have hb : ((if x=u ∨ x=v then 1 else 0 : ℕ):ℝ)≤1 := by split_ifs <;> norm_num
  49     have hn' := hN₂ (Fintype.card V) ((le_max_right _ _).trans hn)
  50     have hg := hdeg x
  51     simp only [← SimpleGraph.ncard_neighborSet] at hc hg ⊢
  52     nlinarith
