  12 theorem exp_mul_unit_interval {x : ℝ} (hx : x ∈ Set.Icc (0 : ℝ) 1) (s : ℝ) :
  13     Real.exp (s * x) ≤ 1 + x * (Real.exp s - 1) := by
  14   have h := convexOn_exp.2 (Set.mem_univ (0 : ℝ)) (Set.mem_univ s)
  15     (sub_nonneg.mpr hx.2) hx.1 (show 1 - x + x = 1 by ring)
  16   simp only [smul_eq_mul, mul_zero, zero_add, Real.exp_zero, mul_one] at h
  17   calc
  18     _ = Real.exp (x * s) := by rw [mul_comm]
  19     _ ≤ 1 - x + x * Real.exp s := h
  20     _ = _ := by ring

  22 theorem finiteLaw_unit_interval_mgf
  23     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X : Ω → ℝ)
  24     (hx : ∀ ω, X ω ∈ Set.Icc (0 : ℝ) 1) (s : ℝ) :
  25     μ.expectation (fun ω => Real.exp (s * X ω)) ≤
  26       Real.exp (μ.expectation X * (Real.exp s - 1)) := by
  27   calc
  28     _ ≤ μ.expectation (fun ω => 1 + X ω * (Real.exp s - 1)) :=
  29       finiteLaw_expectation_mono μ _ _ (fun ω => exp_mul_unit_interval (hx ω) s)
  30     _ = 1 + μ.expectation X * (Real.exp s - 1) := by
  31       simp only [FiniteLaw.expectation, mul_add, mul_one, Finset.sum_add_distrib,
  32         μ.total, ← mul_assoc, Finset.sum_mul]
  33     _ ≤ _ := by simpa only [add_comm] using Real.add_one_le_exp (μ.expectation X * (Real.exp s - 1))

  42 theorem finiteLaw_independent_unit_interval_mgf
  43     {I : Type*} [Fintype I] [DecidableEq I] {Ω : I → Type*} [∀ i, Fintype (Ω i)]
  44     (μ : ∀ i, FiniteLaw (Ω i)) (X : ∀ i, Ω i → ℝ)
  45     (hx : ∀ i ω, X i ω ∈ Set.Icc (0 : ℝ) 1) (s : ℝ) :
  46     (FiniteLaw.pi μ).expectation (fun σ => Real.exp (s * ∑ i, X i (σ i))) ≤
  47       Real.exp ((∑ i, (μ i).expectation (X i)) * (Real.exp s - 1)) := by
  48   simp only [Finset.mul_sum, Real.exp_sum]
  49   rw [finiteLaw_pi_expectation_product μ (fun i x => Real.exp (s * X i x))]
  50   calc
  51     _ ≤ ∏ i, Real.exp ((μ i).expectation (X i) * (Real.exp s - 1)) := by
  52       apply Finset.prod_le_prod₀
  53       · intro i _
  54         exact Finset.sum_nonneg (fun ω _ => mul_nonneg ((μ i).nonneg ω) (Real.exp_pos _).le)
  55       · intro i _
  56         exact finiteLaw_unit_interval_mgf (μ i) (X i) (hx i) s
  57     _ = _ := by rw [← Real.exp_sum, Finset.sum_mul]
