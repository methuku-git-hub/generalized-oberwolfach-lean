  13 theorem graph_edge_assignment (G:SimpleGraph V) (L:Sym2 V→Finset W) (B:V→Finset W)
  14     (a b:ℝ) (ha:0<a) (hb:0≤b) (hbsmall:b≤a/16) (hn:0<Fintype.card V)
  15     (hL:∀e∈G.edgeFinset,a*Fintype.card V≤((L e).card:ℝ))
  16     (hB:∀v,((B v).card:ℝ)≤b*Fintype.card V)
  17     (hG:∀v,(G.degree v:ℝ)≤b*Fintype.card V):
  18     let K:=⌈4*(G.edgeFinset.card:ℝ)/(a*Fintype.card V)⌉₊;
  19     ∃f:Sym2 V→Option W,
  20       (∀e∈G.edgeFinset,∃w,f e=some w ∧ w∈L e ∧ ∀v∈e.toFinset,w∉B v) ∧
  21       (∀e∈G.edgeFinset,∀e'∈G.edgeFinset,e≠e'→edgeAssignmentConflict e e'→f e≠f e') ∧
  22       (∀w,(G.edgeFinset.filter (fun e=>f e=some w)).card≤K) := by
  23   intro K
  24   by_cases hE:G.edgeFinset.Nonempty
  25   · obtain ⟨e₀,he₀⟩:=hE
  26     have hnR:(0:ℝ)<Fintype.card V:=by exact_mod_cast hn
  27     have hLpos:(0:ℝ)<(L e₀).card:=(mul_pos ha hnR).trans_le (hL e₀ he₀)
  28     have hLposN:0<(L e₀).card:=by exact_mod_cast hLpos
  29     obtain ⟨w₀,hw₀⟩:=Finset.card_pos.mp hLposN
  30     letI:Nonempty W:=⟨w₀⟩
  31     have hEpos:(0:ℝ)<G.edgeFinset.card:=by exact_mod_cast Finset.card_pos.mpr ⟨e₀,he₀⟩
  32     have han:0<a*Fintype.card V:=mul_pos ha hnR
  33     have hquot:0<4*(G.edgeFinset.card:ℝ)/(a*Fintype.card V):=by positivity
  34     have hceil:4*(G.edgeFinset.card:ℝ)/(a*Fintype.card V)≤(K:ℝ):=Nat.le_ceil _
  35     have hK:0<K:=by exact_mod_cast hquot.trans_le hceil
  36     have hbudget:∀e∈G.edgeFinset,((edgeAssignmentForbidden B e).card:ℝ)+
  37         ((G.edgeFinset.filter (edgeAssignmentConflict e)).card:ℝ)+(G.edgeFinset.card:ℝ)/K<(L e).card:=by
  38       intro e he
  39       induction e using Sym2.ind with
  40       | _ x y =>
  41         have hbad:=edge_assignment_forbidden_at_most B (b*Fintype.card V) hB x y
  42         have hconf:=edge_assignment_conflicts_at_most G x y
  43         have hconf':((G.edgeFinset.filter (edgeAssignmentConflict s(x,y))).card:ℝ)≤2*b*Fintype.card V:=by
  44           linarith [hG x,hG y]
  45         exact edge_assignment_numeric_budget a b (Fintype.card V) G.edgeFinset.card K
  46           (edgeAssignmentForbidden B s(x,y)).card (G.edgeFinset.filter (edgeAssignmentConflict s(x,y))).card (L s(x,y)).card
  47           ha hnR hb hbsmall (by exact_mod_cast hK) hceil (by positivity) (by simpa only [mul_assoc] using hbad) hconf' (hL s(x,y) he)
  48     obtain ⟨g,hallowed,hconf,hcap⟩:=edge_assignment_greedy G.edgeFinset L (edgeAssignmentForbidden B) edgeAssignmentConflict
  49       edge_assignment_conflict_symmetric K hK hbudget
  50     refine ⟨fun e=>some (g e),?_,?_,?_⟩
  51     · intro e he
  52       refine ⟨g e,rfl,(hallowed e he).1,?_⟩
  53       intro v hv hvB
  54       have hh:g e∈edgeAssignmentForbidden B e:=by
  55         exact Finset.mem_biUnion.mpr ⟨v,hv,hvB⟩
  56       exact (hallowed e he).2 hh
  57     · intro e he e' he' hne hc heq
  58       exact hconf e he e' he' hne hc (Option.some.inj heq)
  59     · intro w
  60       simpa using hcap w
  61   · have hEmpty:G.edgeFinset=∅:=Finset.not_nonempty_iff_eq_empty.mp hE
  62     refine ⟨fun _=>none,?_,?_,?_⟩
  63     · simp [hEmpty]
  64     · simp [hEmpty]
  65     · simp [hEmpty]
  66 end Oberwolfach
