  32 theorem labelled_family_trade_bound (R : C → Finset V) (S : T → VertexAbsorberTemplate c)
  33     (X : Finset V) (A : Finset (LabelledAbsorberFamily R S X)) (C₀ : C) :
  34     |labelledFamilyTrade R S X A C₀| ≤
  35       (Fintype.card {D : A // ∃ o,(D.val.2.2 o).val=C₀}:ℝ)/A.card := by
  36   apply (finiteSignedAverage_abs_le _ C₀).trans
  37   rw [Fintype.card_coe]
  38   apply div_le_div_of_nonneg_right _ (Nat.cast_nonneg _)
  39   calc
  40     _ ≤ ∑ D : A,(if ∃ o,(D.val.2.2 o).val=C₀ then (1:ℝ) else 0) := by
  41       apply Finset.sum_le_sum
  42       intro D _

  48           intro hm
  49           obtain ⟨i,_,he⟩ := Finset.mem_image.mp hm
