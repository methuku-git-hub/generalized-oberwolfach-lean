  13 theorem scaled_degree_mean_error {N m q D ε : ℝ}
  14     (hN : 0 < N) (hm : 0 ≤ m) (hdegree : |q - D| ≤ ε * N) :
  15     |q * (m / N) - D * (m / N)| ≤ ε * m := by
  16   rw [← sub_mul, abs_mul, abs_of_nonneg (div_nonneg hm hN.le)]
  17   calc
  18     |q - D| * (m / N) ≤ (ε * N) * (m / N) :=
  19       mul_le_mul_of_nonneg_right hdegree (div_nonneg hm hN.le)
  20     _ = ε * m := by field_simp
