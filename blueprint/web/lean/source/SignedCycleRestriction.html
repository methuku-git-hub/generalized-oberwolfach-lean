  13 theorem signed_cycle_test_restrict {V : Type*} [Fintype V] [DecidableEq V]
  14     (G : SimpleGraph V) (r : ℕ) (a t : SimpleGraph V → ℝ)
  15     (hs : ∀ C,a C≠0 → C≤G ∧ IsEmbeddedCycle C r) :
  16     (∑ C : OrdinaryCycle G r,a C.val.val*t C.val.val)=∑ C : SimpleGraph V,a C*t C := by
  17   have hzero (C : SimpleGraph V) (hC : ¬(C≤G ∧ IsEmbeddedCycle C r)) : a C=0 := by
  18     by_contra hn
  19     exact hC (hs C hn)
  20   symm
  21   calc
  22     _ = ∑ C : EmbeddedCycleFamily V r,a C.val*t C.val := by
  23       apply finite_sum_restrict_of_zero (fun C => IsEmbeddedCycle C r)
  24       intro C hC
  25       rw [hzero C (fun h => hC h.2),zero_mul]
  26     _ = _ := by
  27       apply finite_sum_restrict_of_zero (fun C : EmbeddedCycleFamily V r => C.val≤G)
  28       intro C hC
  29       rw [hzero C.val (fun h => hC h.1),zero_mul]

  31 theorem indexedWheelWeight_injective_bound {V U I : Type*} [Fintype I]
  32     {Q F : SimpleGraph V} {H : V → U → Prop} {r : ℕ}
  33     (W : I → OrdinaryWheel Q F H r) (hW : Function.Injective W) (a : I → ℝ)
  34     (B : ℝ) (hB : 0≤B) (ha : ∀ i,|a i|≤B) (D : OrdinaryWheel Q F H r) :
  35     |indexedWheelWeight W a D|≤B := by
  36   classical
  37   by_cases hD : D∈Set.range W
  38   · obtain ⟨i,rfl⟩ := hD
  39     have he : indexedWheelWeight W a (W i)=a i := by
  40       unfold indexedWheelWeight
  41       simp only [hW.eq_iff,Finset.sum_ite_eq',Finset.mem_univ,ite_true]
  42     rw [he]
  43     exact ha i
  44   · have hz : indexedWheelWeight W a D=0 := by
  45       unfold indexedWheelWeight
  46       apply Finset.sum_eq_zero
  47       intro i _
  48       exact if_neg (fun h => hD ⟨i,h⟩)
  49     rw [hz,abs_zero]
  50     exact hB
