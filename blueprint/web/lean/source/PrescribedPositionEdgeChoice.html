   4 /-! In a graph of maximum degree two, a vertex set meeting every edge has
   5 size at least half the number of edges. -/
   6 set_option autoImplicit false
   7 noncomputable section
   8 namespace Oberwolfach
   9 attribute [local instance] Classical.propDecidable
  10 open scoped BigOperators
  11 variable {X:Type} [Fintype X] [DecidableEq X]

  13 theorem prescribed_position_edge_choice (Q:SimpleGraph X) (S:Finset X)
  14     (hdeg:∀ x,Q.degree x≤2) (hE:2*S.card<Q.edgeFinset.card):
  15     ∃ x y,Q.Adj x y ∧ x∉S ∧ y∉S := by
  16   by_contra hn
  17   have hcover:Q.edgeFinset⊆S.biUnion (fun v=>Q.incidenceFinset v):=by
  18     intro e he
  19     induction e using Sym2.ind with
  20     | _ x y =>
  21       have hxy:Q.Adj x y:=SimpleGraph.mem_edgeFinset.mp he
  22       have hs:x∈S ∨ y∈S:=by
  23         by_contra h
  24         push_neg at h
  25         exact hn ⟨x,y,hxy,h.1,h.2⟩
  26       rcases hs with hx|hy
  27       · apply Finset.mem_biUnion.mpr
  28         refine ⟨x,hx,?_⟩
  29         rw [Q.incidenceFinset_eq_filter x]
  30         simp [he]
  31       · apply Finset.mem_biUnion.mpr
  32         refine ⟨y,hy,?_⟩
  33         rw [Q.incidenceFinset_eq_filter y]
  34         simp [he]
  35   have hc:Q.edgeFinset.card≤2*S.card:=by
  36     calc
  37       Q.edgeFinset.card≤(S.biUnion (fun v=>Q.incidenceFinset v)).card:=Finset.card_le_card hcover
  38       _≤∑ x∈S,(Q.incidenceFinset x).card:=Finset.card_biUnion_le
  39       _≤∑ _x∈S,2:=by
  40         apply Finset.sum_le_sum
  41         intro x hx
  42         rw [Q.card_incidenceFinset_eq_degree]
  43         exact hdeg x
  44       _=2*S.card:=by simp [Nat.mul_comm]
  45   omega
  46 end Oberwolfach
