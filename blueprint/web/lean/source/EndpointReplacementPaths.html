  22     (j : Bool) (x y : V) (hxy : x≠y) (hFxy : ¬F.Adj x y)
  23     (hx : x∈if j then T else S) (hy : y∈if j then T else S)
  24     (n : ℕ) (hn : n<2) (f : Fin n→V)
  25     (hf : ClassSparseGreedyGood Q F (endpointPathClass S T j) {x,y} (endpointPathRequirements x y) f) :
  26     b≤((common (endpointPathClass S T j n) (endpointPathRequirements x y n f)
  27       (fun v=>Q.neighborFinset v)).card:ℝ) := by
  28   have hpair : ({x,y}:Finset V).card=2 := Finset.card_pair hxy
  29   interval_cases n
  30   · cases j
  31     · simpa [endpointPathClass,endpointPathRequirements,common] using hb.pairs x hx y hy hxy hFxy
  32     · have hh := hb.rows ∅ {x,y} (by simp) (by simpa [Finset.insert_subset_iff,Finset.singleton_subset_iff] using And.intro hx hy) (by simp) hpair.le (by simp [hpair])
  33       simpa [endpointPathClass,endpointPathRequirements] using hh
  34   · have he : Finset.univ.image f={f 0} := by
  35       ext v;simp only [Finset.mem_image,Finset.mem_univ,true_and,Finset.mem_singleton]
  36       constructor
  37       · rintro ⟨i,rfl⟩;rw [Subsingleton.elim i 0]
  38       · rintro rfl;exact ⟨0,rfl⟩
  39     have hf0 := (hf.2.1 0).1
  40     cases j
  41     · have hh := hb.rows {x,y} {f 0} (by simpa [Finset.insert_subset_iff,Finset.singleton_subset_iff] using And.intro hx hy)
  42         (by simpa [endpointPathClass] using hf0) hpair.le (by simp) (by simp [hpair])
  43       simpa only [endpointPathClass,Bool.false_or,ne_eq,one_ne_zero,not_false_eq_true,decide_true,ite_true,

  45     · have hh := hb.rows {f 0} {x,y} (by simpa [endpointPathClass] using hf0)
