  14 theorem two_root_coefficient_factor {V : Type*} [DecidableEq V]
  15     (S : Finset V) {r : ℕ} (hr : 1≤r) (hS : S.card=r)
  16     (x y : V) (hxy : x≠y) (n γ : ℝ) (hn : n≠0) :
  17     1/(γ*n^((S∪{x,y}).card-1))=
  18       (1/(γ*n^(r-1)))*(if x∈S then 1 else 1/n)*(if y∈S then 1 else 1/n) := by
  19   have he : S∪{x,y}=insert x (insert y S) := by
  20     ext z
  21     simp only [Finset.mem_union,Finset.mem_insert,Finset.mem_singleton]
  22     tauto
  23   rw [he]
  24   by_cases hx : x∈S <;> by_cases hy : y∈S
  25   · simp [hx,hy,hS]
  26   · have hx' : x∈insert y S := Finset.mem_insert_of_mem hx
  27     rw [Finset.insert_eq_of_mem hx',Finset.card_insert_of_notMem hy,hS]
  28     simp only [hx,hy,if_true,if_false,mul_one,Nat.add_sub_cancel]
  29     have hp : n^r=n^(r-1)*n := by rw [← pow_succ,show r-1+1=r by omega]
  30     rw [hp];field_simp
  31   · have hx' : x∉insert y S := by simp [hx,hxy]
  32     rw [Finset.card_insert_of_notMem hx',Finset.insert_eq_of_mem hy,hS]
  33     simp only [hx,hy,if_true,if_false,mul_one,Nat.add_sub_cancel]
  34     have hp : n^r=n^(r-1)*n := by rw [← pow_succ,show r-1+1=r by omega]
  35     rw [hp];field_simp
  36   · have hx' : x∉insert y S := by simp [hx,hxy]
  37     rw [Finset.card_insert_of_notMem hx',Finset.card_insert_of_notMem hy,hS]
  38     simp only [hx,hy,if_true,if_false,Nat.add_sub_cancel]
  39     have hp : n^(r+1)=n^(r-1)*n*n := by
  40       rw [← pow_succ,← pow_succ,show r-1+1+1=r+1 by omega]
  41     rw [hp];field_simp

  43 theorem root_discount_sum {V : Type*} [Fintype V] [DecidableEq V]
  44     (S : Finset V) (hn : 0<Fintype.card V) :
  45     (∑ x : V,if x∈S then (1:ℝ) else 1/Fintype.card V)≤S.card+1 := by
  46   have hn' : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr hn
  47   have hfirst : (∑ x : V,if x∈S then (1:ℝ) else 0)=(S.card:ℝ) := by
  48     rw [← Finset.sum_filter]
  49     have hf : Finset.univ.filter (fun x => x∈S)=S := by ext;simp
  50     rw [hf,Finset.sum_const,nsmul_eq_mul,mul_one]
  51   calc
  52     _ ≤ ∑ x : V,((if x∈S then (1:ℝ) else 0)+1/Fintype.card V) := by
  53       apply Finset.sum_le_sum
  54       intro x _
  55       split_ifs
  56       · exact le_add_of_nonneg_right (by positivity)
  57       · simp
  58     _ = _ := by
  59       rw [Finset.sum_add_distrib,hfirst,Finset.sum_const,Finset.card_univ,nsmul_eq_mul]
  60       field_simp

  62 theorem two_root_coefficient_sum {V : Type*} [Fintype V] [DecidableEq V]
  63     (S : Finset V) {r : ℕ} (hr : 1≤r) (hS : S.card=r)
  64     (γ : ℝ) (hγ : 0<γ) (hn : 0<Fintype.card V)
  65     (a : V → V → ℝ) (ha : ∀ x y,0≤a x y)
  66     (hdiag : ∀ x,a x x=0)
  67     (hb : ∀ x y,x≠y → a x y≤1/(γ*(Fintype.card V:ℝ)^((S∪{x,y}).card-1))) :
  68     (∑ x,∑ y,a x y)≤(r+1)^2/(γ*(Fintype.card V:ℝ)^(r-1)) := by
  69   have hn' : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr hn
  70   let b (x : V) : ℝ := if x∈S then 1 else 1/Fintype.card V
  71   let C : ℝ := 1/(γ*(Fintype.card V:ℝ)^(r-1))
  72   have hC : 0≤C := by dsimp [C];positivity
  73   have hb0 (x : V) : 0≤b x := by dsimp [b];split_ifs <;> positivity
  74   have hsum : (∑ x,b x)≤r+1 := by simpa only [hS] using root_discount_sum S hn
  75   have hterm (x y : V) : a x y≤C*b x*b y := by
  76     by_cases hxy : x=y
  77     · subst y;rw [hdiag];positivity
  78     · have hh := hb x y hxy
  79       rw [two_root_coefficient_factor S hr hS x y hxy _ γ hn'.ne'] at hh
  80       exact hh
  81   calc
  82     _ ≤ ∑ x,∑ y,C*b x*b y := Finset.sum_le_sum (fun x _ => Finset.sum_le_sum (fun y _ => hterm x y))
  83     _ = C*(∑ x,b x)*(∑ x,b x) := by
  84       simp_rw [← Finset.mul_sum]
  85       rw [← Finset.sum_mul,← Finset.mul_sum]
  86     _ ≤ C*(r+1)*(r+1) := by
  87       apply mul_le_mul
  88       · exact mul_le_mul_of_nonneg_left hsum hC
  89       · exact hsum
  90       · exact Finset.sum_nonneg (fun x _ => hb0 x)
  91       · positivity
  92     _ = _ := by dsimp [C];ring
