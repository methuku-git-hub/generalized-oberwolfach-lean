  15 theorem endpoint_as_local_cycle_exists (hST:Disjoint S T) (C:EndpointTriangle Q F S T):
  16     ∃ D:LocalOrdinaryCycle Q F (S∪T) 3,localOrdinaryVertices D=endpointTriangleVertices C:=by
  17   have hc:C.val.card=3:=endpointTriangleVertices_card hST C
  18   let f:Fin 3≃C.val:=Fintype.equivOfCardEq (by simp only [Fintype.card_fin,Fintype.card_coe,hc])
  19   let e:Fin 3↪V:=⟨fun i=>(f i).val,fun i j h=>f.injective (Subtype.ext h)⟩
  20   have hi:Finset.univ.image e=C.val:=by
  21     ext v
  22     constructor
  23     · rintro hv;obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hv;exact (f i).property
  24     · intro hv;obtain ⟨i,hi⟩:=f.surjective ⟨v,hv⟩
  25       exact Finset.mem_image.mpr ⟨i,Finset.mem_univ _,congrArg Subtype.val hi⟩
  26   obtain ⟨D,hD⟩:=ordinary_cycle_from_embedding Q F (S∪T) (by norm_num : 3≤3) e
  27     (fun i=>C.property.1 (f i).property)
  28     (fun i j=>C.property.2.2.2.2 _ (f i).property _ (f j).property)
  29     (fun i j hij=>C.property.2.2.2.1 _ (f i).property _ (f j).property
  30       (fun he=>hij.ne (e.injective he)))
  31   exact ⟨D,hD.trans hi⟩

  43 theorem endpointFactorGraph_choice (hST:Disjoint S T) (D:Finset (EndpointTriangle Q F S T))
  44     (Y:Finset V) (K:SimpleGraph V)
  45     (hD:∀ C∈D,∀ E∈D,C≠E→Disjoint (endpointTriangleVertices C) (endpointTriangleVertices E))
  46     (hY:D.biUnion endpointTriangleVertices=Y)
  47     (hK:∀ C∈D,∀ x∈endpointTriangleVertices C,∀ y∈endpointTriangleVertices C,¬K.Adj x y):
  48     endpointFactorGraph hST D≤spanningRestriction Q Y ∧
  49       (∀ v,(endpointFactorGraph hST D).degree v≤2) ∧
  50       Disjoint (endpointFactorGraph hST D) K:=by
  51   let M:=D.image (endpointAsLocalCycle hST)
  52   have hM:∀ C∈M,∀ E∈M,C≠E→Disjoint (localOrdinaryVertices C) (localOrdinaryVertices E):=by
  53     intro C hC E hE hne
  54     obtain ⟨C',hC',rfl⟩:=Finset.mem_image.mp hC
  55     obtain ⟨E',hE',rfl⟩:=Finset.mem_image.mp hE
  56     rw [endpointAsLocalCycle_vertices,endpointAsLocalCycle_vertices]
  57     exact hD C' hC' E' hE' (fun he=>hne (congrArg (endpointAsLocalCycle hST) he))
  58   have hcover:M.biUnion localOrdinaryVertices=Y:=by
  59     rw [←hY]
  60     ext v
  61     simp only [M,Finset.mem_biUnion,Finset.mem_image]
  62     constructor
  63     · rintro ⟨_,⟨C,hC,rfl⟩,hv⟩
  64       exact ⟨C,hC,by simpa only [endpointAsLocalCycle_vertices] using hv⟩
  65     · rintro ⟨C,hC,hv⟩
  66       exact ⟨endpointAsLocalCycle hST C,⟨C,hC,rfl⟩,by simpa only [endpointAsLocalCycle_vertices] using hv⟩
  67   have havoid:∀ C∈M,∀ x∈localOrdinaryVertices C,∀ y∈localOrdinaryVertices C,¬K.Adj x y:=by
  68     intro C hC
  69     obtain ⟨C',hC',rfl⟩:=Finset.mem_image.mp hC
  70     simpa only [endpointAsLocalCycle_vertices] using hK C' hC'
  71   exact ⟨localOrdinaryFactorGraph_restriction M Y hcover,localOrdinaryFactorGraph_degree M hM,
  72     localOrdinaryFactorGraph_disjoint M K havoid⟩
