  51 theorem interval_scaled_path_nonneg {X Y Z : Type*}
  52     [Fintype X] [Fintype Y] [Fintype Z]
  53     (L : X → Y → Prop) (R : Y → Z → Prop) (k δ : ℝ)
  54     (hk : 2 ≤ k) (hn : 0 < Fintype.card Y) (hδ : 0 ≤ δ)
  55     (hthreshold : k * Fintype.card Y ≤ (2 * k - 1) * δ)
  56     (hdL : ∀ x, δ ≤ (Finset.univ.filter (L x)).card)
  57     (hdR : ∀ z, δ ≤ (Finset.univ.filter (fun y => R y z)).card)
  58     (hbal : ∀ y, (Finset.univ.filter (fun x => L x y)).card =
  59       (Finset.univ.filter (R y)).card)
  60     {Q : X → Z → ℝ} (hQ : Q ∈ transportMatrices
  61       (fun x => (Finset.univ.filter (L x)).card / k)
  62       (fun z => (Finset.univ.filter (fun y => R y z)).card / k))
  63     (x : X) (y : Y) (z : Z) :
  64     0 ≤ scaledResidualPath (interfaceCapacity L) (interfaceCapacity R)
  65       (endpointReserve L R Q) k x y z := by
  66   obtain ⟨hL, hR⟩ := endpointReserve_capacities L R Q k δ hk hn hδ hthreshold hdL hdR hQ
  67   have hb : ∀ y, (∑ x, interfaceCapacity L x y) = ∑ z, interfaceCapacity R y z := by
  68     intro y
  69     rw [interfaceCapacity_column, interfaceCapacity_row, hbal y]
  70   unfold scaledResidualPath
  71   apply div_nonneg _ (by linarith)
  72   exact (residualPath_bounds _ _
  73     (fun x y => sub_nonneg.mpr (hL x y))
  74     (fun y z => sub_nonneg.mpr (hR y z))
  75     (residual_interfaces_balanced _ _ _ hb) x y z).1
