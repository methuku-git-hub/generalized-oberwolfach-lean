  15 theorem isolator_configuration_hub_root_count {V U : Type*} [Fintype V] [DecidableEq V] [Fintype U]
  16     {P Q F : SimpleGraph V} {H : V → U → Prop} {k : ℕ} {θ α : ℝ} {x y : V}
  17     (u : U) (R : Finset V) (hR : R.card≤k) :
  18     (∑ T : OrdinaryIsolatorConfiguration P Q F H k θ α x y,
  19       if T.hub=u ∧ R⊆T.vertices.val then (1:ℝ) else 0)≤
  20       (Fintype.card V:ℝ)^(k-R.card) := by
  21   let A := {T : OrdinaryIsolatorConfiguration P Q F H k θ α x y // T.hub=u ∧ R⊆T.vertices.val}
  22   let B := {S : FixedSubset V k // R⊆S.val}
  23   let f (T : A) : B := ⟨T.val.vertices,T.property.2⟩
  24   have hf : Function.Injective f := by
  25     intro T T' he
  26     apply Subtype.ext
  27     apply OrdinaryIsolatorConfiguration.encoding_injective
  28     apply Prod.ext
  29     · exact T.property.1.trans T'.property.1.symm
  30     · exact congrArg Subtype.val he
  31   have hc : Fintype.card A≤Fintype.card V^(k-R.card) := by
  32     calc
  33       _ ≤ Fintype.card B := Fintype.card_le_of_injective f hf
  34       _ = (Fintype.card V-R.card).choose (k-R.card) := by
  35         simpa only [← Nat.card_eq_fintype_card] using fixedSubset_containing_card R hR
  36       _ ≤ (Fintype.card V-R.card)^(k-R.card) := Nat.choose_le_pow _ _
  37       _ ≤ _ := Nat.pow_le_pow_left (Nat.sub_le _ _) _
  38   have he : (∑ T : OrdinaryIsolatorConfiguration P Q F H k θ α x y,
  39       if T.hub=u ∧ R⊆T.vertices.val then (1:ℝ) else 0)=(Fintype.card A:ℝ) := by
  40     simp only [← Finset.sum_filter,Finset.sum_const,nsmul_eq_mul,mul_one,Fintype.card_subtype,A]
  41   rw [he]
  42   exact_mod_cast hc
