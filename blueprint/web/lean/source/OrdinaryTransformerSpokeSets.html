  17     ((Finset.univ:Finset {e:V×U // H e.1 e.2}).filter (fun e=>e.val.2=u)).card=
  18       ((Finset.univ:Finset V).filter (fun v=>H v u)).card:=by

  29  theorem ordinary_transformer_spoke_sets (Q F:SimpleGraph V) (H K:V→U→Prop)
  30     (φ:U→U) (r:ℕ) (hr:3≤r)
  31     (p:{e:V×U // H e.1 e.2}→(Σ a:V,Σ b:V,Q.Walk a b))
  32     (hpath:∀ i,(p i).2.2.IsPath∧(p i).2.2.length=r-2)
  33     (hdis:∀ i j,i≠j→Disjoint (p i).2.2.support.toFinset (p j).2.2.support.toFinset)
  34     (hF:∀ i j,∀ v∈(p i).2.2.support,∀ w∈(p j).2.2.support,¬F.Adj v w)
  35     (hH:∀ i,∀ v∈(p i).2.2.support,K v i.val.2∧K v (φ i.val.2))
  36     (hdiv:∀ u,r∣((Finset.univ:Finset V).filter (fun v=>H v u)).card):
  37     let Z:U→Finset V:=fun u=>(Finset.univ.filter (fun i:{e:V×U // H e.1 e.2}=>i.val.2=u)).biUnion
  38       (fun i=>(p i).2.2.support.toFinset)
  39     (∀ u,(Z u).card=(r-1)*((Finset.univ:Finset V).filter (fun v=>H v u)).card) ∧
  40     (∀ u,r∣(Z u).card) ∧ (∀ u u',u≠u'→Disjoint (Z u) (Z u')) ∧
  41     (∀ u,∀ v∈Z u,K v u∧K v (φ u)) ∧
  42     (∀ u,∀ v∈Z u,∀ w∈Z u,¬F.Adj v w):=by
  43   dsimp only
  44   let J:={e:V×U // H e.1 e.2}
  45   let I:U→Finset J:=fun u=>Finset.univ.filter (fun i=>i.val.2=u)
  46   let Z:U→Finset V:=fun u=>(I u).biUnion (fun i=>(p i).2.2.support.toFinset)
  47   have hsize (i:J):(p i).2.2.support.toFinset.card=r-1:=by
  48     rw [List.toFinset_card_of_nodup (hpath i).1.support_nodup,(p i).2.2.length_support,(hpath i).2]
  49     omega
  50   have hZcard (u:U):(Z u).card=(r-1)*((Finset.univ:Finset V).filter (fun v=>H v u)).card:=by
  51     change ((I u).biUnion (fun i=>(p i).2.2.support.toFinset)).card=_
  52     rw [Finset.card_biUnion (by intro i _ j _ hij;exact hdis i j hij)]
  53     simp only [hsize,Finset.sum_const,nsmul_eq_mul]
  54     rw [show (I u).card=((Finset.univ:Finset V).filter (fun v=>H v u)).card from spoke_index_fiber_card H u,Nat.mul_comm]
  55     norm_cast
  56   refine ⟨hZcard,?_,?_,?_,?_⟩
  57   · intro u
  58     rw [hZcard]
  59     exact dvd_mul_of_dvd_right (hdiv u) (r-1)
  60   · intro u u' huu'
  61     apply Finset.disjoint_left.mpr
  62     intro v hv hv'
  63     obtain ⟨i,hi,hvi⟩:=Finset.mem_biUnion.mp hv
  64     obtain ⟨j,hj,hvj⟩:=Finset.mem_biUnion.mp hv'
  65     have hij:i≠j:=by
  66       intro he
  67       have hi':i.val.2=u:=(Finset.mem_filter.mp hi).2
  68       have hj':j.val.2=u':=(Finset.mem_filter.mp hj).2
  69       exact huu' (hi'.symm.trans ((congrArg (fun e:J=>e.val.2) he).trans hj'))
  70     exact Finset.disjoint_left.mp (hdis i j hij) hvi hvj
  71   · intro u v hv
  72     obtain ⟨i,hi,hv⟩:=Finset.mem_biUnion.mp hv
  73     have hi':i.val.2=u:=(Finset.mem_filter.mp hi).2
  74     simpa only [hi'] using hH i v (List.mem_toFinset.mp hv)
  75   · intro u v hv w hw
  76     obtain ⟨i,_,hv⟩:=Finset.mem_biUnion.mp hv
  77     obtain ⟨j,_,hw⟩:=Finset.mem_biUnion.mp hw
  78     exact hF i j v (List.mem_toFinset.mp hv) w (List.mem_toFinset.mp hw)
