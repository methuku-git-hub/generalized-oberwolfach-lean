  17     (∏ e∈ordinaryWheelResources W,Sum.elim (fun _=>pR) (fun _=>pH) e)=pR^r*pH^r := by
  18   have hd : Disjoint ((ordinaryWheelRim W).edgeFinset.image (Sum.inl (β:=V×U)))
  19       ((ordinaryWheelRim W).support.toFinset.image (fun v=>Sum.inr (v,ordinaryWheelHub W))) := by

  21     intro e he hf
  22     obtain ⟨a,_,rfl⟩:=Finset.mem_image.mp he
  23     obtain ⟨b,_,hh⟩:=Finset.mem_image.mp hf
  24     cases hh
  25   have hC : IsEmbeddedCycle (ordinaryWheelRim W) r := W.val.1.val.property
  26   unfold ordinaryWheelResources
  27   rw [Finset.prod_union hd,
  28     Finset.prod_image (by intro x _ y _ h;exact Sum.inl.inj h),
  29     Finset.prod_image (by intro x _ y _ h;exact (Prod.mk.inj (Sum.inr.inj h)).1)]

  33 theorem ordinaryWheelResources_rim_omitted_product (W : OrdinaryWheel P F Hbase r)
  34     (hr : 1≤r) (pR pH : ℝ) (hpR : pR≠0) (e : Sym2 V)
  35     (he : Sum.inl e∈ordinaryWheelResources W) :
  36     (∏ a∈ordinaryWheelResources W\{Sum.inl e},Sum.elim (fun _=>pR) (fun _=>pH) a)=
  37       pR^(r-1)*pH^r := by
  38   have hh := Finset.mul_prod_erase (ordinaryWheelResources W)
  39     (Sum.elim (fun _=>pR) (fun _=>pH)) he
  40   have heq : (ordinaryWheelResources W).erase (Sum.inl e)=ordinaryWheelResources W\{Sum.inl e} := by
  41     ext a;simp [and_comm]
  42   rw [heq,ordinaryWheelResources_constant_product] at hh
  43   apply mul_left_cancel₀ hpR
  44   calc
  45     _ = pR^r*pH^r := hh
  46     _ = (pR*pR^(r-1))*pH^r := by rw [← pow_succ',Nat.sub_add_cancel hr]
  47     _ = _ := by ring
