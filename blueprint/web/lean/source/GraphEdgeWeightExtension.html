   3 /-! Extending weights on graph edges by zero preserves every incident sum.
   4 This makes the total functions used in finite sums a proved representation,
   5 not an extra fractional-matching assumption. -/

  15 theorem graph_weighted_incidence_sum (G : SimpleGraph V) (weight : Sym2 V→ℝ) (v : V) :
  16     (∑ w,if G.Adj v w then weight s(v,w) else 0)=
  17       ∑ e∈G.edgeFinset,if v∈e then weight e else 0 := by
  18   have hl : (∑ w,if G.Adj v w then weight s(v,w) else 0)=
  19       ∑ w∈G.neighborFinset v,weight s(v,w) := by
  20     rw [SimpleGraph.neighborFinset_eq_filter,Finset.sum_filter]
  21   have hr : (∑ e∈G.edgeFinset,if v∈e then weight e else 0)=
  22       ∑ e∈G.incidenceFinset v,weight e := by
  23     rw [SimpleGraph.incidenceFinset_eq_filter,Finset.sum_filter]
  24   rw [hl,hr]
  25   apply Finset.sum_bij (fun w _=>s(v,w))
  26   · intro w hw
  27     simpa using hw
  28   · intro w hw z hz he
  29     exact Sym2.congr_right.mp he
  30   · intro e he
  31     have he' : e∈G.edgeSet ∧ v∈e := by simpa [SimpleGraph.incidenceSet] using he
  32     obtain ⟨w,rfl⟩:=Sym2.mem_iff_exists.mp he'.2
  33     exact ⟨w,by simpa using he'.1,rfl⟩
  34   · intro w hw;rfl

  36 def graphEdgeWeightExtension (G : SimpleGraph V) (weight : G.edgeSet→ℝ) (e : Sym2 V) : ℝ :=
  37   if h:e∈G.edgeSet then weight ⟨e,h⟩ else 0

  39 theorem graphEdgeWeightExtension_edge (G : SimpleGraph V) (weight : G.edgeSet→ℝ)
  40     (e : Sym2 V) (he : e∈G.edgeSet) : graphEdgeWeightExtension G weight e=weight ⟨e,he⟩ := by
  41   simp [graphEdgeWeightExtension,he]

  47 theorem graphEdgeWeightExtension_incident (G : SimpleGraph V) (weight : G.edgeSet→ℝ) (v : V) :
  48     (∑ w,graphEdgeWeightExtension G weight s(v,w))=
  49       ∑ e:G.edgeSet,if v∈e.val then weight e else 0 := by
  50   have hleft : (∑ w,graphEdgeWeightExtension G weight s(v,w))=
  51       ∑ w,if G.Adj v w then graphEdgeWeightExtension G weight s(v,w) else 0 := by
  52     apply Finset.sum_congr rfl
  53     intro w hw
  54     by_cases h:G.Adj v w
  55     · simp [h]
  56     · simp [h,graphEdgeWeightExtension]
  57   rw [hleft,graph_weighted_incidence_sum]
  58   have hh:=Finset.sum_subtype (F:=inferInstance) (p:=fun e:Sym2 V=>e∈G.edgeSet)
  59     G.edgeFinset (fun e=>by simp) (fun e=>if v∈e then graphEdgeWeightExtension G weight e else 0)
  60   rw [hh]
  61   apply Finset.sum_congr rfl
  62   intro e he
  63   rw [graphEdgeWeightExtension_edge G weight e.val e.property]
