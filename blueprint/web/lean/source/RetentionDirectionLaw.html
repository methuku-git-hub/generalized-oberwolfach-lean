   5 /-! Retention of a physical edge and its independent fair direction have
   6 exactly the two-colour direction law. The direction of an unretained edge
   7 is still present in the experiment, for subsequent insertions. -/

  15 def retentionColourProbability (p : ℝ) (a : Fin 2) : ℝ := if a=0 then p else 1-p

  17 theorem retentionColourProbability_nonneg (p : ℝ) (hp : 0≤p) (hp1 : p≤1) (a : Fin 2) :
  18     0≤retentionColourProbability p a := by unfold retentionColourProbability;split_ifs <;> linarith

  20 theorem retentionColourProbability_sum (p : ℝ) : ∑ a,retentionColourProbability p a=1 := by
  21   simp [Fin.sum_univ_two,retentionColourProbability]

  23 def retentionDirectionToColour (z : Bool×Bool) : Fin 2×Bool := (if z.1 then 0 else 1,z.2)
  24 
  25 theorem retentionDirection_map (p : ℝ) (hp : 0≤p) (hp1 : p≤1) :
  26     ((FiniteLaw.retentionCoin p hp hp1).prod FiniteLaw.fairCoin).map retentionDirectionToColour=
  27       FiniteLaw.categoricalDirection (retentionColourProbability p)
  28         (retentionColourProbability_nonneg p hp hp1) (retentionColourProbability_sum p) := by
  29   apply finiteLaw_ext
  30   intro z
  31   rcases z with ⟨a,b⟩
  32   fin_cases a <;> cases b <;>
  33     simp [FiniteLaw.map,FiniteLaw.prod,FiniteLaw.retentionCoin,FiniteLaw.fairCoin,
  34       FiniteLaw.categoricalDirection,retentionDirectionToColour,retentionColourProbability,
  35       Fintype.sum_prod_type,Fintype.sum_bool] <;> ring

  48     (p : ℝ) (hp : 0≤p) (hp1 : p≤1) (E : (V→V→ℕ)→Prop) :
