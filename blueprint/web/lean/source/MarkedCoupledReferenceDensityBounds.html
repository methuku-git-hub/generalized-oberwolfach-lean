  13     (hρlo:ρ₀≤ρ) (hρhi:ρ≤1) (hslo:sstar≤s) (hshi:s≤1/100):
  14     let t₀:=sstar/m
  15     let t:=s/m
  16     let pA0:=ρ₀*sstar/2
  17     let p0:=ρ₀*sstar/(2*m)
  18     let p:=ρ*s/2
  19     let k:=p/m;
  20     0<t₀ ∧ t₀≤1 ∧ 0<pA0 ∧ pA0≤1/2 ∧ 0<p0 ∧ p0≤1/2 ∧
  21     t₀≤t ∧ (m:ℝ)*t≤1 ∧ pA0≤ρ*(m:ℝ)*t/2 ∧
  22     p0≤ρ*(m:ℝ)*t/2 ∧ p0≤ρ*t/2 ∧ p0≤t ∧
  23     p0=ρ₀*t₀/2 ∧ p=ρ*(m:ℝ)*t/2 ∧ k=ρ*t/2 ∧ p0/2=ρ₀*t₀/4 := by

  32   have ht:0<t := by dsimp [t];positivity
  33   have hts:t₀≤t := by dsimp [t₀,t];exact div_le_div_of_nonneg_right hslo hmpos.le
