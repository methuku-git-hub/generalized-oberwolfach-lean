  16  theorem transformer_root_resources_disjoint (G:SimpleGraph V) (H₀:V→U→Prop) (φ:V→V) (ψ:U→U)
  17     (hbalance:∀ v,G.degree v=2*((Finset.univ:Finset U).filter (H₀ v)).card)
  18     (hcross:∀ v∈G.support,∀ w∈G.support,φ v≠w):
  19     Disjoint (ordinaryHostResources G H₀) (transformerImageResources G H₀ φ ψ):=by
  20   apply Finset.disjoint_left.mpr

  22   rcases Finset.mem_union.mp hf with hf|hf
  23   · obtain ⟨i,_,hi⟩:=Finset.mem_image.mp hf
  24     rw [←hi,ordinaryHostResources_rim,SimpleGraph.mem_edgeSet] at he
  25     exact hcross _ (edge_out_adj G i).mem_support_left _ he.mem_support_left rfl

  53   let C':Finset V:=(wheelFamilyRim (fun d=>(D d).second)).support.toFinset
  54   have hC:C.card≤r*Fintype.card (Σ v,G.neighborSet v):=(wheelFamily_vertex_bound (fun d=>(D d).first)).1
  55   have hC':C'.card≤r*Fintype.card (Σ v,G.neighborSet v):=(wheelFamily_vertex_bound (fun d=>(D d).second)).1

  57     apply Set.mem_toFinset.mpr
