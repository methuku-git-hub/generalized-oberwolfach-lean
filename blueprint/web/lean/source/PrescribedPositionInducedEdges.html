  12     (hcross:∀ e∈es,∀ v∈vs,v∉prescribedPairClosed Q e)

  14     (hy:y∈es.flatMap prescribedPairVertices++vs) (hxy:Q.Adj x y):
  15     ∃ e∈es,(x=e.1 ∧ y=e.2) ∨ (x=e.2 ∧ y=e.1) := by

  22       rcases hxe with hxe|hxe <;> rcases hyf with hyf|hyf
  23       · subst x;subst y;exact (Q.irrefl hxy).elim
  24       · exact ⟨e,he,Or.inl ⟨hxe,hyf⟩⟩
  25       · exact ⟨e,he,Or.inr ⟨hxe,hyf⟩⟩
