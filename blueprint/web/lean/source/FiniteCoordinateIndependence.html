  17 theorem finiteLaw_disjoint_coordinate_events
  18     (μ : ∀ e,FiniteLaw (State e)) (S T : Finset E) (hST : Disjoint S T)
  19     (A : (∀ e : S,State e) → Prop) (B : (∀ e : T,State e) → Prop) :
  20     (FiniteLaw.pi μ).probability (fun σ=>A (fun e : S=>σ e) ∧ B (fun e : T=>σ e))=
  21       (FiniteLaw.pi μ).probability (fun σ=>A (fun e : S=>σ e))*
  22         (FiniteLaw.pi μ).probability (fun σ=>B (fun e : T=>σ e)) := by
  23   let : ∀ e,MeasurableSpace (State e) := fun _=>⊤
  24   let : ∀ e,MeasurableSingletonClass (State e) := fun _=>⟨fun _=>trivial⟩
  25   have hcoord : iIndepFun (fun (e : E) (σ : ∀ i,State i)=>σ e) (FiniteLaw.pi μ).toPMF.toMeasure := by
  26     rw [finiteLaw_pi_measure]
  27     exact iIndepFun_pi (fun _=>aemeasurable_id)
  28   have hh := hcoord.indepFun_finset S T hST (fun e=>measurable_pi_apply e)
  29   have hm := hh.measure_inter_preimage_eq_mul {σ | A σ} {σ | B σ}
  30     (Set.toFinite _).measurableSet (Set.toFinite _).measurableSet
  31   have hr := congrArg ENNReal.toReal hm
  32   rw [ENNReal.toReal_mul] at hr
  33   change (FiniteLaw.pi μ).toPMF.toMeasure.real
  34       {σ | A (fun e : S=>σ e) ∧ B (fun e : T=>σ e)} =
  35     (FiniteLaw.pi μ).toPMF.toMeasure.real {σ | A (fun e : S=>σ e)} *
  36       (FiniteLaw.pi μ).toPMF.toMeasure.real {σ | B (fun e : T=>σ e)} at hr
  37   simpa only [finiteLaw_probability_measure] using hr

  39 theorem finiteLaw_disjoint_supported_events
  40     (μ : ∀ e,FiniteLaw (State e)) (S T : Finset E) (hST : Disjoint S T)
  41     (A B : (∀ e,State e) → Prop)
  42     (hA : ∀ σ τ,(∀ e∈S,σ e=τ e) → (A σ ↔ A τ))
  43     (hB : ∀ σ τ,(∀ e∈T,σ e=τ e) → (B σ ↔ B τ)) :
  44     (FiniteLaw.pi μ).probability (fun σ=>A σ ∧ B σ)=
  45       (FiniteLaw.pi μ).probability A*(FiniteLaw.pi μ).probability B := by
  46   let extend (U : Finset E) (τ : ∀ e : U,State e) (e : E) : State e :=
  47     if he : e∈U then τ ⟨e,he⟩ else Classical.choice (finiteLaw_nonempty (μ e))
  48   have hext (U : Finset E) (σ : ∀ e,State e) (e : E) (he : e∈U) :
  49       σ e=extend U (fun e : U=>σ e) e := by simp only [extend,dif_pos he]
  50   have ha (σ : ∀ e,State e) : A (extend S (fun e : S=>σ e)) ↔ A σ :=
  51     (hA σ _ (hext S σ)).symm
  52   have hb (σ : ∀ e,State e) : B (extend T (fun e : T=>σ e)) ↔ B σ :=
  53     (hB σ _ (hext T σ)).symm
  54   have hh := finiteLaw_disjoint_coordinate_events μ S T hST (fun τ=>A (extend S τ))
  55     (fun τ=>B (extend T τ))
  56   simpa only [ha,hb] using hh
