  15 theorem wheelFamily_injective_hub_degree (W : I→OrdinaryWheel Q F H r)
  16     (h : Function.Injective (fun i=>ordinaryWheelHub (W i))) (u : U) :
  17     (Finset.univ.filter (fun v=>wheelFamilySpokes W v u)).card≤r := by
  18   let J:=Finset.univ.filter (fun i=>ordinaryWheelHub (W i)=u)
  19   have he : Finset.univ.filter (fun v=>wheelFamilySpokes W v u)=
  20       J.biUnion (fun i=>(ordinaryWheelRim (W i)).support.toFinset) := by
  21     ext v
  22     simp only [Finset.mem_filter,Finset.mem_univ,true_and,wheelFamilySpokes,
  23       Finset.mem_biUnion,J,Set.mem_toFinset]
  24   have hJ : J.card≤1 := by
  25     apply Finset.card_le_one.mpr
  26     intro i hi j hj
  27     exact h ((Finset.mem_filter.mp hi).2.trans (Finset.mem_filter.mp hj).2.symm)
  28   rw [he]
  29   calc
  30     _ ≤ ∑ i∈J,(ordinaryWheelRim (W i)).support.toFinset.card:=Finset.card_biUnion_le
  31     _ = J.card*r := by
  32       have hc (i : I) : (ordinaryWheelRim (W i)).support.toFinset.card=r:=(W i).val.1.val.property.support_card
  33       simp only [hc,Finset.sum_const,smul_eq_mul]
  34     _ ≤ r := by nlinarith only [Nat.mul_le_mul_right r hJ]
