  15 def markedRowRim (a : Bool×Bool×V) (z : V) : (V×V)⊕(V×V) :=
  16   let xy := if a.2.1 then (z,a.2.2) else (a.2.2,z)
  17   if a.1 then Sum.inr xy else Sum.inl xy

  19 def markedRowResources (S : Finset (Bool×Bool×V)) (J : Finset (Bool×U))
  20     (z : V) : Finset (MarkedResource V U) :=
  21   (S.image (fun a=>Sum.inl (markedRowRim a z))) ∪
  22     (J.image (fun a=>Sum.inr (a.1,z,a.2)))

  24 theorem markedRowResources_rim (S : Finset (Bool×Bool×V)) (J : Finset (Bool×U))
  25     (z : V) (e : (V×V)⊕(V×V)) :
  26     Sum.inl e∈markedRowResources S J z ↔ ∃ a∈S,markedRowRim a z=e := by
  27   simp [markedRowResources]

  29 theorem markedRowResources_spoke (S : Finset (Bool×Bool×V)) (J : Finset (Bool×U))
  30     (z : V) (a : Bool×V×U) :
  31     Sum.inr a∈markedRowResources S J z ↔ a.2.1=z ∧ (a.1,a.2.2)∈J := by
  32   constructor
  33   · intro ha
  34     rcases Finset.mem_union.mp ha with ha|ha
  35     · obtain ⟨x,hx,he⟩ := Finset.mem_image.mp ha
  36       cases he
  37     · obtain ⟨x,hx,he⟩ := Finset.mem_image.mp ha
  38       have he' := Sum.inr.inj he
  39       have hz := congrArg (fun y : Bool×V×U=>y.2.1) he'
  40       have hx' := congrArg (fun y : Bool×V×U=>(y.1,y.2.2)) he'
  41       exact ⟨hz.symm,hx' ▸ hx⟩
  42   · intro ha
  43     apply Finset.mem_union_right
  44     apply Finset.mem_image.mpr
  45     refine ⟨(a.1,a.2.2),ha.2,?_⟩
  46     apply congrArg Sum.inr
  47     exact Prod.ext rfl (Prod.ext ha.1.symm rfl)

  53 theorem markedRowRim_variable (a : Bool×Bool×V) (z : V) :
  54     (Sum.elim Prod.fst Prod.fst (markedRowRim a z)=z) ∨
  55       (Sum.elim Prod.snd Prod.snd (markedRowRim a z)=z) := by
  56   obtain ⟨b,c,x⟩ := a
  57   cases b <;> cases c <;> simp [markedRowRim]

  59 theorem markedRowResources_incidence (S : Finset (Bool×Bool×V)) (J : Finset (Bool×U))
  60     (C : Finset V) (e : MarkedResource V U) :
  61     (C.filter (fun z=>e∈markedRowResources S J z)).card≤2 := by
  62   cases e with
  63   | inl e=>
  64     have hsub : C.filter (fun z=>Sum.inl e∈markedRowResources S J z)⊆
  65         {Sum.elim Prod.fst Prod.fst e,Sum.elim Prod.snd Prod.snd e} := by
  66       intro z hz
  67       obtain ⟨a,ha,he⟩ := (markedRowResources_rim S J z e).mp (Finset.mem_filter.mp hz).2
  68       have hh := markedRowRim_variable a z
  69       rw [he] at hh
  70       simpa only [Finset.mem_insert,Finset.mem_singleton,eq_comm] using hh
  71     exact (Finset.card_le_card hsub).trans Finset.card_le_two
  72   | inr a=>
  73     have hsub : C.filter (fun z=>Sum.inr a∈markedRowResources S J z)⊆{a.2.1} := by
  74       intro z hz
  75       exact Finset.mem_singleton.mpr ((markedRowResources_spoke S J z a).mp (Finset.mem_filter.mp hz).2).1.symm
  76     have hh := Finset.card_le_card hsub
  77     simp only [Finset.card_singleton] at hh
  78     omega

  88 theorem markedRowResources_disjoint (π : Equiv.Perm V) (d : ℕ)
