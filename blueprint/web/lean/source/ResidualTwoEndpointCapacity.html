  11 theorem disjoint_walk_endpoints_card_le (F:SimpleGraph V) (s:Finset I)
  12     (u v:I→V) (P:∀ i,F.Walk (u i) (v i))
  13     (hne:∀i∈s,u i≠v i)
  14     (hP:(↑s:Set I).PairwiseDisjoint (fun i=>(P i).support.toFinset))
  15     (A:Finset V) (hAu:∀i∈s,u i∈A) (hAv:∀i∈s,v i∈A):2*s.card≤A.card := by
  16   let ends:I→Finset V:=fun i=>{u i,v i}
  17   have hends : (↑s:Set I).PairwiseDisjoint ends := by
  18     intro i hi j hj hij
  19     apply (hP hi hj hij).mono <;>
  20       intro x hx <;>
  21       simp only [ends,Finset.mem_insert,Finset.mem_singleton] at hx <;>
  22       rcases hx with rfl|rfl <;> simp
  23   have hsub:s.biUnion ends⊆A := by
  24     intro x hx
  25     obtain ⟨i,hi,hx⟩:=Finset.mem_biUnion.mp hx
  26     simp only [ends,Finset.mem_insert,Finset.mem_singleton] at hx
  27     rcases hx with rfl|rfl
  28     · exact hAu i hi
  29     · exact hAv i hi
  30   have hcount:(s.biUnion ends).card=2*s.card := by
  31     rw [Finset.card_biUnion hends]
  32     have hh:∀ i∈s,(ends i).card=2 := by
  33       intro i hi
  34       simp [ends,hne i hi]
  35     calc
  36       _ = ∑ i∈s,(2:ℕ) := by
  37         apply Finset.sum_congr rfl
  38         intro i hi
  39         exact hh i hi
  40       _ = _ := by simp [mul_comm]
  41   rw [←hcount]
  42   exact Finset.card_le_card hsub
  43 end Oberwolfach
