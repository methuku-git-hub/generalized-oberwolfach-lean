  18 theorem profile_roots_inside_card (a : I → ℕ) (c : ℕ) (ha : ∑ i,a i=c)
  19     (Y : Finset (Σ i,V i)) :
  20     Fintype.card {ρ : ∀ i,FixedSubset (V i) (a i) // profileRootSet ρ⊆Y} ≤ Y.card^c := by
  21   let A := {ρ : ∀ i,FixedSubset (V i) (a i) // profileRootSet ρ⊆Y}
  22   let B := {X : Finset (Σ i,V i) // X∈Y.powersetCard c}
  23   let f (ρ : A) : B := ⟨profileRootSet ρ.val,Finset.mem_powersetCard.mpr
  24     ⟨ρ.property,(profileRootSet_card ρ.val).trans ha⟩⟩
  25   have hi : Function.Injective f := by
  26     intro ρ τ he
  27     apply Subtype.ext
  28     exact profileRootSet_injective a (congrArg (fun x : B => x.val) he)
  29   have hh := Fintype.card_le_of_injective f hi
  30   have hb : Fintype.card B=Y.card.choose c := by simp only [B,Fintype.card_coe,Finset.card_powersetCard]
  31   rw [hb] at hh
  32   exact hh.trans (Nat.choose_le_pow _ _)

  34 theorem profile_roots_common_event {Ω : Type*} [Fintype Ω]
  35     (μ : FiniteLaw Ω) (a : I → ℕ) (c : ℕ) (ha : ∑ i,a i=c)
  36     (Y : Ω → Finset (Σ i,V i)) (M : ℕ) (hM : ∀ ω,(Y ω).card ≤ M)
  37     (Bad : (∀ i,FixedSubset (V i) (a i)) → Ω → Prop) (ε : ℝ) (hε : 0 ≤ ε)
  38     (h : ∀ ρ,μ.probability (fun ω => profileRootSet ρ⊆Y ω ∧ Bad ρ ω) ≤
  39       ε*μ.probability (fun ω => profileRootSet ρ⊆Y ω)) :
  40     1-ε*(M:ℝ)^c ≤ μ.probability (fun ω => ∀ ρ,profileRootSet ρ⊆Y ω → ¬Bad ρ ω) := by
  41   have hb (ω : Ω) :
  42       ((Finset.univ.filter (fun ρ : ∀ i,FixedSubset (V i) (a i) => profileRootSet ρ⊆Y ω)).card:ℝ) ≤ (M:ℝ)^c := by
  43     have hh := (profile_roots_inside_card a c ha (Y ω)).trans (Nat.pow_le_pow_left (hM ω) c)
  44     simp only [Fintype.card_subtype] at hh
  45     exact_mod_cast hh
  46   have hh := finiteLaw_eligible_common_event μ Finset.univ
  47     (fun ρ ω => profileRootSet ρ⊆Y ω) Bad ((M:ℝ)^c) ε hε (fun ω => by convert hb ω using 1 <;> congr!) (fun ρ _ => h ρ)
  48   simpa only [Finset.mem_univ,forall_const] using hh
