  14 theorem sampledColourOrientation_large_cuts_exponential
  15     (α pstar : ℝ) (hα : 0<α) (hα1 : α≤1/2) (hpstar : 0<pstar) (k : ℕ) :
  16     ∃ N : ℕ,1≤N ∧ ∀ (V : Type) [Fintype V] [LinearOrder V],N≤Fintype.card V →
  17     ∀ (P : SimpleGraph V),(∀ v,(1/2+α)*Fintype.card V≤(P.degree v:ℝ)) →
  18     ∀ (ρ : Fin k→ℝ) (hρ : ∀ a,0≤ρ a) (hsum : ∑ a,ρ a=1),
  19     1-8*k*Fintype.card V*Real.exp
  20       (-min (2*(pstar/32)^2) ((α*pstar^2/1024)^2/4)*Fintype.card V)≤
  21     (FiniteLaw.pi (fun _ : Sym2 V=>FiniteLaw.categoricalDirection ρ hρ hsum)).probability
  22       (fun σ=>∀ a : Fin k,pstar≤ρ a → ∀ δ : ℝ,δ≤α/32 → δ≤ρ a/64 → ∀ S T : Finset V,
  23         (1-2*δ)*Fintype.card V<(S.card:ℝ)+T.card →
  24         (α*ρ a/4)*Fintype.card V*min (S.card:ℝ) T.card≤
  25           (matrixCut (sampledColourOrientation P σ a) S T:ℝ)) := by
  26   let ε : ℝ := pstar/32
  27   let q : ℝ := α*pstar^2/1024
  28   let c := min (2*ε^2) (q^2/4)
  29   have hε : 0<ε := by dsimp [ε];positivity
  30   have hq : 0<q := by dsimp [q];positivity
  31   obtain ⟨N,hN1,hN⟩ := sampledColourCut_quadratic_exponential_event k hq
  32   refine ⟨N,hN1,?_⟩
  33   intro V _ _ hnN P hP ρ hρ hsum
  34   let n := Fintype.card V
  35   have hn : 0<n := by dsimp [n];omega
  36   have hn0 : (0:ℝ)<n := by exact_mod_cast hn
  37   have hn1 : (1:ℝ)≤n := by exact_mod_cast (show 1≤n by dsimp [n];omega)
  38   let μ := FiniteLaw.pi (fun _ : Sym2 V=>FiniteLaw.categoricalDirection ρ hρ hsum)
  39   have hd := sampledColourDegrees_common_event P P ρ hρ hsum (t:=ε*n) (by positivity)
  40   have hcuts := hN V hnN P ρ hρ hsum
  41   have hjoint := finiteLaw_good_intersection μ
  42     (fun σ=>∀ a v,
  43       |((∑ y,sampledColourOrientation P σ a v y:ℕ):ℝ)-ρ a/2*P.degree v|<ε*n ∧
  44       |((∑ y,sampledColourOrientation P σ a y v:ℕ):ℝ)-ρ a/2*P.degree v|<ε*n ∧
  45       |(sampledColourDegree P (fun e=>(σ e).1) a v:ℝ)-ρ a*P.degree v|<ε*n)
  46     _ (2*k*Real.exp (-(q^2/4)*n)) hcuts
  47   have hbase := (sub_le_sub_right hd (2*k*Real.exp (-(q^2/4)*n))).trans hjoint
  48   have he : -2*(ε*n)^2/n=-(2*ε^2)*n := by field_simp [ne_of_gt hn0] <;> ring
  49   have heD : Real.exp (-(2*ε^2)*n)≤Real.exp (-c*n) := by
  50     apply Real.exp_le_exp.mpr
  51     have h := mul_le_mul_of_nonneg_right (min_le_left (2*ε^2) (q^2/4)) hn0.le
  52     change c*n≤(2*ε^2)*n at h
  53     linarith only [h]
  54   have heC : Real.exp (-(q^2/4)*n)≤Real.exp (-c*n) := by
  55     apply Real.exp_le_exp.mpr
  56     have h := mul_le_mul_of_nonneg_right (min_le_right (2*ε^2) (q^2/4)) hn0.le
  57     change c*n≤(q^2/4)*n at h
  58     linarith only [h]
  59   have hbD : 6*k*n*Real.exp (-2*(ε*n)^2/n)≤6*k*n*Real.exp (-c*n) := by
  60     rw [he]
  61     exact mul_le_mul_of_nonneg_left heD (by positivity)
  62   have hbC : 2*k*Real.exp (-(q^2/4)*n)≤2*k*n*Real.exp (-c*n) := by
  63     apply mul_le_mul _ heC (by positivity) (by positivity)
  64     have h := mul_le_mul_of_nonneg_left hn1 (show (0:ℝ)≤2*k by positivity)
  65     simpa only [mul_one] using h
  66   have hbound : 1-8*k*n*Real.exp (-c*n)≤
  67       (1-6*k*n*Real.exp (-2*(ε*n)^2/n))-2*k*Real.exp (-(q^2/4)*n) := by
  68     linarith only [hbD,hbC]
  69   apply (hbound.trans hbase).trans
  70   apply finiteLaw_probability_mono
  71   intro σ hσ a ha δ hδα hδρ S T hlarge
  72   have hrow (v : V) : ρ a*n/4-ε*n≤((∑ y,sampledColourOrientation P σ a v y:ℕ):ℝ) := by
  73     have hh := (abs_lt.mp (hσ.1 a v).1).1
  74     have hdeg := mul_le_mul_of_nonneg_left (hP v) (div_nonneg (hρ a) (by norm_num : (0:ℝ)≤2))
  75     have hpos := mul_nonneg (hρ a) (mul_nonneg hα.le hn0.le)
  76     nlinarith only [hh,hdeg,hpos]
  77   have hcol (v : V) : ρ a*n/4-ε*n≤((∑ y,sampledColourOrientation P σ a y v:ℕ):ℝ) := by
  78     have hh := (abs_lt.mp (hσ.1 a v).2.1).1
  79     have hdeg := mul_le_mul_of_nonneg_left (hP v) (div_nonneg (hρ a) (by norm_num : (0:ℝ)≤2))
  80     have hpos := mul_nonneg (hρ a) (mul_nonneg hα.le hn0.le)
  81     nlinarith only [hh,hdeg,hpos]
  82   have hh := matrixCut_large_of_degree_and_quadratic_strong P (sampledColourOrientation P σ a)
  83     hα.le hα1 (hρ a) hδα hδρ
  84     (by dsimp [ε,n];nlinarith only [mul_le_mul_of_nonneg_right ha hn0.le])
  85     (by
  86       have hs : pstar^2≤(ρ a)^2 := by nlinarith only [ha,hpstar,hρ a]
  87       dsimp [q]
  88       nlinarith only [mul_le_mul_of_nonneg_left hs hα.le])
  89     (by intro x y;unfold sampledColourOrientation;split_ifs <;> omega)
  90     hP hrow hcol (fun X Y=>hσ.2 a X Y) S T hlarge
  91   exact hh
