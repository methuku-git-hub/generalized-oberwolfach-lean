  15 theorem ordinaryAmbientMap_support_location (G : SimpleGraph A) (i : A ↪ V) (j : B ↪ V)
  16     (f : A → B) (R : Finset V) (hi : ∀ a,i a∈R) (hj : ∀ b,j b∈R) :
  17     ∀ v∈(G.map i).support,v∈R ∧ ordinaryAmbientMap i j f v∈R := by
  18   intro v hv
  19   rw [SimpleGraph.support_map] at hv
  20   obtain ⟨a,ha,rfl⟩ := hv
  21   exact ⟨hi a,by rw [ordinaryAmbientMap_apply];exact hj (f a)⟩

  23 theorem ordinaryAmbientMap_disjoint_support (G : SimpleGraph A) (i : A ↪ V) (j : B ↪ V)
  24     (f : A → B) (hd : ∀ a b,i a≠j b) :
  25     ∀ v∈(G.map i).support,∀ w∈(G.map i).support,ordinaryAmbientMap i j f v≠w := by

  41 theorem ordinaryAmbientMap_edge_roots_separated (G : SimpleGraph A) (F : SimpleGraph V)
  42     (i : A ↪ V) (j : B ↪ V) (f : A → B)
  43     (h₀ : ∀ a b,G.Adj a b→¬F.Adj (i a) (i b))
  44     (h₁ : ∀ a b,G.Adj a b→¬F.Adj (j (f a)) (j (f b)))
  45     (hcross : ∀ a b,¬F.Adj (i a) (j b)) :
  46     ∀ v w,(G.map i).Adj v w→
  47       ∀ a∈({v,w,ordinaryAmbientMap i j f v,ordinaryAmbientMap i j f w}:Finset V),
  48       ∀ b∈({v,w,ordinaryAmbientMap i j f v,ordinaryAmbientMap i j f w}:Finset V),¬F.Adj a b := by
  49   intro v w hvw
  50   obtain ⟨x,y,hxy,rfl,rfl⟩ := (SimpleGraph.map_adj _ _ _ _).mp hvw
  51   rw [ordinaryAmbientMap_apply,ordinaryAmbientMap_apply]
  52   have hxx : ¬F.Adj (i x) (i x) := fun h=>h.ne rfl
  53   have hyy : ¬F.Adj (i y) (i y) := fun h=>h.ne rfl
  54   have hffx : ¬F.Adj (j (f x)) (j (f x)) := fun h=>h.ne rfl
  55   have hffy : ¬F.Adj (j (f y)) (j (f y)) := fun h=>h.ne rfl
  56   intro a ha b hb
  57   simp only [Finset.mem_insert,Finset.mem_singleton] at ha hb
  58   rcases ha with rfl|rfl|rfl|rfl <;> rcases hb with rfl|rfl|rfl|rfl
  59   all_goals first | assumption | exact h₀ x y hxy | exact h₀ y x hxy.symm |
  60     exact h₁ x y hxy | exact h₁ y x hxy.symm | exact hcross _ _ |
  61     exact fun h=>hcross _ _ h.symm
