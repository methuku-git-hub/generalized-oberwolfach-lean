  12 theorem vortex_ambient_subset_step {ε : ℝ} (he : 0 < ε) (he1 : ε ≤ 1) :
  13     ∃ N : ℕ, ∀ (V : Type w) (B R : Finset V) (A : V → Finset V),
  14       N ≤ B.card → (R.card : ℝ) ≤ 2 * B.card / ε →
  15       ∃ C : Finset V, C ⊆ B ∧ C.card = Nat.floor (ε * B.card) ∧
  16         ∀ x ∈ R, |((A x ∩ C).card : ℝ) / C.card -
  17           ((A x ∩ B).card : ℝ) / B.card| < (B.card : ℝ) ^ (-1 / 3 : ℝ) := by
  18   obtain ⟨N, hN⟩ := vortex_uniform_subset_step.{w, w} he he1
  19   refine ⟨N, ?_⟩
  20   intro V B R A hn hR
  21   let : DecidableEq {x // x ∈ B} := Classical.decEq _
  22   let A' (x : V) := (A x).subtype (fun y => y ∈ B)
  23   obtain ⟨S, hS⟩ := hN {y // y ∈ B} (by simpa using hn) V R A' (by simpa using hR)
  24   let C := S.1.map (Function.Embedding.subtype (fun y => y ∈ B))
  25   have hc : C.card = Nat.floor (ε * B.card) := by
  26     simpa only [C, Finset.card_map, Fintype.card_coe] using S.2
  27   refine ⟨C, subtype_sample_subset B S.1, hc, ?_⟩
  28   intro x hx
  29   have hh := hS x hx
  30   dsimp only [A'] at hh
  31   rw [subtype_sample_intersection_card (A x) B S.1, subtype_restriction_card (A x) B] at hh
  32   simpa only [Fintype.card_coe, ← hc] using hh

  34 theorem vortex_graph_step {ε : ℝ} (he : 0 < ε) (he1 : ε ≤ 1) :
  35     ∃ N : ℕ, ∀ (V : Type w) [Fintype V] (G : SimpleGraph V) (B R : Finset V),
  36       N ≤ B.card → (R.card : ℝ) ≤ 2 * B.card / ε →
  37       ∃ C : Finset V, C ⊆ B ∧ C.card = Nat.floor (ε * B.card) ∧
  38         ∀ x ∈ R, |((G.neighborFinset x ∩ C).card : ℝ) / C.card -
  39           ((G.neighborFinset x ∩ B).card : ℝ) / B.card| < (B.card : ℝ) ^ (-1 / 3 : ℝ) := by
  40   obtain ⟨N, hN⟩ := vortex_ambient_subset_step.{w} he he1
  41   exact ⟨N, fun V _ G B R hn hR => hN V B R (fun x => G.neighborFinset x) hn hR⟩
