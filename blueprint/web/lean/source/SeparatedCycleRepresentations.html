  11 theorem permutationSeparated_iff_range {V I:Type} (π:Equiv.Perm V) (b:ℕ) (f:I→V)
  12     (hf:Function.Injective f):
  13     PermutationSeparated π b f ↔ ∀ x∈Set.range f,∀ y∈Set.range f,x≠y→¬PermutationClose π b x y:=by
  14   constructor
  15   · intro h x hx y hy hxy
  16     obtain ⟨i,rfl⟩:=hx
  17     obtain ⟨j,rfl⟩:=hy
  18     exact h i j (fun hij=>hxy (congrArg f hij))
  19   · intro h i j hij
  20     exact h (f i) ⟨i,rfl⟩ (f j) ⟨j,rfl⟩ (fun hh=>hij (hf hh))
