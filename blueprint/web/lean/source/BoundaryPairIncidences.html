  14 theorem disjoint_embedding_pair_unique {A:Finset V} {B:Finset U}
  15     (v w:B↪A) (hdis:Disjoint (Set.range v) (Set.range w)) (a:V) (u z:B)
  16     (hu:a=(v u).val∨a=(w u).val) (hz:a=(v z).val∨a=(w z).val):u=z:=by
  17   rcases hu with hu|hu <;> rcases hz with hz|hz
  18   · exact v.injective (Subtype.ext (hu.symm.trans hz))
  19   · exact (Set.disjoint_left.mp hdis ⟨u,rfl⟩ ⟨z,Subtype.ext (hz.symm.trans hu)⟩).elim

  21   · exact w.injective (Subtype.ext (hu.symm.trans hz))
