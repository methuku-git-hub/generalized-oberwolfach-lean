  13 theorem ordinary_normalized_admissible_vertex_load {V:Type} [Fintype V] [DecidableEq V]
  14     (P F:SimpleGraph V) (r:ℕ) (hr:2≤r) (hn:0<Fintype.card V)
  15     (weight:OrdinaryCycle P r→ℝ) (hweight:∀ C,0≤weight C)
  16     (B:ℝ) (hB:0≤B) (hb:∀ C,weight C≤B/(Fintype.card V:ℝ)^(r-2))
  17     (L:ℝ) (hdeg:∀ v,(F.degree v:ℝ)≤L)
  18     (he:∀ x y,(∑ C,if C.val.val.Adj x y then weight C else 0)=if P.Adj x y then 1 else 0)
  19     (D ε:ℝ) (hD:0<D) (hnear:∀ v,|(P.degree v:ℝ)-D|≤ε*Fintype.card V) (v:V):
  20     |(2/D)*(∑ C,if v∈C.val.val.support ∧
  21         (∀ x∈C.val.val.support,∀ y∈C.val.val.support,¬F.Adj x y) then weight C else 0)-1|≤
  22       (6*(r:ℝ)^4*B*L+ε*Fintype.card V)/D := by
  23   let S:ℝ:=∑ C,if v∈C.val.val.support ∧
  24     (∀ x∈C.val.val.support,∀ y∈C.val.val.support,¬F.Adj x y) then weight C else 0
  25   have hs:|S-(P.degree v:ℝ)/2|≤3*(r:ℝ)^4*B*L:=by
  26     have hh:=ordinary_admissible_cycle_vertex_load P F r hr hn weight hweight B hB hb L hdeg he v
  27     convert hh using 1 <;> congr!
  28   have ha:=abs_le.mp hs
  29   have hd:=abs_le.mp (hnear v)
  30   have hraw:|2*S-D|≤6*(r:ℝ)^4*B*L+ε*Fintype.card V:=by
  31     apply abs_le.mpr
  32     constructor <;> linarith only [ha.1,ha.2,hd.1,hd.2]
  33   have heq:(2/D)*S-1=(2*S-D)/D:=by field_simp
  34   change |(2/D)*S-1|≤_
  35   rw [heq,abs_div,abs_of_pos hD]
  36   exact div_le_div_of_nonneg_right hraw hD.le
  37 end Oberwolfach
