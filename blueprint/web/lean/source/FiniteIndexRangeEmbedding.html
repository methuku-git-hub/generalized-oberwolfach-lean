   7 theorem finite_index_range_embedding
   8     {V W I:Type*} [DecidableEq V] [Fintype I] (source:I↪V) (target:I↪W):
   9     ∃f:{x // x∈Finset.univ.image source}↪W,
  10       ∀k:I,f ⟨source k,Finset.mem_image.mpr ⟨k,Finset.mem_univ _,rfl⟩⟩=target k:=by
  11   classical
  12   let idx (x:{x // x∈Finset.univ.image source}):I:=Classical.choose (Finset.mem_image.mp x.property)
  13   have hidx (x:{x // x∈Finset.univ.image source}):source (idx x)=x.val:=
  14     (Classical.choose_spec (Finset.mem_image.mp x.property)).2
  15   let φ (x:{x // x∈Finset.univ.image source}):W:=target (idx x)
  16   have hinj:Function.Injective φ:=by
  17     intro x y hxy
  18     have hidxeq:idx x=idx y:=target.injective hxy
  19     apply Subtype.ext
  20     calc
  21       x.val=source (idx x):=(hidx x).symm
  22       _=source (idx y):=congrArg source hidxeq
  23       _=y.val:=hidx y
  24   refine ⟨⟨φ,hinj⟩,?_⟩
  25   intro k
  26   let x:{x // x∈Finset.univ.image source}:=⟨source k,Finset.mem_image.mpr ⟨k,Finset.mem_univ _,rfl⟩⟩
  27   change target (idx x)=target k
  28   congr 1
  29   apply source.injective
  30   exact hidx x
