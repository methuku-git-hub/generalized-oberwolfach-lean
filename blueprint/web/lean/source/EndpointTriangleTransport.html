  15 def endpointTriangleLift (e:W↪V) (Q F:SimpleGraph V) (S T:Finset V)
  16     (C:EndpointTriangle (Q.comap e) (F.comap e) (embeddingPullFinset e S) (embeddingPullFinset e T)) :
  17     EndpointTriangle Q F S T := by
  18   refine ⟨C.val.image e,?_,?_,?_,?_,?_⟩
  19   · rintro v hv
  20     obtain ⟨w,hw,rfl⟩:=Finset.mem_image.mp hv
  21     simpa only [Finset.mem_union,embeddingPullFinset_mem] using C.property.1 hw
  22   · have he:(C.val.image e)∩S=(C.val∩embeddingPullFinset e S).image e:=by
  23       ext v;simp only [Finset.mem_inter,Finset.mem_image,embeddingPullFinset_mem]
  24       constructor
  25       · rintro ⟨⟨w,hw,rfl⟩,hs⟩;exact ⟨w,⟨hw,hs⟩,rfl⟩
  26       · rintro ⟨w,⟨hw,hs⟩,rfl⟩;exact ⟨⟨w,hw,rfl⟩,hs⟩
  27     rw [he,Finset.card_image_of_injective _ e.injective]
  28     exact C.property.2.1
  29   · have he:(C.val.image e)∩T=(C.val∩embeddingPullFinset e T).image e:=by
  30       ext v;simp only [Finset.mem_inter,Finset.mem_image,embeddingPullFinset_mem]
  31       constructor
  32       · rintro ⟨⟨w,hw,rfl⟩,ht⟩;exact ⟨w,⟨hw,ht⟩,rfl⟩
  33       · rintro ⟨w,⟨hw,ht⟩,rfl⟩;exact ⟨⟨w,hw,rfl⟩,ht⟩
  34     rw [he,Finset.card_image_of_injective _ e.injective]
  35     exact C.property.2.2.1
  36   · rintro v hv w hw hne
  37     obtain ⟨v',hv',rfl⟩:=Finset.mem_image.mp hv
  38     obtain ⟨w',hw',rfl⟩:=Finset.mem_image.mp hw
  39     exact C.property.2.2.2.1 v' hv' w' hw' (fun he=>hne (congrArg e he))
  40   · rintro v hv w hw
  41     obtain ⟨v',hv',rfl⟩:=Finset.mem_image.mp hv
  42     obtain ⟨w',hw',rfl⟩:=Finset.mem_image.mp hw
  43     exact C.property.2.2.2.2 v' hv' w' hw'

  49 theorem endpointTriangleLift_bijective (e:W↪V) (Q F:SimpleGraph V) (S T:Finset V)
  50     (hS:∀ v∈S,∃ w,e w=v) (hT:∀ v∈T,∃ w,e w=v) :
  51     Function.Bijective (endpointTriangleLift e Q F S T) := by
  52   constructor
  53   · intro C D h
  54     apply Subtype.ext
  55     have hh:=congrArg Subtype.val h
  56     change C.val.image e=D.val.image e at hh
  57     have hi:=congrArg (embeddingPullFinset e) hh
  58     simpa using hi
  59   · intro C
  60     have hsupp:∀ v∈C.val,∃ w,e w=v:=by
  61       intro v hv
  62       exact (Finset.mem_union.mp (C.property.1 hv)).elim (hS v) (hT v)
  63     let X:=embeddingPullFinset e C.val
  64     have hXS:X∩embeddingPullFinset e S=embeddingPullFinset e (C.val∩S):=by
  65       ext v;simp [X]
  66     have hXT:X∩embeddingPullFinset e T=embeddingPullFinset e (C.val∩T):=by
  67       ext v;simp [X]
  68     let hX:EndpointTriangle (Q.comap e) (F.comap e) (embeddingPullFinset e S) (embeddingPullFinset e T):=by
  69       refine ⟨X,?_,?_,?_,?_,?_⟩
  70       · intro w hw
  71         simpa only [Finset.mem_union,embeddingPullFinset_mem] using
  72           C.property.1 ((embeddingPullFinset_mem _ _ _).mp hw)
  73       · rw [hXS,embeddingPullFinset_card e (C.val∩S) (fun v hv=>hS v (Finset.mem_inter.mp hv).2)]
  74         exact C.property.2.1
  75       · rw [hXT,embeddingPullFinset_card e (C.val∩T) (fun v hv=>hT v (Finset.mem_inter.mp hv).2)]
  76         exact C.property.2.2.1
  77       · intro v hv w hw hne
  78         exact C.property.2.2.2.1 (e v) ((embeddingPullFinset_mem _ _ _).mp hv)
  79           (e w) ((embeddingPullFinset_mem _ _ _).mp hw) (e.injective.ne hne)
  80       · intro v hv w hw
  81         exact C.property.2.2.2.2 (e v) ((embeddingPullFinset_mem _ _ _).mp hv)
  82           (e w) ((embeddingPullFinset_mem _ _ _).mp hw)
  83     refine ⟨hX,Subtype.ext ?_⟩
  84     exact embeddingPullFinset_image e C.val hsupp

  92 theorem endpointTriangleLift_weighted_degree (e:W↪V) (Q F:SimpleGraph V) (S T:Finset V)
  93     (hS:∀ v∈S,∃ w,e w=v) (hT:∀ v∈T,∃ w,e w=v)
  94     (wt:EndpointTriangle Q F S T→ℝ) (v:W) :
  95     hypergraphWeightedDegree endpointTriangleVertices (fun C=>wt (endpointTriangleLift e Q F S T C)) v=
  96       hypergraphWeightedDegree endpointTriangleVertices wt (e v) := by
  97   let E:=endpointTriangleEquiv e Q F S T hS hT
  98   have hh:=Equiv.sum_comp E (fun C=>if e v∈endpointTriangleVertices C then wt C else 0)
  99   unfold hypergraphWeightedDegree

 103   have he:(e v∈endpointTriangleVertices (E C)) ↔ v∈endpointTriangleVertices C:=by
