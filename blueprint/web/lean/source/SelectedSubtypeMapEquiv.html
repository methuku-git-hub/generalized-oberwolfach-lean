  10     (hPQ : ∀ a, Q (e a) ↔ P a) :
  11     {a : A // a ∈ M ∧ P a} ≃ {b : B // b ∈ M.map e ∧ Q b} := by
  12   classical
  13   apply Equiv.ofBijective (fun a => ⟨e a.val, Finset.mem_map.mpr ⟨a.val, a.property.1, rfl⟩,
  14     (hPQ a.val).2 a.property.2⟩)
  15   constructor
  16   · intro x y h
  17     apply Subtype.ext
  18     exact e.injective (congrArg Subtype.val h)
  19   · intro y
  20     obtain ⟨a, ha, hay⟩ := Finset.mem_map.mp y.property.1
  21     have hp : P a := (hPQ a).1 (by simpa [hay] using y.property.2)
  22     refine ⟨⟨a, ha, hp⟩, ?_⟩
  23     exact Subtype.ext hay
