  13 abbrev transferConstructionGraph (r : ℕ) (hr : 3 ≤ r) :=

  16 abbrev transferRowStart (r i : ℕ) := r+1+i*(r-1)

  18 theorem transferPredecessors_row_formula (r i j : ℕ) (hr : 3 ≤ r) (hj : j < r-1) :
  19     transferPredecessors r (transferRowStart r i+j)=
  20       if j=0 then {0,i+1} else if j=1 then
  21         if r=3 then {0,i+1,transferRowStart r i} else {0,i+1}
  22       else {if j=2 then transferRowStart r i else transferRowStart r i+j-1}∪
  23         (if j=r-2 then {transferRowStart r i+1} else ∅) := by
  24   have hz : transferRowStart r i+j≠0 := by unfold transferRowStart;omega
  25   have hzr : ¬transferRowStart r i+j≤r := by unfold transferRowStart;omega
  26   have hsub : transferRowStart r i+j-r-1=(r-1)*i+j := by simp only [transferRowStart,Nat.mul_comm];omega
  27   have hdiv : ((r-1)*i+j)/(r-1)=i := by
  28     rw [Nat.mul_add_div (by omega),Nat.div_eq_of_lt hj,Nat.add_zero]
  29   have hmod : ((r-1)*i+j)%(r-1)=j := by rw [Nat.mul_add_mod_self_left,Nat.mod_eq_of_lt hj]
  30   unfold transferPredecessors
  31   rw [if_neg hz,if_neg hzr]
  32   dsimp only
  33   rw [hsub,hdiv,hmod]
  34   split_ifs <;> ext x <;> simp only [Finset.mem_union,Finset.mem_insert,Finset.mem_singleton,Finset.notMem_empty,Nat.add_sub_cancel]
  35   all_goals omega

  53 theorem transferConstructionGraph_row_endpoints (r i : ℕ) (hr : 3 ≤ r) :
  54     (transferConstructionGraph r hr).Adj 0 (transferRowStart r i) ∧
  55     (transferConstructionGraph r hr).Adj (i+1) (transferRowStart r i) ∧
  56     (transferConstructionGraph r hr).Adj 0 (transferRowStart r i+1) ∧
  57     (transferConstructionGraph r hr).Adj (i+1) (transferRowStart r i+1) := by
  58   have h0 := transferPredecessors_row_formula r i 0 hr (by omega)
  59   have h1 := transferPredecessors_row_formula r i 1 hr (by omega)
  60   simp only [Nat.add_zero,ite_true] at h0
  61   simp only [show (1:ℕ)≠0 by omega,ite_false,ite_true] at h1
  62   refine ⟨Or.inl ?_,Or.inl ?_,Or.inl ?_,Or.inl ?_⟩
  63   · rw [h0];simp
  64   · rw [h0];simp
  65   · rw [h1];split_ifs <;> simp
  66   · rw [h1];split_ifs <;> simp

  69     (hj : 0 < j) (hjr : j+1 < r) :

  71       (transferRowStart r i+transferColumnOrder r j)
  72       (transferRowStart r i+transferColumnOrder r (j+1)) := by
