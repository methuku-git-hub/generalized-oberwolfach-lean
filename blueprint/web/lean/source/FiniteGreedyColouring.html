  14 theorem finiteGreedyColouring
  15     {V : Type*} [Fintype V] (G : SimpleGraph V) {q : ℕ} (hq : 0<q)
  16     (hdegree : ∀ v,(Finset.univ.filter (fun u=>G.Adj v u)).card<q) :
  17     ∃ f : V → Fin q,∀ u v,G.Adj u v → f u≠f v := by
  18   classical
  19   have hpartial (S : Finset V) :
  20       ∃ f : V → Fin q,∀ u∈S,∀ v∈S,G.Adj u v → f u≠f v := by
  21     induction S using Finset.induction_on with
  22     | empty =>
  23       exact ⟨fun _=>⟨0,hq⟩,by simp⟩
  24     | @insert x S hx ih =>
  25       obtain ⟨f,hf⟩ := ih
  26       let B := (S.filter (fun u=>G.Adj x u)).image f
  27       have hB : B.card<q := by
  28         calc
  29           _ ≤ (S.filter (fun u=>G.Adj x u)).card := Finset.card_image_le
  30           _ ≤ (Finset.univ.filter (fun u=>G.Adj x u)).card :=
  31             Finset.card_le_card (Finset.filter_subset_filter _ (Finset.subset_univ S))
  32           _ < _ := hdegree x
  33       obtain ⟨c,hc⟩ : ∃ c : Fin q,c∉B := by
  34         by_contra hn
  35         push Not at hn
  36         have he : B=Finset.univ := Finset.eq_univ_iff_forall.mpr hn
  37         simpa [he] using hB
  38       let g := Function.update f x c
  39       refine ⟨g,?_⟩
  40       intro u hu v hv hadj
  41       by_cases hux : u=x
  42       · subst u
  43         have hvx : v≠x := fun he=>G.irrefl (he ▸ hadj)
  44         have hvS : v∈S := (Finset.mem_insert.mp hv).resolve_left hvx
  45         have hfv : f v∈B := Finset.mem_image.mpr
  46           ⟨v,Finset.mem_filter.mpr ⟨hvS,hadj⟩,rfl⟩
  47         simp only [g,Function.update_self,Function.update_of_ne hvx]
  48         exact fun he=>hc (he.symm ▸ hfv)
  49       · have huS : u∈S := (Finset.mem_insert.mp hu).resolve_left hux
  50         by_cases hvx : v=x
  51         · subst v
  52           have hfu : f u∈B := Finset.mem_image.mpr
  53             ⟨u,Finset.mem_filter.mpr ⟨huS,hadj.symm⟩,rfl⟩
  54           simp only [g,Function.update_self,Function.update_of_ne hux]
  55           exact fun he=>hc (he ▸ hfu)
  56         · have hvS : v∈S := (Finset.mem_insert.mp hv).resolve_left hvx
  57           simp only [g,Function.update_of_ne hux,Function.update_of_ne hvx]
  58           exact hf u huS v hvS hadj
  59   obtain ⟨f,hf⟩ := hpartial Finset.univ
  60   exact ⟨f,fun u v=>hf u (Finset.mem_univ _) v (Finset.mem_univ _)⟩
  61 
  62 end Oberwolfach
