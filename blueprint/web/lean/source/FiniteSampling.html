  14 /-- A probability law on a finite sample space, given by its point masses. -/
  15 structure FiniteLaw (Ω : Type*) [Fintype Ω] where
  16   mass : Ω → ℝ
  17   nonneg : ∀ ω, 0 ≤ mass ω
  18   total : ∑ ω, mass ω = 1
  19 
  20 namespace FiniteLaw
  21 variable {Ω Γ : Type*} [Fintype Ω] [Fintype Γ]
  22 
  23 def expectation (μ : FiniteLaw Ω) (X : Ω → ℝ) : ℝ :=
  24   ∑ ω, μ.mass ω * X ω
  25 
  26 def probability (μ : FiniteLaw Ω) (A : Ω → Prop) : ℝ := by
  27   classical
  28   exact ∑ ω, if A ω then μ.mass ω else 0

  30 /-- Independent product of finite laws. -/
  31 def prod (μ : FiniteLaw Ω) (ν : FiniteLaw Γ) : FiniteLaw (Ω × Γ) where
  32   mass ω := μ.mass ω.1 * ν.mass ω.2
  33   nonneg ω := mul_nonneg (μ.nonneg _) (ν.nonneg _)
  34   total := by
  35     rw [Fintype.sum_prod_type]
  36     simp_rw [← Finset.mul_sum, ν.total, mul_one]
  37     exact μ.total
  38 
  39 /-- Independent categorical choices with possibly different state types
  40 and different probability vectors at different coordinates. -/
  41 def pi {E : Type*} [Fintype E] [DecidableEq E]
  42     {State : E → Type*} [∀ e, Fintype (State e)]
  43     (μ : ∀ e, FiniteLaw (State e)) : FiniteLaw (∀ e, State e) where
  44   mass σ := ∏ e, (μ e).mass (σ e)
  45   nonneg σ := Finset.prod_nonneg (fun e _ => (μ e).nonneg (σ e))
  46   total := by
  47     rw [← Fintype.prod_sum]
  48     simp only [(μ _).total, Finset.prod_const_one]
  49 
  50 end FiniteLaw

  53 theorem finiteLaw_expectation_pattern
  54     {E : Type*} [Fintype E] [DecidableEq E]
  55     {State : E → Type*} [∀ e, Fintype (State e)]
  56     (μ : ∀ e, FiniteLaw (State e)) (F : Finset E) (s : ∀ e, State e) :
  57     (pi μ).expectation (patternValue F s) = ∏ e ∈ F, (μ e).mass (s e) := by
  58   classical
  59   let g (e : E) (x : State e) :=
  60     if e ∈ F then (μ e).mass x * (if x = s e then 1 else 0) else (μ e).mass x
  61   have hpoint (σ : ∀ e, State e) :
  62       (∏ e, (μ e).mass (σ e)) * patternValue F s σ = ∏ e, g e (σ e) := by
  63     have hprod : patternValue F s σ =
  64         ∏ e, if e ∈ F then (if σ e = s e then (1 : ℝ) else 0) else 1 := by
  65       unfold patternValue
  66       simp
  67     rw [hprod, ← Finset.prod_mul_distrib]
  68     apply Finset.prod_congr rfl
  69     intro e _
  70     by_cases he : e ∈ F <;> simp [g, he]
  71   have hsum (e : E) : ∑ x, g e x = if e ∈ F then (μ e).mass (s e) else 1 := by
  72     by_cases he : e ∈ F
  73     · simp [g, he, mul_ite]
  74     · simp only [g, ite_eq_right he]
  75       exact (μ e).total
  76   unfold expectation
  77   change (∑ σ, (∏ e, (μ e).mass (σ e)) * patternValue F s σ) = _
  78   simp_rw [hpoint]
  79   rw [← Fintype.prod_sum]
  80   simp_rw [hsum]
  81   simp

  83 /-- Fixed candidate families and fixed coefficients may be summed after
  84 omitting root coordinates from every product. This is an unconditional
  85 expectation identity, not conditioning on root presence. -/
  86 theorem finiteLaw_expectation_omitted_root_sum
  87     {E I : Type*} [Fintype E] [DecidableEq E]
  88     {State : E → Type*} [∀ e, Fintype (State e)]
  89     (μ : ∀ e, FiniteLaw (State e)) (C : Finset I)
  90     (F : I → Finset E) (R : Finset E)
  91     (s : I → ∀ e, State e) (w : I → ℝ) :
  92     (pi μ).expectation (fun σ => ∑ i ∈ C, w i * patternValue (F i \ R) (s i) σ) =
  93       ∑ i ∈ C, w i * ∏ e ∈ F i \ R, (μ e).mass (s i e) := by
  94   unfold expectation
  95   simp_rw [Finset.mul_sum]
  96   rw [Finset.sum_comm]
  97   apply Finset.sum_congr rfl
  98   intro i _
  99   calc
 100     _ = w i * (pi μ).expectation (patternValue (F i \ R) (s i)) := by
 101       unfold expectation
 102       rw [Finset.mul_sum]
 103       apply Finset.sum_congr rfl
 104       intro σ _
 105       ring

 108 /-- A joint event implies one event in the first coordinate, uniformly
 109 over all later admissible choices. In particular, the auxiliary random
 110 coordinate is eliminated once for the entire family of later choices. -/
 111 theorem finiteLaw_common_event_projection
 112     {Ω Γ X Y : Type*} [Fintype Ω] [Fintype Γ]
 113     (μ : FiniteLaw Ω) (ν : FiniteLaw Γ)
 114     (E : Ω → Γ → Prop) (admissible : Ω → X → Prop)
 115     (conclusion : Ω → X → Y → Prop)
 116     (hE : ∀ ω γ, E ω γ → ∀ x, admissible ω x → ∃ y, conclusion ω x y) :
 117     (μ.prod ν).probability (fun z => E z.1 z.2) ≤
 118       μ.probability (fun ω => ∀ x, admissible ω x → ∃ y, conclusion ω x y) := by
 119   classical
 120   unfold probability
 121   rw [Fintype.sum_prod_type]
 122   apply Finset.sum_le_sum
 123   intro ω _
 124   by_cases hgood : ∀ x, admissible ω x → ∃ y, conclusion ω x y
 125   · rw [ite_eq_left hgood]
 126     calc
 127       _ ≤ ∑ γ, μ.mass ω * ν.mass γ := by
 128         apply Finset.sum_le_sum
 129         intro γ _
 130         change (if E ω γ then μ.mass ω * ν.mass γ else 0) ≤ _
 131         split_ifs
 132         · rfl
 133         · exact mul_nonneg (μ.nonneg ω) (ν.nonneg γ)
 134       _ = μ.mass ω := by rw [← Finset.mul_sum, ν.total, mul_one]
 135   · have hbad (γ : Γ) : ¬E ω γ := fun h => hgood (hE ω γ h)
 136     simp [hgood, hbad]

 138 /-- One physical edge receives a categorical colour and an independent
 139 fair direction. Both possible directions belong to this one coordinate. -/
 140 def FiniteLaw.categoricalDirection {k : ℕ} (ρ : Fin k → ℝ)
 141     (hρ : ∀ a, 0 ≤ ρ a) (htotal : ∑ a, ρ a = 1) : FiniteLaw (Fin k × Bool) where
 142   mass s := ρ s.1 / 2
 143   nonneg s := div_nonneg (hρ s.1) (by norm_num)
 144   total := by
 145     rw [Fintype.sum_prod_type]
 146     simp only [Fintype.sum_bool]
 147     convert htotal using 1
 148     apply Finset.sum_congr rfl
 149     intro a _
 150     ring

 152 /-- The indicator that a particular physical edge has colour `a` and its
 153 chosen direction meets a fixed cut. `forward` and `reverse` record the two
 154 endpoint tests; they are allowed to hold simultaneously. -/
 155 def directedCutContribution {k : ℕ} (a : Fin k) (forward reverse : Prop)
 156     [Decidable forward] [Decidable reverse] (s : Fin k × Bool) : ℝ :=
 157   if s.1 = a then
 158     if s.2 then (if forward then 1 else 0) else (if reverse then 1 else 0)
 159   else 0

 162     {k : ℕ} (a : Fin k) (forward reverse : Prop)
 163     [Decidable forward] [Decidable reverse] (s : Fin k × Bool) :
 164     directedCutContribution a forward reverse s ∈ Set.Icc (0 : ℝ) 1 := by

 168 /-- If both directions of a physical edge meet a cut, its expected
 169 contribution is `ρ a`, not two independent Bernoulli contributions. -/
 170 theorem categoricalDirection_expected_cut
 171     {k : ℕ} (ρ : Fin k → ℝ) (hρ : ∀ a, 0 ≤ ρ a) (htotal : ∑ a, ρ a = 1)
 172     (a : Fin k) (forward reverse : Prop)
 173     [Decidable forward] [Decidable reverse] :
 174     (FiniteLaw.categoricalDirection ρ hρ htotal).expectation
 175       (directedCutContribution a forward reverse) =
 176       ρ a / 2 * ((if forward then 1 else 0) + (if reverse then 1 else 0)) := by
 177   classical
 178   unfold expectation directedCutContribution
 179   simp only [FiniteLaw.categoricalDirection, Fintype.sum_prod_type, Fintype.sum_bool]
 180   by_cases hf : forward <;> by_cases hr : reverse <;>
 181     (simp [hf, hr, mul_ite, Finset.sum_add_distrib] <;> ring)
