  11 theorem triangle_row_cover_parameters (ε η κ C : ℝ)
  12     (hε:0<ε) (hη:0<η) (hκ:0<κ) (hC:0<C) :
  13     ∃ θ:ℝ,0<θ ∧ ∀ a b n:ℝ,0<a → κ*a≤b → 0≤n → n≤θ*a →
  14       4*(3*n)/a+3*(3*n)/b≤ε/2 ∧ 5*C*(3*n)/a≤η/2 := by
  15   let D:ℝ:=12+9/κ
  16   have hD:0<D:=by dsimp [D];positivity
  17   let θ:=min (ε/(2*D)) (η/(30*C))
  18   have hθ:0<θ:=lt_min (by positivity) (by positivity)
  19   refine ⟨θ,hθ,?_⟩
  20   intro a b n ha hba hn hnθ
  21   have hb:0<b:=(mul_pos hκ ha).trans_le hba
  22   have hratio:n/a≤θ:=(div_le_iff₀ ha).mpr hnθ
  23   have hnb:n/b≤(n/a)/κ:=by
  24     rw [div_div]
  25     apply div_le_div_of_nonneg_left hn (mul_pos ha hκ)
  26     simpa only [mul_comm] using hba
  27   constructor
  28   · have h₁:D*(n/a)≤D*θ:=mul_le_mul_of_nonneg_left hratio hD.le
  29     have h₂:D*θ≤ε/2:=by
  30       have hh:=mul_le_mul_of_nonneg_left (min_le_left (ε/(2*D)) (η/(30*C))) hD.le
  31       have he:D*(ε/(2*D))=ε/2:=by field_simp <;> ring
  32       simpa only [θ,he] using hh
  33     calc
  34       _ = 12*(n/a)+9*(n/b):=by ring
  35       _ ≤ 12*(n/a)+9*((n/a)/κ):=add_le_add (le_refl _) (mul_le_mul_of_nonneg_left hnb (by norm_num : (0:ℝ)≤9))
  36       _ = D*(n/a):=by dsimp [D];ring
  37       _ ≤ ε/2:=h₁.trans h₂
  38   · have h₁:15*C*(n/a)≤15*C*θ:=mul_le_mul_of_nonneg_left hratio (by positivity)
  39     have h₂:15*C*θ≤η/2:=by
  40       have hh:=mul_le_mul_of_nonneg_left (min_le_right (ε/(2*D)) (η/(30*C))) (show 0≤15*C by positivity)
  41       have he:15*C*(η/(30*C))=η/2:=by field_simp <;> ring
  42       simpa only [θ,he] using hh
  43     have he:5*C*(3*n)/a=15*C*(n/a):=by ring
  44     rw [he]
  45     exact h₁.trans h₂
