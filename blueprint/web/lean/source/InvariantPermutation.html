  10 def invariantSetPermutation (π : Equiv.Perm V) (A : Set V)
  11     (hA : ∀ v,π v∈A ↔ v∈A) : Equiv.Perm A where
  12   toFun x := ⟨π x,(hA x).mpr x.property⟩
  13   invFun x := ⟨π.symm x,(hA (π.symm x)).mp (by simpa using x.property)⟩
  14   left_inv x := by apply Subtype.ext;exact π.symm_apply_apply x
  15   right_inv x := by apply Subtype.ext;exact π.apply_symm_apply x
