  13 theorem transversal_graph_weight_bound {I X V : Type*}
  14     [Fintype I] [Fintype X] [Fintype V] [DecidableEq I] [DecidableEq V]
  15     (cl : V → I) (val : V → X)
  16     (hinj : Function.Injective (fun v => (cl v, val v)))
  17     (C : (I → X) → SimpleGraph V) (hC : Function.Injective C)
  18     (w : SimpleGraph V → ℝ) (hw : ∀ H, 0 ≤ w H)
  19     (hs : ∀ H, 0 < w H → H ∈ Set.range C)
  20     (M : ℝ) (hM : 0 ≤ M) (hb : ∀ f, w (C f) ≤ M)
  21     (S : Finset V) (E : SimpleGraph V → Prop)
  22     (hE : ∀ f, E (C f) → ∀ v ∈ S, f (cl v) = val v) :
  23     (∑ H, if E H then w H else 0) ≤
  24       (Fintype.card X : ℝ) ^ (Fintype.card I - S.card) * M := by
  25   classical
  26   have hsum : (∑ f, if E (C f) then w (C f) else 0) =
  27       ∑ H, if E H then w H else 0 := by
  28     apply Fintype.sum_of_injective C hC
  29     · intro H hH
  30       have hz : w H = 0 := by
  31         by_contra hn
  32         exact hH (hs H (lt_of_le_of_ne (hw H) (Ne.symm hn)))
  33       simp only [hz, ite_self]
  34     · intro f; rfl
  35   rw [← hsum]
  36   calc
  37     _ ≤ ∑ f, if (∀ v ∈ S, f (cl v) = val v) then w (C f) else 0 := by
  38       apply Finset.sum_le_sum
  39       intro f _
  40       by_cases he : E (C f)
  41       · rw [if_pos he, if_pos (hE f he)]
  42       · simp only [he, ite_false]
  43         split_ifs
  44         · exact hw _
  45         · exact le_refl _
  46     _ ≤ _ := by
  47       convert prescribedVertices_weight_le S cl val hinj (fun f => w (C f)) M hM hb using 1
  48       apply Finset.sum_congr rfl
  49       intro f _
  50       split_ifs <;> rfl

  52 theorem normalized_transversal_power (n B : ℝ) (d s : ℕ)
  53     (hn : n ≠ 0) (hs : 2 ≤ s) (hsd : s ≤ d) :
  54     n ^ (d - s) * (B / n ^ (d - 2)) = B / n ^ (s - 2) := by
  55   have he : d - 2 = (d - s) + (s - 2) := by omega
