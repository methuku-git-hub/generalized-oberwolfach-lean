  16 theorem sequential_choices_count {V : Type*} [Fintype V]
  17     (Good : ∀ n : ℕ,(Fin n → V) → Prop) (K : ℕ) (a : ℝ) (ha : 0≤a)
  18     (hzero : ∀ f : Fin 0 → V,Good 0 f)
  19     (hstep : ∀ n<K,∀ f : Fin n → V,Good n f →
  20       a≤(Fintype.card {v : V // Good (n+1) (Fin.snoc f v)}:ℝ)) :
  21     a^K≤(Fintype.card {f : Fin K → V // Good K f}:ℝ) := by
  22   have hind : ∀ n≤K,a^n≤(Fintype.card {f : Fin n → V // Good n f}:ℝ) := by
  23     intro n
  24     induction n with
  25     | zero=>
  26       intro _
  27       simp only [pow_zero]
  28       have hnon : Nonempty {f : Fin 0 → V // Good 0 f} := ⟨⟨Fin.elim0,hzero _⟩⟩
  29       exact_mod_cast (Fintype.card_pos_iff.mpr hnon : 0<Fintype.card {f : Fin 0 → V // Good 0 f})
  30     | succ n ih=>
  31       intro hn
  32       let B := {f : Fin n → V // Good n f}
  33       let C (f : B) := {v : V // Good (n+1) (Fin.snoc f.val v)}
  34       let next : (Σ f : B,C f) → {g : Fin (n+1) → V // Good (n+1) g} :=
  35         fun z=>⟨Fin.snoc z.1.val z.2.val,z.2.property⟩
  36       have hinj : Function.Injective next := by
  37         intro x y he
  38         have hh := congrArg Subtype.val he
  39         have hf : x.1=y.1 := by
  40           apply Subtype.ext
  41           funext i
  42           have hi := congrFun hh i.castSucc
  43           simpa only [next,Fin.snoc_castSucc] using hi
  44         have hv : x.2.val=y.2.val := by
  45           have hi := congrFun hh (Fin.last n)
  46           simpa only [next,Fin.snoc_last] using hi
  47         cases x with
  48         | mk f v=>
  49           cases y with
  50           | mk g w=>
  51             dsimp only at hf
  52             subst g
  53             exact congrArg (Sigma.mk f) (Subtype.ext hv)
  54       have hc : (Fintype.card (Σ f : B,C f):ℝ)≤
  55           Fintype.card {g : Fin (n+1) → V // Good (n+1) g} := by
  56         exact_mod_cast Fintype.card_le_of_injective next hinj
  57       have hs : (Fintype.card B:ℝ)*a≤Fintype.card (Σ f : B,C f) := by
  58         rw [Fintype.card_sigma,Nat.cast_sum]
  59         calc
  60           _ = ∑ _f : B,a := by simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul]
  61           _ ≤ _ := Finset.sum_le_sum (fun f _=>hstep n (by omega) f.val f.property)
  62       calc
  63         a^(n+1)=a^n*a := pow_succ _ _
  64         _ ≤ (Fintype.card B:ℝ)*a := mul_le_mul_of_nonneg_right (ih (by omega)) ha
  65         _ ≤ _ := hs.trans hc
  66   exact hind K le_rfl
