  11     L+1≤M 0 := by
  12   have hdec (i : ℕ) (hi : i<L) : M (i+1)+1≤M i := by
  13     have hposR : (0:ℝ)<M i := by exact_mod_cast (lt_of_lt_of_le Nat.zero_lt_one (hpos i (by omega)))
  14     have hh := (hstep i hi).trans_lt (by simpa only [one_mul] using mul_lt_mul_of_pos_right hη hposR)
  15     have hh' : M (i+1)<M i := by exact_mod_cast hh
  16     omega
  17   have hsum : ∀ i≤L,M i+i≤M 0 := by
  18     intro i
  19     induction i with
  20     | zero=>intro hi;omega
  21     | succ i ih=>
  22       intro hi
  23       have hp := ih (by omega)
  24       have hs := hdec i (by omega)
  25       omega
  26   have hh := hsum L le_rfl
  27   have hp := hpos L le_rfl
