  25       (∑ x,(J₀ v x:ℝ))+(∑ x,(J₀ x v:ℝ))+(∑ x,(Jₖ v x:ℝ))+(∑ x,(Jₖ x v:ℝ)):=by

  29         (∑ e:Z,if J (t e) (h e)≠0∧t e=v then (1:ℝ) else 0)+
