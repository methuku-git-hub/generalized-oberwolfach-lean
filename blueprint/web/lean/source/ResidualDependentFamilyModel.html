   2 /-! Construct the embedding theorem's source pieces for an arbitrary
   3 family of separately typed two-factor subgraphs. -/
   4 set_option autoImplicit false
   5 noncomputable section
   6 namespace Oberwolfach
   7 attribute [local instance] Classical.propDecidable
   8 variable {I:Type} [Fintype I] [DecidableEq I] {X:I→Type}
   9   [∀i,Fintype (X i)] [∀i,DecidableEq (X i)]

  11 theorem two_factor_family_subgraph_dependent_models
  12     (F H:∀i,SimpleGraph (X i)) (R:∀i,Finset (X i))
  13     (hF:∀i,(F i).IsRegularOfDegree 2) (hHF:∀i,H i≤F i)
  14     (hisol:∀i x,x∉R i→∀y,¬(H i).Adj x y):
  15     ∃ (D:∀i,X i→X i→Prop) (pieces:∀i,List (PrescribedSourcePiece (X i))),
  16       (∀i,(pieces i).Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset)) ∧
  17       (∀i p,p∈pieces i→p.Shape ∧ p.OriginalValid (D i)) ∧
  18       (∀i,prescribedDependentVertices pieces i=R i) ∧
  19       (∀i,prescribedDependentGraph pieces i=H i) ∧
  20       (∀i x y,D i x y→(F i).Adj x y) := by
  21   choose D pieces hpair hvalid hvertices hgraph hD using
  22     fun i=>two_factor_subgraph_dependent_model (F i) (H i) (hF i) (hHF i) (R i) (hisol i)
  23   refine ⟨D,pieces,hpair,hvalid,?_,?_,hD⟩
  24   · intro i
  25     exact hvertices i
  26   · intro i
  27     exact hgraph i
  28 end Oberwolfach
