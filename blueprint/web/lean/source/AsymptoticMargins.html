  11 theorem tendsto_polynomial_stretchedExp
  12     (s : ℝ) {c p : ℝ} (hc : 0 < c) (hp : 0 < p) :
  13     Tendsto (fun x : ℝ => x ^ s * Real.exp (-c * x ^ p)) atTop (𝓝 0) := by
  14   have h := (tendsto_rpow_mul_exp_neg_mul_atTop_nhds_zero (s / p) c hc).comp
  15     (tendsto_rpow_atTop hp)
  16   apply h.congr'
  17   filter_upwards [eventually_ge_atTop (0 : ℝ)] with x hx
  18   dsimp only [Function.comp_apply]
  19   rw [← Real.rpow_mul hx]
  20   congr 2
  21   field_simp

  23 theorem eventually_stretchedExp_le_inv_pow
  24     (C K : ℝ) {c p : ℝ} (hc : 0 < c) (hp : 0 < p) (q : ℕ) :
  25     ∃ N : ℕ, ∀ n : ℕ, N ≤ n →
  26       C * (n : ℝ) ^ K * Real.exp (-c * (n : ℝ) ^ p) ≤ 1 / (n : ℝ) ^ q := by
  27   have ht := ((tendsto_polynomial_stretchedExp (K + q) hc hp).const_mul C).comp
  28     tendsto_natCast_atTop_atTop
  29   simp only [mul_zero] at ht
  30   have hb : ∀ᶠ n : ℕ in atTop,
  31       C * ((n : ℝ) ^ (K + q) * Real.exp (-c * (n : ℝ) ^ p)) < 1 :=
  32     ht.eventually (eventually_lt_nhds (show (0 : ℝ) < 1 by norm_num))
  33   obtain ⟨N, hN⟩ := eventually_atTop.mp hb
  34   refine ⟨max N 1, fun n hn => ?_⟩
  35   have hnpos : 0 < (n : ℝ) := by exact_mod_cast (lt_of_lt_of_le Nat.zero_lt_one (le_trans (le_max_right _ _) hn))
  36   apply (le_div_iff₀ (pow_pos hnpos q)).mpr
  37   have heq : (C * (n : ℝ) ^ K * Real.exp (-c * (n : ℝ) ^ p)) * (n : ℝ) ^ q =
  38       C * ((n : ℝ) ^ (K + q) * Real.exp (-c * (n : ℝ) ^ p)) := by
  39     rw [Real.rpow_add hnpos, Real.rpow_natCast]
  40     ring
  41   rw [heq]
  42   exact (hN n (le_trans (le_max_left _ _) hn)).le

  44 /-- A union bound over all ordered pairs of vertex subsets still decays
  45 faster than every inverse power when each cut has quadratic exponential
  46 failure. The constant and threshold are chosen before the graph order. -/
  47 theorem eventually_allCuts_failure_small {C c : ℝ} (hC : 0 ≤ C) (hc : 0 < c) (q : ℕ) :
  48     ∃ N : ℕ, ∀ n : ℕ, N ≤ n →
  49       C * (4 : ℝ) ^ n * Real.exp (-c * (n : ℝ) ^ 2) ≤ 1 / (n : ℝ) ^ q := by
  50   obtain ⟨N, hN⟩ := eventually_stretchedExp_le_inv_pow C 0
  51     (c := c / 2) (p := 2) (by positivity) (by norm_num) q
  52   refine ⟨max N (Nat.ceil (2 * Real.log 4 / c)), fun n hn => ?_⟩
  53   have hn0 : (0 : ℝ) ≤ n := Nat.cast_nonneg n
  54   have hceil : (Nat.ceil (2 * Real.log 4 / c) : ℝ) ≤ n := by
  55     exact_mod_cast le_trans (le_max_right _ _) hn
  56   have hh : 2 * Real.log 4 / c ≤ (n : ℝ) := (Nat.le_ceil _).trans hceil
  57   have hlog : 2 * Real.log 4 ≤ (n : ℝ) * c := (div_le_iff₀ hc).mp hh
  58   have hmul := mul_le_mul_of_nonneg_right hlog hn0
  59   have hexp : (n : ℝ) * Real.log 4 + -c * (n : ℝ) ^ 2 ≤ -(c / 2) * (n : ℝ) ^ 2 := by
  60     nlinarith
  61   have heq : (4 : ℝ) ^ n = Real.exp ((n : ℝ) * Real.log 4) := by
  62     rw [Real.exp_nat_mul, Real.exp_log (by norm_num : (0 : ℝ) < 4)]
  63   calc
  64     _ = C * Real.exp ((n : ℝ) * Real.log 4 + -c * (n : ℝ) ^ 2) := by
  65       rw [heq, Real.exp_add]
  66       ring
  67     _ ≤ C * Real.exp (-(c / 2) * (n : ℝ) ^ 2) :=
  68       mul_le_mul_of_nonneg_left (Real.exp_le_exp.mpr hexp) hC
  69     _ ≤ _ := by
  70       simpa only [Real.rpow_zero, mul_one, Real.rpow_two] using
  71         hN n (le_trans (le_max_left _ _) hn)
