  14 theorem candidate_vertex_avoiding (F:SimpleGraph V) (C W R:Finset V)
  15     (d c:ℝ) (hF:∀ v,(F.degree v:ℝ)≤d) (hC:c≤(C.card:ℝ))
  16     (hbudget:W.card+(R.card:ℝ)*(d+1)<c):
  17     ∃ v∈C,v∉W∧v∉R∧∀ w∈R,¬F.Adj v w:=by
  18   let D:=W∪(R∪R.biUnion (fun v=>F.neighborFinset v))
  19   have hR:=forbidden_used_neighbours_card F R d hF
  20   have hu:(D.card:ℝ)≤W.card+(R∪R.biUnion (fun v=>F.neighborFinset v)).card:=by
  21     exact_mod_cast Finset.card_union_le W (R∪R.biUnion (fun v=>F.neighborFinset v))
  22   have hlt:D.card<C.card:=by exact_mod_cast (show (D.card:ℝ)<C.card by linarith only [hR,hu,hC,hbudget])
  23   obtain ⟨v,hv,hvD⟩:=Finset.exists_mem_notMem_of_card_lt_card hlt
  24   have hout:v∉W∧v∉R∧v∉R.biUnion (fun v=>F.neighborFinset v):=by
  25     simpa only [D,Finset.mem_union,not_or] using hvD
  26   refine ⟨v,hv,hout.1,hout.2.1,?_⟩
  27   intro w hw hadj
  28   exact hout.2.2 (Finset.mem_biUnion.mpr ⟨w,hw,(F.mem_neighborFinset w v).mpr hadj.symm⟩)

  30 theorem separated_candidate_vertices (F:SimpleGraph V) (C:I→Finset V) (W R:Finset V)
  31     (d c:ℝ) (hd:0≤d) (hF:∀ v,(F.degree v:ℝ)≤d) (hC:∀ i,c≤((C i).card:ℝ))
  32     (hbudget:W.card+((R.card:ℝ)+Fintype.card I)*(d+1)<c):
  33     ∃ p:I↪V,(∀ i,p i∈C i∧p i∉W∧p i∉R∧∀ v∈R,¬F.Adj (p i) v)∧
  34       ∀ i j,¬F.Adj (p i) (p j):=by
  35   let Good:I→V→Prop:=fun i v=>v∈C i∧v∉R∧∀ w∈R,¬F.Adj v w
  36   have hc:∀ i,∀ B:Finset V,(B.card:ℝ)≤W.card+Fintype.card I*1*(d+1)→
  37       ∃ v,Good i v∧({v}:Finset V).card≤1∧Disjoint ({v}:Finset V) B:=by
  38     intro i B hB
  39     obtain ⟨v,hv,hvB,hvR,hvF⟩:=candidate_vertex_avoiding F (C i) B R d c hF (hC i)
  40       (by linarith only [hB,hbudget])
  41     exact ⟨v,⟨hv,hvR,hvF⟩,by simp,by simpa using hvB⟩
  42   obtain ⟨p,hp,hsep⟩:=finite_conflict_separated_choices (fun v:V=>({v}:Finset V)) Good W 1 F d hd hF (by simpa only [Nat.cast_one] using hc)
  43   have hinj:Function.Injective p:=by
  44     intro i j he
  45     by_contra hij
  46     have hh:=(hsep i j hij).1
  47     simpa only [he,Finset.disjoint_singleton_left,Finset.mem_singleton,not_true_eq_false] using hh
  48   refine ⟨⟨p,hinj⟩,?_,?_⟩
  49   · intro i
  50     exact ⟨(hp i).1.1,by simpa using (hp i).2.2,(hp i).1.2.1,(hp i).1.2.2⟩
  51   · intro i j
  52     by_cases he:i=j
  53     · subst j;exact F.loopless.irrefl _
  54     · exact (hsep i j he).2 _ (by simp) _ (by simp)
