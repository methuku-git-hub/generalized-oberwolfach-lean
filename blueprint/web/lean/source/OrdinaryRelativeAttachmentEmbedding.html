  17 theorem ordinary_relative_attachment_embedding (Q F : SimpleGraph V) (H : V → U → Prop)
  18     (A W R : Finset V) (B : Finset U) (φ : V₀ ↪ V) (ψ : U₀ ↪ U)
  19     (G₀ : SimpleGraph V₀) (H₀ : V₀ → U₀ → Prop) (ρ : I → V₀)
  20     (hφ : ∀ a,φ a∈R) (hψ : ∀ b,ψ b∈B)
  21     (hG₀ : ∀ a b,G₀.Adj a b → Q.Adj (φ a) (φ b)) (hH₀ : ∀ a b,H₀ a b → H (φ a) (ψ b))
  22     (r : ℕ) (hr : 3≤r) (d c cH : ℝ) (hd : 0≤d)
  23     (hF : ∀ v,(F.degree v:ℝ)≤d)
  24     (hcount : ∀ v w,c≤((A∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  25     (hhub : ∀ T:Finset V,T.card≤r → (∀ v∈T,∀ w∈T,¬F.Adj v w) →
  26       cH≤((common Finset.univ T (fun v => Finset.univ.filter (H v))).card:ℝ))
  27     (hbudget : W.card+((R.card:ℝ)+Fintype.card I*r+r+6)*(d+1)<c)
  28     (hB : (B.card:ℝ)+Fintype.card I<cH) :
  29     ∃ (f : V₀⊕(I×Fin r) ↪ V) (g : U₀⊕I ↪ U),
  30       (∀ a,f (.inl a)=φ a) ∧ (∀ b,g (.inl b)=ψ b) ∧
  31       (∀ i w,f (.inr (i,w))∈A ∧ f (.inr (i,w))∉W ∧ f (.inr (i,w))∉R ∧
  32         ∀ z∈R,¬F.Adj (f (.inr (i,w))) z) ∧
  33       (∀ i,g (.inr i)∉B) ∧
  34       (∀ i j w w',¬F.Adj (f (.inr (i,w))) (f (.inr (j,w')))) ∧
  35       (∀ a b,(ordinaryAttachmentRim G₀ (SimpleGraph.cycleGraph r) ⟨0,by omega⟩ ρ).Adj a b → Q.Adj (f a) (f b)) ∧
  36       ∀ a b,ordinaryAttachmentSpokes H₀ ⟨0,by omega⟩ ρ a b → H (f a) (g b) := by
  37   let z : Fin r := ⟨0,by omega⟩
  38   let C := SimpleGraph.cycleGraph r
  39   let P : I → (Fin r ↪ V) → U → Prop := fun i f a =>
  40     (∀ j k,j≠z → k≠z → C.Adj j k → Q.Adj (f j) (f k)) ∧
  41     (∀ j,C.Adj z j → Q.Adj (φ (ρ i)) (f j)) ∧ H (φ (ρ i)) a ∧ (∀ j,j≠z → H (f j) a) ∧
  42     (∀ j k,¬F.Adj (f j) (f k)) ∧
  43     ∀ j,f j∈A ∧ f j∉W ∧ f j∉R ∧ ∀ v∈R,¬F.Adj (f j) v
  44   obtain ⟨f,a,hf,hsep⟩ := separated_labelled_gadgets_finite r F R B P (by
  45     intro i S T hRS hBT hS hT
  46     have hSc : (S.card:ℝ)≤R.card+Fintype.card I*r := by exact_mod_cast hS
  47     have hTc : (T.card:ℝ)≤B.card+Fintype.card I := by exact_mod_cast hT
  48     obtain ⟨f,a,hC,hcross,hrootH,hH,ha,hind,havoid⟩ := ordinary_relative_attachment_root Q F H A W S T
  49       (φ (ρ i)) (hRS (hφ (ρ i))) r hr d c cH hd hF hcount
  50       (by nlinarith only [hbudget,hSc,hd]) hhub (hTc.trans_lt hB)
  51     refine ⟨f,a,⟨hC,hcross,hrootH,hH,hind,?_⟩,(fun j => (havoid j).2.2.1),ha,?_⟩
  52     · intro j
  53       exact ⟨(havoid j).1,(havoid j).2.1,(fun h => (havoid j).2.2.1 (hRS h)),
  54         fun v hv => (havoid j).2.2.2 v (hRS hv)⟩
  55     · intro j v hv
  56       exact (havoid j).2.2.2 v hv)
  57   have hfroot : ∀ i w b,f i w≠φ b := by intro i w b he;exact (hf i).2.1 w (he ▸ hφ b)
  58   have haroot : ∀ i b,a i≠ψ b := by intro i b he;exact (hf i).2.2 (he ▸ hψ b)
  59   have hainj : Function.Injective a := by intro i j he;by_contra hne;exact (hsep i j hne).1 he
  60   have hfsep : ∀ i j,i≠j → ∀ w w',f i w≠f j w' := fun i j hij w w' => ((hsep i j hij).2 w w').1
  61   let f' := expansionRowEmbedding φ f hfroot hfsep
  62   let g' := expansionHubEmbedding ψ a haroot hainj
  63   refine ⟨f',g',(fun _ => rfl),(fun _ => rfl),(fun i => (hf i).1.2.2.2.2.2),(fun i => (hf i).2.2),?_,
  64     attachmentRowEmbedding_hom φ f hfroot hfsep G₀ C z ρ Q hG₀ (fun i => (hf i).1.1) (fun i => (hf i).1.2.1),
  65     attachmentEmbedding_spokes φ f hfroot hfsep ψ a haroot hainj H₀ z ρ H hH₀
  66       (fun i => (hf i).1.2.2.2.1) (fun i => (hf i).1.2.2.1)⟩
  67   intro i j w w'
  68   by_cases hij : i=j
  69   · subst j;exact (hf i).1.2.2.2.2.1 w w'
  70   · exact ((hsep i j hij).2 w w').2
