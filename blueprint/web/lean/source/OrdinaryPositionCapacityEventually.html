   8 theorem ordinary_position_capacity_for_large_orders
   9     (θ C ε lam d a:ℝ) (hθ:0<θ) (hC:1≤C)
  10     (hε:0≤ε) (hlam:0≤lam) (hd:0<d) (ha:0<a)
  11     (hearly:C*(ε+ε^2/a)≤1/100000)
  12     (hlate:C/d+C*lam+C*lam/a≤θ/100000):
  13     ∃ N:ℕ,∀ n:ℕ,N≤n→0<n→∀ A Z E:ℝ,
  14       0≤E→A≤(2/d+lam)*n→Z≤(2*θ*ε+2*lam)*n→
  15       E≤C*(θ*ε^2+lam*ε)*(n:ℝ)^2+C*d*n→
  16       8*A+8*((⌈4*E/(a*n)⌉₊:ℕ):ℝ)+4*Z+10≤θ*n/4 := by
  17   obtain ⟨N,hN⟩:=exists_nat_ge (16*(8*(4*C*d/a+1)+10)/θ)
  18   refine ⟨N,?_⟩
  19   intro n hn hn0 A Z E hE0 hA hZ hE
  20   have hnR:(0:ℝ)<n:=by exact_mod_cast hn0
  21   have hbound:16*(8*(4*C*d/a+1)+10)/θ≤(n:ℝ):=
  22     hN.trans (by exact_mod_cast hn)
  23   have hlarge:8*(4*C*d/a+1)+10≤θ*(n:ℝ)/16:=by
  24     have hh:16*(8*(4*C*d/a+1)+10)≤(n:ℝ)*θ:=(div_le_iff₀ hθ).mp hbound
  25     linarith
  26   exact ordinary_actual_position_capacity n θ C ε lam d a A Z E hnR hθ.le hC hε hlam hd ha hE0 hearly hlate hA hZ hE hlarge
  27 end Oberwolfach
