   9 theorem transitionPathWeight_product {X : Type*}
  10     (R : ℕ → X → X → ℝ) (i k : ℕ) (p : Fin (k+1) → X) :
  11     transitionPathWeight R i k p=∏ j : Fin k, R (i+j.val) (p j.castSucc) (p j.succ) := by
  12   induction k generalizing i with
  13   | zero => simp [transitionPathWeight]
  14   | succ k ih =>
  15     rw [transitionPathWeight,ih,Fin.prod_univ_succ]
  16     simp only [Fin.val_zero,Nat.add_zero,Fin.castSucc_zero,Fin.succ_zero_eq_one,
  17       Fin.val_succ,Fin.tail,Fin.succ_castSucc]
  18     congr 1
  19     apply Finset.prod_congr rfl
  20     intro j _
  21     congr 1 <;> omega
