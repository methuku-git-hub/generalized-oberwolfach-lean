  14 theorem embedding_predicate_card {I X : Type*} [Fintype I] [Fintype X]
  15     (f : I↪X) (P : X→Prop) :
  16     (Finset.univ.filter (fun i=>P (f i))).card≤(Finset.univ.filter P).card := by
  17   classical
  18   rw [← Finset.card_map f]
  19   apply Finset.card_le_card
  20   intro x hx
  21   obtain ⟨i,hi,rfl⟩:=Finset.mem_map.mp hx
  22   exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,(Finset.mem_filter.mp hi).2⟩
