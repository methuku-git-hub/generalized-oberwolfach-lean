  13 theorem factor_cycle_orders_align_with_source_pieces
  14     {X:Type} [Fintype X] [DecidableEq X] (G:SimpleGraph X) (hG:G.IsRegularOfDegree 2)
  15     (pieces:List (PrescribedSourcePiece X)) (D:X→X→Prop)
  16     (hpair:pieces.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset))
  17     (hvalid:∀p∈pieces,p.OriginalValid D)
  18     (hEdges:sourcePieceEdgesUnion pieces=G.edgeFinset)
  19     (hArcs:∀x y,D x y↔(x,y)∈sourcePieceArcsUnion pieces):
  20     ∃ e:∀C:G.ConnectedComponent,SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph,
  21       ∀x,D x (factorCycleOrientation G e x):=by
  22   classical
  23   obtain ⟨hcover,hin⟩:=actual_source_orientation_covers_and_has_unique_incoming G pieces D hpair hvalid hEdges hArcs
  24   have hex (C:G.ConnectedComponent):
  25       ∃e:SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph,
  26         D (e ⟨0,by have h:3≤Fintype.card C:=(component_regular_two_cycle_iso G hG C).1;omega⟩).val
  27           (e ⟨1,by have h:3≤Fintype.card C:=(component_regular_two_cycle_iso G hG C).1;omega⟩).val:=by
  28     have h3:3≤Fintype.card C:=(component_regular_two_cycle_iso G hG C).1
  29     obtain ⟨e₀⟩:=(component_regular_two_cycle_iso G hG C).2
  30     let z0:Fin (Fintype.card C):=⟨0,by omega⟩
  31     let z1:Fin (Fintype.card C):=⟨1,by omega⟩
  32     have hrot:finRotate (Fintype.card C) z0=z1:=by
  33       apply (fin_rotate_succ_iff h3 z0 z1).1
  34       simp only [z0,z1,Fin.val_mk,Nat.zero_add,Nat.mod_eq_of_lt (by omega : 1<Fintype.card C)]
  35     have hcycle:(SimpleGraph.cycleGraph (Fintype.card C)).Adj z0 z1:=by
  36       apply (cycleGraph_modular_adj (by omega) _ _).2
  37       exact Or.inl ((fin_rotate_succ_iff h3 z0 z1).2 hrot)
  38     have hadj:G.Adj (e₀ z0).val (e₀ z1).val:=(e₀.map_rel_iff.mpr hcycle)
  39     rcases hcover _ _ hadj with hforward | hreverse
  40     · exact ⟨e₀,hforward⟩
  41     · obtain ⟨e₁,he0,he1⟩:=component_cycle_iso_with_first_edge G hG C (e₀ z1) (e₀ z0) hadj.symm
  42       refine ⟨e₁,?_⟩
  43       simpa only [he0,he1] using hreverse
  44   choose e he using hex
  45   refine ⟨e,factor_orientation_aligns_from_component_edges G hG e D hcover hin ?_⟩
  46   intro C
  47   have h3:3≤Fintype.card C:=(component_regular_two_cycle_iso G hG C).1
  48   let z0:Fin (Fintype.card C):=⟨0,by omega⟩
  49   let z1:Fin (Fintype.card C):=⟨1,by omega⟩
  50   have hrot:finRotate (Fintype.card C) z0=z1:=by
  51     apply (fin_rotate_succ_iff h3 z0 z1).1
  52     simp only [z0,z1,Fin.val_mk,Nat.zero_add,Nat.mod_eq_of_lt (by omega : 1<Fintype.card C)]
  53   refine ⟨e C z0,?_⟩
  54   rw [factor_cycle_orientation_on_component G e C (e C z0)]
  55   simpa only [RelIso.symm_apply_apply,hrot] using he C
  56 end Oberwolfach
