  13 theorem finite_image_incidence_bound (S:Finset A) (f:A↪B) (g:V↪V')
  14     (tail head:A→V) (tail' head':B→V')
  15     (ht:∀ a∈S,tail' (f a)=g (tail a)) (hh:∀ a∈S,head' (f a)=g (head a))
  16     (b:ℝ) (hb:0≤b)
  17     (h:∀ v,((S.filter (fun a=>v=tail a∨v=head a)).card:ℝ)≤b):
  18     ∀ v,(((S.image f).filter (fun a=>v=tail' a∨v=head' a)).card:ℝ)≤b:=by
  19   intro v
  20   rw [Finset.filter_image,Finset.card_image_of_injective _ f.injective]
  21   by_cases hv:∃ w,g w=v
  22   · obtain ⟨w,rfl⟩:=hv
  23     have he:S.filter (fun a=>g w=tail' (f a)∨g w=head' (f a))=
  24         S.filter (fun a=>w=tail a∨w=head a):=by
  25       ext a
  26       simp only [Finset.mem_filter]
  27       apply and_congr_right
  28       intro ha
  29       rw [ht a ha,hh a ha]
  30       simp only [EmbeddingLike.apply_eq_iff_eq]
  31     rw [he]
  32     exact h w
  33   · have he:S.filter (fun a=>v=tail' (f a)∨v=head' (f a))=∅:=by
  34       apply Finset.eq_empty_iff_forall_notMem.mpr
  35       intro a ha
  36       obtain ⟨ha,ht'|hh'⟩:=Finset.mem_filter.mp ha
  37       · exact hv ⟨tail a,(ht a ha).symm.trans ht'.symm⟩
  38       · exact hv ⟨head a,(hh a ha).symm.trans hh'.symm⟩
  39     rw [he,Finset.card_empty,Nat.cast_zero]
  40     exact hb
