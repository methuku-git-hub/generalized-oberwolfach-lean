  10 theorem forward_cycle_graph_walk (N:ℕ) (hN:3≤N):
  11     ∃(z:Fin N) (p:(SimpleGraph.cycleGraph N).Walk z z),p.IsCycle ∧ p.length=N ∧
  12       ∀k<p.length,finRotate N (p.getVert k)=p.getVert (k+1):=by
  13   obtain ⟨m,hm⟩:=Nat.exists_eq_add_of_le hN
  14   have hm':N=m+3:=by omega
  15   rw [hm']
  16   let p:=(SimpleGraph.cycleGraph.cycle m).reverse
  17   have hp:∀k≤m+3,p.getVert k=⟨k%(m+3),Nat.mod_lt _ (by omega)⟩:=by
  18     intro k hk
  19     dsimp only [p]
  20     rw [SimpleGraph.Walk.getVert_reverse,SimpleGraph.cycleGraph.length_cycle,
  21       SimpleGraph.cycleGraph.getVert_cycle (by omega)]
  22     congr 1
  23     have hh:m+3-(m+3-k)=k:=by omega
  24     rw [hh]
  25   refine ⟨0,p,SimpleGraph.cycleGraph.isCycle_cycle.reverse,by simp [p],?_⟩
  26   intro k hk
  27   have hk':k<m+3:=by simpa [p] using hk
  28   rw [hp k (by omega),hp (k+1) (by omega),finRotate_apply]
  29   apply Fin.ext
  30   simp only [Fin.val_add,Fin.val_mk,Fin.val_one]
  31   rw [Nat.mod_eq_of_lt hk']
  32 end Oberwolfach
