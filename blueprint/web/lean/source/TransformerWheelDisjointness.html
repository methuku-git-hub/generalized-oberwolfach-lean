  15  theorem private_support_intersection (R A B:Finset V) (hAB:Disjoint A B)
  16     (hAR:Disjoint A R) (hBR:Disjoint B R) (x:V)
  17     (hxA:x∈R∪A) (hxB:x∈R∪B):x∈R:=by
  18   by_contra hx
  19   exact Finset.disjoint_left.mp hAB ((Finset.mem_union.mp hxA).resolve_left hx)
  20     ((Finset.mem_union.mp hxB).resolve_left hx)

  22  theorem transformer_wheel_pair_disjoint (W W':OrdinaryWheel Q F H r)
  23     (R S S':Finset V) (hSS:Disjoint S S')
  24     (hSR:Disjoint S R) (hS'R:Disjoint S' R)
  25     (hW:∀ x∈(ordinaryWheelRim W).support,x∈R∪S)
  26     (hW':∀ x∈(ordinaryWheelRim W').support,x∈R∪S')
  27     (hrim:∀ x∈R,∀ y∈R,¬((ordinaryWheelRim W).Adj x y∧(ordinaryWheelRim W').Adj x y))
  28     (hspoke:ordinaryWheelHub W=ordinaryWheelHub W'→
  29       ∀ x∈R,¬(x∈(ordinaryWheelRim W).support∧x∈(ordinaryWheelRim W').support)):
  30     Disjoint (ordinaryWheelResources W) (ordinaryWheelResources W'):=by
  31   apply (ordinaryWheelResources_disjoint_iff W W').mpr
  32   constructor
  33   · apply disjoint_iff_inf_le.mpr
  34     intro x y h
  35     have hx:=private_support_intersection R S S' hSS hSR hS'R x
  36       (hW x h.1.mem_support_left) (hW' x h.2.mem_support_left)
  37     have hy:=private_support_intersection R S S' hSS hSR hS'R y
  38       (hW y h.1.mem_support_right) (hW' y h.2.mem_support_right)
  39     exact (hrim x hx y hy h).elim
  40   · intro he
  41     apply Finset.disjoint_left.mpr
  42     intro x hx hx'
  43     have h1:=Set.mem_toFinset.mp hx
  44     have h2:=Set.mem_toFinset.mp hx'
  45     exact hspoke he x (private_support_intersection R S S' hSS hSR hS'R x (hW x h1) (hW' x h2)) ⟨h1,h2⟩
