   6 theorem ordinary_sampled_triangle_margins (α pmin smin η b L:ℝ)
   7     (hα:0<α) (hp:0<pmin) (hs:0<smin) (hη:0<η) (hb:0≤b) (hL:0≤L):
   8     ∃ β rowErr t ε:ℝ,0<β∧0<rowErr∧0<t∧0<ε ∧
   9     ∃ N:ℕ,∀ n:ℕ,(N:ℝ)≤n → ∀ D:ℝ,(n:ℝ)/2≤D →
  10       β*n≤(pmin^4*smin^4)*(4*(min α (1/4)))*n-rowErr*n-4*L ∧
  11       t/(pmin^3*(smin*smin)^2)+(6*(3:ℝ)^4*b*L+ε*n)/D≤η := by
  12   let cmin:=pmin^3*(smin*smin)^2
  13   let β:=2*(min α (1/4))*pmin^4*smin^4
  14   let rowErr:=β/4
  15   let t:=η*cmin/4
  16   let ε:=η/8
  17   have hc:0<cmin:=by dsimp [cmin];positivity
  18   have hβ:0<β:=by
  19     have hx:0<min α (1/4:ℝ):=lt_min hα (by norm_num)
  20     dsimp [β];positivity
  21   have hrow:0<rowErr:=by dsimp [rowErr];positivity
  22   have ht:0<t:=by dsimp [t];positivity
  23   have hε:0<ε:=by dsimp [ε];positivity
  24   obtain ⟨N,hN⟩:=exists_nat_gt (8*L/β+48*(3:ℝ)^4*b*L/η+1)
  25   refine ⟨β,rowErr,t,ε,hβ,hrow,ht,hε,N,?_⟩
  26   intro n hn D hD
  27   have hterm1:0≤8*L/β:=by positivity
  28   have hterm2:0≤48*(3:ℝ)^4*b*L/η:=by positivity
  29   have hnreal:(0:ℝ)<n:=by nlinarith [hN,hn]
  30   have hDpos:0<D:=by linarith
  31   have hβn:8*L≤β*n:=by
  32     have hh:8*L/β≤(n:ℝ):=by linarith [hN,hn,hterm2]
  33     simpa only [mul_comm] using (div_le_iff₀ hβ).mp hh
  34   have hconf:48*(3:ℝ)^4*b*L≤η*n:=by
  35     have hh:48*(3:ℝ)^4*b*L/η≤(n:ℝ):=by linarith [hN,hn,hterm1]
  36     simpa only [mul_comm] using (div_le_iff₀ hη).mp hh
  37   constructor
  38   · dsimp [rowErr,β]
  39     dsimp [β] at hβn
  40     nlinarith only [hβn,hL]
  41   · have hfrac:(n:ℝ)/D≤2:=(div_le_iff₀ hDpos).mpr (by linarith)
  42     have hconfD:6*(3:ℝ)^4*b*L/D≤η/4:=by
  43       apply (div_le_iff₀ hDpos).mpr
  44       have hh:0≤η*(2*D-n):=mul_nonneg hη.le (by linarith)
  45       nlinarith only [hconf,hh]
  46     have herr:(ε*n)/D≤η/4:=by
  47       dsimp [ε]
  48       have hh:=mul_le_mul_of_nonneg_left hfrac (show 0≤η/8 by positivity)
  49       have heq:(η/8*n)/D=(η/8)*(n/D):=by ring
  50       rw [heq]
  51       calc
  52         _ ≤ (η/8)*2 := hh
  53         _ = η/4 := by ring
  54     have htdiv:t/cmin=η/4:=by dsimp [t];field_simp
  55     change t/cmin+(6*(3:ℝ)^4*b*L+ε*n)/D≤η
  56     rw [add_div,htdiv]
  57     linarith only [hconfD,herr,hη]
