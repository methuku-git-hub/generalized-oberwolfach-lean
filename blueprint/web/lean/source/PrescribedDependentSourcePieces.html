  15 theorem prescribed_lifted_pieces_apart (pieces:∀ i,List (PrescribedSourcePiece (X i)))
  16     (hapart:∀ i,(pieces i).Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset)):
  17     ∀ is:List I,is.Nodup→(prescribedLiftDependentPieces is pieces).Pairwise PrescribedPiecesApart := by
  18   intro is
  19   induction is with
  20   | nil => intro hn;simp [prescribedLiftDependentPieces]
  21   | cons i is ih =>
  22     intro hn
  23     have hn':=List.nodup_cons.mp hn
  24     simp only [prescribedLiftDependentPieces,List.flatMap_cons,List.pairwise_append]
  25     refine ⟨?_,ih hn'.2,?_⟩
  26     · rw [List.pairwise_map]
  27       apply (hapart i).imp
  28       intro p q h
  29       right
  30       simp only [prescribed_piece_map_vertices]
  31       apply Finset.disjoint_left.mpr
  32       intro z hz hz'
  33       obtain ⟨a,ha,haz⟩:=List.mem_map.mp (List.mem_toFinset.mp hz)
  34       obtain ⟨b,hb,hbz⟩:=List.mem_map.mp (List.mem_toFinset.mp hz')
  35       have hab:a=b:=sigma_mk_injective (haz.trans hbz.symm)
  36       subst b
  37       exact (Finset.disjoint_left.mp h (List.mem_toFinset.mpr ha)) (List.mem_toFinset.mpr hb)
  38     · intro p hp q hq
  39       obtain ⟨p',hp',rfl⟩:=List.mem_map.mp hp
  40       obtain ⟨j,hj,hq⟩:=List.mem_flatMap.mp hq
  41       obtain ⟨q',hq',rfl⟩:=List.mem_map.mp hq
  42       left
  43       intro h
  44       have hij:i=j:=h
  45       subst j
