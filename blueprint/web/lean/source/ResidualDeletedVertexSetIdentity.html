  11 theorem deleted_path_family_interiors_identity (F:SimpleGraph V)
  12     (s₁ s₂:Finset I) (hs:Disjoint s₁ s₂) (u v:I→V) (P:∀i,F.Walk (u i) (v i))
  13     (c d:ℕ) (T:I→Finset ℕ) (hT:∀i,T i⊆Finset.range d):
  14     let Z:=DeletedPathIndex s₁ s₂ d T
  15     let U:Z→V:=deletedPathStart F s₁ s₂ u v P c d T
  16     let W:Z→V:=deletedPathEnd F s₁ s₂ u v P c d T
  17     let paths:∀z:Z,F.Walk (U z) (W z):=deletedPathWalk F s₁ s₂ u v P c d T (fun i _=>hT i);
  18     (Finset.univ.biUnion (fun z=>(paths z).support.toFinset \ {U z,W z})) =
  19       (s₁∪s₂).biUnion (pathFamilyLocalDeletion F s₁ u v P c d T hT) := by
  20   dsimp only
  21   ext x
  22   constructor
  23   · intro hx
  24     obtain ⟨z,_,hx⟩:=Finset.mem_biUnion.mp hx
  25     cases z with
  26     | inl i =>
  27       refine Finset.mem_biUnion.mpr ⟨i.1,Finset.mem_union_left _ i.2,?_⟩
  28       simpa [pathFamilyLocalDeletion,i.2,centralPathInterior,deletedPathWalk,deletedPathStart,deletedPathEnd] using hx
  29     | inr z =>
  30       rcases z with ⟨i,r⟩
  31       have hinot:i.1∉s₁ := by intro h;exact Finset.disjoint_left.mp hs h i.2
  32       refine Finset.mem_biUnion.mpr ⟨i.1,Finset.mem_union_right _ i.2,?_⟩
  33       simp only [pathFamilyLocalDeletion,hinot,ite_false]
  34       apply Finset.mem_biUnion.mpr
  35       refine ⟨r,by simp,?_⟩
  36       simpa [deletedPathWalk,deletedPathStart,deletedPathEnd] using hx
  37   · intro hx
  38     obtain ⟨i,hi,hx⟩:=Finset.mem_biUnion.mp hx
  39     by_cases hi₁:i∈s₁
  40     · let z:DeletedPathIndex s₁ s₂ d T:=.inl ⟨i,hi₁⟩
  41       refine Finset.mem_biUnion.mpr ⟨z,Finset.mem_univ _,?_⟩
  42       simpa [z,pathFamilyLocalDeletion,hi₁,centralPathInterior,deletedPathWalk,deletedPathStart,deletedPathEnd] using hx

  44         rcases Finset.mem_union.mp hi with h|h
  45         · exact (hi₁ h).elim
  46         · exact h
  47       simp only [pathFamilyLocalDeletion,hi₁,ite_false] at hx

  49       let z:DeletedPathIndex s₁ s₂ d T:=.inr ⟨⟨i,hi₂⟩,r⟩
