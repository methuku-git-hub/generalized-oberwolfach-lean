   9 theorem interior_image_separated_by_boundary_and_avoidance
  10     {X Y:Type*} [DecidableEq X] [DecidableEq Y]
  11     (A D:Finset X) (hDA:D⊆A) (R:Finset Y)
  12     (fA:{x // x∈A}↪Y) (fC:{x // x∉D}→Y)
  13     (hbound:∀x (hx:x∈A) (hxD:x∉D),fA ⟨x,hx⟩=fC ⟨x,hxD⟩)
  14     (hinside:∀x (hx:x∈D),fA ⟨x,hDA hx⟩∈R)
  15     (havoid:∀x (hx:x∉D),x∉A→fC ⟨x,hx⟩∉R) :
  16     ∀x (hx:x∈D) y (hy:y∉D),fA ⟨x,hDA hx⟩≠fC ⟨y,hy⟩ := by
  17   intro x hx y hy heq
  18   by_cases hyA:y∈A
  19   · rw [←hbound y hyA hy] at heq
  20     have hxy:x=y:=congrArg Subtype.val (fA.injective heq)
  21     exact hy (hxy ▸ hx)
  22   · exact havoid y hy hyA (heq ▸ hinside x hx)
  23 end Oberwolfach
