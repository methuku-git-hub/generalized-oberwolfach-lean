  14 theorem clique_assigned_rejected_iff
  15     {V C : Type*} (G : SimpleGraph V) (T : Finset V)
  16     (hT : ∀ u∈T,∀ v∈T,u≠v → G.Adj u v) (σ : V → C×Bool) (c : C) :
  17     (colourAssignedIn T σ c ∧ ¬colourRejectedIn G T σ c) ↔
  18       ∃ v∈T,coinRetainedColour G σ v=some c := by
  19   constructor
  20   · rintro ⟨⟨v,hv,hc⟩,hn⟩
  21     refine ⟨v,hv,(coinRetainedColour_eq_some G σ v c).mpr ⟨hc,?_⟩⟩
  22     by_contra hbad
  23     exact hn ⟨v,hv,hc,hbad⟩
  24   · rintro ⟨v,hv,hkeep⟩
  25     obtain ⟨hc,hsurv⟩ := (coinRetainedColour_eq_some G σ v c).mp hkeep
  26     refine ⟨⟨v,hv,hc⟩,?_⟩
  27     rintro ⟨u,hu,huc,hbad⟩
  28     by_cases he : u=v
  29     · subst u
  30       exact hbad hsurv
  31     · exact hsurv.2 u (hT v hv u hu (Ne.symm he)) (huc.trans hc.symm)

  33 def nibbleAvailableColours {V C J : Type*} (G : SimpleGraph V)
  34     (Q : Finset J) (T : J → Finset V) (L : Finset C) (σ : V → C×Bool) : Finset C :=
  35   L.filter (fun c=>∀ j∈Q,∀ v∈T j,coinRetainedColour G σ v≠some c)

  37 theorem nibbleAvailableColours_mem_iff
  38     {V C J : Type*} (G : SimpleGraph V) (Q : Finset J) (T : J → Finset V)
  39     (hT : ∀ j∈Q,∀ u∈T j,∀ v∈T j,u≠v → G.Adj u v)
  40     (L : Finset C) (σ : V → C×Bool) (c : C) :
  41     c∈nibbleAvailableColours G Q T L σ ↔ c∈L ∧
  42       (assignedSetIndices Q T σ c).card=(rejectedSetIndices G Q T σ c).card := by
  43   classical
  44   have hsub := rejectedSetIndices_subset_assigned G Q T σ c
  45   constructor
  46   · intro hc
  47     obtain ⟨hc,hno⟩ := Finset.mem_filter.mp hc
  48     refine ⟨hc,?_⟩
  49     apply congrArg Finset.card
  50     apply Finset.Subset.antisymm _ hsub
  51     intro j hj
  52     obtain ⟨hj,ha⟩ := Finset.mem_filter.mp hj
  53     refine Finset.mem_filter.mpr ⟨hj,?_⟩
  54     by_contra hn
  55     obtain ⟨v,hv,hkeep⟩ := (clique_assigned_rejected_iff G (T j) (hT j hj) σ c).mp ⟨ha,hn⟩
  56     exact hno j hj v hv hkeep
  57   · rintro ⟨hc,hcard⟩
  58     have he := Finset.eq_of_subset_of_card_le hsub hcard.le
  59     apply Finset.mem_filter.mpr
  60     refine ⟨hc,?_⟩
  61     intro j hj v hv hkeep
  62     obtain ⟨ha,hn⟩ := (clique_assigned_rejected_iff G (T j) (hT j hj) σ c).mpr ⟨v,hv,hkeep⟩
  63     have hja : j∈assignedSetIndices Q T σ c := Finset.mem_filter.mpr ⟨hj,ha⟩
  64     have hjb : j∈rejectedSetIndices G Q T σ c := he.symm ▸ hja
  65     exact hn (Finset.mem_filter.mp hjb).2

  67 theorem threshold_diagonal_sum {A B k : ℕ} (hB : B≤A) (hA : A≤k) :
  68     (∑ a∈Finset.range (k+1),
  69       ((if a≤A ∧ a≤B then (1 : ℝ) else 0)-
  70         (if a+1≤A ∧ a≤B then (1 : ℝ) else 0)))=
  71       if A=B then 1 else 0 := by
  72   have hp (a : ℕ) :
  73       ((if a≤A ∧ a≤B then (1 : ℝ) else 0)-
  74         (if a+1≤A ∧ a≤B then (1 : ℝ) else 0))=
  75         if A=B ∧ a=A then 1 else 0 := by
  76     by_cases h₁ : a≤A <;> by_cases h₂ : a≤B <;>
  77       by_cases h₃ : a+1≤A <;> by_cases h₄ : A=B ∧ a=A <;>
  78         simp only [h₁,h₂,h₃,h₄,true_and,false_and,and_true,and_false,
  79           ite_true,ite_false] <;> try norm_num <;> omega
  80   simp_rw [hp]
  81   by_cases he : A=B
  82   · subst B
  83     simp [Finset.mem_range.mpr (Nat.lt_succ_of_le hA)]
  84   · simp [he]

  86 theorem nibbleAvailableColours_card
  87     {V C J : Type*} (G : SimpleGraph V) (Q : Finset J) (T : J → Finset V)
  88     (hT : ∀ j∈Q,∀ u∈T j,∀ v∈T j,u≠v → G.Adj u v)
  89     (L : Finset C) (σ : V → C×Bool) :
  90     ((nibbleAvailableColours G Q T L σ).card : ℝ)=
  91       ∑ a∈Finset.range (Q.card+1),
  92         (nibbleColourCount G Q T L a a σ-nibbleColourCount G Q T L (a+1) a σ) := by
  93   classical
  94   have hcard : ((nibbleAvailableColours G Q T L σ).card : ℝ)=
  95       ∑ c∈L,if (assignedSetIndices Q T σ c).card=
  96         (rejectedSetIndices G Q T σ c).card then (1 : ℝ) else 0 := by
  97     rw [Finset.sum_boole]
  98     apply congrArg (fun n : ℕ=>(n : ℝ))
  99     apply congrArg Finset.card
 100     ext c
 101     simp only [nibbleAvailableColours_mem_iff G Q T hT,Finset.mem_filter]
 102   rw [hcard]
 103   have hc (a b : ℕ) : nibbleColourCount G Q T L a b σ=
 104       ∑ c∈L,if a≤(assignedSetIndices Q T σ c).card ∧
 105         b≤(rejectedSetIndices G Q T σ c).card then (1 : ℝ) else 0 :=
 106     (Finset.sum_boole _ _).symm
 107   simp_rw [hc,← Finset.sum_sub_distrib]
 108   rw [Finset.sum_comm]
 109   apply Finset.sum_congr rfl
 110   intro c _
 111   exact (threshold_diagonal_sum
 112     (Finset.card_le_card (rejectedSetIndices_subset_assigned G Q T σ c))
 113     (Finset.card_le_card (Finset.filter_subset _ _))).symm
