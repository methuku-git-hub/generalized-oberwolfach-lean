  15 theorem ordinary_relative_common_lower (P Q F : SimpleGraph V) (T S : Finset V)
  16     (h j : ℕ) (δ μ p q s ε : ℝ)
  17     (hδ : δ≤1) (hμ : 0≤μ) (hp : 0<p) (hp1 : p≤1) (hq : p≤q) (hs : p≤s)
  18     (hS : S.card≤h) (hj : j≤2)
  19     (hdeg : ∀ v,(δ+μ)*Fintype.card V≤(P.degree v:ℝ))
  20     (hF : ∀ v,(F.degree v:ℝ)≤μ*Fintype.card V)
  21     (hmargin : 0≤1-(h:ℝ)*(1-δ))
  22     (hrelative : q^S.card*s^j*((common Finset.univ S (fun x=>(P\F).neighborFinset x)).card:ℝ)-
  23       ε*Fintype.card V≤((common T S (fun x=>Q.neighborFinset x)).card:ℝ)) :
  24     (p^(h+2)*(1-(h:ℝ)*(1-δ))-ε)*Fintype.card V≤
  25       ((common T S (fun x=>Q.neighborFinset x)).card:ℝ) := by
  26   have hdeg' (v : V) : δ*Fintype.card V≤((P\F).degree v:ℝ) := by
  27     have hh : (P.degree v:ℝ)≤(P\F).degree v+F.degree v := by
  28       exact_mod_cast graph_degree_le_sdiff_add P F v
  29     nlinarith only [hh,hdeg v,hF v]
  30   have hc := common_neighbour_card_lower (Finset.univ : Finset V) S
  31     (fun x=>(P\F).neighborFinset x) δ (fun v _=>by
  32       simpa only [Finset.inter_univ,Finset.univ_inter,Finset.card_univ,
  33         SimpleGraph.card_neighborFinset_eq_degree] using hdeg' v)
  34   simp only [Finset.card_univ] at hc
  35   have hSr : (S.card:ℝ)≤h := by exact_mod_cast hS
  36   have hcoef : 1-(h:ℝ)*(1-δ)≤1-(S.card:ℝ)*(1-δ) := by
  37     nlinarith only [mul_le_mul_of_nonneg_right hSr (sub_nonneg.mpr hδ)]
  38   have hc' : (1-(h:ℝ)*(1-δ))*Fintype.card V≤
  39       ((common Finset.univ S (fun x=>(P\F).neighborFinset x)).card:ℝ) :=
  40     (mul_le_mul_of_nonneg_right hcoef (Nat.cast_nonneg _)).trans hc
  41   have hpq : p^S.card≤q^S.card := pow_le_pow_left₀ hp.le hq _
  42   have hps : p^j≤s^j := pow_le_pow_left₀ hp.le hs _
  43   have hpowers : p^(h+2)≤q^S.card*s^j := by
  44     calc
  45       _ ≤ p^(S.card+j) := pow_le_pow_of_le_one hp.le hp1 (Nat.add_le_add hS hj)
  46       _ = p^S.card*p^j := pow_add _ _ _
  47       _ ≤ _ := mul_le_mul hpq hps (by positivity) (pow_nonneg (hp.le.trans hq) _)
  48   have hm := mul_le_mul hpowers hc' (mul_nonneg hmargin (Nat.cast_nonneg _))
  49     (mul_nonneg (pow_nonneg (hp.le.trans hq) _) (pow_nonneg (hp.le.trans hs) _))
  50   nlinarith only [hm,hrelative]

  52 /-- A uniform density margin for the triangle and longer-cycle cases. -/
  53 def ordinaryLocalBaseDegree (m : ℕ) : ℝ := if m=2 then 3/4 else 1/2
  54 
  55 def ordinaryLocalRootBound (m : ℕ) : ℕ := if m=2 then 3 else 2

  57 theorem ordinary_local_margin (m : ℕ) (α : ℝ) (hα : 0<α) :
  58     let u := min α (1/4)
  59     let δ := ordinaryLocalBaseDegree m+u/2
  60     δ≤1 ∧ 0<1-(ordinaryLocalRootBound m:ℝ)*(1-δ) := by
  61   dsimp
  62   have hu : 0<min α (1/4:ℝ) := lt_min hα (by norm_num)
  63   have hu1 : min α (1/4:ℝ)≤1/4 := min_le_right _ _
  64   unfold ordinaryLocalBaseDegree ordinaryLocalRootBound
  65   split_ifs <;> norm_num <;> constructor <;> linarith
