  16     (a : Fin (∑ i,L i) → X → X → ℝ) (d : Fin (∑ i,L i) → X → ℝ)

  29       ∀ x z,κ*(d (finRotate (∑ i,L i) (finRotate (∑ i,L i) (finRotate (∑ i,L i) i))) z/(∑ z,d 0 z)) ≤
  30       ∑ y,∑ w,cyclicCapacityTransition a d i x y*cyclicCapacityTransition a d (finRotate (∑ i,L i) i) y w*
  31         cyclicCapacityTransition a d (finRotate (∑ i,L i) (finRotate (∑ i,L i) i)) w z) :
  32     ∃ w : (Fin (∑ i,L i) → X) → ℝ,
  33       (∀ v,0≤w v) ∧
  34       (∀ i x y,(∑ v,if v i=x ∧ v (finRotate (∑ i,L i) i)=y then w v else 0)=a i x y) ∧
  35       (∀ v,(∃ i,a i (v i) (v (finRotate (∑ i,L i) i))=0) → w v=0) ∧
  36       (∀ v,(∀ i,0<a i (v i) (v (finRotate (∑ i,L i) i))) →
  37         longCycleLower L a₀ d₀ d₁/n^((∑ i,L i)-2) ≤ w v ∧
  38           w v ≤ longCycleUpper L a₁ d₀ d₁/n^((∑ i,L i)-2)) := by

  70   have hm (i : Fin 3) (j : ℕ) (_ : s i≤j) (hji : j+3≤s i+L i)
