  15 theorem cycle_sample_support_probability {V Ω : Type*}
  16     [Fintype V] [DecidableEq V] [Fintype Ω] {m r : ℕ}
  17     [Nonempty (FixedSubset V m)] (hm : m≤Fintype.card V)
  18     (μ : FiniteLaw Ω) (S : Ω → FixedSubset V m)
  19     (hlaw : μ.map S=FiniteLaw.uniform (FixedSubset V m))
  20     (J : Ω → SimpleGraph V) (hJ : ∀ ω,(J ω).support⊆(S ω).val)
  21     (good : Ω → Prop) (C : SimpleGraph V) (hC : IsEmbeddedCycle C r) :
  22     μ.probability (fun ω => good ω ∧ C≤J ω)≤((m:ℝ)/Fintype.card V)^r := by
  23   let T := C.support.toFinset
  24   have hT : T.card=r := hC.support_card
  25   calc
  26     _ ≤ μ.probability (fun ω => T⊆(S ω).val) := by
  27       apply finiteLaw_probability_mono
  28       intro ω hω x hx
  29       have hx' : x∈C.support := Set.mem_toFinset.mp hx
  30       exact hJ ω (SimpleGraph.support_mono hω.2 hx')
  31     _ = (FiniteLaw.uniform (FixedSubset V m)).probability (fun A => T⊆A.val) := by
  32       rw [← finiteLaw_map_probability μ S (fun A => T⊆A.val),hlaw]
  33     _ ≤ ((m:ℝ)/Fintype.card V)^T.card := fixedSubset_joint_inclusion_le hm T
  34     _ = _ := by rw [hT]
