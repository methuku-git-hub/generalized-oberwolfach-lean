  11 def prescribedPairClosed (Q:SimpleGraph X) (e:X×X):Finset X:=
  12   prescribedClosedNeighbourhood Q e.1∪prescribedClosedNeighbourhood Q e.2
  13 
  14 def prescribedPairsApart (Q:SimpleGraph X) (e f:X×X):Prop:=
  15   f.1∉prescribedPairClosed Q e ∧ f.2∉prescribedPairClosed Q e

  17 theorem prescribed_position_greedy_pairs (Q:SimpleGraph X) (D:X→X→Prop)
  18     (hdeg:∀ x,Q.degree x≤2) (hrep:∀x y,Q.Adj x y→D x y ∨ D y x):
  19     ∀ m:ℕ,∀ S:Finset X,2*S.card+8*m≤Q.edgeFinset.card→
  20       ∃ es:List (X×X),es.length=m ∧
  21         (∀ e∈es,D e.1 e.2 ∧ Q.Adj e.1 e.2 ∧ e.1∉S ∧ e.2∉S) ∧
  22         es.Pairwise (prescribedPairsApart Q) := by
  23   intro m
  24   induction m with
  25   | zero =>
  26     intro S h
  27     exact ⟨[],rfl,by simp,by simp⟩
  28   | succ m ih =>
  29     intro S h
  30     have hlt:2*S.card<Q.edgeFinset.card:=by omega
  31     obtain ⟨u,v,huv,hu,hv⟩:=prescribed_position_edge_choice Q S hdeg hlt
  32     have hor:=hrep u v huv
  33     have hex:∃ e:X×X,D e.1 e.2 ∧ Q.Adj e.1 e.2 ∧ e.1∉S ∧ e.2∉S:=by
  34       rcases hor with hor|hor
  35       · exact ⟨(u,v),hor,huv,hu,hv⟩
  36       · exact ⟨(v,u),hor,huv.symm,hv,hu⟩
  37     obtain ⟨e,heD,heQ,he1,he2⟩:=hex
  38     let T:=S∪prescribedPairClosed Q e
  39     have hpairCard:(prescribedPairClosed Q e).card≤4:=prescribed_edge_closed_neighbourhood_card Q hdeg e.1 e.2 heQ
  40     have hT:T.card≤S.card+4:=(Finset.card_union_le _ _).trans (Nat.add_le_add_left hpairCard _)
  41     have hbudget:2*T.card+8*m≤Q.edgeFinset.card:=by omega
  42     obtain ⟨rest,hlen,hrest,hpair⟩:=ih T hbudget
  43     refine ⟨e::rest,by simp [hlen],?_,?_⟩
  44     · intro f hf
  45       rcases List.mem_cons.mp hf with hf|hf
  46       · subst f;exact ⟨heD,heQ,he1,he2⟩
  47       · obtain ⟨hfD,hfQ,hf1,hf2⟩:=hrest f hf
  48         refine ⟨hfD,hfQ,?_,?_⟩
  49         · intro hx;exact hf1 (Finset.mem_union_left _ hx)
  50         · intro hx;exact hf2 (Finset.mem_union_left _ hx)
  51     · apply List.pairwise_cons.mpr
  52       refine ⟨?_,hpair⟩
  53       intro f hf
  54       have hf1: f.1∉T:=(hrest f hf).2.2.1
  55       have hf2: f.2∉T:=(hrest f hf).2.2.2
  56       exact ⟨fun hx=>hf1 (Finset.mem_union_right _ hx),fun hx=>hf2 (Finset.mem_union_right _ hx)⟩
  57 end Oberwolfach
