  13 theorem one_position_indicator_total {I V : Type*} [Fintype I] [Fintype V] [DecidableEq I]
  14     (i : I) (v : V) :
  15     (∑ C : I → V,if C i=v then (1:ℝ) else 0)=(Fintype.card V:ℝ)^(Fintype.card I-1) := by
  16   let S : Set (I → V) := {C | ∀ j∈({i}:Finset I),C j=v}
  17   have he := Finset.sum_congr_set S
  18     (fun C => if C i=v then (1:ℝ) else 0) (fun _ : S => (1:ℝ))
  19     (by intro C hC;exact if_pos (hC i (by simp)))
  20     (by intro C hC;apply if_neg;intro h;apply hC;intro j hj;simpa only [Finset.mem_singleton.mp hj] using h)
  21   rw [he]
  22   simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul,mul_one]
  23   change (Fintype.card {C : I → V // ∀ j∈({i}:Finset I),C j=(fun _ : I => v) j}:ℝ)=_
  24   rw [fixedCoordinates_card,Finset.card_singleton,Nat.cast_pow]

  26 theorem marked_cycle_spoke_count {V E K : Type*} [Fintype V] [DecidableEq V]
  27     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (π : Equiv.Perm V) (b : ℕ)
  28     (j : Bool) (v : V) :
  29     (∑ C : MarkedCycleTuple tail head ktail khead m π b,markedSpokeIndicator C.val j v)≤
  30       (m+1)*(Fintype.card V:ℝ)^m := by
  31   apply (finite_subtype_sum_le (IsMarkedCycleTuple tail head ktail khead m π b)
  32     (fun C => markedSpokeIndicator C j v) (fun C => markedSpokeIndicator_nonneg C j v)).trans
  33   have hn : 0≤(Fintype.card V:ℝ)^m := by positivity
  34   cases j
  35   · simp only [markedSpokeIndicator,Bool.false_eq_true,ite_false]
  36     rw [Finset.sum_comm]
  37     have hh (i : Fin m) : (∑ C : Fin (m+1) → V,if C i.succ=v then (1:ℝ) else 0)=
  38         (Fintype.card V:ℝ)^m := by
  39       have he := one_position_indicator_total i.succ v
  40       simp only [Fintype.card_fin,Nat.add_sub_cancel] at he
  41       convert he using 1
  42       apply Finset.sum_congr rfl
  43       intro C _
  44       split_ifs <;> rfl
  45     simp only [hh,Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul]
  46     nlinarith
  47   · simp only [markedSpokeIndicator,ite_true]
  48     have hh := one_position_indicator_total (0:Fin (m+1)) v
  49     simp only [Fintype.card_fin,Nat.add_sub_cancel] at hh
  50     have he : (∑ C : Fin (m+1) → V,if C 0=v then (1:ℝ) else 0)=(Fintype.card V:ℝ)^m := by
  51       convert hh using 1
  52       apply Finset.sum_congr rfl
  53       intro C _
  54       split_ifs <;> rfl
  55     rw [he]
  56     nlinarith [show (0:ℝ)≤m by positivity]
