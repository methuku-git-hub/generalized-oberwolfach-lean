  31     (T T':Finset V) (hT:Disjoint T T') (P Q:Finset (Finset V⊕(V×U)))
  32     (hP:∀ e∈P,MarkedResourceSupported tail head ktail khead π T e)
  33     (hQ:∀ e∈Q,MarkedResourceSupported tail head ktail khead π T' e):Disjoint P Q:=by
  34   apply Finset.disjoint_left.mpr
  35   intro e he hf
  36   cases e with
  37   | inl e=>
