   8 theorem marked_successor_margin_cutoff (α:ℝ) (hα:0<α):
   9     ∃ K:ℕ, 2≤K ∧ ∀ k:ℕ,K≤k→successorCycleRatio k+α/4≤1/2+α/2 := by
  10   obtain ⟨K,hK⟩:=exists_nat_gt (1/α)
  11   refine ⟨max K 2,le_max_right _ _,?_⟩
  12   intro k hk
  13   have hKk:K≤k := (le_max_left K 2).trans hk
  14   have hKkR:(K:ℝ)≤k := by exact_mod_cast hKk
  15   have hkreal:(1/α:ℝ)<k := lt_of_lt_of_le hK hKkR
  16   have hkα:1<α*(k:ℝ) := by
  17     have hh:=mul_lt_mul_of_pos_left hkreal hα
  18     simpa [ne_of_gt hα] using hh
  19   have hk0:(0:ℝ)≤k := Nat.cast_nonneg _
  20   have hden:(0:ℝ)<2*((k:ℝ)+2)-1 := by linarith
  21   have hrat:successorCycleRatio k≤1/2+α/4 := by
  22     unfold successorCycleRatio
  23     apply (div_le_iff₀ hden).mpr
  24     nlinarith
  25   linarith
