   8 theorem permutation_invariant_image {V W : Type*} [DecidableEq V]
   9     (f : W→V) (π : Equiv.Perm V) (ρ : Equiv.Perm W)
  10     (hcomm : ∀ x,f (ρ x)=π (f x)) (A : Finset W) (hA : ∀ x,ρ x∈A ↔ x∈A) :
  11     ∀ x,π x∈A.image f ↔ x∈A.image f := by
  12   intro x
  13   constructor
  14   · intro hx
  15     obtain ⟨y,hy,he⟩ := Finset.mem_image.mp hx
  16     have hy' : ρ.symm y∈A := (hA (ρ.symm y)).mp (by simpa only [ρ.apply_symm_apply] using hy)
  17     have he' : f (ρ.symm y)=x := π.injective (by rw [← hcomm,ρ.apply_symm_apply,he])
  18     exact Finset.mem_image.mpr ⟨ρ.symm y,hy',he'⟩
  19   · intro hx
  20     obtain ⟨y,hy,he⟩ := Finset.mem_image.mp hx
  21     exact Finset.mem_image.mpr ⟨ρ y,(hA y).mpr hy,(hcomm y).trans (congrArg π he)⟩
  22 
  23 end Oberwolfach
