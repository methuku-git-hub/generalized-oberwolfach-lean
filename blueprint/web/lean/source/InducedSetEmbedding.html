  15 def inducedSetEmbedding (B : Finset V) {k : ℕ} (S : FixedSubset {v // v∈B} k) : FixedSubset V k :=
  16   ⟨S.val.map (Function.Embedding.subtype _),by rw [Finset.card_map,S.property]⟩

  18 theorem inducedSetEmbedding_injective (B : Finset V) (k : ℕ) :
  19     Function.Injective (inducedSetEmbedding B (k:=k)) := by
  20   intro S T h
  21   apply Subtype.ext
  22   have hh := congrArg Subtype.val h
  23   exact Finset.map_injective (Function.Embedding.subtype _) hh

  30 theorem inducedSetEmbedding_degree (P : SimpleGraph V) (B : Finset V) {k : ℕ}
  31     (S : FixedSubset {v // v∈B} k) (v : {v // v∈B}) :
  32     (((P.induce (B:Set V)).neighborFinset v)∩S.val).card=
  33       (P.neighborFinset v.val∩(inducedSetEmbedding B S).val).card := by
  34   have he : (P.induce (B:Set V)).neighborFinset v=(P.neighborFinset v.val).subtype (fun x=>x∈B) := by
  35     ext x
  36     simp only [SimpleGraph.mem_neighborFinset,SimpleGraph.induce_adj,Finset.mem_subtype]
  37   rw [he]
  38   convert subtype_sample_intersection_card (P.neighborFinset v.val) B S.val using 1 <;> congr!

  41 theorem inducedSetEmbedding_conflict_free (F : SimpleGraph V) (B : Finset V) {k : ℕ}
  42     (S : FixedSubset {v // v∈B} k)
  43     (h : ∀ x∈S.val,∀ y∈S.val,¬(F.induce (B:Set V)).Adj x y) :
  44     ∀ x∈(inducedSetEmbedding B S).val,∀ y∈(inducedSetEmbedding B S).val,¬F.Adj x y := by
  45   intro x hx y hy hxy
  46   obtain ⟨x',hx',rfl⟩ := Finset.mem_map.mp hx
  47   obtain ⟨y',hy',rfl⟩ := Finset.mem_map.mp hy
  48   exact h x' hx' y' hy' hxy

  51 theorem induced_degree_le_ambient (F : SimpleGraph V) (B : Finset V) (v : {v // v∈B}) :
  52     (F.induce (B:Set V)).degree v≤F.degree v.val := by
  53   rw [induced_degree_intersection,← SimpleGraph.card_neighborFinset_eq_degree]
