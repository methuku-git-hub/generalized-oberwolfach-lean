  21 theorem endpoint_source_counts (P R H F : SimpleGraph V) (A B : Finset V) (x : V)
  22     (hAB : Disjoint A B) (hR : R≤P\F)
  23     (p q s ε L κ : ℝ) (hp : 0<p) (hp1 : p≤1) (hq : p≤q) (hs : p≤s)
  24     (hκ : 0<κ) (hB : κ*A.card≤(B.card:ℝ)) (hL : 0≤L) (hLa : L≤(A.card:ℝ)/24)
  25     (hε : ε≤min (p^4/16) (p^3/2))
  26     (hdeg : ∀ v∈insert x A,(3/4:ℝ)*A.card≤((A∩P.neighborFinset v).card:ℝ))
  27     (hF : ∀ v,(F.degree v:ℝ)≤L)
  28     (hrows : ∀ X J : Finset V,x∈X → X⊆insert x A → J⊆B →
  29       X.card≤3 → J.card≤2 → X.card+J.card≤4 →
  30       q^X.card*s^J.card*((common A X (fun v=>(P\F).neighborFinset v)).card:ℝ)-ε*A.card≤
  31         ((common (common A X (fun v=>R.neighborFinset v)) J (fun v=>H.neighborFinset v)).card:ℝ))
  32     (hindices : ∀ X : Finset V,x∈X → X⊆insert x A → (X.card=1 ∨ X.card=3) →
  33       (∀ v∈X,∀ w∈X,¬F.Adj v w) →
  34       (s^X.card-ε)*B.card≤((common B X (fun v=>H.neighborFinset v)).card:ℝ)) :
  35     EndpointNeighbourCounts (endpointHostGraph R H A B) F
  36       (common A {x} (fun v=>R.neighborFinset v)) (common B {x} (fun v=>H.neighborFinset v))
  37       (endpointCountCoefficient p κ*A.card) := by
  38   let S := common A {x} (fun v=>R.neighborFinset v)
  39   let T := common B {x} (fun v=>H.neighborFinset v)
  40   have hSA : S⊆A := fun v hv=>((mem_common_iff _ _ _ _).mp hv).1
  41   have hTB : T⊆B := fun v hv=>((mem_common_iff _ _ _ _).mp hv).1
  42   have hSx (v : V) (hv : v∈S) : R.Adj x v := by
  43     simpa only [Finset.mem_singleton,forall_eq,SimpleGraph.mem_neighborFinset] using ((mem_common_iff _ _ _ _).mp hv).2
  44   have hSF (v : V) (hv : v∈S) : ¬F.Adj x v := (hR (hSx v hv)).2
  45   have hscale : endpointCountCoefficient p κ*A.card≤(p^3/2)*B.card := by
  46     calc
  47       _ ≤ (κ*p^3/2)*A.card := mul_le_mul_of_nonneg_right (min_le_right _ _) (Nat.cast_nonneg _)
  48       _ = (p^3/2)*(κ*A.card) := by ring
  49       _ ≤ _ := mul_le_mul_of_nonneg_left hB (by positivity)
  50   constructor
  51   · apply hscale.trans
  52     apply endpoint_relative_index_lower H B {x} p s ε hp hp1 hs (by simp)
  53       (hε.trans (min_le_right _ _))
  54     apply hindices {x} (by simp) (by simp) (Or.inl (by simp))
  55     intro v hv w hw
  56     simp only [Finset.mem_singleton] at hv hw
  57     subst v;subst w;exact F.loopless.irrefl x
  58   · intro Y J hY hJ hYc hJc hYJ
  59     rw [endpointHostGraph_row_common R H A B hAB x Y J hY (hJ.trans hTB)]
  60     have hXY : insert x Y⊆insert x A := Finset.insert_subset_insert x (hY.trans hSA)
  61     have hcard : (insert x Y).card≤3 := (Finset.card_insert_le _ _).trans (by omega)
  62     have hcardJ : (insert x Y).card+J.card≤4 := by have hh:=Finset.card_insert_le x Y;omega
  63     have hh := endpoint_relative_row_lower P R H F A (insert x Y) J p q s ε L hp hp1 hq hs
  64       hcard hcardJ hL hLa (hε.trans (min_le_left _ _))
  65       (fun v hv=>hdeg v (hXY hv)) (fun v _=>hF v)
  66       (hrows _ _ (by simp) hXY (hJ.trans hTB) hcard hJc hcardJ)
  67     exact (mul_le_mul_of_nonneg_right (min_le_left (p^4/16) (κ*p^3/2)) (Nat.cast_nonneg _)).trans hh
  68   · intro v hv w hw hvw hvwF
  69     rw [endpointHostGraph_index_common R H A B hAB x {v,w}
  70       (by simp only [Finset.insert_subset_iff,Finset.singleton_subset_iff];exact ⟨hSA hv,hSA hw⟩)]
  71     have hxv : x≠v := (hSx v hv).ne
  72     have hxw : x≠w := (hSx w hw).ne
  73     have hcard : (insert x {v,w}:Finset V).card=3 := by simp [hxv,hxw,hvw]
  74     have hX : insert x {v,w}⊆insert x A := by
  75       apply Finset.insert_subset_insert
  76       simp only [Finset.insert_subset_iff,Finset.singleton_subset_iff]
  77       exact ⟨hSA hv,hSA hw⟩
  78     have hfree : ∀ z∈insert x ({v,w}:Finset V),∀ z'∈insert x ({v,w}:Finset V),¬F.Adj z z' := by
  79       have hxvF := hSF v hv
  80       have hxwF := hSF w hw
  81       have hvxF : ¬F.Adj v x := fun h=>hxvF h.symm
  82       have hwxF : ¬F.Adj w x := fun h=>hxwF h.symm
  83       have hwvF : ¬F.Adj w v := fun h=>hvwF h.symm
  84       intro z hz z' hz'
  85       simp only [Finset.mem_insert,Finset.mem_singleton] at hz hz'
  86       rcases hz with rfl|rfl|rfl <;> rcases hz' with rfl|rfl|rfl <;> simp_all
  87     exact hscale.trans (endpoint_relative_index_lower H B (insert x {v,w}) p s ε hp hp1 hs
  88       hcard.le (hε.trans (min_le_right _ _)) (hindices _ (by simp) hX (Or.inr hcard) hfree))
