  15 theorem sparse_rooted_cycle_embedding (Q F : SimpleGraph V) (A W R : Finset V)
  16     (x : V) (hx : x∈R) (r : ℕ) (hr : 3≤r) (d c : ℝ) (hd : 0≤d)
  17     (hF : ∀ v,(F.degree v:ℝ)≤d)
  18     (hcount : ∀ v w,c≤((A∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  19     (hbudget : W.card+((R.card:ℝ)+r+5)*(d+1)<c) :
  20     ∃ f : Fin r ↪ V,f ⟨0,by omega⟩=x ∧
  21       (∀ i j,(SimpleGraph.cycleGraph r).Adj i j → Q.Adj (f i) (f j)) ∧
  22       (∀ i j,¬F.Adj (f i) (f j)) ∧
  23       ∀ i,i≠⟨0,by omega⟩ → f i∈A ∧ f i∉W ∧ f i∉R ∧ ∀ v∈R,¬F.Adj (f i) v := by
  24   have hd1 : 0≤d+1 := by linarith
  25   obtain ⟨y,hyA,hyW,hyR,hxy,_hxy,hyF⟩ := sparse_common_neighbour_avoiding Q F A W R x x d c hd hF
  26     (hcount x x) (by nlinarith only [hbudget,hd1,(show (0:ℝ)≤r from Nat.cast_nonneg _)])
  27   let D := R∪R.biUnion (fun v => F.neighborFinset v)
  28   let W' := W∪D
  29   let Z : Finset V := {x,y}
  30   have hZ : ∀ a∈Z,∀ b∈Z,¬F.Adj a b := by
  31     apply graph_independent_insert F ({y}:Finset V) x
  32     · intro a ha b hb
  33       have ha' : a=y := Finset.mem_singleton.mp ha
  34       have hb' : b=y := Finset.mem_singleton.mp hb
  35       rw [ha',hb']
  36       exact F.loopless.irrefl y
  37     · intro a ha h
  38       have ha' : a=y := Finset.mem_singleton.mp ha
  39       rw [ha'] at h
  40       exact hyF x hx h.symm
  41   have hW' : (W'.card:ℝ)≤W.card+(R.card:ℝ)*(d+1) := by
  42     have hD := forbidden_used_neighbours_card F R d hF
  43     have hh : (W'.card:ℝ)≤W.card+(D.card:ℝ) := by exact_mod_cast Finset.card_union_le W D
  44     exact hh.trans (by dsimp [D];linarith only [hD])
  45   have hZc : (Z.card:ℝ)≤2 := by
  46     have hh := Finset.card_insert_le x ({y}:Finset V)
  47     simp only [Finset.card_singleton] at hh
  48     exact_mod_cast hh
  49   have hrR : ((r-1:ℕ):ℝ)≤r := by exact_mod_cast Nat.sub_le r 1
  50   have hpbudget : W'.card+((Z.card:ℝ)+(r-1:ℕ))*(d+1)<c := by
  51     nlinarith only [hbudget,hW',hZc,hrR,hd1]
  52   obtain ⟨p,hp,hlen,hclass,hind,_hZp⟩ := sparse_fixed_length_forbidden_path Q F A W' Z x y
  53     hxy.ne (by simp [Z]) (by simp [Z]) hZ (r-1) (by omega) d c hd hF hcount hpbudget
  54   let f := pathCycleEmbedding p hp hlen
  55   have hf0 : f ⟨0,by omega⟩=x := pathCycleEmbedding_first _ p hp hlen
  56   refine ⟨f,hf0,pathCycleEmbedding_hom hr p hp hlen hxy,?_,?_⟩
  57   · intro i j
  58     exact hind _ (p.getVert_mem_support _) _ (p.getVert_mem_support _)
  59   · intro i hi
  60     have hfx : f i≠x := by intro he;exact hi (f.injective (he.trans hf0.symm))
  61     rcases hclass (f i) (p.getVert_mem_support _) with hx'|hy'|hrest
  62     · exact (hfx hx').elim
  63     · rw [hy']
  64       exact ⟨hyA,hyW,hyR,hyF⟩
  65     · have hnotW : f i∉W := fun h => hrest.2.1 (Finset.mem_union_left _ h)
  66       have hnotR : f i∉R := fun h => hrest.2.1
  67         (Finset.mem_union_right _ (Finset.mem_union_left _ h))
  68       refine ⟨hrest.1,hnotW,hnotR,?_⟩
  69       intro v hv hFiv
  70       apply hrest.2.1
  71       exact Finset.mem_union_right _ (Finset.mem_union_right _ (Finset.mem_biUnion.mpr
  72         ⟨v,hv,(F.mem_neighborFinset v (f i)).mpr hFiv.symm⟩))
