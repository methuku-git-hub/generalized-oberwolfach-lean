   9 theorem positive_class_complement_dichotomy
  10     {I:Type} [Fintype I] [DecidableEq I] (ρ:I→ℝ) (r:ℝ)
  11     (hnonneg:∀i,0≤ρ i) (htotal:∑i,ρ i=1) (hlow:∀i,r≤ρ i) (a:I):
  12     ρ a=1 ∨ r≤1-ρ a:=by
  13   classical
  14   by_cases h:∃b:I,b≠a
  15   · right
  16     obtain ⟨b,hba⟩:=h
  17     have hmem:b∈(Finset.univ:Finset I).erase a:=by simp [hba]
  18     have hb:ρ b≤∑i∈(Finset.univ:Finset I).erase a,ρ i:=
  19       Finset.single_le_sum (fun i _=>hnonneg i) hmem
  20     have hs:=Finset.sum_erase_add Finset.univ ρ (Finset.mem_univ a)
  21     rw [htotal] at hs
  22     linarith [hlow b]
  23   · left
  24     have hsing:(Finset.univ:Finset I)={a}:=by
  25       ext i
  26       simp only [Finset.mem_univ,Finset.mem_singleton,true_iff]
  27       by_contra hia
  28       exact h ⟨i,hia⟩
  29     rw [hsing] at htotal
  30     simpa using htotal
  31 end Oberwolfach
