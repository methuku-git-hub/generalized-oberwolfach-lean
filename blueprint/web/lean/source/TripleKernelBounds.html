  14 theorem bipTripleKernel_bounds (H : A → B → Prop) (q : B → ℝ)
  15     (L : ℝ) (hL : 0≤L) (hq : ∀ u,0≤q u ∧ q u≤L) (x s : A) (u t : B) :
  16     0≤bipTripleKernel H q x u s t ∧
  17       bipTripleKernel H q x u s t≤L^3*(Fintype.card A:ℝ)*Fintype.card B := by
  18   unfold bipTripleKernel
  19   have hpoint (v : B) : 0≤bipAdj H x v*bipAdj H s v*(q v)^3*bipTripleIntersection H u v t ∧
  20       bipAdj H x v*bipAdj H s v*(q v)^3*bipTripleIntersection H u v t≤L^3*Fintype.card A := by
  21     have hqv := (hq v).1
  22     have hI := bipTripleIntersection_bounds H u v t
  23     constructor
  24     · exact mul_nonneg (mul_nonneg (mul_nonneg (bipAdj_nonneg H x v) (bipAdj_nonneg H s v))
  25         (pow_nonneg hqv 3)) hI.1
  26     · unfold bipAdj
  27       split_ifs <;> simp only [one_mul,zero_mul]
  28       all_goals first
  29       | positivity
  30       | exact mul_le_mul (pow_le_pow_left₀ hqv (hq v).2 3) hI.2 hI.1 (by positivity)
  31   constructor
  32   · exact Finset.sum_nonneg (fun v _=>(hpoint v).1)
  33   · calc
  34       _ ≤ ∑ _v : B,L^3*Fintype.card A := Finset.sum_le_sum (fun v _=>(hpoint v).2)
  35       _ = _ := by simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul];ring

  38     (L δ : ℝ) (hL : 0≤L) (hδ : 0≤δ)

  45   have he : bipTripleKernel H q x u s t-(Fintype.card A:ℝ)*p u*p t*bipWeightedCodegree H q x s=
  46       ∑ v,(bipAdj H x v*bipAdj H s v*(q v)^3)*
  47         (bipTripleIntersection H u v t-(Fintype.card A:ℝ)*p u*p v*p t) := by
  48     unfold bipTripleKernel bipWeightedCodegree
  49     simp only [mul_sub,Finset.sum_sub_distrib,Finset.mul_sum]
  50     congr 1
  51     apply Finset.sum_congr rfl;intro v _
  52     have hqp : (q v)^3*p v=(q v)^2 := by
  53       calc
  54         _ = (q v)^2*(q v*p v) := by ring
  55         _ = _ := by rw [hpq v,mul_one]
  56     calc
  57       _ = bipAdj H x v*bipAdj H s v*(Fintype.card A:ℝ)*p u*p t*(q v)^2 := by ring
  58       _ = _ := by rw [← hqp];ring
  59   rw [he]
  60   have hfactor (v : B) : |bipAdj H x v*bipAdj H s v*(q v)^3|≤L^3 := by
  61     have hqv := (hq v).1
  62     unfold bipAdj
  63     split_ifs <;> simp only [one_mul,zero_mul,abs_zero]
  64     all_goals first
  65     | positivity
  66     | rw [abs_of_nonneg (pow_nonneg hqv 3)]
  67       exact pow_le_pow_left₀ hqv (hq v).2 3
  68   exact (weighted_sum_abs_le _ _ (L^3) (by positivity) hfactor).trans
  69     (mul_le_mul_of_nonneg_left (bipTripleIntersection_error_sum H p hp δ hδ herr u t hut) (by positivity))
