  15 def longReservedComponentBlock
  16     (e:SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph)
  17     (c d t i:ℕ) (ht:t≤d) (hfit:(c*d+5)*(i+1)≤Fintype.card C) : Fin (c*t+5) ↪ C where
  18   toFun j:=e ⟨(c*d+5)*i+j.val,by
  19     have hprod:c*t≤c*d:=Nat.mul_le_mul_left c ht
  20     simp only [Nat.mul_add,Nat.mul_one] at hfit
  21     omega⟩
  22   inj' := by
  23     intro x y hxy
  24     have hh:=e.injective hxy
  25     apply Fin.ext
  26     have hv:=congrArg Fin.val hh
  27     simp only at hv
  28     omega

  47 theorem long_reserved_component_blocks_disjoint
  48     (e:SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph)
  49     (c d t t' i j:ℕ) (ht:t≤d) (ht':t'≤d)
  50     (hi:(c*d+5)*(i+1)≤Fintype.card C) (hj:(c*d+5)*(j+1)≤Fintype.card C) (hij:i≠j):
  51     Disjoint (Finset.univ.image (longReservedComponentBlock G C e c d t i ht hi))
  52       (Finset.univ.image (longReservedComponentBlock G C e c d t' j ht' hj)) := by
  53   apply Finset.disjoint_left.mpr
  54   intro x hxi hxj
  55   obtain ⟨a,_,ha⟩:=Finset.mem_image.mp hxi
  56   obtain ⟨b,_,hb⟩:=Finset.mem_image.mp hxj
  57   have heq:=e.injective (ha.trans hb.symm)
  58   have hv:=congrArg Fin.val heq
  59   have haMem:(c*d+5)*i+a.val∈longReservedExtendedBlock c d t i:=by
  60     simp only [longReservedExtendedBlock,Finset.mem_Icc]
  61     omega
  62   have hbMem:(c*d+5)*j+b.val∈longReservedExtendedBlock c d t' j:=by
  63     simp only [longReservedExtendedBlock,Finset.mem_Icc]
  64     omega
  65   have hdis:=long_reserved_extended_blocks_disjoint c d t t' i j ht ht' hij
  66   have hval:(c*d+5)*i+a.val=(c*d+5)*j+b.val:=by simpa [longReservedComponentBlock] using hv
  67   exact (Finset.disjoint_left.mp hdis) haMem (hval ▸ hbMem)
