  12 theorem cycle_hypergraph_near_perfect
  13     (k B:ℕ) (hk:2≤k) (K a γ ε:ℝ) (hK:0<K) (ha:0<a)
  14     (hγ:0<γ) (hγ1:γ≤1) (hε:0≤ε) (hmargin:a*k≤1-ε):
  15     ∃ n₀:ℕ,∀ n D:ℕ,n₀≤n→n≤D→D≤n^B→
  16     ∀ (V E:Type) [Fintype V] [Fintype E] [DecidableEq V],
  17     ∀ (R:E→Finset V),
  18     n≤Fintype.card V→Fintype.card V≤n^B→
  19     (∀ e,(R e).card=k)→
  20     (∀ v,(1-ε)*D≤(Fintype.card (HypergraphEdgesAt R v):ℝ) ∧
  21       Fintype.card (HypergraphEdgesAt R v)≤D)→
  22     (∀ v z,v≠z→((Finset.univ.filter (fun e=>v∈R e ∧ z∈R e)).card:ℝ)≤K*D/n)→
  23     ∃ M:Finset E,(∀ e∈M,∀ f∈M,e≠f→Disjoint (R e) (R f)) ∧
  24       (1-γ)*(1-ε)*Fintype.card V≤(k:ℝ)*M.card := by
  25   obtain ⟨N,hN⟩:=cycle_hypergraph_matching_cardinality k B hk K a γ hK ha hγ
  26   refine ⟨max N 1,?_⟩
  27   intro n D hn hnD hDB V E _ _ _ R hnV hV hsize hdegrees hcodeg
  28   have hnpos:0<n:=by omega
  29   have hDpos:(0:ℝ)<D:=by exact_mod_cast (lt_of_lt_of_le hnpos hnD)
  30   have hkpos:(0:ℝ)<k:=by exact_mod_cast (show 0<k by omega)
  31   have hmarginpos:0<1-ε:=(mul_pos ha hkpos).trans_le hmargin
  32   have hpositive:∀ v,0<Fintype.card (HypergraphEdgesAt R v):=by
  33     intro v
  34     have hh: (0:ℝ)<Fintype.card (HypergraphEdgesAt R v):=
  35       (mul_pos hmarginpos hDpos).trans_le (hdegrees v).1
  36     exact_mod_cast hh
  37   have hsum: (1-ε)*(D:ℝ)*Fintype.card V≤(k:ℝ)*Fintype.card E:=by
  38     have hh:=(resource_set_sum_bounds R Finset.univ ((1-ε)*(D:ℝ)) D
  39       (fun v=>⟨(hdegrees v).1,by exact_mod_cast (hdegrees v).2⟩)).1
  40     simpa only [Finset.card_univ,Finset.inter_univ,hsize,Finset.sum_const,
  41       nsmul_eq_mul,Nat.cast_ofNat,mul_comm,mul_left_comm,mul_assoc] using hh
  42   have hnV':(n:ℝ)≤Fintype.card V:=by exact_mod_cast hnV
  43   have hE: a*D*n≤(Fintype.card E:ℝ):=by
  44     have h1:=mul_le_mul_of_nonneg_right hmargin (mul_nonneg hDpos.le (Nat.cast_nonneg (Fintype.card V)))
  45     have h2:=mul_le_mul_of_nonneg_left hnV' (mul_nonneg ha.le hDpos.le)
  46     have hh:=mul_le_mul_of_nonneg_right h2 hkpos.le
  47     nlinarith
  48   obtain ⟨M,hM,hcard⟩:=hN n D ((le_max_left N 1).trans hn) hnD hDB V E R hV hsize hpositive
  49     (fun v=>(hdegrees v).2) hcodeg hE
  50   refine ⟨M,hM,?_⟩
  51   have hscale:=mul_le_mul_of_nonneg_left hsum (sub_nonneg.mpr hγ1)
  52   have hmatch:=mul_le_mul_of_nonneg_left hcard hkpos.le
  53   have hmatch':(1-γ)*(k:ℝ)*Fintype.card E≤(k:ℝ)*(M.card:ℝ)*D:=by
  54     have h:=mul_le_mul_of_nonneg_right hmatch hDpos.le
  55     field_simp at h
  56     nlinarith
  57   have hfinal: ((1-γ)*(1-ε)*Fintype.card V)*D≤((k:ℝ)*M.card)*D:=by nlinarith
  58   exact le_of_mul_le_mul_right hfinal hDpos
  59 end Oberwolfach
