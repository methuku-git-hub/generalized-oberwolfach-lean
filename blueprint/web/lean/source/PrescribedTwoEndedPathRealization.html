  13 theorem prescribed_two_ended_path_realization
  14     (as:List X) (ha:as.Nodup) (hne:1≤as.length) (A:Finset X) (f:X→V)
  15     (hinj:Set.InjOn f A) (hsource:∀ a∈as,a∉A)
  16     (a b:X) (haA:a∈A) (hbA:b∈A) (hab:a≠b)
  17     (S L:Finset V) (β:ℝ) (D:V→V→Prop) (hext:IsExtendingList D L β)
  18     (F:SimpleGraph V) (M:ℝ) (hM:0≤M) (hdeg:∀ v,(F.degree v:ℝ)≤M)
  19     (hbudget:((A.image f∪S).card:ℝ)+as.length+2+2*M<β*Fintype.card V):
  20     ∃ g:X→V,Set.InjOn g (↑(A∪as.toFinset):Set X) ∧ (∀ u∈A,g u=f u) ∧
  21     (∀ u∈as,g u∈L ∧ g u∉S) ∧
  22     DirectedPathFrom D F true (g a) ((as.map g)++[g b]) := by
  23   have hfa:f a≠f b:=by intro h;exact hab (hinj haA hbA h)
  24   let B:=A.image f∪S
  25   have hbudget':(B.card:ℝ)+((as.length-1)+1:ℕ)+2+2*M<β*Fintype.card V:=by
  26     have hh:as.length-1+1=as.length:=Nat.sub_add_cancel hne
  27     simpa [hh,B] using hbudget
  28   obtain ⟨ys,hyslen,hysnodup,hys,hpath⟩:=extending_list_embed_internal_path D L β hext F M hM hdeg (as.length-1) B (f a) (f b) hfa hbudget'
  29   have hlen:as.length=ys.length:=by rw [hyslen,Nat.sub_add_cancel hne]
  30   have htarget:∀ z∈ys,z∉A.image f:=by
  31     intro z hz hmem
  32     exact (hys z hz).2.1 (Finset.mem_union_left S hmem)
  33   let g:=updatePairedImages f (as.zip ys)
  34   have hb:=prescribed_embedding_zip_update as ys hlen ha hysnodup A f hinj hsource htarget
  35   have hmap:=prescribed_embedding_update_map_list as ys hlen ha hysnodup A f hinj hsource htarget
  36   refine ⟨g,hb.1,hb.2.1,?_,?_⟩
  37   · intro u hu
  38     obtain ⟨i,hi⟩:=List.mem_iff_get.mp hu
  39     have hp:=prescribed_embedding_zip_update_at_position as ys hlen ha hysnodup A f hinj hsource htarget i
  40     have hz:ys.get ⟨i.val,by omega⟩∈ys:=List.get_mem _ _
  41     have hgood:=hys _ hz
  42     have hnotS:(ys.get ⟨i.val,by omega⟩)∉S:=fun h=>hgood.2.1 (Finset.mem_union_right _ h)
  43     rw [←hi]
  44     change updatePairedImages f (as.zip ys) (as.get i)∈L ∧
  45       updatePairedImages f (as.zip ys) (as.get i)∉S
  46     rw [hp]
  47     exact ⟨hgood.1,hnotS⟩
  48   · have hga:g a=f a:=hb.2.1 a haA
  49     have hgb:g b=f b:=hb.2.1 b hbA
  50     rw [hga,hgb,hmap]
  51     exact prescribed_path_through_to_from D F (f b) ys (f a) hpath
  52 end Oberwolfach
