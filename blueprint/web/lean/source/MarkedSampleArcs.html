  21 theorem markedPhysicalRim_eq_one_iff (P : SimpleGraph V) (π : Equiv.Perm V) (b : ℕ)
  22     (σ : Sym2 V→Option (Bool×Bool)) (a : Bool) (x y : V) :
  23     markedPhysicalRim P π b σ a x y=1 ↔
  24       ¬PermutationClose π b x y ∧ P.Adj x (if a then π y else y) ∧
  25         σ s(x,if a then π y else y)=some (a,orientationBit x (if a then π y else y)) := by
  26   unfold markedPhysicalRim permutationSeparatedMatrix markedPhysicalOrientation
  27   split_ifs <;> simp_all

  29 theorem markedPhysicalRim_physical_unique (P : SimpleGraph V) (π : Equiv.Perm V) (b : ℕ)
  30     (σ : Sym2 V→Option (Bool×Bool)) (a a' : Bool) (x y x' y' : V)
  31     (h : markedPhysicalRim P π b σ a x y=1) (h' : markedPhysicalRim P π b σ a' x' y'=1)
  32     (he : s(x,if a then π y else y)=s(x',if a' then π y' else y')) :
  33     a=a' ∧ x=x' ∧ y=y' := by
  34   have hh := (markedPhysicalRim_eq_one_iff P π b σ a x y).mp h
  35   have hh' := (markedPhysicalRim_eq_one_iff P π b σ a' x' y').mp h'
  36   have hs := hh.2.2
  37   rw [he,hh'.2.2] at hs
  38   have hstates := Prod.mk.inj (Option.some.inj hs)
  39   have ha : a=a' := hstates.1.symm
  40   subst a'
  41   rcases Sym2.eq_iff.mp he with ⟨hxx,hyy⟩|⟨hxy,hyx⟩
  42   · refine ⟨rfl,hxx,?_⟩
  43     cases a
  44     · exact hyy
  45     · exact π.injective hyy
  46   · have hbits : orientationBit x (if a then π y else y)=
  47         orientationBit (if a then π y else y) x := by
  48       simpa only [← hxy,← hyx] using hstates.2.symm
  49     exact ((orientationBit_ne_swap x _ hh.2.1.ne) hbits).elim

  55 theorem markedSampleArc_physical_injective (P : SimpleGraph V) (π : Equiv.Perm V) (b : ℕ)
  56     (σ : Sym2 V→Option (Bool×Bool)) :
  57     Function.Injective (markedPhysicalEdge
  58       (fun e : MarkedSampleArc P π b σ false=>e.val.1)
  59       (fun e : MarkedSampleArc P π b σ false=>e.val.2)
  60       (fun e : MarkedSampleArc P π b σ true=>e.val.1)
  61       (fun e : MarkedSampleArc P π b σ true=>e.val.2) π) := by
  62   have hpair (x y x' y' : V) (h : ({x,y}:Finset V)={x',y'}) : s(x,y)=s(x',y') := by
  63     apply Sym2.eq_iff.mpr
  64     apply Set.pair_eq_pair_iff.mp
  65     simpa only [Finset.coe_pair] using congrArg (fun S : Finset V=>(S:Set V)) h
  66   intro e f he
  67   cases e with
  68   | inl e=>cases f with
  69     | inl f=>
  70       have hh := markedPhysicalRim_physical_unique P π b σ false false e.val.1 e.val.2 f.val.1 f.val.2
  71         e.property f.property (hpair _ _ _ _ he)
  72       exact congrArg Sum.inl (Subtype.ext (Prod.ext hh.2.1 hh.2.2))
  73     | inr f=>
  74       have hh := markedPhysicalRim_physical_unique P π b σ false true e.val.1 e.val.2 f.val.1 f.val.2
  75         e.property f.property (hpair _ _ _ _ he)
  76       cases hh.1
  77   | inr e=>cases f with
  78     | inl f=>
  79       have hh := markedPhysicalRim_physical_unique P π b σ true false e.val.1 e.val.2 f.val.1 f.val.2
  80         e.property f.property (hpair _ _ _ _ he)
  81       cases hh.1
  82     | inr f=>
  83       have hh := markedPhysicalRim_physical_unique P π b σ true true e.val.1 e.val.2 f.val.1 f.val.2
  84         e.property f.property (hpair _ _ _ _ he)
  85       exact congrArg Sum.inr (Subtype.ext (Prod.ext hh.2.1 hh.2.2))
