  23     (∀ z h,ordinarySpokeLoad (ordinaryAverageTransfer hr Q F H v u u') z h=

  32       if ordinaryWheelHub W=h ∧ z∈(ordinaryWheelRim W).support then 1 else 0
  33     let K : ℝ := (if z=v then (1:ℝ) else 0)*((if h=u then 1 else 0)-(if h=u' then 1 else 0))
