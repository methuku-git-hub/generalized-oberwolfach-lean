  15 theorem embedded_cycle_from_path_with_support (G : SimpleGraph V)
  16     (r : ℕ) (hr : 3≤r) (x y : V) (p : G.Walk x y)
  17     (hp : p.IsPath) (hlen : p.length=r-1) (hxy : G.Adj x y) :
  18     ∃ C : SimpleGraph V,IsEmbeddedCycle C r ∧ C≤G ∧ C.Adj x y ∧
  19       (∀ v,v∈C.support ↔ v∈p.support) ∧
  20       (∀ i : ℕ,i<p.length → C.Adj (p.getVert i) (p.getVert (i+1))) := by
  21   let f : Fin r↪V:=⟨fun i=>p.getVert i.val,by
  22     intro i j he
  23     apply Fin.ext
  24     exact hp.getVert_injOn (by change i.val≤p.length;rw [hlen];omega)
  25       (by change j.val≤p.length;rw [hlen];omega) he⟩
  26   let C:=(SimpleGraph.cycleGraph r).map f
  27   have hC : IsEmbeddedCycle C r:=⟨hr,f,rfl⟩
  28   have hadj (i j : Fin r) (h : (SimpleGraph.cycleGraph r).Adj i j) : G.Adj (f i) (f j):=by
  29     change G.Adj (p.getVert i.val) (p.getVert j.val)
  30     rcases cycleGraph_adj_natural_cases r hr i j h with hij|hji|⟨hi,hj⟩|⟨hj,hi⟩
  31     · rw [←hij];exact p.adj_getVert_succ (by rw [hlen];have hj:=j.isLt;omega)
  32     · rw [←hji];exact (p.adj_getVert_succ (by rw [hlen];have hi:=i.isLt;omega)).symm
  33     · rw [hi,hj,←hlen,p.getVert_zero,p.getVert_length];exact hxy
  34     · rw [hj,hi,←hlen,p.getVert_zero,p.getVert_length];exact hxy.symm
  35   have hCG : C≤G:=by
  36     intro v w h
  37     obtain ⟨i,j,hij,rfl,rfl⟩:=(SimpleGraph.map_adj f _ _ _).mp h
  38     exact hadj i j hij
  39   have hclose : C.Adj x y:=by
  40     apply (SimpleGraph.map_adj f _ _ _).mpr
  41     refine ⟨⟨0,by omega⟩,⟨r-1,by omega⟩,?_,p.getVert_zero,?_⟩
  42     · apply (cycleGraph_modular_adj (by omega) _ _).mpr
  43       right
  44       change (r-1+1)%r=0
  45       rw [show r-1+1=r by omega,Nat.mod_self]
  46     · change p.getVert (r-1)=y
  47       rw [←hlen,p.getVert_length]
  48   have hsupport : ∀ v,v∈C.support ↔ v∈p.support:=by
  49     intro v
  50     rw [cycle_map_support hr f]
  51     constructor
  52     · rintro ⟨i,rfl⟩;exact p.getVert_mem_support _
  53     · intro hv
  54       obtain ⟨i,hiv,hi⟩:=SimpleGraph.Walk.mem_support_iff_exists_getVert.mp hv
  55       exact ⟨⟨i,by rw [hlen] at hi;omega⟩,hiv⟩
  56   refine ⟨C,hC,hCG,hclose,hsupport,?_⟩
  57   intro i hi
  58   have hir : i+1<r:=by rw [hlen] at hi;omega
  59   apply (SimpleGraph.map_adj f _ _ _).mpr
  60   refine ⟨⟨i,by omega⟩,⟨i+1,hir⟩,?_,rfl,rfl⟩
  61   apply (cycleGraph_modular_adj (by omega) _ _).mpr
  62   left
  63   change (i+1)%r=i+1
  64   exact Nat.mod_eq_of_lt hir
