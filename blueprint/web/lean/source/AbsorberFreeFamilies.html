  25 theorem labelledAbsorberFamily_ext (R : C → Finset V) (S : T → VertexAbsorberTemplate c)
  26     (X : Finset V) (D E : LabelledAbsorberFamily R S X) (ht : D.1=E.1)
  27     (hf : ∀ j,D.2.1.val j=E.2.1.val j) (hg : ∀ o,(D.2.2 o).val=(E.2.2 o).val) : D=E := by
  28   rcases D with ⟨t,D⟩
  29   rcases E with ⟨u,E⟩
  30   change t=u at ht
  31   subst u
  32   exact congrArg (Sigma.mk t) (labelledCycleRecord_ext R (S t).val X D E hf hg)

  34 theorem free_absorber_family_card (R : C → Finset V) (S : T → VertexAbsorberTemplate c) (J : ℕ)
  35     (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J) :
  36     Fintype.card (FreeAbsorberFamily R S) ≤ Fintype.card T*Fintype.card V^(c*c)*J^c := by
  37   rw [Fintype.card_sigma]
  38   calc
  39     _ ≤ ∑ _t : T,Fintype.card V^(c*c)*J^c :=
  40       Finset.sum_le_sum (fun t _ => free_absorber_records_card R (S t) J hJ)
  41     _ = _ := by simp only [Finset.sum_const,Finset.card_univ,smul_eq_mul];ring

  43 theorem family_forget_absorber_roots_fibre_card (R : C → Finset V) (S : T → VertexAbsorberTemplate c)
  44     (X : Finset V) (J : ℕ)
  45     (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J) (F : FreeAbsorberFamily R S) :
  46     Fintype.card {D : LabelledAbsorberFamily R S X // familyForgetAbsorberRoots R S X D=F} ≤
  47       Fintype.card T*X.card^c*J^(c+1) := by
  48   obtain ⟨code,hcode⟩ := finite_fibre_codes R J hJ
  49   let A := {D : LabelledAbsorberFamily R S X // familyForgetAbsorberRoots R S X D=F}
  50   let enc (D : A) : T × (Fin c → X) × (Fin (c+1) → Fin J) :=
  51     (D.val.1,fun i => ⟨D.val.2.1.val (Sum.inr i),rootedLabelMap_root_mem X D.val.2.1 i⟩,
  52      fun i => code (D.val.2.2 (Sum.inr i)).val)
  53   have hinj : Function.Injective enc := by
  54     intro D E he
  55     have ht : D.val.1=E.val.1 := congrArg Prod.fst he
  56     have hf (j : Fin (c*c) ⊕ Fin c) : D.val.2.1.val j=E.val.2.1.val j := by
  57       rcases j with j | j
  58       · have hd := congrArg (fun F : FreeAbsorberFamily R S => F.2.1 j) D.property
  59         have he' := congrArg (fun F : FreeAbsorberFamily R S => F.2.1 j) E.property
  60         exact hd.trans he'.symm
  61       · exact congrArg (fun z : X => z.val) (congrFun (congrArg (fun z => z.2.1) he) j)
  62     apply Subtype.ext
  63     apply labelledAbsorberFamily_ext R S X D.val E.val ht hf
  64     intro o
  65     rcases o with i | i
  66     · have hd := congrArg (fun F : FreeAbsorberFamily R S => (F.2.2 i).val) D.property
  67       have he' := congrArg (fun F : FreeAbsorberFamily R S => (F.2.2 i).val) E.property
  68       exact hd.trans he'.symm
  69     · apply hcode
  70       · rw [(D.val.2.2 (Sum.inr i)).property,(E.val.2.2 (Sum.inr i)).property]
  71         have hs := congrArg (fun t => (S t).val (Sum.inr i)) ht
  72         exact (congrArg (Finset.image D.val.2.1.val) hs).trans
  73           (congrArg (Finset.image · ((S E.val.1).val (Sum.inr i))) (funext hf))
  74       · exact congrFun (congrArg (fun z => z.2.2) he) i
  75   have hh := Fintype.card_le_of_injective enc hinj
  76   simpa only [Fintype.card_prod,Fintype.card_fun,Fintype.card_fin,Fintype.card_coe,mul_assoc] using hh
