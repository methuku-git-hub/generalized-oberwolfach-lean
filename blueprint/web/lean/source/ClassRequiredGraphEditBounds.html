  10 theorem graph_le_sup_degree_and_edges
  11     {V:Type} [Fintype V] [DecidableEq V] (G A B:SimpleGraph V) (h:G≤A⊔B):
  12     (∀v,G.degree v≤A.degree v+B.degree v) ∧
  13       G.edgeFinset.card≤A.edgeFinset.card+B.edgeFinset.card:=by
  14   constructor
  15   · intro v
  16     have hn:G.neighborFinset v⊆(A⊔B).neighborFinset v:=by
  17       intro w hw
  18       exact (SimpleGraph.mem_neighborFinset _ _ _).mpr (h ((SimpleGraph.mem_neighborFinset _ _ _).mp hw))
  19     exact (Finset.card_le_card hn).trans (graph_sup_degree_le A B v)
  20   · have he:(A⊔B).edgeFinset=A.edgeFinset∪B.edgeFinset:=by
  21       ext e
  22       simp only [Finset.mem_union,SimpleGraph.mem_edgeFinset,SimpleGraph.edgeSet_sup,Set.mem_union]
  23     have hm:=Finset.card_le_card (SimpleGraph.edgeFinset_mono h)
  24     rw [he] at hm
  25     exact hm.trans (Finset.card_union_le _ _)

  28     {V:Type} [Fintype V] [LinearOrder V]
  29     (P S₀ S Used Sel:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ) (I:Finset V) (D:V→V→Prop)
  30     (hSP:S≤P):

  33     let E:=spanningRestriction (S\S₀) I

  35       K.edgeFinset.card≤K₀.edgeFinset.card+E.edgeFinset.card:=by
