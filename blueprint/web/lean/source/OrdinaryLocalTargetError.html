  15 theorem ordinaryLocalTarget_error (P F : SimpleGraph V) (hr : 2≤r) (hn : 0<Fintype.card V)
  16     (weight : OrdinaryCycle P r→ℝ) (hweight : ∀ C,0≤weight C)
  17     (B L D δ : ℝ) (hB : 0≤B) (hL : 0≤L) (hδ : 0≤δ)
  18     (hweightB : ∀ C,weight C≤B/(Fintype.card V:ℝ)^(r-2))
  19     (hF : ∀ v,(F.degree v:ℝ)≤L)
  20     (hedge : ∀ x y,(∑ C,if C.val.val.Adj x y then weight C else 0)=if P.Adj x y then 1 else 0)
  21     (hD : (Fintype.card V:ℝ)/2≤D) (v : V) (hdeg : |(P.degree v:ℝ)-D|≤δ*Fintype.card V) :
  22     |ordinaryLocalTarget P F weight D v-1|≤2*δ+12*(r:ℝ)^4*B*L/Fintype.card V := by
  23   let n : ℝ := Fintype.card V
  24   have hnR : 0<n := Nat.cast_pos.mpr hn
  25   have hDpos : 0<D := (half_pos hnR).trans_le hD
  26   have hload := ordinary_admissible_cycle_vertex_load P F r hr hn weight hweight B hB hweightB L hF hedge v
  27   have hid : ordinaryLocalTarget P F weight D v-(P.degree v:ℝ)/D=
  28       (2/D)*((∑ C,if v∈C.val.val.support ∧
  29         (∀ x∈C.val.val.support,∀ y∈C.val.val.support,¬F.Adj x y) then weight C else 0)-(P.degree v:ℝ)/2) := by
  30     unfold ordinaryLocalTarget;ring
  31   have hmid : |ordinaryLocalTarget P F weight D v-(P.degree v:ℝ)/D|≤6*(r:ℝ)^4*B*L/D := by
  32     rw [hid,abs_mul,abs_of_nonneg (show 0≤2/D by positivity)]
  33     have hh := mul_le_mul_of_nonneg_left hload (show 0≤2/D by positivity)
  34     convert hh using 1 <;> ring
  35   have hlast : |(P.degree v:ℝ)/D-1|≤δ*n/D := by
  36     have he : (P.degree v:ℝ)/D-1=((P.degree v:ℝ)-D)/D := by field_simp
  37     rw [he,abs_div,abs_of_pos hDpos]
  38     exact div_le_div_of_nonneg_right hdeg hDpos.le
  39   have hlo : 6*(r:ℝ)^4*B*L/D≤12*(r:ℝ)^4*B*L/n := by
  40     have hh := div_le_div_of_nonneg_left (show 0≤6*(r:ℝ)^4*B*L by positivity) (half_pos hnR) hD
  41     convert hh using 1 <;> field_simp <;> ring
  42   have hhi : δ*n/D≤2*δ := by
  43     have hh := div_le_div_of_nonneg_left (show 0≤δ*n by positivity) (half_pos hnR) hD
  44     have he : δ*n/(n/2)=2*δ := by field_simp
  45     exact hh.trans_eq he
  46   have ht := abs_sub_le (ordinaryLocalTarget P F weight D v) ((P.degree v:ℝ)/D) 1
  47   linarith only [ht,hmid,hlast,hlo,hhi]
