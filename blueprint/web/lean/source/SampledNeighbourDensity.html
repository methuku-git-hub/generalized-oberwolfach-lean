   9 theorem sampled_neighbour_density (n p θ α t size deg count : ℝ)
  10     (hn : 0≤n) (hp : 0≤p) (hα : 0<α) (hθ : 0≤θ+α/2) (hθ1 : θ+α/2≤1)
  11     (ht : 0≤t) (htsize : t≤p*n/2) (htmargin : t≤α*p*n/4)
  12     (hsize : |size-p*n|≤t) (hdeg : (θ+α)*n≤deg)
  13     (hcount : |count-p*deg|≤t) :
  14     p*n/2≤size ∧ (θ+α/2)*size≤count := by
  15   have hlo := (abs_le.mp hsize).1
  16   have hhi := (abs_le.mp hsize).2
  17   have hclo := (abs_le.mp hcount).1
  18   have hd := mul_le_mul_of_nonneg_left hdeg hp
  19   have hs := mul_le_mul_of_nonneg_left (show size≤p*n+t by linarith only [hhi]) hθ
  20   have hterr := mul_le_mul_of_nonneg_right hθ1 ht
  21   constructor
  22   · linarith only [hlo,htsize]
  23   · nlinarith only [hs,hclo,hd,hterr,htmargin]
  24 
  25 end Oberwolfach
