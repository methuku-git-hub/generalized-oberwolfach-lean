  12 theorem finite_fiber_membership_sum {E V : Type*} [Fintype E] [DecidableEq V]
  13     (f : E → V) (a : E → ℝ) (S : Finset V) :
  14     (∑ e,if f e∈S then a e else 0)=∑ x∈S,∑ e,if f e=x then a e else 0 := by

  16   apply Finset.sum_congr rfl
  17   intro e _
  18   by_cases he : f e∈S <;> simp [he,Finset.sum_ite_eq']

  20 theorem occurrence_root_incident_cost {E V : Type*} [Fintype E] [Fintype V] [DecidableEq V]
  21     (f g : E → V) (D : E → Prop) (d : ℝ)
  22     (hout : ∀ x,(∑ e,if D e ∧ f e=x then (1:ℝ) else 0)≤d)
  23     (hin : ∀ y,(∑ e,if D e ∧ g e=y then (1:ℝ) else 0)≤d)
  24     (S T : Finset V) (M N : ℝ) (hM : 0≤M) (hN : 0≤N) :
  25     (∑ e,if D e then (if f e∈S ∨ g e∈T then M else N) else 0)≤
  26       d*((S.card+T.card:ℝ)*M+(Fintype.card V:ℝ)*N) := by
  27   let a (e : E) : ℝ := if D e then 1 else 0
  28   have hf (x : V) : (∑ e,if f e=x then a e else 0)≤d := by
  29     convert hout x using 1
  30     apply Finset.sum_congr rfl
  31     intro e _
  32     dsimp only [a]
  33     split_ifs <;> simp_all
  34   have hg (x : V) : (∑ e,if g e=x then a e else 0)≤d := by
  35     convert hin x using 1
  36     apply Finset.sum_congr rfl
  37     intro e _
  38     dsimp only [a]
  39     split_ifs <;> simp_all
  40   have hS : (∑ e,if f e∈S then a e else 0)≤S.card*d := by
  41     rw [finite_fiber_membership_sum]
  42     calc
  43       _ ≤ ∑ _x∈S,d := Finset.sum_le_sum (fun x _=>hf x)
  44       _ = _ := by simp only [Finset.sum_const,nsmul_eq_mul]
  45   have hT : (∑ e,if g e∈T then a e else 0)≤T.card*d := by
  46     rw [finite_fiber_membership_sum]
  47     calc
  48       _ ≤ ∑ _x∈T,d := Finset.sum_le_sum (fun x _=>hg x)
  49       _ = _ := by simp only [Finset.sum_const,nsmul_eq_mul]
  50   have htotal : (∑ e,a e)≤Fintype.card V*d := by
  51     have hh := finite_fiber_membership_sum f a Finset.univ
  52     simp only [Finset.mem_univ,ite_true] at hh
  53     rw [hh]
  54     calc
  55       _ ≤ ∑ _x : V,d := Finset.sum_le_sum (fun x _=>hf x)
  56       _ = _ := by simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul]
  57   calc
  58     _ ≤ ∑ e,((if f e∈S then a e else 0)*M+(if g e∈T then a e else 0)*M+a e*N) := by
  59       apply Finset.sum_le_sum
  60       intro e _
  61       dsimp only [a]
  62       split_ifs <;> simp_all <;> linarith
  63     _ = (∑ e,if f e∈S then a e else 0)*M+(∑ e,if g e∈T then a e else 0)*M+(∑ e,a e)*N := by
  64       simp only [Finset.sum_add_distrib,Finset.sum_mul]
  65     _ ≤ _ := by
  66       have h1 := mul_le_mul_of_nonneg_right hS hM
  67       have h2 := mul_le_mul_of_nonneg_right hT hM
  68       have h3 := mul_le_mul_of_nonneg_right htotal hN
  69       nlinarith only [h1,h2,h3]
