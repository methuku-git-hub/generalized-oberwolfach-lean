  25 theorem active_typed_resource_physical_injective
  26     (P:SimpleGraph V) (π:Equiv.Perm V) (b:ℕ)
  27     (σ:Sym2 V→Option (Bool×Bool)) (H:Bool→V→U→Prop)
  28     (hdisjoint:∀v u,¬(H false v u ∧ H true v u)):
  29     Function.Injective (fun x:{z // activeTypedResource P π b σ H z}=>
  30       typedResourcePhysical π x.val) := by
  31   intro x y he
  32   apply Subtype.ext
  33   obtain ⟨x,hx⟩:=x
  34   obtain ⟨y,hy⟩:=y
  35   cases x with
  36   | inl x =>
  37     cases y with
  38     | inl y =>
  39       rcases x with ⟨a,x₁,x₂⟩
  40       rcases y with ⟨a',y₁,y₂⟩
  41       have he' : s(x₁,if a then π x₂ else x₂)=s(y₁,if a' then π y₂ else y₂) := by
  42         simpa [typedResourcePhysical] using he
  43       have hh:=markedPhysicalRim_physical_unique P π b σ a a' x₁ x₂ y₁ y₂ hx hy he'
  44       rcases hh with ⟨rfl,rfl,rfl⟩
  45       rfl
  46     | inr y => simp [typedResourcePhysical] at he
  47   | inr x =>
  48     cases y with
  49     | inl y => simp [typedResourcePhysical] at he
  50     | inr y =>
  51       rcases x with ⟨a,x₁,x₂⟩
  52       rcases y with ⟨a',y₁,y₂⟩
  53       have he' : (x₁,x₂)=(y₁,y₂) := by simpa [typedResourcePhysical] using he
  54       cases he'
  55       change H a x₁ x₂ at hx
  56       change H a' x₁ x₂ at hy
  57       have ha:a=a' := by
  58         cases a <;> cases a'
  59         · rfl
  60         · exact False.elim (hdisjoint x₁ x₂ ⟨hx,hy⟩)
  61         · exact False.elim (hdisjoint x₁ x₂ ⟨hy,hx⟩)
  62         · rfl
  63       cases ha
  64       rfl
