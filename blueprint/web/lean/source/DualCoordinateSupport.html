  12 theorem paired_coordinate_support_bounds
  13     (S:R→Finset E) (f g:T↪R) (B:ℕ)
  14     (hread:∀ r,(S r).card≤B)
  15     (haffect:∀ e,(Finset.univ.filter (fun r=>e∈S r)).card≤B):
  16     (∀ t,((S (f t))∪(S (g t))).card≤2*B) ∧
  17     (∀ e,(Finset.univ.filter (fun t=>e∈(S (f t))∪(S (g t)))).card≤2*B):=by
  18   constructor
  19   · intro t
  20     exact (Finset.card_union_le _ _).trans (by have h1:=hread (f t);have h2:=hread (g t);omega)
  21   · intro e
  22     let A:=Finset.univ.filter (fun r=>e∈S r)
  23     let T₁:=Finset.univ.filter (fun t=>e∈S (f t))
  24     let T₂:=Finset.univ.filter (fun t=>e∈S (g t))
  25     have h1:T₁.card≤B:=by
  26       have hsub:T₁.image f⊆A:=by
  27         intro r hr
  28         obtain ⟨t,ht,rfl⟩:=Finset.mem_image.mp hr
  29         exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,(Finset.mem_filter.mp ht).2⟩
  30       calc
  31         T₁.card=(T₁.image f).card:=(Finset.card_image_of_injective T₁ f.injective).symm
  32         _≤A.card:=Finset.card_le_card hsub
  33         _≤B:=haffect e
  34     have h2:T₂.card≤B:=by
  35       have hsub:T₂.image g⊆A:=by
  36         intro r hr
  37         obtain ⟨t,ht,rfl⟩:=Finset.mem_image.mp hr
  38         exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,(Finset.mem_filter.mp ht).2⟩
  39       calc
  40         T₂.card=(T₂.image g).card:=(Finset.card_image_of_injective T₂ g.injective).symm
  41         _≤A.card:=Finset.card_le_card hsub
  42         _≤B:=haffect e
  43     have heq:Finset.univ.filter (fun t=>e∈S (f t)∪S (g t))=T₁∪T₂:=by
  44       ext t
  45       simp [T₁,T₂]
  46     rw [heq]
