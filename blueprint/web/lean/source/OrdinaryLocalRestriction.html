  20 def localOrdinaryRestrictionParent {Q Q' F : SimpleGraph V} {T T' : Finset V}
  21     (hQ : Q'≤Q) (hT : T'⊆T) : LocalOrdinaryCycle Q' F T' r ↪ LocalOrdinaryCycle Q F T r where
  22   toFun C := ⟨⟨C.val.val,C.val.property.trans hQ⟩,
  23     fun v hv=>hT (C.property.1 v hv),C.property.2⟩
  24   inj' := by intro C D he;apply Subtype.ext;apply Subtype.ext;exact congrArg (fun C=>C.val.val) he

  26 def localOrdinaryRestrictionEquiv (Q Q' F : SimpleGraph V) (T T' : Finset V)
  27     (hQ : Q'≤Q) (hT : T'⊆T) : LocalOrdinaryCycle Q' F T' r ≃
  28       {C : LocalOrdinaryCycle Q F T r // localOrdinaryRestrictionGood Q' T' C} where
  29   toFun C := ⟨localOrdinaryRestrictionParent hQ hT C,C.val.property,
  30     fun v hv=>C.property.1 v (Set.mem_toFinset.mp hv)⟩
  31   invFun C := ⟨⟨C.val.val.val,C.property.1⟩,
  32     fun v hv=>C.property.2 (Set.mem_toFinset.mpr hv),C.val.property.2⟩
  33   left_inv _ := rfl
  34   right_inv _ := rfl
