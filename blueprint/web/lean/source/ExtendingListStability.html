   9 variable {V:Type} [Fintype V] [DecidableEq V]

  11 def directedOutNeighbours (D:V→V→Prop) (x:V):Finset V:=Finset.univ.filter (D x)
  12 def directedInNeighbours (D:V→V→Prop) (x:V):Finset V:=Finset.univ.filter (fun y=>D y x)

  14 def IsExtendingList (D:V→V→Prop) (L:Finset V) (lam:ℝ):Prop:=
  15   0≤lam ∧ lam*Fintype.card V≤(L.card:ℝ) ∧
  16   (∀ x,lam*Fintype.card V≤((directedOutNeighbours D x∩L).card:ℝ) ∧
  17         lam*Fintype.card V≤((directedInNeighbours D x∩L).card:ℝ)) ∧
  18   ∀ x y,x≠y→lam*Fintype.card V≤
  19     ((directedOutNeighbours D x∩directedInNeighbours D y∩L).card:ℝ)

  21 private theorem card_triple_intersection_loss (A B C A' B' C':Finset V):
  22     (A∩B∩C).card≤(A'∩B'∩C').card+(A\A').card+(B\B').card+(C\C').card := by
  23   have hsub:A∩B∩C⊆(A'∩B'∩C')∪(A\A')∪(B\B')∪(C\C'):=by
  24     intro x hx
  25     simp only [Finset.mem_inter] at hx
  26     by_cases ha:x∈A' <;> by_cases hb:x∈B' <;> by_cases hc:x∈C' <;>
  27       simp [ha,hb,hc,hx.1.1,hx.1.2,hx.2]
  28   have hh:=Finset.card_le_card hsub
  29   have h₁:=Finset.card_union_le ((A'∩B'∩C')∪(A\A')∪(B\B')) (C\C')
  30   have h₂:=Finset.card_union_le ((A'∩B'∩C')∪(A\A')) (B\B')
  31   have h₃:=Finset.card_union_le (A'∩B'∩C') (A\A')
  32   omega

  34 theorem extending_list_after_neighbourhood_changes
  35     (D D':V→V→Prop) (L L':Finset V) (β lam r s:ℝ)
  36     (hlam:0≤lam) (hr:0≤r) (hs:0≤s)
  37     (hL:((L\L').card:ℝ)≤s)
  38     (hout:∀ x,((directedOutNeighbours D x\directedOutNeighbours D' x).card:ℝ)≤r)
  39     (hin:∀ x,((directedInNeighbours D x\directedInNeighbours D' x).card:ℝ)≤r)
  40     (hbudget:lam*Fintype.card V+2*r+s≤β*Fintype.card V)
  41     (hext:IsExtendingList D L β):IsExtendingList D' L' lam := by
  42   refine ⟨hlam,?_,?_,?_⟩
  43   · have hh:=card_triple_intersection_loss (Finset.univ:Finset V) Finset.univ L Finset.univ Finset.univ L'
  44     simp only [Finset.univ_inter,Finset.sdiff_self,Finset.card_empty,zero_add] at hh
  45     have hh':(L.card:ℝ)≤L'.card+(L\L').card:=by exact_mod_cast hh
  46     nlinarith [hext.2.1]
  47   · intro x
  48     have ho:=card_triple_intersection_loss (directedOutNeighbours D x) Finset.univ L
  49       (directedOutNeighbours D' x) Finset.univ L'
  50     have hi:=card_triple_intersection_loss (directedInNeighbours D x) Finset.univ L
  51       (directedInNeighbours D' x) Finset.univ L'
  52     simp only [Finset.inter_univ,Finset.sdiff_self,Finset.card_empty,add_zero] at ho hi
  53     have ho':((directedOutNeighbours D x∩L).card:ℝ)≤
  54       (directedOutNeighbours D' x∩L').card+(directedOutNeighbours D x\directedOutNeighbours D' x).card+(L\L').card:=by exact_mod_cast ho
  55     have hi':((directedInNeighbours D x∩L).card:ℝ)≤
  56       (directedInNeighbours D' x∩L').card+(directedInNeighbours D x\directedInNeighbours D' x).card+(L\L').card:=by exact_mod_cast hi
  57     exact ⟨by nlinarith [(hext.2.2.1 x).1,hout x],by nlinarith [(hext.2.2.1 x).2,hin x]⟩
  58   · intro x y hxy
  59     have hh:=card_triple_intersection_loss (directedOutNeighbours D x) (directedInNeighbours D y) L
  60       (directedOutNeighbours D' x) (directedInNeighbours D' y) L'
  61     have hh':((directedOutNeighbours D x∩directedInNeighbours D y∩L).card:ℝ)≤
  62       (directedOutNeighbours D' x∩directedInNeighbours D' y∩L').card+
  63       (directedOutNeighbours D x\directedOutNeighbours D' x).card+
  64       (directedInNeighbours D y\directedInNeighbours D' y).card+(L\L').card:=by exact_mod_cast hh
  65     nlinarith [hext.2.2.2 x y hxy,hout x,hin y]
