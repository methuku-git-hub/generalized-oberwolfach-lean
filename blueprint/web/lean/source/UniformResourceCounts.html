  16 theorem uniform_resources_containing_count (R : C → Finset V) (c J : ℕ)
  17     (hR : ∀ C₀,(R C₀).card=c) (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J)
  18     (S : Finset V) (hS : S.card ≤ c) :
  19     Fintype.card {C₀ : C // S⊆R C₀} ≤ J*Fintype.card V^(c-S.card) := by
  20   obtain ⟨code,hcode⟩ := finite_fibre_codes R J hJ
  21   let B := {U : FixedSubset V c // S⊆U.val}
  22   let enc (C₀ : {C₀ : C // S⊆R C₀}) : B × Fin J :=
  23     (⟨⟨R C₀.val,hR C₀.val⟩,C₀.property⟩,code C₀.val)
  24   have hinj : Function.Injective enc := by
  25     intro D E he
  26     apply Subtype.ext
  27     apply hcode
  28     · exact congrArg (fun z : B × Fin J => z.1.val.val) he
  29     · exact congrArg Prod.snd he
  30   have hB : Fintype.card B ≤ Fintype.card V^(c-S.card) := by
  31     have he : Fintype.card B=(Fintype.card V-S.card).choose (c-S.card) := by
  32       convert fixedSubset_containing_card S hS using 1 <;> congr!
  33     rw [he]
  34     exact (Nat.choose_le_pow _ _).trans (Nat.pow_le_pow_left (Nat.sub_le _ _) _)
  35   have hh := Fintype.card_le_of_injective enc hinj
  36   rw [Fintype.card_prod,Fintype.card_fin] at hh
  37   exact hh.trans (by simpa only [Nat.mul_comm] using Nat.mul_le_mul_right J hB)

  39 theorem uniform_resources_pair_count (R : C → Finset V) (c J : ℕ) (hc : 2 ≤ c)
  40     (hR : ∀ C₀,(R C₀).card=c) (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J)
  41     (u v : V) (huv : u≠v) :
  42     Fintype.card {C₀ : C // u∈R C₀ ∧ v∈R C₀} ≤ J*Fintype.card V^(c-2) := by
  43   have hcard : ({u,v}:Finset V).card=2 := by simp [huv]
  44   have hh := uniform_resources_containing_count R c J hR hJ {u,v} (by simpa only [hcard] using hc)
  45   simpa only [hcard,Finset.insert_subset_iff,Finset.singleton_subset_iff] using hh
