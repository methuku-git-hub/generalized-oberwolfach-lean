  14 theorem marked_indegree_flip_reduces (head:E→Y) (M D:E→Bool) (y:Y)
  15     (h : ∀ e, head e=y → D e=true → M e=true) :
  16     (∑ e, if head e=y ∧ flipMarkedArcs M D e=true then (1:ℤ) else 0) =
  17     (∑ e, if head e=y ∧ M e=true then (1:ℤ) else 0) -
  18       (∑ e, if head e=y ∧ D e=true then (1:ℤ) else 0) := by
  19   rw [← Finset.sum_sub_distrib]
  20   apply Finset.sum_congr rfl
  21   intro e _
  22   by_cases hy : head e=y
  23   · cases hm : M e <;> cases hd : D e <;>
  24       simp_all [flipMarkedArcs]
  25   · simp [hy]
