  11 theorem long_reserved_prescribed_walks
  12     {V I J:Type*} [Fintype V] [DecidableEq V] [Fintype I] [Fintype J]
  13     (G:SimpleGraph V) (hG:G.IsRegularOfDegree 2)
  14     (components:I↪G.ConnectedComponent)
  15     (e:∀i:I,SimpleGraph.cycleGraph (Fintype.card (components i)) ≃g (components i).toSimpleGraph)
  16     (c d:ℕ) (t:J→ℕ) (ht:∀a,t a≤d)
  17     (hcapacity:Fintype.card J≤∑i:I,Fintype.card (components i)/(c*d+5)):
  18     ∃index:J↪LongReservedBlockIndex G components c d,
  19     ∃ext:∀a:J,Fin (c*t a+5)↪V,
  20     ∃inner:∀a:J,Fin (c*t a+1)↪V,
  21       (∀a,ext a=longReservedSigmaBlock G components c d e (index a) (t a) (ht a)) ∧
  22       (∀(a:J) (j:Fin (c*t a+1)),inner a j=ext a ⟨j.val+2,by omega⟩) ∧
  23       (∀a j,ext a j∈(components (index a).1).supp) ∧
  24       (∀a b:J,a≠b→Disjoint (Finset.univ.image (ext a)) (Finset.univ.image (ext b))) ∧
  25       (∀a:J,∃p:G.Walk (inner a ⟨0,by omega⟩) (inner a ⟨c*t a,by omega⟩),
  26         p.IsPath ∧ p.length=c*t a ∧ p.support=List.ofFn (inner a) ∧
  27         ∃q:G.Walk (ext a ⟨0,by omega⟩) (ext a ⟨c*t a+4,by omega⟩),
  28           q.IsPath ∧ q.length=c*t a+4 ∧ q.support=List.ofFn (ext a)) := by
  29   obtain ⟨index,ext,inner,horient,hext,hinner,hpath,hmem,hdis⟩:=
  30     long_reserved_prescribed_paths G hG components e c d t ht hcapacity
  31   refine ⟨index,ext,inner,horient,hinner,hmem,hdis,?_⟩
  32   intro a
  33   obtain ⟨p,hp,hplen,hpsupp⟩:=finite_sequence_simple_walk G (c*t a) (inner a) (hpath a)
  34   obtain ⟨q,hq,hqlen,hqsupp⟩:=finite_sequence_simple_walk G (c*t a+4) (ext a) (hext a)
  35   exact ⟨p,hp,hplen,hpsupp,q,hq,hqlen,hqsupp⟩
