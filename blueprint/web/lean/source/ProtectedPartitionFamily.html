  15 theorem protectedRegularPartition_family_highProbability
  16     {k : ℕ} (hk : 0 < k) {α : ℝ} (hα : 0 < α)
  17     (r ξ : Fin k → ℝ) (hr : ∀ a, 0 < r a) (hξ : ∀ a, 0 < ξ a) (B : ℕ) :
  18     ∃ N₀ : ℕ, ∀ n ≥ N₀, ∀ (G : SimpleGraph (Fin n)) (D : ℕ)
  19       (m : Fin k → ℕ) (ρ : Fin k → ℝ)
  20       (hρ : ∀ a, 0 ≤ ρ a) (htotal : ∑ a, ρ a = 1),
  21       G.IsRegularOfDegree D → Even D → (1 / 2 + α) * n ≤ (D : ℝ) →
  22       (∑ a, m a = D / 2) → (∀ a, ρ a / 2 * D = (m a : ℝ)) →
  23       (∀ a, r a ≤ ρ a) → ∀ I : Fin k → Finset (Fin n),
  24       ((∑ a, (I a).card : ℕ) : ℝ) ≤ α / 100000 * n →
  25       1 - 1 / (n : ℝ) ^ 10 ≤
  26         (FiniteLaw.pi (fun _ : Sym2 (Fin n) => FiniteLaw.categorical ρ hρ htotal)).probability
  27           (fun c => ∀ A : Fin k → SimpleGraph (Fin n),
  28             (∀ a, A a ≤ edgeColourGraph G c a) →
  29             (∑ a, (A a).edgeFinset.card) ≤ B →
  30             ∃ H : Fin k → SimpleGraph (Fin n),
  31               EdgePartition G H ∧ (∀ a, (H a).IsRegularOfDegree (2 * m a)) ∧
  32               (∀ a, A a ≤ H a) ∧
  33               (∀ a b x, x ∈ I b → ∀ y, y ∈ I b →
  34                 ((H a).Adj x y ↔ (edgeColourGraph G c a).Adj x y)) ∧
  35               (∀ a x, ((Finset.univ.filter (fun y =>
  36                 Xor ((H a).Adj x y) ((edgeColourGraph G c a).Adj x y))).card : ℝ) ≤ ξ a * n)) := by
  37   obtain ⟨r₀, hr₀, hrle⟩ := positive_common_lower_bound hk r hr
  38   obtain ⟨ξ₀, hξ₀, hξle⟩ := positive_common_lower_bound hk ξ hξ
  39   obtain ⟨N₀, hN⟩ := protectedRegularPartition_highProbability k hα hr₀ hξ₀ B
  40   refine ⟨N₀, fun n hn G D m ρ hρ htotal hreg heven hmin hsum hm hrlow I hI => ?_⟩
  41   let Z := protectedInducedUnion G I
  42   have hZ (x : Fin n) : (Z.degree x : ℝ) ≤ α / 100000 * n := by
  43     have hh : (Z.degree x : ℝ) ≤ (∑ a, (I a).card : ℕ) := by
  44       exact_mod_cast protectedInducedUnion_degree G I x
  45     exact hh.trans hI
  46   have hp := hN n hn G Z D m ρ hρ htotal hreg heven hmin hsum hm
  47     (fun a => (hrle a).trans (hrlow a)) hZ
  48   apply hp.trans
  49   apply finiteLaw_probability_mono
  50   intro c hc A hA hsize
  51   let U := Finset.univ.biUnion (fun a => (A a).edgeFinset)
  52   let F := SimpleGraph.fromEdgeSet (U : Set (Sym2 (Fin n)))
  53   have hF (x : Fin n) : F.degree x ≤ B := by
  54     have hu : U.card ≤ ∑ a, (A a).edgeFinset.card := Finset.card_biUnion_le
  55     have hh := (graphFromFiniteEdges_degree U x).trans (hu.trans hsize)
  56     simpa only [← SimpleGraph.ncard_neighborSet] using hh
  57   obtain ⟨H, hH, hregH, hfixed, hdiff⟩ := hc F (fun x => by simpa only [← SimpleGraph.ncard_neighborSet] using hF x)
  58   have hret (a : Fin k) : A a ≤ H a := by
  59     intro x y hxy
  60     have he : s(x,y) ∈ U := by
  61       apply Finset.mem_biUnion.mpr
  62       exact ⟨a, Finset.mem_univ _, by simpa using hxy⟩
  63     have hf : F.Adj x y := ⟨he, (A a).ne_of_adj hxy⟩
  64     exact (hfixed a x y (Or.inr hf)).mpr (hA a hxy)
  65   have hind (a b : Fin k) (x : Fin n) (hx : x ∈ I b) (y : Fin n) (hy : y ∈ I b) :
  66       (H a).Adj x y ↔ (edgeColourGraph G c a).Adj x y := by
  67     by_cases hg : G.Adj x y
  68     · have hz : Z.Adj x y := ⟨hg, b, hx, hy⟩
  69       exact hfixed a x y (Or.inl hz)
  70     · have hh : ¬ (H a).Adj x y := fun h => hg ((hH x y).1.mpr ⟨a, h⟩)
  71       have hc' : ¬ (edgeColourGraph G c a).Adj x y := fun h => hg h.1
  72       simp [hh, hc']
  73   exact ⟨H, hH, hregH, hret, hind, fun a x =>
  74     (hdiff a x).trans (mul_le_mul_of_nonneg_right (hξle a) (Nat.cast_nonneg n))⟩
