  18 def ordinaryWheelResources (W : OrdinaryWheel P F Hbase r) : Finset (Sym2 V⊕(V×U)) :=
  19   (ordinaryWheelRim W).edgeFinset.image Sum.inl ∪
  20     (ordinaryWheelRim W).support.toFinset.image (fun v=>Sum.inr (v,ordinaryWheelHub W))

  22 theorem ordinaryWheelResources_rim (W : OrdinaryWheel P F Hbase r) (e : Sym2 V) :
  23     Sum.inl e∈ordinaryWheelResources W ↔ e∈(ordinaryWheelRim W).edgeSet := by
  24   classical
  25   simp [ordinaryWheelResources]

  27 theorem ordinaryWheelResources_spoke (W : OrdinaryWheel P F Hbase r) (v : V) (u : U) :
  28     Sum.inr (v,u)∈ordinaryWheelResources W ↔ ordinaryWheelHub W=u ∧ v∈(ordinaryWheelRim W).support := by

  38 def ordinaryResourceState (Q : SimpleGraph V) (H : V → U → Prop) : Sym2 V⊕(V×U) → Bool :=
  39   Sum.elim (fun e=>decide (e∈Q.edgeSet)) (fun vu=>decide (H vu.1 vu.2))

  41 def ordinaryPotentialLoad (w : OrdinaryWheel P F Hbase r → ℝ)
  42     (root : Sym2 V⊕(V×U)) (Q : SimpleGraph V) (H : V → U → Prop) : ℝ :=
  43   ∑ W, (if root∈ordinaryWheelResources W then w W else 0)*
  44     patternValue (ordinaryWheelResources W\{root}) (fun _=>true) (ordinaryResourceState Q H)

  46 theorem ordinaryPotentialLoad_stability
  47     (w : OrdinaryWheel P F Hbase r → ℝ) (hw : ∀ W,0≤w W)
  48     (root : Sym2 V⊕(V×U)) (Q Q' : SimpleGraph V) (H H' : V → U → Prop)
  49     (D : Finset (Sym2 V⊕(V×U)))
  50     (hD : ∀ e,e∉D → ordinaryResourceState Q H e=ordinaryResourceState Q' H' e) :
  51     |ordinaryPotentialLoad w root Q' H'-ordinaryPotentialLoad w root Q H|≤
  52       ∑ e∈D,∑ W,if root∈ordinaryWheelResources W ∧ e∈ordinaryWheelResources W ∧ e≠root then w W else 0 := by
  53   classical
  54   have hh := omitted_root_stability Finset.univ ordinaryWheelResources {root} D (fun _ _=>true)
  55     (fun W=>if root∈ordinaryWheelResources W then w W else 0)
  56     (ordinaryResourceState Q H) (ordinaryResourceState Q' H')
  57     (by intro W _;split_ifs;exact hw W;norm_num)
  58     (by intro W _ e _ _ he;exact hD e he)
  59   change |ordinaryPotentialLoad w root Q' H'-ordinaryPotentialLoad w root Q H|≤_ at hh
  60   apply hh.trans_eq
  61   apply Finset.sum_congr rfl
  62   intro e _
  63   apply Finset.sum_congr rfl
  64   intro W _
  65   simp only [Finset.mem_sdiff,Finset.mem_singleton]
  66   split_ifs <;> simp_all

  68 theorem ordinaryPotentialLoad_present
  69     (w : OrdinaryWheel P F Hbase r → ℝ) (root : Sym2 V⊕(V×U))
  70     (Q : SimpleGraph V) (H : V → U → Prop) (hr : ordinaryResourceState Q H root=true) :
  71     ordinaryPotentialLoad w root Q H=
  72       ∑ W,if root∈ordinaryWheelResources W ∧
  73         (∀ e∈ordinaryWheelResources W,ordinaryResourceState Q H e=true) then w W else 0 := by
  74   unfold ordinaryPotentialLoad
  75   apply Finset.sum_congr rfl
  76   intro W _
  77   rw [patternValue_indicator]
  78   have he : (∀ e∈ordinaryWheelResources W\{root},ordinaryResourceState Q H e=true) ↔
  79       (∀ e∈ordinaryWheelResources W,ordinaryResourceState Q H e=true) := by
