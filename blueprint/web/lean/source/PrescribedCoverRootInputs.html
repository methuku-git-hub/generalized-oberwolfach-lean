  17     (hRQ:R≤Q) (hKH:∀ v u,K v u→H v u)
  18     (hdec:HasOrdinaryWheelDecomposition R K r) (v:V)

  33 theorem prescribed_cover_root_geometry (C:Z→OrdinaryWheel Q F H r)
  34     (I:Finset V) (J:Finset U) (T:Finset V) (W:Finset U)
  35     (E₀:Finset (Sym2 V⊕(V×U)))
  36     (hcovered:E₀⊆ordinaryHostResources (wheelFamilyRim C) (wheelFamilySpokes C))
  37     (hrim:∀ x∈T,∀ y∈I,Q.Adj x y→Sum.inl s(x,y)∈E₀)
  38     (hrow:∀ x∈T,∀ u∈J,H x u→Sum.inr (x,u)∈E₀)
  39     (hcol:∀ u∈W,∀ x∈I,H x u→Sum.inr (x,u)∈E₀):
  40     (∀ x∈T,(Q\wheelFamilyRim C).neighborFinset x⊆Finset.univ\I) ∧
  41     (∀ x∈T,∀ u,H x u∧¬wheelFamilySpokes C x u→u∈Finset.univ\J) ∧
  42     (∀ u∈W,Finset.univ.filter (fun v=>H v u∧¬wheelFamilySpokes C v u)⊆Finset.univ\I):=by
  43   refine ⟨?_,?_,?_⟩
  44   · intro x hx y hy
  45     obtain ⟨hQ,hnot⟩:=((Q\wheelFamilyRim C).mem_neighborFinset x y).mp hy
  46     refine Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,?_⟩
  47     intro hyI
  48     have he:=hcovered (hrim x hx y hyI hQ)
  49     exact hnot ((ordinaryHostResources_rim (wheelFamilyRim C) (wheelFamilySpokes C) s(x,y)).mp he)
  50   · intro x hx u hu
  51     refine Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,?_⟩
  52     intro huJ
  53     have he:=hcovered (hrow x hx u huJ hu.1)
  54     exact hu.2 ((ordinaryHostResources_spoke (wheelFamilyRim C) (wheelFamilySpokes C) x u).mp he)
  55   · intro u hu x hx
  56     obtain ⟨hH,hnot⟩:=(Finset.mem_filter.mp hx).2
  57     refine Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,?_⟩
  58     intro hxI
  59     have he:=hcovered (hcol u hu x hxI hH)
  60     exact hnot ((ordinaryHostResources_spoke (wheelFamilyRim C) (wheelFamilySpokes C) x u).mp he)

  63     (hdis:∀ i j,i≠j→Disjoint (ordinaryWheelResources (C i)) (ordinaryWheelResources (C j)))
