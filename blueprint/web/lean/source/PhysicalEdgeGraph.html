   3 /-! The simple graph determined by a finite family of physical edges.
   4 An incidence-cardinality bound implies the ordinary graph degree bound. -/
   5 set_option autoImplicit false
   6 set_option backward.isDefEq.respectTransparency false
   7 noncomputable section
   8 namespace Oberwolfach
   9 attribute [local instance] Classical.propDecidable
  10 variable {V : Type*} [Fintype V] [DecidableEq V]
  11 
  12 def physicalEdgeGraph (S : Finset (Finset V)) : SimpleGraph V where
  13   Adj x y := x≠y ∧ ({x,y}:Finset V)∈S
  14   symm := ⟨by intro x y h;exact ⟨h.1.symm,by simpa only [Finset.pair_comm] using h.2⟩⟩
  15   loopless := ⟨by intro x h;exact h.1 rfl⟩
  16 
  17 theorem physicalEdgeGraph_degree_le_incident (S : Finset (Finset V)) (v : V) :
  18     (physicalEdgeGraph S).degree v≤(S.filter (fun e => v∈e)).card := by
  19   let G := physicalEdgeGraph S
  20   let f : {u : V // G.Adj v u} → S.filter (fun e => v∈e) := fun u =>
  21     ⟨{v,u.val},Finset.mem_filter.mpr ⟨u.property.2,by simp⟩⟩
  22   have hf : Function.Injective f := by
  23     intro u w he
  24     apply Subtype.ext
  25     have hp : ({v,u.val}:Finset V)={v,w.val} := congrArg Subtype.val he
  26     have hu : u.val∈({v,w.val}:Finset V) := by rw [← hp];simp
  27     rcases Finset.mem_insert.mp hu with hv|hu
  28     · exact (u.property.1 hv.symm).elim
  29     · exact Finset.mem_singleton.mp hu
  30   have hh := Fintype.card_le_of_injective f hf
  31   have he : Fintype.card {u : V // G.Adj v u}=G.degree v := by
  32     convert G.card_neighborSet_eq_degree v using 1 <;> congr!
  33   rw [he,Fintype.card_coe] at hh
  34   exact hh
  35 
  36 end Oberwolfach
