  13 theorem large_sets_half_intersection {V : Type*} [Fintype V] [DecidableEq V]
  14     (A B : Finset V) (hA : 3*Fintype.card V≤4*A.card) (hB : 3*Fintype.card V≤4*B.card) :
  15     Fintype.card V≤2*(A∩B).card := by
  16   have hh := Finset.card_union_add_card_inter A B
  17   have hU := Finset.card_le_univ (A∪B)
  18   omega

  20 theorem tripartite_triangle_bijections {X Y Z : Type*} [Fintype X] [Fintype Y] [Fintype Z]
  21     (E₀₁ : X→Y→Prop) (E₀₂ : X→Z→Prop) (E₁₂ : Y→Z→Prop)
  22     (hXY : Fintype.card X=Fintype.card Y) (hXZ : Fintype.card X=Fintype.card Z)
  23     (h₀₁r : ∀ x,3*Fintype.card Y≤4*(Finset.univ.filter (E₀₁ x)).card)
  24     (h₀₁c : ∀ y,3*Fintype.card X≤4*(Finset.univ.filter (fun x=>E₀₁ x y)).card)
  25     (h₀₂r : ∀ x,3*Fintype.card Z≤4*(Finset.univ.filter (E₀₂ x)).card)
  26     (h₀₂c : ∀ z,3*Fintype.card X≤4*(Finset.univ.filter (fun x=>E₀₂ x z)).card)
  27     (h₁₂r : ∀ y,3*Fintype.card Z≤4*(Finset.univ.filter (E₁₂ y)).card)
  28     (h₁₂c : ∀ z,3*Fintype.card Y≤4*(Finset.univ.filter (fun y=>E₁₂ y z)).card) :
  29     ∃ σ : X≃Y,∃ τ : X≃Z,∀ x,E₀₁ x (σ x) ∧ E₀₂ x (τ x) ∧ E₁₂ (σ x) (τ x) := by
  30   classical
  31   obtain ⟨σ,hσ⟩ := dense_bipartite_perfect_matching E₀₁ hXY
  32     (fun x=>by have hh:=h₀₁r x;omega) (fun y=>by have hh:=h₀₁c y;omega)
  33   let E : X→Z→Prop := fun x z=>E₀₂ x z ∧ E₁₂ (σ x) z
  34   have hrow (x : X) : Fintype.card Z≤2*(Finset.univ.filter (E x)).card := by
  35     have he : Finset.univ.filter (E x)=(Finset.univ.filter (E₀₂ x))∩(Finset.univ.filter (E₁₂ (σ x))) := by
  36       ext z;simp [E]
  37     rw [he]
  38     exact large_sets_half_intersection _ _ (h₀₂r x) (h₁₂r (σ x))
  39   have hcol (z : Z) : Fintype.card X≤2*(Finset.univ.filter (fun x=>E x z)).card := by
  40     have hcard : (Finset.univ.filter (fun x=>E₁₂ (σ x) z)).card=
  41         (Finset.univ.filter (fun y=>E₁₂ y z)).card := by
  42       apply Finset.card_bij (fun x _=>σ x)
  43       · intro x hx;simpa using hx
  44       · intro x _ y _ h;exact σ.injective h
  45       · intro y hy;exact ⟨σ.symm y,by simpa using hy,σ.apply_symm_apply y⟩
  46     have he : Finset.univ.filter (fun x=>E x z)=
  47         (Finset.univ.filter (fun x=>E₀₂ x z))∩(Finset.univ.filter (fun x=>E₁₂ (σ x) z)) := by
  48       ext x;simp [E]
  49     rw [he]
  50     apply large_sets_half_intersection _ _ (h₀₂c z)
  51     rw [hcard,hXY]
  52     exact h₁₂c z
  53   obtain ⟨τ,hτ⟩ := dense_bipartite_perfect_matching E hXZ (fun x=>by convert hrow x using 1 <;> congr!) (fun z=>by convert hcol z using 1 <;> congr!)
  54   exact ⟨σ,τ,fun x=>⟨hσ x,hτ x⟩⟩
