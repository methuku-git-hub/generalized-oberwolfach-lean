  24 theorem markedPathChoice_closing
  25     (P : SimpleGraph V) (π : Equiv.Perm V) (b r : ℕ) (j : Bool)
  26     (T₀ TK : Finset V) (v v' : V) (f : Fin (r+4)→V)
  27     (hf : MixedGreedyGood P π b (markedPathChoiceClass j T₀ TK) (markedPathChoiceHead π j)
  28       (markedPathChoiceRequests π j (r+4) v v') {v,v'} f) :
  29     P.Adj (f 1) (if j then π v else v) ∧ P.Adj (f 1) (if j then π v' else v') := by
  30   have h := hf.2.2.2 (1:Fin (r+4))
  31   have hv : (1:Fin (r+4)).val=1 := by simp [Nat.mod_eq_of_lt (show 1<r+4 by omega)]

  33   have h' : P.Adj (if j then π v else v) (f 1) ∧ P.Adj (if j then π v' else v') (f 1) := by
  34     simpa [markedPathChoiceRequests,markedPathChoiceHead] using h
  35   exact ⟨h'.1.symm,h'.2.symm⟩

  48 theorem markedPathChoice_internal
  49     (P : SimpleGraph V) (π : Equiv.Perm V) (b r : ℕ) (j : Bool)
  50     (T₀ TK : Finset V) (v v' : V) (f : Fin (r+4)→V)
  51     (hf : MixedGreedyGood P π b (markedPathChoiceClass j T₀ TK) (markedPathChoiceHead π j)
  52       (markedPathChoiceRequests π j (r+4) v v') {v,v'} f)
  53     (n : ℕ) (hn : 3≤n) (hnr : n<r+4) :
  54     P.Adj (f ⟨n-1,by omega⟩) (markedPathChoiceHead π j n (f ⟨n,hnr⟩)) := by
  55   have h := hf.2.2.2 ⟨n,hnr⟩
  56   apply h
  57   change f ⟨n-1,by omega⟩∈markedPathChoiceRequests π j (r+4) v v' n (greedyPrefix f ⟨n,hnr⟩)
  58   unfold markedPathChoiceRequests
  59   rw [if_neg (by omega),if_neg (by omega),dif_neg (by omega),dif_pos (by omega)]
  60   exact Finset.mem_union_left _ (Finset.mem_singleton_self _)
