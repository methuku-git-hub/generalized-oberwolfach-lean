  13 abbrev OrdinaryCycle {V : Type*} (G : SimpleGraph V) (r : ℕ) :=
  14   {C : EmbeddedCycleFamily V r // C.val≤G}

  16 def ordinaryCycleIncidence {V : Type*} (G : SimpleGraph V) (r : ℕ)
  17     (C : OrdinaryCycle G r) (e : G.edgeSet) : ℝ :=
  18   if e.val∈C.val.val.edgeSet then 1 else 0

  20 theorem ordinaryCycleIncidence_nonneg {V : Type*} (G : SimpleGraph V) (r : ℕ)
  21     (C : OrdinaryCycle G r) (e : G.edgeSet) : 0≤ordinaryCycleIncidence G r C e := by
  22   dsimp [ordinaryCycleIncidence]
  23   split_ifs <;> norm_num

  25 theorem ordinary_cycle_incidence_count {V : Type*} [Fintype V] [DecidableEq V]
  26     (G : SimpleGraph V) (r : ℕ) (e : G.edgeSet) :
  27     (∑ C : OrdinaryCycle G r,ordinaryCycleIncidence G r C e)≤
  28       (r:ℝ)^2*(Fintype.card V:ℝ)^(r-2) := by
  29   apply (finite_subtype_sum_le (fun C : EmbeddedCycleFamily V r => C.val≤G)
  30     (fun C => if e.val∈C.val.edgeSet then (1:ℝ) else 0)
  31     (fun C => by split_ifs <;> norm_num)).trans
  32   rcases e with ⟨e,he⟩
  33   induction e using Sym2.ind with
  34   | _ u v =>
  35     have huv : G.Adj u v := he
  36     have hh := ordinary_cycle_edge_count r u v huv.ne
  37     convert hh using 1
  38     apply Finset.sum_congr rfl
  39     intro C _
  40     simp only [SimpleGraph.mem_edgeSet]

  42 theorem ordinary_cycle_positive_baseline {V : Type*} [Fintype V] [DecidableEq V]
  43     (G : SimpleGraph V) (r : ℕ) (hr : 0<r) (κ : ℝ) (hκ : 0<κ)
  44     (hn : 0<Fintype.card V) :
  45     ‖∑ C : OrdinaryCycle G r,
  46       ((κ/(4*(r:ℝ)^2))/(Fintype.card V:ℝ)^(r-2)) • ordinaryCycleIncidence G r C‖≤κ/4 := by
  47   have hnr : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr hn
  48   have hrr : (0:ℝ)<r := Nat.cast_pos.mpr hr
  49   have hp : 0<(Fintype.card V:ℝ)^(r-2) := pow_pos hnr _
  50   have hb : 0≤(κ/(4*(r:ℝ)^2))/(Fintype.card V:ℝ)^(r-2) := by positivity
  51   apply (pi_norm_le_iff_of_nonneg (show 0≤κ/4 by positivity)).mpr
  52   intro e
  53   rw [Finset.sum_apply]
  54   simp only [Pi.smul_apply,smul_eq_mul,← Finset.mul_sum,Real.norm_eq_abs]
  55   have hs : 0≤∑ C : OrdinaryCycle G r,ordinaryCycleIncidence G r C e :=
  56     Finset.sum_nonneg (fun C _ => ordinaryCycleIncidence_nonneg G r C e)
  57   rw [abs_of_nonneg (mul_nonneg hb hs)]
  58   have hh := mul_le_mul_of_nonneg_left (ordinary_cycle_incidence_count G r e) hb
  59   have he : ((κ/(4*(r:ℝ)^2))/(Fintype.card V:ℝ)^(r-2))*
  60       ((r:ℝ)^2*(Fintype.card V:ℝ)^(r-2))=κ/4 := by field_simp
  61   exact he ▸ hh
