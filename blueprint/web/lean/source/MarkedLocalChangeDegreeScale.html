   9 theorem marked_local_change_degree_scale
  10     {V W X Y:Type} [Fintype V] [Fintype W] [Fintype X] [Fintype Y]
  11     [DecidableEq V] [DecidableEq W] [DecidableEq X] [DecidableEq Y]
  12     (f:W↪V) (e:X↪Y) (FR:SimpleGraph V) (FS:SimpleGraph Y)
  13     (iota ζGlobal ζLocal:ℝ)
  14     (hζLocal:0≤ζLocal) (hsize:iota*Fintype.card V≤(Fintype.card W:ℝ))
  15     (hsmall:ζGlobal ≤ iota*ζLocal)
  16     (hFR:∀ v,(FR.degree v:ℝ)≤ζGlobal*Fintype.card V)
  17     (hFS:∀ y,(FS.degree y:ℝ)≤ζGlobal*Fintype.card V):
  18     (∀ w,((FR.comap f).degree w:ℝ)≤ζLocal*Fintype.card W) ∧
  19     (∀ x,((FS.comap e).degree x:ℝ)≤ζLocal*Fintype.card W) := by
  20   have hn:(0:ℝ)≤Fintype.card V:=Nat.cast_nonneg _
  21   have hglobal:=mul_le_mul_of_nonneg_right hsmall hn
  22   have hlocal:=mul_le_mul_of_nonneg_left hsize hζLocal
  23   have hbound:ζGlobal*Fintype.card V≤ζLocal*Fintype.card W:=by nlinarith
  24   constructor
  25   · intro w
  26     exact (Nat.cast_le.mpr (graph_comap_degree_le FR f w)).trans ((hFR (f w)).trans hbound)
  27   · intro x
  28     exact (Nat.cast_le.mpr (graph_comap_degree_le FS e x)).trans ((hFS (e x)).trans hbound)
  29 end Oberwolfach
