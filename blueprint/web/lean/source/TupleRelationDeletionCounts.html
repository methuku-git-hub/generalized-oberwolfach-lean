  17 theorem tuple_relation_indicator_count (D : V → V → Prop) (i j : I) (hij : i≠j) :
  18     (∑ z : I → V,if D (z i) (z j) then (1:ℝ) else 0)=
  19       (∑ x,∑ y,bipAdj D x y)*(Fintype.card V:ℝ)^(Fintype.card I-2) := by
  20   have expand (z : I → V) : (if D (z i) (z j) then (1:ℝ) else 0)=
  21       ∑ x,∑ y,if D x y then (if z i=x ∧ z j=y then (1:ℝ) else 0) else 0 := by
  22     have he (x y : V) : (if D x y then (if z i=x ∧ z j=y then (1:ℝ) else 0) else 0)=
  23         if x=z i then (if y=z j then bipAdj D x y else 0) else 0 := by
  24       by_cases hx : x=z i <;> by_cases hy : y=z j <;> by_cases hd : D x y <;> simp_all [bipAdj,eq_comm]
  25     simp only [he]
  26     simp [bipAdj]
  27   simp_rw [expand]
  28   rw [Finset.sum_comm]
  29   calc
  30     _ = ∑ x,∑ y,∑ z : I → V,if D x y then (if z i=x ∧ z j=y then (1:ℝ) else 0) else 0 := by
  31       apply Finset.sum_congr rfl
  32       intro x _
  33       exact Finset.sum_comm
  34     _ = ∑ x,∑ y,bipAdj D x y*(Fintype.card V:ℝ)^(Fintype.card I-2) := by
  35       apply Finset.sum_congr rfl
  36       intro x _
  37       apply Finset.sum_congr rfl
  38       intro y _
  39       by_cases hd : D x y
  40       · simp only [hd,ite_true,bipAdj,one_mul]
  41         convert two_position_indicator_total i j hij x y using 1 <;> congr!
  42       · simp only [hd,ite_false,bipAdj,zero_mul,Finset.sum_const_zero]
  43     _ = _ := by simp_rw [← Finset.sum_mul]

  90 theorem finite_predicate_loss_union {A J : Type*} [Fintype A] [Fintype J]
  91     (P Q : A → Prop) (bad : J → A → Prop)
  92     (hbad : ∀ a,P a → ¬Q a → ∃ j,bad j a) :
  93     (Fintype.card {a // P a}:ℝ)-Fintype.card {a // Q a}≤
  94       ∑ j,∑ a,if bad j a then (1:ℝ) else 0 := by
  95   have he (T : A → Prop) : (Fintype.card {a // T a}:ℝ)=∑ a,if T a then (1:ℝ) else 0 := by
  96     simp only [Fintype.card_subtype,Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  97   rw [he P,he Q,← Finset.sum_sub_distrib,Finset.sum_comm]

  99   intro a _
 100   have hnon : 0≤∑ j,if bad j a then (1:ℝ) else 0 :=
 101     Finset.sum_nonneg (fun j _=>by split_ifs <;> norm_num)
 102   by_cases hp : P a

 112 theorem finite_constraint_loss {A J : Type*} [Fintype A] [Fintype J]
 113     (base : A → Prop) (P Q : J → A → Prop) (M : ℝ)
 114     (hM : ∀ j,(∑ a,if P j a ∧ ¬Q j a then (1:ℝ) else 0)≤M) :
 115     (Fintype.card {a // base a ∧ ∀ j,P j a}:ℝ)-
 116       Fintype.card {a // base a ∧ ∀ j,Q j a}≤(Fintype.card J:ℝ)*M := by
 117   have hh := finite_predicate_loss_union (fun a=>base a ∧ ∀ j,P j a)
 118     (fun a=>base a ∧ ∀ j,Q j a) (fun j a=>P j a ∧ ¬Q j a) (by
 119       intro a hp hq
 120       have hnot : ¬∀ j,Q j a := fun h=>hq ⟨hp.1,h⟩
 121       push_neg at hnot
 122       obtain ⟨j,hj⟩ := hnot
 123       exact ⟨j,hp.2 j,hj⟩)
 124   have hbound : (∑ j,∑ a,if P j a ∧ ¬Q j a then (1:ℝ) else 0)≤(Fintype.card J:ℝ)*M := by
 125     calc
 126       _ ≤ ∑ _j : J,M := Finset.sum_le_sum (fun j _=>hM j)
 127       _ = _ := by simp
 128   convert (hh.trans (by convert hbound using 1 <;> congr!)) using 1 <;> congr!
