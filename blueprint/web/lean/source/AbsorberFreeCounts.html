  16 theorem labelledCycleRecord_ext {h : ℕ} {O : Type*} [Fintype O] [DecidableEq O]
  17     (R : C → Finset V) (S : O → Finset (Fin h ⊕ Fin c)) (X : Finset V)
  18     (D E : LabelledCycleRecord R S X) (hf : ∀ j,D.1.val j=E.1.val j)
  19     (hg : ∀ o,(D.2 o).val=(E.2 o).val) : D=E := by
  20   rcases D with ⟨f,d⟩
  21   rcases E with ⟨g,e⟩
  22   have he : f=g := Subtype.ext (funext hf)
  23   subst g
  24   have he' : d=e := funext (fun o => Subtype.ext (hg o))
  25   subst e
  26   rfl

  28 theorem free_absorber_records_card (R : C → Finset V) (S : VertexAbsorberTemplate c) (J : ℕ)
  29     (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J) :
  30     Fintype.card (FreeAbsorberRecord R S) ≤ Fintype.card V^(c*c)*J^c := by
  31   obtain ⟨code,hcode⟩ := finite_fibre_codes R J hJ
  32   let enc (D : FreeAbsorberRecord R S) : (Fin (c*c) → V) × (Fin c → Fin J) :=
  33     (D.1,fun i => code (D.2 i).val)
  34   have hinj : Function.Injective enc := by
  35     rintro ⟨f,d⟩ ⟨g,e⟩ he
  36     have hf : f=g := DFunLike.coe_injective (congrArg Prod.fst he)
  37     subst g
  38     have hid : d=e := by
  39       funext i
  40       apply Subtype.ext
  41       apply hcode
  42       · exact (d i).property.trans (e i).property.symm
  43       · exact congrFun (congrArg Prod.snd he) i
  44     subst e
  45     rfl
  46   have hh := Fintype.card_le_of_injective enc hinj
  47   simpa only [Fintype.card_prod,Fintype.card_fun,Fintype.card_fin] using hh

  49 theorem forget_absorber_roots_fibre_card (R : C → Finset V) (S : VertexAbsorberTemplate c)
  50     (X : Finset V) (J : ℕ)
  51     (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J) (F : FreeAbsorberRecord R S) :
  52     Fintype.card {D : LabelledCycleRecord R S.val X // forgetAbsorberRoots R S X D=F} ≤
  53       X.card^c*J^(c+1) := by
  54   obtain ⟨code,hcode⟩ := finite_fibre_codes R J hJ
  55   let A := {D : LabelledCycleRecord R S.val X // forgetAbsorberRoots R S X D=F}
  56   let enc (D : A) : (Fin c → X) × (Fin (c+1) → Fin J) :=
  57     (fun i => ⟨D.val.1.val (Sum.inr i),rootedLabelMap_root_mem X D.val.1 i⟩,
  58      fun i => code (D.val.2 (Sum.inr i)).val)
  59   have hinj : Function.Injective enc := by
  60     intro D E he
  61     have hf (j : Fin (c*c) ⊕ Fin c) : D.val.1.val j=E.val.1.val j := by
  62       rcases j with j | j
  63       · have hd := congrArg (fun F : FreeAbsorberRecord R S => F.1 j) D.property
  64         have he' := congrArg (fun F : FreeAbsorberRecord R S => F.1 j) E.property
  65         exact hd.trans he'.symm
  66       · exact congrArg (fun z : X => z.val) (congrFun (congrArg Prod.fst he) j)
  67     apply Subtype.ext
  68     apply labelledCycleRecord_ext R S.val X D.val E.val hf
  69     intro o
  70     rcases o with i | i
  71     · have hd := congrArg (fun F : FreeAbsorberRecord R S => (F.2 i).val) D.property
  72       have he' := congrArg (fun F : FreeAbsorberRecord R S => (F.2 i).val) E.property
  73       exact hd.trans he'.symm
  74     · apply hcode
  75       · rw [(D.val.2 (Sum.inr i)).property,(E.val.2 (Sum.inr i)).property]
  76         exact congrArg (Finset.image · (S.val (Sum.inr i))) (funext hf)
  77       · exact congrFun (congrArg Prod.snd he) i
  78   have hh := Fintype.card_le_of_injective enc hinj
  79   simpa only [Fintype.card_prod,Fintype.card_fun,Fintype.card_fin,Fintype.card_coe] using hh
