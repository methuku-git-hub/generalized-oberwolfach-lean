  10 theorem matching_power_scales (k h D n : ℕ) (hk : 2≤k) (hh : 2≤h)
  11     (hDn : n≤D) (hn : h^(64*k)≤n) (K : ℝ) (hK : 0<K) (hKh : 2*K≤h) :
  12     let p : ℝ := h^16
  13     let q := D/h^(16*k-12)
  14     h*p^(2*k)≤n ∧ h*p^k*q≤(D : ℝ)*n ∧
  15     (D : ℝ)^2*p*h≤4*(p^k)^2*q^2 ∧ (K*D/n)/(q : ℝ)≤1 := by
  16   dsimp only
  17   let m := h^(16*k-12)
  18   let q := D/m
  19   have hh0 : 0<h := by omega
  20   have hhR : (1 : ℝ)≤h := by exact_mod_cast (show 1≤h by omega)
  21   have hn0 : 0<n := lt_of_lt_of_le (pow_pos hh0 _) hn
  22   have hD0 : 0<D := hn0.trans_le hDn
  23   have hm0 : 0<m := pow_pos hh0 _
  24   have hmD : m≤D := by
  25     calc
  26       _ ≤ h^(64*k) := Nat.pow_le_pow_right (by omega) (by omega)
  27       _ ≤ n := hn
  28       _ ≤ D := hDn
  29   have hq0 : 0<q := Nat.div_pos hmD hm0
  30   have hfloor : D<m*q+m := by
  31     have hd := Nat.mod_lt D hm0
  32     have he := Nat.mod_add_div D m
  33     dsimp only [q]
  34     omega
  35   have hmq : (D : ℝ)≤2*(m : ℝ)*q := by
  36     have hd : (D : ℝ)<(m : ℝ)*q+m := by exact_mod_cast hfloor

  38     have hmR : (0 : ℝ)≤m := Nat.cast_nonneg m
  39     nlinarith
  40   have hnR : (h : ℝ)^(64*k)≤n := by exact_mod_cast hn
  41   have hbound (j : ℕ) (hj : j≤64*k) : (h : ℝ)^j≤n :=
  42     (pow_le_pow_right₀ hhR hj).trans hnR
  43   have hp2 : (h : ℝ)*((h : ℝ)^16)^(2*k)≤n := by
  44     rw [← pow_mul]
  45     have heq : (h : ℝ)*(h : ℝ)^(16*(2*k))=(h : ℝ)^(16*(2*k)+1) := by rw [pow_succ];ring
  46     rw [heq]
  47     exact hbound _ (by omega)
  48   have hp1 : (h : ℝ)*((h : ℝ)^16)^k≤n := by
  49     rw [← pow_mul]
  50     have heq : (h : ℝ)*(h : ℝ)^(16*k)=(h : ℝ)^(16*k+1) := by rw [pow_succ];ring
  51     rw [heq]
  52     exact hbound _ (by omega)
  53   have hqD : (q : ℝ)≤D := Nat.cast_le.mpr (Nat.div_le_self _ _)
  54   have hprod : (h : ℝ)*((h : ℝ)^16)^k*(q : ℝ)≤(D : ℝ)*n := by
  55     calc
  56       _ ≤ (n : ℝ)*q := mul_le_mul_of_nonneg_right hp1 (Nat.cast_nonneg _)
  57       _ ≤ (n : ℝ)*D := mul_le_mul_of_nonneg_left hqD (Nat.cast_nonneg _)
  58       _ = _ := by ring
  59   have hmP : (m : ℝ)^2*(h : ℝ)^16*h≤(((h : ℝ)^16)^k)^2 := by
  60     dsimp only [m]
  61     push_cast
  62     rw [← pow_mul,← pow_mul,← pow_mul,← pow_add]
  63     rw [← pow_succ]
  64     exact pow_le_pow_right₀ hhR (by omega)
  65   have hselection : (D : ℝ)^2*(h : ℝ)^16*h≤4*(((h : ℝ)^16)^k)^2*(q : ℝ)^2 := by

  67     have h1 := mul_le_mul_of_nonneg_right hsq (show 0≤(h : ℝ)^16*h by positivity)
  68     have h2 := mul_le_mul_of_nonneg_right hmP (show 0≤4*(q : ℝ)^2 by positivity)
  69     nlinarith
  70   have hcode : (K*D/n)/(q : ℝ)≤1 := by

  72       dsimp only [m]
