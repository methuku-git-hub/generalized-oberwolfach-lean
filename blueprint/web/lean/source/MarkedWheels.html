  17 def MarkedHubCompatible {V U : Type*} {m : ℕ}
  18     (H₀ Hₖ : V → U → Prop) (C : Fin (m+1) → V) (u : U) : Prop :=
  19   Hₖ (C 0) u ∧ ∀ i : Fin m,H₀ (C i.succ) u

  21 abbrev MarkedWheel {V U E K : Type*}
  22     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (π : Equiv.Perm V) (b : ℕ)
  23     (H₀ Hₖ : V → U → Prop) :=
  24   (C : MarkedCycleTuple tail head ktail khead m π b) × {u // MarkedHubCompatible H₀ Hₖ C.val u}

  26 def markedSpokeIndicator {V : Type*} [DecidableEq V] {m : ℕ}
  27     (C : Fin (m+1) → V) (j : Bool) (v : V) : ℝ :=
  28   if j then (if C 0=v then 1 else 0) else ∑ i : Fin m,if C i.succ=v then 1 else 0

  58 theorem markedHubCompatible_missing_spoke {V U : Type*} [DecidableEq V] {m : ℕ}
  59     (H₀ Hₖ : V → U → Prop) (C : Fin (m+1) → V) (u : U)
  60     (hc : MarkedHubCompatible H₀ Hₖ C u) (j : Bool) (v : V)
  61     (hmiss : ¬(if j then Hₖ v u else H₀ v u)) : markedSpokeIndicator C j v=0 := by
  62   cases j
  63   · simp only [markedSpokeIndicator,Bool.false_eq_true,ite_false] at *
  64     apply Finset.sum_eq_zero
  65     intro i _
  66     apply if_neg
  67     intro he
  68     exact hmiss (he ▸ hc.2 i)
  69   · simp only [markedSpokeIndicator,ite_true] at *
  70     apply if_neg
  71     intro he
  72     exact hmiss (he ▸ hc.1)
