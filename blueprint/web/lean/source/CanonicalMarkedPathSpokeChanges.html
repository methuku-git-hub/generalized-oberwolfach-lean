  12 theorem canonicalCommonMarkedPath_spoke_changes
  13     {V U E K:Type*} [Fintype V] [DecidableEq V]
  14     (tail head:E→V) (ktail khead:K→V) (m:ℕ) (π:Equiv.Perm V) (b:ℕ)
  15     (H H':Bool→V→U→Prop) (i:Fin (m+1)) (v v':V) (u u':U) (δ:ℝ)
  16     (hcol:∀ b u,(∑ x,if ¬(H b x u↔H' b x u) then (1:ℝ) else 0)≤δ):
  17     (Fintype.card (CanonicalCommonMarkedPath tail head ktail khead m π b (H false) (H true) i v v' u u'):ℝ)-
  18       Fintype.card (CanonicalCommonMarkedPath tail head ktail khead m π b (H' false) (H' true) i v v' u u')≤
  19       2*(m:ℝ)*δ*(Fintype.card V:ℝ)^(m-1):=by
  20   let base:=fun z:Fin m→V=>
  21     IsMarkedCycleTuple tail head ktail khead m π b (Fin.insertNth i v z)∧
  22     IsMarkedCycleTuple tail head ktail khead m π b (Fin.insertNth i v' z)
  23   have he (J:Bool→V→U→Prop):
  24       CanonicalCommonMarkedPath tail head ktail khead m π b (J false) (J true) i v v' u u'≃
  25       {z:Fin m→V // base z∧∀ a,J (markedPositionType (i.succAbove a)) (z a) u∧
  26         J (markedPositionType (i.succAbove a)) (z a) u'}:=by
  27     apply Equiv.subtypeEquivRight
  28     intro z
  29     have hs (a:Fin m) (w:U):
  30         (if markedPositionType (i.succAbove a) then J true (z a) w else J false (z a) w)↔
  31           J (markedPositionType (i.succAbove a)) (z a) w:=by
  32       cases markedPositionType (i.succAbove a) <;> simp
  33     simp only [hs,base,and_assoc]
  34   rw [Fintype.card_congr (he H),Fintype.card_congr (he H')]
  35   have hh:=twoHubTuple_spoke_loss H H' u u' (fun a:Fin m=>markedPositionType (i.succAbove a)) base δ hcol
  36   simp only [Fintype.card_fin] at hh
  37   simp only [Fintype.card_eq_nat_card] at hh ⊢
  38   convert hh using 1 <;> ring
