  15     let T:=Finset.univ.filter (fun C:(F u).ConnectedComponent=>Fintype.card C=i.val+1)
  16     0≤longAssignedKeptFraction F hF K θ i u*(n:ℝ)/((i.val+1:ℕ):ℝ) ∧
  17       longAssignedKeptFraction F hF K θ i u*(n:ℝ)/((i.val+1:ℕ):ℝ)≤T.card :=by
  18   intro T
  19   let f:=ordinaryAssignedFraction F hF K u (some i)
  20   have hf:0≤f:=by dsimp [f,ordinaryAssignedFraction];positivity
  21   have hkept:0≤longAssignedKeptFraction F hF K θ i u ∧
  22       longAssignedKeptFraction F hF K θ i u≤f:=by
  23     dsimp [longAssignedKeptFraction]
  24     split_ifs <;> constructor <;> simp_all only [f,le_refl]
  25   have hr:0<((i.val+1:ℕ):ℝ):=by positivity
  26   constructor
  27   · exact div_nonneg (mul_nonneg hkept.1 (by positivity)) hr.le
  28   · have h:=mul_le_mul_of_nonneg_right hkept.2 (show (0:ℝ)≤n by positivity)
  29     have hd:=div_le_div_of_nonneg_right h hr.le
