  15 theorem selectionPotential_zero_bound {I : Type*} [Fintype I] {N : ℕ}
  16     (active : Fin N→I→Prop) (q M L : ℝ) (hq : 0≤q)
  17     (hcount : (Fintype.card I:ℝ)≤L)
  18     (hremaining : ∀ x,(selectionRemaining active 0 x:ℝ)≤M) :
  19     selectionPotential active q 0 (fun _=>0)≤L*Real.exp (q*M) := by
  20   have hbase : 1+q≤Real.exp q := by simpa only [add_comm] using Real.add_one_le_exp q
  21   have hp (x : I) : (1+q)^selectionRemaining active 0 x≤Real.exp (q*M) := by
  22     calc
  23       _ ≤ (Real.exp q)^selectionRemaining active 0 x := pow_le_pow_left₀ (by linarith) hbase _
  24       _ = Real.exp (q*(selectionRemaining active 0 x:ℝ)) := by rw [← Real.exp_nat_mul,mul_comm]
  25       _ ≤ _ := Real.exp_le_exp.mpr (mul_le_mul_of_nonneg_left (hremaining x) hq)
  26   calc
  27     _ = ∑ x,(1+q)^selectionRemaining active 0 x := by simp [selectionPotential]
  28     _ ≤ ∑ _x : I,Real.exp (q*M) := Finset.sum_le_sum (fun x _=>hp x)
  29     _ = (Fintype.card I:ℝ)*Real.exp (q*M) := by simp
  30     _ ≤ _ := mul_le_mul_of_nonneg_right hcount (Real.exp_pos _).le

  45 theorem bounded_degree_selection_order (p : ℝ) (hp : 0<p) :
  46     ∃ n₀ : ℕ,2≤n₀ ∧ ∀ n : ℕ,n₀≤n →
  47     ∀ (I : Type*) [Fintype I] (N : ℕ) (active : Fin N→I→Prop),
  48       (Fintype.card I:ℝ)≤p*(n:ℝ)^2 →
  49       (∀ x,(selectionRemaining active 0 x:ℝ)≤p*n) →
  50       selectionPotential active (Real.sqrt p/4) 0 (fun _=>0)<
  51         (2:ℝ)^(Nat.floor (p*Real.sqrt p*n)+1) := by
  52   have hsqrt : 0<Real.sqrt p := Real.sqrt_pos.2 hp
  53   let c := p*Real.sqrt p/4
  54   have hc : 0<c := by dsimp [c];positivity
  55   obtain ⟨n₁,hn₁⟩ := eventually_stretchedExp_le_inv_pow p 2 hc (p:=1) (by norm_num) 1
  56   refine ⟨max 2 n₁,le_max_left _ _,?_⟩
  57   intro n hn I _ N active hcount hremaining
  58   have hn2 : 2≤n := (le_max_left _ _).trans hn
  59   have hnR : (0:ℝ)<n := by exact_mod_cast (show 0<n by omega)
  60   let s := p*Real.sqrt p*n
  61   have hs : 0<s := by dsimp [s];positivity
  62   have htail := hn₁ n ((le_max_right _ _).trans hn)
  63   simp only [Real.rpow_two,Real.rpow_one,pow_one] at htail
  64   have hinv : 1/(n:ℝ)<1 := (div_lt_one hnR).mpr (by exact_mod_cast (show 1<n by omega))
  65   have ht : p*(n:ℝ)^2*Real.exp (-s/4)<1 := by
  66     have he : -c*(n:ℝ)= -s/4 := by dsimp [c,s];ring
  67     rw [he] at htail
  68     exact htail.trans_lt hinv
  69   have hsmall : p*(n:ℝ)^2*Real.exp (s/4)<Real.exp (s/2) := by
  70     have hh := mul_lt_mul_of_pos_right ht (Real.exp_pos (s/2))
  71     rw [one_mul,mul_assoc,← Real.exp_add] at hh
  72     have he : -s/4+s/2=s/4 := by ring
  73     rwa [he] at hh
  74   have hpot := selectionPotential_zero_bound active (Real.sqrt p/4) (p*n) (p*(n:ℝ)^2)
  75     (by positivity) hcount hremaining
  76   have he : (Real.sqrt p/4)*(p*n)=s/4 := by dsimp [s];ring
  77   rw [he] at hpot
  78   exact (hpot.trans_lt hsmall).trans_le (exp_half_le_two_pow s hs.le (Nat.floor s+1)
  79     (le_of_lt (by exact_mod_cast Nat.lt_floor_add_one s)))

  81 theorem selection_candidate_scale (p n : ℝ) (hp : 0<p) (hn : 0<n) :
  82     let s := p*Real.sqrt p*n
  83     let t := Nat.ceil (4*s)
  84     0<t ∧ (p^2*n)≤(Real.sqrt p/4)*t ∧ ((Nat.floor s:ℕ):ℝ)≤s := by
  85   intro s t
  86   have hsqrt : 0<Real.sqrt p := Real.sqrt_pos.2 hp
  87   have hs : 0<s := by dsimp [s];positivity
  88   have ht : 4*s≤(t:ℝ) := Nat.le_ceil _
  89   have ht0 : 0<(t:ℝ) := (by linarith : (0:ℝ)<4*s).trans_le ht
  90   have hsq := Real.sq_sqrt hp.le
  91   have hm := mul_le_mul_of_nonneg_left ht (by positivity : (0:ℝ)≤Real.sqrt p/4)
  92   refine ⟨by exact_mod_cast ht0,?_,Nat.floor_le hs.le⟩
  93   have he : (Real.sqrt p/4)*(4*s)=p^2*n := by
  94     dsimp [s]
  95     nlinarith only [congrArg (fun z=>p*z*n) hsq]
  96   simpa only [he] using hm
