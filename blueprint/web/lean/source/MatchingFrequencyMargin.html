   7 theorem matching_frequency_margin
   8     (r actual η s n θ f error γ ε:ℝ)
   9     (hr:1≤r) (hn:0≤n) (hε:0≤ε) (hε1:ε≤1) (hγ:0≤γ) (hγε:γ≤ε)
  10     (he0:0≤error) (he:error≤ε*n) (htarget:|(1-θ)*f*n|≤n)
  11     (hbalance:η*s=(1-θ)*f)
  12     (h:|r*actual-η*s*n|≤error+γ*(|η*s*n|+error)):
  13     |actual-(1-θ)*f*n/r|≤3*ε*n :=by
  14   have h₁:=matching_frequency_error r actual η s n θ f error γ (by linarith) hbalance h
  15   have ht0:0≤|(1-θ)*f*n|:=abs_nonneg _
  16   have hfirst:γ*(|(1-θ)*f*n|+error)≤ε*(n+ε*n):=by
  17     apply le_trans (mul_le_mul_of_nonneg_left (add_le_add htarget he) hγ)
  18     apply mul_le_mul_of_nonneg_right hγε
  19     positivity
  20   have hprod:ε*(ε*n)≤ε*n:=by
  21     have hh:=mul_le_mul_of_nonneg_right hε1 (mul_nonneg hε hn)
  22     nlinarith
  23   have hnum:error+γ*(|(1-θ)*f*n|+error)≤3*ε*n:=by nlinarith
  24   have hdiv:(error+γ*(|(1-θ)*f*n|+error))/r≤3*ε*n:=by
  25     apply (div_le_iff₀ (by linarith : 0<r)).2
  26     have hh:=mul_le_mul_of_nonneg_left hr (show 0≤3*ε*n by positivity)
  27     nlinarith
  28   exact h₁.trans hdiv
