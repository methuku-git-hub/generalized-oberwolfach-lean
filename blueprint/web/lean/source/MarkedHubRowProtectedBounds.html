  29     have hsub:({W.1.val i,twistedTupleHead π W.1.val i}:Finset V)⊆I:=by

  62     (hMrows:∀ W∈M,markedWheelRows W⊆F)
  63     (hFI:Disjoint F I) (hI:∀ x,π x∈I↔x∈I) (hYJ:∀ i,Disjoint (Y i) J):
  64     ((M∪Finset.univ.biUnion C).biUnion markedWheelPhysicalResources).filter (markedInnerResource I J)=∅:=by

  66   intro e he
