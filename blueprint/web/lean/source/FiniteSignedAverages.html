  11 def finiteSignedAverage {I W : Type*} [Fintype I] (a : I → W → ℝ) (w : W) : ℝ :=
  12   (∑ i,a i w)/(Fintype.card I:ℝ)

  14 theorem finiteSignedAverage_test {I W : Type*} [Fintype I] [Fintype W]
  15     (a : I → W → ℝ) (t : W → ℝ) :
  16     (∑ w,finiteSignedAverage a w*t w)=
  17       (∑ i,∑ w,a i w*t w)/(Fintype.card I:ℝ) := by
  18   unfold finiteSignedAverage
  19   calc
  20     _ = ∑ w,∑ i,(a i w*t w)/(Fintype.card I:ℝ) := by
  21       apply Finset.sum_congr rfl
  22       intro w _
  23       rw [← Finset.sum_div,← Finset.sum_mul]
  24       ring
  25     _ = ∑ i,∑ w,(a i w*t w)/(Fintype.card I:ℝ) := Finset.sum_comm
  26     _ = _ := by simp only [Finset.sum_div]

  28 theorem finiteSignedAverage_constant_test {I W : Type*} [Fintype I] [Fintype W]
  29     (hI : 0<Fintype.card I) (a : I → W → ℝ) (t : W → ℝ) (K : ℝ)
  30     (ht : ∀ i,(∑ w,a i w*t w)=K) :
  31     (∑ w,finiteSignedAverage a w*t w)=K := by
  32   rw [finiteSignedAverage_test]
