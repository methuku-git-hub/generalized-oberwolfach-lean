  17 theorem hypergraphColorPartition_exists
  18     (R : E → Finset V) (k : ℕ) (hk : 2≤k) (hR : ∀ e,(R e).card=k)
  19     (hdegree : ∀ v,0<Fintype.card (HypergraphEdgesAt R v))
  20     (D : ℝ) (hD : 0≤D)
  21     (hcodegree : ∀ v u,u≠v → ((Finset.univ.filter (fun e=>v∈R e ∧ u∈R e)).card:ℝ)≤D)
  22     (w : J → E → ℝ) (hw : ∀ j e,0≤w j e) (hW : ∀ j,0<∑ e,w j e)
  23     (L : J → ℝ) (hL : ∀ j v,hypergraphWeightedDegree R (w j) v≤L j)
  24     (t : V → ℝ) (s : J → ℝ) (ht : ∀ v,0≤t v) (hs : ∀ j,0≤s j)
  25     (hsmall :
  26       (∑ v : V,∑ _c : C,2*Real.exp (-2*(t v)^2/(D*((k:ℝ)-1)*Fintype.card (HypergraphEdgesAt R v))))+
  27       (∑ j : J,∑ _c : C,2*Real.exp (-2*(s j)^2/(L j*k*∑ e,w j e)))<1) :
  28     ∃ σ : V → C,
  29       (∀ v c,|monochromaticWeightedSum (hypergraphRootRemainder R v) (fun _=>1) c σ-
  30         (Fintype.card (HypergraphEdgesAt R v):ℝ)/(Fintype.card C:ℝ)^(k-1)|<t v) ∧
  31       (∀ j c,|(∑ e∈hypergraphColorClass R σ c,w j e)-(∑ e,w j e)/(Fintype.card C:ℝ)^k|<s j) := by
  32   let μ := FiniteLaw.pi (fun _ : V=>FiniteLaw.uniform C)
  33   let A : (V×C)⊕(J×C) → (V → C) → Prop := Sum.elim
  34     (fun vc σ=>t vc.1≤|monochromaticWeightedSum (hypergraphRootRemainder R vc.1) (fun _=>1) vc.2 σ-
  35       (Fintype.card (HypergraphEdgesAt R vc.1):ℝ)/(Fintype.card C:ℝ)^(k-1)|)
  36     (fun jc σ=>s jc.1≤|monochromaticWeightedSum R (w jc.1) jc.2 σ-
  37       (∑ e,w jc.1 e)/(Fintype.card C:ℝ)^k|)
  38   let p : (V×C)⊕(J×C) → ℝ := Sum.elim
  39     (fun vc=>2*Real.exp (-2*(t vc.1)^2/(D*((k:ℝ)-1)*Fintype.card (HypergraphEdgesAt R vc.1))))
  40     (fun jc=>2*Real.exp (-2*(s jc.1)^2/(L jc.1*k*∑ e,w jc.1 e)))
  41   have hb (a : (V×C)⊕(J×C)) (_ha : a∈Finset.univ) : μ.probability (A a)≤p a := by
  42     cases a with
  43     | inl vc =>
  44       rcases vc with ⟨v,c⟩
  45       exact hypergraphRootColor_tail R k hk hR v (hdegree v) D hD (hcodegree v) c (t v) (ht v)
  46     | inr jc =>
  47       rcases jc with ⟨j,c⟩
  48       exact monochromaticWeightedSum_tail R (w j) (hw j) k (by omega) hR (hW j) (L j) (hL j) c (s j) (hs j)
  49   have hp : (∑ a,p a)<1 := by
  50     simpa only [Fintype.sum_sum_type,Fintype.sum_prod_type,p,Sum.elim_inl,Sum.elim_inr] using hsmall
  51   obtain ⟨σ,_hσ,hgood⟩ := finiteLaw_common_event_exists μ Finset.univ A p hb hp
  52   refine ⟨σ,?_,?_⟩
  53   · intro v c
  54     exact lt_of_not_ge (hgood (Sum.inl (v,c)) (Finset.mem_univ _))
  55   · intro j c
  56     have hh := lt_of_not_ge (hgood (Sum.inr (j,c)) (Finset.mem_univ _))
  57     rw [hypergraphColorClass_weight]
  58     exact hh
