  20 theorem three_injective_coin_maps_influence {E:Type} [DecidableEq E]
  21     (f g h:V→E) (hf:Function.Injective f) (hg:Function.Injective g) (hh:Function.Injective h) (e:E):
  22     (Finset.univ.filter (fun v=>e∈({f v,g v,h v}:Finset E))).card≤3:=by
  23   have h1 (f:V→E) (hf:Function.Injective f):(Finset.univ.filter (fun v=>e=f v)).card≤1:=by
  24     apply Finset.card_le_one.mpr
  25     intro v hv w hw
  26     exact hf ((Finset.mem_filter.mp hv).2.symm.trans (Finset.mem_filter.mp hw).2)
  27   have he:Finset.univ.filter (fun v=>e∈({f v,g v,h v}:Finset E))=
  28       (Finset.univ.filter (fun v=>e=f v))∪((Finset.univ.filter (fun v=>e=g v))∪(Finset.univ.filter (fun v=>e=h v))):=by
  29     ext v;simp
  30   rw [he]
  31   have ha:=Finset.card_union_le (Finset.univ.filter (fun v=>e=f v))
  32     ((Finset.univ.filter (fun v=>e=g v))∪(Finset.univ.filter (fun v=>e=h v)))
  33   have hb:=Finset.card_union_le (Finset.univ.filter (fun v=>e=g v)) (Finset.univ.filter (fun v=>e=h v))
  34   have hc:=h1 f hf;have hd:=h1 g hg;have hi:=h1 h hh
  35   omega
