  30 theorem ordinaryRowResources_incidence (S:Finset V) (J:Finset U) (C:Finset V)
  31     (e:Sym2 V⊕(V×U)):(C.filter (fun z=>e∈ordinaryRowResources S J z)).card≤2:=by
  32   cases e with
  33   | inl e=>
  34     induction e using Sym2.ind with
  35     | _ a b=>
  36       have hsub:C.filter (fun z=>Sum.inl s(a,b)∈ordinaryRowResources S J z)⊆{a,b}:=by
  37         intro z hz
  38         obtain ⟨x,hx,he⟩:=(ordinaryRowResources_rim S J z s(a,b)).mp (Finset.mem_filter.mp hz).2
  39         have hh:=Sym2.eq_iff.mp he
  40         simp only [Finset.mem_insert,Finset.mem_singleton]
  41         rcases hh with h|h <;> tauto
  42       exact (Finset.card_le_card hsub).trans Finset.card_le_two
  43   | inr a=>
  44     have hsub:C.filter (fun z=>Sum.inr a∈ordinaryRowResources S J z)⊆{a.1}:=by
  45       intro z hz
  46       exact Finset.mem_singleton.mpr ((ordinaryRowResources_spoke S J z a).mp (Finset.mem_filter.mp hz).2).1.symm
  47     have hh:=Finset.card_le_card hsub
  48     simp only [Finset.card_singleton] at hh
  49     omega
