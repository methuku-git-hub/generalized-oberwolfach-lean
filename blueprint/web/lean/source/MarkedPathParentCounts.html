  16     (hdeg₀ : ∀ x : V,(1/2+α/2)*T₀.card≤((P.neighborFinset x∩T₀).card:ℝ))
  17     (hdegK : ∀ x : V,(1/2+α/2)*TK.card≤((P.neighborFinset x∩TK.image π).card:ℝ)) :
  18     a^(r+4)≤(Fintype.card {z : Fin (r+4)→V // DenseRootMarkedPath P π b j T₀ TK v v' z}:ℝ) := by
  19   have hh := marked_path_choice_count P π b (r+4) j T₀ TK v v' α a hα hα1 ha
  20     (by simpa only [Nat.cast_add,Nat.cast_ofNat,add_assoc,show (4:ℝ)+2=6 by norm_num] using hbudget) hdeg₀ hdegK
  21   let I := {f : Fin (r+4)→V //
  22     MixedGreedyGood P π b (markedPathChoiceClass j T₀ TK) (markedPathChoiceHead π j)
  23       (markedPathChoiceRequests π j (r+4) v v') {v,v'} f}
  24   let J := {z : Fin (r+4)→V // DenseRootMarkedPath P π b j T₀ TK v v' z}
  25   let inj : I→J := fun f=>⟨fun i=>f.val (pathChoiceOrder (r+2) i),
  26     markedPathChoice_to_parent P π b r hr j T₀ TK v v' f.val f.property⟩
  27   have hi : Function.Injective inj := by
  28     intro f g he
  29     apply Subtype.ext
  30     funext i
  31     have hh := congrArg (fun z : J=>z.val ((pathChoiceOrder (r+2)).symm i)) he
  32     simpa only [inj,Equiv.apply_symm_apply] using hh
  33   exact hh.trans (by exact_mod_cast Fintype.card_le_of_injective inj hi)

  44       (Fintype.card {z : Fin (r+4)→V // DenseRootMarkedPath P π b j T₀ TK v v' z}:ℝ) := by
  45   obtain ⟨N,hN⟩ := exists_nat_ge ((((r:ℝ)+6)*(2*(b:ℝ)+1))/(α*θ/4))
  46   refine ⟨N,?_⟩
