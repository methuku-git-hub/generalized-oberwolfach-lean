  18 theorem decompositionAt_transfer {S T : Set ℕ} {α β : ℝ} {n : ℕ}
  19     (hST : S ⊆ T) (hdeg : allowedThreshold T + β ≤ allowedThreshold S + α)
  20     (hT : DecompositionAt T β n) : DecompositionAt S α n := by
  21   classical
  22   intro D hD hbound F hF G hG
  23   apply hT D hD (le_trans ?_ hbound) F ?_ G hG
  24   · exact mul_le_mul_of_nonneg_right hdeg (Nat.cast_nonneg n)
  25   · intro i
  26     exact ⟨(hF i).1, allowed_lengths_mono hST (hF i).2⟩

  28 /-- A finite exceptional set and a common tail, chosen using only alpha. -/
  29 def SmallThresholdTail : Prop :=
  30   ∀ α : ℝ, 0 < α → ∃ E : Finset ℕ, ∃ T : Set ℕ,
  31     T.Nonempty ∧ (∀ r ∈ E, 3 ≤ r) ∧ (∀ r ∈ T, 3 ≤ r) ∧
  32     (∀ r, 3 ≤ r → r ∉ E → r ∈ T) ∧
  33     (∀ r ∈ T, cycleThreshold r ≤ (1 / 2 : ℝ) + α / 2)

  35 /-- The fixed-set theorem and the explicitly stated threshold estimates imply
  36 uniformity of the order threshold over ALL allowed length sets. -/
  37 theorem uniform_of_fixed_set
  38     (hfixed : FixedSetTheorem)
  39     (hlower : ∀ r : ℕ, 3 ≤ r → (1 / 2 : ℝ) ≤ cycleThreshold r)
  40     (hupper : ∀ r : ℕ, 3 ≤ r → cycleThreshold r ≤ 1)
  41     (htail : SmallThresholdTail) : MainTheorem := by
  42   classical
  43   intro α hα
  44   obtain ⟨E, T, hTne, hE, hT, hcover, hsmall⟩ := htail α hα
  45   have hhalf : 0 < α / 2 := by linarith
  46   have hf : ∀ B : {B : Finset ℕ // B ⊆ E},
  47       ∃ N : ℕ, ∀ n ≥ N, DecompositionAt (T ∪ (B.val : Set ℕ)) (α / 2) n := by
  48     intro B
  49     apply hfixed (T ∪ (B.val : Set ℕ))
  50     · exact hTne.mono Set.subset_union_left
  51     · intro r hr
  52       rcases hr with hr | hr
  53       · exact hT r hr
  54       · exact hE r (B.property hr)
  55     · exact hhalf
  56   choose bound hbound using hf
  57   let bs : Finset {B : Finset ℕ // B ⊆ E} := E.powerset.attach.map
  58     ⟨fun B => ⟨B.val, Finset.mem_powerset.mp B.property⟩, by
  59       intro B C h; apply Subtype.ext
  60       exact congrArg (fun z : {B : Finset ℕ // B ⊆ E} => z.val) h⟩
  61   let N := bs.sup bound
  62   refine ⟨N, ?_⟩
  63   intro S hSne hS n hn
  64   let B : {B : Finset ℕ // B ⊆ E} := ⟨E.filter (fun r => r ∈ S), Finset.filter_subset _ _⟩
  65   have hBmem : B ∈ bs := by
  66     apply Finset.mem_map.mpr
  67     refine ⟨⟨B.val, Finset.mem_powerset.mpr B.property⟩, by simp, ?_⟩
  68     rfl
  69   have hBN : bound B ≤ N := Finset.le_sup hBmem
  70   let Sstar := T ∪ (B.val : Set ℕ)
  71   have hSstar : S ⊆ Sstar := by
  72     intro r hr
  73     by_cases hrE : r ∈ E
  74     · exact Or.inr (Finset.mem_filter.mpr ⟨hrE, hr⟩)
  75     · exact Or.inl (hcover r (hS r hr) hrE)
  76   have hbdd : BddAbove (cycleThreshold '' S) := by
  77     refine ⟨1, ?_⟩
  78     rintro y ⟨r, hr, rfl⟩
  79     exact hupper r (hS r hr)
  80   have hle : ∀ r ∈ S, cycleThreshold r ≤ allowedThreshold S := by
  81     intro r hr
  82     exact le_csSup hbdd ⟨r, hr, rfl⟩
  83   have hbase : (1 / 2 : ℝ) ≤ allowedThreshold S := by
  84     obtain ⟨r, hr⟩ := hSne
  85     exact (hlower r (hS r hr)).trans (hle r hr)
  86   have hsup : allowedThreshold Sstar ≤ allowedThreshold S + α / 2 := by
  87     apply csSup_le (hTne.mono Set.subset_union_left |>.image cycleThreshold)
  88     rintro y ⟨r, hr, rfl⟩
  89     rcases hr with hr | hr
  90     · linarith [hsmall r hr]
  91     · have hrS : r ∈ S := (Finset.mem_filter.mp hr).2
  92       linarith [hle r hrS]
  93   apply decompositionAt_transfer hSstar (show allowedThreshold Sstar + α / 2 ≤
  94       allowedThreshold S + α by linarith)
  95   exact hbound B n (hBN.trans hn)

  97 /-- A cofinite upper bound supplies the finite exceptional-set interface. -/
  98 theorem smallThresholdTail_of_cofinite
  99     (ht : ∀ α : ℝ, 0 < α → ∃ R : ℕ, ∀ r ≥ R,
 100       cycleThreshold r ≤ (1 / 2 : ℝ) + α / 2) : SmallThresholdTail := by
 101   classical
 102   intro α hα
 103   obtain ⟨R, hR⟩ := ht α hα
 104   let E := (Finset.range R).filter (fun r => 3 ≤ r)
 105   let T : Set ℕ := {r | 3 ≤ r ∧ R ≤ r}
 106   refine ⟨E, T, ⟨max 3 R, le_max_left _ _, le_max_right _ _⟩, ?_, ?_, ?_, ?_⟩
 107   · intro r hr; exact (Finset.mem_filter.mp hr).2
 108   · intro r hr; exact hr.1
 109   · intro r hr hnot
 110     refine ⟨hr, ?_⟩
 111     by_contra hge
 112     have : r ∈ E := Finset.mem_filter.mpr ⟨Finset.mem_range.mpr (by omega), hr⟩
 113     exact hnot this
 114   · intro r hr; exact hR r hr.2
