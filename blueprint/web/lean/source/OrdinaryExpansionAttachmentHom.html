  13 def attachmentCollapse (z : W) (ρ : I → V) : V ⊕ (I×W) → V ⊕ (I×W)
  14   | .inl a => .inl a
  15   | .inr (i,w) => attachmentWheelEmbedding z ρ i w

  17 def expansionAttachmentRoots (x : E → V) (v : J → V) : E ⊕ J → V := Sum.elim x v

  19 theorem attachmentCollapse_private_preimage (z : W) (ρ : I → V)
  20     (a : V ⊕ (I×W)) (i : I) (w : W) :
  21     attachmentCollapse z ρ a=.inr (i,w) ↔ a=.inr (i,w) ∧ w≠z := by
  22   cases a with
  23   | inl a => simp [attachmentCollapse]
  24   | inr p =>
  25     rcases p with ⟨j,w'⟩
  26     rw [attachmentCollapse,attachmentWheelEmbedding_private]
  27     simp only [Sum.inr.injEq,Prod.mk.injEq]
  28     aesop

  45 theorem ordinaryExpansion_attachment_rim_hom (G : SimpleGraph V) (C : SimpleGraph W) (z o : W)
  46     (hzo : C.Adj z o) (x y : E → V) (v : J → V) (hG : ∀ e,G.Adj (x e) (y e))
  47     (a b : ExpansionRow V E J W) (h : (ordinaryExpansionRim C z o x y).Adj a b) :
  48     (ordinaryAttachmentRim G C z (expansionAttachmentRoots x v)).Adj
  49       (attachmentCollapse z (expansionAttachmentRoots x v) a)
  50       (attachmentCollapse z (expansionAttachmentRoots x v) b) := by
  51   have hcross (a : V) (p : (E ⊕ J) × W) (h : expansionCrossAdj z o x y a p) :
  52       (ordinaryAttachmentRim G C z (expansionAttachmentRoots x v)).Adj
  53         (.inl a) (attachmentCollapse z (expansionAttachmentRoots x v) (.inr p)) := by
  54     rcases p with ⟨i,w⟩
  55     cases i with
  56     | inr j => exact h.elim
  57     | inl e =>
  58       rcases h with ⟨ha,hw⟩|⟨ha,hw⟩
  59       · subst a;subst w
  60         simp [attachmentCollapse,attachmentWheelEmbedding,ordinaryAttachmentRim,expansionAttachmentRoots,hzo.ne.symm,hzo]
  61       · subst a;subst w
  62         simpa [attachmentCollapse,attachmentWheelEmbedding,ordinaryAttachmentRim,expansionAttachmentRoots] using (hG e).symm
  63   cases a with
  64   | inl a =>
  65     cases b with
  66     | inl b => exact h.elim
  67     | inr p => exact hcross a p h
  68   | inr p =>
  69     cases b with
  70     | inl b => exact (hcross b p h).symm
  71     | inr q =>
  72       rcases p with ⟨i,w⟩;rcases q with ⟨j,w'⟩
  73       obtain ⟨hij,h⟩ := h
  74       subst j
  75       have hC := expansionPrivateAdj_adj C z o i w w' h
  76       by_cases hw : w=z <;> by_cases hw' : w'=z
  77       · exact (hC.ne (hw.trans hw'.symm)).elim
  78       · subst w
  79         simp [attachmentCollapse,attachmentWheelEmbedding,ordinaryAttachmentRim,hw',hC]
  80       · subst w'
  81         simp [attachmentCollapse,attachmentWheelEmbedding,ordinaryAttachmentRim,hw,hC.symm]
  82       · simp [attachmentCollapse,attachmentWheelEmbedding,ordinaryAttachmentRim,hw,hw',hC]
