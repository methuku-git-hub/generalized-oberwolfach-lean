   1 import Oberwolfach.FiniteHighEventsCommon
   2 /-! Split an ordered source path at its prescribed vertices, retaining the
   3 prescribed boundary vertices of every intervening free run. Empty runs may
   4 be discarded after the prescribed edges have been reserved. -/
   5 set_option autoImplicit false
   6 noncomputable section
   7 namespace Oberwolfach
   8 attribute [local instance] Classical.propDecidable
   9 
  10 structure PrescribedPathSegment (X:Type) where
  11   left:Option X
  12   free:List X
  13   right:Option X
  14 
  15 variable {X:Type} [DecidableEq X]
  16 
  17 def prescribedPathSegments (A:Finset X):Option X→List X→List X→List (PrescribedPathSegment X)
  18   | left,acc,[]=>[⟨left,acc.reverse,none⟩]
  19   | left,acc,x::xs=>
  20     if x∈A then ⟨left,acc.reverse,some x⟩::prescribedPathSegments A (some x) [] xs
  21     else prescribedPathSegments A left (x::acc) xs

  23 theorem prescribed_path_segments_free_vertices (A:Finset X):
  24     ∀ (xs:List X) (left:Option X) (acc:List X),
  25       (prescribedPathSegments A left acc xs).flatMap PrescribedPathSegment.free =
  26         acc.reverse++xs.filter (fun x=>x∉A) := by
  27   intro xs
  28   induction xs with
  29   | nil => intro left acc;simp [prescribedPathSegments]
  30   | cons x xs ih =>
  31     intro left acc
  32     by_cases hx:x∈A
  33     · simp [prescribedPathSegments,hx,ih]
  34     · simp [prescribedPathSegments,hx,ih,List.reverse_cons,List.append_assoc]
  35 theorem prescribed_path_segment_free_mem (A:Finset X) (xs:List X)
  36     (seg:PrescribedPathSegment X) (hseg:seg∈prescribedPathSegments A none [] xs):
  37     ∀ x∈seg.free,x∈xs ∧ x∉A := by
  38   intro x hx
  39   have hm:x∈(prescribedPathSegments A none [] xs).flatMap PrescribedPathSegment.free:=by
  40     exact List.mem_flatMap.mpr ⟨seg,hseg,hx⟩
  41   simpa [prescribed_path_segments_free_vertices] using hm

  43 theorem prescribed_path_segment_boundaries (A:Finset X):
  44     ∀ (xs:List X) (left:Option X) (acc:List X),
  45       (∀ y,left=some y→y∈A) →
  46       ∀ seg∈prescribedPathSegments A left acc xs,
  47         (∀ y,seg.left=some y→y∈A) ∧ (∀ y,seg.right=some y→y∈A) := by
  48   intro xs
  49   induction xs with
  50   | nil =>
  51     intro left acc hl seg hs
  52     simp only [prescribedPathSegments,List.mem_singleton] at hs
  53     subst seg
  54     exact ⟨hl,by simp⟩
  55   | cons x xs ih =>
  56     intro left acc hl seg hs
  57     by_cases hx:x∈A
  58     · simp only [prescribedPathSegments,hx,ite_true,List.mem_cons] at hs
  59       rcases hs with hs|hs
  60       · subst seg
  61         constructor
  62         · exact hl
  63         · intro y hy
  64           cases hy
  65           exact hx
  66       · exact ih (some x) [] (by intro y hy;cases hy;exact hx) seg hs
  67     · simp only [prescribedPathSegments,hx,ite_false] at hs
  68       exact ih left (x::acc) hl seg hs
  69 theorem prescribed_path_segment_free_nodup (A:Finset X) (xs:List X) (hxs:xs.Nodup)
  70     (seg:PrescribedPathSegment X) (hseg:seg∈prescribedPathSegments A none [] xs):
  71     seg.free.Nodup := by
  72   have hf:(xs.filter (fun x=>x∉A)).Nodup:=hxs.filter _
  73   have hflat:(prescribedPathSegments A none [] xs).flatMap PrescribedPathSegment.free |>.Nodup:=by
  74     simpa [prescribed_path_segments_free_vertices] using hf
  75   exact (List.nodup_flatMap.mp hflat).1 seg hseg

  77 theorem prescribed_path_segment_boundaries_distinct (A:Finset X):
  78     ∀ (xs:List X) (left:Option X) (acc:List X),xs.Nodup→
  79       (∀ y,left=some y→y∉xs) →
  80       ∀ seg∈prescribedPathSegments A left acc xs,
  81         ∀ a b,seg.left=some a→seg.right=some b→a≠b := by
  82   intro xs
  83   induction xs with
  84   | nil =>
  85     intro left acc hn hl seg hs a b ha hb
  86     simp only [prescribedPathSegments,List.mem_singleton] at hs
  87     subst seg
  88     simp at hb
  89   | cons x xs ih =>
  90     intro left acc hn hl seg hs a b ha hb
  91     have hnodup:xs.Nodup:=(List.nodup_cons.mp hn).2
  92     have hxnot:x∉xs:=(List.nodup_cons.mp hn).1
  93     by_cases hx:x∈A
  94     · simp only [prescribedPathSegments,hx,ite_true,List.mem_cons] at hs
  95       rcases hs with hs|hs
  96       · subst seg
  97         simp only at ha hb
  98         cases hb
  99         intro he
 100         subst a
 101         exact hl x ha (by simp)
 102       · exact ih (some x) [] hnodup (by intro y hy;cases hy;exact hxnot) seg hs a b ha hb
 103     · simp only [prescribedPathSegments,hx,ite_false] at hs
 104       apply ih left (x::acc) hnodup (by intro y hy hym;exact hl y hy (by simp [hym])) seg hs a b ha hb
