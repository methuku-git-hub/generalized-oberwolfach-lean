   1 import Oberwolfach.RegularCycleIsomorphism
   2 /-! Glue maps on distinct factor components when their vertex images are
   3 pairwise disjoint. The source domain is the disjoint union of those actual
   4 components. -/
   5 set_option autoImplicit false
   6 noncomputable section
   7 namespace Oberwolfach
   8 attribute [local instance] Classical.propDecidable
   9 variable {n:ℕ} {A:Type}
  10 
  11 theorem component_family_embeddings_glue
  12     (F:SimpleGraph (Fin n)) (C:A→F.ConnectedComponent)
  13     (hC:Function.Injective C) (f:∀a,C a↪Fin n)
  14     (hdis:∀a b,a≠b→Disjoint (Finset.univ.image (f a)) (Finset.univ.image (f b))):
  15     ∃src dst:(Σa,C a)↪Fin n,
  16       (∀z,src z=z.2.val) ∧ (∀z,dst z=f z.1 z.2) :=by
  17   let src:(Σa,C a)↪Fin n:=⟨fun z=>z.2.val,by
  18     rintro ⟨a,x⟩ ⟨b,y⟩ heq
  19     change x.val=y.val at heq
  20     have hy:x.val∈(C b).supp:=by rw [heq];exact y.property
  21     have hc:C a=C b:=SimpleGraph.ConnectedComponent.eq_of_common_vertex x.property hy
  22     have hab:a=b:=hC hc
  23     subst b
  24     have hxy:x=y:=Subtype.ext heq
  25     subst y
  26     rfl⟩
  27   let dst:(Σa,C a)↪Fin n:=⟨fun z=>f z.1 z.2,by
  28     rintro ⟨a,x⟩ ⟨b,y⟩ heq
  29     by_cases hab:a=b
  30     · subst b
  31       have hxy:x=y:=(f a).injective heq
  32       subst y
  33       rfl
  34     · have ha:f a x∈Finset.univ.image (f a):=Finset.mem_image.mpr ⟨x,Finset.mem_univ _,rfl⟩
  35       have hb:f a x∈Finset.univ.image (f b):=Finset.mem_image.mpr ⟨y,Finset.mem_univ _,heq.symm⟩
  36       exact ((Finset.disjoint_left.mp (hdis a b hab)) ha hb).elim⟩
  37   exact ⟨src,dst,fun _=>rfl,fun _=>rfl⟩
  38 end Oberwolfach
