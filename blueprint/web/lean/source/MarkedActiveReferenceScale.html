  11     (a b:I→ℝ) (ha:∀ i,0≤a i) (hb:∀ i,0<b i):
  12     ∃ k:ℝ,0<k∧∀ x:ℝ,0≤x→x≤k→∀ i,a i*x≤b i:=by
  13   classical
  14   let C:ℝ:=1+∑ i,a i/b i
  15   have hsum:0≤∑ i,a i/b i:=Finset.sum_nonneg (fun i _=>div_nonneg (ha i) (hb i).le)
  16   have hC:0<C:=by dsimp [C];linarith
  17   refine ⟨1/C,by positivity,?_⟩
  18   intro x hx hxk i
  19   have hi:a i/b i≤∑ j,a j/b j:=Finset.single_le_sum
  20     (fun j _=>div_nonneg (ha j) (hb j).le) (Finset.mem_univ i)
  21   have hiC:a i/b i≤C:=by dsimp [C];linarith
  22   have hi':a i≤C*b i:=(div_le_iff₀ (hb i)).mp hiC
  23   have hix:=mul_le_mul_of_nonneg_left hxk (ha i)
  24   apply hix.trans
  25   rw [mul_one_div]
  26   apply (div_le_iff₀ hC).mpr
  27   nlinarith only [hi']
