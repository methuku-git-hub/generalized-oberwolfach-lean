  13 theorem vortex_uniform_subset_step {ε : ℝ} (he : 0 < ε) (he1 : ε ≤ 1) :
  14     ∃ N : ℕ, ∀ (V : Type u) [Fintype V], N ≤ Fintype.card V →
  15       ∀ (J : Type v) (tests : Finset J) (A : J → Finset V),
  16       (tests.card : ℝ) ≤ 2 * Fintype.card V / ε →
  17       ∃ S : FixedSubset V (Nat.floor (ε * Fintype.card V)), ∀ j ∈ tests,
  18         |((A j ∩ S.1).card : ℝ) / (Nat.floor (ε * Fintype.card V) : ℕ) -
  19           ((A j).card : ℝ) / Fintype.card V| <
  20           (Fintype.card V : ℝ) ^ (-1 / 3 : ℝ) := by
  21   obtain ⟨N, hN⟩ := eventually_vortex_step_margins he
  22   refine ⟨N, ?_⟩
  23   intro V _ hn J tests A htests
  24   obtain ⟨hn2, hlarge, hsmall⟩ := hN (Fintype.card V) hn
  25   obtain ⟨hrpos, hr, hrlo⟩ := vortex_floor_step_bounds he1 hlarge
  26   apply fixedSubset_normalized_counts_exists hr hrpos tests A (by positivity)
  27   have hexp := vortex_normalized_exponent (by omega : 0 < Fintype.card V) hrlo
  28   have hcoef : 2 * (tests.card : ℝ) ≤ 4 * Fintype.card V / ε := by
  29     calc
  30       _ ≤ 2 * (2 * Fintype.card V / ε) := mul_le_mul_of_nonneg_left htests (by norm_num)
  31       _ = _ := by ring
  32   calc
  33     _ ≤ (4 * Fintype.card V / ε) *
  34         Real.exp (-ε * (Fintype.card V : ℝ) ^ (1 / 3 : ℝ)) :=
  35       mul_le_mul hcoef (Real.exp_le_exp.mpr hexp) (Real.exp_pos _).le (by positivity)
  36     _ < 1 := hsmall
