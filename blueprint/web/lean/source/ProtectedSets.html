  10 def protectedInducedUnion {V : Type*} {k : ℕ} (G : SimpleGraph V)
  11     (I : Fin k → Finset V) : SimpleGraph V where
  12   Adj x y := G.Adj x y ∧ ∃ a, x ∈ I a ∧ y ∈ I a
  13   symm := by constructor; intro x y h; obtain ⟨a, hx, hy⟩ := h.2; exact ⟨h.1.symm, a, hy, hx⟩
  14   loopless := by constructor; intro x h; exact (G.ne_of_adj h.1) rfl

  16 theorem protectedInducedUnion_degree {V : Type*} [Fintype V] {k : ℕ}
  17     (G : SimpleGraph V) (I : Fin k → Finset V) (x : V) :
  18     (protectedInducedUnion G I).degree x ≤ ∑ a, (I a).card := by
  19   have hs : (protectedInducedUnion G I).neighborFinset x ⊆ Finset.univ.biUnion I := by
  20     intro y hy
  21     have hh : (protectedInducedUnion G I).Adj x y := by simpa using hy
  22     obtain ⟨_, a, _, ha⟩ := hh
  23     exact Finset.mem_biUnion.mpr ⟨a, Finset.mem_univ a, ha⟩
  24   exact (Finset.card_le_card hs).trans (Finset.card_biUnion_le)

  26 theorem graphFromFiniteEdges_degree {V : Type*} [Fintype V]
  27     (A : Finset (Sym2 V)) (x : V) :
  28     (SimpleGraph.fromEdgeSet (A : Set (Sym2 V))).degree x ≤ A.card := by
  29   let H := SimpleGraph.fromEdgeSet (A : Set (Sym2 V))
  30   have hs : H.edgeFinset ⊆ A := by
  31     intro e he
  32     have hh := (SimpleGraph.mem_edgeFinset.mp he)
  33     rw [SimpleGraph.edgeSet_fromEdgeSet] at hh
  34     exact hh.1
  35   exact (H.degree_le_card_edgeFinset x).trans (Finset.card_le_card hs)

  37 theorem positive_common_lower_bound {k : ℕ} (hk : 0 < k) (ρ : Fin k → ℝ)
  38     (hρ : ∀ a, 0 < ρ a) : ∃ r : ℝ, 0 < r ∧ ∀ a, r ≤ ρ a := by
  39   have hu : (Finset.univ : Finset (Fin k)).Nonempty := ⟨⟨0, hk⟩, Finset.mem_univ _⟩
