  21 theorem markedPackingResources_embedding_graphs
  22     (f:V↪V') (g:U↪U')
  23     (M:Finset (MarkedWheel tail head ktail khead m π b H₀ HK))
  24     (N:Finset (MarkedWheel tail' head' ktail' khead' m' π' b' H₀' HK'))
  25     (he:N.biUnion markedWheelPhysicalResources=(M.biUnion markedWheelPhysicalResources).image (markedResourceEmbedding f g)):
  26     markedPackingRimGraph N=(markedPackingRimGraph M).map f∧
  27       (∀ x u,markedPackingSpokes N x u↔∃ v w,f v=x∧g w=u∧markedPackingSpokes M v w):=by
  28   have hr:markedPackingRims N=(markedPackingRims M).image (Finset.image f):=by
  29     ext e
  30     rw [←markedPackingResources_rim,he]
  31     simp only [Finset.mem_image]
  32     constructor
  33     · rintro ⟨a,ha,he⟩
  34       cases a with
  35       | inl a=>exact ⟨a,(markedPackingResources_rim M a).mp ha,Sum.inl.inj he⟩
  36       | inr a=>exact (Sum.inr_ne_inl he).elim
  37     · rintro ⟨a,ha,he⟩
  38       exact ⟨Sum.inl a,(markedPackingResources_rim M a).mpr ha,congrArg Sum.inl he⟩
  39   constructor
  40   · unfold markedPackingRimGraph
  41     rw [hr,physicalEdgeGraph_image_map]
  42   · intro x u
  43     rw [←markedPackingResources_spoke,he]
  44     simp only [Finset.mem_image]
  45     constructor
  46     · rintro ⟨a,ha,he⟩
  47       cases a with
  48       | inl a=>exact (Sum.inl_ne_inr he).elim
  49       | inr a=>
  50         have h: (f a.1,g a.2)=(x,u):=Sum.inr.inj he
  51         exact ⟨a.1,a.2,congrArg Prod.fst h,congrArg Prod.snd h,(markedPackingResources_spoke M _ _).mp ha⟩
  52     · rintro ⟨v,w,hv,hw,hM⟩
  53       exact ⟨Sum.inr (v,w),(markedPackingResources_spoke M v w).mpr hM,by
  54         change Sum.inr (f v,g w)=Sum.inr (x,u)
  55         rw [hv,hw]⟩

  57 theorem markedPackingResources_embedding_rim_degree
  58     (f:V↪V') (g:U↪U')
  59     (M:Finset (MarkedWheel tail head ktail khead m π b H₀ HK))
  60     (N:Finset (MarkedWheel tail' head' ktail' khead' m' π' b' H₀' HK'))
  61     (he:N.biUnion markedWheelPhysicalResources=(M.biUnion markedWheelPhysicalResources).image (markedResourceEmbedding f g))
  62     (v:V):
  63     (markedPackingRimGraph N).degree (f v)=(markedPackingRimGraph M).degree v:=by
  64   simp only [←SimpleGraph.ncard_neighborSet]
  65   rw [(markedPackingResources_embedding_graphs f g M N he).1,SimpleGraph.neighborSet_map,
  66     Set.ncard_image_of_injective _ f.injective]
