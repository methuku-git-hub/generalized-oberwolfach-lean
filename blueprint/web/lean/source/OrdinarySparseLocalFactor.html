  29       HasResourceDecomposition (localOrdinaryVertices (Q:=Q) (F:=F) (T:=T) (r:=m+1)) T := by

  54     nlinarith only [(div_le_iff₀ hγ).mp hh]
  55   have hbudget : (m+3:ℝ)*(ε*n+1)≤γ/2*n := by
  56     have he : (m+3:ℝ)*ε=γ/4 := by dsimp [ε];field_simp

  58   obtain ⟨k,hkT⟩ := hdiv

  60   have hkappa : κ'*n≤(k:ℝ) := by

  64     nlinarith only [hsize,hkTreal]
