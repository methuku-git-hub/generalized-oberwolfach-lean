  15 theorem normalized_choose_ratio_product (a t k b d : ℝ) (ha : 0 < a) (ht : 0 < t) (hk : 0 < k)
  16     (m n r : ℕ) (hm : (m:ℝ)=a*t-b) (hn : (n:ℝ)=a*k-d) (hrm : r ≤ m) (hrn : r ≤ n) :
  17     (k/t)^r*((m.choose r:ℝ)/n.choose r)=
  18       ∏ j : Fin r,(1-(b+j.val)/(a*t))/(1-(d+j.val)/(a*k)) := by
  19   have hp : (m.choose r:ℝ)/n.choose r=∏ j : Fin r,((m-j.val:ℕ):ℝ)/(n-j.val:ℕ) := by
  20     rw [choose_ratio_product]
  21     exact (Fin.prod_univ_eq_prod_range (fun j => ((m-j:ℕ):ℝ)/(n-j:ℕ)) r).symm
  22   calc
  23     _ = ∏ j : Fin r,(k/t)*(((m-j.val:ℕ):ℝ)/(n-j.val:ℕ)) := by
  24       rw [Finset.prod_mul_distrib]
  25       simp only [Finset.prod_const,Finset.card_univ,Fintype.card_fin,hp]
  26     _ = _ := by
  27       apply Finset.prod_congr rfl
  28       intro j _
  29       have hjm : j.val ≤ m := by omega
  30       have hjn : j.val < n := by omega
  31       have hjnR : (j.val:ℝ)<n := by exact_mod_cast hjn
  32       have hden : 0 < a*k-d-j.val := by linarith [hn]
  33       have hratio : (d+j.val)/(a*k)<1 := (div_lt_one (mul_pos ha hk)).mpr (by linarith)
  34       have hne : 1-(d+j.val)/(a*k)≠0 := by linarith
  35       rw [Nat.cast_sub hjm,Nat.cast_sub hjn.le,hm,hn]
  36       field_simp [ha.ne',ht.ne',hk.ne',hden.ne',hne]
  37       <;> ring

  39 theorem normalized_profile_inclusion_error {I : Type*} [Fintype I]
  40     (a b d : I → ℝ) (ha : ∀ i,0 < a i) (hb : ∀ i,0 ≤ b i) (hd : ∀ i,0 ≤ d i)
  41     (t k : ℝ) (ht : 0 < t) (hk : 0 < k) (m n r : I → ℕ)
  42     (hm : ∀ i,(m i:ℝ)=a i*t-b i) (hn : ∀ i,(n i:ℝ)=a i*k-d i)
  43     (hrm : ∀ i,r i ≤ m i) (hrn : ∀ i,r i ≤ n i)
  44     (hsmall : (∑ j : Σ i,Fin (r i),(d j.1+j.2.val)/(a j.1*k)) ≤ 1/2) :
  45     |(k/t)^(∑ i,r i)*(∏ i,((m i).choose (r i):ℝ)/(n i).choose (r i))-1| ≤
  46       2*((∑ j : Σ i,Fin (r i),(b j.1+j.2.val)/(a j.1*t))+
  47         (∑ j : Σ i,Fin (r i),(d j.1+j.2.val)/(a j.1*k))) := by
  48   let x := fun j : Σ i,Fin (r i) => (b j.1+j.2.val)/(a j.1*t)
  49   let y := fun j : Σ i,Fin (r i) => (d j.1+j.2.val)/(a j.1*k)
  50   have hx (j : Σ i,Fin (r i)) : 0 ≤ x j ∧ x j ≤ 1 := by
  51     refine ⟨div_nonneg (add_nonneg (hb j.1) (Nat.cast_nonneg _)) (mul_pos (ha _) ht).le,?_⟩
  52     apply (div_le_one (mul_pos (ha _) ht)).mpr
  53     have hj : (j.2.val:ℝ) ≤ m j.1 := by exact_mod_cast (show j.2.val ≤ m j.1 by have hh:=hrm j.1;omega)
  54     linarith [hm j.1]
  55   have hy (j : Σ i,Fin (r i)) : 0 ≤ y j ∧ y j ≤ 1 := by
  56     refine ⟨div_nonneg (add_nonneg (hd j.1) (Nat.cast_nonneg _)) (mul_pos (ha _) hk).le,?_⟩
  57     apply (div_le_one (mul_pos (ha _) hk)).mpr
  58     have hj : (j.2.val:ℝ) ≤ n j.1 := by exact_mod_cast (show j.2.val ≤ n j.1 by have hh:=hrn j.1;omega)
  59     linarith [hn j.1]
  60   have he : (k/t)^(∑ i,r i)*(∏ i,((m i).choose (r i):ℝ)/(n i).choose (r i))=
  61       (∏ j : Σ i,Fin (r i),(1-x j))/(∏ j : Σ i,Fin (r i),(1-y j)) := by
  62     rw [← Finset.prod_pow_eq_pow_sum,← Finset.prod_mul_distrib,← Finset.prod_div_distrib,Fintype.prod_sigma]
  63     apply Finset.prod_congr rfl
