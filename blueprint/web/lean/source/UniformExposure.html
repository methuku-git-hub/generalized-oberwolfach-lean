  14 theorem finiteLaw_sigma_mgf_bound
  15     {Ω : Type*} [Fintype Ω] {Γ : Ω → Type*} [∀ ω, Fintype (Γ ω)]
  16     (μ : FiniteLaw Ω) (ν : ∀ ω, FiniteLaw (Γ ω)) (X : (Σ ω, Γ ω) → ℝ)
  17     {c v : ℝ} (hc : 0 ≤ c)
  18     (hdiam : ∀ x y, |(ν x).expectation (fun z => X ⟨x,z⟩) -
  19       (ν y).expectation (fun z => X ⟨y,z⟩)| ≤ c)
  20     (htail : ∀ x u, (ν x).expectation (fun z =>
  21       Real.exp (u * (X ⟨x,z⟩ - (ν x).expectation (fun w => X ⟨x,w⟩)))) ≤
  22         Real.exp (u ^ 2 * v / 8)) (u : ℝ) :
  23     (μ.sigma ν).expectation (fun z => Real.exp (u * (X z - (μ.sigma ν).expectation X))) ≤
  24       Real.exp (u ^ 2 * (v + c ^ 2) / 8) := by
  25   let g (x : Ω) := (ν x).expectation (fun z => X ⟨x,z⟩)
  26   have hmean : (μ.sigma ν).expectation X = μ.expectation g := finiteLaw_sigma_expectation μ ν X
  27   have hfactor (x : Ω) :
  28       (ν x).expectation (fun z => Real.exp (u * (X ⟨x,z⟩ - (μ.sigma ν).expectation X))) =
  29       Real.exp (u * (g x - μ.expectation g)) *
  30         (ν x).expectation (fun z => Real.exp (u * (X ⟨x,z⟩ - g x))) := by
  31     rw [← finiteLaw_expectation_const_mul]
  32     apply congrArg (ν x).expectation
  33     funext z
  34     rw [← Real.exp_add, hmean]
  35     congr 1
  36     ring
  37   rw [finiteLaw_sigma_expectation]
  38   simp_rw [hfactor]
  39   calc
  40     _ ≤ μ.expectation (fun x => Real.exp (u ^ 2 * v / 8) *
  41         Real.exp (u * (g x - μ.expectation g))) := by
  42       apply finiteLaw_expectation_mono
  43       intro x
  44       have hh := mul_le_mul_of_nonneg_left (htail x u)
  45         (Real.exp_pos (u * (g x - μ.expectation g))).le
  46       simpa only [g, mul_comm] using hh
  47     _ = Real.exp (u ^ 2 * v / 8) * μ.expectation
  48         (fun x => Real.exp (u * (g x - μ.expectation g))) :=
  49       finiteLaw_expectation_const_mul μ _ _
  50     _ ≤ Real.exp (u ^ 2 * v / 8) * Real.exp (u ^ 2 * c ^ 2 / 8) := by
  51       apply mul_le_mul_of_nonneg_left (finiteLaw_mgf_of_diameter μ g hc hdiam u)
  52       positivity
  53     _ = _ := by rw [← Real.exp_add]; congr 1; ring

  55 theorem finiteLaw_mgf_absolute_tail
  56     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X : Ω → ℝ)
  57     {v : ℝ} (hv : 0 ≤ v)
  58     (hmgf : ∀ u, μ.expectation (fun ω => Real.exp (u * (X ω - μ.expectation X))) ≤
  59       Real.exp (u ^ 2 * v / 8)) {t : ℝ} (ht : 0 ≤ t) :
  60     μ.probability (fun ω => t ≤ |X ω - μ.expectation X|) ≤
  61       2 * Real.exp (-2 * t ^ 2 / v) := by
  62   let : MeasurableSpace Ω := ⊤
  63   let q : ℝ≥0 := ⟨v / 4, by positivity⟩
  64   have hq : (q : ℝ) = v / 4 := rfl
  65   have hsub : HasSubgaussianMGF (fun ω => X ω - μ.expectation X) q μ.toPMF.toMeasure := by
  66     refine ⟨fun _ => Integrable.of_finite, ?_⟩
  67     intro u
  68     simp only [mgf, finiteLaw_integral]
  69     convert hmgf u using 1
  70     rw [hq]
  71     congr 1
  72     ring
  73   have hh := subgaussian_absolute_tail hsub ht
  74   rw [finiteLaw_probability_measure] at hh
  75   have he : -t ^ 2 / (2 * (q : ℝ)) = -2 * t ^ 2 / v := by rw [hq]; ring
  76   simpa only [he] using hh
