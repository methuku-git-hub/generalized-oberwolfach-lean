  11 def DirectedPathThrough (D:V→V→Prop) (F:SimpleGraph V) (x y:V):List V→Prop
  12   | []=>D x y ∧ ¬F.Adj x y
  13   | z::zs=>D x z ∧ ¬F.Adj x z ∧ DirectedPathThrough D F z y zs

  15 theorem extending_list_embed_internal_path (D:V→V→Prop) (L:Finset V) (β:ℝ)

  17     (hdeg:∀ v,(F.degree v:ℝ)≤M) :
  18     ∀ k:ℕ,∀ (B:Finset V) (x y:V),x≠y→
  19       (B.card:ℝ)+(k+1:ℕ)+2+2*M<β*Fintype.card V→

  21         (∀ z∈xs,z∈L ∧ z∉B ∧ z≠x ∧ z≠y) ∧ DirectedPathThrough D F x y xs := by

  23   induction k with
