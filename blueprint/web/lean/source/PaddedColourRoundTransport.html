  32 theorem paddedRetainedOriginal_eq_some
  33     {V E C : Type*} [Fintype E]
  34     (R : E → Finset V) (L : E → Finset C) (s l : ℕ)
  35     (σ : (E ⊕ ColourPaddingVertex R L s) → PaddedColour R L s l × Bool)
  36     (e : E) (c : C) : paddedRetainedOriginal R L s l σ e=some c ↔
  37       coinRetainedColour (hypergraphLineGraph (paddedResources R L s)) σ (Sum.inl e)=some (Sum.inl c) := by
  38   unfold paddedRetainedOriginal
  39   cases h : coinRetainedColour (hypergraphLineGraph (paddedResources R L s)) σ (Sum.inl e) with
  40   | none => simp
  41   | some b => cases b <;> simp [originalColour]

  74       ¬coinColourSurvives (hypergraphLineGraph (paddedResources R L s)) σ (Sum.inl e) := by

 129 theorem padded_residual_colour_degree_bound
 130     {V E C : Type*} [Fintype V] [Fintype E] [Fintype C]
 131     (R : E → Finset V) (L : E → Finset C) (s l : ℕ)
 132     (σ : (E ⊕ ColourPaddingVertex R L s) → PaddedColour R L s l × Bool)
 133     (hσ : ∀ e,(σ (Sum.inl e)).1∈paddedColourList R L s l (Sum.inl e))
 134     (z : V) (c : C) :
 135     (hypergraphColourClique
 136       (fun e : {e // paddedRetainedOriginal R L s l σ e=none}=>R e.val)
 137       (fun e=>paddedAvailableOriginal R L s l σ e.val) z c).card≤
 138       (remainingColourVertices (hypergraphLineGraph (paddedResources R L s))
 139         (realColourClass (paddedResources R L s) (paddedColourList R L s l)
 140           (Finset.univ.map Function.Embedding.inl) z (Sum.inl c))
 141         (fun e=>(paddedResources R L s e).erase z)
 142         (hypergraphVertexClique (paddedResources R L s)) (Sum.inl c) σ).card := by
 143   classical
 144   apply Finset.card_le_card_of_injOn
 145     (f:=fun e : {e // paddedRetainedOriginal R L s l σ e=none}=>Sum.inl e.val)
 146   · intro e he
 147     have hm : z∈R e.val ∧ c∈paddedAvailableOriginal R L s l σ e.val := by
 148       simpa only [Finset.mem_coe,hypergraphColourClique,Finset.mem_filter,Finset.mem_univ,true_and] using he
 149     have ha : c∈L e.val ∧ ∀ v∈R e.val,∀ f∈hypergraphVertexClique (paddedResources R L s) v,
 150         coinRetainedColour (hypergraphLineGraph (paddedResources R L s)) σ f≠some (Sum.inl c) := by
 151       simpa only [paddedAvailableOriginal,Finset.mem_filter] using hm.2
 152     simp only [Finset.mem_coe,remainingColourVertices,Finset.mem_filter]
