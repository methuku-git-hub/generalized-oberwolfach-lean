  11 theorem power_count_ratio_bound (h s : ℕ) (hs : s ≤ h) (a k b K M z : ℝ)
  12     (ha : 0 < a) (hk : 0 < k) (hb : 1 ≤ b) (hK : 0 ≤ K)
  13     (hM : a*k^h ≤ M) (hz : z ≤ K*(b*k)^(h-s)) :
  14     z/M ≤ (K*b^h/a)/k^s := by
  15   have hM0 : 0 < M := lt_of_lt_of_le (by positivity) hM
  16   have hb0 : 0 ≤ b := by linarith
  17   have hpow : b^(h-s) ≤ b^h := pow_le_pow_right₀ hb (Nat.sub_le _ _)
  18   have hz' : z ≤ K*b^h*k^(h-s) := by
  19     rw [mul_pow] at hz
  20     exact hz.trans (by nlinarith [mul_le_mul_of_nonneg_right hpow (by positivity : 0 ≤ K*k^(h-s))])
  21   have he : (K*b^h/a/k^s)*(a*k^h)=K*b^h*k^(h-s) := by
  22     have hh : k^h=k^(h-s)*k^s := by rw [← pow_add,Nat.sub_add_cancel hs]
  23     rw [hh]
  24     field_simp
  25   apply (div_le_iff₀ hM0).mpr
  26   calc
  27     z ≤ K*b^h*k^(h-s) := hz'
  28     _ = (K*b^h/a/k^s)*(a*k^h) := he.symm
  29     _ ≤ (K*b^h/a/k^s)*M := mul_le_mul_of_nonneg_left hM (by positivity)
