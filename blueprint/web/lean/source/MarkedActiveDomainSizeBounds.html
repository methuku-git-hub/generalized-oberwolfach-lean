  15 theorem marked_active_domain_size_bounds
  16     (f:A↪V) (g:B↪U) (P:SimpleGraph V) (hA:0<Fintype.card A)
  17     (rho gamma omega eps:ℝ) (hrho:0<rho) (hrho1:rho≤1)
  18     (hg:0≤gamma) (ho:0≤omega) (hepsG:eps≤gamma/4) (heps1:eps≤1/4)
  19     (hdeg:∀ x,(1/2+gamma/2)*Fintype.card V≤(P.degree x:ℝ))
  20     (hreg:∀ x,|(P.degree x:ℝ)-2*Fintype.card U/rho|≤eps*Fintype.card V)
  21     (hsize:(3/4:ℝ)*Fintype.card V≤Fintype.card A)
  22     (hubs:omega*Fintype.card V≤Fintype.card U)
  23     (hloseG:(Fintype.card U:ℝ)-Fintype.card B≤(gamma*rho/16)*Fintype.card V)
  24     (hloseO:(Fintype.card U:ℝ)-Fintype.card B≤(omega/2)*Fintype.card V):
  25     (Fintype.card A:ℝ)/2≤2*Fintype.card B/rho∧
  26     2*Fintype.card B/rho≤2*Fintype.card A∧
  27     (omega/2)*Fintype.card A≤Fintype.card B∧Fintype.card B≤Fintype.card A:=by
  28   have hn:(Fintype.card A:ℝ)≤Fintype.card V:=Nat.cast_le.mpr (Fintype.card_le_of_injective f f.injective)
  29   have hm:(Fintype.card B:ℝ)≤Fintype.card U:=Nat.cast_le.mpr (Fintype.card_le_of_injective g g.injective)
  30   have hn0:(0:ℝ)≤Fintype.card V:=Nat.cast_nonneg _
  31   have hm0:(0:ℝ)≤Fintype.card U:=Nat.cast_nonneg _
  32   have ha0:(0:ℝ)≤Fintype.card A:=Nat.cast_nonneg _
  33   obtain ⟨x⟩:=Fintype.card_pos_iff.mp hA
  34   have hregx:=abs_le.mp (hreg (f x))
  35   have hdegx:=hdeg (f x)
  36   have hmax:(P.degree (f x):ℝ)≤Fintype.card V:=by
  37     rw [←SimpleGraph.card_neighborFinset_eq_degree]
  38     exact_mod_cast (Finset.card_le_univ (s:=P.neighborFinset (f x)))
  39   have heG:=mul_le_mul_of_nonneg_right hepsG hn0
  40   have he1:=mul_le_mul_of_nonneg_right heps1 hn0
  41   have hDlo:(1/2+gamma/4)*Fintype.card V≤2*Fintype.card U/rho:=by
  42     nlinarith only [hregx.2,hdegx,heG]
  43   have hDhi:2*Fintype.card U/rho≤(5/4:ℝ)*Fintype.card V:=by
  44     nlinarith only [hregx.1,hmax,he1]
  45   have hDlo':((1/2+gamma/4)*Fintype.card V)*rho≤2*Fintype.card U:=(le_div_iff₀ hrho).mp hDlo
  46   have hDhi':2*(Fintype.card U:ℝ)≤((5/4:ℝ)*Fintype.card V)*rho:=(div_le_iff₀ hrho).mp hDhi
  47   have hγrho:0≤gamma*rho*(Fintype.card V:ℝ):=by positivity
  48   have hnρ:=mul_le_mul_of_nonneg_right hn hrho.le
  49   have hmBound:(Fintype.card U:ℝ)≤(5/8:ℝ)*Fintype.card V:=by
  50     have hρ:=mul_le_mul_of_nonneg_left hrho1 (show 0≤(5/4:ℝ)*Fintype.card V by positivity)
  51     nlinarith only [hDhi',hρ]
  52   refine ⟨?_,?_,?_,?_⟩
  53   · apply (le_div_iff₀ hrho).mpr
  54     nlinarith only [hDlo',hloseG,hnρ,hγrho]
  55   · apply (div_le_iff₀ hrho).mpr
  56     have hs:=mul_le_mul_of_nonneg_right hsize hrho.le
  57     have hnrho:0≤(Fintype.card V:ℝ)*rho:=by positivity
  58     nlinarith only [hDhi',hm,hs,hnrho]
  59   · have hs:=mul_le_mul_of_nonneg_left hn (show 0≤omega/2 by positivity)
  60     linarith only [hs,hubs,hloseO]
  61   · exact_mod_cast (show (Fintype.card B:ℝ)≤Fintype.card A by linarith only [hm,hmBound,hsize,hn0])
