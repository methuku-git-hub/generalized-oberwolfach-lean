  15 def replaceEmbeddingVertex (g : W ↪ V) (z : W) (x : V) (hx : ∀ w,g w≠x) : W ↪ V where
  16   toFun w := if w=z then x else g w
  17   inj' := by
  18     intro a b he
  19     by_cases ha : a=z <;> by_cases hb : b=z
  20     · exact ha.trans hb.symm
  21     · simp only [ha,hb,if_true,if_false] at he
  22       exact False.elim (hx b he.symm)
  23     · simp only [ha,hb,if_true,if_false] at he
  24       exact False.elim (hx a he)
  25     · simp only [ha,hb,if_false] at he
  26       exact g.injective he

  34 theorem ordinary_attachment_root_embedding (β : ℝ) (hβ : 0<β) (r b : ℕ) (hr : 3≤r) :
  35     ∃ n₀ : ℕ,∀ n : ℕ,n₀≤n →
  36     ∀ (V U : Type*) [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U],Fintype.card V=n →
  37     ∀ (G : SimpleGraph V) (H : V → U → Prop),
  38     (∀ a,(1/2+β)*(n:ℝ)≤G.degree a) →
  39     (∀ a,(1-1/(2*(r:ℝ)))*(Fintype.card U:ℝ)≤(Finset.univ.filter (H a)).card) →
  40     ∀ (S : Finset V) (B : Finset U) (x : V),S.card≤b → x∈S → 2*B.card<Fintype.card U →
  41     ∃ (f : Fin r ↪ V) (u : U),
  42       (∀ i j,i≠⟨0,by omega⟩ → j≠⟨0,by omega⟩ →
  43         (SimpleGraph.cycleGraph r).Adj i j → G.Adj (f i) (f j)) ∧
  44       (∀ j,(SimpleGraph.cycleGraph r).Adj ⟨0,by omega⟩ j → G.Adj x (f j)) ∧
  45       H x u ∧ (∀ j,j≠⟨0,by omega⟩ → H (f j) u) ∧
  46       (∀ j,f j∉S) ∧ u∉B := by
  47   obtain ⟨n₁,hcycle⟩ := rooted_cycle_inside_large_subset β hβ r b hr
  48   refine ⟨max n₁ (b+r+1),?_⟩
  49   intro n hn V U _ _ _ _ hV G H hdeg hH S B x hS hx hB
  50   obtain ⟨g,hg0,hhom,_hgA,havoid⟩ := hcycle n ((le_max_left _ _).trans hn) V hV.le G hdeg
  51     Finset.univ (S.erase x) (by simp;positivity) ((Finset.card_erase_le).trans hS) x
  52     (Finset.mem_univ _) (Finset.notMem_erase _ _)
  53   let z : Fin r := ⟨0,by omega⟩
  54   have hgn (w : Fin r) (hw : w≠z) : g w∉S := by
  55     intro hmem
  56     have hne : g w≠x := by intro he;exact hw (g.injective (he.trans hg0.symm))
  57     exact havoid w (Finset.mem_erase.mpr ⟨hne,hmem⟩)
  58   let A := Finset.univ.image g
  59   have hAc : A.card≤r := Finset.card_image_le.trans (by simp)
  60   obtain ⟨u,hu,hAu⟩ := common_spoke_hub_avoiding H A B r (by omega) hAc (fun v _ => hH v) hB
  61   have hSc : (S∪A).card<n := by
  62     have hh := Finset.card_union_le S A
  63     have hn' := (le_max_right n₁ (b+r+1)).trans hn
  64     omega
  65   obtain ⟨x',_hx',hx'SA⟩ := Finset.exists_mem_notMem_of_card_lt_card
  66     (show (S∪A).card<(Finset.univ:Finset V).card by simpa [hV] using hSc)
  67   have hx'g : ∀ w,g w≠x' := by
  68     intro w he
  69     apply hx'SA
  70     exact Finset.mem_union_right _ (Finset.mem_image.mpr ⟨w,Finset.mem_univ _,he⟩)
  71   let f := replaceEmbeddingVertex g z x' hx'g
  72   have hf (w : Fin r) (hw : w≠z) : f w=g w := replaceEmbeddingVertex_ne g z x' hx'g w hw
  73   refine ⟨f,u,?_,?_,?_,?_,?_,hu⟩
  74   · intro i j hi hj hij
  75     rw [hf i hi,hf j hj]
  76     exact hhom i j hij
  77   · intro j hj
  78     have hjz : j≠z := hj.ne.symm
  79     rw [hf j hjz,←hg0]
  80     exact hhom z j hj
  81   · apply hAu
  82     exact Finset.mem_image.mpr ⟨z,Finset.mem_univ _,hg0⟩
  83   · intro j hj
  84     rw [hf j hj]
  85     exact hAu _ (Finset.mem_image.mpr ⟨j,Finset.mem_univ _,rfl⟩)
  86   · intro j
  87     by_cases hj : j=z
  88     · subst j
  89       rw [show f z=x' from replaceEmbeddingVertex_at g z x' hx'g]
  90       exact fun h => hx'SA (Finset.mem_union_left _ h)
  91     · rw [hf j hj]
  92       exact hgn j hj
