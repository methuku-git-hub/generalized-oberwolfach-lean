   8 theorem matching_frequency_quotient_margin
   9     (r actual n θ f error γ ε:ℝ)
  10     (hr:1≤r) (hn:0≤n) (hε:0≤ε) (hε1:ε≤1) (hγ:0≤γ) (hγε:γ≤ε)
  11     (he0:0≤error) (he:error≤ε*n) (htarget:|(1-θ)*f*n|≤n)
  12     (h:|actual-(1-θ)*f*n/r|≤(error+γ*(|(1-θ)*f*n|+error))/r):
  13     |actual-(1-θ)*f*n/r|≤3*ε*n :=by
  14   have ht0:0≤|(1-θ)*f*n|:=abs_nonneg _
  15   have hfirst:γ*(|(1-θ)*f*n|+error)≤ε*(n+ε*n):=by
  16     apply le_trans (mul_le_mul_of_nonneg_left (add_le_add htarget he) hγ)
  17     apply mul_le_mul_of_nonneg_right hγε
  18     positivity
  19   have hprod:ε*(ε*n)≤ε*n:=by
  20     have hh:=mul_le_mul_of_nonneg_right hε1 (mul_nonneg hε hn)
  21     nlinarith
  22   have hnum:error+γ*(|(1-θ)*f*n|+error)≤3*ε*n:=by nlinarith
  23   have hdiv:(error+γ*(|(1-θ)*f*n|+error))/r≤3*ε*n:=by
  24     apply (div_le_iff₀ (by linarith : 0<r)).2
  25     have hh:=mul_le_mul_of_nonneg_left hr (show 0≤3*ε*n by positivity)
  26     nlinarith
  27   exact h.trans hdiv
  28 end Oberwolfach
