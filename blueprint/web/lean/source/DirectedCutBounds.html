  12 def matrixCut {V : Type*} (M : V → V → ℕ) (S T : Finset V) : ℕ :=
  13   ∑ x ∈ S, ∑ y ∈ T, M x y

  15 theorem matrixCut_transpose {V : Type*} (M : V → V → ℕ) (S T : Finset V) :
  16     matrixCut M S T = matrixCut (fun x y => M y x) T S := by
  17   exact Finset.sum_comm

  19 theorem matrixCut_row_lower {V : Type*} [Fintype V]
  20     (M : V → V → ℕ) (hM : ∀ x y, M x y ≤ 1) (S T : Finset V) {d : ℝ}
  21     (hrow : ∀ x ∈ S, d ≤ ((∑ y, M x y : ℕ) : ℝ)) :
  22     (S.card : ℝ) * (d + T.card - Fintype.card V) ≤ (matrixCut M S T : ℝ) := by
  23   have hpoint (x : V) (hx : x ∈ S) :
  24       d + T.card - Fintype.card V ≤ ((∑ y ∈ T, M x y : ℕ) : ℝ) := by
  25     have hc : ∑ y ∈ Finset.univ \ T, M x y ≤ Fintype.card V - T.card := by
  26       calc
  27         _ ≤ ∑ _y ∈ Finset.univ \ T, 1 := Finset.sum_le_sum (fun y _ => hM x y)
  28         _ = _ := by simp [Finset.card_sdiff_of_subset (Finset.subset_univ T)]
  29     have hs := Finset.sum_sdiff (f := fun y => M x y) (Finset.subset_univ T)
  30     have ht := Finset.card_le_univ T
  31     have hcr : ((∑ y ∈ Finset.univ \ T, M x y : ℕ) : ℝ) ≤
  32         (Fintype.card V : ℝ) - T.card := by
  33       exact_mod_cast hc
  34     have hsr : ((∑ y ∈ Finset.univ \ T, M x y : ℕ) : ℝ) +
  35         ((∑ y ∈ T, M x y : ℕ) : ℝ) = ((∑ y, M x y : ℕ) : ℝ) := by
  36       exact_mod_cast hs
  37     linarith [hrow x hx]
  38   simpa only [matrixCut, Nat.cast_sum, Finset.sum_const, nsmul_eq_mul] using
  39     Finset.sum_le_sum hpoint

  41 theorem matrixCut_small_side {V : Type*} [Fintype V]
  42     (M : V → V → ℕ) (hM : ∀ x y, M x y ≤ 1)
  43     {d c δ s : ℝ} (hrow : ∀ x, d ≤ ((∑ y, M x y : ℕ) : ℝ))
  44     (hcol : ∀ y, d ≤ ((∑ x, M x y : ℕ) : ℝ))
  45     (hmargin : c ≤ d - s - 2 * δ * Fintype.card V)
  46     (S T : Finset V)
  47     (hlarge : (1 - 2 * δ) * Fintype.card V < (S.card : ℝ) + T.card)
  48     (hsmall : min (S.card : ℝ) T.card ≤ s) :
  49     c * min (S.card : ℝ) T.card ≤ (matrixCut M S T : ℝ) := by
  50   have hle (A B : Finset V) (N : V → V → ℕ)
  51       (hN : ∀ x y, N x y ≤ 1)
  52       (hdeg : ∀ x, d ≤ ((∑ y, N x y : ℕ) : ℝ))
  53       (hA : (A.card : ℝ) ≤ s)
  54       (hab : (1 - 2 * δ) * Fintype.card V < (A.card : ℝ) + B.card) :
  55       c * A.card ≤ (matrixCut N A B : ℝ) := by
  56     have hh := matrixCut_row_lower N hN A B (fun x _ => hdeg x)
  57     have hb : c ≤ d + B.card - Fintype.card V := by linarith
  58     have hm := mul_le_mul_of_nonneg_left hb (show (0 : ℝ) ≤ A.card by positivity)
  59     nlinarith
  60   rcases le_total (S.card : ℝ) T.card with h | h
  61   · rw [min_eq_left h] at *
  62     exact hle S T M hM hrow hsmall hlarge
  63   · rw [min_eq_right h] at *
  64     rw [matrixCut_transpose]
  65     exact hle T S (fun x y => M y x) (fun x y => hM y x) hcol hsmall (by linarith)

  67 theorem matrixCut_upper_of_degrees {V : Type*} [Fintype V]
  68     (M : V → V → ℕ) {b : ℝ}
  69     (hrow : ∀ x, ((∑ y, M x y : ℕ) : ℝ) ≤ b)
  70     (hcol : ∀ y, ((∑ x, M x y : ℕ) : ℝ) ≤ b)
  71     (S T : Finset V) :
  72     (matrixCut M S T : ℝ) ≤ b * min (S.card : ℝ) T.card := by
  73   have hle (N : V → V → ℕ) (A B : Finset V)
  74       (h : ∀ x, ((∑ y, N x y : ℕ) : ℝ) ≤ b) :
  75       (matrixCut N A B : ℝ) ≤ b * A.card := by
  76     have hh (x : V) : ((∑ y ∈ B, N x y : ℕ) : ℝ) ≤ b := by
  77       apply le_trans _ (h x)
  78       exact_mod_cast Finset.sum_le_univ_sum_of_nonneg (fun y => Nat.zero_le (N x y))
  79     have hs := Finset.sum_le_sum (fun x (_ : x ∈ A) => hh x)
  80     simpa only [matrixCut, Nat.cast_sum, Finset.sum_const, nsmul_eq_mul, mul_comm] using hs
  81   rcases le_total (S.card : ℝ) T.card with h | h
  82   · rw [min_eq_left h]; exact hle M S T hrow
  83   · rw [min_eq_right h, matrixCut_transpose]
  84     exact hle (fun x y => M y x) T S hcol

  86 theorem matrixCut_loss_bound {V : Type*} [Fintype V]
  87     (M L W : V → V → ℕ) {b : ℝ}
  88     (hpoint : ∀ x y, M x y ≤ L x y + W x y)
  89     (hrow : ∀ x, ((∑ y, W x y : ℕ) : ℝ) ≤ b)
  90     (hcol : ∀ y, ((∑ x, W x y : ℕ) : ℝ) ≤ b)
  91     (S T : Finset V) :
  92     (matrixCut M S T : ℝ) - b * min (S.card : ℝ) T.card ≤ (matrixCut L S T : ℝ) := by
  93   have hh : matrixCut M S T ≤ matrixCut L S T + matrixCut W S T := by
  94     simp only [matrixCut, ← Finset.sum_add_distrib]
  95     exact Finset.sum_le_sum (fun x _ => Finset.sum_le_sum (fun y _ => hpoint x y))
  96   have hr : (matrixCut M S T : ℝ) ≤ (matrixCut L S T : ℝ) + matrixCut W S T := by exact_mod_cast hh
  97   linarith [matrixCut_upper_of_degrees W hrow hcol S T]
