  14 def colourMask {C : Type*} (c : C) (x : C×Bool) : Option Bool :=
  15   if x.1=c then some x.2 else none

  17 theorem colourMask_eq_none {C : Type*} (c : C) (x : C×Bool) :
  18     colourMask c x=none ↔ x.1≠c := by
  19   unfold colourMask
  20   by_cases hc : x.1=c <;> simp [hc]

  58 theorem colourMask_mass_none {C : Type*} [Fintype C]
  59     (μ : FiniteLaw (C×Bool)) (c : C) :
  60     (μ.map (colourMask c)).mass none=1-μ.probability (fun x=>x.1=c) := by
  61   rw [finiteLaw_map_mass_probability]
  62   have he (x : C×Bool) := colourMask_eq_none c x
  63   rw [finiteLaw_probability_congr μ _ _ he]
  64   have h := finiteLaw_probability_compl μ (fun x=>x.1=c)
  65   linarith
