  14 def colourRestriction {V : Type*} {k : ℕ} (c : Sym2 V → Fin k)
  15     (M : V → V → ℕ) (a : Fin k) (x y : V) : ℕ :=
  16   if c s(x,y) = a then M x y else 0

  18 def edgeColourGraph {V : Type*} {k : ℕ} (G : SimpleGraph V)
  19     (c : Sym2 V → Fin k) (a : Fin k) : SimpleGraph V where
  20   Adj x y := G.Adj x y ∧ c s(x,y) = a
  21   symm := by
  22     constructor
  23     intro x y h
  24     exact ⟨h.1.symm, by simpa only [Sym2.eq_swap] using h.2⟩
  25   loopless := by constructor; intro x h; exact (G.ne_of_adj h.1) rfl

  28     (M : V → V → ℕ) (x y : V) :

  38 theorem colourRestriction_underlying {V : Type*} {k : ℕ} (G : SimpleGraph V)
  39     (c : Sym2 V → Fin k) (M : V → V → ℕ)
  40     (hphysical : ∀ x y, M x y + M y x = if G.Adj x y then 1 else 0)
  41     (a : Fin k) (x y : V) :
  42     0 < colourRestriction c M a x y + colourRestriction c M a y x ↔
  43       (edgeColourGraph G c a).Adj x y := by
  44   dsimp only [colourRestriction, edgeColourGraph]
  45   rw [show s(y,x) = s(x,y) from Sym2.eq_swap]
  46   by_cases hc : c s(x,y) = a
  47   · simp only [hc, ite_true, and_true, hphysical]
  48     split_ifs <;> simp_all
  49   · simp [hc]

  62 theorem boundedNat_sum_stability {V : Type*} [Fintype V]
  63     (f g : V → ℕ) (hf : ∀ x, f x ≤ 1) (hg : ∀ x, g x ≤ 1)
  64     (D : Finset V) (hsame : ∀ x, x ∉ D → f x = g x) :
  65     |((∑ x, f x : ℕ) : ℝ) - ((∑ x, g x : ℕ) : ℝ)| ≤ D.card := by
  66   have hp (x : V) : |(f x : ℝ) - g x| ≤ if x ∈ D then 1 else 0 := by
  67     by_cases h : x ∈ D
  68     · rw [ite_eq_left h]
  69       have hfr : (f x : ℝ) ≤ 1 := by exact_mod_cast hf x
  70       have hgr : (g x : ℝ) ≤ 1 := by exact_mod_cast hg x
  71       exact abs_le.mpr ⟨by linarith [Nat.cast_nonneg (α := ℝ) (f x)], by linarith [Nat.cast_nonneg (α := ℝ) (g x)]⟩
  72     · simp [h, hsame x h]
  73   rw [Nat.cast_sum, Nat.cast_sum, ← Finset.sum_sub_distrib]
  74   apply (Finset.abs_sum_le_sum_abs _ _).trans
  75   have hs := Finset.sum_le_sum (fun x (_ : x ∈ Finset.univ) => hp x)
  76   simpa using hs

  78 theorem colourRestriction_degree_stability
  79     {V : Type*} [Fintype V] {k : ℕ} (c : Sym2 V → Fin k)
  80     (M N : V → V → ℕ) (hM : ∀ x y, M x y ≤ 1) (hN : ∀ x y, N x y ≤ 1)
  81     (a : Fin k) (x : V) :
  82     |((∑ y, colourRestriction c M a x y : ℕ) : ℝ) -
  83       ((∑ y, colourRestriction c N a x y : ℕ) : ℝ)| ≤
  84       (Finset.univ.filter (fun y => M x y ≠ N x y ∨ M y x ≠ N y x)).card := by
  85   apply boundedNat_sum_stability
  86   · intro y; exact (colourRestriction_le c M a x y).trans (hM x y)
  87   · intro y; exact (colourRestriction_le c N a x y).trans (hN x y)
  88   · intro y hy
  89     have hh : M x y = N x y ∧ M y x = N y x := by simpa using hy
  90     have he := hh.1
  91     simp only [colourRestriction, he]
