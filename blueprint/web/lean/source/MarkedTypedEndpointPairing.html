  14 theorem marked_typed_endpoint_pairing_of_bounds
  15     {A X Y V : Type*} [Fintype A] [Fintype X] [Fintype Y] [Fintype V]
  16     (EA : A → Y → Prop) (EX : X → Y → Prop)
  17     (PA : A → Prop) (PX : X → Prop) (Q : Y → Prop)
  18     (hA : Fintype.card A=Fintype.card Y) (hX : Fintype.card X=Fintype.card Y)
  19     (hclass : Fintype.card {a // PA a}=Fintype.card {y // Q y})
  20     (hallowed : ∀ x y, EX x y → PX x → ¬Q y)
  21     (π : Equiv.Perm V) (b M : ℕ) (labelA : A → V) (labelX : X → V)
  22     (hfA : ∀ v, Fintype.card {a // labelA a=v}≤M)
  23     (hfX : ∀ v, Fintype.card {x // labelX x=v}≤M)
  24     (d a lam : ℝ) (ha : 0<a) (hlam : 0<lam)
  25     (hdegree : 2*a+(M*(2*b+1):ℕ)≤d)
  26     (hmargin : (M*(2*b+1):ℕ)<lam*a)
  27     (hgap : ((Finset.univ.filter PX).card:ℝ)+2*a≤(Finset.univ.filter (fun y => ¬Q y)).card)
  28     (B₁ : BipartiteBlockBounds EA PA Q d a lam)
  29     (B₂ : BipartiteBlockBounds EA (fun x => ¬PA x) (fun y => ¬Q y) d a lam)
  30     (B₃ : BipartiteBlockBounds EX PX (fun y => ¬Q y) d a lam)
  31     (B₄ : BipartiteBlockBounds EX (fun x => ¬PX x) (fun y => ¬Q y) d a lam)
  32     (B₅ : BipartiteBlockBounds EX (fun x => ¬PX x) Q d a lam) :
  33     ∃ f : A ≃ Y, ∃ g : X ≃ Y,
  34       (∀ x, EA x (f x)) ∧ (∀ x, EX x (g x)) ∧
  35       (∀ x, Q (f x) ↔ PA x) ∧ (∀ x, PX x → ¬Q (g x)) ∧
  36       ∀ y, ¬PermutationClose π b (labelA (f.symm y)) (labelX (g.symm y)) := by
  37   classical
  38   have hC : (0:ℝ)≤(M*(2*b+1):ℕ) := Nat.cast_nonneg _
  39   have hd : a≤d := by linarith
  40   obtain ⟨f₁,hf₁⟩ := bipartite_block_perfect_matching EA PA Q d a lam ha hd hlam B₁ hclass
  41   have hclass₀ : Fintype.card {x // ¬PA x}=Fintype.card {y // ¬Q y} := by
  42     rw [Fintype.card_subtype_compl,Fintype.card_subtype_compl,hA,hclass]
  43   obtain ⟨f₀,hf₀⟩ := bipartite_block_perfect_matching EA (fun x => ¬PA x)
  44     (fun y => ¬Q y) d a lam ha hd hlam B₂ (by convert hclass₀ using 1 <;> congr!)
  45   obtain ⟨f,hf,hftype⟩ := combine_class_matchings PA Q EA f₁ f₀ hf₁ hf₀
  46   let J (x : X) (y : Y) := PermutationClose π b (labelX x) (labelA (f.symm y))
  47   have hJrow (x : X) : ((Finset.univ.filter (J x)).card:ℝ)≤(M*(2*b+1):ℕ) := by
  48     have hh := label_permutation_close_card (fun y => labelA (f.symm y)) π b M
  49       (label_fibre_bound_reindex labelA f.symm M hfA) (labelX x)
  50     have hn : (Finset.univ.filter (J x)).card≤M*(2*b+1) := by
  51       simpa only [J,Fintype.card_subtype] using hh
  52     exact_mod_cast hn
  53   have hJcol (y : Y) : ((Finset.univ.filter (fun x => J x y)).card:ℝ)≤(M*(2*b+1):ℕ) := by
  54     have hh := label_permutation_close_card labelX π b M hfX (labelA (f.symm y))
  55     have he : (Finset.univ.filter (fun x => J x y))=
  56         Finset.univ.filter (fun x => PermutationClose π b (labelA (f.symm y)) (labelX x)) := by
  57       ext x; simp only [Finset.mem_filter,Finset.mem_univ,true_and,J]
  58       exact permutationClose_symm π b _ _
  59     rw [he]
  60     have hn : (Finset.univ.filter (fun x => PermutationClose π b (labelA (f.symm y)) (labelX x))).card≤M*(2*b+1) := by
  61       simpa only [Fintype.card_subtype] using hh
  62     exact_mod_cast hn
  63   obtain ⟨g,hg⟩ := three_block_matching_after_deletion EX J PX Q hX d a lam
  64     (M*(2*b+1):ℕ) ha hlam hdegree hmargin hgap B₃ B₄ B₅ hJrow hJcol
  65   refine ⟨f,g,hf,fun x => (hg x).1,hftype,fun x hx => hallowed x (g x) (hg x).1 hx,?_⟩
  66   intro y hclose
  67   have hh := (hg (g.symm y)).2
  68   apply hh
  69   dsimp [J]
  70   rw [g.apply_symm_apply]
  71   exact (permutationClose_symm π b _ _).mp hclose
