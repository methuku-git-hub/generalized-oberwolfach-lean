  14 structure EndpointPairing (G F:SimpleGraph V) (A B:Finset V) where
  15   disjoint_parts : Disjoint A B
  16   left : B ↪ A
  17   right : B ↪ A
  18   adjacent : ∀ u,G.Adj (left u).val u.val ∧ G.Adj (right u).val u.val
  19   avoid : ∀ u,¬F.Adj (left u).val (right u).val
  20   disjoint_ranges : Disjoint (Set.range left) (Set.range right)
  21   cover : ∀ v:A,(∃ u,left u=v) ∨ ∃ u,right u=v

  23 theorem endpoint_pairing_of_degree_cuts (c:ℝ) (hc:0<c) (b:ℕ):
  24     ∃ N:ℕ,∀ (V:Type) [Fintype V] [DecidableEq V] (G F:SimpleGraph V) (A B:Finset V),
  25       N≤B.card→A.card=2*B.card→Disjoint A B→
  26       (∀ v,F.degree v≤b)→
  27       (∀ v∈A,c*B.card≤((B∩G.neighborFinset v).card:ℝ))→
  28       (∀ u∈B,c*A.card≤((A∩G.neighborFinset u).card:ℝ))→
  29       (∀ S⊆A,∀ T⊆B,c/16*B.card≤(S.card:ℝ)→c/16*B.card≤(T.card:ℝ)→
  30         c*S.card*T.card≤(bipartiteMatrixCut (fun v w=>if G.Adj v w then 1 else 0) S T:ℝ))→
  31       Nonempty (EndpointPairing G F A B):=by
  32   obtain ⟨N,hN⟩:=ordinary_neighbour_pair_assignment c hc b
  33   refine ⟨N,?_⟩
  34   intro V _ _ G F A B hn hAB hdis hF hrow hcol hcut
  35   let eA:A↪V:=⟨Subtype.val,Subtype.val_injective⟩
  36   let eB:B↪V:=⟨Subtype.val,Subtype.val_injective⟩
  37   have hrow' (v:A):c*B.card≤((Finset.univ.filter (fun u:B=>G.Adj v.val u.val)).card:ℝ):=by
  38     rw [finite_domain_predicate_card]
  39     have he:B.filter (G.Adj v.val)=B∩G.neighborFinset v.val:=by ext u;simp
  40     rw [he];exact hrow v.val v.property
  41   have hcol' (u:B):2*c*B.card≤((Finset.univ.filter (fun v:A=>G.Adj v.val u.val)).card:ℝ):=by
  42     rw [finite_domain_predicate_card A (fun v=>G.Adj v u.val)]
  43     have he:A.filter (fun v=>G.Adj v u.val)=A∩G.neighborFinset u.val:=by
  44       ext v;simp only [Finset.mem_filter,Finset.mem_inter,SimpleGraph.mem_neighborFinset,G.adj_comm]
  45     rw [he]
  46     have hh:=hcol u.val u.property
  47     rw [hAB,Nat.cast_mul,Nat.cast_ofNat] at hh
  48     nlinarith only [hh]
  49   have hcut' (S:Finset A) (T:Finset B)
  50       (hS:c/16*B.card≤(S.card:ℝ)) (hT:c/16*B.card≤(T.card:ℝ)):
  51       c*S.card*T.card≤(bipartiteMatrixCut (fun v:A=>fun u:B=>if G.Adj v.val u.val then 1 else 0) S T:ℝ):=by
  52     have hh:=hcut (S.map eA) (by rintro v hv;obtain ⟨x,hx,rfl⟩:=Finset.mem_map.mp hv;exact x.property)
  53       (T.map eB) (by rintro u hu;obtain ⟨x,hx,rfl⟩:=Finset.mem_map.mp hu;exact x.property)
  54       (by simpa only [Finset.card_map] using hS) (by simpa only [Finset.card_map] using hT)
  55     simpa only [Finset.card_map,bipartiteMatrixCut,Finset.sum_map,eA,eB,Function.Embedding.coeFn_mk] using hh
  56   obtain ⟨v,w,hadj,hranges,hcover⟩:=hN A B B.card hn
  57     (by simpa only [Fintype.card_coe] using hAB) (Fintype.card_coe B)
  58     (fun v:A=>fun u:B=>G.Adj v.val u.val) (F.comap Subtype.val)
  59     hrow' hcol' (fun v=>(embedding_comap_degree_le eA F v).trans (hF v.val)) hcut'
  60   exact ⟨⟨hdis,v,w,fun u=>⟨(hadj u).1,(hadj u).2.1⟩,fun u=>(hadj u).2.2,hranges,hcover⟩⟩

  65 def graph (P:EndpointPairing G F A B):SimpleGraph V where
  66   Adj x y := ∃ u:B,(x=u.val ∧ (y=(P.left u).val ∨ y=(P.right u).val)) ∨
  67     (y=u.val ∧ (x=(P.left u).val ∨ x=(P.right u).val))
  68   symm := ⟨by
  69     intro x y
  70     rintro ⟨u,h|h⟩
  71     · exact ⟨u,Or.inr h⟩
  72     · exact ⟨u,Or.inl h⟩⟩
  73   loopless := ⟨by
  74     intro x
  75     have hnot (u:B):u.val∉A:=fun hu=>Finset.disjoint_left.mp P.disjoint_parts hu u.property
  76     rintro ⟨u,⟨h,h'|h'⟩|⟨h,h'|h'⟩⟩
  77     · apply hnot u;rw [←h,h'];exact (P.left u).property
  78     · apply hnot u;rw [←h,h'];exact (P.right u).property
  79     · apply hnot u;rw [←h,h'];exact (P.left u).property
  80     · apply hnot u;rw [←h,h'];exact (P.right u).property⟩
