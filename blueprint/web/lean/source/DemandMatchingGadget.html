   3 /-! The bounded bipartite degree problem is encoded as a bistochastic
   4 matching problem. Each original edge has one pair of auxiliary vertices;
   5 matching that pair means the original edge is unused. -/

  13 abbrev DemandSlots {X : Type*} (f : X → ℕ) := Σ x, Fin (f x)
  14 
  15 def demandMatchingMatrix {X Y E : Type*}
  16     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ) (z : E → ℝ) :
  17     (DemandSlots f ⊕ E) → (DemandSlots g ⊕ E) → ℝ
  18   | Sum.inl a, Sum.inl _ => 0
  19   | Sum.inl a, Sum.inr e => if a.1 = tail e then z e / f (tail e) else 0
  20   | Sum.inr e, Sum.inl b => if head e = b.1 then z e / g (head e) else 0
  21   | Sum.inr e, Sum.inr e' => if e = e' then 1 - z e else 0
  22 
  23 def DemandMatchingAllowed {X Y E : Type*}
  24     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ) :
  25     (DemandSlots f ⊕ E) → (DemandSlots g ⊕ E) → Prop
  26   | Sum.inl a, Sum.inl _ => False
  27   | Sum.inl a, Sum.inr e => a.1 = tail e
  28   | Sum.inr e, Sum.inl b => head e = b.1
  29   | Sum.inr e, Sum.inr e' => e = e'

  31 theorem demandMatchingMatrix_nonneg {X Y E : Type*}
  32     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ) (z : E → ℝ)
  33     (hz : ∀ e, 0 ≤ z e ∧ z e ≤ 1) :
  34     ∀ a b, 0 ≤ demandMatchingMatrix tail head f g z a b := by
  35   intro a b
  36   cases a <;> cases b <;> simp only [demandMatchingMatrix]
  37   · exact le_refl _
  38   · split_ifs; exact div_nonneg (hz _).1 (Nat.cast_nonneg _); exact le_refl _
  39   · split_ifs; exact div_nonneg (hz _).1 (Nat.cast_nonneg _); exact le_refl _
  40   · split_ifs; exact sub_nonneg.mpr (hz _).2; exact le_refl _

  42 theorem demandMatchingMatrix_pos_allowed {X Y E : Type*}
  43     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ) (z : E → ℝ)
  44     (a : DemandSlots f ⊕ E) (b : DemandSlots g ⊕ E)
  45     (h : 0 < demandMatchingMatrix tail head f g z a b) :
  46     DemandMatchingAllowed tail head f g a b := by
  47   cases a <;> cases b <;> simp only [demandMatchingMatrix, DemandMatchingAllowed] at h ⊢
  48   · exact (lt_irrefl 0) h
  49   all_goals split_ifs at h with hh; exact hh; exact False.elim ((lt_irrefl 0) h)

  51 theorem sum_slots_indicator {X : Type*} [Fintype X] (f : X → ℕ) (x : X) (a : ℝ) :
  52     (∑ s : DemandSlots f, if x = s.1 then a else 0) = (f x : ℝ) * a := by
  53   classical
  54   rw [Fintype.sum_sigma]
  55   have hinner (y : X) : (∑ _i : Fin (f y), if x = y then a else 0) =
  56       if x = y then (f y : ℝ) * a else 0 := by
  57     by_cases hxy : x = y <;> simp [hxy]
  58   simp_rw [hinner]
  59   simp

  61 theorem demand_zero_forces_edge_zero {X E : Type*} [Fintype E]
  62     (tail : E → X) (f : X → ℕ) (z : E → ℝ) (hz : ∀ e, 0 ≤ z e)
  63     (hrow : ∀ x, (∑ e, if tail e = x then z e else 0) = f x)
  64     (e : E) (hf : f (tail e) = 0) : z e = 0 := by
  65   have hle := Finset.single_le_sum
  66     (f := fun a => if tail a = tail e then z a else 0) (s := Finset.univ) (a := e)
  67     (fun a _ => by split_ifs; exact hz a; exact le_refl _) (Finset.mem_univ e)
  68   simp only [ite_true, hrow, hf, Nat.cast_zero] at hle
  69   exact le_antisymm hle (hz e)

  71 theorem demandMatchingMatrix_row {X Y E : Type*}
  72     [Fintype X] [Fintype Y] [Fintype E]
  73     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ) (z : E → ℝ)
  74     (hz : ∀ e, 0 ≤ z e)
  75     (hrow : ∀ x, (∑ e, if tail e = x then z e else 0) = f x)
  76     (hcol : ∀ y, (∑ e, if head e = y then z e else 0) = g y)
  77     (a : DemandSlots f ⊕ E) : ∑ b, demandMatchingMatrix tail head f g z a b = 1 := by
  78   classical
  79   rw [Fintype.sum_sum_type]
  80   cases a with
  81   | inl a =>
  82     simp only [demandMatchingMatrix, Finset.sum_const_zero, zero_add]
  83     have hf : (0 : ℝ) < f a.1 := by exact_mod_cast Nat.zero_lt_of_lt a.2.isLt
  84     calc
  85       _ = (∑ e, if tail e = a.1 then z e else 0) / f a.1 := by
  86         rw [Finset.sum_div]
  87         apply Finset.sum_congr rfl
  88         intro e _
  89         by_cases he : tail e = a.1
  90         · simp only [he, ite_true]
  91         · simp only [he, Ne.symm he, ite_false, zero_div]
  92       _ = 1 := by rw [hrow, div_self hf.ne']
  93   | inr e =>
  94     simp only [demandMatchingMatrix]
  95     rw [sum_slots_indicator]
  96     simp only [Finset.sum_ite_eq, Finset.mem_univ, ite_true]
  97     by_cases hg : g (head e) = 0
  98     · rw [demand_zero_forces_edge_zero head g z hz hcol e hg]
  99       simp
 100     · rw [mul_div_cancel₀ _ (by exact_mod_cast hg)]
 101       ring

 103 theorem demandMatchingMatrix_transpose {X Y E : Type*}
 104     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ) (z : E → ℝ)
 105     (a : DemandSlots f ⊕ E) (b : DemandSlots g ⊕ E) :
 106     demandMatchingMatrix tail head f g z a b = demandMatchingMatrix head tail g f z b a := by
 107   cases a <;> cases b <;> simp only [demandMatchingMatrix, eq_comm]
 108   rename_i e e'
 109   by_cases he : e = e'
 110   · subst e'; simp
 111   · simp [he, Ne.symm he]

 113 theorem demandMatchingMatrix_column {X Y E : Type*}
 114     [Fintype X] [Fintype Y] [Fintype E]
 115     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ) (z : E → ℝ)
 116     (hz : ∀ e, 0 ≤ z e)
 117     (hrow : ∀ x, (∑ e, if tail e = x then z e else 0) = f x)
 118     (hcol : ∀ y, (∑ e, if head e = y then z e else 0) = g y)
 119     (b : DemandSlots g ⊕ E) : ∑ a, demandMatchingMatrix tail head f g z a b = 1 := by

 121     _ = ∑ a, demandMatchingMatrix head tail g f z b a :=

 125 theorem demandMatching_parts_card {X Y E : Type*}
 126     [Fintype X] [Fintype Y] [Fintype E]
 127     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ) (z : E → ℝ)
 128     (hrow : ∀ x, (∑ e, if tail e = x then z e else 0) = f x)
 129     (hcol : ∀ y, (∑ e, if head e = y then z e else 0) = g y) :
 130     Fintype.card (DemandSlots f ⊕ E) = Fintype.card (DemandSlots g ⊕ E) := by
 131   classical
 132   have hf : (∑ x, (f x : ℝ)) = ∑ e, z e := by
 133     simp_rw [← hrow]
 134     rw [Finset.sum_comm]
 135     simp
 136   have hg : (∑ y, (g y : ℝ)) = ∑ e, z e := by
 137     simp_rw [← hcol]
 138     rw [Finset.sum_comm]
 139     simp
 140   have htot : ∑ x, f x = ∑ y, g y := by exact_mod_cast hf.trans hg.symm
 141   simp only [Fintype.card_sum, Fintype.card_sigma, Fintype.card_fin, DemandSlots]
 142   rw [htot]
