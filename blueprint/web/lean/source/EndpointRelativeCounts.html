  16 theorem mem_common_iff {I : Type*} (A : Finset V) (X : Finset I) (N : I→Finset V) (v : V) :
  17     v∈common A X N ↔ v∈A ∧ ∀ i∈X,v∈N i := by simp only [common,Finset.mem_filter]

  19 theorem common_neighbour_deletion_lower (P F : SimpleGraph V) (A X : Finset V) (δ L : ℝ)
  20     (hdeg : ∀ v∈X,δ*A.card≤((A∩P.neighborFinset v).card:ℝ))
  21     (hF : ∀ v∈X,(F.degree v:ℝ)≤L) :
  22     (1-(X.card:ℝ)*(1-δ))*A.card-X.card*L≤
  23       ((common A X (fun v=>(P\F).neighborFinset v)).card:ℝ) := by
  24   let D := X.biUnion (fun v=>F.neighborFinset v)
  25   have hsub : common A X (fun v=>P.neighborFinset v)⊆
  26       common A X (fun v=>(P\F).neighborFinset v)∪D := by
  27     intro z hz
  28     by_cases hd : z∈D
  29     · exact Finset.mem_union_right _ hd
  30     · apply Finset.mem_union_left
  31       obtain ⟨hzA,hzP⟩ := (mem_common_iff _ _ _ _).mp hz
  32       apply (mem_common_iff _ _ _ _).mpr
  33       refine ⟨hzA,?_⟩
  34       intro v hv
  35       simp only [SimpleGraph.mem_neighborFinset,SimpleGraph.sdiff_adj]
  36       refine ⟨(P.mem_neighborFinset v z).mp (hzP v hv),?_⟩
  37       intro h
  38       exact hd (Finset.mem_biUnion.mpr ⟨v,hv,(F.mem_neighborFinset v z).mpr h⟩)
  39   have hD : (D.card:ℝ)≤X.card*L := by
  40     calc
  41       _ ≤ ∑ v∈X,((F.neighborFinset v).card:ℝ) := by exact_mod_cast (Finset.card_biUnion_le : D.card≤_)
  42       _ ≤ ∑ _v∈X,L := Finset.sum_le_sum (fun v hv=>by simpa only [SimpleGraph.card_neighborFinset_eq_degree] using hF v hv)
  43       _ = _ := by simp
  44   have hcard : ((common A X (fun v=>P.neighborFinset v)).card:ℝ)≤
  45       (common A X (fun v=>(P\F).neighborFinset v)).card+D.card := by
  46     exact_mod_cast (Finset.card_le_card hsub).trans (Finset.card_union_le _ _)
  47   linarith only [common_neighbour_card_lower A X (fun v=>P.neighborFinset v) δ hdeg,hD,hcard]

  49 theorem endpoint_surviving_common_lower (P F : SimpleGraph V) (A X : Finset V) (L : ℝ)
  50     (hX : X.card≤3) (hL : 0≤L) (hLa : L≤(A.card:ℝ)/24)
  51     (hdeg : ∀ v∈X,(3/4:ℝ)*A.card≤((A∩P.neighborFinset v).card:ℝ))
  52     (hF : ∀ v∈X,(F.degree v:ℝ)≤L) :
  53     (A.card:ℝ)/8≤((common A X (fun v=>(P\F).neighborFinset v)).card:ℝ) := by
  54   have hh := common_neighbour_deletion_lower P F A X (3/4) L hdeg hF
  55   have hXr : (X.card:ℝ)≤3 := by exact_mod_cast hX
  56   have hcoef : 1-(X.card:ℝ)*(1-(3/4:ℝ))≥1/4 := by linarith only [hXr]
  57   have hmul := mul_le_mul_of_nonneg_right hcoef (Nat.cast_nonneg A.card)
  58   have hloss := mul_le_mul_of_nonneg_right hXr hL
  59   nlinarith only [hh,hmul,hloss,hLa]

  61 theorem endpoint_relative_row_lower (P R H F : SimpleGraph V) (A X J : Finset V)
  62     (p q s ε L : ℝ) (hp : 0<p) (hp1 : p≤1) (hq : p≤q) (hs : p≤s)
  63     (hX : X.card≤3) (hXJ : X.card+J.card≤4) (hL : 0≤L) (hLa : L≤(A.card:ℝ)/24)
  64     (hε : ε≤p^4/16)
  65     (hdeg : ∀ v∈X,(3/4:ℝ)*A.card≤((A∩P.neighborFinset v).card:ℝ))
  66     (hF : ∀ v∈X,(F.degree v:ℝ)≤L)
  67     (hrelative : q^X.card*s^J.card*((common A X (fun v=>(P\F).neighborFinset v)).card:ℝ)-ε*A.card≤
  68       ((common (common A X (fun v=>R.neighborFinset v)) J (fun v=>H.neighborFinset v)).card:ℝ)) :
  69     (p^4/16)*A.card≤
  70       ((common (common A X (fun v=>R.neighborFinset v)) J (fun v=>H.neighborFinset v)).card:ℝ) := by
  71   have hc := endpoint_surviving_common_lower P F A X L hX hL hLa hdeg hF
  72   have hprod : p^4≤q^X.card*s^J.card := by
  73     calc
  74       _ ≤ p^(X.card+J.card) := pow_le_pow_of_le_one hp.le hp1 hXJ
  75       _ = p^X.card*p^J.card := pow_add _ _ _
  76       _ ≤ _ := mul_le_mul (pow_le_pow_left₀ hp.le hq _) (pow_le_pow_left₀ hp.le hs _)
  77         (pow_nonneg hp.le _) (pow_nonneg (hp.le.trans hq) _)
  78   have hm := mul_le_mul hprod hc (by positivity : (0:ℝ)≤A.card/8)
  79     (mul_nonneg (pow_nonneg (hp.le.trans hq) _) (pow_nonneg (hp.le.trans hs) _))
  80   have he := mul_le_mul_of_nonneg_right hε (Nat.cast_nonneg A.card)
  81   nlinarith only [hm,he,hrelative]

  83 theorem endpoint_relative_index_lower (H : SimpleGraph V) (B X : Finset V)
  84     (p s ε : ℝ) (hp : 0<p) (hp1 : p≤1) (hs : p≤s) (hX : X.card≤3)
  85     (hε : ε≤p^3/2)
  86     (hrelative : (s^X.card-ε)*B.card≤((common B X (fun v=>H.neighborFinset v)).card:ℝ)) :
  87     (p^3/2)*B.card≤((common B X (fun v=>H.neighborFinset v)).card:ℝ) := by
  88   have hpow : p^3≤s^X.card := (pow_le_pow_of_le_one hp.le hp1 hX).trans (pow_le_pow_left₀ hp.le hs _)

  90     (Nat.cast_nonneg _)).trans hrelative
