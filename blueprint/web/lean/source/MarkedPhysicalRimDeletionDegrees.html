  14 theorem deleted_pair_outdegree {I V:Type} [Fintype I] [Fintype V] [DecidableEq V]
  15     (a c:I→V) (hinj:Function.Injective (fun i=>({a i,c i}:Finset V))) (D:SimpleGraph V) (x:V):
  16     (∑ i,if D.Adj (a i) (c i)∧a i=x then (1:ℝ) else 0)≤D.degree x:=by
  17   have hh:(Finset.univ.filter (fun i=>D.Adj (a i) (c i)∧a i=x)).card≤(D.neighborFinset x).card:=by
  18     apply Finset.card_le_card_of_injOn c
  19     · intro i hi
  20       obtain ⟨_,hi,he⟩:=Finset.mem_filter.mp hi
  21       change c i∈D.neighborFinset x
  22       apply (D.mem_neighborFinset x (c i)).mpr
  23       exact he ▸ hi
  24     · intro i hi j hj hc
  25       obtain ⟨_,hi,he⟩:=Finset.mem_filter.mp hi
  26       obtain ⟨_,hj,hf⟩:=Finset.mem_filter.mp hj
  27       apply hinj
  28       change ({a i,c i}:Finset V)={a j,c j}
  29       rw [he,hf,hc]
  30   rw [SimpleGraph.card_neighborFinset_eq_degree] at hh
  31   have hsum:(∑ i,if D.Adj (a i) (c i)∧a i=x then (1:ℝ) else 0)=
  32       ((Finset.univ.filter (fun i=>D.Adj (a i) (c i)∧a i=x)).card:ℝ):=by
  33     simp only [Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  34   rw [hsum]
  35   exact_mod_cast hh

  46 theorem marked_physical_rim_deletion_degrees {V E K:Type} [Fintype V] [Fintype E] [Fintype K]
  47     [DecidableEq V] (tail head:E→V) (ktail khead:K→V) (π:Equiv.Perm V)
  48     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π)) (D:SimpleGraph V)
  49     (d:ℝ) (hD:∀ v,(D.degree v:ℝ)≤d):
  50     (∀ v,(∑ e,if D.Adj (tail e) (head e)∧tail e=v then (1:ℝ) else 0)≤d)∧
  51     (∀ v,(∑ e,if D.Adj (tail e) (head e)∧head e=v then (1:ℝ) else 0)≤d)∧
  52     (∀ v,(∑ e,if D.Adj (ktail e) (π (khead e))∧ktail e=v then (1:ℝ) else 0)≤d)∧
  53     (∀ v,(∑ e,if D.Adj (ktail e) (π (khead e))∧khead e=v then (1:ℝ) else 0)≤d):=by
  54   have hO:Function.Injective (fun e:E=>({tail e,head e}:Finset V)):=by
  55     intro e f he
  56     exact Sum.inl.inj (hp (show markedPhysicalEdge tail head ktail khead π (Sum.inl e)=
  57       markedPhysicalEdge tail head ktail khead π (Sum.inl f) from he))
  58   have hK:Function.Injective (fun e:K=>({ktail e,π (khead e)}:Finset V)):=by
  59     intro e f he
  60     exact Sum.inr.inj (hp (show markedPhysicalEdge tail head ktail khead π (Sum.inr e)=
  61       markedPhysicalEdge tail head ktail khead π (Sum.inr f) from he))
  62   refine ⟨fun v=>(deleted_pair_outdegree tail head hO D v).trans (hD v),
  63     fun v=>(deleted_pair_indegree tail head hO D v).trans (hD v),
  64     fun v=>(deleted_pair_outdegree ktail (fun e=>π (khead e)) hK D v).trans (hD v),?_⟩
  65   intro v
  66   simpa only [Equiv.apply_eq_iff_eq] using
  67     (deleted_pair_indegree ktail (fun e=>π (khead e)) hK D (π v)).trans (hD (π v))
