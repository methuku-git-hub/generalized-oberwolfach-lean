  13 theorem triangle_transformer_sample_factors (z : ℕ) (hz : 3∣z) (β ρ : ℝ)
  14     (hβ : 0<β) (hβ1 : β≤1) (hρ : 0<ρ) :
  15     ∃ s₀ : ℕ,∀ s : ℕ,s₀≤s → 3∣s → ∃ N : ℕ,∀ (V : Type) [Fintype V] [DecidableEq V],
  16       N≤Fintype.card V → ∀ (G : SimpleGraph V) (A Z : Finset V),Z.card=z →
  17       ρ*Fintype.card V≤(A.card:ℝ) →
  18       (∀ v∈A∪Z,(3/4+β/2)*(A.card:ℝ)≤((G.neighborFinset v∩A).card:ℝ)) →
  19       ∃ S : Finset V,S.card=s ∧ S⊆A ∧ Disjoint S Z ∧
  20       ∃ M₁ M₂ : Finset (OrdinaryCycle G 3),
  21         (∀ C∈M₁,∀ D∈M₁,C≠D → Disjoint (ordinaryCycleVertices G 3 C) (ordinaryCycleVertices G 3 D)) ∧
  22         M₁.biUnion (ordinaryCycleVertices G 3)=S∪Z ∧
  23         (∀ C∈M₂,∀ D∈M₂,C≠D → Disjoint (ordinaryCycleVertices G 3 C) (ordinaryCycleVertices G 3 D)) ∧
  24         M₂.biUnion (ordinaryCycleVertices G 3)=S ∧
  25         (∀ C∈M₁,∀ D∈M₂,Disjoint C.val.val.edgeFinset D.val.val.edgeFinset) ∧
  26         (∀ C∈M₁,∀ v w,C.val.val.Adj v w → v∈S ∨ w∈S) := by
  27   obtain ⟨K,hK⟩:=triangle_transformer_sample z β ρ hβ hβ1 hρ
  28   obtain ⟨L,hL⟩:=two_triangle_factors_at_hub (β/2) (by positivity) z
  29   refine ⟨max K L,?_⟩
  30   intro s hs hdiv
  31   obtain ⟨N,hN⟩:=hK s ((le_max_left K L).trans hs)
  32   refine ⟨N,?_⟩
  33   intro V _ _ hn G A Z hZ hA hdeg
  34   obtain ⟨S,hScard,hSA,hSZ,hSdeg⟩:=hN V hn G A Z hZ hA hdeg
  35   let T:=S∪Z
  36   let W:Type:=(T:Set V)
  37   let Q:=G.induce (T:Set V)
  38   let S' : Finset W:=Finset.univ.filter (fun v=>v.val∈S)
  39   let f : W↪V:=Function.Embedding.subtype _
  40   have himage : S'.image f=S := by
  41     ext v
  42     constructor
  43     · intro hv
  44       obtain ⟨w,hw,rfl⟩:=Finset.mem_image.mp hv
  45       exact (Finset.mem_filter.mp hw).2
  46     · intro hv
  47       exact Finset.mem_image.mpr ⟨⟨v,Finset.mem_union_left _ hv⟩,
  48         Finset.mem_filter.mpr ⟨Finset.mem_univ _,hv⟩,rfl⟩
  49   have hS' : S'.card=s := by
  50     rw [←Finset.card_image_of_injective S' f.injective,himage,hScard]
  51   have hW : Fintype.card W=s+z := by
  52     have hc : Fintype.card W=T.card := by
  53       convert Fintype.card_coe T using 1 <;> congr!
  54     rw [hc]
  55     simp only [T,Finset.card_union_of_disjoint hSZ,hScard,hZ]
  56   have hQdeg (v : W) : (Q.neighborFinset v∩S').card=(G.neighborFinset v.val∩S).card := by
  57     apply Finset.card_bij (fun w _=>w.val)
  58     · intro w hw
  59       have hh:=Finset.mem_inter.mp hw
  60       have hq : Q.Adj v w := by simpa using hh.1
  61       have hq' : G.Adj v.val w.val:=hq
  62       exact Finset.mem_inter.mpr ⟨by simpa using hq',(Finset.mem_filter.mp hh.2).2⟩
  63     · intro w _ u _ h;exact Subtype.ext h
  64     · intro w hw
  65       have hh:=Finset.mem_inter.mp hw
  66       let u : W:=⟨w,Finset.mem_union_left _ hh.2⟩
  67       refine ⟨u,Finset.mem_inter.mpr ⟨?_,Finset.mem_filter.mpr ⟨Finset.mem_univ _,hh.2⟩⟩,rfl⟩
  68       have hg : G.Adj v.val w := by simpa using hh.1
  69       have hq : Q.Adj v u:=hg
  70       simpa using hq
  71   obtain ⟨M₁,M₂,hpack₁,hcover₁,hpack₂,hcover₂,hcross,hold⟩:=hL W Q S'
  72     (by rw [hS'];exact (le_max_right K L).trans hs) (by rwa [hS'])
  73     (by rw [hW];exact Nat.dvd_add hdiv hz) (by rw [hW,hS'];omega)
  74     (by intro v;rw [hS',hQdeg];convert hSdeg v.val v.property using 1 <;> ring)
  75   have hQ : Q.map f≤G := by
  76     intro v w h
  77     rw [SimpleGraph.map_adj] at h
  78     obtain ⟨a,b,hab,rfl,rfl⟩:=h
  79     exact hab
  80   let lift:=ordinaryCycleLift f hQ (r:=3)
  81   have hfirst:=ordinary_cycle_family_lift f hQ M₁ hpack₁
  82   have hsecond:=ordinary_cycle_family_lift f hQ M₂ hpack₂
  83   have hall : Finset.univ.image f=T := by
  84     ext v
  85     constructor
  86     · intro hv
  87       obtain ⟨w,_,rfl⟩:=Finset.mem_image.mp hv
  88       exact w.property
  89     · intro hv
  90       exact Finset.mem_image.mpr ⟨⟨v,hv⟩,Finset.mem_univ _,rfl⟩
  91   refine ⟨S,hScard,hSA,hSZ,M₁.image lift,M₂.image lift,hfirst.1,?_,hsecond.1,?_,?_,?_⟩
  92   · rw [hfirst.2,hcover₁,hall]
  93   · rw [hsecond.2,hcover₂,himage]
  94   · intro C hC D hD
  95     obtain ⟨C,hCM,rfl⟩:=Finset.mem_image.mp hC
  96     obtain ⟨D,hDM,rfl⟩:=Finset.mem_image.mp hD
  97     exact ordinaryCycleLift_disjoint_edges f hQ C D (hcross C hCM D hDM)
  98   · intro C hC v w h
  99     obtain ⟨C,hCM,rfl⟩:=Finset.mem_image.mp hC
 100     change (C.val.val.map f).Adj v w at h
 101     rw [SimpleGraph.map_adj] at h
 102     obtain ⟨a,b,hab,rfl,rfl⟩:=h
 103     rcases hold C hCM a b hab with ha|hb
 104     · exact Or.inl (Finset.mem_filter.mp ha).2
 105     · exact Or.inr (Finset.mem_filter.mp hb).2
