  16 theorem ordinary_hub_cover_avoids_other_rows (M:Finset (OrdinaryWheel Q F H r))
  17     (hr:0<r) (u:Fin n→U) (hu:Function.Injective u) (T:Fin n→Finset V)
  18     (X:Finset V) (hX:∀ i,T i⊆X)
  19     (hunion:M.biUnion ordinaryWheelResources=(ordinaryWheelRimGraph M).edgeFinset.image Sum.inl ∪
  20       Finset.univ.biUnion (fun i=>(T i).image (fun v=>Sum.inr (v,u i)))):
  21     (∀ v,v∉X→∀ w,¬(ordinaryWheelRimGraph M).Adj v w) ∧
  22     (∀ v,v∉X→∀ a,¬ordinaryPackingSpokes M v a):=by
  23   constructor
  24   · intro v hv w hvw
  25     obtain ⟨W,hW,hh⟩:=(ordinaryWheelRimGraph_edges M s(v,w)).mp hvw
  26     obtain ⟨i,_,hi⟩:=ordinary_hub_packing_wheel_support M hr u hu T hunion W hW
  27     exact hv (hX i (hi v hh.mem_support_left))
  28   · intro v hv a hh
  29     obtain ⟨i,_,hi⟩:=(ordinary_hub_packing_spokes M u T hunion v a).mp hh
  30     exact hv (hX i hi)

  34     (X:Finset V) (hX:∀ i,T i⊆X)

  47 theorem ordinary_hub_deletion_fixed_domain (M:Finset (OrdinaryWheel Q F H r))
  48     (u:Fin n→U) (T:Fin n→Finset V) (A:Finset V) (B:Finset U) (hB:∀ i,u i∉B)
  49     (hunion:M.biUnion ordinaryWheelResources=(ordinaryWheelRimGraph M).edgeFinset.image Sum.inl ∪
  50       Finset.univ.biUnion (fun i=>(T i).image (fun v=>Sum.inr (v,u i))))
  51     (hdeg:∀ v,(ordinaryWheelRimGraph M).degree v≤2*n):
  52     (∀ v,(changed (A∩Q.neighborFinset v) (A∩(Q\ordinaryWheelRimGraph M).neighborFinset v)).card≤2*n) ∧
  53     (∀ v,B∩Finset.univ.filter (fun a=>H v a∧¬ordinaryPackingSpokes M v a)=B∩Finset.univ.filter (H v)) ∧
  54     (∀ a∈B,A∩Finset.univ.filter (fun v=>H v a∧¬ordinaryPackingSpokes M v a)=A∩Finset.univ.filter (fun v=>H v a)):=by
  55   have hno (v:V) (a:U) (ha:a∈B):¬ordinaryPackingSpokes M v a:=by
  56     intro hh
  57     obtain ⟨i,rfl,_⟩:=(ordinary_hub_packing_spokes M u T hunion v a).mp hh
  58     exact hB i ha
  59   refine ⟨?_,?_,?_⟩
  60   · intro v
  61     have hs:changed (A∩Q.neighborFinset v) (A∩(Q\ordinaryWheelRimGraph M).neighborFinset v)⊆
  62         (ordinaryWheelRimGraph M).neighborFinset v:=by
  63       intro w hw
  64       simp only [changed,Finset.mem_union,Finset.mem_sdiff,Finset.mem_inter,
  65         SimpleGraph.mem_neighborFinset,SimpleGraph.sdiff_adj] at hw ⊢
  66       tauto
  67     exact (Finset.card_le_card hs).trans (by simpa only [SimpleGraph.card_neighborFinset_eq_degree] using hdeg v)
  68   · intro v
  69     ext a
  70     simp only [Finset.mem_inter,Finset.mem_filter,Finset.mem_univ,true_and]
  71     constructor
  72     · rintro ⟨ha,hva,_⟩;exact ⟨ha,hva⟩
  73     · rintro ⟨ha,hva⟩;exact ⟨ha,hva,hno v a ha⟩
  74   · intro a ha
  75     ext v
  76     simp only [Finset.mem_inter,Finset.mem_filter,Finset.mem_univ,true_and,hno v a ha,not_false_eq_true,and_true]

  78 theorem ordinary_wheel_avoids_removed_packing (M:Finset (OrdinaryWheel Q F H r))
  79     (W:OrdinaryWheel (Q\ordinaryWheelRimGraph M) F
  80       (fun v a=>H v a∧¬ordinaryPackingSpokes M v a) r):
  81     Disjoint (ordinaryWheelResources W) (M.biUnion ordinaryWheelResources):=by
  82   apply Finset.disjoint_left.mpr
  83   intro e he heM
  84   obtain ⟨Z,hZ,heZ⟩:=Finset.mem_biUnion.mp heM
  85   cases e with
  86   | inl e=>
  87     have hw:=(ordinaryWheelResources_rim W e).mp he
  88     have hz:=(ordinaryWheelResources_rim Z e).mp heZ
  89     have hD:(ordinaryWheelRimGraph M).edgeSet e:=
  90       (ordinaryWheelRimGraph_edges M e).mpr ⟨Z,hZ,hz⟩
  91     have hQ:=SimpleGraph.edgeSet_mono W.val.1.property hw
  92     have hsd:(Q\ordinaryWheelRimGraph M).edgeSet=Q.edgeSet\(ordinaryWheelRimGraph M).edgeSet:=by
  93       ext a
  94       induction a using Sym2.inductionOn with
  95       | _ v w=>simp only [SimpleGraph.mem_edgeSet,SimpleGraph.sdiff_adj,Set.mem_sdiff]
  96     rw [hsd] at hQ
  97     exact hQ.2 hD
  98   | inr va=>
  99     obtain ⟨hW,hw⟩:=(ordinaryWheelResources_spoke W va.1 va.2).mp he
 100     obtain ⟨hZhub,hz⟩:=(ordinaryWheelResources_spoke Z va.1 va.2).mp heZ
 101     exact (W.property.1 va.1 hw).2 ⟨Z,hZ,hZhub.trans hW.symm,hz⟩
