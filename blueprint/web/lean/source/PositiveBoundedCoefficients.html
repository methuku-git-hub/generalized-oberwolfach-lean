  10 theorem subspace_positive_bounded_coefficients
  11     {I E : Type*} [Fintype I] [Fintype E]
  12     (S : Submodule ℝ (E → ℝ)) (P : I → E → ℝ) (hP : ∀ i,P i∈S)
  13     (center w : E → ℝ) (hw : w∈S)
  14     (κ ε B b : ℝ) (hκ : 0<κ) (hb : 0<b) (hε : ε<κ/4)
  15     (hwclose : ‖w-center‖ ≤ κ/4)
  16     (hpre : ‖∑ i,b • P i‖ ≤ κ/4)
  17     (happrox : ∀ w'∈S,‖w'-center‖ ≤ κ → ∃ a : I → ℝ,
  18       (∀ i,0≤a i ∧ a i≤B) ∧ ‖(∑ i,a i • P i)-w'‖ ≤ ε) :
  19     ∃ a : I → ℝ,(∀ i,0<a i ∧ b≤a i ∧ a i≤b+B) ∧ (∑ i,a i • P i)=w := by
  20   let v := ∑ i,b • P i
  21   have hv : v∈S := S.sum_mem (fun i _ => S.smul_mem b (hP i))
  22   have hwv : w-v∈S := S.sub_mem hw hv
  23   obtain ⟨a,ha,he⟩ := subspace_bounded_coefficients_exact S P hP (w-v) hwv B (κ/4) ε
  24     (by positivity) hε (by
  25       intro w' hw' hclose
  26       apply happrox w' hw'
  27       have heq : w'-center=(w'-(w-v))+(w-center)-v := by abel
  28       rw [heq]
  29       have hn := (norm_sub_le ((w'-(w-v))+(w-center)) v).trans
  30         (add_le_add (norm_add_le (w'-(w-v)) (w-center)) (le_refl ‖v‖))
  31       have hvnorm : ‖v‖ ≤ κ/4 := hpre
  32       linarith)
  33   refine ⟨fun i => b+a i,?_,?_⟩
  34   · intro i
  35     exact ⟨by linarith [(ha i).1],by linarith [(ha i).1],by linarith [(ha i).2]⟩
  36   · simp only [add_smul,Finset.sum_add_distrib,he]
  37     change v+(w-v)=w
  38     abel
