  15 theorem marked_spoke_reference_change_after_deletion_on_fixed_domains
  16     (π : Equiv.Perm V) (d : ℕ) (σ : RawMarkedOutcome V U Γ Color)
  17     (g : Γ) (j : Color) (H : Bool→V→U→Prop)
  18     (DR : SimpleGraph V) (DS : SimpleGraph (V⊕U)) (A:Finset V) (B:Finset U) (eR eC δ : ℝ)
  19     (hrow : ∀ b x,(∑ u∈B,if ¬(rawMarkedPairSpokes π d σ g j b x u↔H b x u) then (1:ℝ) else 0)≤eR)
  20     (hcol : ∀ b u,(∑ x∈A,if ¬(rawMarkedPairSpokes π d σ g j b x u↔H b x u) then (1:ℝ) else 0)≤eC)
  21     (hDS : ∀ z,(DS.degree z:ℝ)≤δ) :
  22     (∀ b x,(∑ u∈B,if ¬(rawMarkedPairSpokes π d (rawMarkedDeleteRims σ DR.edgeFinset) g j b x u↔
  23       markedDeletePhysicalSpokes H DS b x u) then (1:ℝ) else 0)≤eR+δ) ∧
  24     (∀ b u,(∑ x∈A,if ¬(rawMarkedPairSpokes π d (rawMarkedDeleteRims σ DR.edgeFinset) g j b x u↔
  25       markedDeletePhysicalSpokes H DS b x u) then (1:ℝ) else 0)≤eC+δ) := by
  26   have tri (a b c : Prop) : (if ¬(a↔c) then (1:ℝ) else 0)≤
  27       (if ¬(a↔b) then (1:ℝ) else 0)+(if ¬(b↔c) then (1:ℝ) else 0) := by
  28     by_cases ha:a <;> by_cases hb:b <;> by_cases hc:c <;> simp [ha,hb,hc]
  29   constructor
  30   · intro b x
  31     have hchange := (marked_spoke_physical_deletion_row_change H DS B b x).trans (hDS (Sum.inl x))
  32     calc
  33       _ ≤ (∑ u∈B,if ¬(rawMarkedPairSpokes π d σ g j b x u↔H b x u) then (1:ℝ) else 0)+
  34           (∑ u∈B,if ¬(H b x u↔markedDeletePhysicalSpokes H DS b x u) then (1:ℝ) else 0) := by
  35         rw [←Finset.sum_add_distrib]
  36         apply Finset.sum_le_sum
  37         intro u _
  38         exact tri (rawMarkedPairSpokes π d σ g j b x u) (H b x u) (markedDeletePhysicalSpokes H DS b x u)
  39       _ ≤ eR+δ := add_le_add (hrow b x) hchange
