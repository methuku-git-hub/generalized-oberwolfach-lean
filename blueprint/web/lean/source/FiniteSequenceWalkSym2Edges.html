  10 theorem finite_sequence_walk_sym2_edge_set
  11     {V:Type*} [DecidableEq V] (G:SimpleGraph V) (n:ℕ) (f:Fin (n+1)→V)
  12     {a b:V} (p:G.Walk a b) (hlen:p.length=n) (hsupp:p.support=List.ofFn f):
  13     p.edges.toFinset=Finset.univ.image (fun i:Fin n=>s(f i.castSucc,f i.succ)):=by
  14   classical
  15   have hv:∀j:ℕ,(hj:j≤n)→p.getVert j=f ⟨j,by omega⟩:=by
  16     intro j hj
  17     have hh:=p.getVert_eq_support_getElem? (n:=j) (by omega)
  18     rw [hsupp] at hh
  19     simpa only [List.getElem?_ofFn,dif_pos (show j<n+1 by omega),Option.some.injEq] using hh
  20   have hedge (i:Fin n):
  21       p.edges[i.val]'(by simpa [SimpleGraph.Walk.length_edges,hlen] using i.isLt)=s(f i.castSucc,f i.succ):=by
  22     have hi:i.val<p.edges.length:=by simpa [SimpleGraph.Walk.length_edges,hlen] using i.isLt
  23     rw [SimpleGraph.Walk.getElem_edges hi,hv i.val (by omega),hv (i.val+1) (by omega)]
  24     have h1:(⟨i.val,by omega⟩:Fin (n+1))=i.castSucc:=Fin.ext rfl
  25     have h2:(⟨i.val+1,by omega⟩:Fin (n+1))=i.succ:=Fin.ext rfl
  26     rw [h1,h2]
  27   ext z
  28   simp only [List.mem_toFinset,Finset.mem_image,Finset.mem_univ,true_and]
  29   constructor
  30   · intro hz
  31     obtain ⟨i,hi,he⟩:=List.mem_iff_getElem.mp hz
  32     have hin:i<n:=by simpa [SimpleGraph.Walk.length_edges,hlen] using hi
  33     refine ⟨⟨i,hin⟩,?_⟩
  34     exact (hedge ⟨i,hin⟩).symm.trans he
  35   · rintro ⟨i,he⟩
  36     have hi:i.val<p.edges.length:=by simpa [SimpleGraph.Walk.length_edges,hlen] using i.isLt
  37     exact he ▸ (hedge i ▸ List.getElem_mem hi)
  38 end Oberwolfach
