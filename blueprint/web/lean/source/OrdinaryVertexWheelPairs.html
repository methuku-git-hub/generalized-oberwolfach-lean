  16  structure OrdinaryVertexWheelPair (Q F:SimpleGraph V) (H:V→U→Prop)
  17     (r:ℕ) (x y a b:V) (u:U) where
  18   first : OrdinaryWheel Q F H r
  19   second : OrdinaryWheel Q F H r
  20   first_hub : ordinaryWheelHub first=u
  21   second_hub : ordinaryWheelHub second=u
  22   first_a : (ordinaryWheelRim first).Adj x a
  23   first_b : (ordinaryWheelRim first).Adj x b
  24   second_a : (ordinaryWheelRim second).Adj y a
  25   second_b : (ordinaryWheelRim second).Adj y b
  26   second_rim : ordinaryWheelRim second=(ordinaryWheelRim first).map (Equiv.swap x y).toEmbedding
  27   second_support : ∀ v,v∈(ordinaryWheelRim second).support↔v=y∨(v∈(ordinaryWheelRim first).support∧v≠x)
  28   rim_difference : ∀ e:Sym2 V,
  29     (if Sum.inl e∈ordinaryWheelResources first then (1:ℤ) else 0)-
  30       (if Sum.inl e∈ordinaryWheelResources second then 1 else 0)=
  31     (if e∈({s(x,a),s(x,b)}:Finset (Sym2 V)) then 1 else 0)-
  32       (if e∈({s(y,a),s(y,b)}:Finset (Sym2 V)) then 1 else 0)
  33   spoke_difference : ∀ v z,
  34     (if Sum.inr (v,z)∈ordinaryWheelResources first then (1:ℤ) else 0)-
  35       (if Sum.inr (v,z)∈ordinaryWheelResources second then 1 else 0)=
  36     (if v=x∧z=u then 1 else 0)-(if v=y∧z=u then 1 else 0)

  55         (if Sum.inr (v,z)∈ordinaryWheelResources C' then 1 else 0)=

  61     ⟨C,C',hCu,hC'u,hCxa,hCxb,hC'ya,hC'yb,hmap,hsupport,hrim,hsp⟩
  62   refine ⟨D,?_⟩

  65     simpa only [rootWheelInternal,Finset.mem_sdiff,Set.mem_toFinset] using hv
  66   exact (hclass v hv'.1).resolve_left hv'.2

  93       have hu:((W∪B).card:ℝ)≤W.card+B.card:=by exact_mod_cast Finset.card_union_le W B
