  18 theorem fixedPartition_two_placement {V : Type*} [Fintype V]
  19     {k : ℕ} {d : Fin k → ℕ} [Nonempty (FixedPartition V k d)]
  20     (x y : V) (hxy : x ≠ y) (a b : Fin k) (hab : a ≠ b) :
  21     (FiniteLaw.uniform (FixedPartition V k d)).probability
  22       (fun P => P.1 x = a ∧ P.1 y = b) =
  23       ((d a : ℝ) * d b) / ((Fintype.card V : ℝ) * (Fintype.card V - 1)) := by
  24   have hn : 2 ≤ Fintype.card V := by
  25     simpa only [Fintype.card_fin] using Fintype.card_le_of_injective _ (pairEmbedding x y hxy).injective
  26   have h := fixedPartition_joint_placement (d := d) (pairEmbedding x y hxy) (pairEmbedding a b hab)
  27   simpa only [pairEmbedding, Function.Embedding.coeFn_mk, Fin.forall_fin_two,
  28     Matrix.cons_val_zero, Matrix.cons_val_one, Fin.prod_univ_two, Fintype.card_fin,
  29     Nat.descFactorial_succ, Nat.descFactorial_zero, Nat.sub_zero, mul_one,
  30     Nat.cast_mul, Nat.cast_sub (show 1 ≤ Fintype.card V by omega), Nat.cast_one,
  31     mul_comm] using h

  33 theorem fixedPartition_joint_placement_lower {I V : Type*} [Fintype I] [Fintype V]
  34     {k : ℕ} {d : Fin k → ℕ} [Nonempty (FixedPartition V k d)]
  35     (x : I ↪ V) (a : I ↪ Fin k) (m : ℕ) (hm : ∀ i, d (a i) = m)
  36     (hn : 0 < Fintype.card V) :
  37     ((m : ℝ) / Fintype.card V) ^ Fintype.card I ≤
  38       (FiniteLaw.uniform (FixedPartition V k d)).probability
  39         (fun P => ∀ i, P.1 (x i) = a i) := by
  40   rw [fixedPartition_joint_placement]
  41   simp only [hm, Finset.prod_const, Finset.card_univ, Nat.cast_pow, div_pow]
  42   have hinj : Fintype.card I ≤ Fintype.card V := Fintype.card_le_of_injective _ x.injective
  43   have hden : (0 : ℝ) < (Fintype.card V).descFactorial (Fintype.card I) := by
  44     exact_mod_cast Nat.descFactorial_pos.mpr hinj
  45   have hpow : ((Fintype.card V).descFactorial (Fintype.card I) : ℝ) ≤
  46       (Fintype.card V : ℝ) ^ Fintype.card I := by
  47     exact_mod_cast Nat.descFactorial_le_pow (Fintype.card V) (Fintype.card I)
  48   exact div_le_div_of_nonneg_left (by positivity) hden hpow
