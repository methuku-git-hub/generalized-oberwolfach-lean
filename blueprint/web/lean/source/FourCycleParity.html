  12 def fourBarrierAllowed (a b : Fin 3) : Prop :=
  13   ¬ (a = 0 ∧ b = 2) ∧ ¬ (a = 2 ∧ b = 0) ∧ ¬ (a = 1 ∧ b = 1)
  14 
  15 theorem four_cycle_colour_parity :
  16     ∀ c : Fin 4 → Fin 3, (∀ i, fourBarrierAllowed (c i) (c (cycleNext i))) →
  17       Even (∑ i : Fin 4, if c i = 0 ∧ c (cycleNext i) = 0 then (1 : ℕ) else 0) := by
  18   unfold fourBarrierAllowed
  19   decide
  20 
  21 def colourInternalEdge {V : Type*} (c : V → Fin 3) (e : Sym2 V) : Prop :=
  22   ∀ x ∈ e, c x = 0
  23 
  24 theorem colourInternalEdge_pair {V : Type*} (c : V → Fin 3) (u v : V) :
  25     colourInternalEdge c s(u, v) ↔ c u = 0 ∧ c v = 0 := by
  26   simp [colourInternalEdge, Sym2.mem_iff]

  28 theorem IsCycleCopy.four_internal_even {n : ℕ} {H : SimpleGraph (Fin n)}
  29     (hH : IsCycleCopy H 4) (c : Fin n → Fin 3)
  30     (hallowed : ∀ u v, H.Adj u v → fourBarrierAllowed (c u) (c v)) :
  31     Even ((H.edgeFinset.filter (colourInternalEdge c)).card) := by
  32   obtain ⟨hr, f, hf⟩ := hH
  33   rw [cycle_copy_filtered_edge_count hr f hf]
  34   simp only [colourInternalEdge_pair]
  35   apply four_cycle_colour_parity (fun i => c (f i))
  36   intro i
  37   exact hallowed _ _ ((hf _ _).mpr ⟨i, cycleNext i, rfl, rfl, Or.inl rfl⟩)
  38 
  39 theorem HasCycleDecomposition.four_internal_even {n : ℕ} {G : SimpleGraph (Fin n)}
  40     (hG : HasCycleDecomposition G 4) (c : Fin n → Fin 3)
  41     (hallowed : ∀ u v, G.Adj u v → fourBarrierAllowed (c u) (c v)) :
  42     Even ((G.edgeFinset.filter (colourInternalEdge c)).card) := by
  43   obtain ⟨q, H, hpart, hH⟩ := hG
  44   rw [hpart.filtered_edge_count]
  45   apply Finset.even_sum
  46   intro i hi
  47   exact (hH i).four_internal_even c (fun u v huv => hallowed u v (hpart.piece_le i huv))
