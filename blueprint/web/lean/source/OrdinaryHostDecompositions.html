  15  theorem wheelFamily_host_decomposition [Fintype I] (W:I→OrdinaryWheel Q F H r)
  16     (hW:∀ i j,i≠j→Disjoint (ordinaryWheelResources (W i)) (ordinaryWheelResources (W j))):
  17     HasResourceDecomposition (ordinaryWheelResources (P:=Q) (F:=F) (Hbase:=H) (r:=r))
  18       (ordinaryHostResources (wheelFamilyRim W) (wheelFamilySpokes W)):=by
  19   refine ⟨Finset.univ.image W,?_,?_⟩
  20   · intro A hA B hB hAB
  21     obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hA
  22     obtain ⟨j,_,rfl⟩:=Finset.mem_image.mp hB
  23     exact hW i j (fun he=>hAB (congrArg W he))
  24   · rw [wheelFamily_resources]
  25     ext e
  26     simp only [Finset.mem_biUnion,Finset.mem_image,Finset.mem_univ,true_and]
  27     constructor
  28     · rintro ⟨C,⟨i,rfl⟩,he⟩;exact ⟨i,he⟩
  29     · rintro ⟨i,he⟩;exact ⟨W i,⟨i,rfl⟩,he⟩

  38  theorem selected_factor_host_decomposition {T:Finset V}
  39     (M:Finset (LocalOrdinaryCycle Q F T r)) (A:Finset V) (u:U)
  40     (hM:∀ C∈M,∀ C'∈M,C≠C'→Disjoint (localOrdinaryVertices C) (localOrdinaryVertices C'))
  41     (hA:M.biUnion localOrdinaryVertices=A) (hH:∀ v∈A,H v u):
  42     HasResourceDecomposition (ordinaryWheelResources (P:=Q) (F:=F) (Hbase:=H) (r:=r))
  43       (ordinaryHostResources (localOrdinaryFactorGraph M) (fun v z=>v∈A∧z=u)):=by
  44   let j:Unit↪U:=⟨fun _=>u,fun _ _ _=>Subsingleton.elim _ _⟩
  45   obtain ⟨W,_,hWr,hWs,hdis,_⟩:=selected_hub_factor_wheels Q F H r
  46     (fun _:Unit=>T) (fun _:Unit=>A) j (fun _:Unit=>M) (fun _=>hM) (fun _=>hA)
  47     (fun _=>hH) (fun a b h=>False.elim (h (Subsingleton.elim a b)))
  48   have hrim:wheelFamilyRim W=localOrdinaryFactorGraph M:=hWr.trans (by simp)
  49   have hspoke:wheelFamilySpokes W=(fun v z=>v∈A∧z=u):=by
  50     funext v z
  51     apply propext
  52     rw [hWs]
  53     constructor
  54     · rintro ⟨_,hz,hv⟩;exact ⟨hv,hz.symm⟩
  55     · rintro ⟨hv,hz⟩;exact ⟨(),hz.symm,hv⟩
  56   have hh:=wheelFamily_host_decomposition W hdis
  57   rwa [hrim,hspoke] at hh

  63       M₀.biUnion localOrdinaryVertices=A)
  64     (h₁:(∀ C∈M₁,∀ D∈M₁,C≠D→Disjoint (localOrdinaryVertices C) (localOrdinaryVertices D))∧
  65       M₁.biUnion localOrdinaryVertices=A∪Z)
  66     (hdis:Disjoint (localOrdinaryFactorGraph M₀) (localOrdinaryFactorGraph M₁))
  67     (hH:∀ v∈A∪Z,H v u∧H v u'):
  68     let R:=localOrdinaryFactorGraph M₁⊔localOrdinaryFactorGraph M₀
  69     HasResourceDecomposition (ordinaryWheelResources (P:=Q) (F:=F) (Hbase:=H) (r:=r))
  70       (ordinaryHostResources R (fun v z=>(v∈A∪Z∧z=u)∨(v∈A∧z=u')))∧
  71     HasResourceDecomposition (ordinaryWheelResources (P:=Q) (F:=F) (Hbase:=H) (r:=r))
  72       (ordinaryHostResources R (fun v z=>(v∈A∪Z∧z=u')∨(v∈A∧z=u))):=by
  73   dsimp only
  74   have h0u:=selected_factor_host_decomposition (H:=H) M₀ A u h₀.1 h₀.2

  79   have h1u':=selected_factor_host_decomposition (H:=H) M₁ (A∪Z) u' h₁.1 h₁.2 (fun v hv=>(hH v hv).2)
