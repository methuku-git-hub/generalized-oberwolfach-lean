  11 theorem prescribed_subtype_complete_factor_glue
  12     (embed:X↪Y) (A:Finset Y) (A₀ B:Finset X) (ψ:Y→V) (f g Φ:X→V) (L C Z:Finset V)
  13     (hA₀:∀x,x∈A₀↔embed x∈A) (hA₀B:A₀⊆B)
  14     (hψ:Set.InjOn ψ (↑A:Set Y)) (hΦ:Function.Injective Φ)
  15     (hf:∀x∈A₀,f x=ψ (embed x)) (hg:∀x∈A₀,g x=f x) (hΦg:∀x∈B,Φ x=g x)
  16     (himage:B.image g=A₀.image f∪C∪Z)
  17     (hOldC:Disjoint (A.image ψ) C) (hOldZ:Disjoint (A.image ψ) Z)
  18     (hfree:∀x,x∉B→Φ x∈L) (hL:Disjoint L (A.image ψ)):
  19     ∃h:Y→V,Set.InjOn h (↑(A∪Finset.univ.image embed):Set Y) ∧
  20       (∀y∈A,h y=ψ y) ∧ (∀x,h (embed x)=Φ x) := by
  21   have hΦB:Set.InjOn Φ (↑B:Set X):=fun x _ y _ h=>hΦ h
  22   have hselected:∀x∈B\A₀,Φ x∉A.image ψ:=
  23     prescribed_selected_images_avoid_old A₀ B f g Φ (A.image ψ) C Z hA₀B hΦB hg hΦg himage hOldC hOldZ
  24   apply prescribed_subtype_embeddings_glue embed A ψ Φ hψ hΦ ?_ ?_
  25   · intro x hx
  26     have hx₀:x∈A₀:=(hA₀ x).mpr hx
  27     calc
  28       Φ x=g x:=hΦg x (hA₀B hx₀)
  29       _=f x:=hg x hx₀
  30       _=ψ (embed x):=hf x hx₀
  31   · intro x hx
  32     have hx₀:x∉A₀:=by intro h;exact hx ((hA₀ x).mp h)
  33     by_cases hxB:x∈B
  34     · exact hselected x (Finset.mem_sdiff.mpr ⟨hxB,hx₀⟩)
  35     · exact (Finset.disjoint_left.mp hL) (hfree x hxB)
  36 end Oberwolfach
