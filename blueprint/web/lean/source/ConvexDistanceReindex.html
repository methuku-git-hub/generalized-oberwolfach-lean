  13 theorem finiteLaw_convexDistance_reindex
  14     {I J Ω : Type*} [Fintype I] [Fintype J] [Fintype Ω]
  15     (e : I ≃ J) (μ : J → FiniteLaw Ω)
  16     (ih : ∀ B : Finset (I → Ω),
  17       0<(FiniteLaw.pi (fun i=>μ (e i))).probability (fun x=>x∈B) →
  18       (FiniteLaw.pi (fun i=>μ (e i))).expectation
  19         (fun x=>Real.exp (convexDistanceSq B x/4))≤
  20       ((FiniteLaw.pi (fun i=>μ (e i))).probability (fun x=>x∈B))⁻¹)
  21     (A : Finset (J → Ω))
  22     (hA : 0<(FiniteLaw.pi μ).probability (fun x=>x∈A)) :
  23     (FiniteLaw.pi μ).expectation (fun x=>Real.exp (convexDistanceSq A x/4))≤
  24       ((FiniteLaw.pi μ).probability (fun x=>x∈A))⁻¹ := by
  25   classical
  26   let P := Equiv.piCongrLeft (fun _ : J=>Ω) e
  27   let R : (J → Ω) → (I → Ω) := fun x i=>x (e i)
  28   have hRP (x : I → Ω) : R (P x)=x := by
  29     funext i
  30     exact Equiv.piCongrLeft_apply_apply _ _ _ _
  31   have hPR (x : J → Ω) : P (R x)=x := by
  32     funext j
  33     obtain ⟨i,rfl⟩ := e.surjective j
  34     exact Equiv.piCongrLeft_apply_apply _ _ _ _
  35   let B := A.image R
  36   have hmem (x : I → Ω) : x∈B ↔ P x∈A := by
  37     constructor
  38     · intro hx
  39       obtain ⟨y,hy,he⟩ := Finset.mem_image.mp hx
  40       rw [← he,hPR]
  41       exact hy
  42     · intro hx
  43       exact Finset.mem_image.mpr ⟨P x,hx,hRP x⟩
  44   have hp : (FiniteLaw.pi μ).probability (fun x=>x∈A)=
  45       (FiniteLaw.pi (fun i=>μ (e i))).probability (fun x=>x∈B) := by
  46     rw [finiteLaw_pi_reindex_probability e μ]
  47     congr 1
  48     funext x
  49     exact propext (hmem x).symm
  50   have hd (x : I → Ω) : convexDistanceSq B x=convexDistanceSq A (P x) := by
  51     have hh := convexDistanceSq_reindex e A (P x)
  52     change convexDistanceSq B (R (P x))=convexDistanceSq A (P x) at hh
  53     rwa [hRP] at hh
  54   have he : (FiniteLaw.pi μ).expectation (fun x=>Real.exp (convexDistanceSq A x/4))=
  55       (FiniteLaw.pi (fun i=>μ (e i))).expectation
  56         (fun x=>Real.exp (convexDistanceSq B x/4)) := by
  57     rw [finiteLaw_pi_reindex_expectation e μ]
  58     congr 1
  59     funext x
  60     rw [hd]
  61   rw [he,hp]
  62   exact ih B (hp ▸ hA)
