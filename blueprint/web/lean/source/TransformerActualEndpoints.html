  14  def rimIncidenceEquiv (G:SimpleGraph V):
  15     (Σ v,G.neighborSet v)≃(G.edgeFinset×Fin 2):=
  16   (neighborSigmaDartEquiv G).trans (canonicalEdgeDartEquiv G).symm

  18  theorem rimIncidenceEquiv_edge (G:SimpleGraph V) (d:Σ v,G.neighborSet v):
  19     (rimIncidenceEquiv G d).1.val=s(d.1,d.2.val):=
  20   canonicalEdgeDartEquiv_inverse_edge G (neighborSigmaDartEquiv G d)

  22  theorem rimIncidenceEquiv_root (G:SimpleGraph V) (d:Σ v,G.neighborSet v):
  23     d.1=if (rimIncidenceEquiv G d).2=0 then (rimIncidenceEquiv G d).1.val.out.1
  24       else (rimIncidenceEquiv G d).1.val.out.2:=by
  25   have hh:(canonicalEdgeDartEquiv G) (rimIncidenceEquiv G d)=neighborSigmaDartEquiv G d:=
  26     (canonicalEdgeDartEquiv G).apply_symm_apply _
  27   have hfst:=congrArg (fun x:G.Dart=>x.fst) hh
  28   change (canonicalEdgeDart G (rimIncidenceEquiv G d)).fst=d.1 at hfst
  29   unfold canonicalEdgeDart at hfst
  30   split_ifs at hfst ⊢ <;> exact hfst.symm

  32  theorem actual_rim_endpoint_adjacency (G Q:SimpleGraph V) (φ:V→V)
  33     (p:G.edgeFinset→(Σ a:V,Σ c:V,Q.Walk a c))
  34     (hends:∀ e,Q.Adj e.val.out.1 (p e).1 ∧ Q.Adj (φ e.val.out.1) (p e).1 ∧
  35       Q.Adj e.val.out.2 (p e).2.1 ∧ Q.Adj (φ e.val.out.2) (p e).2.1):
  36     ∀ d:Σ v,G.neighborSet v,
  37       Q.Adj d.1 (graphPathEndpoint (p (rimIncidenceEquiv G d).1) (rimIncidenceEquiv G d).2) ∧
  38       Q.Adj (φ d.1) (graphPathEndpoint (p (rimIncidenceEquiv G d).1) (rimIncidenceEquiv G d).2):=by
  39   intro d
  40   have hroot:=rimIncidenceEquiv_root G d
  41   have hend:=hends (rimIncidenceEquiv G d).1
  42   unfold graphPathEndpoint
  43   split_ifs with he
  44   · simp only [he,ite_true] at hroot
  45     rw [hroot]
  46     exact ⟨hend.1,hend.2.1⟩
  47   · simp only [he,ite_false] at hroot
  48     rw [hroot]
  49     exact ⟨hend.2.2.1,hend.2.2.2⟩
