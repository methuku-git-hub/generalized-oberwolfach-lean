  23 theorem commonMarkedCycleConstraint_all (R : Bool → V → V → Prop) (H : Bool → V → U → Prop)
  24     (u u' : U) (C : Fin (m+1) → V) :
  25     (∀ a,commonMarkedCycleConstraint R H u u' a C) ↔
  26       (∀ i,R (decide (i=Fin.last m)) (C i) (C (finRotate (m+1) i))) ∧
  27       MarkedHubCompatible (H false) (H true) C u ∧ MarkedHubCompatible (H false) (H true) C u' := by
  28   constructor
  29   · intro h
  30     refine ⟨fun i=>h (Sum.inl i),?_,?_⟩
  31     · apply (markedHubCompatible_iff_positions _ _ _ _).mpr
  32       intro i
  33       have hh := h (Sum.inr (false,i))
  34       dsimp only [commonMarkedCycleConstraint,Sum.elim_inr,Bool.false_eq_true,ite_false] at hh
  35       cases hi : markedPositionType i <;> simpa only [hi,Bool.false_eq_true,ite_false,ite_true] using hh
  36     · apply (markedHubCompatible_iff_positions _ _ _ _).mpr
  37       intro i
  38       have hh := h (Sum.inr (true,i))
  39       dsimp only [commonMarkedCycleConstraint,Sum.elim_inr,ite_true] at hh
  40       cases hi : markedPositionType i <;> simpa only [hi,Bool.false_eq_true,ite_false,ite_true] using hh
  41   · rintro ⟨hr,hu,hu'⟩ a
  42     cases a with
  43     | inl i => exact hr i
  44     | inr s =>
  45       rcases s with ⟨side,i⟩
  46       have h (v : U) (hv : MarkedHubCompatible (H false) (H true) C v) : H (markedPositionType i) (C i) v := by
  47         have hh := (markedHubCompatible_iff_positions _ _ _ _).mp hv i
  48         cases hi : markedPositionType i <;> simpa only [hi,Bool.false_eq_true,ite_false,ite_true] using hh
  49       cases side
  50       · exact h u hu
  51       · exact h u' hu'

  53 theorem isMarkedCycleTuple_iff_relations
  54     (tail head : E → V) (ktail khead : K → V) (π : Equiv.Perm V) (b : ℕ)
  55     (C : Fin (m+1) → V) :
  56     IsMarkedCycleTuple tail head ktail khead m π b C ↔
  57       PermutationSeparated π b C ∧ ∀ i,
  58         (if i=Fin.last m then ∃ e,ktail e=C i ∧ khead e=C (finRotate (m+1) i)
  59           else ∃ e,tail e=C i ∧ head e=C (finRotate (m+1) i)) := by
  60   constructor
  61   · intro h
  62     refine ⟨h.1,fun i=>?_⟩
  63     refine Fin.lastCases ?_ (fun j=>?_) i
  64     · simpa only [ite_true,finRotate_last] using h.2.2
  65     · simpa only [Fin.castSucc_ne_last,ite_false,finRotate_castSucc_eq_succ] using h.2.1 j
  66   · rintro ⟨hsep,h⟩
  67     refine ⟨hsep,?_,?_⟩
  68     · intro j
  69       simpa only [Fin.castSucc_ne_last,ite_false,finRotate_castSucc_eq_succ] using h j.castSucc
  70     · simpa only [ite_true,finRotate_last] using h (Fin.last m)
