  17 theorem profile_factor_avoiding_pairs (a : I → ℕ) (c J : ℕ) (hc : 3 ≤ c)
  18     (ha : ∀ i,1 ≤ a i) (hsize : ∑ i,a i=c)
  19     (S : T → VertexAbsorberTemplate c) (a₀ b B : ℝ)
  20     (ha₀ : 0 < a₀) (hb : 0 < b) (hB : 0 ≤ B) :
  21     ∃ η : ℝ,0 < η ∧ ∀ d : ℕ,∃ k₀ : ℕ,∀ k : ℕ,k₀ ≤ k →
  22       ∀ (V : I → Type) [∀ i,Fintype (V i)] [∀ i,DecidableEq (V i)]
  23         (C : Type) [Fintype C] [DecidableEq C],
  24       (∀ i,Fintype.card (V i)=a i*k) →
  25       ∀ (R : C → Finset (Σ i,V i)) (w : C → ℝ),
  26       (∀ C₀ i,(Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈R C₀)).card=a i) →
  27       (∀ Y : Finset (Σ i,V i),Fintype.card {C₀ : C // R C₀=Y} ≤ J) →
  28       (∀ C₀,b/(k:ℝ)^(c-1) ≤ w C₀ ∧ w C₀ ≤ B/(k:ℝ)^(c-1)) →
  29       (∀ v,|1-hypergraphWeightedDegree R w v| ≤ η) →
  30       (∀ σ : ∀ i,FixedSubset (V i) (a i),
  31         a₀*(k:ℝ)^(c*c) ≤ Fintype.card (LabelledAbsorberFamily R S (profileRootSet σ))) →
  32       ∀ E : Finset ((Σ i,V i)×(Σ i,V i)),
  33       (∀ e∈E,e.1≠e.2) → E.card ≤ c*k →
  34       (∀ v,(E.filter (fun e => v=e.1 ∨ v=e.2)).card ≤ d) →
  35       ∃ D : Finset C,
  36         ((∀ x∈D,∀ y∈D,x≠y → Disjoint (R x) (R y)) ∧ D.biUnion R=Finset.univ) ∧
  37         ∀ C₀∈D,∀ e∈E,¬(e.1∈R C₀ ∧ e.2∈R C₀) := by
  38   obtain ⟨ηf,hηf,kf,hf⟩ := profile_factor_extraction a c J hc ha hsize S (a₀/2) b B (by positivity) hb hB
  39   refine ⟨ηf/2,by positivity,?_⟩
  40   intro d
  41   let K := ((d:ℝ)+1)*(J:ℝ)*(c:ℝ)^(c-2)*B
  42   let L := absorberPairDeletionCoefficient c J (Fintype.card T) d
  43   have hK : 0 ≤ K := by dsimp [K];positivity
  44   have hL : 0 ≤ L := by dsimp [L,absorberPairDeletionCoefficient];positivity
  45   obtain ⟨N,hN⟩ := exists_nat_ge (max (2*K/ηf) (2*L/a₀))
  46   refine ⟨max kf (max N 1),?_⟩
  47   intro k hk V _ _ C _ _ hn R w hR hJ hw herr habs E hE hEsize hEd
  48   have hkf : kf ≤ k := (le_max_left _ _).trans hk
  49   have hkN : N ≤ k := (le_max_left N 1).trans ((le_max_right _ _).trans hk)
  50   have hk1 : 1 ≤ k := (le_max_right N 1).trans ((le_max_right _ _).trans hk)
  51   have hkpos : (0:ℝ)<k := by exact_mod_cast (show 0<k by omega)
  52   have hkNR : (N:ℝ) ≤ k := by exact_mod_cast hkN
  53   have hKsmall : K/(k:ℝ) ≤ ηf/2 := by
  54     have hh := (le_max_left (2*K/ηf) (2*L/a₀)).trans (hN.trans hkNR)
  55     have hh' := (div_le_iff₀ hηf).mp hh
  56     apply (div_le_iff₀ hkpos).mpr
  57     nlinarith
  58   have hLsmall : L ≤ (a₀/2)*k := by
  59     have hh := (le_max_right (2*K/ηf) (2*L/a₀)).trans (hN.trans hkNR)
  60     have hh' := (div_le_iff₀ ha₀).mp hh
  61     nlinarith
  62   have hnall : Fintype.card (Σ i,V i)=c*k := by
  63     simp only [Fintype.card_sigma,hn,← Finset.sum_mul,hsize]
  64   have hRC (C₀ : C) : (R C₀).card=c := by rw [sigma_finset_card];simp only [hR];exact hsize
  65   have hw0 (C₀ : C) : 0 ≤ w C₀ := (by positivity : (0:ℝ) ≤ b/(k:ℝ)^(c-1)).trans (hw C₀).1
  66   let P := fun C₀ => ∀ e : E,¬(e.val.1∈R C₀ ∧ e.val.2∈R C₀)
  67   let C' := {C₀ : C // P C₀}
  68   let R' := candidateResourceSet R P
  69   let w' := fun D : C' => w D.val
  70   have hR' (D : C') (i) :
  71       (Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈R' D)).card=a i := hR D.val i
  72   have hJ' : ∀ Y : Finset (Σ i,V i),Fintype.card {D : C' // R' D=Y} ≤ J :=
  73     candidate_resource_multiplicity R P J hJ
  74   have hw' (D : C') : b/(k:ℝ)^(c-1) ≤ w' D ∧ w' D ≤ B/(k:ℝ)^(c-1) := hw D.val
  75   have herr' (v : Σ i,V i) : |1-hypergraphWeightedDegree R' w' v| ≤ ηf := by
  76     obtain ⟨hlo,hhi⟩ := candidate_resource_pair_degree_loss_scale R c J hc hRC hJ w B k hB hkpos
  77       (by exact_mod_cast hnall) (fun C₀ => ⟨hw0 C₀,(hw C₀).2⟩) E hE v d (hEd v) (by simpa only [hnall] using hEsize)
  78     change _ ≤ K/k at hhi
  79     have hs := abs_le.mp (herr v)
  80     apply abs_le.mpr
  81     constructor
  82     all_goals change hypergraphWeightedDegree R w v-hypergraphWeightedDegree R' w' v ≤ K/k at hhi
  83     all_goals change 0 ≤ hypergraphWeightedDegree R w v-hypergraphWeightedDegree R' w' v at hlo
  84     all_goals linarith
  85   have habs' (σ : ∀ i,FixedSubset (V i) (a i)) :
  86       (a₀/2)*(k:ℝ)^(c*c) ≤ Fintype.card (LabelledAbsorberFamily R' S (profileRootSet σ)) := by
  87     have hX : (profileRootSet σ).card=c := (profileRootSet_card σ).trans hsize
  88     have hloss := candidate_absorber_pair_loss_scale (by omega : 2≤c) R S (profileRootSet σ) J hJ E hE d
  89       (fun v _ => hEd v) (by simpa only [hnall] using hEsize) k (by exact_mod_cast hnall) hX
  90     change (Fintype.card (LabelledAbsorberFamily R S (profileRootSet σ)):ℝ) ≤
  91       Fintype.card (LabelledAbsorberFamily R' S (profileRootSet σ))+L*(k:ℝ)^(c*c-1) at hloss
  92     have hpow : (k:ℝ)*(k:ℝ)^(c*c-1)=(k:ℝ)^(c*c) := by
  93       rw [← pow_succ']
  94       congr 1
  95       have hh : 1≤c*c := by nlinarith
  96       omega
  97     have hsmall : L*(k:ℝ)^(c*c-1) ≤ (a₀/2)*(k:ℝ)^(c*c) := by
  98       calc
  99         _ ≤ ((a₀/2)*k)*(k:ℝ)^(c*c-1) := mul_le_mul_of_nonneg_right hLsmall (by positivity)
 100         _ = _ := by rw [mul_assoc,hpow]
 101     linarith [habs σ]
 102   have hfactor := hf k hkf V C' hn R' w' hR' hJ' hw' herr' habs'
 103   obtain ⟨D,hD,hP⟩ := candidate_resource_decomposition_lift R P Finset.univ hfactor
 104   exact ⟨D,hD,fun C₀ hC e he => hP C₀ hC ⟨e,he⟩⟩
