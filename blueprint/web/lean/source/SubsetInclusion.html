  12 theorem fixedSubset_joint_inclusion {V : Type*} [Fintype V] {r : ℕ}
  13     [Nonempty (FixedSubset V r)] (hr : r ≤ Fintype.card V) (T : Finset V) :
  14     (FiniteLaw.uniform (FixedSubset V r)).probability (fun S => T ⊆ S.1) =
  15       (r.choose T.card : ℝ) / (Fintype.card V).choose T.card := by
  16   by_cases hk : T.card ≤ r
  17   · rw [fixedSubset_inclusion_probability T hk]
  18     have hrpos : ((Fintype.card V).choose r : ℝ) ≠ 0 := by
  19       exact_mod_cast (Nat.choose_pos hr).ne'
  20     have hkpos : ((Fintype.card V).choose T.card : ℝ) ≠ 0 := by
  21       exact_mod_cast (Nat.choose_pos T.card_le_univ).ne'
  22     apply (div_eq_div_iff hrpos hkpos).mpr
  23     have hh := Nat.choose_mul (n := Fintype.card V) hk
  24     exact_mod_cast (by nlinarith [hh] :
  25       (Fintype.card V - T.card).choose (r - T.card) * (Fintype.card V).choose T.card =
  26         r.choose T.card * (Fintype.card V).choose r)
  27   · have hzero : r.choose T.card = 0 := Nat.choose_eq_zero_of_lt (by omega)
  28     rw [hzero, Nat.cast_zero, zero_div]
  29     unfold FiniteLaw.probability
  30     apply Finset.sum_eq_zero
  31     intro S _
  32     have hn : ¬ T ⊆ S.1 := by
  33       intro hh
  34       have hc := Finset.card_le_card hh
  35       rw [S.2] at hc
  36       exact hk hc
  37     simp [hn]
