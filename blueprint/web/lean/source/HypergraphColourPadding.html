   4 /-! Finite padding of each resource and each original colour to a supplied
   5 colour degree. Auxiliary edges use one resource and have private filler

  15 abbrev ColourPaddingVertex {V E C : Type*} [Fintype E]
  16     (R : E → Finset V) (L : E → Finset C) (s : ℕ) :=
  17   (v : V) × (c : C) × Fin (s-(hypergraphColourClique R L v c).card)

  19 abbrev PaddedColour {V E C : Type*} [Fintype E]
  20     (R : E → Finset V) (L : E → Finset C) (s l : ℕ) :=
  21   C ⊕ (ColourPaddingVertex R L s × Fin (l-1))

  23 def paddedResources {V E C : Type*} [Fintype E]
  24     (R : E → Finset V) (L : E → Finset C) (s : ℕ) :
  25     E ⊕ ColourPaddingVertex R L s → Finset V
  26   | .inl e => R e
  27   | .inr d => {d.1}

  29 def paddedColourList {V E C : Type*} [Fintype E]
  30     (R : E → Finset V) (L : E → Finset C) (s l : ℕ) :
  31     E ⊕ ColourPaddingVertex R L s → Finset (PaddedColour R L s l)
  32   | .inl e => (L e).map Function.Embedding.inl
  33   | .inr d => {Sum.inl d.2.1} ∪ Finset.univ.map
  34       ⟨fun i=>Sum.inr (d,i),fun _ _ h=>Prod.mk.inj (Sum.inr.inj h) |>.2⟩

  36 theorem paddedColourList_original {V E C : Type*} [Fintype E]
  37     (R : E → Finset V) (L : E → Finset C) (s l : ℕ)
  38     (e : E ⊕ ColourPaddingVertex R L s) (c : C) :
  39     Sum.inl c∈paddedColourList R L s l e ↔
  40       match e with
  41       | .inl f => c∈L f
  42       | .inr d => c=d.2.1 := by
  43   cases e <;> simp [paddedColourList]

  52 theorem paddedColourList_card {V E C : Type*} [Fintype E]
  53     (R : E → Finset V) (L : E → Finset C) (s : ℕ)
  54     {l : ℕ} (hl : 0<l) (hcard : ∀ e,(L e).card=l)
  55     (e : E ⊕ ColourPaddingVertex R L s) : (paddedColourList R L s l e).card=l := by
  56   classical
  57   cases e with
  58   | inl e => simpa only [paddedColourList,Finset.card_map] using hcard e
  59   | inr d =>
  60     have hd : Disjoint ({Sum.inl d.2.1} : Finset (PaddedColour R L s l))
  61         (Finset.univ.map ⟨fun i : Fin (l-1)=>Sum.inr (d,i),
  62           fun _ _ h=>Prod.mk.inj (Sum.inr.inj h) |>.2⟩) := by simp
  63     rw [paddedColourList,Finset.card_union_of_disjoint hd]
  64     simp only [Finset.card_singleton,Finset.card_map,Finset.card_univ,Fintype.card_fin]
  65     omega

  75 theorem paddedColourClique_card_original
  76     {V E C : Type*} [Fintype V] [Fintype E] [Fintype C]
  77     (R : E → Finset V) (L : E → Finset C) (s l : ℕ)
  78     (hdegree : ∀ v c,(hypergraphColourClique R L v c).card≤s)
  79     (v : V) (c : C) :
  80     (hypergraphColourClique (paddedResources R L s) (paddedColourList R L s l) v (Sum.inl c)).card=s := by
  81   classical
  82   have hpad : (∑ d : ColourPaddingVertex R L s,if v=d.1 ∧ c=d.2.1 then (1 : ℕ) else 0)=
  83       s-(hypergraphColourClique R L v c).card := by
  84     rw [Fintype.sum_sigma]
  85     simp_rw [Fintype.sum_sigma]
  86     change (∑ v' : V,∑ c' : C,∑ _i : Fin (s-(hypergraphColourClique R L v' c').card),
  87       if v=v' ∧ c=c' then (1 : ℕ) else 0)=_
  88     simp only [Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul,
  89       mul_ite,mul_one,mul_zero,ite_and]
  90     simp [eq_comm]
  91   have he :
  92       (hypergraphColourClique (paddedResources R L s) (paddedColourList R L s l) v (Sum.inl c)).card=
  93       (hypergraphColourClique R L v c).card+(s-(hypergraphColourClique R L v c).card) := by
  94     have hset : hypergraphColourClique (paddedResources R L s) (paddedColourList R L s l) v (Sum.inl c)=
  95         Finset.univ.filter (fun e=>v∈paddedResources R L s e ∧ Sum.inl c∈paddedColourList R L s l e) := by
  96       ext e
  97       simp only [hypergraphColourClique,Finset.mem_filter]
  98     rw [hset]
  99     conv_lhs => rw [Finset.card_eq_sum_ones,Finset.sum_filter]
 100     rw [Fintype.sum_sum_type]
 101     simp only [paddedColourList_original,paddedResources,Finset.mem_singleton]
 102     rw [hpad]
 103     congr 1
 104     simp only [hypergraphColourClique,Finset.card_eq_sum_ones,Finset.sum_filter]
 105   rw [he]
 106   exact Nat.add_sub_of_le (hdegree v c)

 108 theorem paddedColourClique_card_private
 109     {V E C : Type*} [Fintype V] [Fintype E] [Fintype C]
 110     (R : E → Finset V) (L : E → Finset C) (s l : ℕ)
 111     (v : V) (d : ColourPaddingVertex R L s) (i : Fin (l-1)) :
 112     (hypergraphColourClique (paddedResources R L s) (paddedColourList R L s l) v (Sum.inr (d,i))).card≤1 := by
 113   classical
 114   apply (Finset.card_le_card (show
 115     hypergraphColourClique (paddedResources R L s) (paddedColourList R L s l) v (Sum.inr (d,i))⊆
 116       {Sum.inr d} from ?_)).trans (by simp)
 117   intro e he
 118   have hm : Sum.inr (d,i)∈paddedColourList R L s l e := by
 119     have hmem : v∈paddedResources R L s e ∧ Sum.inr (d,i)∈paddedColourList R L s l e := by
 120       simpa only [hypergraphColourClique,Finset.mem_filter,Finset.mem_univ,true_and] using he
 121     exact hmem.2
 122   exact Finset.mem_singleton.mpr ((paddedColourList_private R L s l e d i).mp hm)

 124 theorem paddedColourClique_degree_bound
 125     {V E C : Type*} [Fintype V] [Fintype E] [Fintype C]
 126     (R : E → Finset V) (L : E → Finset C) {s : ℕ} (hs : 1≤s) (l : ℕ)
 127     (hdegree : ∀ v c,(hypergraphColourClique R L v c).card≤s)
 128     (v : V) (c : PaddedColour R L s l) :
 129     (hypergraphColourClique (paddedResources R L s) (paddedColourList R L s l) v c).card≤s := by
 130   cases c with
 131   | inl c => exact le_of_eq (paddedColourClique_card_original R L s l hdegree v c)
 132   | inr p => exact (paddedColourClique_card_private R L s l v p.1 p.2).trans hs

 134 theorem paddedResources_codegree
 135     {V E C : Type*} [Fintype V] [Fintype E] [Fintype C]
 136     (R : E → Finset V) (L : E → Finset C) (s : ℕ)
 137     (v w : V) (hvw : v≠w) :
 138     (Finset.univ.filter (fun e=>v∈paddedResources R L s e ∧ w∈paddedResources R L s e)).card=
 139       (Finset.univ.filter (fun e=>v∈R e ∧ w∈R e)).card := by
 140   classical
 141   simp only [Finset.card_eq_sum_ones,Finset.sum_filter]
 142   rw [Fintype.sum_sum_type]
 143   have hz (d : ColourPaddingVertex R L s) :
 144       ¬(v∈paddedResources R L s (Sum.inr d) ∧ w∈paddedResources R L s (Sum.inr d)) := by
 145     simpa only [paddedResources,Finset.mem_singleton] using
 146       (show ¬(v=d.1 ∧ w=d.1) from fun h=>hvw (h.1.trans h.2.symm))

 150 theorem paddedResources_degree_formula
 151     {V E C : Type*} [Fintype V] [Fintype E] [Fintype C]
 152     (R : E → Finset V) (L : E → Finset C) (s : ℕ) (v : V) :
 153     (hypergraphVertexClique (paddedResources R L s) v).card=
 154       (hypergraphVertexClique R v).card+∑ c : C,(s-(hypergraphColourClique R L v c).card) := by
 155   classical
 156   have hpad : (∑ d : ColourPaddingVertex R L s,if v=d.1 then (1 : ℕ) else 0)=
 157       ∑ c : C,(s-(hypergraphColourClique R L v c).card) := by
 158     rw [Fintype.sum_sigma]
 159     simp_rw [Fintype.sum_sigma]
 160     change (∑ v' : V,∑ c' : C,∑ _i : Fin (s-(hypergraphColourClique R L v' c').card),
 161       if v=v' then (1 : ℕ) else 0)=_
 162     simp only [Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul,
 163       mul_ite,mul_one,mul_zero]
 164     simp_rw [Finset.sum_ite_irrel,Finset.sum_const_zero]
 165     simp
 166   unfold hypergraphVertexClique
 167   conv_lhs => rw [Finset.card_eq_sum_ones,Finset.sum_filter,Fintype.sum_sum_type]
 168   simp only [paddedResources,Finset.mem_singleton]
 169   rw [hpad]
 170   congr 1
 171   simp only [Finset.card_eq_sum_ones,Finset.sum_filter]

 173 theorem paddedResources_degree_bound
 174     {V E C : Type*} [Fintype V] [Fintype E] [Fintype C]
 175     (R : E → Finset V) (L : E → Finset C) (s : ℕ) {D : ℕ}
 176     (hdegree : ∀ v,(hypergraphVertexClique R v).card≤D) (v : V) :
 177     (hypergraphVertexClique (paddedResources R L s) v).card≤D+Fintype.card C*s := by
 178   rw [paddedResources_degree_formula]
 179   apply add_le_add (hdegree v)
 180   calc
 181     _ ≤ ∑ _c : C,s := Finset.sum_le_sum (fun c _=>Nat.sub_le _ _)
 182     _ = _ := by simp
