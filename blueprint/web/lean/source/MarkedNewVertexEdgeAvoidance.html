  16  theorem marked_rim_edge_avoids_forbidden (F:Finset (Finset V)) (Y:Finset V)
  17     (hY:Disjoint Y ((F.biUnion id)∪(F.biUnion id).image π.symm))
  18     (a:E⊕K) (hnew:Sum.elim tail ktail a∈Y∨Sum.elim head khead a∈Y):
  19     markedPhysicalEdge tail head ktail khead π a∉F:=by
  20   intro ha
  21   have hmem:∀ v∈markedPhysicalEdge tail head ktail khead π a,v∈F.biUnion id:=
  22     fun v hv=>Finset.mem_biUnion.mpr ⟨_,ha,hv⟩
  23   have htail:Sum.elim tail ktail a∈F.biUnion id:=by
  24     apply hmem
  25     cases a <;> simp [markedPhysicalEdge]
  26   rcases hnew with hnew|hnew
  27   · exact Finset.disjoint_left.mp hY hnew (Finset.mem_union_left _ htail)
  28   · cases a with
  29     | inl a=>
  30       have hh:head a∈F.biUnion id:=hmem _ (by simp [markedPhysicalEdge])
  31       exact Finset.disjoint_left.mp hY hnew (Finset.mem_union_left _ hh)
  32     | inr a=>
  33       have hh:π (khead a)∈F.biUnion id:=hmem _ (by simp [markedPhysicalEdge])
  34       have hi:khead a∈(F.biUnion id).image π.symm:=Finset.mem_image.mpr ⟨π (khead a),hh,π.symm_apply_apply _⟩
  35       exact Finset.disjoint_left.mp hY hnew (Finset.mem_union_right _ hi)
