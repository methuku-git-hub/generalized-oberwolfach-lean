  14 theorem ordinary_rim_edge_wheel_from_common_counts (Q F:SimpleGraph V) (H:V→U→Prop)
  15     (A W:Finset V) (x y:V) (u:U) (hxy:Q.Adj x y) (hFxy:¬F.Adj x y)
  16     (hHx:H x u) (hHy:H y u) (hHA:∀ a∈A,H a u)
  17     (r:ℕ) (hr:3≤r) (L b:ℝ) (hL:0≤L) (hF:∀ a,(F.degree a:ℝ)≤L)
  18     (hcount:∀ a c,b≤((A∩Q.neighborFinset a∩Q.neighborFinset c).card:ℝ))
  19     (hbudget:W.card+((r:ℝ)+2)*(L+1)<b) :
  20     ∃ C:OrdinaryWheel Q F H r,ordinaryWheelHub C=u ∧
  21       (ordinaryWheelRim C).Adj x y ∧
  22       (∀ a∈(ordinaryWheelRim C).support,a=x ∨ a=y ∨ a∈A∧a∉W) := by
  23   have hroot:∀ a∈({x,y}:Finset V),∀ c∈({x,y}:Finset V),¬F.Adj a c:=by
  24     intro a ha c hc
  25     simp only [Finset.mem_insert,Finset.mem_singleton] at ha hc
  26     rcases ha with rfl|rfl <;> rcases hc with rfl|rfl
  27     · exact F.loopless.irrefl _
  28     · exact hFxy
  29     · exact fun h=>hFxy h.symm
  30     · exact F.loopless.irrefl _
  31   have hcard:(({x,y}:Finset V).card:ℝ)≤2:=by exact_mod_cast (Finset.card_le_two (a:=x) (b:=y))
  32   have hk:((r-1:ℕ):ℝ)≤r:=by exact_mod_cast Nat.sub_le r 1
  33   obtain ⟨p,hp,hlen,hclass,hind,hroots⟩:=sparse_fixed_length_forbidden_path Q F A W {x,y}
  34     x y hxy.ne (by simp) (by simp) hroot (r-1) (by omega) L b hL hF hcount (by
  35       have hh:=mul_le_mul_of_nonneg_right (show (({x,y}:Finset V).card:ℝ)+(r-1:ℕ)≤(r:ℝ)+2 by linarith only [hcard,hk])
  36         (show 0≤L+1 by linarith only [hL])
  37       linarith only [hbudget,hh])
  38   obtain ⟨C,hC,hCQ,hCxy,hCs,_⟩:=embedded_cycle_from_path_with_support Q r hr x y p hp hlen hxy
  39   have hH (a:V) (ha:a∈C.support):H a u:=by
  40     rcases hclass a ((hCs a).mp ha) with rfl|rfl|ha
  41     · exact hHx
  42     · exact hHy
  43     · exact hHA a ha.1
  44   have hFC:∀ a∈C.support,∀ c∈C.support,¬F.Adj a c:=by
  45     intro a ha c hc
  46     exact hind a ((hCs a).mp ha) c ((hCs c).mp hc)
  47   refine ⟨⟨(⟨⟨C,hC⟩,hCQ⟩,u),hH,hFC⟩,rfl,hCxy,?_⟩
  48   intro a ha
  49   rcases hclass a ((hCs a).mp ha) with he|he|ha
  50   · exact Or.inl he
  51   · exact Or.inr (Or.inl he)
  52   · exact Or.inr (Or.inr ⟨ha.1,ha.2.1⟩)

  54 theorem ordinary_spoke_wheel_from_common_counts (Q F:SimpleGraph V) (H:V→U→Prop)
  55     (A W:Finset V) (x:V) (u:U) (hHx:H x u) (hHA:∀ a∈A,H a u)
  56     (r:ℕ) (hr:3≤r) (L b:ℝ) (hL:0≤L) (hF:∀ a,(F.degree a:ℝ)≤L)
  57     (hcount:∀ a c,b≤((A∩Q.neighborFinset a∩Q.neighborFinset c).card:ℝ))
  58     (hbudget:W.card+((r:ℝ)+2)*(L+1)<b) :
  59     ∃ C:OrdinaryWheel Q F H r,ordinaryWheelHub C=u ∧ x∈(ordinaryWheelRim C).support ∧
  60       (∀ a∈(ordinaryWheelRim C).support,a=x ∨ a∈A∧a∉W) := by
  61   obtain ⟨y,hyA,hyW,hyx,hxy,_,hFy⟩:=sparse_common_neighbour_avoiding Q F A W {x}
  62     x x L b hL hF (hcount x x) (by
  63       simp only [Finset.card_singleton,Nat.cast_one]
  64       have hr0:(0:ℝ)≤r:=Nat.cast_nonneg _
  65       nlinarith only [hbudget,hL,hr0])
  66   obtain ⟨C,hCu,hCxy,hCs⟩:=ordinary_rim_edge_wheel_from_common_counts Q F H A W x y u hxy
  67     (fun h=>hFy x (Finset.mem_singleton_self _) h.symm) hHx (hHA y hyA) hHA
  68     r hr L b hL hF hcount hbudget
  69   refine ⟨C,hCu,hCxy.mem_support_left,?_⟩
  70   intro a ha
  71   rcases hCs a ha with he|he|ha
  72   · exact Or.inl he
  73   · exact Or.inr (he ▸ ⟨hyA,hyW⟩)
  74   · exact Or.inr ha
