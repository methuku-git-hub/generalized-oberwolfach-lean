   9 theorem deleted_path_lengths (F:SimpleGraph V) (s₁ s₂:Finset I)
  10     (u v:I→V) (P:∀i,F.Walk (u i) (v i))
  11     (hp:∀i∈s₁∪s₂,(P i).IsPath) (c d:ℕ) (hc:0<c)
  12     (hlen:∀i∈s₁∪s₂,(P i).length=c*d+4)
  13     (T:I→Finset ℕ) (hT:∀i∈s₂,T i⊆Finset.range d):
  14     (∀i:{i // i∈s₁},(deletedPathWalk F s₁ s₂ u v P c d T hT (.inl i)).length=c*d) ∧
  15     (∀i:{i // i∈s₂},∀r:{r // r∈selectedSegmentRuns d (T i.1)},
  16       (deletedPathWalk F s₁ s₂ u v P c d T hT (.inr ⟨i,r⟩)).length=c*(r.1.2+1-r.1.1)) := by
  17   constructor
  18   · intro i
  19     have hi:i.1∈s₁∪s₂:=Finset.mem_union_left _ i.2
  20     have hright:2+c*d≤(P i.1).length:=by rw [hlen i.1 hi];omega
  21     have hh:=walk_interval_length (P i.1) 2 (2+c*d) (by omega) hright
  22     change (walkInterval (P i.1) 2 (2+c*d) (by omega)).length=c*d
  23     convert hh using 1 <;> simp
  24   · intro i r

  28     have hb:r.1.2<d:=by

  30       simpa using hh
