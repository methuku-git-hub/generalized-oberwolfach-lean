  12  theorem balanced_trim_numerical (a b s:ℕ) (hs:1≤s):
  13     let k:=min a (b/s)
  14     k≤a∧s*k≤b∧
  15       ((a:ℝ)-k+(b:ℝ)-s*k≤|(b:ℝ)-s*a|+s-1):=by
  16   intro k
  17   have hs0:0<s:=by omega
  18   have hq:s*(b/s)≤b:=Nat.mul_div_le b s
  19   have hr:b%s<s:=Nat.mod_lt b hs0
  20   have hb:s*(b/s)+b%s=b:=by simpa only [Nat.mul_comm,add_comm] using Nat.mod_add_div b s
  21   have hka:k≤a:=min_le_left _ _
  22   have hkb:k≤b/s:=min_le_right _ _
  23   refine ⟨hka,(Nat.mul_le_mul_left s hkb).trans hq,?_⟩
  24   have hsR:(1:ℝ)≤s:=by exact_mod_cast hs
  25   by_cases hab:a≤b/s
  26   · have hk:k=a:=min_eq_left hab
  27     rw [hk]
  28     have hsa:s*a≤b:=(Nat.mul_le_mul_left s hab).trans hq
  29     have hp':(s:ℝ)*a≤b:=by exact_mod_cast hsa
  30     rw [abs_of_nonneg (by linarith)]
  31     linarith
  32   · have hk:k=b/s:=min_eq_right (by omega)
  33     rw [hk]
  34     have hqa:b/s+1≤a:=by omega
  35     have hqaR:((b/s:ℕ):ℝ)+1≤a:=by exact_mod_cast hqa
  36     have hrR:((b%s:ℕ):ℝ)≤s-1:=by
  37       have hh:((b%s:ℕ):ℝ)+1≤s:=by exact_mod_cast (show b%s+1≤s by omega)
  38       linarith
  39     have hbR:(s:ℝ)*(b/s:ℕ)+(b%s:ℕ)=b:=by exact_mod_cast hb
  40     have hn:(0:ℝ)≤((s:ℝ)-1)*((a:ℝ)-(b/s:ℕ)-1):=mul_nonneg (by linarith) (by linarith)
  41     have hneg:(b:ℝ)-s*a≤0:=by nlinarith only [hsR,hqaR,hrR,hbR,hn]
  42     rw [abs_of_nonpos hneg]
  43     nlinarith only [hsR,hqaR,hrR,hbR,hn]

  45  theorem balanced_subsets_preserving_roots (A B ZA ZB:Finset V) (s:ℕ) (hs:1≤s)
  46     (hZA:ZA⊆A) (hZB:ZB⊆B) (hZ:ZB.card=s*ZA.card):
  47     ∃ A' B':Finset V,ZA⊆A'∧A'⊆A∧ZB⊆B'∧B'⊆B∧
  48       B'.card=s*A'.card∧A'.card=min A.card (B.card/s)∧
  49       ((A.card:ℝ)-A'.card+(B.card:ℝ)-B'.card≤|(B.card:ℝ)-s*A.card|+s-1):=by
  50   let k:=min A.card (B.card/s)
  51   have hka:k≤A.card:=min_le_left _ _
  52   have hkb:s*k≤B.card:=(balanced_trim_numerical A.card B.card s hs).2.1
  53   have hZk:ZA.card≤k:=by
  54     apply le_min (Finset.card_le_card hZA)
  55     apply (Nat.le_div_iff_mul_le (by omega)).mpr
  56     simpa only [Nat.mul_comm,hZ] using (Finset.card_le_card hZB)
  57   obtain ⟨A',hZA',hA',hAc⟩:=Finset.exists_subsuperset_card_eq hZA hZk hka
  58   obtain ⟨B',hZB',hB',hBc⟩:=Finset.exists_subsuperset_card_eq hZB
  59     (by rw [hZ];exact Nat.mul_le_mul_left s hZk) hkb
  60   refine ⟨A',B',hZA',hA',hZB',hB',by rw [hAc,hBc],hAc,?_⟩
  61   rw [hAc,hBc,Nat.cast_mul]
  62   exact (balanced_trim_numerical A.card B.card s hs).2.2
