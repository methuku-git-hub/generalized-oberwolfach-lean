  13 theorem packed_marked_wheels_exhaust_selected_segments
  14     {X V U E L K J:Type} [Fintype X] [DecidableEq X] [Fintype V] [DecidableEq V] [DecidableEq U] [DecidableEq K] [DecidableEq J]
  15     {F:SimpleGraph X} (data:ActualResidualData (K:=K) (J:=J) F)
  16     (tail head:E→V) (ktail khead:L→V) (m b:ℕ) (π:Equiv.Perm V) (H₀ Hₖ:V→U→Prop)
  17     (S:Finset (MarkedWheel tail head ktail khead m π b H₀ Hₖ)) (w:U) (root:K→V)
  18     (hHub:∀W∈S,W.2.val=w)
  19     (hDis:∀W∈S,∀Z∈S,W≠Z→Disjoint (markedWheelPhysicalResources W) (markedWheelPhysicalResources Z))
  20     (hT:∀i∈data.s₂,data.T i=packedPhysicalSpokePositions tail head ktail khead m b data.d π H₀ Hₖ S w (root i))
  21     (hRoots:∀W∈S,∃i∈data.s₂,∃j<data.d,W.1.val 0=(π^j) (root i))
  22     (Wsel:actualSelectedSegmentIndex data→{W // W∈S})
  23     (hRoot:∀z,(Wsel z).val.1.val 0=(π^z.2.val.val) (root z.1.1.val)):
  24     ∀W∈S,∃z,(Wsel z).val=W:=by
  25   intro W hW
  26   obtain ⟨i,hi,j,hj,hjroot⟩:=hRoots W hW
  27   have hspoke:Sum.inr ((π^j) (root i),w)∈markedWheelPhysicalResources W:=by
  28     apply (markedWheelPhysicalResources_spoke W _ _).mpr
  29     refine ⟨hHub W hW,?_⟩
  30     exact Finset.mem_image.mpr ⟨0,Finset.mem_univ _,hjroot⟩
  31   have hjT:j∈data.T i:=by
  32     rw [hT i hi]
  33     apply Finset.mem_filter.mpr
  34     exact ⟨Finset.mem_range.mpr hj,⟨W,hW,hspoke⟩⟩
  35   obtain ⟨a,b',hab,haj,hjb⟩:=selected_segment_in_maximal_run data.d (data.T i) (data.hT i) j hjT
  36   let r:SelectedRunIndex data.s₂ data.d data.T:=⟨⟨i,hi⟩,⟨(a,b'),hab⟩⟩
  37   let z:actualSelectedSegmentIndex data:=⟨r,⟨⟨j,by omega⟩,haj,hjb⟩⟩
  38   have hzspoke:Sum.inr ((π^j) (root i),w)∈markedWheelPhysicalResources (Wsel z).val:=by
  39     apply (markedWheelPhysicalResources_spoke (Wsel z).val _ _).mpr
  40     refine ⟨hHub _ (Wsel z).property,?_⟩
  41     apply Finset.mem_image.mpr
  42     refine ⟨0,Finset.mem_univ _,?_⟩
  43     simpa [z,r] using hRoot z
  44   by_cases heq:(Wsel z).val=W
  45   · exact ⟨z,heq⟩
  46   · have hdj:=hDis (Wsel z).val (Wsel z).property W hW heq
  47     exact (Finset.disjoint_left.mp hdj hzspoke hspoke).elim
