  14 theorem finite_weighted_union_bound {I J : Type*} [Fintype I] [Fintype J]
  15     (w : I → ℝ) (hw : ∀ i,0≤w i) (base : I → Prop) (E : I → J → Prop) :
  16     (∑ i,if base i ∧ ∃ j,E i j then w i else 0)≤
  17       ∑ j,∑ i,if base i ∧ E i j then w i else 0 := by
  18   rw [Finset.sum_comm]
  19   apply Finset.sum_le_sum
  20   intro i _
  21   by_cases h : base i ∧ ∃ j,E i j
  22   · obtain ⟨hb,j,hj⟩:=h
  23     rw [if_pos ⟨hb,j,hj⟩]
  24     have hh:=Finset.single_le_sum (s:=Finset.univ)
  25       (f:=fun j=>if base i ∧ E i j then w i else 0)
  26       (fun j _=>by split_ifs;exact hw i;exact le_refl _) (Finset.mem_univ j)
  27     simpa only [hb,hj,and_self,ite_true] using hh
  28   · rw [if_neg h]
  29     exact Finset.sum_nonneg (fun j _=>by split_ifs;exact hw i;exact le_refl _)

  37 theorem ordinary_cycle_three_vertex_weight (x y z : V) (hxy : x≠y) (hxz : x≠z) (hyz : y≠z)
  38     (E : OrdinaryCycle P r → Prop)
  39     (hE : ∀ C,E C → x∈C.val.val.support ∧ y∈C.val.val.support ∧ z∈C.val.val.support) :
  40     (∑ C,if E C then weight C else 0)≤(r:ℝ)^4*B/Fintype.card V := by
  41   have hc : ({x,y,z}:Finset V).card=3 := by simp [hxy,hxz,hyz]
  42   have hh := ordinary_cycle_event_weight_bound P r (by omega) hn weight hweight B hB hb {x,y,z}
  43     (by rw [hc];norm_num) E (by
  44       intro C h v hv
  45       rcases hE C h with ⟨hx,hy,hz⟩
  46       simp only [Finset.coe_insert,Finset.coe_singleton,Set.mem_insert_iff,Set.mem_singleton_iff] at hv
  47       rcases hv with rfl|rfl|rfl <;> assumption)
  48   simp only [hc,show 3-2=1 from rfl,pow_one] at hh
  49   apply hh.trans
  50   apply div_le_div_of_nonneg_right _ (by positivity)
  51   apply mul_le_mul_of_nonneg_right _ hB
  52   exact pow_le_pow_right₀ (by exact_mod_cast (show 1≤r by omega)) (by omega)
