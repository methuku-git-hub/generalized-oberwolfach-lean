  10 theorem regular_two_buffered_path_family_target_embedding

  12     (G:SimpleGraph V) (hG:G.IsRegularOfDegree 2) (H:SimpleGraph W)
  13     (n:I→ℕ) (ext:∀i,Fin (n i+5)↪V) (inner:∀i,Fin (n i+1)↪V)
  14     (hinner:∀i k,inner i k=ext i ⟨k.val+2,by omega⟩)
  15     (hwalk:∀i,∃p:G.Walk (ext i ⟨0,by omega⟩) (ext i ⟨n i+4,by omega⟩),
  16       p.length=n i+4 ∧ p.support=List.ofFn (ext i))
  17     (hdis:∀i j,i≠j→Disjoint (Finset.univ.image (ext i)) (Finset.univ.image (ext j)))
  18     (A:Finset I) (target:∀i,Fin (n i+1)↪W)
  19     (htarget:∀i (k:Fin (n i)),H.Adj (target i ⟨k.val,by omega⟩) (target i ⟨k.val+1,by omega⟩))
  20     (htargetDis:∀i∈A,∀j∈A,i≠j→Disjoint (Set.range (target i)) (Set.range (target j))):
  21     ∃φ:{x // x∈A.biUnion (fun i=>Finset.univ.image (inner i))}↪W,
  22       (∀i (hi:i∈A) (k:Fin (n i+1)),

  24       (∀x y,G.Adj x.val y.val→H.Adj (φ x) (φ y)):=by
  25   classical
  26   have hstep:∀i (k:Fin (n i+4)),
  27       G.Adj (ext i ⟨k.val,by omega⟩) (ext i ⟨k.val+1,by omega⟩):=by
  28     intro i k
  29     obtain ⟨p,hlen,hsupport⟩:=hwalk i
  30     exact walk_ofFn_support_adjacent G (n i+4) (ext i) p hlen hsupport k
  31   have hexists (i:I):=regular_two_buffered_path_target_embedding G hG H (n i) (ext i) (inner i)
  32     (hinner i) (hstep i) (target i) (htarget i)
  33   let f (i:I):{x // x∈Finset.univ.image (inner i)}↪W:=Classical.choose (hexists i)
  34   have hf (i:I):=
  35     Classical.choose_spec (hexists i)
  36   have hfdis:∀i∈A,∀j∈A,i≠j→Disjoint (Set.range (f i)) (Set.range (f j)):=by
  37     intro i hi j hj hij
  38     exact (htargetDis i hi j hj hij).mono (hf i).2.1 (hf j).2.1
  39   have hlocal:∀i∈A,∀x y (hx:x∈Finset.univ.image (inner i)) (hy:y∈Finset.univ.image (inner i)),
  40       G.Adj x y→H.Adj (f i ⟨x,hx⟩) (f i ⟨y,hy⟩):=by
  41     intro i hi x y hx hy hxy
  42     exact (hf i).2.2 ⟨x,hx⟩ ⟨y,hy⟩ hxy
  43   obtain ⟨φ,hφ,hgraph⟩:=regular_two_buffered_walk_embeddings_glue G hG H n ext inner hinner hwalk hdis A f hfdis hlocal
  44   refine ⟨φ,?_,hgraph⟩

  46   let hx:inner i k∈Finset.univ.image (inner i):=Finset.mem_image.mpr ⟨k,Finset.mem_univ _,rfl⟩
  47   exact (hφ i hi (inner i k) hx).trans ((hf i).1 k)
