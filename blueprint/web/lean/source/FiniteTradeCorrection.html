  15 theorem finiteLaw_abs_expectation_bound {Ω : Type*} [Fintype Ω]
  16     (μ : FiniteLaw Ω) (f : Ω → ℝ) : |μ.expectation f| ≤ μ.expectation (fun ω => |f ω|) := by
  17   calc
  18     _ ≤ ∑ ω,|μ.mass ω*f ω| := Finset.abs_sum_le_sum_abs _ _
  19     _ = _ := by simp only [abs_mul,abs_of_nonneg (μ.nonneg _),FiniteLaw.expectation]

  23 def correctedTradeWeight (μ : FiniteLaw Ω) (w : C → ℝ) (t : Ω → ℝ) (T : Ω → C → ℝ) (c : C) : ℝ :=

  26 theorem correctedTradeWeight_exact (μ : FiniteLaw Ω) (R : C → Finset V) (X : Ω → Finset V)
  27     (w : C → ℝ) (t : Ω → ℝ) (T : Ω → C → ℝ)
  28     (hT : ∀ ω v,hypergraphWeightedDegree R (T ω) v=if v∈X ω then 1 else 0)
  29     (ht : ∀ v,μ.expectation (fun ω => t ω*(if v∈X ω then 1 else 0))=1-hypergraphWeightedDegree R w v) :
  30     ∀ v,hypergraphWeightedDegree R (correctedTradeWeight μ w t T) v=1 := by
  31   intro v
  32   have havg := averagedWeight_incidence μ (fun ω c => t ω*T ω c) 1 (fun c => v∈R c)
  33   simp only [averagedWeight,div_one] at havg
  34   have hpoint (ω : Ω) : (∑ c,if v∈R c then t ω*T ω c else 0)=
  35       t ω*hypergraphWeightedDegree R (T ω) v := by
  36     simp only [hypergraphWeightedDegree,Finset.mul_sum,mul_ite,mul_zero]
  37   have havg' : (∑ c,if v∈R c then μ.expectation (fun ω => t ω*T ω c) else 0)=
  38       1-hypergraphWeightedDegree R w v := by
  39     calc
  40       _ = μ.expectation (fun ω => ∑ c,if v∈R c then t ω*T ω c else 0) := by
  41         convert havg using 1 <;> congr!
  42       _ = μ.expectation (fun ω => t ω*(if v∈X ω then 1 else 0)) := by
  43         apply congrArg μ.expectation
  44         funext ω
  45         rw [hpoint,hT]
  46       _ = _ := ht v
  47   have heq (c : C) : (if v∈R c then correctedTradeWeight μ w t T c else 0)=
  48       (if v∈R c then w c else 0)+(if v∈R c then μ.expectation (fun ω => t ω*T ω c) else 0) := by
  49     by_cases hc : v∈R c <;> simp [hc,correctedTradeWeight]
  50   unfold hypergraphWeightedDegree
  51   simp_rw [heq,Finset.sum_add_distrib]
  52   rw [havg']
  53   change hypergraphWeightedDegree R w v+(1-hypergraphWeightedDegree R w v)=1
  54   ring

  56 theorem correctedTradeWeight_difference (μ : FiniteLaw Ω) (w : C → ℝ) (t : Ω → ℝ)
  57     (T : Ω → C → ℝ) (A B : ℝ) (hA : 0 ≤ A)
  58     (ht : ∀ ω,|t ω| ≤ A) (hT : ∀ c,μ.expectation (fun ω => |T ω c|) ≤ B) (c : C) :
  59     |correctedTradeWeight μ w t T c-w c| ≤ A*B := by
  60   simp only [correctedTradeWeight,add_sub_cancel_left]
  61   calc
  62     _ ≤ μ.expectation (fun ω => |t ω*T ω c|) := finiteLaw_abs_expectation_bound μ _
  63     _ ≤ μ.expectation (fun ω => A*|T ω c|) := finiteLaw_expectation_mono μ _ _
  64       (by intro ω;rw [abs_mul];exact mul_le_mul_of_nonneg_right (ht ω) (abs_nonneg _))
  65     _ = A*μ.expectation (fun ω => |T ω c|) := finiteLaw_expectation_const_mul μ A _
  66     _ ≤ A*B := mul_le_mul_of_nonneg_left (hT c) hA

  68 theorem finiteLaw_trade_bound_from_hit_probability (μ : FiniteLaw Ω)
  69     (T : Ω → C → ℝ) (hit : C → Ω → Prop) (L U κ : ℝ) (hU : 0 ≤ U)
  70     (hT : ∀ ω c,|T ω c| ≤ L+(if hit c ω then U else 0))
  71     (hhit : ∀ c,μ.probability (hit c) ≤ κ) (c : C) :
  72     μ.expectation (fun ω => |T ω c|) ≤ L+U*κ := by
  73   have hh := finiteLaw_expectation_mono μ _ _ (fun ω => hT ω c)
  74   have he : μ.expectation (fun ω => L+(if hit c ω then U else 0))=L+U*μ.probability (hit c) := by
  75     simp only [FiniteLaw.expectation,mul_add,Finset.sum_add_distrib,← Finset.sum_mul,μ.total,one_mul]
  76     congr 1
  77     exact finiteLaw_expectation_indicator_value μ (hit c) U
  78   rw [he] at hh
  79   exact hh.trans (add_le_add le_rfl (mul_le_mul_of_nonneg_left (hhit c) hU))

  81 theorem correctedTradeWeight_two_sided (μ : FiniteLaw Ω) (w : C → ℝ) (t : Ω → ℝ)
  82     (T : Ω → C → ℝ) (A B l u : ℝ) (hA : 0 ≤ A)
  83     (ht : ∀ ω,|t ω| ≤ A) (hT : ∀ c,μ.expectation (fun ω => |T ω c|) ≤ B)
  84     (hw : ∀ c,l ≤ w c ∧ w c ≤ u) (hsmall : A*B ≤ l/2) (c : C) :
  85     l/2 ≤ correctedTradeWeight μ w t T c ∧ correctedTradeWeight μ w t T c ≤ u+l/2 := by
  86   have hh := correctedTradeWeight_difference μ w t T A B hA ht hT c
  87   have hb := abs_le.mp hh
