  12 theorem prescribed_free_path_from_ordered_vertices
  13     (as:List X) (x:V) (xs:List V) (hlen:as.length=(x::xs).length)
  14     (ha:as.Nodup) (hb:(x::xs).Nodup) (A:Finset X) (f:X→V)
  15     (hinj:Set.InjOn f A) (hsource:∀ a∈as,a∉A)
  16     (htarget:∀ z∈x::xs,z∉A.image f)
  17     (D:V→V→Prop) (F:SimpleGraph V) (hpath:DirectedPathFrom D F true x xs):
  18     let g:=updatePairedImages f (as.zip (x::xs));
  19     Set.InjOn g (↑(A∪as.toFinset):Set X) ∧ (∀ a∈A,g a=f a) ∧
  20     (∀ i:ℕ,∀ hi:i+1<as.length,
  21       D (g (as[i]'(by omega))) (g (as[i+1]'(by omega))) ∧
  22       ¬F.Adj (g (as[i]'(by omega))) (g (as[i+1]'(by omega)))) := by
  23   let g:=updatePairedImages f (as.zip (x::xs))
  24   have hbase:=prescribed_embedding_zip_update as (x::xs) hlen ha hb A f hinj hsource htarget
  25   refine ⟨hbase.1,hbase.2.1,?_⟩
  26   intro i hi
  27   have hi0:i<as.length:=by omega
  28   have hi1:i+1<as.length:=hi
  29   have hh0:=prescribed_embedding_zip_update_at_position as (x::xs) hlen ha hb A f hinj hsource htarget ⟨i,hi0⟩
  30   have hh1:=prescribed_embedding_zip_update_at_position as (x::xs) hlen ha hb A f hinj hsource htarget ⟨i+1,hi1⟩
  31   have htargetlen:i+1<(x::xs).length:=by omega
  32   have hedges:=prescribed_path_consecutive D F true xs x hpath i htargetlen
  33   simp only [Bool.true_eq,if_true] at hedges
  34   change D (g (as[i])) (g (as[i+1])) ∧ ¬F.Adj (g (as[i])) (g (as[i+1]))
  35   change g (as[i])=(x::xs)[i] at hh0
  36   change g (as[i+1])=(x::xs)[i+1] at hh1
  37   rw [hh0,hh1]
  38   exact hedges
  39 end Oberwolfach
