   5 /-! Being at cyclic permutation distance at most b is expressed by at
   6 most b forward steps in either direction, including equality. This
   7 works for all cycle lengths, including fixed points. -/

  14 def PermutationClose {V : Type*} (π : Equiv.Perm V) (b : ℕ) (x y : V) : Prop :=
  15   x = y ∨ ∃ j : Fin b, (π ^ (j.val + 1)) x = y ∨ (π ^ (j.val + 1)) y = x
  16 
  17 theorem permutationClose_symm {V : Type*} (π : Equiv.Perm V) (b : ℕ) (x y : V) :
  18     PermutationClose π b x y ↔ PermutationClose π b y x := by
  19   simp only [PermutationClose, eq_comm, or_comm]

  21 theorem permutationClose_neighbour_card {V : Type*} [Fintype V]
  22     (π : Equiv.Perm V) (b : ℕ) (x : V) :
  23     Fintype.card {y // PermutationClose π b x y} ≤ 2 * b + 1 := by
  24   classical
  25   let A := Finset.univ.image (fun j : Fin b => (π ^ (j.val + 1)) x)
  26   let B := Finset.univ.image (fun j : Fin b => (π ^ (j.val + 1)).symm x)
  27   let S := insert x (A ∪ B)
  28   have hmem (y : V) (h : PermutationClose π b x y) : y ∈ S := by
  29     rcases h with h | ⟨j, h | h⟩
  30     · subst y
  31       exact Finset.mem_insert_self _ _
  32     · apply Finset.mem_insert_of_mem
  33       exact Finset.mem_union_left _ (Finset.mem_image.mpr ⟨j, Finset.mem_univ _, h⟩)
  34     · apply Finset.mem_insert_of_mem
  35       apply Finset.mem_union_right
  36       apply Finset.mem_image.mpr
  37       refine ⟨j, Finset.mem_univ _, ?_⟩
  38       exact ((π ^ (j.val + 1)).eq_symm_apply.mpr h).symm
  39   let f : {y // PermutationClose π b x y} → S := fun y => ⟨y.1, hmem y.1 y.2⟩
  40   have hf : Function.Injective f := by
  41     intro y z he
  42     apply Subtype.ext
  43     exact congrArg (fun w : S => w.val) he
  44   have ha : A.card ≤ b := (Finset.card_image_le).trans (by simp)
  45   have hb : B.card ≤ b := (Finset.card_image_le).trans (by simp)
  46   calc
  47     _ ≤ Fintype.card S := Fintype.card_le_of_injective f hf
  48     _ = S.card := Fintype.card_coe S
  49     _ ≤ (A ∪ B).card + 1 := Finset.card_insert_le _ _
  50     _ ≤ 2 * b + 1 := by have hh := Finset.card_union_le A B; omega
  51 
  52 end Oberwolfach
