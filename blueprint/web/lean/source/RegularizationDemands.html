  14 theorem regularization_demand_totals {V : Type*} [Fintype V]
  15     (M : V → V → ℕ) (k : ℕ)
  16     (hl : ∀ x, k ≤ ∑ y, M x y) (hr : ∀ y, k ≤ ∑ x, M x y) :
  17     ∑ x, ((∑ y, M x y) - k) = ∑ y, ((∑ x, M x y) - k) := by
  18   calc
  19     _ = (∑ x, ∑ y, M x y) - ∑ _x : V, k :=
  20       Finset.sum_tsub_distrib Finset.univ (fun x _ => hl x)
  21     _ = (∑ y, ∑ x, M x y) - ∑ _y : V, k := by rw [Finset.sum_comm]
  22     _ = _ := (Finset.sum_tsub_distrib Finset.univ (fun y _ => hr y)).symm

  24 theorem matrix_regularization_deletion_law (β : ℝ) (hβ : 0 < β) :
  25     ∃ C η₀ : ℝ, 0 < C ∧ 0 < η₀ ∧ ∀ η : ℝ, 0 < η → η < η₀ →
  26       ∃ m₀ : ℕ, ∀ (V : Type) [Fintype V] [DecidableEq V], m₀ ≤ Fintype.card V →
  27       ∀ (M : V → V → ℕ), (∀ x y, M x y ≤ 1) → ∀ d : ℝ,
  28       (∀ x, (1 / 2 + β) * Fintype.card V ≤ ((∑ y, M x y : ℕ) : ℝ)) →
  29       (∀ y, (1 / 2 + β) * Fintype.card V ≤ ((∑ x, M x y : ℕ) : ℝ)) →
  30       (∀ x, d - η * Fintype.card V ≤ ((∑ y, M x y : ℕ) : ℝ) ∧
  31         ((∑ y, M x y : ℕ) : ℝ) ≤ d + η * Fintype.card V) →
  32       (∀ y, d - η * Fintype.card V ≤ ((∑ x, M x y : ℕ) : ℝ) ∧
  33         ((∑ x, M x y : ℕ) : ℝ) ≤ d + η * Fintype.card V) →
  34       let k := ⌊d - Real.sqrt η * Fintype.card V⌋₊
  35       (∀ x, k ≤ ∑ y, M x y) ∧ (∀ y, k ≤ ∑ x, M x y) ∧
  36       ∃ ν : FiniteLaw (BipartiteEdges M → Bool),
  37         (∀ D, 0 < ν.mass D →
  38           (∀ x, (∑ e, if e.1 = x ∧ D e = true then (1 : ℕ) else 0) =
  39             (∑ y, M x y) - k) ∧
  40           (∀ y, (∑ e, if e.2.1 = y ∧ D e = true then (1 : ℕ) else 0) =
  41             (∑ x, M x y) - k)) ∧
  42         (∀ e, ν.probability (fun D => D e = true) ≤ C * Real.sqrt η) := by
  43   obtain ⟨δ, h, C, η₀, hδ, hδ8, hh, hhβ, hC, hCeq, hη₀, hsmall⟩ :=
  44     regularization_constants β hβ
  45   refine ⟨C, η₀, hC, hη₀, ?_⟩
  46   intro η hη hηsmall
  47   obtain ⟨hs, hs4, hsd, hp, hp1⟩ := hsmall η hη hηsmall
  48   obtain ⟨m₀, _, hm₀⟩ := regularization_order_threshold η hη
  49   refine ⟨m₀, ?_⟩
  50   intro V _ _ hm M hM d hleft hright hnearleft hnearright
  51   obtain ⟨hn, hnlarge⟩ := hm₀ _ hm
  52   have hn' : (0 : ℝ) < Fintype.card V := by exact_mod_cast hn
  53   have hηsq : η = (Real.sqrt η) ^ 2 := (Real.sq_sqrt hη.le).symm
  54   have hηs : η ≤ Real.sqrt η := by nlinarith
  55   obtain ⟨v⟩ := Fintype.card_pos_iff.mp hn
  56   have harg : 0 ≤ d - Real.sqrt η * Fintype.card V :=
  57     regularization_floor_argument hn' hβ.le hs.le hs4 hηsq
  58       (hleft v) (hnearleft v).2
  59   let k := ⌊d - Real.sqrt η * Fintype.card V⌋₊
  60   have hfl (x : V) := regularization_floor_demand (∑ y, M x y)
  61     hn' harg hηs hnlarge (hnearleft x)
  62   have hfr (y : V) := regularization_floor_demand (∑ x, M x y)
  63     hn' harg hηs hnlarge (hnearright y)
  64   refine ⟨fun x => (hfl x).1, fun y => (hfr y).1, ?_⟩
  65   have hscale := regularization_relative_scale hs hηsq hδ hδ8 hsd
  66   have hcap := regularization_capacity_scale hs hs4 hηsq hh hCeq
  67   have result := interval_random_demands (fun e : BipartiteEdges M => e.1)
  68     (fun e : BipartiteEdges M => e.2.1)
  69     (fun x => (∑ y, M x y) - k) (fun y => (∑ x, M x y) - k)
  70     ((Real.sqrt η - η) * Fintype.card V)
  71     ((Real.sqrt η + 2 * η) * Fintype.card V) δ h (C * Real.sqrt η)
  72     hn (mul_pos hscale.1 hn') hp hp1 hδ.le (by linarith)
  73     (by nlinarith [mul_le_mul_of_nonneg_right hscale.2 hn'.le])
  74     (by nlinarith [mul_le_mul_of_nonneg_right hcap hn'.le])
  75     (fun x => (hfl x).2) (fun y => (hfr y).2)
  76     (regularization_demand_totals M k (fun x => (hfl x).1) (fun y => (hfr y).1))
  77     (matrixEdges_normalized_large_cuts M hM β δ h hn hhβ hleft hright)
  78   exact result
