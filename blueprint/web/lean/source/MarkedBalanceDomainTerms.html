  15 theorem markedBalanceTermSet_domain_card
  16     (tail head:E→V) (ktail khead:K→V) (H₀ HK:V→U→Prop)
  17     (X:Finset V) (Z:Finset U) (r:V⊕U) (j:Fin 6):
  18     let PE:E→Prop:=fun e=>match r with
  19       | Sum.inl v=>(if tail e=v then head e else tail e)∈X
  20       | Sum.inr _=>True
  21     let PK:K→Prop:=fun e=>match r with
  22       | Sum.inl v=>(if ktail e=v then khead e else ktail e)∈X
  23       | Sum.inr _=>True
  24     let P:V→U→Prop:=fun x u=>match r with
  25       | Sum.inl _=>u∈Z
  26       | Sum.inr _=>x∈X;
  27     ((markedBalanceTermSet tail head ktail khead H₀ HK r j).filter (fun e=>
  28       (if markedActiveAuxTail tail ktail H₀ HK e=r then markedActiveAuxHead head khead H₀ HK e
  29         else markedActiveAuxTail tail ktail H₀ HK e)∈ordinaryInnerVertices X Z)).card=
  30     (markedBalanceTermSet (fun e:{e:E//PE e}=>tail e.val) (fun e:{e:E//PE e}=>head e.val)
  31       (fun e:{e:K//PK e}=>ktail e.val) (fun e:{e:K//PK e}=>khead e.val)
  32       (fun x u=>H₀ x u∧P x u) (fun x u=>HK x u∧P x u) r j).card:=by
  33   intro PE PK P
  34   rw [markedBalanceTermSet_restriction_card PE PK P]
  35   congr 1
  36   apply Finset.filter_congr
  37   intro a ha
  38   have ht:markedBalanceTerm tail head ktail khead H₀ HK r j a:=(Finset.mem_filter.mp ha).2
  39   rcases r with v|u <;> rcases a with (e|e)|s
  40   · simp only [markedActiveAuxTail,markedActiveAuxHead,Sum.elim_inl,Sum.inl.injEq,
  41       markedActiveRetained,PE]
  42     split_ifs <;> simp only [ordinaryInnerVertices_left]
  43   · simp only [markedActiveAuxTail,markedActiveAuxHead,Sum.elim_inl,Sum.elim_inr,Sum.inl.injEq,
  44       markedActiveRetained,PK]
  45     split_ifs <;> simp only [ordinaryInnerVertices_left]
  46   · change s.val.1=v∧_ at ht
