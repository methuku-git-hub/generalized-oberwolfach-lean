  12 theorem finiteLaw_pi_blocks_probability {I J : Type*} [Fintype I] [Fintype J]
  13     [DecidableEq I] [DecidableEq J] {State : I→J→Type*} [∀ i j,Fintype (State i j)]
  14     (μ : ∀ i j,FiniteLaw (State i j)) (E : (∀ i j,State i j)→Prop) :
  15     (FiniteLaw.pi (fun a : I×J=>μ a.1 a.2)).probability (fun σ=>E (fun i j=>σ (i,j)))=
  16       (FiniteLaw.pi (fun i=>FiniteLaw.pi (μ i))).probability E := by
  17   let e : (∀ a : I×J,State a.1 a.2)≃(∀ i j,State i j) :=
  18     {toFun:=fun σ i j=>σ (i,j)
  19      invFun:=fun τ a=>τ a.1 a.2
  20      left_inv:=by intro σ;rfl
  21      right_inv:=by intro τ;rfl}
  22   unfold FiniteLaw.probability
  23   rw [← e.symm.sum_comp]
  24   apply Finset.sum_congr rfl
  25   intro σ _
  26   change (if E σ then ∏ a : I×J,(μ a.1 a.2).mass (σ a.1 a.2) else 0)=
  27     if E σ then ∏ i,∏ j,(μ i j).mass (σ i j) else 0
  28   rw [Fintype.prod_prod_type]
  29 
  30 end Oberwolfach
