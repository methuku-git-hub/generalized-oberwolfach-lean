  10 theorem directed_source_edge_images_subset_host
  11     {X V:Type*} [Fintype X] [Fintype V] [DecidableEq X] [DecidableEq V]
  12     (Q:SimpleGraph X) (H:SimpleGraph V) (R:X→X→Prop) (f:X→V)
  13     (hrep:∀x y,Q.Adj x y→R x y∨R y x)
  14     (harc:∀x y,R x y→H.Adj (f x) (f y)) :
  15     Q.edgeFinset.image (Sym2.map f)⊆H.edgeFinset := by
  16   classical
  17   intro e he
  18   obtain ⟨a,ha,rfl⟩:=Finset.mem_image.mp he
  19   induction a using Sym2.inductionOn with
  20   | _ x y =>
  21     have hxy:Q.Adj x y:=by simpa using ha
  22     apply SimpleGraph.mem_edgeFinset.mpr
  23     rcases hrep x y hxy with h|h
  24     · exact harc x y h
  25     · exact (harc y x h).symm

  27 theorem actual_residual_directed_new_edges_subset_host
  28     {I X V K J:Type} [Fintype I] [Fintype X] [Fintype V]
  29     [DecidableEq I] [DecidableEq X] [DecidableEq V] [DecidableEq K] [DecidableEq J]
  30     (F:I→SimpleGraph X) (a:∀i,ActualResidualData (K:=K) (J:=J) (F i))
  31     (pieces:∀i,List (PrescribedSourcePiece {x // x∈(a i).remaining}))
  32     (hgraph:∀i,prescribedDependentGraph pieces i=((a i).graph).induce (↑(a i).remaining:Set X))
  33     (H:SimpleGraph V) (Dstar:V→V→Prop) (hHost:∀x y,Dstar x y→H.Adj x y)
  34     (Φ:∀i,{x // x∈(a i).remaining}→V)
  35     (harc:∀i x y,prescribedDependentDigraph pieces i x y→Dstar (Φ i x) (Φ i y)) :
  36     ∀i,(((a i).graph.induce (↑(a i).remaining:Set X)).edgeFinset).image
  37       (Sym2.map (Φ i))⊆H.edgeFinset := by
  38   intro i e he
  39   obtain ⟨z,hz,heq⟩:=Finset.mem_image.mp he
  40   induction z using Sym2.inductionOn with
  41   | _ x y =>
  42     have hxy:((a i).graph.induce (↑(a i).remaining:Set X)).Adj x y:=SimpleGraph.mem_edgeFinset.mp hz
  43     have hxy':(prescribedDependentGraph pieces i).Adj x y:=by rw [hgraph i];exact hxy
  44     have hor:=prescribed_dependent_source_representation pieces i x y hxy'
  45     have hed:Sym2.map (Φ i) s(x,y)∈H.edgeFinset:=by
  46       change s(Φ i x,Φ i y)∈H.edgeFinset
  47       apply SimpleGraph.mem_edgeFinset.mpr
  48       rcases hor with h|h
  49       · exact hHost _ _ (harc i x y h)
  50       · exact (hHost _ _ (harc i y x h)).symm
  51     exact heq ▸ hed
  52 end Oberwolfach
