  15 def filteredCoinConflictSet {V C : Type*} (G : SimpleGraph V) (T : Finset V)
  16     (P : V → (V → C×Bool) → Prop) (σ : V → C×Bool) : Finset V :=
  17   coinConflictSet G (T.filter (fun v=>P v σ)) σ

  19 theorem filteredCoinConflict_certificate
  20     {V C : Type*} [Fintype V] (G : SimpleGraph V) (T : Finset V)
  21     (hT : ∀ u∈T,∀ v∈T,u≠v → G.Adj u v)
  22     (P : V → (V → C×Bool) → Prop) (W : (V → C×Bool) → V → Finset V)
  23     {k : ℕ} {c : ℝ}
  24     (hsize : ∀ σ v,v∈T → P v σ → (W σ v).card≤k)
  25     (hload : ∀ σ i,((T.filter (fun v=>P v σ ∧ i∈W σ v)).card : ℝ)≤c)
  26     (hkeep : ∀ σ v,v∈T → P v σ → ∀ τ,(∀ i∈W σ v,σ i=τ i) → P v τ) :
  27     HasCoordinateCertificateBound
  28       (fun σ=>((filteredCoinConflictSet G T P σ).card : ℝ))
  29       (2+c) ((3+(k : ℝ))*(T.card : ℝ)) := by
  30   classical
  31   intro σ
  32   let S := T.filter (fun v=>P v σ)
  33   have hST : S⊆T := Finset.filter_subset _ _
  34   have hS : ∀ u∈S,∀ v∈S,u≠v → G.Adj u v :=
  35     fun u hu v hv hne=>hT u (hST hu) v (hST hv) hne
  36   obtain ⟨w₁,hw₁,hs₁,hchange₁⟩ := coinConflict_certificate G S hS σ
  37   have hqrange (τ : V → C×Bool) :
  38       ((S.filter (fun v=>P v τ)).card : ℝ)≤T.card := by
  39     exact_mod_cast (Finset.card_le_card ((Finset.filter_subset _ _).trans hST))
  40   have hqsize (τ : V → C×Bool) (v : V) (hv : v∈S) (hp : P v τ) : (W τ v).card≤k :=
  41     hsize τ v (hST hv) hp
  42   have hqload (τ : V → C×Bool) (i : V) :
  43       ((S.filter (fun v=>P v τ ∧ i∈W τ v)).card : ℝ)≤c := by
  44     have hsub : S.filter (fun v=>P v τ ∧ i∈W τ v)⊆
  45         T.filter (fun v=>P v τ ∧ i∈W τ v) := by
  46       intro v hv
  47       obtain ⟨hvS,hvP,hi⟩ := Finset.mem_filter.mp hv
  48       exact Finset.mem_filter.mpr ⟨hST hvS,hvP,hi⟩
  49     exact (show ((S.filter (fun v=>P v τ ∧ i∈W τ v)).card : ℝ)≤
  50       (T.filter (fun v=>P v τ ∧ i∈W τ v)).card by
  51         exact_mod_cast Finset.card_le_card hsub).trans (hload τ i)
  52   have hqkeep (τ : V → C×Bool) (v : V) (hv : v∈S) (hp : P v τ)
  53       (τ' : V → C×Bool) (he : ∀ i∈W τ v,τ i=τ' i) : P v τ' :=
  54     hkeep τ v (hST hv) hp τ' he
  55   obtain ⟨w₂,hw₂,hs₂,hchange₂⟩ :=
  56     finiteWitness_coordinate_certificate_on S P W hqrange hqsize hqload hqkeep σ
  57   refine ⟨fun i=>w₁ i+w₂ i,?_,?_,?_⟩
  58   · intro i
  59     constructor <;> linarith [(hw₁ i).1,(hw₁ i).2,(hw₂ i).1,(hw₂ i).2]
  60   · rw [Finset.sum_add_distrib]
  61     have hcard : (S.card : ℝ)≤T.card := by exact_mod_cast Finset.card_le_card hST
  62     nlinarith
  63   · intro τ
  64     let B := coinConflictSet G S τ
  65     let A := S.filter (fun v=>P v τ)
  66     let R := filteredCoinConflictSet G T P τ
  67     have hBS : B⊆S := Finset.filter_subset _ _
  68     have hAS : A⊆S := Finset.filter_subset _ _
  69     have hInter : B∩A⊆R := by
  70       intro v hv
  71       obtain ⟨hvB,hvA⟩ := Finset.mem_inter.mp hv
  72       obtain ⟨hvS,hbad⟩ := Finset.mem_filter.mp hvB
  73       have hP := (Finset.mem_filter.mp hvA).2
  74       exact Finset.mem_filter.mpr ⟨Finset.mem_filter.mpr ⟨hST hvS,hP⟩,hbad⟩
  75     have hUnion : B∪A⊆S := Finset.union_subset hBS hAS
  76     have hcount : B.card+A.card≤R.card+S.card := by
  77       have h := Finset.card_union_add_card_inter B A
  78       have h₁ := Finset.card_le_card hInter
  79       have h₂ := Finset.card_le_card hUnion
  80       omega
  81     have hcountR : (B.card : ℝ)+(A.card : ℝ)≤(R.card : ℝ)+(S.card : ℝ) := by
  82       exact_mod_cast hcount
  83     have hAtStart : S.filter (fun v=>P v σ)=S := by
  84       ext v
  85       simp only [Finset.mem_filter]
  86       constructor
  87       · exact And.left
  88       · intro hv
  89         exact ⟨hv,(Finset.mem_filter.mp hv).2⟩
  90     have hfirst := hchange₁ τ
  91     have hsecond := hchange₂ τ
  92     have hstart : ((S.filter (fun v=>P v σ)).card : ℝ)=(S.card : ℝ) :=
  93       congrArg (fun F : Finset V=>(F.card : ℝ)) hAtStart
  94     have hstart' : ((filteredCoinConflictSet G T P σ).card : ℝ)=
  95         ((coinConflictSet G S σ).card : ℝ) := rfl
  96     have hchanges : (∑ i,if σ i≠τ i then w₁ i+w₂ i else 0)=
  97         (∑ i,if σ i≠τ i then w₁ i else 0)+(∑ i,if σ i≠τ i then w₂ i else 0) := by
  98       rw [← Finset.sum_add_distrib]
  99       apply Finset.sum_congr rfl
 100       intro i _
 101       split_ifs <;> ring
 102     change ((filteredCoinConflictSet G T P σ).card : ℝ)≤(R.card : ℝ)+_
 103     change ((coinConflictSet G S σ).card : ℝ)≤(B.card : ℝ)+_ at hfirst
 104     change ((S.filter (fun v=>P v σ)).card : ℝ)≤(A.card : ℝ)+_ at hsecond
 105     rw [hstart] at hsecond
 106     have hf : ((coinConflictSet G S σ).card : ℝ)≤(B.card : ℝ)+
 107         (∑ i,if σ i≠τ i then w₁ i else 0) := by
 108       convert hfirst using 1
 109       congr 1
 110       apply Finset.sum_congr rfl
 111       intro i _
 112       by_cases he : σ i=τ i <;> simp [he]
 113     have hs : (S.card : ℝ)≤(A.card : ℝ)+
 114         (∑ i,if σ i≠τ i then w₂ i else 0) := by
 115       convert hsecond using 1
 116       congr 1
 117       apply Finset.sum_congr rfl
 118       intro i _
 119       by_cases he : σ i=τ i <;> simp [he]
 120     have hmain : ((filteredCoinConflictSet G T P σ).card : ℝ)≤
 121         (R.card : ℝ)+((∑ i,if σ i≠τ i then w₁ i else 0)+
 122           (∑ i,if σ i≠τ i then w₂ i else 0)) := by
 123       rw [hstart']
 124       linarith only [hf,hs,hcountR]
 125     calc
 126       _ ≤ _ := hmain
 127       _ = (R.card : ℝ)+(∑ i,if σ i≠τ i then w₁ i+w₂ i else 0) :=
 128         congrArg (fun z=>(R.card : ℝ)+z) hchanges.symm
 129       _ = _ := by
 130         congr 1
 131         apply Finset.sum_congr rfl
 132         intro i _
 133         by_cases he : σ i=τ i <;> simp [he]
