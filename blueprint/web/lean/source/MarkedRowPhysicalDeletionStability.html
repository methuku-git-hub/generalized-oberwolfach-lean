  17 theorem marked_row_physical_coordinate_degree
  18     (π:Equiv.Perm V) (D:SimpleGraph V) (mark incoming:Bool) (v:V):
  19     (∑ x:V,if markedPairPhysical π (markedRowRim (mark,incoming,v) x)∈D.edgeFinset then (1:ℝ) else 0)=
  20       D.degree (if mark && incoming then π v else v):=by
  21   have hsum (z:V):(∑ x:V,if D.Adj z x then (1:ℝ) else 0)=D.degree z:=by
  22     rw [←SimpleGraph.card_neighborFinset_eq_degree,SimpleGraph.neighborFinset_eq_filter]
  23     simp only [Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  24   have hsumπ (z:V):(∑ x:V,if D.Adj z (π x) then (1:ℝ) else 0)=D.degree z:=
  25     (Equiv.sum_comp π (fun x=>if D.Adj z x then (1:ℝ) else 0)).trans (hsum z)
  26   cases mark <;> cases incoming <;>
  27     simp only [markedRowRim,markedPairPhysical,Bool.false_eq_true,ite_false,ite_true,
  28       Sum.elim_inl,Sum.elim_inr,SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet,
  29       Bool.false_and,Bool.true_and,Bool.and_false,Bool.and_true]
  30   · exact hsum v
  31   · simpa [D.adj_comm] using hsum v
  32   · exact hsumπ v
  33   · simpa [D.adj_comm] using hsum (π v)

  35 theorem marked_row_rim_physical_deletion_change
  36     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ) (σ:RawMarkedOutcome V U Γ Color)
  37     (D:SimpleGraph V) (A:Finset V) (a:Bool×Bool×V):
  38     (∑ x∈A,if ¬(rawMarkedPairRims P π d σ (markedRowRim a x)↔
  39       rawMarkedPairRims P π d (rawMarkedDeleteRims σ D.edgeFinset) (markedRowRim a x)) then (1:ℝ) else 0)
  40       ≤D.degree (if a.1 && a.2.1 then π a.2.2 else a.2.2):=by
  41   calc
  42     _ ≤ ∑ x∈A,if markedPairPhysical π (markedRowRim a x)∈D.edgeFinset then (1:ℝ) else 0:=by
  43       apply Finset.sum_le_sum
  44       intro x _
  45       rw [rawMarkedDeleteRims_rims]
  46       by_cases he:rawMarkedPairRims P π d σ (markedRowRim a x) <;>
  47         by_cases hd:markedPairPhysical π (markedRowRim a x)∈D.edgeFinset <;> simp [he,hd]
  48     _ ≤ ∑ x:V,if markedPairPhysical π (markedRowRim a x)∈D.edgeFinset then (1:ℝ) else 0:=
  49       Finset.sum_le_sum_of_subset_of_nonneg (Finset.subset_univ A) (by intro x _ _;positivity)
  50     _ = _:=marked_row_physical_coordinate_degree π D a.1 a.2.1 a.2.2
