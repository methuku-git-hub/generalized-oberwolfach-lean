  21     (H₀' HK' : V→U→Prop) (h₀ : ∀ v u,H₀ v u→H₀' v u) (hK : ∀ v u,HK v u→HK' v u) :
  22     MarkedEndpointPairing A₀ A₁ X₀ X₁ Y₀ Y₁ H₀' HK' π b where
  23   hubs_disjoint:=P.hubs_disjoint
  24   incoming:=P.incoming
  25   outgoing:=P.outgoing
  26   incoming_adj:=fun a=>⟨(P.incoming_adj a).1,h₀ _ _ (P.incoming_adj a).2⟩
  27   outgoing_adj:=by
  28     intro x
  29     refine ⟨(P.outgoing_adj x).1,?_⟩
  30     have hh:=(P.outgoing_adj x).2

  32     · exact hK _ _ hh
  33     · exact h₀ _ _ hh
  34   separated:=P.separated
