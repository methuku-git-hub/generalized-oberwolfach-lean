  15 def HasCoordinateCertificateBound {I Ω : Type*} [Fintype I]
  16     (R : (I → Ω) → ℝ) (c D : ℝ) : Prop :=
  17   ∀ x,∃ w : I → ℝ,(∀ i,0≤w i ∧ w i≤c) ∧ (∑ i,w i)≤D ∧
  18     ∀ y,R x≤R y+∑ i,if x i≠y i then w i else 0

  20 theorem finiteLaw_probability_or_le
  21     {Γ : Type*} [Fintype Γ] (ν : FiniteLaw Γ) (A B : Γ → Prop) :
  22     ν.probability (fun x=>A x ∨ B x)≤ν.probability A+ν.probability B := by
  23   classical
  24   unfold FiniteLaw.probability
  25   rw [← Finset.sum_add_distrib]
  26   apply Finset.sum_le_sum
  27   intro x _
  28   by_cases hA : A x <;> by_cases hB : B x <;> simp [hA,hB,ν.nonneg x]

  30 theorem finiteLaw_certificate_separated_tails
  31     {I Ω : Type*} [Fintype I] [Fintype Ω]
  32     (μ : I → FiniteLaw Ω) (R : (I → Ω) → ℝ)
  33     {c D t m : ℝ} (hc : 0<c) (hD : 0<D) (ht : 0≤t)
  34     (hcert : HasCoordinateCertificateBound R c D) :
  35     (FiniteLaw.pi μ).probability (fun x=>R x≤m)*
  36       (FiniteLaw.pi μ).probability (fun x=>m+t≤R x)≤
  37       Real.exp (-t^2/(4*c*D)) := by
  38   classical
  39   let ν := FiniteLaw.pi μ
  40   let A := Finset.univ.filter (fun x=>R x≤m)
  41   have hmem (x : I → Ω) : x∈A ↔ R x≤m := by simp [A]
  42   have hp : ν.probability (fun x=>x∈A)=ν.probability (fun x=>R x≤m) :=
  43     finiteLaw_probability_congr ν _ _ hmem
  44   have hp0 := finiteLaw_probability_nonneg ν (fun x=>R x≤m)
  45   by_cases hpos : 0<ν.probability (fun x=>R x≤m)
  46   · have hAne := finiteLaw_positive_finset_nonempty ν A (hp.symm ▸ hpos)
  47     have hcost (x : I → Ω) (hx : m+t≤R x) :
  48         t^2/(c*D)≤convexDistanceSq A x := by
  49       obtain ⟨w,hw,hs,hchange⟩ := hcert x
  50       have hh := convexDistanceSq_value_certificate R A hAne x w hc.le ht hw hs
  51         (fun y hy=>(hmem y).mp hy) hx hchange
  52       apply (div_le_iff₀ (mul_pos hc hD)).mpr
  53       nlinarith
  54     have hmono := finiteLaw_probability_mono ν (fun x=>m+t≤R x)
  55       (fun x=>t^2/(c*D)≤convexDistanceSq A x) hcost
  56     have htail := finiteLaw_convexDistance_tail μ A (t^2/(c*D))
  57     change ν.probability (fun x=>x∈A)*_≤_ at htail
  58     rw [hp] at htail
  59     have he : -(t^2/(c*D))/4=-t^2/(4*c*D) := by ring
  60     rw [he] at htail
  61     exact (mul_le_mul_of_nonneg_left hmono hp0).trans htail
  62   · have hz : ν.probability (fun x=>R x≤m)=0 := le_antisymm (le_of_not_gt hpos) hp0
  63     change ν.probability (fun x=>R x≤m)*_≤_
  64     rw [hz,zero_mul]
  65     exact (Real.exp_pos _).le

  67 theorem finiteLaw_certificate_median_tail
  68     {I Ω : Type*} [Fintype I] [Fintype Ω]
  69     (μ : I → FiniteLaw Ω) (R : (I → Ω) → ℝ)
  70     {c D t m : ℝ} (hc : 0<c) (hD : 0<D) (ht : 0≤t)
  71     (hcert : HasCoordinateCertificateBound R c D)
  72     (hlo : 1/2≤(FiniteLaw.pi μ).probability (fun x=>R x≤m))
  73     (hhi : 1/2≤(FiniteLaw.pi μ).probability (fun x=>m≤R x)) :
  74     (FiniteLaw.pi μ).probability (fun x=>t≤|R x-m|)≤
  75       4*Real.exp (-t^2/(4*c*D)) := by
  76   classical
  77   let ν := FiniteLaw.pi μ
  78   have hu := finiteLaw_certificate_separated_tails μ R hc hD ht hcert (m:=m)
  79   have hl := finiteLaw_certificate_separated_tails μ R hc hD ht hcert (m:=m-t)
  80   have he : m-t+t=m := by ring
  81   rw [he] at hl
  82   have hu' : ν.probability (fun x=>m+t≤R x)≤2*Real.exp (-t^2/(4*c*D)) := by
  83     have hh := mul_le_mul_of_nonneg_right hlo
  84       (finiteLaw_probability_nonneg ν (fun x=>m+t≤R x))
  85     nlinarith
  86   have hl' : ν.probability (fun x=>R x≤m-t)≤2*Real.exp (-t^2/(4*c*D)) := by
  87     have hh := mul_le_mul_of_nonneg_left hhi
  88       (finiteLaw_probability_nonneg ν (fun x=>R x≤m-t))
  89     nlinarith
  90   have hm := finiteLaw_probability_mono ν (fun x=>t≤|R x-m|)
  91     (fun x=>R x≤m-t ∨ m+t≤R x) (by
  92       intro x hx
  93       obtain h|h := le_abs.mp hx
  94       · right;linarith
  95       · left;linarith)
  96   have ho := finiteLaw_probability_or_le ν (fun x=>R x≤m-t) (fun x=>m+t≤R x)
  97   linarith
