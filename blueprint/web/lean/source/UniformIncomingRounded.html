  13 theorem uniform_incoming_rounding
  14     (tail : E→X) (head : E→Y) (eligible : E→Prop) (t : Y→ℕ)
  15     (ht : ∀ y, t y ≤ (eligibleIncoming head eligible y).card) :
  16     ∃ D : E→Bool,
  17       (∀ e, D e=true → eligible e) ∧
  18       (∀ y, (∑ e, if head e=y ∧ D e=true then (1:ℕ) else 0) = t y) ∧
  19       (∀ x, (∑ e, if tail e=x ∧ D e=true then (1:ℕ) else 0) ≤
  20         ⌈∑ e, if tail e=x then uniformIncomingWeight head eligible t e else 0⌉₊) := by
  21   obtain ⟨D,hrow,hcol,hsupport⟩ := bipartite_rounding_with_tail_ceilings_support tail head t
  22     (uniformIncomingWeight head eligible t) (uniform_incoming_weight_bounds head eligible t ht)
  23     (uniform_incoming_weight_columns head eligible t ht)
  24   refine ⟨D, ?_, hcol, hrow⟩
  25   intro e he
  26   by_contra hn
  27   have hz : uniformIncomingWeight head eligible t e = 0 := by simp [uniformIncomingWeight,hn]
  28   have hf := hsupport e hz
  29   rw [he] at hf
  30   cases hf
  31 theorem uniform_incoming_weight_upper
  32     (head : E→Y) (eligible : E→Prop) (t : Y→ℕ) (tau rho : ℝ)
  33     (htau : 0≤tau) (hrho : 0<rho)
  34     (ht : ∀ y, (t y:ℝ)≤tau)
  35     (hsize : ∀ y, rho≤((eligibleIncoming head eligible y).card:ℝ)) (e : E) :
  36     uniformIncomingWeight head eligible t e ≤ tau/rho := by
  37   unfold uniformIncomingWeight
  38   split_ifs with he
  39   · exact div_le_div₀ htau (ht (head e)) hrho (hsize (head e))
  40   · exact div_nonneg htau hrho.le

  42 theorem uniform_incoming_rounding_bounded
  43     (tail : E→X) (head : E→Y) (eligible : E→Prop) (t : Y→ℕ)
  44     (ht : ∀ y, t y ≤ (eligibleIncoming head eligible y).card)
  45     (tau rho : ℝ) (M : ℕ) (htau : 0≤tau) (hrho : 0<rho)
  46     (htauBound : ∀ y, (t y:ℝ)≤tau)
  47     (hsize : ∀ y, rho≤((eligibleIncoming head eligible y).card:ℝ))
  48     (hdegree : ∀ x, (Finset.univ.filter (fun e=>tail e=x)).card≤M) :
  49     ∃ D : E→Bool,
  50       (∀ e, D e=true → eligible e) ∧
  51       (∀ y, (∑ e, if head e=y ∧ D e=true then (1:ℕ) else 0) = t y) ∧
  52       (∀ x, (∑ e, if tail e=x ∧ D e=true then (1:ℕ) else 0) ≤ ⌈(M:ℝ)*(tau/rho)⌉₊) := by
  53   obtain ⟨D,hD,hheads,htails⟩ := uniform_incoming_rounding tail head eligible t ht
  54   refine ⟨D,hD,hheads,?_⟩
  55   intro x
  56   apply (htails x).trans
  57   apply Nat.ceil_mono
  58   let S := Finset.univ.filter (fun e=>tail e=x)
  59   have hc : 0≤tau/rho := div_nonneg htau hrho.le
  60   calc
  61     (∑ e, if tail e=x then uniformIncomingWeight head eligible t e else 0) =
  62         ∑ e∈S, uniformIncomingWeight head eligible t e := by
  63           simp only [S,Finset.sum_filter]
  64     _ ≤ ∑ _e∈S, tau/rho := by
  65       apply Finset.sum_le_sum
  66       intro e _
  67       exact uniform_incoming_weight_upper head eligible t tau rho htau hrho htauBound hsize e
  68     _ = (S.card:ℝ)*(tau/rho) := by simp
  69     _ ≤ (M:ℝ)*(tau/rho) := by
  70       apply mul_le_mul_of_nonneg_right _ hc
  71       exact_mod_cast hdegree x
  72 end Oberwolfach
