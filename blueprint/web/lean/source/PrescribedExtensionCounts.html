  13 theorem common_domain_restriction_card {T:Type*} [DecidableEq T]
  14     (A X:Finset T) (p:T→Prop) [DecidablePred p]:
  15     (A.filter p).card≤(X.filter p).card+(A\X).card:=by
  16   have hs:A.filter p⊆X.filter p∪(A\X):=by
  17     intro a ha
  18     by_cases hx:a∈X
  19     · exact Finset.mem_union_left _ (Finset.mem_filter.mpr ⟨hx,(Finset.mem_filter.mp ha).2⟩)
  20     · exact Finset.mem_union_right _ (Finset.mem_sdiff.mpr ⟨(Finset.mem_filter.mp ha).1,hx⟩)
  21   exact (Finset.card_le_card hs).trans (Finset.card_union_le _ _)

  23 theorem prescribed_edge_extension_counts (Q F:SimpleGraph V) (H:V→U→Prop)
  24     (A X:Finset V) (U₀ B:Finset U) (β:ℝ)
  25     (hXA:((A\X).card:ℝ)≤β*A.card/2) (hBU:((U₀\B).card:ℝ)≤β*U₀.card/2)
  26     (hrows:∀ S:Finset V,∀ J:Finset U,S.card≤2 → J.card≤1 →
  27       β*A.card≤((common (common A S (fun v=>Q.neighborFinset v)) J
  28         (fun u=>Finset.univ.filter (fun v=>H v u))).card:ℝ))
  29     (hindices:∀ S:Finset V,S.card≤2 → (∀ v∈S,∀ w∈S,¬F.Adj v w) →
  30       β*U₀.card≤((common U₀ S (fun v=>Finset.univ.filter (H v))).card:ℝ)) :
  31     (∀ u a c,β*A.card/2≤(((X.filter (fun v=>H v u))∩Q.neighborFinset a∩Q.neighborFinset c).card:ℝ)) ∧
  32     (∀ a c,¬F.Adj a c → β*U₀.card/2≤((B.filter (fun u=>H a u∧H c u)).card:ℝ)) :=by
  33   constructor
  34   · intro u a c
  35     have he (Z:Finset V):common (common Z ({a,c}:Finset V) (fun v=>Q.neighborFinset v)) ({u}:Finset U)
  36         (fun u=>Finset.univ.filter (fun v=>H v u))=Z.filter (fun v=>H v u∧Q.Adj a v∧Q.Adj c v):=by
  37       ext v;simp [mem_common_iff];tauto
  38     have he' (Z:Finset V):Z.filter (fun v=>H v u∧Q.Adj a v∧Q.Adj c v)=
  39         (Z.filter (fun v=>H v u))∩Q.neighborFinset a∩Q.neighborFinset c:=by
  40       ext v;simp;tauto
  41     have hh:=hrows {a,c} {u} Finset.card_le_two (by simp)
  42     rw [he] at hh
  43     have hc:((A.filter (fun v=>H v u∧Q.Adj a v∧Q.Adj c v)).card:ℝ)≤
  44       (X.filter (fun v=>H v u∧Q.Adj a v∧Q.Adj c v)).card+(A\X).card:=by
  45       exact_mod_cast common_domain_restriction_card A X (fun v=>H v u∧Q.Adj a v∧Q.Adj c v)
  46     rw [he' X] at hc
  47     linarith only [hh,hc,hXA]
  48   · intro a c hfac
  49     have hsep:∀ v∈({a,c}:Finset V),∀ w∈({a,c}:Finset V),¬F.Adj v w:=by
  50       intro v hv w hw h
  51       simp only [Finset.mem_insert,Finset.mem_singleton] at hv hw
  52       rcases hv with rfl|rfl <;> rcases hw with rfl|rfl
  53       · exact F.loopless.irrefl _ h
  54       · exact hfac h
  55       · exact hfac h.symm
  56       · exact F.loopless.irrefl _ h
  57     have he:common U₀ ({a,c}:Finset V) (fun v=>Finset.univ.filter (H v))=
  58         U₀.filter (fun u=>H a u∧H c u):=by ext u;simp [mem_common_iff]
  59     have hh:=hindices {a,c} Finset.card_le_two hsep
  60     rw [he] at hh
  61     have hc:((U₀.filter (fun u=>H a u∧H c u)).card:ℝ)≤
  62       (B.filter (fun u=>H a u∧H c u)).card+(U₀\B).card:=by
  63       exact_mod_cast common_domain_restriction_card U₀ B (fun u=>H a u∧H c u)
  64     linarith only [hh,hc,hBU]
