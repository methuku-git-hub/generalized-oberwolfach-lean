  13 theorem cyclicInterfaceGraph_fractional_cycles_with_pair_bounds {k : ℕ} {X : Type*}
  14     [Fintype X] [DecidableEq X] [Nonempty X]
  15     (L : Fin (k + 2) → X → X → Prop) (R : Fin (k + 2) → X → X → Prop)
  16     (δ α : ℝ) (hn : 0 < Fintype.card X) (hα : 0 < α)
  17     (hthreshold : ((k + 2) + (2 * (k + 2) - 1) * α) * Fintype.card X ≤
  18       (2 * (k + 2) - 1) * δ)
  19     (hdL : ∀ i x, δ ≤ (Finset.univ.filter (L i x)).card)
  20     (hdY : ∀ i y, δ ≤ (Finset.univ.filter (fun x => L i x y)).card)
  21     (hY : ∀ i y, (Finset.univ.filter (fun x => L i x y)).card =
  22       (Finset.univ.filter (R i y)).card)
  23     (hX : ∀ i z, (Finset.univ.filter (fun y => R i y z)).card =
  24       (Finset.univ.filter (L (finRotate (k + 2) i) z)).card) :
  25     ∃ w : SimpleGraph ((Fin (k + 2) × X) ⊕ (Fin (k + 2) × X)) → ℝ,
  26       (∀ H, 0 ≤ w H) ∧
  27       (∀ H, 0 < w H → H ≤ cyclicInterfaceGraph L R ∧
  28         IsEmbeddedCycle H ((k + 2) * 2) ∧
  29           ∃ p : Fin (k + 2) → X × X × X, H = alternatingTupleGraph p) ∧
  30       (∀ u v, (∑ H, if H.Adj u v then w H else 0) =
  31         if (cyclicInterfaceGraph L R).Adj u v then 1 else 0) ∧
  32       (∀ H, H ≤ cyclicInterfaceGraph L R →
  33         (∃ p : Fin (k + 2) → X × X × X, H = alternatingTupleGraph p) →
  34         ((k + 2 : ℝ) * cyclicSectorLower k α) / (Fintype.card X : ℝ) ^ (2 * k + 2) ≤ w H ∧
  35         w H ≤ ((k + 2 : ℝ) * cyclicSectorUpper k α) /
  36           (Fintype.card X : ℝ) ^ (2 * k + 2)) ∧
  37       (∀ a b c, a ≠ b → a ≠ c → b ≠ c →
  38         (∑ H, if H.Adj a b ∧ H.Adj a c then w H else 0) ≤
  39           ((k + 2 : ℝ) * cyclicSectorUpper k α) / Fintype.card X) ∧
  40       (∀ a b c d, a ≠ b → a ≠ c → a ≠ d → b ≠ c → b ≠ d → c ≠ d →
  41         (∑ H, if H.Adj a b ∧ H.Adj c d then w H else 0) ≤
  42           ((k + 2 : ℝ) * cyclicSectorUpper k α) / (Fintype.card X : ℝ) ^ 2) := by
  43   obtain ⟨w, hw, hs, he, hpoint⟩ := cyclicInterfaceGraph_fractional_cycles_bounded
  44     L R δ α hn hα rfl hthreshold hdL hdY hY hX
  45   have hB : 0 ≤ (k + 2 : ℝ) * cyclicSectorUpper k α :=
  46     (mul_pos (by positivity) (cyclicSectorConstants_pos k α hα).2).le
  47   have hn' : (0 : ℝ) < Fintype.card X := by exact_mod_cast hn
  48   have hsupport (H) (hH : 0 < w H) : H ∈ Set.range (cyclicSelectionGraph (k := k) (X := X)) := by
  49     obtain ⟨p, hp⟩ := (hs H hH).2.2
  50     exact ⟨cyclicSelectionOfTuple p, (cyclicSelectionGraph_ofTuple p).trans hp.symm⟩
  51   have hbound (f : Fin (k + 2) ⊕ Fin (k + 2) → X) :
  52       w (cyclicSelectionGraph f) ≤ ((k + 2 : ℝ) * cyclicSectorUpper k α) /
  53         (Fintype.card X : ℝ) ^ (2 * k + 2) := by
  54     by_cases hz : w (cyclicSelectionGraph f) = 0
  55     · rw [hz]
  56       exact div_nonneg hB (pow_pos hn' _).le
  57     · have hp := lt_of_le_of_ne (hw _) (Ne.symm hz)
  58       exact (hpoint _ (hs _ hp).1 (hs _ hp).2.2).2
  59   refine ⟨w, hw, hs, he, hpoint, ?_, ?_⟩
  60   · intro a b c hab hac hbc
  61     exact cyclic_graph_adjacent_pair_load w hw hsupport _ hB hn hbound a b c hab hac hbc
  62   · intro a b c d hab hac had hbc hbd hcd
  63     exact cyclic_graph_disjoint_pair_load w hw hsupport _ hB hn hbound a b c d hab hac had hbc hbd hcd
