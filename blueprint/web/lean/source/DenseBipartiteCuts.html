  12 theorem matrixCut_large_of_minDegree {V : Type*} [Fintype V]
  13     (M : V → V → ℕ) (hM : ∀ x y, M x y ≤ 1) (β δ : ℝ)
  14     (hrow : ∀ x, (1 / 2 + β) * Fintype.card V ≤ ((∑ y, M x y : ℕ) : ℝ))
  15     (hcol : ∀ y, (1 / 2 + β) * Fintype.card V ≤ ((∑ x, M x y : ℕ) : ℝ))
  16     (S T : Finset V)
  17     (hlarge : (1 - 2 * δ) * Fintype.card V < (S.card : ℝ) + T.card) :
  18     (β - δ) * Fintype.card V * min (S.card : ℝ) T.card ≤ (matrixCut M S T : ℝ) := by
  19   have hsmallLarge (A B : Finset V) (N : V → V → ℕ)
  20       (hN : ∀ x y, N x y ≤ 1)
  21       (hdeg : ∀ x, (1 / 2 + β) * Fintype.card V ≤ ((∑ y, N x y : ℕ) : ℝ))
  22       (hab : (A.card : ℝ) ≤ B.card)
  23       (hlarge : (1 - 2 * δ) * Fintype.card V < (A.card : ℝ) + B.card) :
  24       (β - δ) * Fintype.card V * A.card ≤ (matrixCut N A B : ℝ) := by
  25     have hb : (1 / 2 - δ) * Fintype.card V ≤ (B.card : ℝ) := by linarith
  26     have hh := matrixCut_row_lower N hN A B (fun x _ => hdeg x)
  27     have hm := mul_le_mul_of_nonneg_left hb (show (0 : ℝ) ≤ A.card by positivity)
  28     nlinarith
  29   rcases le_total (S.card : ℝ) T.card with h | h
  30   · rw [min_eq_left h]
  31     exact hsmallLarge S T M hM hrow h hlarge
  32   · rw [min_eq_right h, matrixCut_transpose]
  33     exact hsmallLarge T S (fun x y => M y x) (fun x y => hM y x) hcol h (by linarith)

  35 theorem matrixEdges_normalized_large_cuts {V : Type*} [Fintype V] [DecidableEq V]
  36     (M : V → V → ℕ) (hM : ∀ x y, M x y ≤ 1) (β δ h : ℝ)
  37     (hn : 0 < Fintype.card V) (hh : h ≤ β - δ)
  38     (hrow : ∀ x, (1 / 2 + β) * Fintype.card V ≤ ((∑ y, M x y : ℕ) : ℝ))
  39     (hcol : ∀ y, (1 / 2 + β) * Fintype.card V ≤ ((∑ x, M x y : ℕ) : ℝ))
  40     (S T : Finset V)
  41     (hlarge : 1 - 2 * δ < (S.card : ℝ) / Fintype.card V + (T.card : ℝ) / Fintype.card V) :
  42     h * Fintype.card V * Fintype.card V *
  43       min ((S.card : ℝ) / Fintype.card V) ((T.card : ℝ) / Fintype.card V) ≤
  44     ((Finset.univ.filter (fun e : BipartiteEdges M => e.1 ∈ S ∧ e.2.1 ∈ T)).card : ℝ) := by
  45   have hn' : (0 : ℝ) < Fintype.card V := by exact_mod_cast hn
  46   have hl : (1 - 2 * δ) * Fintype.card V < (S.card : ℝ) + T.card := by
  47     rw [← add_div] at hlarge
  48     exact (lt_div_iff₀ hn').mp hlarge
  49   have hcut := matrixCut_large_of_minDegree M hM β δ hrow hcol S T hl
  50   have hm0 : 0 ≤ min (S.card : ℝ) T.card := le_min (Nat.cast_nonneg _) (Nat.cast_nonneg _)
  51   have hcut' : h * Fintype.card V * min (S.card : ℝ) T.card ≤ (matrixCut M S T : ℝ) :=
  52     (mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_right hh hn'.le) hm0).trans hcut
  53   rw [min_div_div_right hn'.le]
  54   have hcancel : h * Fintype.card V * Fintype.card V *
  55       (min (S.card : ℝ) T.card / Fintype.card V) =
  56       h * Fintype.card V * min (S.card : ℝ) T.card := by field_simp
  57   rw [hcancel, bipartiteEdges_filter_card M (fun x y => x ∈ S ∧ y ∈ T)]
  58   convert hcut' using 1
  59   simp only [matrixCut]
  60   congr 1
  61   calc
  62     _ = ∑ x : V, if x ∈ S then ∑ y ∈ T, M x y else 0 := by
  63       apply Finset.sum_congr rfl
  64       intro x _
  65       by_cases hx : x ∈ S
  66       · simp [hx, ← Finset.sum_filter]
  67       · simp [hx]
  68     _ = _ := by rw [← Finset.sum_filter]; simp
