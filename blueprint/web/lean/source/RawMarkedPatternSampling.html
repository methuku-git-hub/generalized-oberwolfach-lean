  20 theorem rawMarkedResourceCoordinates_disjoint
  21     (π : Equiv.Perm V) (d : ℕ) (a b : MarkedResource V U)
  22     (h : match a,b with
  23       | Sum.inl e,Sum.inl f=>markedPairPhysical π e≠markedPairPhysical π f
  24       | Sum.inr x,Sum.inr y=>x.2.2≠y.2.2 ∨ ¬PermutationClose π (3*d) x.2.1 y.2.1
  25       | _,_=>True) :
  26     Disjoint (rawMarkedResourceCoordinates π d a) (rawMarkedResourceCoordinates π d b) := by
  27   cases a with
  28   | inl a=>cases b with
  29     | inl b=>simpa only [rawMarkedResourceCoordinates,Finset.disjoint_singleton,ne_eq,Sum.inl.injEq] using h
  30     | inr b=>simp [rawMarkedResourceCoordinates]
  31   | inr a=>cases b with
  32     | inl b=>simp [rawMarkedResourceCoordinates]
  33     | inr b=>
  34       apply Finset.disjoint_left.mpr
  35       intro e he hf
  36       obtain ⟨x,hx,he⟩ := Finset.mem_image.mp he
  37       obtain ⟨y,hy,hf⟩ := Finset.mem_image.mp hf
  38       have hxy := Sum.inr.inj (he.trans hf.symm)
  39       exact Finset.disjoint_left.mp (rawIntervalFamilyCoordinates_disjoint π d a.2 b.2 h) hx (hxy ▸ hy)
