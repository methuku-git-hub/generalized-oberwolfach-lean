  13  theorem fixed_class_root_order (m:ℕ) (A B X:Finset V) (hAB:Disjoint A B)
  14     (hX:X⊆A∪B) (hA:(X∩A).card=1) (hB:(X∩B).card=m):
  15     ∃ x:Fin (m+1)→V,Function.Injective x∧Finset.univ.image x=X∧x 0∈A∧∀ i:Fin m,x i.succ∈B:=by
  16   obtain ⟨a,ha⟩:=Finset.card_eq_one.mp hA
  17   have haX:a∈X∩A:=by rw [ha];simp
  18   let e:Fin m≃{v:V // v∈X∩B}:=(Fintype.equivFinOfCardEq (by simpa only [Fintype.card_coe] using hB)).symm
  19   let x:Fin (m+1)→V:=Fin.cons a (fun i=>(e i).val)
  20   have haB:a∉B:=fun h=>Finset.disjoint_left.mp hAB (Finset.mem_inter.mp haX).2 h
  21   have hx:Function.Injective x:=by
  22     intro i j hij
  23     revert j
  24     refine Fin.cases ?_ (fun i=>?_) i
  25     · intro j hij
  26       revert hij
  27       refine Fin.cases (fun _=>rfl) (fun j hij=>?_) j
  28       have he:a=(e j).val:=hij
  29       exact False.elim (haB (he.symm ▸ (Finset.mem_inter.mp (e j).property).2))
  30     · intro j hij
  31       revert hij
  32       refine Fin.cases (fun hij=>?_) (fun j hij=>?_) j
  33       · have he:(e i).val=a:=hij
  34         exact False.elim (haB (he ▸ (Finset.mem_inter.mp (e i).property).2))
  35       · exact congrArg Fin.succ (e.injective (Subtype.ext hij))
  36   have hset:Finset.univ.image x=X:=by
  37     ext v
  38     constructor
  39     · intro hv
  40       obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hv
  41       refine Fin.cases ?_ (fun i=>?_) i
  42       · exact (Finset.mem_inter.mp haX).1
  43       · exact (Finset.mem_inter.mp (e i).property).1
  44     · intro hv
  45       rcases Finset.mem_union.mp (hX hv) with hvA|hvB
  46       · have he:v=a:=by have hh: v∈X∩A:=Finset.mem_inter.mpr ⟨hv,hvA⟩;rw [ha] at hh;exact Finset.mem_singleton.mp hh
  47         exact Finset.mem_image.mpr ⟨0,Finset.mem_univ _,he.symm⟩
  48       · let w:{v:V // v∈X∩B}:=⟨v,Finset.mem_inter.mpr ⟨hv,hvB⟩⟩
  49         refine Finset.mem_image.mpr ⟨(e.symm w).succ,Finset.mem_univ _,?_⟩
  50         change (e (e.symm w)).val=v
  51         rw [e.apply_symm_apply]
  52   exact ⟨x,hx,hset,(Finset.mem_inter.mp haX).2,fun i=>(Finset.mem_inter.mp (e i).property).2⟩
