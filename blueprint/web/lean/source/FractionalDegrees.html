  13 def bipartiteNetworkTail {X Y E : Type*} (tail : E → X) :
  14     X ⊕ E ⊕ Y → Bool ⊕ X ⊕ Y
  15   | .inl _ => .inl false
  16   | .inr (.inl e) => .inr (.inl (tail e))
  17   | .inr (.inr y) => .inr (.inr y)
  18 
  19 def bipartiteNetworkHead {X Y E : Type*} (head : E → Y) :
  20     X ⊕ E ⊕ Y → Bool ⊕ X ⊕ Y
  21   | .inl x => .inr (.inl x)
  22   | .inr (.inl e) => .inr (.inr (head e))
  23   | .inr (.inr _) => .inl true

  25 def bipartiteNetworkCapacity {X Y E : Type*}
  26     (f : X → ℝ) (g : Y → ℝ) (c : E → ℝ) : X ⊕ E ⊕ Y → ℝ
  27   | .inl x => f x
  28   | .inr (.inl e) => c e
  29   | .inr (.inr y) => g y

  31 theorem bipartiteNetwork_divergence
  32     {X Y E : Type*} [Fintype X] [Fintype Y] [Fintype E]
  33     [DecidableEq X] [DecidableEq Y]
  34     (tail : E → X) (head : E → Y) (q : X ⊕ E ⊕ Y → ℝ) :
  35     flowDivergence (bipartiteNetworkTail tail) (bipartiteNetworkHead head) q
  36       (.inl false) = ∑ x, q (.inl x) ∧
  37     flowDivergence (bipartiteNetworkTail tail) (bipartiteNetworkHead head) q
  38       (.inl true) = -(∑ y, q (.inr (.inr y))) ∧
  39     (∀ x, flowDivergence (bipartiteNetworkTail tail) (bipartiteNetworkHead head) q
  40       (.inr (.inl x)) = (∑ e, if tail e = x then q (.inr (.inl e)) else 0) -
  41         q (.inl x)) ∧
  42     (∀ y, flowDivergence (bipartiteNetworkTail tail) (bipartiteNetworkHead head) q
  43       (.inr (.inr y)) = q (.inr (.inr y)) -
  44         ∑ e, if head e = y then q (.inr (.inl e)) else 0) := by
  45   simp [flowDivergence, vertexDifference, Fintype.sum_sum_type,
  46     bipartiteNetworkTail, bipartiteNetworkHead, mul_ite, Finset.sum_neg_distrib,
  47     sub_eq_add_neg, add_comm]

  50 theorem bipartiteNetwork_cut
  51     {X Y E : Type*} [Fintype X] [Fintype Y] [Fintype E]
  52     [DecidableEq X] [DecidableEq Y]
  53     (tail : E → X) (head : E → Y) (f : X → ℝ) (g : Y → ℝ) (c : E → ℝ)
  54     (R : Finset (Bool ⊕ X ⊕ Y)) (hs : Sum.inl false ∈ R) (ht : Sum.inl true ∉ R) :
  55     networkCutCapacity (bipartiteNetworkTail tail) (bipartiteNetworkHead head)
  56       (bipartiteNetworkCapacity f g c) R =
  57       (∑ x, if Sum.inr (Sum.inl x) ∉ R then f x else 0) +
  58       (∑ e, if Sum.inr (Sum.inl (tail e)) ∈ R ∧
  59         Sum.inr (Sum.inr (head e)) ∉ R then c e else 0) +
  60       ∑ y, if Sum.inr (Sum.inr y) ∈ R then g y else 0 := by
  61   simp [networkCutCapacity, Fintype.sum_sum_type, bipartiteNetworkTail,
  62     bipartiteNetworkHead, bipartiteNetworkCapacity, hs, ht, add_assoc]

  64 /-- Real demands are feasible exactly under the bipartite cut inequalities.
  65 This direction supplies the fractional edge weights used later in the
  66 manuscript; the integer version is proved separately by Hall's theorem. -/
  67 theorem bipartite_fractionalDegrees
  68     {X Y E : Type*} [Fintype X] [Fintype Y] [Fintype E]
  69     [DecidableEq X] [DecidableEq Y]
  70     (tail : E → X) (head : E → Y) (c : E → ℝ)
  71     (f : X → ℝ) (g : Y → ℝ)
  72     (hc : ∀ e, 0 ≤ c e) (hf : ∀ x, 0 ≤ f x) (hg : ∀ y, 0 ≤ g y)
  73     (htotal : ∑ x, f x = ∑ y, g y)
  74     (hcut : ∀ S : Finset X, ∀ T : Finset Y,
  75       ∑ x ∈ S, f x ≤ (∑ y ∈ T, g y) +
  76         ∑ e, if tail e ∈ S ∧ head e ∉ T then c e else 0) :
  77     ∃ q : E → ℝ, CapacityBounded c q ∧
  78       (∀ x, (∑ e, if tail e = x then q e else 0) = f x) ∧
  79       (∀ y, (∑ e, if head e = y then q e else 0) = g y) := by
  80   classical
  81   let a := bipartiteNetworkTail (Y := Y) tail
  82   let b := bipartiteNetworkHead (X := X) head
  83   let cap := bipartiteNetworkCapacity f g c
  84   have hcap : ∀ e, 0 ≤ cap e := by
  85     rintro (x | e | y)
  86     · exact hf x
  87     · exact hc e
  88     · exact hg y
  89   obtain ⟨q, hq, R, hs, ht, heq⟩ :=
  90     real_maxFlow_minCut a b cap hcap (.inl false) (.inl true) (by simp)
  91   let S := Finset.univ.filter (fun x => Sum.inr (Sum.inl x) ∈ R)
  92   let T := Finset.univ.filter (fun y => Sum.inr (Sum.inr y) ∈ R)
  93   have hsplit : ∑ x, f x = (∑ x ∈ S, f x) +
  94       ∑ x, if Sum.inr (Sum.inl x) ∉ R then f x else 0 := by
  95     simp only [S, Finset.sum_filter, ← Finset.sum_add_distrib]
  96     apply Finset.sum_congr rfl
  97     intro x _
  98     by_cases hx : Sum.inr (Sum.inl x) ∈ R <;> simp [hx]
  99   have hcutR : ∑ x, f x ≤ networkCutCapacity a b cap R := by
 100     have hh := hcut S T
 101     simp only [S, T, Finset.mem_filter, Finset.mem_univ, true_and,
 102       Finset.sum_filter] at hh
 103     rw [bipartiteNetwork_cut tail head f g c R hs ht]
 104     rw [hsplit]
 105     simpa only [S, Finset.sum_filter] using (by linarith : _)
 106   obtain ⟨hsource, _, hleft, hright⟩ := bipartiteNetwork_divergence tail head q
 107   have hqf (x : X) : q (.inl x) ≤ f x := (hq.1 (.inl x)).2
 108   have hqg (y : Y) : q (.inr (.inr y)) ≤ g y := (hq.1 (.inr (.inr y))).2
 109   have hsourceTotal : ∑ x, q (.inl x) = ∑ x, f x := by
 110     apply le_antisymm (Finset.sum_le_sum (fun x _ => hqf x))
 111     rw [← hsource]
 112     exact heq ▸ hcutR
 113   have hleftEq (x : X) :
 114       (∑ e, if tail e = x then q (.inr (.inl e)) else 0) = q (.inl x) := by
 115     have hz := hq.2 (.inr (.inl x)) (by simp) (by simp)
 116     rw [hleft] at hz
 117     linarith
 118   have hrightEq (y : Y) :
 119       (∑ e, if head e = y then q (.inr (.inl e)) else 0) = q (.inr (.inr y)) := by
 120     have hz := hq.2 (.inr (.inr y)) (by simp) (by simp)
 121     rw [hright] at hz
 122     linarith
 123   have hmiddleTotal : ∑ x, q (.inl x) = ∑ e, q (.inr (.inl e)) := by
 124     simp_rw [← hleftEq]
 125     rw [Finset.sum_comm]
 126     simp
 127   have hsinkTotal : ∑ y, q (.inr (.inr y)) = ∑ y, g y := by
 128     calc
 129       ∑ y, q (.inr (.inr y)) = ∑ e, q (.inr (.inl e)) := by
 130         simp_rw [← hrightEq]
 131         rw [Finset.sum_comm]
 132         simp
 133       _ = ∑ y, g y := by rw [← hmiddleTotal, hsourceTotal, htotal]
 134   have hsourceFull (x : X) : q (.inl x) = f x :=
 135     (Finset.sum_eq_sum_iff_of_le (fun x _ => hqf x)).mp hsourceTotal x (Finset.mem_univ x)
 136   have hsinkFull (y : Y) : q (.inr (.inr y)) = g y :=
 137     (Finset.sum_eq_sum_iff_of_le (fun y _ => hqg y)).mp hsinkTotal y (Finset.mem_univ y)
 138   exact ⟨fun e => q (.inr (.inl e)), fun e => hq.1 (.inr (.inl e)),
 139     fun x => (hleftEq x).trans (hsourceFull x),
 140     fun y => (hrightEq y).trans (hsinkFull y)⟩

 142 theorem bipartite_fractionalDegrees_cutNecessary
 143     {X Y E : Type*} [Fintype X] [Fintype Y] [Fintype E]
 144     [DecidableEq X] [DecidableEq Y]
 145     (tail : E → X) (head : E → Y) (c q : E → ℝ)
 146     (f : X → ℝ) (g : Y → ℝ) (hq : CapacityBounded c q)
 147     (hleft : ∀ x, (∑ e, if tail e = x then q e else 0) = f x)
 148     (hright : ∀ y, (∑ e, if head e = y then q e else 0) = g y)
 149     (S : Finset X) (T : Finset Y) :
 150     ∑ x ∈ S, f x ≤ (∑ y ∈ T, g y) +
 151       ∑ e, if tail e ∈ S ∧ head e ∉ T then c e else 0 := by
 152   have hs : ∑ x ∈ S, f x = ∑ e, if tail e ∈ S then q e else 0 := by
 153     simp_rw [← hleft]
 154     rw [Finset.sum_comm]
 155     simp
 156   have ht : ∑ y ∈ T, g y = ∑ e, if head e ∈ T then q e else 0 := by
 157     simp_rw [← hright]
 158     rw [Finset.sum_comm]
 159     simp
 160   rw [hs, ht, ← Finset.sum_add_distrib]
 161   apply Finset.sum_le_sum
 162   intro e _
 163   by_cases hx : tail e ∈ S <;> by_cases hy : head e ∈ T <;> simp [hx, hy]
 164   · exact (hq e).2
 165   · exact (hq e).1

 168     {X Y E : Type*} [Fintype X] [Fintype Y] [Fintype E]
 169     [DecidableEq X] [DecidableEq Y]
 170     (tail : E → X) (head : E → Y) (c : E → ℝ)
 171     (f : X → ℝ) (g : Y → ℝ)
 172     (hc : ∀ e, 0 ≤ c e) (hf : ∀ x, 0 ≤ f x) (hg : ∀ y, 0 ≤ g y)
 173     (htotal : ∑ x, f x = ∑ y, g y) :
 174     (∃ q : E → ℝ, CapacityBounded c q ∧
 175       (∀ x, (∑ e, if tail e = x then q e else 0) = f x) ∧
 176       (∀ y, (∑ e, if head e = y then q e else 0) = g y)) ↔
 177     (∀ S : Finset X, ∀ T : Finset Y,
 178       ∑ x ∈ S, f x ≤ (∑ y ∈ T, g y) +
 179         ∑ e, if tail e ∈ S ∧ head e ∉ T then c e else 0) := by

 181   · rintro ⟨q, hq, hleft, hright⟩ S T
 182     exact bipartite_fractionalDegrees_cutNecessary tail head c q f g hq hleft hright S T
 183   · exact bipartite_fractionalDegrees tail head c f g hc hf hg htotal

 185 /-- The fractional conclusion of the normalized-demand lemma, with the
 186 original part sizes and real demands. The lower demand estimates in the
 187 manuscript imply nonnegativity; only the displayed upper estimates are
 188 needed in the cut calculation. -/
 189 theorem bipartite_nearUniformFractionalDegrees
 190     {X Y E : Type*} [Fintype X] [Fintype Y] [Fintype E]
 191     [DecidableEq X] [DecidableEq Y]
 192     (tail : E → X) (head : E → Y) (f : X → ℝ) (g : Y → ℝ)
 193     {δ δ₀ M h : ℝ}
 194     (ha : 0 < Fintype.card X) (hb : 0 < Fintype.card Y)
 195     (hδ : 0 ≤ δ) (hδ₀ : δ ≤ δ₀) (hδsmall : δ₀ < 1 / 4)
 196     (hM : 0 ≤ M)
 197     (hMC : M ≤ h * Fintype.card X * Fintype.card Y / 2)
 198     (hftotal : ∑ x, f x = M) (hgtotal : ∑ y, g y = M)
 199     (hfnonneg : ∀ x, 0 ≤ f x) (hgnonneg : ∀ y, 0 ≤ g y)
 200     (hf : ∀ x, f x ≤ (1 + δ) * M / Fintype.card X)
 201     (hg : ∀ y, g y ≤ (1 + δ) * M / Fintype.card Y)
 202     (hlarge : ∀ (S : Finset X) (T : Finset Y),
 203       1 - 2 * δ₀ < (S.card : ℝ) / Fintype.card X + (T.card : ℝ) / Fintype.card Y →
 204       h * Fintype.card X * Fintype.card Y *
 205           min ((S.card : ℝ) / Fintype.card X) ((T.card : ℝ) / Fintype.card Y) ≤
 206         ((Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∈ T)).card : ℝ)) :
 207     ∃ q : E → ℝ, (∀ e, q e ∈ Set.Icc 0 1) ∧
 208       (∀ x, (∑ e, if tail e = x then q e else 0) = f x) ∧
 209       (∀ y, (∑ e, if head e = y then q e else 0) = g y) := by
 210   classical
 211   apply bipartite_fractionalDegrees tail head (fun _ => 1) f g
 212     (fun _ => zero_le_one) hfnonneg hgnonneg (hftotal.trans hgtotal.symm)
 213   intro S T
 214   let U := Tᶜ
 215   have hFs : ∑ x ∈ S, f x ≤ M := by
 216     rw [← hftotal]
 217     exact Finset.sum_le_univ_sum_of_nonneg hfnonneg
 218   have hFt : ∑ y ∈ U, g y ≤ M := by
 219     rw [← hgtotal]
 220     exact Finset.sum_le_univ_sum_of_nonneg hgnonneg
 221   have hFsbound : ∑ x ∈ S, f x ≤
 222       (1 + δ) * M * ((S.card : ℝ) / Fintype.card X) := by
 223     calc
 224       _ ≤ ∑ _ ∈ S, (1 + δ) * M / Fintype.card X :=
 225         Finset.sum_le_sum (fun x _ => hf x)
 226       _ = _ := by simp only [Finset.sum_const, nsmul_eq_mul]; ring
 227   have hFtbound : ∑ y ∈ U, g y ≤
 228       (1 + δ) * M * ((U.card : ℝ) / Fintype.card Y) := by
 229     calc
 230       _ ≤ ∑ _ ∈ U, (1 + δ) * M / Fintype.card Y :=
 231         Finset.sum_le_sum (fun y _ => hg y)
 232       _ = _ := by simp only [Finset.sum_const, nsmul_eq_mul]; ring
 233   have hdeficit := normalizedDemand_cut hδ hδ₀ hδsmall hM
 234     (show 2 * M ≤ h * Fintype.card X * Fintype.card Y by linarith)
 235     (div_nonneg (Nat.cast_nonneg S.card) (Nat.cast_pos.mpr ha).le)
 236     (div_nonneg (Nat.cast_nonneg U.card) (Nat.cast_pos.mpr hb).le)
 237     hFs hFt hFsbound hFtbound
 238     (Nat.cast_nonneg (Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∈ U)).card)
 239     (hlarge S U)
 240   have hsum : (∑ y ∈ T, g y) + (∑ y ∈ U, g y) = M := by
 241     rw [show U = Tᶜ from rfl, Finset.sum_add_sum_compl, hgtotal]
 242   have he : ((Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∈ U)).card : ℝ) =
 243       ∑ e, if tail e ∈ S ∧ head e ∉ T then (1 : ℝ) else 0 := by
 244     simp only [Finset.card_filter, Nat.cast_sum, Nat.cast_ite, Nat.cast_one,
 245       Nat.cast_zero, U, Finset.mem_compl]
 246   rw [he] at hdeficit
 247   linarith
