  16  theorem fresh_transformer_hubs (F:SimpleGraph V) (H:V→U→Prop)
  17     (S:I→Finset V) (B:Finset U) (k:ℕ) (b:ℝ)
  18     (hS:∀ i,(S i).card≤k) (hSF:∀ i,∀ v∈S i,∀ w∈S i,¬F.Adj v w)
  19     (hcount:∀ T:Finset V,T.card≤k→(∀ v∈T,∀ w∈T,¬F.Adj v w)→
  20       b≤((common (Finset.univ:Finset U) T (fun v=>Finset.univ.filter (H v))).card:ℝ))
  21     (hbudget:B.card+Fintype.card I<b):
  22     ∃ h:I↪U,(∀ i,h i∉B) ∧ (∀ i,∀ v∈S i,H v (h i)):=by
  23   have hchoose:∀ i,∀ C:Finset U,C.card≤B.card+Fintype.card I*1→
  24       ∃ u,(∀ v∈S i,H v u) ∧ ({u}:Finset U).card≤1 ∧ Disjoint ({u}:Finset U) C:=by
  25     intro i C hC
  26     have hCr:(C.card:ℝ)≤B.card+Fintype.card I:=by exact_mod_cast (by simpa using hC)
  27     have hlt:C.card<(common (Finset.univ:Finset U) (S i) (fun v=>Finset.univ.filter (H v))).card:=by
  28       exact_mod_cast (hCr.trans_lt hbudget).trans_le (hcount (S i) (hS i) (hSF i))
  29     obtain ⟨u,hu,huC⟩:=Finset.exists_mem_notMem_of_card_lt_card hlt
  30     refine ⟨u,?_,by simp,by simpa using huC⟩
  31     simpa only [common,Finset.mem_filter,Finset.mem_univ,true_and] using hu
  32   obtain ⟨h,hh,hd⟩:=finite_disjoint_choices (fun u:U=>({u}:Finset U))
  33     (fun i u=>∀ v∈S i,H v u) B 1 hchoose
  34   have hinj:Function.Injective h:=by
  35     intro i j hij
  36     by_contra hne
  37     have hn:Disjoint ({h i}:Finset U) {h j}:=hd i j hne
  38     simpa [hij] using hn
