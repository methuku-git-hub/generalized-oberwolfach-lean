  20 theorem ordinary_wheel_fixed_hub_event (S : Finset V) (hS : 2≤S.card) (u : U)
  21     (P : OrdinaryWheel Q F H r → Prop)
  22     (hP : ∀ W,P W → (↑S:Set V)⊆(ordinaryWheelRim W).support ∧ ordinaryWheelHub W=u) :
  23     (∑ W,if P W then w W else 0)≤(r:ℝ)^S.card*B/(Fintype.card V:ℝ)^(S.card-1) := by
  24   have hh := ordinary_wheel_event_weight_bound Q F H r hr hn w B hB hw S hS {u} P
  25     (by intro W hp;exact ⟨(hP W hp).1,Finset.mem_singleton.mpr (hP W hp).2⟩)
  26   simp only [Finset.card_singleton,Nat.cast_one,one_mul] at hh
  27   convert hh using 1
  28   all_goals
  29     apply Finset.sum_congr
  30     · ext W;simp
  31     · intro W _;split_ifs <;> rfl

  33 theorem ordinary_wheel_free_hub_event (hu : Fintype.card U≤Fintype.card V)
  34     (S : Finset V) (hS : 2≤S.card) (P : OrdinaryWheel Q F H r → Prop)
  35     (hP : ∀ W,P W → (↑S:Set V)⊆(ordinaryWheelRim W).support) :
  36     (∑ W,if P W then w W else 0)≤(r:ℝ)^S.card*B/(Fintype.card V:ℝ)^(S.card-2) := by
  37   have hh := ordinary_wheel_event_weight_bound Q F H r hr hn w B hB hw S hS Finset.univ P
  38     (by intro W hp;exact ⟨hP W hp,Finset.mem_univ _⟩)
  39   simp only [Finset.card_univ] at hh
  40   have hn0 : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr hn
  41   apply hh.trans
  42   calc
  43     _ ≤ (Fintype.card V:ℝ)*(r:ℝ)^S.card*B/(Fintype.card V:ℝ)^(S.card-1) := by
  44       apply div_le_div_of_nonneg_right _ (by positivity)
  45       apply mul_le_mul_of_nonneg_right _ hB
  46       apply mul_le_mul_of_nonneg_right _ (by positivity)
  47       exact_mod_cast hu
  48     _ = _ := by
  49       have hp : (Fintype.card V:ℝ)^(S.card-1)=(Fintype.card V:ℝ)^(S.card-2)*Fintype.card V := by
  50         rw [← pow_succ,show S.card-2+1=S.card-1 by omega]
  51       rw [hp]
  52       field_simp
