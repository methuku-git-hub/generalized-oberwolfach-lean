  14  theorem injective_occurrence_sum {I X : Type*} [Fintype I] [DecidableEq X]
  15     (f : I → X) (hf : Function.Injective f) (x : X) :
  16     (∑ i,if f i=x then (1:ℝ) else 0)=if x∈Finset.univ.image f then 1 else 0 := by
  17   classical
  18   by_cases h : ∃ i,f i=x
  19   · obtain ⟨i,hi⟩ := h
  20     have hmem : x∈Finset.univ.image f := Finset.mem_image.mpr ⟨i,Finset.mem_univ _,hi⟩
  21     rw [if_pos hmem,Finset.sum_eq_single i]
  22     · simp only [hi,ite_true]
  23     · intro j _ hji
  24       exact if_neg (fun hj=>hji (hf (hj.trans hi.symm)))
  25     · simp
  26   · have hmem : x∉Finset.univ.image f := by simpa only [Finset.mem_image,Finset.mem_univ,true_and] using h
  27     rw [if_neg hmem]
  28     apply Finset.sum_eq_zero
  29     intro i _
  30     exact if_neg (fun hi=>h ⟨i,hi⟩)

  60     (hp : Function.Injective (markedPhysicalEdge tail head ktail khead π))
  61     (C : MarkedCycleTuple tail head ktail khead m π b) (e : E⊕K) :

  63       if markedPhysicalEdge tail head ktail khead π e∈
  64         markedCyclePhysicalEdges tail head ktail khead m π b C then 1 else 0 := by
