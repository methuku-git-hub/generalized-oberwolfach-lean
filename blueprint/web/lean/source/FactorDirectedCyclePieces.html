  13       (D:V→V→Prop),
  14       (∀ c,(PrescribedSourcePiece.cycle (root c) (tail c)).Valid D ⊥) ∧
  15       (∀ c x,x∈root c::tail c ↔ x∈c.supp) ∧
  16       (∀ c c' x,x∈root c::tail c→x∈root c'::tail c'→c=c') ∧
  17       (∀ x,∃ c,x∈root c::tail c) ∧
  18       (∀ x y,D x y→F.Adj x y) ∧
  19       (∀ x y,F.Adj x y ↔ ∃ c,x∈root c::tail c ∧
  20         s(x,y)∈(PrescribedSourcePiece.cycle (root c) (tail c)).edges) := by

  23   refine ⟨root,tail,D,?_,hverts,hdisj,hcover,?_,?_⟩
  24   · intro c
  25     change (root c::tail c).Nodup ∧ 3≤(root c::tail c).length ∧

  27     refine ⟨(hcycle c).1,(hcycle c).2,?_⟩
  28     apply (directed_cycle_through_iff_chain D ⊥ (root c) (tail c)).2
