   9 theorem ordinary_common_scales_for_bounded_lengths
  10     (K:ℕ) (hK:3≤K) (α p ω:ℝ) (hα:0<α) (hp:0<p) (hp1:p≤1) (hω:0<ω):
  11     ∃ β row:ℝ,0<β ∧ 0<row ∧
  12       β≤2*(min (α/2) (1/4))*p^4*p^4 ∧ β≤ω*p^9/4 ∧
  13       ∀m:ℕ,3≤m→m≤K→
  14         β≤(α/2)*p^2*p^2 ∧ β≤ω*p^(m+7)/4 ∧ row≤p^(m+7)/2 := by
  15   let A:ℝ:=(min (α/2) (1/4))*p^8
  16   let B:ℝ:=ω*p^(K+7)/4
  17   let β:ℝ:=min A B/2
  18   let row:ℝ:=p^(K+7)/4
  19   have hA:0<A:=by dsimp [A];positivity
  20   have hB:0<B:=by dsimp [B];positivity
  21   have hβ:0<β:=by dsimp [β];positivity
  22   have hβA:β≤A:=by dsimp [β];have hh:=min_le_left A B;linarith
  23   have hβB:β≤B:=by dsimp [β];have hh:=min_le_right A B;linarith
  24   have hrow:0<row:=by dsimp [row];positivity
  25   have h8:p^8≤p^4:=pow_le_pow_of_le_one hp.le hp1 (by omega:4≤8)
  26   have h9:p^(K+7)≤p^9:=pow_le_pow_of_le_one hp.le hp1 (by omega:9≤K+7)
  27   have hmin:0<min (α/2) (1/4):=by positivity
  28   have hminα:min (α/2) (1/4)≤α/2:=min_le_left _ _
  29   refine ⟨β,row,hβ,hrow,?_,?_,?_⟩
  30   · dsimp [A] at hβA
  31     nlinarith [pow_nonneg hp.le 4]
  32   · dsimp [B] at hβB
  33     nlinarith
  34   · intro m hm hmK
  35     have hpow:p^(K+7)≤p^(m+7):=pow_le_pow_of_le_one hp.le hp1 (by omega)
  36     refine ⟨?_,?_,?_⟩
  37     · dsimp [A] at hβA
  38       have hstep:=mul_le_mul_of_nonneg_left h8 hmin.le
  39       have halpha:=mul_le_mul_of_nonneg_right hminα (pow_nonneg hp.le 4)
  40       nlinarith [pow_nonneg hp.le 4]
  41     · dsimp [B] at hβB
  42       nlinarith
  43     · dsimp [row]
  44       nlinarith [pow_nonneg hp.le (m+7)]
  45 end Oberwolfach
