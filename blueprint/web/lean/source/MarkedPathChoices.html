  40 def markedPathChoiceRequests {V : Type*} [DecidableEq V] (π : Equiv.Perm V) (j : Bool)
  41     (m : ℕ) (v v' : V) (n : ℕ) (f : Fin n→V) : Finset V :=
  42   if n=0 then {v,v'} else
  43   if n=1 then {if j then π v else v,if j then π v' else v'} else
  44   if h : n=2 then {f ⟨0,by omega⟩} else
  45   if h : 2<n then {f ⟨n-1,by omega⟩} ∪ (if n+1=m then {f ⟨1,by omega⟩} else ∅) else ∅
