  13 theorem retained_neighbour_count_low {V : Type*} [Fintype V] [DecidableEq V]
  14     (G : SimpleGraph V) (S : Finset V) (μ : Sym2 V → FiniteLaw Bool) (v : V)
  15     (δ α : ℝ) (hα : 0<α) (hS : 0<S.card)
  16     (hμ : ∀ e,1-α/8≤(μ e).mass true)
  17     (hdeg : (δ+α/2)*S.card≤((G.neighborFinset v∩S).card:ℝ)) :
  18     (FiniteLaw.pi μ).probability (fun σ =>
  19       (retainedNeighbourCount G S σ v:ℝ)≤(δ+α/4)*S.card)≤
  20       2*Real.exp (-α^2*S.card/32) := by
  21   have hm : (0:ℝ)<S.card := Nat.cast_pos.mpr hS
  22   have hc : ((G.neighborFinset v∩S).card:ℝ)≤S.card := by
  23     exact_mod_cast Finset.card_le_card (Finset.inter_subset_right (s₁:=G.neighborFinset v) (s₂:=S))
  24   have hmean : (δ+3*α/8)*S.card≤∑ x∈G.neighborFinset v∩S,(μ s(v,x)).mass true := by
  25     have hh : (1-α/8)*((G.neighborFinset v∩S).card:ℝ)≤
  26         ∑ x∈G.neighborFinset v∩S,(μ s(v,x)).mass true := by
  27       calc
  28         _ = ∑ _x∈G.neighborFinset v∩S,(1-α/8) := by simp;ring
  29         _ ≤ _ := Finset.sum_le_sum (fun x _ => hμ s(v,x))
  30     have hc' := mul_le_mul_of_nonneg_left hc hα.le
  31     nlinarith
  32   apply le_trans (finiteLaw_probability_mono _ _ _ ?_)
  33     (le_trans (retained_neighbour_count_tail G S μ v (α*S.card/8) (by positivity)) ?_)
  34   · intro σ hσ
  35     have hh := le_abs_self ((∑ x∈G.neighborFinset v∩S,(μ s(v,x)).mass true)-
  36       (retainedNeighbourCount G S σ v:ℝ))
  37     rw [abs_sub_comm] at hh
  38     linarith
  39   · have he : -2*(α*S.card/8)^2/S.card = -α^2*S.card/32 := by field_simp;ring
  40     rw [he]

  43     (G : SimpleGraph V) (S : Finset V) (μ : Sym2 V → FiniteLaw Bool)
  44     (δ α : ℝ) (hα : 0<α) (hS : 0<S.card)
  45     (hμ : ∀ e,1-α/8≤(μ e).mass true)
  46     (hdeg : ∀ v∈S,(δ+α/2)*S.card≤((G.neighborFinset v∩S).card:ℝ)) :
  47     (FiniteLaw.pi μ).probability (fun σ => ∃ v∈S,
  48       (retainedNeighbourCount G S σ v:ℝ)≤(δ+α/4)*S.card)≤
  49       2*S.card*Real.exp (-α^2*S.card/32) := by

  51     (fun v σ => (retainedNeighbourCount G S σ v:ℝ)≤(δ+α/4)*S.card)).trans

  53     _ ≤ ∑ _v∈S,2*Real.exp (-α^2*S.card/32) :=
