  11 theorem long_binary_partition_parameters
  12     {U:Type} [Fintype U] (ρ₀ ρ s:ℝ) (D:ℕ) (a b:U→ℝ)
  13     (hD:0<D) (heven:Even D) (hρlo:ρ₀≤ρ) (hρhi:ρ≤1)
  14     (hsize:ρ*D=2*Fintype.card U):
  15     let p:=ρ*(1/100)/2
  16     let k:=ρ*s/2
  17     let p₂:=(∑w,b w)/(D:ℝ)
  18     let k₂:=(∑w,a w)/(D:ℝ)
  19     let u:=2*(∑w,(1-1/100-s-a w-b w))/(D:ℝ)
  20     let m:=binaryClassMultiplicity D (Fintype.card U);
  21     (∑i,m i)=D/2 ∧
  22       (∀i,expandedBinaryProbability p k p₂ k₂ u i/2*D=(m i:ℝ)) ∧
  23       expandedBinaryProbability p k p₂ k₂ u 0=ρ ∧
  24       expandedBinaryProbability p k p₂ k₂ u 1=1-ρ:=by
  25   intro p k p₂ k₂ u m
  26   have hDreal:0<(D:ℝ):=by exact_mod_cast hD
  27   obtain ⟨_,hmass⟩:=long_expanded_class_density_and_floor ρ₀ ρ (D:ℝ) s a b hDreal hρlo hsize
  28   have hsum:2*p+2*k+2*p₂+2*k₂+u=ρ:=by
  29     exact (expanded_physical_class_mass p k p₂ k₂ u).symm.trans hmass
  30   obtain ⟨hs,hm0,hm1⟩:=binary_class_multiplicity_exact D (Fintype.card U) ρ heven hρhi hsize
  31   have h0:expandedBinaryProbability p k p₂ k₂ u 0=ρ:=by
  32     simp [expandedBinaryProbability];exact hsum
  33   have h1:expandedBinaryProbability p k p₂ k₂ u 1=1-ρ:=by
  34     simp [expandedBinaryProbability];linarith
  35   refine ⟨hs,?_,h0,h1⟩
  36   intro i
  37   fin_cases i
  38   · simpa [h0] using hm0
  39   · simpa [h1] using hm1
  40 end Oberwolfach
