  14 def FiniteLaw.conditionEvent {Ω : Type*} [Fintype Ω]
  15     (μ : FiniteLaw Ω) (A : Ω → Prop) : FiniteLaw Ω :=
  16   μ.conditionOnMap (fun x=>decide (A x)) true

  18 theorem finiteLaw_boolean_event_mass {Ω : Type*} [Fintype Ω]
  19     (μ : FiniteLaw Ω) (A : Ω → Prop) :
  20     (μ.map (fun x=>decide (A x))).mass true=μ.probability A := by
  21   unfold FiniteLaw.map FiniteLaw.probability
  22   apply Finset.sum_congr rfl
  23   intro x _
  24   by_cases h : A x <;> simp [h]

  26 theorem finiteLaw_conditionEvent_weight {Ω : Type*} [Fintype Ω]
  27     (μ : FiniteLaw Ω) (A : Ω → Prop) (x : Ω) :
  28     μ.probability A*(μ.conditionEvent A).mass x=if A x then μ.mass x else 0 := by
  29   have h := finiteLaw_conditionOnMap_weight μ (fun x=>decide (A x)) true x
  30   rw [finiteLaw_boolean_event_mass] at h
  31   simpa only [FiniteLaw.conditionEvent,decide_eq_true_eq] using h

  33 theorem finiteLaw_conditionEvent_probability {Ω : Type*} [Fintype Ω]
  34     (μ : FiniteLaw Ω) (A B : Ω → Prop) :
  35     μ.probability A*(μ.conditionEvent A).probability B=μ.probability (fun x=>A x ∧ B x) := by
  36   unfold FiniteLaw.probability
  37   rw [Finset.mul_sum]
  38   apply Finset.sum_congr rfl
  39   intro x _
  40   by_cases hb : B x
  41   · simp only [hb,ite_true,and_true]
  42     exact finiteLaw_conditionEvent_weight μ A x
  43   · simp [hb]

  45 theorem finiteLaw_pi_conditionEvent_probability
  46     {I : Type*} [Fintype I] {Ω : I → Type*} [∀ i,Fintype (Ω i)]
  47     (μ : ∀ i,FiniteLaw (Ω i)) (A : ∀ i,Ω i → Prop) (B : (∀ i,Ω i) → Prop) :
  48     (∏ i,(μ i).probability (A i))*
  49       (FiniteLaw.pi (fun i=>(μ i).conditionEvent (A i))).probability B=
  50         (FiniteLaw.pi μ).probability (fun x=>(∀ i,A i (x i)) ∧ B x) := by
  51   classical
  52   unfold FiniteLaw.probability
  53   rw [Finset.mul_sum]
  54   apply Finset.sum_congr rfl
  55   intro x _
  56   by_cases hb : B x
  57   · simp only [hb,ite_true,and_true,FiniteLaw.pi]
  58     rw [← Finset.prod_mul_distrib]
  59     have hcoord (i : I) := finiteLaw_conditionEvent_weight (μ i) (A i) (x i)
  60     simp only [FiniteLaw.probability] at hcoord
  61     simp_rw [hcoord]
  62     by_cases ha : ∀ i,A i (x i)
  63     · simp [ha]
  64     · rw [if_neg ha]
  65       obtain ⟨i,hi⟩ := not_forall.mp ha
  66       exact Finset.prod_eq_zero (Finset.mem_univ i) (by simp [hi])
  67   · simp [hb]
