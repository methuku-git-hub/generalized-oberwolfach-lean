  20 def attachmentResourceMap (z : W) (ρ : E⊕J → V) :
  21     (Sym2 (ExpansionRow V E J W) ⊕ (ExpansionRow V E J W × ExpansionHub U E J)) →
  22     (Sym2 (ExpansionRow V E J W) ⊕ (ExpansionRow V E J W × ExpansionHub U E J)) :=
  23   Sum.map (Sym2.map (attachmentCollapse z ρ)) (Prod.map (attachmentCollapse z ρ) id)

  25 theorem ordinaryExpansion_attachment_resources_image
  26     (G : SimpleGraph V) (H : V → U → Prop) (C : SimpleGraph W) (z o : W) (hzo : C.Adj z o)
  27     (x y : E → V) (v : J → V) (u : J → U)
  28     (hG : ∀ e,G.Adj (x e) (y e)) (hH : ∀ j,H (v j) (u j))
  29     (hGc : ∀ a b,G.Adj a b → ∃ e,s(x e,y e)=s(a,b))
  30     (hHc : ∀ a b,H a b → ∃ j,v j=a ∧ u j=b) :
  31     (ordinaryHostResources (ordinaryExpansionRim C z o x y) (ordinaryExpansionSpokes z v u)).image
  32       (attachmentResourceMap z (expansionAttachmentRoots x v))=
  33     ordinaryHostResources (ordinaryAttachmentRim G C z (expansionAttachmentRoots x v))
  34       (ordinaryAttachmentSpokes H z (expansionAttachmentRoots x v)) := by
  35   ext e
  36   constructor
  37   · intro himage
  38     obtain ⟨e,he,rfl⟩ := Finset.mem_image.mp himage
  39     cases e with
  40     | inl e =>
  41       rw [ordinaryHostResources_rim] at he
  42       induction e using Sym2.ind with
  43       | _ a b =>
  44         exact (ordinaryHostResources_rim _ _ _).mpr
  45           (ordinaryExpansion_attachment_rim_hom G C z o hzo x y v hG a b he)
  46     | inr p =>
  47       rcases p with ⟨a,b⟩
  48       rw [ordinaryHostResources_spoke] at he
  49       exact (ordinaryHostResources_spoke _ _ _ _).mpr
  50         (ordinaryExpansion_attachment_spoke_hom H z x v u hH a b he)
  51   · intro he
  52     cases e with
  53     | inl e =>
  54       rw [ordinaryHostResources_rim] at he
  55       induction e using Sym2.ind with
  56       | _ a b =>
  57         obtain ⟨a',b',hab,ha,hb⟩ := ordinaryExpansion_attachment_rim_preimage G C z o x y v hGc a b he
  58         refine Finset.mem_image.mpr ⟨.inl s(a',b'),(ordinaryHostResources_rim _ _ _).mpr hab,?_⟩
  59         change Sum.inl s(attachmentCollapse z (expansionAttachmentRoots x v) a',
  60           attachmentCollapse z (expansionAttachmentRoots x v) b')=Sum.inl s(a,b)
  61         rw [ha,hb]
  62     | inr p =>
  63       rcases p with ⟨a,b⟩
  64       rw [ordinaryHostResources_spoke] at he
  65       obtain ⟨a',hab,ha⟩ := ordinaryExpansion_attachment_spoke_preimage H z x v u hHc a b he
  66       refine Finset.mem_image.mpr ⟨.inr (a',b),(ordinaryHostResources_spoke _ _ _ _).mpr hab,?_⟩
  67       change Sum.inr (attachmentCollapse z (expansionAttachmentRoots x v) a',b)=Sum.inr (a,b)
  68       rw [ha]

  70 theorem ordinaryExpansion_attachment_resources_injective
  71     (C : SimpleGraph W) (z o : W) (hzo : z≠o) (x y : E → V) (v : J → V) (u : J → U)
  72     (hxy : Function.Injective (fun e => s(x e,y e))) (hvu : Function.Injective (fun j => (v j,u j))) :
  73     Set.InjOn (attachmentResourceMap (U:=U) z (expansionAttachmentRoots x v))
  74       (ordinaryHostResources (ordinaryExpansionRim C z o x y) (ordinaryExpansionSpokes z v u)) := by
  75   intro e he e' he' hh
  76   cases e with
  77   | inl e =>
  78     cases e' with
  79     | inr p => cases hh
  80     | inl e' =>
  81       rw [Finset.mem_coe,ordinaryHostResources_rim] at he he'
  82       exact congrArg Sum.inl (ordinaryExpansion_attachment_rim_injective C z o hzo x y v hxy he he' (Sum.inl.inj hh))
  83   | inr p =>
  84     cases e' with
  85     | inl e' => cases hh
  86     | inr q =>
  87       rcases p with ⟨a,b⟩;rcases q with ⟨c,d⟩
  88       rw [Finset.mem_coe,ordinaryHostResources_spoke] at he he'
  89       exact congrArg Sum.inr (ordinaryExpansion_attachment_spoke_injective z x v u hvu he he' (Sum.inr.inj hh))

  91 theorem actual_expansion_attachment_bijection

  94     let J := {e : V×U // H e.1 e.2}

  98     let u := fun j : J => j.val.2

 100     let T := ordinaryHostResources (ordinaryAttachmentRim G C z (expansionAttachmentRoots x v))
 101       (ordinaryAttachmentSpokes H z (expansionAttachmentRoots x v))
 102     let f := attachmentResourceMap (U:=U) z (expansionAttachmentRoots x v)
 103     Set.InjOn f S ∧ S.image f=T := by
 104   dsimp only
 105   have hout (e : G.edgeFinset) : s(e.val.out.1,e.val.out.2)=e.val := by
 106     rw [Sym2.mk];exact e.val.out_eq
 107   have hxy : Function.Injective (fun e : G.edgeFinset => s(e.val.out.1,e.val.out.2)) := by
 108     intro e e' he
 109     apply Subtype.ext
 110     simpa only [hout] using he
 111   have hvu : Function.Injective (fun j : {e : V×U // H e.1 e.2} => (j.val.1,j.val.2)) := by
 112     intro j j' h
 113     exact Subtype.ext h
 114   have hGc : ∀ a b,G.Adj a b → ∃ e : G.edgeFinset,s(e.val.out.1,e.val.out.2)=s(a,b) := by
 115     intro a b hab
