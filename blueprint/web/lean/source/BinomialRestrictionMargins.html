  11 theorem uniform_binomial_restriction_tail (K a b : ℝ) (hK : 0≤K) (ha : 0<a) (hb : 0<b) (L : ℕ) :
  12     ∃ m N : ℕ,1≤m ∧ 1≤N ∧ ∀ n : ℕ,N≤n → ∀ M : ℕ,m≤M →
  13     ∀ p : ℝ,0≤p → p≤K*(n:ℝ)^(-b) →
  14       (M.choose (Nat.ceil (a*M)):ℝ)*p^(Nat.ceil (a*M))≤1/(n:ℝ)^L := by
  15   obtain ⟨N,hN1,hN⟩ := eventually_binomial_deletion_tail K a b hK ha hb
  16   let c := a*b/2
  17   have hc : 0<c := by dsimp [c];positivity
  18   refine ⟨max 1 (Nat.ceil ((L:ℝ)/c)),N,le_max_left _ _,hN1,?_⟩
  19   intro n hn M hM p hp hrate
  20   have hnR : (1:ℝ)≤n := by exact_mod_cast hN1.trans hn
  21   have hLM : (L:ℝ)≤c*M := by
  22     have hceil : (L:ℝ)/c≤(M:ℝ) := (Nat.le_ceil _).trans (by exact_mod_cast (show Nat.ceil ((L:ℝ)/c)≤M by omega))
  23     have hh := (div_le_iff₀ hc).mp hceil
  24     linarith
  25   have hh := hN n hn M (Nat.ceil (a*M)) p hp hrate (Nat.le_ceil _)
