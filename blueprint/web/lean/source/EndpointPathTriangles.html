  13 theorem endpoint_path_triangle (Q F : SimpleGraph V) (S T : Finset V) (hST : Disjoint S T)
  14     (j : Bool) (x y z : V) (hz : z∈({x,y}:Finset V)) (hzclass : z∈if j then T else S)
  15     (f : Fin 2→V)
  16     (hf : ClassSparseGreedyGood Q F (endpointPathClass S T j) {x,y} (endpointPathRequirements x y) f) :
  17     ∃ C : EndpointTriangle Q F S T,endpointTriangleVertices C=insert z (Finset.univ.image f) := by
  18   obtain ⟨he,hfedge⟩ := endpoint_path_edges Q F S T j x y f hf
  19   have himage : Finset.univ.image f={f 0,f 1} := by
  20     ext v
  21     simp only [Finset.mem_image,Finset.mem_univ,true_and,Finset.mem_insert,Finset.mem_singleton]
  22     constructor
  23     · rintro ⟨i,rfl⟩;fin_cases i <;> simp
  24     · rintro (rfl|rfl)
  25       · exact ⟨0,rfl⟩
  26       · exact ⟨1,rfl⟩
  27   have h0z : f 0≠z := fun h=>(hf.2.1 0).2 (h.symm ▸ hz)
  28   have h1z : f 1≠z := fun h=>(hf.2.1 1).2 (h.symm ▸ hz)
  29   have h01 : f 0≠f 1 := fun h=>by have hh:=hf.1 h;norm_num at hh
  30   have hF0 := hf.2.2.2.1 0 z hz
  31   have hF1 := hf.2.2.2.1 1 z hz
  32   have hF01 := hf.2.2.1 0 1
  33   have hc0 := (hf.2.1 0).1
  34   have hc1 := (hf.2.1 1).1
  35   cases j
  36   · obtain ⟨C,hC⟩ := endpoint_triangle_from_vertices Q F S T hST (f 0) z (f 1)
  37       (by simpa [endpointPathClass] using hc0) hzclass
  38       (by simpa [endpointPathClass] using hc1) h1z.symm (he z hz 0).symm hfedge (he z hz 1)
  39       hF0 hF01 (fun h=>hF1 h.symm)
  40     refine ⟨C,hC.trans ?_⟩
  41     rw [himage,Finset.insert_comm]
  42   · obtain ⟨C,hC⟩ := endpoint_triangle_from_vertices Q F S T hST z (f 0) (f 1) hzclass
  43       (by simpa [endpointPathClass] using hc0) (by simpa [endpointPathClass] using hc1) h01
  44       (he z hz 0) (he z hz 1) hfedge (fun h=>hF0 h.symm) (fun h=>hF1 h.symm) hF01
  45     exact ⟨C,hC.trans (by rw [himage])⟩
