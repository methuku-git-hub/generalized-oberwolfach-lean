  17 abbrev RootedLabelMap (V : Type*) [DecidableEq V] (h c : ℕ) (X : Finset V) :=
  18   {f : Fin h ⊕ Fin c → V // Function.Injective f ∧ Finset.univ.image (fun i => f (Sum.inr i))=X}

  20 def freeLabelPositions (S : Finset (Fin h ⊕ Fin c)) : Finset (Fin h) :=
  21   Finset.univ.filter (fun i => Sum.inl i∈S)

  23 theorem rootedLabelMap_root_mem (X : Finset V) (f : RootedLabelMap V h c X) (i : Fin c) :
  24     f.val (Sum.inr i)∈X := by
  25   have hh : f.val (Sum.inr i)∈Finset.univ.image (fun i => f.val (Sum.inr i)) :=
  26     Finset.mem_image.mpr ⟨i,Finset.mem_univ _,rfl⟩
  27   simpa only [f.property.2] using hh

  29 theorem rooted_label_maps_constrained_count (X Y : Finset V) (S : Finset (Fin h ⊕ Fin c)) :
  30     Fintype.card {f : RootedLabelMap V h c X // ∀ j∈S,f.val j∈Y} ≤
  31       X.card^c*Y.card^(freeLabelPositions S).card*
  32         Fintype.card V^(h-(freeLabelPositions S).card) := by
  33   let A := {f : RootedLabelMap V h c X // ∀ j∈S,f.val j∈Y}
  34   let P := freeLabelPositions S
  35   let enc (f : A) : (Fin c → X) × (P → Y) × ({j : Fin h // j∉P} → V) :=
  36     (fun i => ⟨f.val.val (Sum.inr i),rootedLabelMap_root_mem X f.val i⟩,
  37      fun j => ⟨f.val.val (Sum.inl j.val),f.property _ (Finset.mem_filter.mp j.property).2⟩,
  38      fun j => f.val.val (Sum.inl j.val))
  39   have hinj : Function.Injective enc := by
  40     intro f g he
  41     apply Subtype.ext
  42     apply Subtype.ext
  43     funext j
  44     rcases j with j | j
  45     · by_cases hj : j∈P
  46       · exact congrArg (fun z : Y => z.val) (congrFun (congrArg (fun z => z.2.1) he) ⟨j,hj⟩)
  47       · exact congrFun (congrArg (fun z => z.2.2) he) ⟨j,hj⟩
  48     · exact congrArg (fun z : X => z.val) (congrFun (congrArg Prod.fst he) j)
  49   have hh := Fintype.card_le_of_injective enc hinj
  50   simpa only [Fintype.card_prod,Fintype.card_fun,Fintype.card_fin,Fintype.card_coe,
  51     Fintype.card_subtype_compl,mul_assoc] using hh
