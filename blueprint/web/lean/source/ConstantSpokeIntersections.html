  14 def commonSpokeNeighbours (H : A → B → Prop) (S : Finset B) : Finset A :=
  15   Finset.univ.filter (fun x=>∀ u∈S,H x u)

  60 theorem sparseIntersectionData_of_constant_density
  61     (H : A → B → Prop) (p p₀ δ D : ℝ) (hp₀ : 0<p₀) (hp : p₀≤p) (hp1 : p≤1)
  62     (hδ : 0≤δ) (hD : 0≤D)
  63     (hinter : ∀ S : Finset B,1≤S.card → S.card≤3 →
  64       |((commonSpokeNeighbours H S).card:ℝ)-p^S.card*Fintype.card A|≤δ*Fintype.card A)
  65     (hcodeg : ∀ x,∃ E : Finset A,(E.card:ℝ)≤D ∧ ∀ y,y∉E →
  66       |((Finset.univ.filter (fun u=>H x u ∧ H y u)).card:ℝ)-p^2*Fintype.card B|≤
  67         δ*p^2*Fintype.card B) :
  68     SparseIntersectionData H (fun _=>p) (fun _=>1/p) (max 1 (1/p₀)) δ D := by
  69   classical
  70   have hp0 : 0<p := lt_of_lt_of_le hp₀ hp
  71   have hp2 : 0<p^2 := sq_pos_of_pos hp0
  72   refine ⟨le_max_left _ _,hδ,hD,fun _=>⟨hp0.le,hp1⟩,?_,?_,?_,?_,?_,?_⟩
  73   · intro u
  74     exact ⟨by positivity,le_trans (one_div_le_one_div_of_le hp₀ hp) (le_max_right _ _)⟩
  75   · intro u
  76     exact one_div_mul_cancel (ne_of_gt hp0)
  77   · intro u
  78     have hh := hinter {u} (by simp) (by simp)
  79     simpa only [Finset.card_singleton,pow_one,commonSpokeNeighbours_singleton,mul_comm p] using hh
  80   · intro u v huv
  81     have hh := hinter {v,u} (by simp [Ne.symm huv]) (by simp [Ne.symm huv])
  82     have hc : ({v,u}:Finset B).card=2 := by simp [Ne.symm huv]
  83     rw [hc,commonSpokeNeighbours_pair] at hh
  84     convert hh using 1 <;> ring
  85   · intro u v t huv hut hvt
  86     have hc : ({u,v,t}:Finset B).card=3 := by simp [huv,hut,hvt]
  87     have hh := hinter {u,v,t} (by rw [hc];omega) (by rw [hc])
  88     rw [hc,commonSpokeNeighbours_triple] at hh
