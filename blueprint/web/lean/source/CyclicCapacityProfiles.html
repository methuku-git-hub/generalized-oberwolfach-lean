  10 def cyclicIndex (c : ℕ) [NeZero c] (j : ℕ) : Fin c := ⟨j%c,Nat.mod_lt _ (NeZero.pos c)⟩

  12 @[simp] theorem cyclicIndex_fin {c : ℕ} [NeZero c] (i : Fin c) : cyclicIndex c i.val=i := by
  13   apply Fin.ext
  14   exact Nat.mod_eq_of_lt i.isLt

  20 theorem cyclicIndex_succ (c : ℕ) [NeZero c] (j : ℕ) :

  37 theorem balanced_capacity_mass_constant {X : Type*} [Fintype X]
  38     (a : ℕ → X → X → ℝ) (d : ℕ → X → ℝ)
  39     (hrow : ∀ i x,∑ y,a i x y=d i x)
  40     (hcol : ∀ i y,∑ x,a i x y=d (i+1) y) :
  41     ∀ i,(∑ x,d i x)=∑ x,d 0 x := by
  42   intro i
  43   induction i with
  44   | zero => rfl
  45   | succ i ih =>
  46     calc
  47       (∑ y,d (i+1) y) = ∑ y,∑ x,a i x y := by simp only [hcol]
  48       _ = ∑ x,∑ y,a i x y := Finset.sum_comm
  49       _ = ∑ x,d i x := by simp only [hrow]
  50       _ = ∑ x,d 0 x := ih

  52 theorem periodicCapacity_balance {c : ℕ} [NeZero c] {X : Type*} [Fintype X]
  53     (a : Fin c → X → X → ℝ) (d : Fin c → X → ℝ)
  54     (hrow : ∀ i x,∑ y,a i x y=d i x)
  55     (hcol : ∀ i y,∑ x,a i x y=d (finRotate c i) y) :
  56     (∀ j x,∑ y,periodicCapacity a j x y=periodicDegree d j x) ∧
  57       (∀ j y,∑ x,periodicCapacity a j x y=periodicDegree d (j+1) y) := by
  58   constructor
  59   · intro j x
  60     exact hrow _ _
  61   · intro j y
  62     simp only [periodicCapacity,periodicDegree,cyclicIndex_succ]
  63     exact hcol _ _

  65 theorem cyclic_capacity_mass_constant {c : ℕ} [NeZero c] {X : Type*} [Fintype X]
  66     (a : Fin c → X → X → ℝ) (d : Fin c → X → ℝ)
  67     (hrow : ∀ i x,∑ y,a i x y=d i x)
  68     (hcol : ∀ i y,∑ x,a i x y=d (finRotate c i) y) (i : Fin c) :
  69     (∑ x,d i x)=∑ x,d 0 x := by
  70   obtain ⟨hr,hc⟩ := periodicCapacity_balance a d hrow hcol
  71   simpa only [periodicDegree,cyclicIndex_fin,cyclicIndex_zero] using
  72     balanced_capacity_mass_constant (periodicCapacity a) (periodicDegree d) hr hc i.val

  77 theorem transitionWalk_three {X : Type*} [Fintype X] (R : ℕ → X → X → ℝ) (j : ℕ) (x z : X) :
  78     transitionWalk R j 3 x z=∑ y,∑ w,R j x y*R (j+1) y w*R (j+2) w z := by
  79   change transitionCompose (R j) (transitionCompose (R (j+1))
  80     (transitionCompose (R (j+2)) identityTransition)) x z=_
  81   rw [transitionCompose_identity_right]
  82   simp [transitionCompose,Finset.mul_sum,mul_assoc]

  84 theorem periodicCapacity_three {c : ℕ} [NeZero c] {X : Type*} [Fintype X]
  85     (a : Fin c → X → X → ℝ) (d : Fin c → X → ℝ) (j : ℕ) (x z : X) :
  86     transitionWalk (capacityTransition (periodicCapacity a) (periodicDegree d)) j 3 x z=
  87       ∑ y,∑ w,cyclicCapacityTransition a d (cyclicIndex c j) x y*
  88         cyclicCapacityTransition a d (finRotate c (cyclicIndex c j)) y w*
  89         cyclicCapacityTransition a d (finRotate c (finRotate c (cyclicIndex c j))) w z := by
  90   rw [transitionWalk_three]
  91   simp only [capacityTransition,periodicCapacity,periodicDegree,cyclicCapacityTransition,
  92     show j+2=j+1+1 from rfl,cyclicIndex_succ]

  94 theorem cyclicIndex_mem_image_iff {c : ℕ} [NeZero c] (J : Finset (Fin c)) (j : ℕ) (hj : j<c) :
  95     cyclicIndex c j∈J ↔ j∈J.image Fin.val := by
