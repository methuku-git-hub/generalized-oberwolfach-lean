  13 def componentVertexEquiv {V : Type*} (G : SimpleGraph V) :
  14     (Σ C : G.ConnectedComponent, C) ≃ V where
  15   toFun z := z.2.val
  16   invFun v := ⟨G.connectedComponentMk v, ⟨v, rfl⟩⟩
  17   left_inv := by
  18     rintro ⟨C, ⟨v, hv⟩⟩
  19     change G.connectedComponentMk v = C at hv
  20     subst C
  21     rfl
  22   right_inv := by intro v; rfl

  24 theorem component_orders_sum {V : Type*} [Fintype V] [DecidableEq V] (G : SimpleGraph V) :
  25     ∑ C : G.ConnectedComponent, Fintype.card C = Fintype.card V := by
  26   rw [← Fintype.card_sigma]
  27   exact Fintype.card_congr (componentVertexEquiv G)

  29 theorem component_isCycleCopy {n : ℕ} (F : SimpleGraph (Fin n))
  30     (hF : F.IsRegularOfDegree 2) (C : F.ConnectedComponent) :
  31     IsCycleCopy C.toSimpleGraph.spanningCoe (Fintype.card C) := by
  32   obtain ⟨hthree, ⟨e⟩⟩ := component_regular_two_cycle_iso F hF C
  33   let f : Fin (Fintype.card C) ↪ Fin n := {
  34     toFun := fun i => (e i).val
  35     inj' := by intro i j hij; apply e.injective; exact Subtype.ext hij }
  36   apply (isCycleCopy_iff_map _).mpr
  37   refine ⟨hthree, f, ?_⟩
  38   ext u v
  39   rw [SimpleGraph.ConnectedComponent.adj_spanningCoe_toSimpleGraph, SimpleGraph.map_adj]
  40   constructor
  41   · rintro ⟨hu, huv⟩
  42     have hv : v ∈ C.supp := (C.mem_supp_congr_adj huv).mp hu
  43     refine ⟨e.symm ⟨u, hu⟩, e.symm ⟨v, hv⟩, e.symm.map_rel_iff.mpr huv, ?_, ?_⟩ <;>
  44       simp only [f, Function.Embedding.coeFn_mk, RelIso.apply_symm_apply]
  45   · rintro ⟨i, j, hij, rfl, rfl⟩
  46     exact ⟨(e i).property, e.map_rel_iff.mpr hij⟩

  48 theorem reachable_order_eq_component_card {V : Type*} [Fintype V] (G : SimpleGraph V)
  49     (C : G.ConnectedComponent) (v : V) (hv : v ∈ C.supp) :
  50     Nat.card {x : V // G.Reachable v x} = Fintype.card C := by
  51   have heq : ({x | G.Reachable v x} : Set V) = C.supp := by
  52     ext x
  53     change G.Reachable v x ↔ G.connectedComponentMk x = C
  54     change G.connectedComponentMk v = C at hv
  55     rw [← SimpleGraph.ConnectedComponent.eq, hv]
  56     exact eq_comm
  57   change ({x | G.Reachable v x} : Set V).ncard = _
  58   rw [heq]
  59   rw [← Nat.card_eq_fintype_card]
  60   exact Nat.card_congr (Equiv.refl _)

  62 theorem AllowedCycleLengths_iff_component_orders {n : ℕ}
  63     (F : SimpleGraph (Fin n)) (S : Set ℕ) :
  64     AllowedCycleLengths F S ↔ ∀ C : F.ConnectedComponent, Fintype.card C ∈ S := by
  65   constructor
  66   · intro h C
  67     obtain ⟨v, hv⟩ := C.nonempty_supp
  68     rw [← reachable_order_eq_component_card F C v hv]
  69     exact h v
  70   · intro h v
  71     rw [reachable_order_eq_component_card F (F.connectedComponentMk v) v rfl]
  72     exact h _

  74 theorem two_regular_cycle_partition {n : ℕ} (F : SimpleGraph (Fin n))
  75     (hF : F.IsRegularOfDegree 2) :
  76     ∃ q : ℕ, ∃ H : Fin q → SimpleGraph (Fin n), ∃ r : Fin q → ℕ,
  77       EdgePartition F H ∧ (∀ i, IsCycleCopy (H i) (r i)) ∧
  78       (∑ i, r i) = n ∧
  79       (∀ S : Set ℕ, AllowedCycleLengths F S ↔ ∀ i, r i ∈ S) := by
  80   let q := Fintype.card F.ConnectedComponent
  81   let e : Fin q ≃ F.ConnectedComponent := (Fintype.equivFin F.ConnectedComponent).symm
  82   let H (i : Fin q) := (e i).toSimpleGraph.spanningCoe
  83   let r (i : Fin q) := Fintype.card (e i)
  84   refine ⟨q, H, r, ?_, ?_, ?_, ?_⟩
  85   · intro u v
  86     have hi (i : Fin q) : (H i).Adj u v ↔ u ∈ (e i).supp ∧ F.Adj u v :=
  87       SimpleGraph.ConnectedComponent.adj_spanningCoe_toSimpleGraph (e i)
  88     constructor
  89     · constructor
  90       · intro huv
  91         refine ⟨e.symm (F.connectedComponentMk u), (hi _).mpr ⟨?_, huv⟩⟩
  92         simp only [Equiv.apply_symm_apply]
  93         rfl
  94       · rintro ⟨i, huv⟩
  95         exact ((hi i).mp huv).2
  96     · intro i j hui huj
  97       apply e.injective
  98       have hi' := ((hi i).mp hui).1
  99       have hj' := ((hi j).mp huj).1
 100       exact SimpleGraph.ConnectedComponent.eq_of_common_vertex hi' hj'
 101   · intro i
 102     exact component_isCycleCopy F hF (e i)
 103   · change (∑ i, Fintype.card (e i)) = n
 104     calc
 105       (∑ i, Fintype.card (e i)) = ∑ C : F.ConnectedComponent, Fintype.card C :=
 106         Fintype.sum_equiv e _ _ (fun _ => rfl)
 107       _ = n := by simpa only [Fintype.card_fin] using component_orders_sum F
 108   · intro S
 109     rw [AllowedCycleLengths_iff_component_orders]
 110     exact ⟨fun h i => h (e i), fun h C => by
 111       obtain ⟨i, rfl⟩ := e.surjective C
 112       exact h i⟩
