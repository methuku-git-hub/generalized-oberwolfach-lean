  10 theorem root_packing_remaining_divisible {V:Type} [Fintype V] [DecidableEq V]
  11     (G:SimpleGraph V) (r:ℕ) (S T:Finset V) (M:Finset (OrdinaryCycle G r))
  12     (hpart:S∪T=Finset.univ) (hdis:Disjoint S T)
  13     (hS:r∣S.card) (hT:r∣T.card)
  14     (hpack:∀ C∈M,∀ E∈M,C≠E→Disjoint (ordinaryCycleVertices G r C) (ordinaryCycleVertices G r E))
  15     (hcover:T⊆M.biUnion (ordinaryCycleVertices G r)):
  16     r∣(S\M.biUnion (ordinaryCycleVertices G r)).card:=by
  17   let U:=M.biUnion (ordinaryCycleVertices G r)
  18   have hU:U.card=M.card*r:=by
  19     dsimp [U]
  20     rw [Finset.card_biUnion (by intro C hC E hE hne;exact hpack C hC E hE hne)]
  21     simp [ordinaryCycleVertices_card]
  22   have hTU:T⊆U:=hcover
  23   have hUdiv:r∣U.card:=by rw [hU];exact dvd_mul_left _ _
  24   have hUT:U\T=U∩S:=by
  25     ext v
  26     have hv:v∈S∨v∈T:=by
  27       have hh: v∈S∪T:=by rw [hpart];exact Finset.mem_univ _
  28       exact Finset.mem_union.mp hh
  29     simp only [Finset.mem_sdiff,Finset.mem_inter]
  30     constructor
  31     · rintro ⟨hu,hnt⟩
  32       rcases hv with hs|ht
  33       · exact ⟨hu,hs⟩
  34       · exact (hnt ht).elim
  35     · rintro ⟨hu,hs⟩
  36       exact ⟨hu,fun ht=>(Finset.disjoint_left.mp hdis) hs ht⟩
  37   have hinside:r∣(S∩U).card:=by
  38     have hh:r∣U.card-T.card:=Nat.dvd_sub hUdiv hT
  39     rw [←Finset.card_sdiff_of_subset hTU,hUT,Finset.inter_comm] at hh
  40     exact hh
  41   have he:(S\U).card=S.card-(S∩U).card:=by simpa only [Finset.inter_comm] using (Finset.card_sdiff (s:=U) (t:=S))
  42   rw [he]
  43   exact Nat.dvd_sub hS hinside
  44 end Oberwolfach
