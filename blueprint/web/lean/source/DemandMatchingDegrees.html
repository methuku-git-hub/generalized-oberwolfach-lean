  12 theorem demandMatching_row_count {X Y E : Type*}
  13     [Fintype E] (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ)
  14     (σ : (DemandSlots f ⊕ E) ≃ (DemandSlots g ⊕ E))
  15     (hσ : ∀ a, DemandMatchingAllowed tail head f g a (σ a)) (x : X) :
  16     (∑ e, if tail e = x ∧ σ (Sum.inr e) ≠ Sum.inr e then (1 : ℕ) else 0) = f x := by
  17   classical
  18   have hslot (a : DemandSlots f) : ∃ e, tail e = a.1 ∧ σ (Sum.inl a) = Sum.inr e := by
  19     have ha := hσ (Sum.inl a)
  20     cases he : σ (Sum.inl a) with
  21     | inl b => rw [he] at ha; exact False.elim ha
  22     | inr e =>
  23       rw [he] at ha
  24       exact ⟨e, ha.symm, rfl⟩
  25   choose F hF hmap using hslot
  26   have hused (i : Fin (f x)) : σ (Sum.inr (F ⟨x, i⟩)) ≠ Sum.inr (F ⟨x, i⟩) := by
  27     intro he
  28     have hh := σ.injective ((hmap ⟨x, i⟩).trans he.symm)
  29     cases hh
  30   let t : Fin (f x) → {e : E // tail e = x ∧ σ (Sum.inr e) ≠ Sum.inr e} :=
  31     fun i => ⟨F ⟨x, i⟩, hF ⟨x, i⟩, hused i⟩
  32   have ht : Function.Bijective t := by
  33     constructor
  34     · intro i j hij
  35       have hv : F ⟨x, i⟩ = F ⟨x, j⟩ := congrArg Subtype.val hij
  36       have hh := σ.injective ((hmap ⟨x, i⟩).trans ((congrArg Sum.inr hv).trans (hmap ⟨x, j⟩).symm))
  37       simpa only [Sum.inl.injEq, Sigma.mk.inj_iff, heq_eq_eq, true_and] using hh
  38     · intro e
  39       cases hp : σ.symm (Sum.inr e.val) with
  40       | inl a =>
  41         have he : σ (Sum.inl a) = Sum.inr e.val := by
  42           rw [← hp, Equiv.apply_symm_apply]
  43         have ha := hσ (Sum.inl a)
  44         rw [he] at ha
  45         have hax : a.1 = x := ha.trans e.property.1
  46         rcases a with ⟨a, i⟩
  47         dsimp only at hax
  48         subst a
  49         refine ⟨i, Subtype.ext ?_⟩
  50         exact Sum.inr.inj ((hmap ⟨x, i⟩).symm.trans he)
  51       | inr a =>
  52         have he : σ (Sum.inr a) = Sum.inr e.val := by
  53           rw [← hp, Equiv.apply_symm_apply]
  54         have ha := hσ (Sum.inr a)
  55         rw [he] at ha
  56         change a = e.val at ha
  57         exact False.elim (e.property.2 (ha ▸ he))
  58   have hc := Fintype.card_congr (Equiv.ofBijective t ht)
  59   simp only [Fintype.card_fin] at hc
  60   rw [hc]
  61   simp [Fintype.card_subtype]

  69 theorem demandMatching_column_count {X Y E : Type*}
  70     [Fintype E] (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ)
  71     (σ : (DemandSlots f ⊕ E) ≃ (DemandSlots g ⊕ E))
  72     (hσ : ∀ a, DemandMatchingAllowed tail head f g a (σ a)) (y : Y) :
  73     (∑ e, if head e = y ∧ σ (Sum.inr e) ≠ Sum.inr e then (1 : ℕ) else 0) = g y := by
  74   have hsym (b) : DemandMatchingAllowed head tail g f b (σ.symm b) := by
  75     have hh := hσ (σ.symm b)
  76     rw [Equiv.apply_symm_apply] at hh
  77     exact (demandMatchingAllowed_symm tail head f g _ _).mp hh
  78   have hh := demandMatching_row_count head tail g f σ.symm hsym y
  79   convert hh using 1
  80   apply Finset.sum_congr rfl
  81   intro e _
  82   have he : σ (Sum.inr e) = Sum.inr e ↔ σ.symm (Sum.inr e) = Sum.inr e := by
  83     rw [Equiv.apply_eq_iff_eq_symm_apply]
  84     exact eq_comm
  85   by_cases hl : head e = y ∧ σ (Sum.inr e) ≠ Sum.inr e
  86   · have hr : head e = y ∧ σ.symm (Sum.inr e) ≠ Sum.inr e :=
  87       ⟨hl.1, (not_congr he).mp hl.2⟩
  88     rw [if_pos hl, if_pos hr]
  89   · have hr : ¬ (head e = y ∧ σ.symm (Sum.inr e) ≠ Sum.inr e) :=
  90       fun hr => hl ⟨hr.1, (not_congr he).mpr hr.2⟩
  91     rw [if_neg hl, if_neg hr]
