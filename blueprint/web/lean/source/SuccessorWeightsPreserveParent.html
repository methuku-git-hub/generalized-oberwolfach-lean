   9 theorem successor_weight_family_preserve_parent
  10     (W:ℕ→Type) [∀ i,Fintype (W i)] (m b:ℕ)
  11     (G:∀ i,SimpleGraph (W i)) (π:∀ i,Equiv.Perm (W i)) (D:ℕ→ℝ)
  12     (τ A B:ℝ) (L:ℕ)
  13     (weight:∀ i,(Fin (m+1)→W i)→ℝ)
  14     (hw:∀ i≤L,SuccessorSequenceWeightConditions (G i) (π i) b (D i) τ A B (weight i))
  15     (parent:(Fin (m+1)→W 0)→ℝ)
  16     (hparent:SuccessorSequenceWeightConditions (G 0) (π 0) b (D 0) τ A B parent):
  17     ∃ updated:∀ i,(Fin (m+1)→W i)→ℝ,
  18       updated 0=parent ∧ (∀ i,0 < i → updated i=weight i) ∧
  19       ∀ i≤L,SuccessorSequenceWeightConditions (G i) (π i) b (D i) τ A B (updated i) := by
  20   let updated:=Function.update weight 0 parent
  21   refine ⟨updated,?_,?_,?_⟩
  22   · simp [updated]
  23   · intro i hi
  24     simp [updated,ne_of_gt hi]
  25   · intro i hi
  26     by_cases h:i=0
  27     · subst i
  28       simpa [updated] using hparent
  29     · simpa [updated,h] using hw i hi
  30 end Oberwolfach
