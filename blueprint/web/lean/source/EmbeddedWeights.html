  12 def embeddedWeight {A B : Type*} [Fintype A]
  13     (f : A ↪ B) (w : A → ℝ) (b : B) : ℝ := ∑ a, if f a = b then w a else 0

  15 theorem embeddedWeight_apply {A B : Type*} [Fintype A]
  16     (f : A ↪ B) (w : A → ℝ) (a : A) : embeddedWeight f w (f a) = w a := by
  17   unfold embeddedWeight
  18   simp only [f.injective.eq_iff, Finset.sum_ite_eq', Finset.mem_univ, ite_true]
  19 
  20 theorem embeddedWeight_not_in_range {A B : Type*} [Fintype A]
  21     (f : A ↪ B) (w : A → ℝ) (b : B) (hb : b ∉ Set.range f) : embeddedWeight f w b = 0 := by
  22   unfold embeddedWeight
  23   apply Finset.sum_eq_zero
  24   intro a _
  25   exact if_neg (fun h => hb ⟨a,h⟩)
  26 
  27 theorem embeddedWeight_nonneg {A B : Type*} [Fintype A]
  28     (f : A ↪ B) (w : A → ℝ) (hw : ∀ a, 0 ≤ w a) (b : B) : 0 ≤ embeddedWeight f w b := by
  29   apply Finset.sum_nonneg
  30   intro a _
  31   split_ifs
  32   · exact hw a
  33   · exact le_refl _

  35 theorem embeddedWeight_bound {A B : Type*} [Fintype A]
  36     (f : A ↪ B) (w : A → ℝ) (C : ℝ) (hC : 0 ≤ C) (hw : ∀ a, w a ≤ C) (b : B) :
  37     embeddedWeight f w b ≤ C := by
  38   by_cases hb : b ∈ Set.range f
  39   · obtain ⟨a,rfl⟩ := hb
  40     rw [embeddedWeight_apply]
  41     exact hw a
  42   · rw [embeddedWeight_not_in_range f w b hb]
  43     exact hC

  45 theorem embeddedWeight_incidence {A B : Type*} [Fintype A] [Fintype B]
  46     (f : A ↪ B) (w : A → ℝ) (E : B → Prop) :
  47     (∑ b, if E b then embeddedWeight f w b else 0) = ∑ a, if E (f a) then w a else 0 := by
  48   have h := finite_marginal_test w f (embeddedWeight f w) (fun _ => rfl)
  49     (fun b => if E b then (1 : ℝ) else 0)
  50   simpa only [mul_ite, mul_one, mul_zero] using h.symm
