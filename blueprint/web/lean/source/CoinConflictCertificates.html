  18 theorem coinConflict_certificate
  19     (G : SimpleGraph V) (T : Finset V)
  20     (hclique : ∀ u∈T,∀ v∈T,u≠v → G.Adj u v) :
  21     HasCoordinateCertificateBound (fun σ : V → C×Bool=>(coinConflictSet G T σ).card) 2
  22       (3*(T.card : ℝ)) := by
  23   classical
  24   intro σ
  25   let B := coinConflictSet G T σ
  26   let f := cliqueConflictWitness G T (fun v=>(σ v).1)
  27   let W := B.image f
  28   let w := cliqueWitnessWeight T W
  29   have hBT : B⊆T := Finset.filter_subset _ _
  30   refine ⟨w,cliqueWitnessWeight_bounds T W,?_,?_⟩
  31   · have hh := cliqueWitnessWeight_changed_sum T W Finset.univ
  32     simp only [Finset.mem_univ,ite_true,Finset.inter_univ] at hh
  33     have hW : W.card≤T.card := Finset.card_image_le.trans (Finset.card_le_card hBT)
  34     have hd : (W\T).card≤W.card := Finset.card_le_card Finset.sdiff_subset
  35     change (∑ v,cliqueWitnessWeight T W v)≤_
  36     rw [hh]
  37     have hWr : (W.card : ℝ)≤T.card := by exact_mod_cast hW
  38     have hdr : ((W\T).card : ℝ)≤W.card := by exact_mod_cast hd
  39     linarith
  40   · intro τ
  41     let B' := coinConflictSet G T τ
  42     let E := Finset.univ.filter (fun v=>σ v≠τ v)
  43     let L := (B\B').filter (fun v=>σ v=τ v)
  44     have hL (v : V) (hv : v∈L) : v∈B ∧ v∉B' ∧ σ v=τ v := by
  45       obtain ⟨hs,he⟩ := Finset.mem_filter.mp hv
  46       exact ⟨(Finset.mem_sdiff.mp hs).1,(Finset.mem_sdiff.mp hs).2,he⟩
  47     have hgood (v : V) (hv : v∈L) : coinColourSurvives G τ v := by
  48       have h := hL v hv
  49       by_contra hn
  50       exact h.2.1 (Finset.mem_filter.mpr ⟨hBT h.1,hn⟩)
  51     have hwitness (v : V) (hv : v∈L) : G.Adj v (f v) ∧ (σ (f v)).1=(σ v).1 := by
  52       have h := hL v hv
  53       have hcoin : (σ v).2=true := by rw [h.2.2];exact (hgood v hv).1
  54       have hbad : ¬coinColourSurvives G σ v := (Finset.mem_filter.mp h.1).2
  55       have hconf : ∃ u,G.Adj v u ∧ (σ u).1=(σ v).1 := by
  56         by_contra hn
  57         apply hbad
  58         refine ⟨hcoin,?_⟩
  59         intro u hu he
  60         exact hn ⟨u,hu,he⟩
  61       exact cliqueConflictWitness_spec G T (fun v=>(σ v).1)
  62         (Finset.mem_filter.mpr ⟨hBT h.1,hconf⟩)
  63     have hchanged (v : V) (hv : v∈L) : f v∈E := by
  64       have h := hL v hv
  65       have hw := hwitness v hv
  66       apply Finset.mem_filter.mpr
  67       refine ⟨Finset.mem_univ _,?_⟩
  68       intro heq
  69       apply (hgood v hv).2 (f v) hw.1
  70       exact (congrArg Prod.fst heq).symm.trans
  71         (hw.2.trans (congrArg Prod.fst h.2.2))
  72     have hinj : Set.InjOn f (L : Set V) := by
  73       intro u hu v hv huv
  74       by_contra hne
  75       have hU := hL u hu
  76       have hV := hL v hv
  77       have hwu := hwitness u hu
  78       have hwv := hwitness v hv
  79       have hcol : (σ u).1=(σ v).1 := by
  80         calc
  81           _ = (σ (f u)).1 := hwu.2.symm
  82           _ = (σ (f v)).1 := congrArg (fun z=>(σ z).1) huv
  83           _ = _ := hwv.2
  84       apply (hgood u hu).2 v (hclique u (hBT hU.1) v (hBT hV.1) hne)
  85       exact (congrArg Prod.fst hV.2.2).symm.trans
  86         (hcol.symm.trans (congrArg Prod.fst hU.2.2))
  87     have himage : L.image f⊆(T∩E)∪((W\T)∩E) := by
  88       intro u hu
  89       obtain ⟨v,hv,rfl⟩ := Finset.mem_image.mp hu
  90       have he := hchanged v hv
  91       by_cases ht : f v∈T
  92       · exact Finset.mem_union_left _ (Finset.mem_inter.mpr ⟨ht,he⟩)
  93       · apply Finset.mem_union_right
  94         exact Finset.mem_inter.mpr ⟨Finset.mem_sdiff.mpr
  95           ⟨Finset.mem_image.mpr ⟨v,(hL v hv).1,rfl⟩,ht⟩,he⟩
  96     have hLcard : L.card≤(T∩E).card+((W\T)∩E).card := by
  97       rw [← Finset.card_image_of_injOn hinj]
  98       exact (Finset.card_le_card himage).trans (Finset.card_union_le _ _)
  99     have hcover : B⊆(B'∪(T∩E))∪L := by
 100       intro v hv
 101       by_cases hv' : v∈B'
 102       · exact Finset.mem_union_left _ (Finset.mem_union_left _ hv')
 103       · by_cases he : σ v=τ v
 104         · exact Finset.mem_union_right _ (Finset.mem_filter.mpr
 105             ⟨Finset.mem_sdiff.mpr ⟨hv,hv'⟩,he⟩)
 106         · exact Finset.mem_union_left _ (Finset.mem_union_right _
 107             (Finset.mem_inter.mpr ⟨hBT hv,Finset.mem_filter.mpr ⟨Finset.mem_univ _,he⟩⟩))
 108     have hBcard : B.card≤B'.card+(T∩E).card+L.card :=
 109       (Finset.card_le_card hcover).trans
 110         ((Finset.card_union_le _ _).trans
 111           (Nat.add_le_add_right (Finset.card_union_le _ _) _))
 112     have hcount : B.card≤B'.card+2*(T∩E).card+((W\T)∩E).card := by omega
 113     have heq : (∑ i,if σ i≠τ i then w i else 0)=
 114         2*((T∩E).card : ℝ)+(((W\T)∩E).card : ℝ) := by
 115       have hh := cliqueWitnessWeight_changed_sum T W E
 116       simpa only [E,Finset.mem_filter,Finset.mem_univ,true_and] using hh
 117     change (B.card : ℝ)≤(B'.card : ℝ)+_
 118     calc
 119       _ ≤ (B'.card : ℝ)+(2*((T∩E).card : ℝ)+(((W\T)∩E).card : ℝ)) := by
 120         exact_mod_cast (show B.card≤B'.card+(2*(T∩E).card+((W\T)∩E).card) by omega)
 121       _ = _ := congrArg (fun z=>(B'.card : ℝ)+z) heq.symm
 122     apply le_of_eq
 123     congr 1
 124     apply Finset.sum_congr rfl
 125     intro i _
 126     by_cases he : σ i=τ i <;> simp [he]
