  15 theorem closedSetNeighbours_mem_iff
  16     {V : Type*} [Fintype V] [DecidableEq V]
  17     (G : SimpleGraph V) (S : Finset V) (v : V) :
  18     v∈closedSetNeighbours G S ↔ ∃ u∈S,u=v ∨ G.Adj u v := by
  19   simp only [closedSetNeighbours,Finset.mem_union,setNeighbours,Finset.mem_filter,
  20     Finset.mem_univ,true_and]
  21   constructor
  22   · rintro (hv|⟨u,hu,huv⟩)
  23     · exact ⟨v,hv,Or.inl rfl⟩
  24     · exact ⟨u,hu,Or.inr huv⟩
  25   · rintro ⟨u,hu,h⟩
  26     rcases h with rfl|h
  27     · exact Or.inl hu
  28     · exact Or.inr ⟨u,hu,h⟩

  30 theorem closedSetNeighbours_incidence
  31     {V I : Type*} [Fintype V] [DecidableEq V] [Fintype I]
  32     (G : SimpleGraph V) (S : I → Finset V) {d b : ℕ}
  33     (hdegree : ∀ v,(Finset.univ.filter (fun u=>G.Adj v u)).card≤d)
  34     (hinc : ∀ v,(Finset.univ.filter (fun i=>v∈S i)).card≤b) (v : V) :
  35     (Finset.univ.filter (fun i=>v∈closedSetNeighbours G (S i))).card≤b*(d+1) := by
  36   classical
  37   have hsub : Finset.univ.filter (fun i=>v∈closedSetNeighbours G (S i))⊆
  38       (closedSetNeighbours G {v}).biUnion (fun u=>Finset.univ.filter (fun i=>u∈S i)) := by
  39     intro i hi
  40     obtain ⟨u,hu,huv⟩ := (closedSetNeighbours_mem_iff G (S i) v).mp (Finset.mem_filter.mp hi).2
  41     apply Finset.mem_biUnion.mpr
  42     refine ⟨u,(closedSetNeighbours_mem_iff G {v} u).mpr ⟨v,by simp,?_⟩,
  43       Finset.mem_filter.mpr ⟨Finset.mem_univ _,hu⟩⟩
  44     exact huv.elim (fun h=>Or.inl h.symm) (fun h=>Or.inr h.symm)
  45   calc
  46     _ ≤ ((closedSetNeighbours G {v}).biUnion (fun u=>Finset.univ.filter (fun i=>u∈S i))).card :=
  47       Finset.card_le_card hsub
  48     _ ≤ (closedSetNeighbours G {v}).card*b :=
  49       Finset.card_biUnion_le_card_mul _ _ _ (fun u _=>hinc u)
  50     _ ≤ (d+1)*b := Nat.mul_le_mul_right _ (by
  51       simpa only [Finset.card_singleton,one_mul] using closedSetNeighbours_card_le G hdegree {v})
  52     _ = _ := Nat.mul_comm _ _

  54 theorem finiteLaw_neighbourhood_localLemma
  55     {V I : Type*} [Fintype V] [DecidableEq V] [Fintype I] [DecidableEq I]
  56     {X : V → Type*} [∀ v,Fintype (X v)]
  57     (G : SimpleGraph V) (μ : ∀ v,FiniteLaw (X v))
  58     (A : I → (∀ v,X v) → Prop) (S : I → Finset V)
  59     {d a b : ℕ} (ha : 0<a) (hb : 0<b)
  60     (hdegree : ∀ v,(Finset.univ.filter (fun u=>G.Adj v u)).card≤d)
  61     (hsize : ∀ i,(S i).card≤a)
  62     (hinc : ∀ v,(Finset.univ.filter (fun i=>v∈S i)).card≤b)
  63     (hs : ∀ i σ τ,(∀ v∈closedSetNeighbours G (closedSetNeighbours G (S i)),σ v=τ v) →
  64       (A i σ ↔ A i τ))
  65     (hp : ∀ i,(FiniteLaw.pi μ).probability (A i)≤1/(4*((a*b*(d+1)^4 : ℕ) : ℝ))) :
  66     ∃ σ,0<(FiniteLaw.pi μ).mass σ ∧ ∀ i,¬A i σ := by
  67   have hsize' (i : I) : (closedSetNeighbours G (closedSetNeighbours G (S i))).card≤a*(d+1)^2 := by
  68     calc
  69       _ ≤ (closedSetNeighbours G (S i)).card*(d+1) := closedSetNeighbours_card_le G hdegree _
  70       _ ≤ ((S i).card*(d+1))*(d+1) := Nat.mul_le_mul_right _ (closedSetNeighbours_card_le G hdegree _)
  71       _ ≤ (a*(d+1))*(d+1) := by gcongr;exact hsize i
  72       _ = _ := by ring
  73   have hinc' (v : V) :
  74       (Finset.univ.filter (fun i=>v∈closedSetNeighbours G (closedSetNeighbours G (S i)))).card≤b*(d+1)^2 := by
  75     have h := closedSetNeighbours_incidence G (fun i=>closedSetNeighbours G (S i)) hdegree
  76       (closedSetNeighbours_incidence G S hdegree hinc) v
  77     convert h using 1 <;> ring
  78   apply finiteLaw_coordinate_localLemma_degree μ A
  79     (fun i=>closedSetNeighbours G (closedSetNeighbours G (S i))) hs
