  15 theorem equalizedColour_probability_at
  16     {V C : Type*} [Fintype V] [Fintype C]
  17     (G : SimpleGraph V) (L : V → Finset C) (hL : ∀ v,(L v).Nonempty)
  18     {l K : ℕ} (hl : 0<l) (hcard : ∀ v,(L v).card=l)
  19     (v : V) (c : C) (hc : c∈L v)
  20     (hbound : (colourNeighbourSet G L v c).card≤K) :
  21     (FiniteLaw.pi (fun u=>(FiniteLaw.uniformOn (L u) (hL u)).withCoin
  22       (equalizationCoin G L l K u) (equalizationCoin_bounds G L hl K u))).probability
  23       (fun σ=>coinRetainedColour G σ v=some c)=
  24       ((1-1/(l : ℝ))^K)/(l : ℝ) := by

  27   simp only [FiniteLaw.uniformOn,hc,ite_true,hcard,equalizationCoin]

  31 theorem finiteLaw_clique_restriction_uniform_loss
  32     {V C J : Type*} [Fintype V] [Fintype C] [Fintype J]
  33     (G : SimpleGraph V) (μ : FiniteLaw (V → C×Bool))
  34     (T T' : J → Finset V) (hsub : ∀ j,T' j⊆T j) (c : C)
  35     {d : ℕ} {p : ℝ} (hp : 0≤p)
  36     (hloss : ∀ j,((T j)\(T' j)).card≤d)
  37     (hpoint : ∀ j,∀ v∈T j,μ.probability (fun σ=>coinRetainedColour G σ v=some c)≤p) :
  38     0≤μ.probability (fun σ=>∀ j,∃ v∈T j,coinRetainedColour G σ v=some c)-
  39       μ.probability (fun σ=>∀ j,∃ v∈T' j,coinRetainedColour G σ v=some c) ∧
  40     μ.probability (fun σ=>∀ j,∃ v∈T j,coinRetainedColour G σ v=some c)-
  41       μ.probability (fun σ=>∀ j,∃ v∈T' j,coinRetainedColour G σ v=some c)≤
  42       (Fintype.card J : ℝ)*(d : ℝ)*p := by
  43   classical
  44   have hh := finiteLaw_clique_restriction_loss G μ Finset.univ T T' (fun j _=>hsub j) c
  45   simp only [Finset.mem_univ,forall_const] at hh
  46   refine ⟨hh.1,hh.2.trans ?_⟩
  47   calc
  48     _ ≤ ∑ j : J,(d : ℝ)*p := by
  49       apply Finset.sum_le_sum
  50       intro j _
  51       calc
  52         _ ≤ ∑ _v∈T j\T' j,p := Finset.sum_le_sum (fun v hv=>hpoint j v (Finset.mem_sdiff.mp hv).1)
  53         _ = (((T j)\T' j).card : ℝ)*p := by simp
  54         _ ≤ _ := mul_le_mul_of_nonneg_right (by exact_mod_cast hloss j) hp
  55     _ = _ := by simp;ring

  57 theorem finiteFamily_restriction_product_loss
  58     {J : Type*} [Fintype J] (m m' : J → ℕ) {l d : ℕ} (hl : 0<l)
  59     (hsub : ∀ j,m' j≤m j) (hsize : ∀ j,m j≤l)
  60     (hloss : ∀ j,m j-m' j≤d) {r : ℝ} (hr : 0≤r ∧ r≤1/(l : ℝ)) :
  61     0≤(∏ j,(m j : ℝ)*r)-(∏ j,(m' j : ℝ)*r) ∧
  62       (∏ j,(m j : ℝ)*r)-(∏ j,(m' j : ℝ)*r)≤
  63         (Fintype.card J : ℝ)*(d : ℝ)/(l : ℝ) := by
  64   classical
  65   have hlr : (0 : ℝ)<l := by exact_mod_cast hl

  67     refine ⟨mul_nonneg (Nat.cast_nonneg _) hr.1,
  68       mul_le_mul_of_nonneg_right (by exact_mod_cast hsub j) hr.1,?_⟩
  69     calc
  70       _ ≤ (m j : ℝ)*(1/(l : ℝ)) := mul_le_mul_of_nonneg_left hr.2 (Nat.cast_nonneg _)
  71       _ ≤ (l : ℝ)*(1/(l : ℝ)) := mul_le_mul_of_nonneg_right
  72         (by exact_mod_cast hsize j) (by positivity)
  73       _ = 1 := by field_simp
  74   have hh := finite_product_perturbation Finset.univ (fun j=>(m j : ℝ)*r)
  75     (fun j=>(m' j : ℝ)*r) (fun j _=>hfac j)
  76   refine ⟨hh.1,hh.2.trans ?_⟩
  77   have hterm (j : J) : (m j : ℝ)*r-(m' j : ℝ)*r≤(d : ℝ)/(l : ℝ) := by
  78     have he : ((m j-m' j : ℕ) : ℝ)=(m j : ℝ)-(m' j : ℝ) := Nat.cast_sub (hsub j)
  79     calc

  81       _ ≤ (d : ℝ)*r := mul_le_mul_of_nonneg_right (by exact_mod_cast hloss j) hr.1
  82       _ ≤ (d : ℝ)*(1/(l : ℝ)) := mul_le_mul_of_nonneg_left hr.2 (Nat.cast_nonneg _)
  83       _ = _ := by ring
  84   calc
  85     _ ≤ ∑ _j : J,(d : ℝ)/(l : ℝ) := Finset.sum_le_sum (fun j _=>hterm j)
  86     _ = _ := by simp;ring
