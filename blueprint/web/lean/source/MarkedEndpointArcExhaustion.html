  14 theorem marked_endpoint_arc_embedding_exhaustive
  15     (P : SimpleGraph V) (π : Equiv.Perm V) (d : ℕ) (A : Finset V)
  16     (mark incoming : Bool) (x : V) (σ : RawMarkedOutcome V U Γ Color)
  17     (e : MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) mark)
  18     (hroot : (if incoming then e.val.2 else e.val.1)=x)
  19     (hother : (if incoming then e.val.1 else e.val.2)∈A) :
  20     ∃ z,markedEndpointArcEmbedding P π d A mark incoming x σ z=e := by
  21   let y := if incoming then e.val.1 else e.val.2
  22   have hpair : (if incoming then y else x,if incoming then x else y)=e.val := by
  23     cases incoming <;> apply Prod.ext <;> first | exact hroot.symm | rfl
  24   have hy : y∈markedEndpointLabels P π d A (mark,incoming,x) σ := by
  25     rw [marked_endpoint_labels_actual]
  26     apply Finset.mem_filter.mpr
  27     refine ⟨hother,?_⟩
  28     have hh := congrArg (fun a:V×V=>markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) mark a.1 a.2) hpair
  29     exact hh.trans e.property
  30   refine ⟨⟨y,hy⟩,Subtype.ext hpair⟩

  32 theorem marked_endpoint_tagged_arc_embedding_exhaustive
  33     (P : SimpleGraph V) (π : Equiv.Perm V) (d : ℕ) (A : Finset V)
  34     (incoming : Bool) (x : V) (σ : RawMarkedOutcome V U Γ Color)
  35     (e : (MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) false)⊕
  36       (MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) true))
  37     (hroot : (if incoming then Sum.elim (fun e=>e.val.2) (fun e=>e.val.2) e
  38       else Sum.elim (fun e=>e.val.1) (fun e=>e.val.1) e)=x)
  39     (hother : (if incoming then Sum.elim (fun e=>e.val.1) (fun e=>e.val.1) e
  40       else Sum.elim (fun e=>e.val.2) (fun e=>e.val.2) e)∈A) :
  41     ∃ z,markedEndpointTaggedArcEmbedding P π d A incoming x σ z=e := by
  42   cases e with
  43   | inl e=>
  44     obtain ⟨z,hz⟩ := marked_endpoint_arc_embedding_exhaustive P π d A false incoming x σ e hroot hother
  45     exact ⟨Sum.inl z,congrArg Sum.inl hz⟩
  46   | inr e=>
  47     obtain ⟨z,hz⟩ := marked_endpoint_arc_embedding_exhaustive P π d A true incoming x σ e hroot hother
  48     exact ⟨Sum.inr z,congrArg Sum.inr hz⟩
