  14 theorem marked_reserved_inner_arc_set_lift
  15     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ) (σ:RawMarkedOutcome V U Γ Color)
  16     (C:Finset (Finset V⊕(V×U))) (I:Finset V) (J:Finset U)
  17     (hI:∀ x,π x∈I↔x∈I) (hC:∀ e∈C,¬markedInnerResource I J e):
  18     let E:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) false
  19     let K:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) true
  20     let tail:=fun e:E=>e.val.1
  21     let head:=fun e:E=>e.val.2
  22     let ktail:=fun e:K=>e.val.1
  23     let khead:=fun e:K=>e.val.2
  24     let DR:=physicalEdgeGraph (markedReservedRimPairs C)
  25     let σ':=rawMarkedDeleteRims σ DR.edgeFinset
  26     let E':=MarkedSampleArc P π (3*d) (fun e=>σ' (Sum.inl e)) false
  27     let K':=MarkedSampleArc P π (3*d) (fun e=>σ' (Sum.inl e)) true
  28     let emb:=markedSampleDeletedArcsEmbedding P π d σ DR.edgeFinset;
  29     ∀ A:Finset (E⊕K),
  30     (∀ a∈A,Sum.elim tail ktail a∈I∧Sum.elim head khead a∈I)→
  31     let A':=Finset.univ.filter (fun a:E'⊕K'=>emb a∈A);
  32     A'.image emb=A ∧ A'.card=A.card :=by
  33   intro E K tail head ktail khead DR σ' E' K' emb A hA A'
  34   have hsurvive:∀ a∈A,∃ a':E'⊕K',emb a'=a:=by
  35     intro a ha
  36     have hh:=hA a ha
  37     cases haType:a with
  38     | inl e=>
  39       obtain ⟨e',he'⟩:=marked_reserved_deletion_preserves_inner_arcs P π d σ C I J hI hC false e
  40         (by simpa only [haType,Sum.elim_inl] using hh.1) (by simpa only [haType,Sum.elim_inl] using hh.2)
  41       refine ⟨Sum.inl e',?_⟩
  42       apply congrArg Sum.inl
  43       apply Subtype.ext
  44       exact he'
  45     | inr e=>
  46       obtain ⟨e',he'⟩:=marked_reserved_deletion_preserves_inner_arcs P π d σ C I J hI hC true e
  47         (by simpa only [haType,Sum.elim_inr] using hh.1) (by simpa only [haType,Sum.elim_inr] using hh.2)
  48       refine ⟨Sum.inr e',?_⟩
  49       apply congrArg Sum.inr
  50       apply Subtype.ext
  51       exact he'
  52   have himage:A'.image emb=A:=by
  53     ext a
  54     constructor
  55     · intro ha
  56       obtain ⟨a',ha',rfl⟩:=Finset.mem_image.mp ha
  57       exact (Finset.mem_filter.mp ha').2
  58     · intro ha
  59       obtain ⟨a',he⟩:=hsurvive a ha
  60       exact Finset.mem_image.mpr ⟨a',Finset.mem_filter.mpr ⟨Finset.mem_univ _,he.symm ▸ ha⟩,he⟩
  61   exact ⟨himage,(Finset.card_image_of_injective A' emb.injective).symm.trans (congrArg Finset.card himage)⟩
  62 end Oberwolfach
