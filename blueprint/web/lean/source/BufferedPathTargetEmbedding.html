  10 theorem regular_two_buffered_path_target_embedding
  11     {V W:Type*} [Fintype V] [DecidableEq V]
  12     (G:SimpleGraph V) (hG:G.IsRegularOfDegree 2) (H:SimpleGraph W) (n:ℕ)
  13     (ext:Fin (n+5)↪V) (inner:Fin (n+1)↪V)
  14     (hinner:∀k,inner k=ext ⟨k.val+2,by omega⟩)
  15     (hstep:∀k:Fin (n+4),G.Adj (ext ⟨k.val,by omega⟩) (ext ⟨k.val+1,by omega⟩))
  16     (target:Fin (n+1)↪W)
  17     (htarget:∀k:Fin n,H.Adj (target ⟨k.val,by omega⟩) (target ⟨k.val+1,by omega⟩)):
  18     ∃f:{x // x∈Finset.univ.image inner}↪W,
  19       (∀k:Fin (n+1),f ⟨inner k,Finset.mem_image.mpr ⟨k,Finset.mem_univ _,rfl⟩⟩=target k) ∧
  20       Set.range f⊆Set.range target ∧
  21       (∀x y,G.Adj x.val y.val→H.Adj (f x) (f y)):=by
  22   classical
  23   obtain ⟨f,hf⟩:=finite_sequence_range_embedding (n+1) inner target
  24   refine ⟨f,hf,?_,?_⟩
  25   · rintro z ⟨x,rfl⟩
  26     obtain ⟨p,_,hp⟩:=Finset.mem_image.mp x.property
  27     refine ⟨p,?_⟩
  28     simpa only [hp] using (hf p).symm
  29   · intro x y hxy
  30     obtain ⟨p,_,hp⟩:=Finset.mem_image.mp x.property
  31     obtain ⟨q,_,hq⟩:=Finset.mem_image.mp y.property
  32     have hfx:f x=target p:=by simpa only [hp] using hf p
  33     have hfy:f y=target q:=by simpa only [hq] using hf q
  34     have hsource:G.Adj (inner p) (inner q):=by simpa only [hp,hq] using hxy
  35     have hindex:=regular_two_buffered_path_adjacent_indices G hG n ext inner hinner hstep p q hsource
  36     rw [hfx,hfy]
  37     rcases hindex with hpq|hqp
  38     · have hpbound:p.val<n:=by omega
  39       have ha:=htarget ⟨p.val,hpbound⟩
  40       have hp':(⟨p.val,by omega⟩:Fin (n+1))=p:=Fin.ext rfl
  41       have hq':(⟨p.val+1,by omega⟩:Fin (n+1))=q:=Fin.ext hpq
  42       simpa only [hp',hq'] using ha
  43     · have hqbound:q.val<n:=by omega
  44       have ha:=(htarget ⟨q.val,hqbound⟩).symm
  45       have hq':(⟨q.val,by omega⟩:Fin (n+1))=q:=Fin.ext rfl
  46       have hp':(⟨q.val+1,by omega⟩:Fin (n+1))=p:=Fin.ext hqp
  47       simpa only [hp',hq'] using ha
  48 end Oberwolfach
