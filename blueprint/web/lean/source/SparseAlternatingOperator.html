  14 def sparseRectangleCoefficient (H : A → B → Prop) (q : B → ℝ) (z : A×B → ℝ)
  15     (x y : A) (u v : B) : ℝ :=
  16   (bipAdj H x u*bipAdj H y u*bipAdj H y v*bipAdj H x v)*
  17     (z (x,u)*q u*(q v)^2)/((Fintype.card A:ℝ)*Fintype.card B)

  19 def sparseAlternatingMap (H : A → B → Prop) (q : B → ℝ) :
  20     (A×B → ℝ) →ₗ[ℝ] (A×B → ℝ) where
  21   toFun z := rectangleCombination (sparseRectangleCoefficient H q z)
  22   map_add' z w := by
  23     ext e
  24     simp only [rectangleCombination,sparseRectangleCoefficient,Pi.add_apply,add_mul,mul_add,add_div,
  25       Finset.sum_add_distrib]
  26   map_smul' c z := by
  27     ext e
  28     simp only [rectangleCombination,sparseRectangleCoefficient,Pi.smul_apply,smul_eq_mul,RingHom.id_apply]
  29     simp only [Finset.mul_sum]
  30     apply Finset.sum_congr rfl;intro x _
  31     apply Finset.sum_congr rfl;intro y _
  32     apply Finset.sum_congr rfl;intro u _
  33     apply Finset.sum_congr rfl;intro v _
  34     ring
