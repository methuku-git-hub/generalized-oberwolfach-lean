  13     (μ : FiniteLaw Ω) (A : Ω → Prop) : μ.probability A≤1 := by
  14   have hc := finiteLaw_probability_compl μ A
  15   linarith [finiteLaw_probability_nonneg μ (fun ω => ¬A ω)]

  17 theorem finiteLaw_prod_probability_slices {Ω Γ : Type*} [Fintype Ω] [Fintype Γ]
  18     (μ : FiniteLaw Ω) (ν : FiniteLaw Γ) (A : Ω → Γ → Prop) :
  19     (μ.prod ν).probability (fun z => A z.1 z.2)=
  20       ∑ ω,μ.mass ω*ν.probability (A ω) := by
  21   unfold FiniteLaw.probability
  22   rw [Fintype.sum_prod_type]
  23   apply Finset.sum_congr rfl
  24   intro ω _
  25   rw [Finset.mul_sum]
  26   apply Finset.sum_congr rfl
  27   intro γ _
  28   change (if A ω γ then μ.mass ω*ν.mass γ else 0)=
  29     μ.mass ω*(if A ω γ then ν.mass γ else 0)
  30   split_ifs <;> simp

  32 theorem finiteLaw_product_rooted_failure {Ω Γ : Type*} [Fintype Ω] [Fintype Γ]
  33     (μ : FiniteLaw Ω) (ν : FiniteLaw Γ) (E A : Ω → Prop) (B : Ω → Γ → Prop)
  34     (b c : ℝ) (hc : 0≤c)
  35     (hA : μ.probability (fun ω => E ω ∧ A ω)≤b*μ.probability E)
  36     (hB : ∀ ω,¬A ω → ν.probability (B ω)≤c) :
  37     (μ.prod ν).probability (fun z => E z.1 ∧ B z.1 z.2)≤
  38       (b+c)*μ.probability E := by
  39   rw [finiteLaw_prod_probability_slices μ ν (fun ω γ => E ω ∧ B ω γ)]
  40   have hp (ω : Ω) : ν.probability (fun γ => E ω ∧ B ω γ)≤
  41       (if E ω ∧ A ω then 1 else 0)+c*(if E ω then 1 else 0) := by
  42     by_cases hE : E ω
  43     · by_cases hbad : A ω
  44       · simp only [hE,true_and,hbad,ite_true,mul_one]
  45         exact (finiteLaw_probability_le_one' ν (B ω)).trans (by linarith)
  46       · simp only [hE,true_and,hbad,ite_false,ite_true,zero_add,mul_one]
  47         exact hB ω hbad
  48     · simp [hE,FiniteLaw.probability]
  49   calc
  50     _ ≤ ∑ ω,μ.mass ω*((if E ω ∧ A ω then 1 else 0)+c*(if E ω then 1 else 0)) :=
  51       Finset.sum_le_sum (fun ω _ => mul_le_mul_of_nonneg_left (hp ω) (μ.nonneg ω))
  52     _ = μ.probability (fun ω => E ω ∧ A ω)+c*μ.probability E := by
  53       unfold FiniteLaw.probability
  54       rw [Finset.mul_sum,← Finset.sum_add_distrib]
  55       apply Finset.sum_congr rfl
  56       intro ω _
  57       by_cases hE : E ω <;> by_cases hbad : A ω <;> simp [hE,hbad] <;> ring
  58     _ ≤ _ := by nlinarith only [hA]
