  15 theorem prescribed_path_physical_edges_avoid (D:V→V→Prop) (F:SimpleGraph V)
  16     (y:V):∀ xs:List V,∀ x:V,DirectedPathThrough D F x y xs→
  17       Disjoint (prescribedPathPhysicalEdges x y xs) F.edgeFinset := by
