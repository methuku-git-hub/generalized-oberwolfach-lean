  17     (α pstar : ℝ) (hα : 0<α) (hα1 : α≤1/2) (hpstar : 0<pstar)
  18     (k b q₀ : ℕ) :
  19     ∃ N : ℕ,∀ (V : Type) [Fintype V] [LinearOrder V],N≤Fintype.card V →
  20     ∀ (P : SimpleGraph V), (∀ v,(1/2+α)*Fintype.card V≤(P.degree v:ℝ)) →
  21     ∀ (ρ : Fin k→ℝ) (hρ : ∀ a,0≤ρ a) (hsum : ∑ a,ρ a=1),
  22     1-1/(Fintype.card V:ℝ)^q₀≤
  23       (FiniteLaw.pi (fun _ : Sym2 V=>FiniteLaw.categoricalDirection ρ hρ hsum)).probability
  24         (fun σ=>∀ a : Fin k,pstar≤ρ a → ∀ (π τ : Equiv.Perm V),
  25           ∀ δ : ℝ,δ≤α/32 → δ≤ρ a/64 → ∀ S T : Finset V,
  26           (1-2*δ)*Fintype.card V<(S.card:ℝ)+T.card →
  27           (α*ρ a/8)*Fintype.card V*min (S.card:ℝ) T.card≤
  28             (matrixCut (permutationSeparatedMatrix π b
  29               (fun x y=>sampledColourOrientation P σ a x (τ y))) S T:ℝ)) := by

  31   obtain ⟨K,hK⟩ := exists_nat_ge ((2*(b:ℝ)+1)*8/(α*pstar))
  32   refine ⟨max N K,?_⟩
  33   intro V _ _ hn P hP ρ hρ hsum
  34   apply (hN V (le_trans (le_max_left _ _) hn) P hP ρ hρ hsum).trans
  35   apply finiteLaw_probability_mono
  36   intro σ hσ a ha π τ δ hδα hδρ S T hlarge
  37   have hcard : (T.image τ).card=T.card := Finset.card_image_iff.mpr (fun x _ y _ h=>τ.injective h)
  38   have hcut := hσ a ha δ hδα hδρ S (T.image τ) (by simpa only [hcard] using hlarge)
  39   rw [hcard,← matrixCut_permute_columns] at hcut
  40   have hloss := permutationSeparatedMatrix_cut_loss π b
  41     (fun x y=>sampledColourOrientation P σ a x (τ y))
  42     (by intro x y;unfold sampledColourOrientation;split_ifs <;> omega) S T
  43   have hKn : (K:ℝ)≤Fintype.card V := by exact_mod_cast (le_trans (le_max_right N K) hn)
  44   have hmargin : 2*(b:ℝ)+1≤α*pstar/8*Fintype.card V := by
  45     have hh := (div_le_iff₀ (mul_pos hα hpstar)).mp (hK.trans hKn)
  46     nlinarith only [hh]

  50   have hm := mul_le_mul_of_nonneg_right hmargin'
  51     (show (0:ℝ)≤min (S.card:ℝ) T.card by positivity)
  52   nlinarith only [hcut,hloss,hm]
