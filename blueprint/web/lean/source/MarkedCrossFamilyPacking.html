  15  theorem marked_cycles_disjoint_of_shared_labels
  16     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  17     (C D:MarkedCycleTuple tail head ktail khead m π b) (S:Finset V)
  18     (hshared:∀ i j,C.val i=D.val j→D.val j∈S)
  19     (hnoarc:∀ j,¬(D.val j∈S∧D.val (finRotate (m+1) j)∈S)):
  20     Disjoint (markedCyclePhysicalEdges tail head ktail khead m π b C)
  21       (markedCyclePhysicalEdges tail head ktail khead m π b D):=by
  22   apply Finset.disjoint_left.mpr
  23   intro e he hf
  24   obtain ⟨i,_,hi⟩:=Finset.mem_image.mp he
  25   obtain ⟨j,_,hj⟩:=Finset.mem_image.mp hf
  26   have hc:=marked_cycle_common_physical_endpoints hp C D i j (hi.trans hj.symm)
  27   exact hnoarc j ⟨hshared i j hc.1,hshared _ _ hc.2⟩

  29  theorem marked_wheels_disjoint_of_shared_labels
  30     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  31     (C D:MarkedWheel tail head ktail khead m π b H₀ HK) (S:Finset V)
  32     (hshared:∀ i j,C.1.val i=D.1.val j→D.1.val j∈S)
  33     (hnoarc:∀ j,¬(D.1.val j∈S∧D.1.val (finRotate (m+1) j)∈S))
  34     (hspoke:∀ i j,C.1.val i=D.1.val j→C.2.val≠D.2.val):
  35     Disjoint (markedWheelPhysicalResources C) (markedWheelPhysicalResources D):=by
  36   have hr:=marked_cycles_disjoint_of_shared_labels hp C.1 D.1 S hshared hnoarc
  37   apply Finset.disjoint_left.mpr
  38   intro e he hf
  39   cases e with
  40   | inl e=>
  41     exact Finset.disjoint_left.mp hr ((markedWheelPhysicalResources_rim C e).mp he)
  42       ((markedWheelPhysicalResources_rim D e).mp hf)
  43   | inr vu=>
  44     rw [markedWheelPhysicalResources_spoke] at he hf
  45     obtain ⟨i,_,hi⟩:=Finset.mem_image.mp he.2
  46     obtain ⟨j,_,hj⟩:=Finset.mem_image.mp hf.2
  47     exact hspoke i j (hi.trans hj.symm) (he.1.trans hf.1.symm)

  49  theorem marked_rim_spoke_cross_packing (n:ℕ)
  50     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  51     (p:Fin (n+1)→V) (v w:V) (jr:Bool) (q:Fin (n+2)→V) (x:V) (js:Bool)
  52     (R:Finset V) (C D:MarkedWheel tail head ktail khead (n+2) π b H₀ HK)
  53     (hC:∀ i,C.1.val i=Fin.cons (α:=fun _=>V) v (Fin.cons w p)
  54       (canonicalMarkedRotation (if jr then 0 else 3) i))
  55     (hD:∀ i,D.1.val i=rootedCycleOrder (markedRootOrder (n+2) js) x q i)
  56     (hv:v∈R) (hw:w∈R) (hx:x∈R)
  57     (hq:∀ i,q i∉R) (hdis:Disjoint (Finset.univ.image p) (Finset.univ.image q))
  58     (hhub:C.2.val≠D.2.val):
  59     Disjoint (markedWheelPhysicalResources C) (markedWheelPhysicalResources D):=by
  60   have hCs:∀ i,C.1.val i∈R∪Finset.univ.image p:=by
  61     intro i
  62     have hi:C.1.val i∈Finset.univ.image C.1.val:=Finset.mem_image.mpr ⟨i,Finset.mem_univ _,rfl⟩
  63     have him:Finset.univ.image C.1.val=insert v (insert w (Finset.univ.image p)):=by
  64       rw [funext hC,two_root_cycle_image]
  65     rw [him] at hi
  66     rcases Finset.mem_insert.mp hi with hi|hi
  67     · exact Finset.mem_union_left _ (hi ▸ hv)
  68     · rcases Finset.mem_insert.mp hi with hi|hi
  69       · exact Finset.mem_union_left _ (hi ▸ hw)
  70       · exact Finset.mem_union_right _ hi
  71   have hDs:∀ i,D.1.val i∈R∪Finset.univ.image q:=by
  72     intro i
  73     have hi:D.1.val i∈Finset.univ.image D.1.val:=Finset.mem_image.mpr ⟨i,Finset.mem_univ _,rfl⟩
  74     rw [funext hD,rootedCycleOrder_vertices] at hi
  75     rw [←hD i] at hi
  76     rcases Finset.mem_insert.mp hi with hi|hi
  77     · exact Finset.mem_union_left _ (hi ▸ hx)
  78     · exact Finset.mem_union_right _ hi
  79   apply marked_wheels_disjoint_of_shared_labels hp C D R ?_
  80     (marked_one_root_wheel_no_old_arc q x R hq (markedRootOrder (n+2) js)
  81       (markedRootOrder_successor (n+2) js) D hD) (fun _ _ _=>hhub)
  82   intro i j hij
  83   have h1:=hij ▸ hCs i
  84   have h2:=hDs j
  85   rcases Finset.mem_union.mp h1 with h1|h1
  86   · exact h1
  87   · rcases Finset.mem_union.mp h2 with h2|h2
  88     · exact h2
  89     · exact False.elim (Finset.disjoint_left.mp hdis h1 h2)
