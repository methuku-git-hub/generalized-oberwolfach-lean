  11 theorem actual_prescribed_old_piece_is_run_endpoint
  12     {X K J:Type} [Fintype X] [DecidableEq X] [DecidableEq K] [DecidableEq J]
  13     {G:SimpleGraph X} (data:ActualResidualData (K:=K) (J:=J) G) (hG:G.IsRegularOfDegree 2)
  14     (x:X) (hx:x∈data.prescribed)
  15     (hA:x∈Finset.univ.biUnion (fun z:actualOldPieceIndex data=>(actualOldPieceWalk data z).support.toFinset)):
  16     ∃r:SelectedRunIndex data.s₂ data.d data.T,x=data.start (.inr r) ∨ x=data.finish (.inr r):=by
  17   have hrem:x∈data.remaining:=(data.graph_facts hG).2.1 hx
  18   have hnot:x∉data.deletedVertices:=(Finset.mem_sdiff.mp hrem).2
  19   rw [actual_old_piece_domain_eq data] at hA
  20   rcases Finset.mem_union.mp hA with hp|hc
  21   · obtain ⟨r,_,hr⟩:=Finset.mem_biUnion.mp hp
  22     by_cases hs:x=data.start (.inr r)
  23     · exact ⟨r,Or.inl hs⟩
  24     by_cases hf:x=data.finish (.inr r)
  25     · exact ⟨r,Or.inr hf⟩
  26     · have hd:x∈data.deletedVertices:=by
  27         apply Finset.mem_union_right
  28         apply Finset.mem_biUnion.mpr
  29         refine ⟨.inr r,Finset.mem_univ _,Finset.mem_sdiff.mpr ⟨hr,?_⟩⟩
  30         simp [hs,hf]
  31       exact False.elim (hnot hd)
  32   · have hd:x∈data.deletedVertices:=Finset.mem_union_left _ hc
  33     exact False.elim (hnot hd)
  34 end Oberwolfach
