  50 theorem markedPairPhysical_fiber_card (π : Equiv.Perm V) (e : Sym2 V) :
  51     (Finset.univ.filter (fun a=>markedPairPhysical π a=e)).card≤4 := by
  52   induction e using Sym2.ind with
  53   | _ x y=>
  54     let T : Finset ((V×V)⊕(V×V)) :=
  55       {Sum.inl (x,y),Sum.inl (y,x),Sum.inr (x,π.symm y),Sum.inr (y,π.symm x)}
  56     have hsub : Finset.univ.filter (fun a=>markedPairPhysical π a=s(x,y))⊆T := by
  57       intro a ha
  58       have ha' := (Finset.mem_filter.mp ha).2
  59       cases a with
  60       | inl a=>
  61         rcases Sym2.eq_iff.mp ha' with ⟨hx,hy⟩|⟨hx,hy⟩
  62         · simp [T,← hx,← hy]
  63         · simp [T,← hx,← hy]
  64       | inr a=>
  65         rcases Sym2.eq_iff.mp ha' with ⟨hx,hy⟩|⟨hx,hy⟩
  66         · have hy' : a.2=π.symm y := (Equiv.eq_symm_apply π).mpr hy
  67           simp [T,← hx,← hy']
  68         · have hy' : a.2=π.symm x := (Equiv.eq_symm_apply π).mpr hy
  69           simp [T,← hx,← hy']
  70     apply (Finset.card_le_card hsub).trans
  71     dsimp [T]
  72     have h₁ := Finset.card_insert_le (Sum.inl (x,y))
  73       ({Sum.inl (y,x),Sum.inr (x,π.symm y),Sum.inr (y,π.symm x)}:Finset ((V×V)⊕(V×V)))
  74     have h₂ := Finset.card_insert_le (Sum.inl (y,x))
  75       ({Sum.inr (x,π.symm y),Sum.inr (y,π.symm x)}:Finset ((V×V)⊕(V×V)))
  76     have h₃ : ({Sum.inr (x,π.symm y),Sum.inr (y,π.symm x)}:Finset ((V×V)⊕(V×V))).card≤2 := Finset.card_le_two
  77     omega
