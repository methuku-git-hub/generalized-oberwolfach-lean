  16     (W:MarkedWheel tail head ktail khead m π b H₀ HK) (z:Fin (m+1)→V) (k:Fin (m+1))
  17     (hW:∀ i,W.1.val i=z (canonicalMarkedRotation k i)):
  18     HK (z (finRotate (m+1) k)) W.2.val∧
  19       ∀ i,i≠finRotate (m+1) k→H₀ (z i) W.2.val:=by
  20   constructor
  21   · simpa only [hW,canonicalMarkedRotation_zero] using W.2.property.1
  22   · intro i hi
  23     let a:Fin (m+1):=(canonicalMarkedRotation k).symm i
  24     have hai:canonicalMarkedRotation k a=i:=Equiv.apply_symm_apply _ _
  25     have ha:a≠0:=by
  26       intro he
  27       rw [he,canonicalMarkedRotation_zero] at hai
  28       exact hi hai.symm
  29     have ht:H₀ (W.1.val a) W.2.val:=by
  30       revert ha
