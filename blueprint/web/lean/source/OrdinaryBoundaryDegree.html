  16 theorem ordinary_boundary_degree (Q R:SimpleGraph V) (H S:V→U→Prop)
  17     (X:Finset V) (Z:Finset U) (x:I↪V) (u:J↪U)
  18     (hx:∀ i,x i∉X) (hu:∀ j,u j∉Z)
  19     (hxcover:∀ a∉X,∃ i,x i=a) (hucover:∀ z∉Z,∃ j,u j=z)
  20     (A:I→Finset V) (B:I→Finset U) (Y:J→Finset V)
  21     (hQr:∀ i a,Q.Adj (x i) a↔a∈A i)
  22     (hHr:∀ i z,H (x i) z↔z∈B i)
  23     (hHc:∀ j a,H a (u j)↔a∈Y j)
  24     (hRQ:R≤Q) (hSH:∀ a z,S a z→H a z)
  25     (hremain:(Q\R).support⊆(X:Set V))
  26     (hcross:∀ a z,H a z∧¬S a z→a∈X∧z∈Z)
  27     (d k:ℕ)
  28     (hinner:∀ w,(spanningRestriction (ordinaryPhysicalGraph R S) (ordinaryInnerVertices X Z)).degree w≤d)
  29     (hmem:∀ w,(Finset.univ.filter (fun i:J⊕I=>
  30       w∈Sum.elim (fun j=>(Y j).image (Sum.inl:V→V⊕U))
  31         (fun i=>(A i).image Sum.inl∪(B i).image Sum.inr) i)).card≤k):
  32     ∀ w∈ordinaryInnerVertices X Z,(ordinaryPhysicalGraph R S).degree w≤d+k:=by
  33   let roots:J⊕I↪V⊕U:=⟨Sum.elim (fun j=>Sum.inr (u j)) (fun i=>Sum.inl (x i)),by
  34     intro a b h
  35     cases a with
  36     | inl a=>
  37       cases b with
  38       | inl b=>exact congrArg Sum.inl (u.injective (Sum.inr.inj h))
  39       | inr b=>exact (Sum.inr_ne_inl h).elim
  40     | inr a=>
  41       cases b with
  42       | inl b=>exact (Sum.inl_ne_inr h).elim
  43       | inr b=>exact congrArg Sum.inr (x.injective (Sum.inl.inj h))⟩
  44   let domains:J⊕I→Finset (V⊕U):=Sum.elim (fun j=>(Y j).image Sum.inl)
  45     (fun i=>(A i).image Sum.inl∪(B i).image Sum.inr)
  46   have hrow (a b:V) (ha:a∉X):R.Adj a b↔Q.Adj a b:=by
  47     refine ⟨fun h=>hRQ h,fun h=>?_⟩
  48     by_contra hn
  49     exact ha (hremain (show a∈(Q\R).support from ⟨b,h,hn⟩))
  50   have hspoke (a:V) (z:U) (h:a∉X∨z∉Z):S a z↔H a z:=by
  51     refine ⟨hSH a z,fun haz=>?_⟩
  52     by_contra hn
  53     have hh:=hcross a z ⟨haz,hn⟩
  54     exact h.elim (fun ha=>ha hh.1) (fun hz=>hz hh.2)
  55   have hout:∀ z∉ordinaryInnerVertices X Z,∀ v,
  56       (ordinaryPhysicalGraph R S).Adj z v↔∃ i,roots i=z∧v∈domains i:=by
  57     intro z hz v
  58     cases z with
  59     | inl a=>
  60       have ha:a∉X:=fun h=>hz ((ordinaryInnerVertices_left X Z a).mpr h)
  61       obtain ⟨i,hi⟩:=hxcover a ha
  62       subst a
  63       cases v with
  64       | inl b=>
  65         change R.Adj (x i) b↔_
  66         rw [hrow _ _ (hx i),hQr]
  67         constructor
  68         · intro hb;exact ⟨Sum.inr i,rfl,by simp [domains,hb]⟩
  69         · rintro ⟨j,hj,hb⟩
  70           cases j with
  71           | inl j=>exact (Sum.inr_ne_inl hj).elim
  72           | inr j=>have he:j=i:=x.injective (Sum.inl.inj hj);subst j;simpa [domains] using hb
  73       | inr z=>
  74         change S (x i) z↔_
  75         rw [hspoke _ _ (Or.inl (hx i)),hHr]
  76         constructor
  77         · intro hb;exact ⟨Sum.inr i,rfl,by simp [domains,hb]⟩
  78         · rintro ⟨j,hj,hb⟩
  79           cases j with
  80           | inl j=>exact (Sum.inr_ne_inl hj).elim
  81           | inr j=>have he:j=i:=x.injective (Sum.inl.inj hj);subst j;simpa [domains] using hb
  82     | inr z=>
  83       have hz':z∉Z:=fun h=>hz ((ordinaryInnerVertices_right X Z z).mpr h)
  84       obtain ⟨j,hj⟩:=hucover z hz'
  85       subst z
  86       cases v with
  87       | inl a=>
  88         change S a (u j)↔_
  89         rw [hspoke _ _ (Or.inr (hu j)),hHc]
  90         constructor
  91         · intro ha;exact ⟨Sum.inl j,rfl,by simp [domains,ha]⟩
  92         · rintro ⟨i,hi,ha⟩
  93           cases i with
  94           | inl i=>have he:i=j:=u.injective (Sum.inr.inj hi);subst i;simpa [domains] using ha
  95           | inr i=>exact (Sum.inl_ne_inr hi).elim
  96       | inr z=>
  97         change False↔_
  98         refine ⟨False.elim,?_⟩
  99         rintro ⟨i,hi,ha⟩
 100         cases i with
 101         | inl i=>simpa [domains] using ha
 102         | inr i=>exact Sum.inl_ne_inr hi
