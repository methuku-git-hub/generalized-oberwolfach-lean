  12 theorem actual_source_orientation_covers_and_has_unique_incoming
  13     {X:Type} [Fintype X] [DecidableEq X] (G:SimpleGraph X)
  14     (pieces:List (PrescribedSourcePiece X)) (D:X→X→Prop)
  15     (hpair:pieces.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset))
  16     (hvalid:∀p∈pieces,p.OriginalValid D)
  17     (hEdges:sourcePieceEdgesUnion pieces=G.edgeFinset)
  18     (hArcs:∀x y,D x y↔(x,y)∈sourcePieceArcsUnion pieces):
  19     (∀x y,G.Adj x y→D x y ∨ D y x) ∧
  20       (∀x y z,D y x→D z x→y=z):=by
  21   classical
  22   constructor
  23   · intro x y hxy
  24     have he:s(x,y)∈sourcePieceEdgesUnion pieces:=by
  25       rw [hEdges]
  26       exact SimpleGraph.mem_edgeFinset.mpr hxy
  27     obtain ⟨p,hp,he⟩:=Finset.mem_biUnion.mp he
  28     rw [prescribed_piece_edges_are_arc_images] at he
  29     obtain ⟨arc,harc,heq⟩:=Finset.mem_image.mp he
  30     rcases arc with ⟨u,v⟩
  31     change s(u,v)=s(x,y) at heq
  32     rcases Sym2.eq_iff.mp heq with ⟨rfl,rfl⟩ | ⟨rfl,rfl⟩
  33     · left
  34       apply (hArcs _ _).2
  35       exact Finset.mem_biUnion.mpr ⟨p,hp,harc⟩
  36     · right
  37       apply (hArcs _ _).2
  38       exact Finset.mem_biUnion.mpr ⟨p,hp,harc⟩
  39   · intro x y z hy hz
  40     obtain ⟨p,hp,hyp⟩:=Finset.mem_biUnion.mp ((hArcs y x).1 hy)
  41     obtain ⟨q,hq,hzq⟩:=Finset.mem_biUnion.mp ((hArcs z x).1 hz)
  42     have hp':p∈pieces:=List.mem_toFinset.mp hp
  43     have hq':q∈pieces:=List.mem_toFinset.mp hq
  44     rcases prescribed_pairwise_rel_or_eq (fun p q:PrescribedSourcePiece X=>Disjoint p.vertices.toFinset q.vertices.toFinset)
  45       (fun _ _ h=>h.symm) pieces hpair p hp' q hq' with heq | hdis
  46     · subst q
  47       exact prescribed_piece_unique_predecessor p D ⊥ (prescribed_original_piece_valid p D (hvalid p hp')) x y z hyp hzq
  48     · have hxp:x∈p.vertices.toFinset:=List.mem_toFinset.mpr (prescribed_source_piece_arc_endpoints p (y,x) hyp).2
  49       have hxq:x∈q.vertices.toFinset:=List.mem_toFinset.mpr (prescribed_source_piece_arc_endpoints q (z,x) hzq).2
  50       exact False.elim (Finset.disjoint_left.mp hdis hxp hxq)
