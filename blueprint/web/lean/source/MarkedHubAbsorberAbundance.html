  13 theorem paired_absorber_order_threshold (m : ℕ) (β : ℝ) (hβ : 0<β) :
  14     ∃ N : ℕ,∀ n : ℕ,N≤n →
  15       2*(m+1:ℝ)^2≤β*n ∧ 2*(m:ℝ)*(2*(m+1)+(m+1)*m)≤β*n := by
  16   obtain ⟨N,hN⟩ := exists_nat_gt (max (2*(m+1:ℝ)^2/β)
  17     (2*(m:ℝ)*(2*(m+1)+(m+1)*m)/β))
  18   refine ⟨N,?_⟩
  19   intro n hn
  20   have hNn : (N:ℝ)≤n := by exact_mod_cast hn
  21   have h₁ := (le_max_left (2*(m+1:ℝ)^2/β) _).trans (hN.le.trans hNn)
  22   have h₂ := (le_max_right _ (2*(m:ℝ)*(2*(m+1)+(m+1)*m)/β)).trans (hN.le.trans hNn)
  23   exact ⟨by nlinarith only [(div_le_iff₀ hβ).mp h₁],
  24     by nlinarith only [(div_le_iff₀ hβ).mp h₂]⟩
