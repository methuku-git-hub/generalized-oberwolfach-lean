  26     Fintype.card (∀ i,FixedSubset (V i) (a i)) ≤ Fintype.card (Σ i,V i)^c := by
  27   let f (σ : ∀ i,FixedSubset (V i) (a i)) : FixedSubset (Σ i,V i) c :=

  29   have hf : Function.Injective f := by

  31     exact profileRootSet_injective a (congrArg Subtype.val he)
  32   have hh := Fintype.card_le_of_injective f hf
  33   have hcard : Fintype.card (FixedSubset (Σ i,V i) c)=(Fintype.card (Σ i,V i)).choose c := by
