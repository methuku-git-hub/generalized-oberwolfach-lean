  15 theorem subtype_filter_card_eq {E : Type*} [Fintype E]
  16     (P Q : E → Prop) [DecidablePred P] :
  17     (Finset.univ.filter (fun e : {e // P e}=>Q e.val)).card=
  18       (Finset.univ.filter (fun e=>P e ∧ Q e)).card := by
  19   classical
  20   apply Finset.card_bij (fun e _=>e.val)
  21   · intro e he
  22     exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,e.property,(Finset.mem_filter.mp he).2⟩
  23   · intro e _ f _ hef
  24     exact Subtype.ext hef
  25   · intro e he
  26     have h := (Finset.mem_filter.mp he).2
  27     exact ⟨⟨e,h.1⟩,Finset.mem_filter.mpr ⟨Finset.mem_univ _,h.2⟩,rfl⟩

  29 theorem partition_colouring_weighted_matching
  30     (k : ℕ) (hk : 0<k) (ε δ : ℝ) (hε : 0<ε) (hε1 : ε≤1/2)
  31     (hδ : 0<δ) (hδ1 : δ≤1) :
  32     ∃ D₀ : ℕ, ∀ D B : ℕ, D₀≤D → (B : ℝ)≤(D : ℝ)^(1-δ) →
  33       ∀ (V E I J T : Type) [Fintype V] [Fintype E] [Fintype I] [Fintype J] [Fintype T]
  34         [Nonempty J],
  35       ∀ (R : E → Finset V) (part : E → I) (split : E → J),
  36       (∀ e,(R e).card=k) →
  37       (∀ e f,part e≠part f → Disjoint (R e) (R f)) →
  38       (∀ i j v,(Finset.univ.filter (fun e=>part e=i ∧ split e=j ∧ v∈R e)).card≤D) →
  39       (∀ i j v u,v≠u →
  40         (Finset.univ.filter (fun e=>part e=i ∧ split e=j ∧ v∈R e ∧ u∈R e)).card≤B) →
  41       ∀ (w : T → E → ℝ) (b : T → I → ℝ) (t : T → ℝ),
  42       (∀ a e,0≤w a e) →
  43       (∀ a i j,(∑ e,if part e=i ∧ split e=j then w a e else 0)≤b a i) →
  44       (∀ a,0≤t a) →
  45       (∑ a,2*Real.exp (-2*(t a)^2/(∑ i,(b a i)^2)))<1 →
  46       ∃ M : Finset E,
  47         (∀ e∈M,∀ f∈M,e≠f → Disjoint (R e) (R f)) ∧
  48         ∀ a,|(∑ e∈M,w a e)-(∑ e,w a e)/
  49           ((Fintype.card J : ℝ)*⌈(1+ε)*(D : ℝ)⌉₊)|<t a := by
  50   classical
  51   obtain ⟨N,hNpos,hN⟩ := near_optimal_hypergraph_edge_colouring k hk ε δ hε hε1 hδ hδ1
  52   refine ⟨N,?_⟩
  53   intro D B hD hB V E I J T _ _ _ _ _ _ R part split hsize hparts hdegree hcodeg w b t hw hb ht hsmall
  54   let l := ⌈(1+ε)*(D : ℝ)⌉₊
  55   have hl : 0<l := by
  56     have hDR : (0 : ℝ)<D := Nat.cast_pos.mpr (lt_of_lt_of_le (by omega : 0<N) hD)
  57     exact Nat.ceil_pos.mpr (by positivity)
  58   letI : Nonempty (Fin l) := Fin.pos_iff_nonempty.mp hl
  59   have hcells : ∀ ij : I×J,∃ f : {e // part e=ij.1 ∧ split e=ij.2} → Fin l,
  60       ∀ e g,(hypergraphLineGraph (fun e : {e // part e=ij.1 ∧ split e=ij.2}=>R e.val)).Adj e g → f e≠f g := by
  61     intro ij
  62     apply hN D B hD hB V {e // part e=ij.1 ∧ split e=ij.2} (fun e=>R e.val) (fun e=>hsize e.val)
  63     · intro v
  64       have heq := subtype_filter_card_eq (fun e=>part e=ij.1 ∧ split e=ij.2) (fun e=>v∈R e)
  65       change (Finset.univ.filter (fun e : {e // part e=ij.1 ∧ split e=ij.2}=>v∈R e.val)).card≤D
  66       have heq' : (Finset.univ.filter (fun e : {e // part e=ij.1 ∧ split e=ij.2}=>v∈R e.val)).card=
  67           (Finset.univ.filter (fun e=>part e=ij.1 ∧ split e=ij.2 ∧ v∈R e)).card := by
  68         convert heq using 1 <;> congr 1 <;> ext e <;> simp only [Finset.mem_filter,Finset.mem_univ,true_and,and_assoc]
  69       rw [heq']
  70       convert hdegree ij.1 ij.2 v using 1 <;> congr 1 <;> ext e <;> simp only [Finset.mem_filter,and_assoc]
  71     · intro v u hvu
  72       have heq := subtype_filter_card_eq (fun e=>part e=ij.1 ∧ split e=ij.2) (fun e=>v∈R e ∧ u∈R e)
  73       have heq' : (Finset.univ.filter (fun e : {e // part e=ij.1 ∧ split e=ij.2}=>v∈R e.val ∧ u∈R e.val)).card=
  74           (Finset.univ.filter (fun e=>part e=ij.1 ∧ split e=ij.2 ∧ v∈R e ∧ u∈R e)).card := by
  75         convert heq using 1 <;> congr 1 <;> ext e <;> simp only [Finset.mem_filter,Finset.mem_univ,true_and,and_assoc]
  76       rw [heq']
  77       convert hcodeg ij.1 ij.2 v u hvu using 1 <;> congr 1 <;> ext e <;> simp only [Finset.mem_filter,and_assoc]
  78   choose χ hχ using hcells
  79   let color (e : E) : J×Fin l := (split e,χ (part e,split e) ⟨e,rfl,rfl⟩)
  80   have hcolor : ∀ e f,e≠f → part e=part f → color e=color f → Disjoint (R e) (R f) := by
  81     intro e f hef hp hc
  82     have hs : split e=split f := congrArg Prod.fst hc
  83     let e' : {g // part g=part e ∧ split g=split e} := ⟨e,rfl,rfl⟩
  84     let f' : {g // part g=part e ∧ split g=split e} := ⟨f,hp.symm,hs.symm⟩
  85     have htransport (i : I) (j : J) (g : E) (hpi : part g=i) (hsj : split g=j) :
  86         χ (part g,split g) ⟨g,rfl,rfl⟩=χ (i,j) ⟨g,hpi,hsj⟩ := by
  87       subst i;subst j;rfl
  88     have hχeq : χ (part e,split e) e'=χ (part e,split e) f' :=
  89       (congrArg Prod.snd hc).trans (htransport _ _ f hp.symm hs.symm)
  90     by_contra hn
  91     exact hχ (part e,split e) e' f' ⟨fun h=>hef (congrArg Subtype.val h),hn⟩ hχeq
  92   have hweight : ∀ a i c,colorClassWeight part color (w a) i c≤b a i := by
  93     intro a i c
  94     apply (Finset.sum_le_sum (fun e _=>show
  95       (if part e=i ∧ color e=c then w a e else 0)≤
  96       (if part e=i ∧ split e=c.1 then w a e else 0) from ?_)).trans (hb a i c.1)
  97     by_cases h : part e=i ∧ color e=c
  98     · have hh : part e=i ∧ split e=c.1 := ⟨h.1,congrArg Prod.fst h.2⟩
  99       rw [if_pos h,if_pos hh]
 100     · rw [if_neg h]
 101       split_ifs
 102       · exact hw a e
 103       · exact le_refl _
 104   obtain ⟨M,hM,hMw⟩ := colorClassLottery_matching R part color hparts hcolor
 105     Finset.univ w b t (fun a _=>hw a) (fun a _=>hweight a) (fun a _=>ht a)
 106     (by simpa using hsmall)
 107   refine ⟨M,hM,?_⟩
 108   intro a
 109   simpa only [Fintype.card_prod,Fintype.card_fin,Nat.cast_mul] using hMw a (Finset.mem_univ _)
