  15 def hypergraphColourClique {V E C : Type*} [Fintype E]
  16     (R : E → Finset V) (L : E → Finset C) (z : V) (c : C) : Finset E :=
  17   Finset.univ.filter (fun e=>z∈R e ∧ c∈L e)

  44     {V E C : Type*} [Fintype E] (R : E → Finset V) (L : E → Finset C)
  45     {k s : ℕ} (hsize : ∀ e,(R e).card≤k)
  46     (c : C) (hdegree : ∀ v,(hypergraphColourClique R L v c).card≤s) (e : E) :
  47     (colourNeighbourSet (hypergraphLineGraph R) L e c).card≤k*s := by
  48   classical
  49   have hsub : colourNeighbourSet (hypergraphLineGraph R) L e c⊆
  50       (R e).biUnion (fun v=>hypergraphColourClique R L v c) := by
  51     intro f hf
  52     obtain ⟨_,⟨_,hdis⟩,hc⟩ := Finset.mem_filter.mp hf
  53     obtain ⟨v,hve,hvf⟩ := Finset.not_disjoint_iff.mp hdis
  54     exact Finset.mem_biUnion.mpr ⟨v,hve,Finset.mem_filter.mpr ⟨Finset.mem_univ _,hvf,hc⟩⟩
  55   calc
  56     _ ≤ ((R e).biUnion (fun v=>hypergraphColourClique R L v c)).card := Finset.card_le_card hsub
  57     _ ≤ (R e).card*s := Finset.card_biUnion_le_card_mul _ _ _ (fun v _=>hdegree v)
