  21     (hRemAgree:∀i (x:{x // x∈(a i).remaining}),f i x.1=Φ i x)
  22     (hAnchorAgree:∀i x,x∈Anchors i → Φ i x=g i x)
  23     (hFree:∀i x,x∉Anchors i → Φ i x∈Lists i)
  24     (hOldAvoid:∀i x,x∈(a i).oldDomain → ψ i x∉A i)
  25     (hAnchorAvoid:∀i x,x∈Anchors i → g i x∉A i)
  26     (hListAvoid:∀i,Disjoint (Lists i) (A i))
  27     (C:Finset (Finset V⊕(V×I))) (Hold Hnew:Bool → V → I → Prop)
  28     (Y B:I → Finset V)
  29     (hMarked:∀v i,Hnew true v i ↔ v∈Y i)

  31     (hFixed:∀v i,Sum.inr (v,i)∈C → Hold true v i → v∈Y i)
  32     (hDisjoint:∀v i,Hold false v i → v∉Y i)
  33     (hLabel:∀v i,Sum.inr (v,i)∈C → Hold false v i → v∈A i)
  34     (hBAvoid:∀v i,Sum.inr (v,i)∈C → Hold false v i → v∉B i):
  35     (∀i,Disjoint (Finset.univ.image (f i)) (A i)) ∧
  36     (∀b v i,Sum.inr (v,i)∈C → Hold b v i → Hnew b v i) :=by
