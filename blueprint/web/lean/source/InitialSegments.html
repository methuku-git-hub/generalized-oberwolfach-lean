  11 def finiteInitialSegment {V : Type*} [Fintype V] (k : ℕ) : Finset V :=
  12   Finset.univ.filter (fun x => ((Fintype.equivFin V) x).val < k)

  14 def finiteInitialSegmentEquiv {V : Type*} [Fintype V] {k : ℕ}
  15     (hk : k ≤ Fintype.card V) : Fin k ≃ {x : V // x ∈ finiteInitialSegment k} where
  16   toFun i := ⟨(Fintype.equivFin V).symm ⟨i.val, lt_of_lt_of_le i.isLt hk⟩, by
  17     simp only [finiteInitialSegment, Finset.mem_filter, Finset.mem_univ, true_and,
  18       Equiv.apply_symm_apply]
  19     exact i.isLt⟩
  20   invFun x := ⟨((Fintype.equivFin V) x.1).val, (Finset.mem_filter.mp x.2).2⟩
  21   left_inv i := by apply Fin.ext; simp
  22   right_inv x := by apply Subtype.ext; simp

  24 theorem finiteInitialSegment_card {V : Type*} [Fintype V] {k : ℕ}
  25     (hk : k ≤ Fintype.card V) : (finiteInitialSegment (V := V) k).card = k := by
  26   have hh := Fintype.card_congr (finiteInitialSegmentEquiv (V := V) hk)
  27   simpa only [Fintype.card_fin, Fintype.card_coe] using hh.symm

  29 theorem finiteInitialSegment_mono {V : Type*} [Fintype V] {j k : ℕ} (hjk : j ≤ k) :
  30     finiteInitialSegment (V := V) j ⊆ finiteInitialSegment k := by
  31   intro x hx
  32   have hx' := (Finset.mem_filter.mp hx).2
  33   exact Finset.mem_filter.mpr ⟨Finset.mem_univ _, lt_of_lt_of_le hx' hjk⟩

  35 theorem finiteInitialSegment_full {V : Type*} [Fintype V] :
  36     finiteInitialSegment (V := V) (Fintype.card V) = Finset.univ := by
  37   ext x
  38   simp only [finiteInitialSegment, Finset.mem_filter, Finset.mem_univ, true_and]
  39   exact iff_true_intro ((Fintype.equivFin V) x).isLt

  41 theorem nested_finsets_of_sizes {V : Type*} [Fintype V] (n : ℕ → ℕ)
  42     (hzero : n 0 = Fintype.card V) (hmono : Antitone n) :
  43     ∃ B : ℕ → Finset V, B 0 = Finset.univ ∧
  44       (∀ i, (B i).card = n i) ∧ ∀ i, B (i + 1) ⊆ B i := by
  45   refine ⟨fun i => finiteInitialSegment (n i), ?_, ?_, ?_⟩
  46   · dsimp only
  47     rw [hzero, finiteInitialSegment_full]
  48   · intro i
  49     apply finiteInitialSegment_card
  50     rw [← hzero]
  51     exact hmono (Nat.zero_le _)
  52   · intro i
  53     exact finiteInitialSegment_mono (hmono (by omega))
