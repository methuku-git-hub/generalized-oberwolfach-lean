  18 theorem typedMarkedResource_product
  19     (W : MarkedWheel tail head ktail khead m π b H₀ Hₖ)
  20     (φ : (E⊕K)⊕(Bool×V×U)→ℝ) (p k s t : ℝ)
  21     (hR : ∀ i,φ (Sum.inl (markedCycleTypedEdge tail head ktail khead m π b W.1 i))=
  22       if i=Fin.last m then k else p)
  23     (hH : ∀ i,φ (Sum.inr (decide (i=0),W.1.val i,W.2.val))=if i=0 then t else s) :
  24     (∏ a∈typedMarkedOccurrenceResources W,φ a)=(p^m*k)*(t*s^m) := by
  25   have he : Finset.univ.filter (fun e=>CanonicalMarkedArcMem W.1 e)=
  26       Finset.univ.image (markedCycleTypedEdge tail head ktail khead m π b W.1) := by
  27     ext e;simp [CanonicalMarkedArcMem]
  28   have hdis : Disjoint
  29       ((Finset.univ.filter (fun e=>CanonicalMarkedArcMem W.1 e)).image (Sum.inl (β:=Bool×V×U)))
  30       (Finset.univ.image (fun i=>Sum.inr (α:=E⊕K) (decide (i=0),W.1.val i,W.2.val))) := by
  31     simp only [Finset.disjoint_left,Finset.mem_image,Finset.mem_filter,Finset.mem_univ,true_and]
  32     rintro a ⟨e,_,rfl⟩ ⟨i,hi⟩
  33     cases hi
  34   rw [typedMarkedOccurrenceResources,Finset.prod_union hdis,
  35     Finset.prod_image (fun _ _ _ _ h=>Sum.inl.inj h),he,
  36     Finset.prod_image (fun _ _ _ _ h=>markedCycleTypedEdge_injective tail head ktail khead m π b W.1 h),
  37     Finset.prod_image (fun i _ j _ h=>permutationSeparated_injective π b W.1.val W.1.property.1
  38       (congrArg (fun a : Bool×V×U=>a.2.1) (Sum.inr.inj h)))]
  39   simp_rw [hR,hH]
  40   rw [Fin.prod_univ_castSucc (fun i : Fin (m+1)=>if i=Fin.last m then k else p),
  41     Fin.prod_univ_succ (fun i : Fin (m+1)=>if i=0 then t else s)]
  42   simp

  49     (hH : ∀ i,φ (Sum.inr (decide (i=0),W.1.val i,W.2.val))=if i=0 then t else s)
  50     (root : (E⊕K)⊕(Bool×V×U)) (hroot : root∈typedMarkedOccurrenceResources W)
  51     (hφ : φ root≠0) :
  52     (∏ a∈typedMarkedOccurrenceResources W\{root},φ a)=(p^m*k)*(t*s^m)/φ root := by
