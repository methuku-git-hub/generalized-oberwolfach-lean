  15     (P:SimpleGraph V) (π:Equiv.Perm V) (d m:ℕ) (raw:RawMarkedOutcome V U Γ Color)
  16     (g:Γ) (j:Color):
  17     let state:=typedMarkedResourceState (rawMarkedPairRims P π d raw) (rawMarkedPairSpokes π d raw g j)
  18     let e:{z // z∈bernoulliVertexSet state}↪_:=Function.Embedding.subtype _
  19     let Parent:=MarkedWheel (Prod.fst:V×V→V) Prod.snd Prod.fst Prod.snd m π (3*d)
  20       (fun _ (_:U)=>True) (fun _ (_:U)=>True)
  21     ∀W:RestrictedResource (fun W:Parent=>typedMarkedOccurrenceResources W) e,
  22     let E:=MarkedSampleArc P π (3*d) (fun x=>raw (Sum.inl x)) false
  23     let K:=MarkedSampleArc P π (3*d) (fun x=>raw (Sum.inl x)) true
  24     let H:=rawMarkedPairSpokes π d raw g j
  25     ∃W':MarkedWheel (fun x:E=>x.val.1) (fun x:E=>x.val.2)
  26       (fun x:K=>x.val.1) (fun x:K=>x.val.2) m π (3*d) (H false) (H true),
