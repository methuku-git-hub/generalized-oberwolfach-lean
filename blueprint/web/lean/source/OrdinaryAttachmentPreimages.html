  13 theorem ordinaryExpansion_attachment_rim_preimage
  14     (G : SimpleGraph V) (C : SimpleGraph W) (z o : W) (x y : E → V) (v : J → V)
  15     (hG : ∀ a b,G.Adj a b → ∃ e,s(x e,y e)=s(a,b))
  16     (a b : ExpansionRow V E J W)
  17     (h : (ordinaryAttachmentRim G C z (expansionAttachmentRoots x v)).Adj a b) :
  18     ∃ a' b',(ordinaryExpansionRim C z o x y).Adj a' b' ∧
  19       attachmentCollapse z (expansionAttachmentRoots x v) a'=a ∧
  20       attachmentCollapse z (expansionAttachmentRoots x v) b'=b := by
  21   have hcross (a : V) (i : E ⊕ J) (w : W)
  22       (h : expansionAttachmentRoots x v i=a ∧ w≠z ∧ C.Adj z w) :
  23       ∃ a' b',(ordinaryExpansionRim C z o x y).Adj a' b' ∧
  24         attachmentCollapse z (expansionAttachmentRoots x v) a'=.inl a ∧
  25         attachmentCollapse z (expansionAttachmentRoots x v) b'=.inr (i,w) := by
  26     cases i with
  27     | inl e =>
  28       by_cases hwo : w=o
  29       · subst w
  30         refine ⟨.inl (x e),.inr (.inl e,o),Or.inl ⟨rfl,rfl⟩,?_,?_⟩
  31         · exact congrArg (Sum.inl (β:=(E⊕J)×W)) h.1
  32         · simp [attachmentCollapse,attachmentWheelEmbedding,h.2.1]
  33       · refine ⟨.inr (.inl e,z),.inr (.inl e,w),⟨rfl,h.2.2,?_⟩,?_,?_⟩
  34         · simp [hwo,h.2.1]
  35         · simpa [attachmentCollapse,attachmentWheelEmbedding,expansionAttachmentRoots] using congrArg (Sum.inl (β:=(E⊕J)×W)) h.1
  36         · simp [attachmentCollapse,attachmentWheelEmbedding,h.2.1]
  37     | inr j =>
  38       refine ⟨.inr (.inr j,z),.inr (.inr j,w),⟨rfl,h.2.2⟩,?_,?_⟩
  39       · simpa [attachmentCollapse,attachmentWheelEmbedding,expansionAttachmentRoots] using congrArg (Sum.inl (β:=(E⊕J)×W)) h.1
  40       · simp [attachmentCollapse,attachmentWheelEmbedding,h.2.1]
  41   cases a with
  42   | inl a =>
  43     cases b with
  44     | inl b =>
  45       obtain ⟨e,he⟩ := hG a b h
  46       rcases Sym2.eq_iff.mp he with ⟨ha,hb⟩|⟨hb,ha⟩
  47       · refine ⟨.inr (.inl e,z),.inl (y e),Or.inr ⟨rfl,rfl⟩,?_,?_⟩
  48         · simp [attachmentCollapse,attachmentWheelEmbedding,expansionAttachmentRoots,ha]
  49         · exact congrArg (Sum.inl (β:=(E⊕J)×W)) hb
  50       · refine ⟨.inl (y e),.inr (.inl e,z),Or.inr ⟨rfl,rfl⟩,?_,?_⟩
  51         · exact congrArg (Sum.inl (β:=(E⊕J)×W)) ha
  52         · simp [attachmentCollapse,attachmentWheelEmbedding,expansionAttachmentRoots,hb]
  53     | inr p => exact hcross a p.1 p.2 h
  54   | inr p =>
  55     cases b with
  56     | inl b =>
  57       obtain ⟨a',b',hab,ha,hb⟩ := hcross b p.1 p.2 h
  58       exact ⟨b',a',hab.symm,hb,ha⟩
  59     | inr q =>
  60       rcases p with ⟨i,w⟩;rcases q with ⟨j,w'⟩
  61       obtain ⟨hij,hw,hw',hC⟩ := h
  62       subst j
  63       refine ⟨.inr (i,w),.inr (i,w'),⟨rfl,?_⟩,?_,?_⟩
  64       · cases i with
  65         | inl e => exact ⟨hC,by simp [hw,hw']⟩
  66         | inr j => exact hC
  67       · simp [attachmentCollapse,attachmentWheelEmbedding,hw]
  68       · simp [attachmentCollapse,attachmentWheelEmbedding,hw']

  70 theorem ordinaryExpansion_attachment_spoke_preimage
  71     (H : V → U → Prop) (z : W) (x : E → V) (v : J → V) (u : J → U)
  72     (hH : ∀ a b,H a b → ∃ j,v j=a ∧ u j=b)
  73     (a : ExpansionRow V E J W) (b : ExpansionHub U E J)
  74     (h : ordinaryAttachmentSpokes H z (expansionAttachmentRoots x v) a b) :
  75     ∃ a',ordinaryExpansionSpokes z v u a' b ∧
  76       attachmentCollapse z (expansionAttachmentRoots x v) a'=a := by
  77   cases a with
  78   | inl a =>
  79     cases b with
  80     | inl b =>
  81       obtain ⟨j,hv,hu⟩ := hH a b h
  82       refine ⟨.inr (.inr j,z),⟨rfl,hu.symm⟩,?_⟩
  83       simp [attachmentCollapse,attachmentWheelEmbedding,expansionAttachmentRoots,hv]
  84     | inr i =>
  85       cases i with
  86       | inl e =>
  87         refine ⟨.inr (.inl e,z),rfl,?_⟩
  88         simpa [attachmentCollapse,attachmentWheelEmbedding,expansionAttachmentRoots] using congrArg (Sum.inl (β:=(E⊕J)×W)) h
  89       | inr j => exact ⟨.inl (v j),rfl,congrArg (Sum.inl (β:=(E⊕J)×W)) h⟩
  90   | inr p =>
  91     rcases p with ⟨i,w⟩
  92     cases b with
  93     | inl b => exact h.elim
  94     | inr j =>
  95       obtain ⟨hw,hij⟩ := h
  96       subst j
  97       refine ⟨.inr (i,w),?_,by simp [attachmentCollapse,attachmentWheelEmbedding,hw]⟩
  98       cases i with
  99       | inl e => rfl
 100       | inr j => exact ⟨hw,rfl⟩
