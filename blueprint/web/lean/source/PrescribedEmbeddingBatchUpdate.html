  16 theorem prescribed_embedding_batch_update (ps:List (X×V)):
  17     ∀ (A:Finset X) (f:X→V),
  18     (ps.map Prod.fst).Nodup→(ps.map Prod.snd).Nodup→
  19     (∀ p∈ps,p.1∉A)→(∀ p∈ps,p.2∉A.image f)→Set.InjOn f A→
  20     Set.InjOn (updatePairedImages f ps) (↑(A∪(ps.map Prod.fst).toFinset):Set X) ∧
  21     (∀ a∈A,updatePairedImages f ps a=f a) ∧
  22     (∀ p∈ps,updatePairedImages f ps p.1=p.2) := by
  23   induction ps with
  24   | nil =>
  25     intro A f ha hb hdom himg hinj
  26     refine ⟨?_,?_,?_⟩
  27     · simpa [updatePairedImages] using hinj
  28     · intro a ha; rfl
  29     · simp
  30   | cons p ps ih =>
  31     rcases p with ⟨x,z⟩
  32     intro A f hsrc htgt hdom himg hinj
  33     have hsrc':(ps.map Prod.fst).Nodup:=by simpa using (List.nodup_cons.mp (by simpa using hsrc)).2
  34     have htgt':(ps.map Prod.snd).Nodup:=by simpa using (List.nodup_cons.mp (by simpa using htgt)).2
  35     have hxnot:x∉ps.map Prod.fst:=(List.nodup_cons.mp (by simpa using hsrc)).1
  36     have hznot:z∉ps.map Prod.snd:=(List.nodup_cons.mp (by simpa using htgt)).1
  37     have hxA:x∉A:=hdom (x,z) (by simp)
  38     have hzA:z∉A.image f:=himg (x,z) (by simp)
  39     let f':=Function.update f x z
  40     let A':=insert x A
  41     have hinj':Set.InjOn f' A':=prescribed_embedding_injective_update A f x z hinj hzA
  42     have hp:=prescribed_embedding_update_preserves A f x z hxA
  43     have hdom':∀ q∈ps,q.1∉A':=by
  44       intro q hq
  45       have hqA:=hdom q (by simp [hq])
  46       have hqx:q.1≠x:=by
  47         intro he
  48         apply hxnot
  49         exact List.mem_map.mpr ⟨q,hq,he⟩
  50       simpa [A',hqA,hqx] 
  51     have himg':∀ q∈ps,q.2∉A'.image f':=by
  52       intro q hq
  53       have hqA:=himg q (by simp [hq])
  54       have hqz:q.2≠z:=by
  55         intro he
  56         apply hznot
  57         exact List.mem_map.mpr ⟨q,hq,he⟩
  58       rw [show A'.image f'=insert z (A.image f) from hp.2]
  59       simp [hqA,hqz]
  60     obtain ⟨hinj'',hold,hnew⟩:=ih A' f' hsrc' htgt' hdom' himg' hinj'
  61     have hsets:A'∪(ps.map Prod.fst).toFinset=A∪(((x,z)::ps).map Prod.fst).toFinset:=by
  62       ext a
  63       simp [A',or_assoc,or_left_comm,or_comm]
  64     constructor
  65     · change Set.InjOn (updatePairedImages f' ps) (↑(A∪(((x,z)::ps).map Prod.fst).toFinset):Set X)
  66       rw [←hsets]
  67       exact hinj''
  68     constructor
  69     · intro a ha
  70       change updatePairedImages f' ps a=f a
  71       have ha':a∈A':=by simp [A',ha]
  72       rw [hold a ha']
  73       exact hp.1 a ha
  74     · intro q hq
  75       change updatePairedImages f' ps q.1=q.2
  76       simp only [List.mem_cons] at hq
  77       rcases hq with rfl|hq
  78       · have hx':x∈A':=by simp [A']
  79         rw [hold x hx']
  80         simp [f']
  81       · exact hnew q hq
  82 end Oberwolfach
