   9 theorem odd_factor_caller_below_cycle_threshold
  10     {r:ℕ} (hr:3≤r) (hodd:Odd r) (β:ℝ)
  11     (hβ:β < 2/((r:ℝ)*(r-1))):
  12     cycleFactorDensity r+β/4<cycleThreshold r:=by
  13   have hre:(3:ℝ)≤r:=by exact_mod_cast hr
  14   have hne:¬Even r:=Nat.not_even_iff_odd.mpr hodd
  15   have h4:r≠4:=by
  16     intro h; subst r; norm_num at hodd
  17   have hlo:=cycleThreshold_ge_ordinaryCycleLowerDensity hr
  18   simp only [ordinaryCycleLowerDensity,h4,hne,ite_false] at hlo
  19   have hd1:(2*(r:ℝ)-2)≠0:=by linarith
  20   have hd2:(r:ℝ)≠0:=by linarith
  21   have hd3:(r:ℝ)-1≠0:=by linarith
  22   have hid:(1/(2*(r:ℝ)-2):ℝ)=1/(2*r)+1/(2*r*(r-1)):=by
  23     field_simp
  24     <;> ring
  25   have hid2:(2/((r:ℝ)*(r-1)))/4=1/(2*r*(r-1)):=by
  26     field_simp
  27     <;> ring
  28   simp only [cycleFactorDensity,hne,ite_false]
  29   linarith
  30 end Oberwolfach
