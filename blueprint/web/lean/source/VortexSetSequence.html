  13 theorem vortex_set_sequence {ε : ℝ} (he : 0 < ε) (he1 : ε < 1) :
  14     ∃ N : ℕ, 2 ≤ N ∧ ∀ (V : Type z) (B₀ : Finset V) (A : V → Finset V) (ℓ : ℕ),
  15       (∀ i < ℓ, N ≤ floorOrbit ε B₀.card i) →
  16       ∃ B : ℕ → Finset V, B 0 = B₀ ∧
  17         (∀ i ≤ ℓ, (B i).card = floorOrbit ε B₀.card i) ∧
  18         ∀ i < ℓ, B (i + 1) ⊆ B i ∧
  19           ∀ x ∈ B (i - 1),
  20             |((A x ∩ B (i + 1)).card : ℝ) / (B (i + 1)).card -
  21               ((A x ∩ B i).card : ℝ) / (B i).card| <
  22               ((B i).card : ℝ) ^ (-1 / 3 : ℝ) := by
  23   obtain ⟨N, hN⟩ := vortex_ambient_subset_step.{z} he he1.le
  24   refine ⟨max N 2, le_max_right _ _, ?_⟩
  25   intro V B₀ A ℓ
  26   induction ℓ with
  27   | zero =>
  28     intro _
  29     refine ⟨fun _ => B₀, rfl, ?_, ?_⟩
  30     · intro i hi
  31       have : i = 0 := by omega
  32       subst i
  33       rfl
  34     · intro i hi
  35       omega
  36   | succ ℓ ih =>
  37     intro hsize
  38     obtain ⟨B, hB₀, hcard, hstep⟩ := ih (fun i hi => hsize i (by omega))
  39     have hcurrent : max N 2 ≤ (B ℓ).card := by rw [hcard ℓ (by omega)]; exact hsize ℓ (by omega)
  40     have hcurrentReal : (1 : ℝ) ≤ (B ℓ).card := by exact_mod_cast (show 1 ≤ (B ℓ).card by omega)
  41     have hroot : ((B (ℓ - 1)).card : ℝ) ≤ 2 * (B ℓ).card / ε := by
  42       by_cases hz : ℓ = 0
  43       · subst ℓ
  44         simp only [Nat.zero_sub]
  45         apply (le_div_iff₀ he).mpr
  46         have hmul := mul_le_mul_of_nonneg_left he1.le (show (0 : ℝ) ≤ (B 0).card by positivity)
  47         nlinarith
  48       · have hsuc : ℓ - 1 + 1 = ℓ := by omega
  49         have hf := Nat.lt_floor_add_one (ε * (floorOrbit ε B₀.card (ℓ - 1) : ℝ))
  50         rw [← floorOrbit_succ, hsuc] at hf
  51         rw [← hcard (ℓ - 1) (by omega), ← hcard ℓ (by omega)] at hf
  52         apply (le_div_iff₀ he).mpr
  53         nlinarith
  54     obtain ⟨C, hCB, hc, hC⟩ := hN V (B ℓ) (B (ℓ - 1)) A (by omega) hroot
  55     let B' := Function.update B (ℓ + 1) C
  56     refine ⟨B', ?_, ?_, ?_⟩
  57     · simpa only [B', Function.update_of_ne (by omega : 0 ≠ ℓ + 1)] using hB₀
  58     · intro i hi
  59       by_cases hi' : i = ℓ + 1
  60       · subst i
  61         simp only [B', Function.update_self]
  62         rw [hc, hcard ℓ (by omega), floorOrbit_succ]
  63       · rw [show B' i = B i by simp [B', hi']]
  64         exact hcard i (by omega)
  65     · intro i hi
  66       by_cases hi' : i = ℓ
  67       · subst i
  68         have hcur : B' ℓ = B ℓ := by simp [B']
  69         have hnext : B' (ℓ + 1) = C := by simp [B']
  70         have hprev : B' (ℓ - 1) = B (ℓ - 1) := by
  71           apply Function.update_of_ne
  72           omega
  73         rw [hcur, hnext, hprev]
  74         exact ⟨hCB, hC⟩
  75       · have hcur : B' i = B i := by apply Function.update_of_ne; omega
  76         have hnext : B' (i + 1) = B (i + 1) := by apply Function.update_of_ne; omega
  77         have hprev : B' (i - 1) = B (i - 1) := by apply Function.update_of_ne; omega
  78         rw [hcur, hnext, hprev]
  79         exact hstep i (by omega)
