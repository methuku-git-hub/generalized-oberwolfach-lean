  11   obtain ⟨N,hN⟩:=exists_nat_gt (1/θ)
  12   let d:=max N (max d₀ 2)
  13   have hdN:N≤d:=le_max_left _ _
  14   have hdd:d₀≤d:=(le_max_left _ _).trans (le_max_right _ _)
  15   have hd2:2≤d:=(le_max_right _ _).trans (le_max_right _ _)
  16   have hdR:(0:ℝ)<d:=by exact_mod_cast (show 0<d by omega)
  17   have hNd:(N:ℝ)≤d:=by exact_mod_cast hdN
