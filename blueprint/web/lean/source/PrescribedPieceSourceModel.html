   4 /-! Construct the source graph, directed arcs and vertex set directly from
   5 the ordered path-and-cycle pieces. -/

  12 def prescribedSourceVertices (ps:List (I×PrescribedSourcePiece X)) (i:I):Finset X:=
  13   Finset.univ.filter (fun x=>∃ p∈ps,p.1=i ∧ x∈p.2.vertices)
  14 
  15 def prescribedSourcePhysical (ps:List (I×PrescribedSourcePiece X)) (i:I):Set (Sym2 X):=
  16   {e | ∃ p∈ps,p.1=i ∧ e∈p.2.edges}
  17 
  18 def prescribedSourceGraph (ps:List (I×PrescribedSourcePiece X)) (i:I):SimpleGraph X:=
  19   SimpleGraph.fromEdgeSet (prescribedSourcePhysical ps i)
  20 
  21 def prescribedSourceDigraph (ps:List (I×PrescribedSourcePiece X)) (i:I) (x y:X):Prop:=
  22   ∃ p∈ps,p.1=i ∧ (x,y)∈p.2.arcs

  24 theorem prescribed_source_model_vertices (ps:List (I×PrescribedSourcePiece X)) (i:I) (x:X):
  25     x∈prescribedSourceVertices ps i ↔ ∃ p∈ps,p.1=i ∧ x∈p.2.vertices := by
  26   simp [prescribedSourceVertices]

  28 theorem prescribed_source_model_edges (ps:List (I×PrescribedSourcePiece X)) (i:I) (e:Sym2 X)
  29     (he:e∈(prescribedSourceGraph ps i).edgeFinset):
  30     ∃ p∈ps,p.1=i ∧ e∈p.2.edges := by
  31   have hh:e∈(prescribedSourceGraph ps i).edgeSet:=SimpleGraph.mem_edgeFinset.mp he
  32   change e∈(SimpleGraph.fromEdgeSet (prescribedSourcePhysical ps i)).edgeSet at hh
  33   rw [SimpleGraph.edgeSet_fromEdgeSet] at hh
  34   exact hh.1

  36 theorem prescribed_source_model_piece_valid (ps:List (I×PrescribedSourcePiece X))
  37     (hshape:∀ p∈ps,p.2.Shape):∀ p∈ps,p.2.OriginalValid (prescribedSourceDigraph ps p.1) := by
  38   intro p hp
  39   cases hpiece:p.2 with
  40   | path xs =>
  41     have hn:xs.Nodup:=by simpa [hpiece,PrescribedSourcePiece.Shape] using hshape p hp
  42     refine ⟨hn,?_⟩
  43     apply (prescribed_directed_edges_chain (prescribedSourceDigraph ps p.1) xs).mpr
  44     intro e he
  45     refine ⟨p,hp,rfl,?_⟩
  46     simpa [hpiece,PrescribedSourcePiece.arcs] using he
  47   | cycle a xs =>
  48     have hh:(a::xs).Nodup ∧ 3≤(a::xs).length:=by simpa [hpiece,PrescribedSourcePiece.Shape] using hshape p hp
  49     refine ⟨hh.1,hh.2,?_⟩
  50     apply (prescribed_directed_edges_chain (prescribedSourceDigraph ps p.1) ((a::xs)++[a])).mpr
  51     intro e he
  52     refine ⟨p,hp,rfl,?_⟩
  53     simpa [hpiece,PrescribedSourcePiece.arcs] using he

  55 theorem prescribed_source_model_arc_graph (ps:List (I×PrescribedSourcePiece X))
  56     (hshape:∀ p∈ps,p.2.Shape):∀ i x y,prescribedSourceDigraph ps i x y→(prescribedSourceGraph ps i).Adj x y := by
  57   intro i x y hxy
  58   obtain ⟨p,hp,hpi,he⟩:=hxy
  59   apply (SimpleGraph.fromEdgeSet_adj _).mpr
  60   constructor
  61   · refine ⟨p,hp,hpi,?_⟩

  65     subst y
  66     exact prescribed_piece_no_loops p.2 (hshape p hp) x he
