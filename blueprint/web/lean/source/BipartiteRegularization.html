  13 theorem bipartite_matrix_random_regularization (β : ℝ) (hβ : 0 < β) :
  14     ∃ C η₀ : ℝ, 0 < C ∧ 0 < η₀ ∧ ∀ η : ℝ, 0 < η → η < η₀ →
  15       ∃ m₀ : ℕ, ∀ (V : Type) [Fintype V] [DecidableEq V], m₀ ≤ Fintype.card V →
  16       ∀ (M : V → V → ℕ), (∀ x y, M x y ≤ 1) → ∀ d : ℝ,
  17       (∀ x, (1 / 2 + β) * Fintype.card V ≤ ((∑ y, M x y : ℕ) : ℝ)) →
  18       (∀ y, (1 / 2 + β) * Fintype.card V ≤ ((∑ x, M x y : ℕ) : ℝ)) →
  19       (∀ x, d - η * Fintype.card V ≤ ((∑ y, M x y : ℕ) : ℝ) ∧
  20         ((∑ y, M x y : ℕ) : ℝ) ≤ d + η * Fintype.card V) →
  21       (∀ y, d - η * Fintype.card V ≤ ((∑ x, M x y : ℕ) : ℝ) ∧
  22         ((∑ x, M x y : ℕ) : ℝ) ≤ d + η * Fintype.card V) →
  23       let k := ⌊d - Real.sqrt η * Fintype.card V⌋₊
  24       ∃ ν : FiniteLaw (BipartiteEdges M → Bool),
  25         (∀ D, 0 < ν.mass D →
  26           (∀ x, (∑ y, retainedBipartiteMatrix M D x y) = k) ∧
  27           (∀ y, (∑ x, retainedBipartiteMatrix M D x y) = k)) ∧
  28         (∀ D x y, retainedBipartiteMatrix M D x y ≤ M x y) ∧
  29         (∀ e : BipartiteEdges M,
  30           ν.probability (fun D => retainedBipartiteMatrix M D e.1 e.2.1 = 0) ≤
  31             C * Real.sqrt η) := by
  32   obtain ⟨C, η₀, hC, hη₀, hall⟩ := matrix_regularization_deletion_law β hβ
  33   refine ⟨C, η₀, hC, hη₀, ?_⟩
  34   intro η hη hsmall
  35   obtain ⟨m₀, hm₀⟩ := hall η hη hsmall
  36   refine ⟨m₀, ?_⟩
  37   intro V _ _ hm M hM d hl hr hnl hnr
  38   obtain ⟨hkl, hkr, ν, hdegrees, hprob⟩ := hm₀ V hm M hM d hl hr hnl hnr
  39   refine ⟨ν, ?_, retainedBipartiteMatrix_le M, ?_⟩
  40   · intro D hD
  41     obtain ⟨hdleft, hdright⟩ := hdegrees D hD
  42     constructor
  43     · intro x
  44       rw [retainedBipartiteMatrix_row]
  45       exact selected_complement_degree (fun e : BipartiteEdges M => e.1 = x)
  46         D _ _ (hkl x) (bipartiteEdges_row_degree M x) (by
  47           convert hdleft x using 1
  48           apply Finset.sum_congr rfl
  49           intro e _
  50           split_ifs <;> rfl)
  51     · intro y
  52       rw [retainedBipartiteMatrix_column]
  53       exact selected_complement_degree (fun e : BipartiteEdges M => e.2.1 = y)
  54         D _ _ (hkr y) (bipartiteEdges_column_degree M y) (by
  55           convert hdright y using 1
  56           apply Finset.sum_congr rfl
  57           intro e _
  58           split_ifs <;> rfl)
  59   · intro e
  60     convert hprob e using 1
  61     congr 1
  62     ext D
  63     exact retainedBipartiteMatrix_edge M hM D e
