  10 theorem marked_row_packing_numeric_threshold (l : ℕ) (β A : ℝ) (hβ : 0<β) (hA : 0<A) :
  11     ∃ K : ℕ, 0<K ∧ ∀ b M : ℕ, ∃ n₀ : ℕ, ∀ n N : ℕ, n₀≤n →
  12       (N:ℝ)≤A*n → ∀ q : ℝ, q≤β*n/4 →
  13       q+(N*(l+4):ℕ)/(K:ℝ)+(l+5)*(2*b+1)+8*(K+2*M:ℕ)<β*n := by
  14   obtain ⟨K,hK⟩ := exists_nat_gt (max 1 (4*A*((l:ℝ)+4)/β))
  15   have hK1 : (1:ℝ)<K := (le_max_left _ _).trans_lt hK
  16   have hKpos : 0<K := by exact_mod_cast (show (0:ℝ)<K by linarith)
  17   have hKreal : (0:ℝ)<K := Nat.cast_pos.mpr hKpos
  18   have hKβ : 4*A*((l:ℝ)+4)<(K:ℝ)*β :=
  19     (div_lt_iff₀ hβ).mp ((le_max_right _ _).trans_lt hK)
  20   refine ⟨K,hKpos,?_⟩
  21   intro b M
  22   let C : ℝ := ((l:ℝ)+5)*(2*b+1)+8*(K+2*M:ℕ)
  23   obtain ⟨n₀,hn₀⟩ := exists_nat_gt (max 1 (4*C/β))
  24   refine ⟨n₀,?_⟩
  25   intro n N hn hN q hq
  26   have hnreal : (n₀:ℝ)≤n := Nat.cast_le.mpr hn
  27   have hnpos : (0:ℝ)<n := by
  28     have hh := (le_max_left _ _).trans_lt (hn₀.trans_le hnreal)
  29     linarith
  30   have hCn : C<β*n/4 := by
  31     have hh := (div_lt_iff₀ hβ).mp ((le_max_right _ _).trans_lt (hn₀.trans_le hnreal))
  32     nlinarith only [hh]
  33   have hdiv : (N*(l+4):ℕ)/(K:ℝ)≤β*n/4 := by
  34     apply (div_le_iff₀ hKreal).mpr
  35     have hnL := mul_le_mul_of_nonneg_right hN (by positivity : 0≤(l:ℝ)+4)
  36     have hkn := mul_lt_mul_of_pos_right hKβ hnpos
  37     push_cast
  38     nlinarith only [hnL,hkn]
  39   dsimp [C] at hCn
  40   nlinarith only [hq,hdiv,hCn,mul_pos hβ hnpos]
