  11 theorem finset_chain_subset {V : Type*} (B : ℕ → Finset V) (k : ℕ)
  12     (hstep : ∀ i < k, B (i + 1) ⊆ B i)
  13     {a b : ℕ} (hab : a ≤ b) (hb : b ≤ k) : B b ⊆ B a := by
  14   revert hb
  15   induction b, hab using Nat.le_induction with
  16   | base => exact fun _ => Finset.Subset.refl _
  17   | succ b hab ih =>
  18     intro hb
  19     exact (hstep b (by omega)).trans (ih (by omega))

  23     |f k - f 0| ≤ ∑ i ∈ Finset.range k, e i := by

  26   | succ k ih =>
  27     have hh := ih (fun i hi => hstep i (by omega))

  30     have ht : |f (k + 1) - f 0| ≤ |f (k + 1) - f k| + |f k - f 0| :=
  31       abs_sub_le _ _ _
