  15 theorem finiteLaw_two_coordinate_values
  16     {V C : Type*} [Fintype V] [Fintype C]
  17     (μ : V → FiniteLaw C) (u v : V) (hne : u≠v) (a b : C) :
  18     (FiniteLaw.pi μ).probability (fun σ=>σ u=a ∧ σ v=b)=(μ u).mass a*(μ v).mass b := by
  19   classical
  20   have hh := finiteLaw_disjoint_supported_events μ {u} {v} (by simpa using hne)
  21     (fun σ=>σ u=a) (fun σ=>σ v=b)
  22     (fun σ τ he=>by rw [he u (by simp)])
  23     (fun σ τ he=>by rw [he v (by simp)])
  24   rw [finiteLaw_pi_coordinate_probability,finiteLaw_pi_coordinate_probability] at hh
  25   exact hh

  27 theorem finite_list_colouring_of_colour_degree
  28     {V C : Type*} [Fintype V] [Fintype C]
  29     (G : SimpleGraph V) (L : V → Finset C) {l d : ℕ}
  30     (hd : 0<d) (hcard : ∀ v,(L v).card=l)
  31     (hdegree : ∀ v c,(colourNeighbourSet G L v c).card≤d)
  32     (hlarge : 16*d≤l) :
  33     ∃ f : V → C,(∀ v,f v∈L v) ∧ ∀ u v,G.Adj u v → f u≠f v := by
  34   classical
  35   have hl : 0<l := by omega
  36   have hL (v : V) : (L v).Nonempty := Finset.card_pos.mp (by rw [hcard];exact hl)
  37   let μ := fun v=>FiniteLaw.uniformOn (L v) (hL v)
  38   let A := fun (p : ListColourConflict G L) (σ : V → C)=>σ p.val.1=p.val.2.2 ∧ σ p.val.2.1=p.val.2.2
  39   have hs (p : ListColourConflict G L) (σ τ : V → C)
  40       (he : ∀ v∈listColourConflictSupport G L p,σ v=τ v) : A p σ ↔ A p τ := by
  41     dsimp only [A]
  42     rw [he p.val.1 (by simp [listColourConflictSupport]),
  43       he p.val.2.1 (by simp [listColourConflictSupport])]
  44   have hp (p : ListColourConflict G L) :
  45       (FiniteLaw.pi μ).probability (A p)≤1/(4*((2*(2*l*d) : ℕ) : ℝ)) := by
  46     have hc : G.Adj p.val.1 p.val.2.1 ∧ p.val.2.2∈L p.val.1 ∧ p.val.2.2∈L p.val.2.1 := by
  47       simpa only [listColourConflicts,Finset.mem_filter,Finset.mem_univ,true_and] using p.property
  48     have hne : p.val.1≠p.val.2.1 := hc.1.ne
  49     rw [show A p=(fun σ=>σ p.val.1=p.val.2.2 ∧ σ p.val.2.1=p.val.2.2) from rfl,
  50       finiteLaw_two_coordinate_values μ _ _ hne]
  51     simp only [μ,FiniteLaw.uniformOn,hc.2.1,hc.2.2,ite_true,hcard]
  52     have hdr : (0 : ℝ)<d := by exact_mod_cast hd
  53     have hlr : (0 : ℝ)<l := by exact_mod_cast hl
  54     have hnum : (16 : ℝ)*(d : ℝ)≤l := by exact_mod_cast hlarge
  55     calc
  56       _ = 1/((l : ℝ)*(l : ℝ)) := by ring
  57       _ ≤ _ := one_div_le_one_div_of_le (by positivity) (by
  58         norm_num only [Nat.cast_mul,Nat.cast_ofNat]
  59         nlinarith [mul_nonneg hlr.le (sub_nonneg.mpr hnum)])
  60   obtain ⟨f,hf,hgood⟩ := finiteLaw_coordinate_localLemma_degree μ A
  61     (listColourConflictSupport G L) hs (by norm_num : 0<2)
  62     (by positivity : 0<2*l*d) (listColourConflictSupport_size G L)
  63     (listColourConflictSupport_incidence G L hcard hdegree) hp
  64   have hlist (v : V) : f v∈L v := by
  65     have hh := finiteLaw_pi_positive_coordinates μ f hf v
  66     by_contra hn
  67     simp only [μ,FiniteLaw.uniformOn,hn,ite_false] at hh
  68     exact (lt_irrefl 0) hh
  69   refine ⟨f,hlist,?_⟩
  70   intro u v huv he
  71   let p : ListColourConflict G L := ⟨(u,v,f u),Finset.mem_filter.mpr
  72     ⟨Finset.mem_univ _,huv,hlist u,by simpa only [he] using hlist v⟩⟩
  73   exact hgood p ⟨rfl,he.symm⟩
