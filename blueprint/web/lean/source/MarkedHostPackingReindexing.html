  21     (ht:∀ a,tail' (e a)=tail a) (hh:∀ a,head' (e a)=head a)
  22     (hkt:∀ a,ktail' (k a)=ktail a) (hkh:∀ a,khead' (k a)=khead a)
  23     (M:Finset (MarkedWheel tail head ktail khead m π b H₀ HK))
  24     (hM:∀ Y∈M,∀ Z∈M,Y≠Z → Disjoint (markedWheelPhysicalResources Y) (markedWheelPhysicalResources Z)) :
  25     ∃ N:Finset (MarkedWheel tail' head' ktail' khead' m π b H₀ HK),
  26       (∀ Y∈N,∀ Z∈N,Y≠Z → Disjoint (markedWheelPhysicalResources Y) (markedWheelPhysicalResources Z)) ∧

  31       MarkedWheel tail' head' ktail' khead' m π b H₀ HK := markedWheelVertexEmbedding f g (fun _=>rfl) e k ht hh hkt hkh (fun _ _ h=>h) (fun _ _ h=>h)

  33     (fun _ _ h=>h) (fun _ _ h=>h) M hM,?_⟩

  41 theorem markedHostReindex_packing_remainder
  42     (tail head:E → V) (ktail khead:K → V) (tail' head':E' → V) (ktail' khead':K' → V)
  43     (H₀ HK:V → U → Prop) (π:Equiv.Perm V) (m b:ℕ) (e:E≃E') (k:K≃K')
  44     (ht:∀ a,tail' (e a)=tail a) (hh:∀ a,head' (e a)=head a)
  45     (hkt:∀ a,ktail' (k a)=ktail a) (hkh:∀ a,khead' (k a)=khead a)
  46     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  47     (hp':Function.Injective (markedPhysicalEdge tail' head' ktail' khead' π))
  48     (hH:∀ v u,¬(H₀ v u∧HK v u)) :
  49     let S:=Finset.univ.filter (fun s:V×U=>H₀ s.1 s.2∨HK s.1 s.2);
  50     MarkedRemainderDivisible tail' head' ktail' khead' HK Finset.univ S m →
  51     ∀ M:Finset (MarkedWheel tail head ktail khead m π b H₀ HK),
  52     (∀ Y∈M,∀ Z∈M,Y≠Z → Disjoint (markedWheelPhysicalResources Y) (markedWheelPhysicalResources Z)) →
  53     ∃ N:Finset (MarkedWheel tail' head' ktail' khead' m π b H₀ HK),
  54       (∀ Y∈N,∀ Z∈N,Y≠Z → Disjoint (markedWheelPhysicalResources Y) (markedWheelPhysicalResources Z)) ∧
  55       N.biUnion markedWheelPhysicalResources=M.biUnion markedWheelPhysicalResources ∧
  56       MarkedRemainderDivisible tail' head' ktail' khead' HK (Finset.univ\N.biUnion markedWheelArcs)
  57         (S\N.biUnion markedWheelSpokePairs) m ∧
  58       markedFiniteResources tail' head' ktail' khead' π (Finset.univ\N.biUnion markedWheelArcs) (S\N.biUnion markedWheelSpokePairs)=
  59         markedFiniteResources tail head ktail khead π (Finset.univ\M.biUnion markedWheelArcs) (S\M.biUnion markedWheelSpokePairs) := by
  60   intro S hd M hM
  61   obtain ⟨N,hN,hR⟩:=markedHostReindex_packing tail head ktail khead tail' head' ktail' khead' H₀ HK π m b e k ht hh hkt hkh M hM
  62   have hNA:N.biUnion markedWheelArcs⊆Finset.univ:=Finset.subset_univ _
  63   have hNS:N.biUnion markedWheelSpokePairs⊆S:=by
  64     intro s hs
  65     obtain ⟨W,_,hs⟩:=Finset.mem_biUnion.mp hs
  66     exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,markedWheelPhysicalResources_spoke_present W s.1 s.2 ((markedWheel_spoke_resource W s).mp hs)⟩
  67   refine ⟨N,hN,hR,marked_packing_remainder_divisible hH Finset.univ S hd N hN hNA hNS,?_⟩
  68   rw [markedFiniteResources_sdiff _ _ _ _ π hp',markedPackingResources_eq_finite,
  69     markedFiniteResources_sdiff _ _ _ _ π hp,markedPackingResources_eq_finite,hR]
  70   have hu:(Finset.univ:Finset (E⊕K)).image (e.toEmbedding.sumMap k.toEmbedding)=Finset.univ:=by
  71     apply Finset.image_univ_of_surjective
  72     intro a
  73     cases a with
  74     | inl a=>exact ⟨Sum.inl (e.symm a),by simp⟩
  75     | inr a=>exact ⟨Sum.inr (k.symm a),by simp⟩
  76   have heq:=markedHostReindex_resources tail head ktail khead tail' head' ktail' khead' π e k ht hh hkt hkh Finset.univ S
  77   rw [hu] at heq
