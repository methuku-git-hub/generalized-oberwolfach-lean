  11 theorem balanced_integer_half_degrees {V : Type*} [Fintype V] (g : V → ℕ)
  12     (htotal : Even (∑ v,g v)) :
  13     ∃ gp gm : V → ℕ,(∀ v,gp v+gm v=g v) ∧ (∑ v,gp v=∑ v,gm v) ∧
  14       ∀ v,|(gp v : ℤ)-(gm v : ℤ)|≤1 := by
  15   classical
  16   let O := Finset.univ.filter (fun v=>g v%2=1)
  17   have hdecomp : (∑ v,g v)=2*(∑ v,g v/2)+O.card := by
  18     have hv (v : V) : g v=2*(g v/2)+(if v∈O then 1 else 0) := by
  19       have hm := Nat.mod_lt (g v) (by omega : 0<2)
  20       simp only [O,Finset.mem_filter,Finset.mem_univ,true_and]
  21       split_ifs <;> omega
  22     calc
  23       _ = ∑ v,(2*(g v/2)+(if v∈O then 1 else 0)) := Finset.sum_congr rfl (fun v _=>hv v)
  24       _ = _ := by
  25         rw [Finset.sum_add_distrib,← Finset.mul_sum,← Finset.card_filter]
  26         have heq : Finset.univ.filter (fun v=>v∈O)=O := by ext v;simp
  27         rw [heq]
  28   obtain ⟨j,hj⟩ := htotal
  29   have hOeven : O.card%2=0 := by omega
  30   obtain ⟨S,hSO,hScard⟩ := Finset.exists_subset_card_eq (s:=O) (n:=O.card/2) (Nat.div_le_self _ _)
  31   let gp (v : V) := g v/2+if v∈S then 1 else 0
  32   have hgp (v : V) : gp v≤g v := by
  33     by_cases hv : v∈S
  34     · have hodd : g v%2=1 := (Finset.mem_filter.mp (hSO hv)).2
  35       dsimp [gp]
  36       rw [if_pos hv]
  37       omega
  38     · simp only [gp,hv,ite_false,add_zero]
  39       exact Nat.div_le_self _ _
  40   let gm (v : V) := g v-gp v
  41   have hsum : 2*(∑ v,gp v)=∑ v,g v := by
  42     have hs : (∑ v,gp v)=(∑ v,g v/2)+S.card := by
  43       simp only [gp,Finset.sum_add_distrib,← Finset.card_filter]
  44       have heq : Finset.univ.filter (fun v=>v∈S)=S := by ext v;simp
  45       rw [heq]
  46     rw [hs,hScard]
  47     omega
  48   refine ⟨gp,gm,fun v=>by dsimp [gm];exact Nat.add_sub_of_le (hgp v),?_,?_⟩
  49   · have hs := Finset.sum_tsub_distrib Finset.univ (fun v _=>hgp v)
  50     change (∑ v,gm v)=(∑ v,g v)-(∑ v,gp v) at hs
  51     omega
  52   · intro v
  53     have hm := Nat.mod_lt (g v) (by omega : 0<2)
  54     have h := hgp v
  55     rw [abs_le]
  56     dsimp [gm,gp]
  57     by_cases hv : v∈S
  58     · have hodd : g v%2=1 := (Finset.mem_filter.mp (hSO hv)).2
  59       simp only [hv,ite_true]
  60       constructor <;> omega
  61     · simp only [hv,ite_false,add_zero]
  62       constructor <;> omega

  64 theorem near_uniform_integer_demands {V : Type*} [Fintype V]
  65     (hn : 0<Fintype.card V) (T : ℕ) :
  66     ∃ f : V → ℕ,(∑ v,f v)=T ∧ ∀ v,|(f v : ℝ)-(T : ℝ)/Fintype.card V|≤1 := by
  67   classical
  68   let n := Fintype.card V
  69   have hm : T%n<n := Nat.mod_lt T hn
  70   obtain ⟨S,hS,hScard⟩ := Finset.exists_subset_card_eq (s:=Finset.univ (α:=V))
  71     (n:=T%n) (by simpa only [Finset.card_univ] using hm.le)
  72   let q := T/n
  73   let f (v : V) := q+if v∈S then 1 else 0
  74   have hsum : (∑ v,f v)=n*q+T%n := by
  75     simp only [f,Finset.sum_add_distrib,Finset.sum_const,Finset.card_univ,smul_eq_mul,← Finset.card_filter]
  76     change n*q+_=n*q+T%n
  77     congr 1
  78     rw [← hScard]
  79     congr 1
  80     ext v
  81     simp
  82   refine ⟨f,?_,?_⟩
  83   · rw [hsum]
  84     exact Nat.div_add_mod T n
  85   · intro v
  86     have hnR : (0 : ℝ)<n := by exact_mod_cast hn
  87     have hlo : (q : ℝ)≤(T : ℝ)/n := by
  88       apply (le_div_iff₀ hnR).mpr
  89       exact_mod_cast (Nat.div_mul_le_self T n)
  90     have hhi : (T : ℝ)/n≤q+1 := by
  91       apply (div_le_iff₀ hnR).mpr
  92       have hh : T≤(q+1)*n := by
  93         have hdiv := Nat.div_add_mod T n
  94         change n*q+T%n=T at hdiv
  95         nlinarith
  96       exact_mod_cast hh
  97     have hv : (q : ℝ)≤f v ∧ (f v : ℝ)≤q+1 := by
  98       dsimp [f]
  99       split_ifs <;> push_cast <;> constructor <;> linarith
 100     change |(f v : ℝ)-(T : ℝ)/n|≤1
 101     rw [abs_le]
 102     constructor <;> linarith [hv.1,hv.2]

 105 /-- A prescribed residue has a representative in every interval of r
 106 consecutive natural numbers. This chooses the hub deletion demands. -/
 107 theorem residue_in_integer_interval (base d r : ℕ) (hr : 0<r) :
 108     ∃ k : ℕ,base≤k ∧ k<base+r ∧ k%r=d%r := by
 109   let q := base/r
 110   have hbase : r*q+base%r=base := Nat.div_add_mod base r
 111   have hb : base%r<r := Nat.mod_lt base hr
 112   have hd : d%r<r := Nat.mod_lt d hr
 113   by_cases h : base%r≤d%r
 114   · refine ⟨r*q+d%r,by omega,by omega,?_⟩
 115     simp only [Nat.add_mod,Nat.mul_mod,Nat.mod_self,zero_mul,Nat.zero_mod,zero_add,Nat.mod_mod]
 116   · refine ⟨r*q+r+d%r,by omega,by omega,?_⟩
 117     simp only [Nat.add_mod,Nat.mul_mod,Nat.mod_self,zero_mul,Nat.zero_mod,zero_add,Nat.mod_mod]
