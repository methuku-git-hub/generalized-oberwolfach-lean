  16 def RetainedMarkedRows (P : SimpleGraph V) (π : Equiv.Perm V) (d h : ℕ)
  17     (A : Finset V) (p k : ℝ) (q : U→Γ→ℝ) (s : U→Color→ℝ) (g : Γ) (j : Color) (ε : ℝ)
  18     (σ : RawMarkedOutcome V U Γ Color) (g₀ : Γ) (t : ℕ)
  19     (K : StartRetentionOutcome (rawStartHubSet π d (fun e=>σ (Sum.inr e)) g₀) t) : Prop :=
  20   ∀ (S T : Finset (Bool×Bool×V)) (J : Finset (Bool×U)),
  21     S.card≤h → T.card≤h → J.card≤h → Set.InjOn (fun a : Bool×U=>a.2) J →
  22     |retainedMarkedRowCount P π d σ g₀ t K (markedRowDomain P π (3*d) A S T) S J g j-
  23       (∏ a∈S,if a.1 then k else p)*(∏ a∈J,if a.1 then q a.2 g else s a.2 j)*
  24         (markedRowDomain P π (3*d) A S T).card|<ε

  35 theorem retainedMarkedRows_of_raw (P : SimpleGraph V) (π : Equiv.Perm V) (d h : ℕ)
  36     (A : Finset V) (p k : ℝ) (q : U→Γ→ℝ) (s : U→Color→ℝ) (g : Γ) (j : Color) (ε ζ : ℝ)
  37     (hζ : 0≤ζ) (σ : RawMarkedOutcome V U Γ Color) (g₀ : Γ) (t : ℕ)
  38     (K : StartRetentionOutcome (rawStartHubSet π d (fun e=>σ (Sum.inr e)) g₀) t)
  39     (hraw : RawMarkedRows P π d h A p k q s g j ε σ)
  40     (hchange : ∀ u,((A.filter (fun x=>retainedRawFamilyState π d (fun e=>σ (Sum.inr e)) g₀ t K (x,u)≠
  41       rawIntervalFamilyState π d (fun e=>σ (Sum.inr e)) (x,u))).card:ℝ)≤ζ*A.card) :
  42     RetainedMarkedRows P π d h A p k q s g j (ε+h*ζ*A.card) σ g₀ t K := by
  43   intro S T J hS hT hJ hinj
  44   let C := markedRowDomain P π (3*d) A S T
  45   have hCA : C⊆A := Finset.filter_subset _ _
  46   have hc := retainedMarkedRowCount_change P π d σ g₀ t K A C hCA S J g j (ζ*A.card) hchange
  47   have hcard : (J.card:ℝ)≤h := by exact_mod_cast hJ
  48   have hb := mul_le_mul_of_nonneg_right hcard (show 0≤ζ*A.card by positivity)
  49   have he := hraw S T J hS hT hJ hinj
  50   have hh := (abs_sub_le (retainedMarkedRowCount P π d σ g₀ t K C S J g j)
  51     (rawMarkedRowCount P π d C S J g j σ)
  52     ((∏ a∈S,if a.1 then k else p)*(∏ a∈J,if a.1 then q a.2 g else s a.2 j)*C.card)).trans_lt
  53       (add_lt_add_of_le_of_lt (hc.trans hb) he)
  54   convert hh using 1 <;> ring
