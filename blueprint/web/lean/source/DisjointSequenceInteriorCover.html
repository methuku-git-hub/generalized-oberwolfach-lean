   9 theorem disjoint_sequences_before_last_in_reserved
  10     {V I:Type*} [DecidableEq V] [Fintype I]
  11     (n:I→ℕ) (f:∀i,Fin (n i+1)↪V) (R:Finset V)
  12     (hdis:∀i j,i≠j→Disjoint (Finset.univ.image (f i)) (Finset.univ.image (f j)))
  13     (hcover:Finset.univ.biUnion (fun i=>Finset.univ.image (f i)) =
  14       R ∪ Finset.univ.image (fun i=>f i (Fin.last (n i)))) :
  15     ∀i (p:Fin (n i+1)),p.val<n i→f i p∈R := by
  16   classical
  17   intro i p hp
  18   have hin:f i p∈Finset.univ.biUnion (fun i=>Finset.univ.image (f i)):=
  19     Finset.mem_biUnion.mpr ⟨i,Finset.mem_univ _,Finset.mem_image.mpr ⟨p,Finset.mem_univ _,rfl⟩⟩
  20   rw [hcover] at hin
  21   rcases Finset.mem_union.mp hin with hR|hEnd
  22   · exact hR
  23   · obtain ⟨j,_,heq⟩:=Finset.mem_image.mp hEnd
  24     by_cases hij:i=j
  25     · subst j
  26       have he:(Fin.last (n i):Fin (n i+1))=p:=(f i).injective heq
  27       have heval:n i=p.val:=congrArg Fin.val he
  28       omega
  29     · have hj:f i p∈Finset.univ.image (f j):=by
  30         exact Finset.mem_image.mpr ⟨Fin.last (n j),Finset.mem_univ _,heq⟩
  31       have hi:f i p∈Finset.univ.image (f i):=Finset.mem_image.mpr ⟨p,Finset.mem_univ _,rfl⟩
  32       exact (Finset.disjoint_left.mp (hdis i j hij) hi hj).elim
