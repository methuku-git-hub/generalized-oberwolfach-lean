  25 theorem commonMarkedPathConstraint_all (R : Bool → V → V → Prop) (H : Bool → V → U → Prop)
  26     (i : Fin (m+1)) (v v' : V) (u u' : U) (z : Fin m → V) :
  27     (∀ a,commonMarkedPathConstraint R H i v v' u u' a z) ↔
  28       (∀ j,R (decide (j=Fin.last m)) (Fin.insertNth (α:=fun _=>V) i v z j)
  29         (Fin.insertNth (α:=fun _=>V) i v z (finRotate (m+1) j))) ∧
  30       (∀ j,R (decide (j=Fin.last m)) (Fin.insertNth (α:=fun _=>V) i v' z j)
  31         (Fin.insertNth (α:=fun _=>V) i v' z (finRotate (m+1) j))) ∧
  32       ∀ k,H (markedPositionType (i.succAbove k)) (z k) u ∧
  33         H (markedPositionType (i.succAbove k)) (z k) u' := by
  34   constructor
  35   · intro h
  36     exact ⟨fun j=>h (Sum.inl (false,j)),fun j=>h (Sum.inl (true,j)),
  37       fun k=>⟨h (Sum.inr (false,k)),h (Sum.inr (true,k))⟩⟩
  38   · rintro ⟨h,h',hs⟩ a
  39     cases a with
  40     | inl s => rcases s with ⟨side,j⟩;cases side;exact h j;exact h' j
  41     | inr s => rcases s with ⟨side,k⟩;cases side;exact (hs k).1;exact (hs k).2
