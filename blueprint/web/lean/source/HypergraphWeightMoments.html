  15 def hypergraphWeightedDegree (R : E → Finset V) (w : E → ℝ) (v : V) : ℝ :=
  16   ∑ e,if v∈R e then w e else 0
  17 
  18 theorem hypergraphWeightedDegree_nonneg (R : E → Finset V) (w : E → ℝ)
  19     (hw : ∀ e,0≤w e) (v : V) : 0≤hypergraphWeightedDegree R w v := by
  20   unfold hypergraphWeightedDegree
  21   apply Finset.sum_nonneg
  22   intro e _
  23   split_ifs <;> first | exact hw e | exact le_refl _
  24 
  25 theorem hypergraphWeightedDegree_sum (R : E → Finset V) (w : E → ℝ) :
  26     (∑ v,hypergraphWeightedDegree R w v)=∑ e,(R e).card*w e := by
  27   unfold hypergraphWeightedDegree
  28   rw [Finset.sum_comm]
  29   apply Finset.sum_congr rfl
  30   intro e _
  31   rw [← Finset.sum_filter]
  32   have he : Finset.univ.filter (fun v=>v∈R e)=R e := by ext;simp
  33   rw [he,Finset.sum_const,nsmul_eq_mul]
  34 
  35 theorem hypergraphWeightedDegree_sum_uniform (R : E → Finset V) (w : E → ℝ)
  36     (k : ℕ) (hR : ∀ e,(R e).card=k) :
  37     (∑ v,hypergraphWeightedDegree R w v)=(k:ℝ)*∑ e,w e := by
  38   rw [hypergraphWeightedDegree_sum]
  39   simp only [hR,← Finset.mul_sum]

  41 theorem hypergraphWeightedDegree_square_sum (R : E → Finset V) (w : E → ℝ)
  42     (hw : ∀ e,0≤w e) (k : ℕ) (hR : ∀ e,(R e).card=k)
  43     (D : ℝ) (hD : ∀ v,hypergraphWeightedDegree R w v≤D) :
  44     (∑ v,(hypergraphWeightedDegree R w v)^2)≤D*k*∑ e,w e := by
  45   calc
  46     _ ≤ ∑ v,D*hypergraphWeightedDegree R w v := by
  47       apply Finset.sum_le_sum
  48       intro v _
  49       have h0 := hypergraphWeightedDegree_nonneg R w hw v
  50       have hh := mul_le_mul_of_nonneg_right (hD v) h0
  51       simpa only [pow_two] using hh
  52     _ = _ := by rw [← Finset.mul_sum,hypergraphWeightedDegree_sum_uniform R w k hR];ring
  53 
  54 theorem hypergraphWeightedDegree_square_sum_pos (R : E → Finset V) (w : E → ℝ)
  55     (hw : ∀ e,0≤w e) (k : ℕ) (hk : 0<k) (hR : ∀ e,(R e).card=k)
  56     (hW : 0<∑ e,w e) : 0<∑ v,(hypergraphWeightedDegree R w v)^2 := by
  57   have hs : 0<∑ v,hypergraphWeightedDegree R w v := by
  58     rw [hypergraphWeightedDegree_sum_uniform R w k hR]
  59     exact mul_pos (Nat.cast_pos.mpr hk) hW
  60   obtain ⟨v,hv,hpos⟩ := (Finset.sum_pos_iff_of_nonneg (fun v _=>hypergraphWeightedDegree_nonneg R w hw v)).mp hs
  61   exact (sq_pos_of_pos hpos).trans_le (Finset.single_le_sum (fun v _=>sq_nonneg _) hv)

  63 theorem hypergraphWeightedDegree_card_bound (R : E → Finset V) (w : E → ℝ)
  64     (b : ℝ) (hb : ∀ e,w e≤b) (v : V) :
  65     hypergraphWeightedDegree R w v≤b*(Finset.univ.filter (fun e=>v∈R e)).card := by
