  28         coinRetainedColour G σ e≠some c)-

  30       (2 : ℝ)^S.card*((S.card^2*(2*k^2+k+2)*B : ℕ) : ℝ)/(l : ℝ) := by

  35     (equalizationCoin G L l (k*s) e)
  36     (equalizationCoin_bounds G L (by omega : 0<l) (k*s) e))
