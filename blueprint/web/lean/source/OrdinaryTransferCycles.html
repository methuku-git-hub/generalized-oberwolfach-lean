  14 theorem cycleGraph_support_eq_univ {r : ℕ} (hr : 3≤r) :
  15     (SimpleGraph.cycleGraph r).support=Set.univ := by
  16   apply Set.eq_univ_of_forall
  17   intro i
  18   by_contra hi
  19   have hz := ((SimpleGraph.cycleGraph r).degree_eq_zero_iff_notMem_support i).mpr hi
  20   have hd := cycleGraph_regular hr i
  21   simp only [← SimpleGraph.ncard_neighborSet] at hz hd
  22   omega

  28 def transferBaseEmbedding {V : Type*} {r : ℕ} (hr : 0<r)
  29     (f : (Fin r × Fin r) ↪ V) : Fin r ↪ V where
  30   toFun i := f (i,⟨0,hr⟩)
  31   inj' := by intro i j h;exact congrArg Prod.fst (f.injective h)

  33 def transferRowEmbedding {V : Type*} {r : ℕ}
  34     (f : (Fin r × Fin r) ↪ V) (i : Fin r) : Fin r ↪ V where
  35   toFun j := f (i,j)
  36   inj' := by intro j k h;exact congrArg Prod.snd (f.injective h)

  38 def transferRootedEmbedding {V : Type*} {r : ℕ} (hr : 0<r)
  39     (f : (Fin r × Fin r) ↪ V) (v : V) (hv : v∉Set.range f) (i : Fin r) : Fin r ↪ V where
  40   toFun j := if j=⟨0,hr⟩ then v else f (i,j)
  41   inj' := by
  42     intro j k h
  43     by_cases hj : j=⟨0,hr⟩ <;> by_cases hk : k=⟨0,hr⟩
  44     · exact hj.trans hk.symm
  45     · simp only [hj,ite_true,hk,ite_false] at h
  46       exact False.elim (hv ⟨(i,k),h.symm⟩)
  47     · simp only [hj,ite_false,hk,ite_true] at h
  48       exact False.elim (hv ⟨(i,j),h⟩)
  49     · simp only [hj,hk,ite_false] at h
  50       exact congrArg Prod.snd (f.injective h)

  52 theorem transfer_rooted_range_difference {V : Type*} {r : ℕ} (hr : 0<r)
  53     (f : (Fin r × Fin r) ↪ V) (v : V) (hv : v∉Set.range f) (i : Fin r) (z : V) :
  54     (if z∈Set.range (transferRootedEmbedding hr f v hv i) then (1:ℝ) else 0)-
  55       (if z∈Set.range (transferRowEmbedding f i) then (1:ℝ) else 0)=
  56       (if z=v then 1 else 0)-(if z=f (i,⟨0,hr⟩) then 1 else 0) := by
  57   have h0 : f (i,⟨0,hr⟩)≠v := fun h => hv ⟨(i,⟨0,hr⟩),h⟩
  58   by_cases hzv : z=v
  59   · subst z
  60     have hin : v∈Set.range (transferRootedEmbedding hr f v hv i) :=
  61       ⟨⟨0,hr⟩,by simp [transferRootedEmbedding]⟩
  62     have hout : v∉Set.range (transferRowEmbedding f i) := by
  63       rintro ⟨j,hj⟩
  64       exact hv ⟨(i,j),hj⟩
  65     simp [hin,hout,h0.symm]
  66   · by_cases hz0 : z=f (i,⟨0,hr⟩)
  67     · subst z
  68       have hin : f (i,⟨0,hr⟩)∈Set.range (transferRowEmbedding f i) := ⟨⟨0,hr⟩,rfl⟩
  69       have hout : f (i,⟨0,hr⟩)∉Set.range (transferRootedEmbedding hr f v hv i) := by
  70         rintro ⟨j,hj⟩
  71         change (if j=⟨0,hr⟩ then v else f (i,j))=f (i,⟨0,hr⟩) at hj
  72         split_ifs at hj with hj0
  73         · exact h0 hj.symm
  74         · exact hj0 (congrArg Prod.snd (f.injective hj))
  75       simp [hin,hout,h0]
  76     · have he : z∈Set.range (transferRootedEmbedding hr f v hv i) ↔
  77           z∈Set.range (transferRowEmbedding f i) := by
  78         constructor
  79         · rintro ⟨j,hj⟩
  80           change (if j=⟨0,hr⟩ then v else f (i,j))=z at hj
  81           split_ifs at hj with hj0
  82           · exact False.elim (hzv hj.symm)
  83           · exact ⟨j,hj⟩
  84         · rintro ⟨j,hj⟩
  85           have hj0 : j≠⟨0,hr⟩ := by intro hh;subst j;exact hz0 hj.symm
  86           exact ⟨j,by simpa only [transferRootedEmbedding,transferRowEmbedding,Function.Embedding.coeFn_mk,if_neg hj0] using hj⟩
  87       simp only [he,hzv,hz0,ite_false,sub_self]
