   9 theorem completed_factor_physical_edges_subset_union
  10     {X Y:Type*} [DecidableEq X] (G:SimpleGraph X) (A D:Finset X) (hDA:D⊆A)
  11     (f:X→Y) (fA:{x // x∈A}→Y) (fC:{x // x∉D}→Y)
  12     (hA:∀x (hx:x∈A),f x=fA ⟨x,hx⟩)
  13     (hC:∀x (hx:x∉D),f x=fC ⟨x,hx⟩)
  14     (hneigh:∀x,x∈D→∀y,G.Adj x y→y∈A) :
  15     partialFactorPhysicalEdgesPoly X G f ⊆
  16       partialFactorPhysicalEdgesPoly {x // x∈A} (G.induce (↑A:Set X)) fA ∪
  17       partialFactorPhysicalEdgesPoly {x // x∉D} (G.induce {x | x∉D}) fC := by
  18   intro e he
  19   obtain ⟨x,y,hxy,rfl⟩:=he
  20   by_cases hx:x∈D
  21   · have hxA:=hDA hx
  22     have hyA:=hneigh x hx y hxy
  23     left
  24     refine ⟨⟨x,hxA⟩,⟨y,hyA⟩,hxy,?_⟩
  25     rw [hA x hxA,hA y hyA]
  26   · by_cases hy:y∈D
  27     · have hxA:=hneigh y hy x hxy.symm
  28       have hyA:=hDA hy
  29       left
  30       refine ⟨⟨x,hxA⟩,⟨y,hyA⟩,hxy,?_⟩
  31       rw [hA x hxA,hA y hyA]
  32     · right
  33       refine ⟨⟨x,hx⟩,⟨y,hy⟩,hxy,?_⟩
  34       rw [hC x hx,hC y hy]

  36 theorem completed_factor_family_physical_edges_disjoint
  37     {U X Y:Type*} [DecidableEq X]
  38     (G:U→SimpleGraph X) (A D:U→Finset X) (hDA:∀u,D u⊆A u)
  39     (f:U→X→Y) (fA:∀u,{x // x∈A u}→Y) (fC:∀u,{x // x∉D u}→Y)
  40     (hA:∀u x (hx:x∈A u),f u x=fA u ⟨x,hx⟩)
  41     (hC:∀u x (hx:x∉D u),f u x=fC u ⟨x,hx⟩)
  42     (hneigh:∀u x,x∈D u→∀y,(G u).Adj x y→y∈A u)
  43     (hAA:∀u v,u≠v→Disjoint
  44       (partialFactorPhysicalEdgesPoly {x // x∈A u} ((G u).induce (↑(A u):Set X)) (fA u))
  45       (partialFactorPhysicalEdgesPoly {x // x∈A v} ((G v).induce (↑(A v):Set X)) (fA v)))
  46     (hAC:∀u v,u≠v→Disjoint
  47       (partialFactorPhysicalEdgesPoly {x // x∈A u} ((G u).induce (↑(A u):Set X)) (fA u))
  48       (partialFactorPhysicalEdgesPoly {x // x∉D v} ((G v).induce {x | x∉D v}) (fC v)))
  49     (hCC:∀u v,u≠v→Disjoint
  50       (partialFactorPhysicalEdgesPoly {x // x∉D u} ((G u).induce {x | x∉D u}) (fC u))
  51       (partialFactorPhysicalEdgesPoly {x // x∉D v} ((G v).induce {x | x∉D v}) (fC v))) :
  52     ∀u v,u≠v→Disjoint (partialFactorPhysicalEdgesPoly X (G u) (f u))
  53       (partialFactorPhysicalEdgesPoly X (G v) (f v)) := by
  54   intro u v huv
  55   apply Set.disjoint_left.mpr
  56   intro e heu hev
  57   have hu:=completed_factor_physical_edges_subset_union (G u) (A u) (D u) (hDA u) (f u) (fA u) (fC u) (hA u) (hC u) (hneigh u) heu
  58   have hv:=completed_factor_physical_edges_subset_union (G v) (A v) (D v) (hDA v) (f v) (fA v) (fC v) (hA v) (hC v) (hneigh v) hev
  59   rcases hu with hu|hu <;> rcases hv with hv|hv
  60   · exact (Set.disjoint_left.mp (hAA u v huv)) hu hv
  61   · exact (Set.disjoint_left.mp (hAC u v huv)) hu hv
  62   · exact (Set.disjoint_left.mp (hAC v u (Ne.symm huv))) hv hu
  63   · exact (Set.disjoint_left.mp (hCC u v huv)) hu hv
  64 end Oberwolfach
