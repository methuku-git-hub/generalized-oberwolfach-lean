   9 theorem log_one_add_lower_quadratic {x : ℝ} (hx : 0 ≤ x) :
  10     x - x ^ 2 / 2 ≤ Real.log (1 + x) := by
  11   let F (y : ℝ) := Real.log (1 + y) - y + y ^ 2 / 2
  12   have hd (y : ℝ) (hy : y ∈ Icc 0 x) :
  13       HasDerivAt F (1 / (1 + y) - 1 + y) y := by
  14     have hn : 1 + y ≠ 0 := ne_of_gt (by linarith [hy.1])
  15     convert ((((hasDerivAt_id y).const_add 1).log hn).sub (hasDerivAt_id y)).add
  16       (((hasDerivAt_id y).pow 2).div_const 2) using 1
  17     · rfl
  18     · dsimp
  19       ring
  20   have hm : MonotoneOn F (Icc 0 x) := by
  21     apply monotoneOn_of_hasDerivWithinAt_nonneg (convex_Icc 0 x)
  22       (fun y hy => (hd y hy).continuousAt.continuousWithinAt)
  23       (fun y hy => (hd y (interior_subset hy)).hasDerivWithinAt)
  24     intro y hy
  25     have hy0 : 0 ≤ y := (interior_subset hy).1
  26     have hn : 0 < 1 + y := by linarith
  27     have he : 1 / (1 + y) - 1 + y = y ^ 2 / (1 + y) := by field_simp; ring
  28     rw [he]
  29     positivity
  30   have hh := hm ⟨le_rfl, hx⟩ ⟨hx,le_rfl⟩ hx
  31   simp only [F, add_zero, Real.log_one, sub_zero, zero_pow (by decide : 2 ≠ 0),
  32     zero_div] at hh
  33   linarith

  35 theorem chernoff_upper_entropy {x : ℝ} (hx : 0 ≤ x) (hx1 : x ≤ 1) :
  36     x ^ 2 / 3 ≤ (1 + x) * Real.log (1 + x) - x := by

  38   have hd (y : ℝ) (hy : y ∈ Icc 0 x) :
  39       HasDerivAt F (Real.log (1 + y) - y + y ^ 2 / 2) y := by
  40     have hn : 1 + y ≠ 0 := ne_of_gt (by linarith [hy.1])
  41     convert (((((hasDerivAt_id y).const_add 1).mul
  42       (((hasDerivAt_id y).const_add 1).log hn)).sub (hasDerivAt_id y)).sub
  43       (((hasDerivAt_id y).pow 2).div_const 2)).add
  44       (((hasDerivAt_id y).pow 3).div_const 6) using 1
  45     · rfl
  46     · dsimp
  47       field_simp [hn]
  48       ring
  49   have hm : MonotoneOn F (Icc 0 x) := by
  50     apply monotoneOn_of_hasDerivWithinAt_nonneg (convex_Icc 0 x)
  51       (fun y hy => (hd y hy).continuousAt.continuousWithinAt)
  52       (fun y hy => (hd y (interior_subset hy)).hasDerivWithinAt)
  53     intro y hy
  54     have h := log_one_add_lower_quadratic (interior_subset hy).1
  55     linarith
  56   have hh := hm ⟨le_rfl,hx⟩ ⟨hx,le_rfl⟩ hx
  57   have hc : x ^ 3 ≤ x ^ 2 := by nlinarith [mul_nonneg (sq_nonneg x) (sub_nonneg.mpr hx1)]
