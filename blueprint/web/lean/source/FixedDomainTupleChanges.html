  17     (hP : ∀ f,P f → ∀ i,f i∈A) :
  18     {f : I→V // P f} ≃ {f : I→A // P (fun i=>(f i).val)} where

  24 theorem fixedDomainTuple_predicate_loss (A : Finset V) (P Q : (I→V)→Prop)
  25     (hP : ∀ f,P f → ∀ i,f i∈A) (hQ : ∀ f,Q f → ∀ i,f i∈A)
  26     (D : I→Finset V) (δ : ℝ) (hD : ∀ i,((D i).card:ℝ)≤δ)
  27     (hbad : ∀ f,P f → ¬Q f → ∃ i,f i∈D i) :
  28     (Fintype.card {f : I→V // P f}:ℝ)-Fintype.card {f : I→V // Q f}≤
  29       Fintype.card I*δ*(A.card:ℝ)^(Fintype.card I-1) := by
  30   rw [Fintype.card_congr (fixedDomainTuplePredicateEquiv A P hP),
  31     Fintype.card_congr (fixedDomainTuplePredicateEquiv A Q hQ)]
  32   have hh := finite_predicate_loss_union
  33     (fun f : I→A=>P (fun i=>(f i).val)) (fun f : I→A=>Q (fun i=>(f i).val))
  34     (fun i (f : I→A)=>(f i).val∈D i) (by intro f hp hq;exact hbad _ hp hq)
  35   apply hh.trans
  36   calc
  37     _ ≤ ∑ _i : I,δ*(Fintype.card A:ℝ)^(Fintype.card I-1) := by
  38       apply Finset.sum_le_sum
  39       intro i hi
  40       apply tuple_unary_deletion_count (fun x : A=>x.val∈D i) i δ
  41       calc
  42         _ ≤ ∑ x : V,if x∈D i then (1:ℝ) else 0 := by

  44             (fun x=>if x∈D i then (1:ℝ) else 0) (fun _=>by split_ifs <;> norm_num)
  45           convert hs using 1 <;> apply Finset.sum_congr rfl <;> intro x hx <;> split_ifs <;> rfl
