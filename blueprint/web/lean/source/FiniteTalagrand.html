  17 theorem finiteLaw_convexDistance_mgf
  18     {I Ω : Type*} [Fintype I] [Fintype Ω]
  19     (μ : I → FiniteLaw Ω) (A : Finset (I → Ω))
  20     (hA : 0<(FiniteLaw.pi μ).probability (fun x=>x∈A)) :
  21     (FiniteLaw.pi μ).expectation (fun x=>Real.exp (convexDistanceSq A x/4))≤
  22       ((FiniteLaw.pi μ).probability (fun x=>x∈A))⁻¹ := by
  23   classical
  24   revert μ A
  25   refine Fintype.induction_empty_option
  26     (P:=fun J _=>∀ (ν : J → FiniteLaw Ω) (B : Finset (J → Ω)),
  27       0<(FiniteLaw.pi ν).probability (fun x=>x∈B) →
  28       (FiniteLaw.pi ν).expectation (fun x=>Real.exp (convexDistanceSq B x/4))≤
  29         ((FiniteLaw.pi ν).probability (fun x=>x∈B))⁻¹) ?_ ?_ ?_ I
  30   · intro J K _ e ih ν B hB
  31     letI : Fintype J := Fintype.ofEquiv K e.symm
  32     exact finiteLaw_convexDistance_reindex e ν (ih (fun i=>ν (e i))) B hB
  33   · intro ν B hB
  34     obtain ⟨x,hx⟩ := finiteLaw_positive_finset_nonempty (FiniteLaw.pi ν) B hB
  35     have hall (y : PEmpty → Ω) : y∈B := (Subsingleton.elim x y) ▸ hx
  36     have hd (y : PEmpty → Ω) : convexDistanceSq B y=0 :=
  37       convexDistanceSq_mem B (hall y)
  38     simp only [hd,zero_div,Real.exp_zero]
  39     rw [finiteLaw_expectation_const]
  40     simp [FiniteLaw.probability,FiniteLaw.pi,hall]
  41   · intro J _ ih ν B hB
  42     have he : (fun a b : Option J=>Classical.propDecidable (a=b))=
  43         (@Option.instDecidableEq J (Classical.decEq J)) := Subsingleton.elim _ _
  44     rw [he] at hB ⊢
  45     exact finiteLaw_convexDistance_product_step ν (ih (fun i=>ν (some i))) B hB

  47 theorem finiteLaw_convexDistance_tail
  48     {I Ω : Type*} [Fintype I] [Fintype Ω]
  49     (μ : I → FiniteLaw Ω) (A : Finset (I → Ω)) (t : ℝ) :
  50     (FiniteLaw.pi μ).probability (fun x=>x∈A)*
  51       (FiniteLaw.pi μ).probability (fun x=>t≤convexDistanceSq A x)≤Real.exp (-t/4) := by
  52   classical
  53   let p := (FiniteLaw.pi μ).probability (fun x=>x∈A)
  54   have hp0 : 0≤p := finiteLaw_probability_nonneg _ _
  55   by_cases hp : 0<p
  56   · have hm := finiteLaw_convexDistance_mgf μ A hp
  57     have hm' : (FiniteLaw.pi μ).expectation (fun x=>Real.exp ((1/4)*convexDistanceSq A x))≤p⁻¹ := by
  58       simpa only [one_div,div_eq_mul_inv,one_mul,mul_comm] using hm
  59     have ht := finiteLaw_exponential_markov (FiniteLaw.pi μ) (convexDistanceSq A) t
  60       (s:=(1/4 : ℝ)) (by norm_num)
  61     have he : -(1/4 : ℝ)*t=-t/4 := by ring
  62     rw [he] at ht
  63     calc
  64       _ ≤ p*(Real.exp (-t/4)*(FiniteLaw.pi μ).expectation
  65           (fun x=>Real.exp ((1/4)*convexDistanceSq A x))) :=
  66         mul_le_mul_of_nonneg_left ht hp.le
  67       _ ≤ p*(Real.exp (-t/4)*p⁻¹) :=
  68         mul_le_mul_of_nonneg_left
  69           (mul_le_mul_of_nonneg_left hm' (Real.exp_pos _).le) hp.le
  70       _ = _ := by rw [mul_left_comm,mul_inv_cancel₀ hp.ne',mul_one]
  71   · have he : p=0 := le_antisymm (le_of_not_gt hp) hp0
  72     change p*_≤_
  73     rw [he,zero_mul]
  74     exact (Real.exp_pos _).le
