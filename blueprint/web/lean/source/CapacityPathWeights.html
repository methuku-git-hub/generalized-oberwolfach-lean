  12 def finiteIntervalTransition {X : Type*} {k : ℕ}
  13     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ) (i : ℕ) (x y : X) : ℝ :=
  14   if hi : i < k then a ⟨i,hi⟩ x y / d ⟨i,by omega⟩ x else identityTransition x y

  16 def intervalDegreeExtend {X : Type*} {k : ℕ}
  17     (d : Fin (k+1) → X → ℝ) (i : ℕ) : X → ℝ :=
  18   if hi : i < k+1 then d ⟨i,hi⟩ else d 0

  20 theorem finiteIntervalTransition_isTransition {X : Type*} [Fintype X] {k : ℕ}
  21     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ)
  22     (ha : ∀ j x y, 0 ≤ a j x y) (hd : ∀ j x, 0 < d j x)
  23     (hrow : ∀ j x, ∑ y, a j x y=d j.castSucc x) :
  24     ∀ i, IsTransition (finiteIntervalTransition a d i) := by
  25   intro i
  26   by_cases hi : i < k
  27   · constructor
  28     · intro x y
  29       simp only [finiteIntervalTransition, dif_pos hi]
  30       exact div_nonneg (ha _ _ _) (hd _ _).le
  31     · intro x
  32       simp only [finiteIntervalTransition,dif_pos hi,← Finset.sum_div,hrow]
  33       exact div_self (hd _ _).ne'
  34   · have heq : finiteIntervalTransition a d i=identityTransition := by
  35       funext x y
  36       simp only [finiteIntervalTransition,dif_neg hi]
  37     rw [heq]
  38     exact identityTransition_isTransition

  40 theorem intervalDegreeExtend_balance {X : Type*} [Fintype X] {k : ℕ}
  41     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ)
  42     (hd : ∀ j x, 0 < d j x)
  43     (hcol : ∀ j y, ∑ x, a j x y=d j.succ y) :
  44     ∀ i < k, transitionApply (intervalDegreeExtend d i) (finiteIntervalTransition a d i)=
  45       intervalDegreeExtend d (i+1) := by
  46   intro i hi
  47   funext y
  48   have hi' : i < k+1 := by omega
  49   have hi'' : i+1 < k+1 := by omega
  50   simp only [transitionApply,intervalDegreeExtend,dif_pos hi',dif_pos hi'',
  51     finiteIntervalTransition,dif_pos hi]
  52   have hcancel (x : X) : d ⟨i,hi'⟩ x*(a ⟨i,hi⟩ x y/d ⟨i,hi'⟩ x)=a ⟨i,hi⟩ x y := by
  53     field_simp [(hd _ _).ne']
  54   simp_rw [hcancel]
  55   exact hcol ⟨i,hi⟩ y

  57 def capacityPathWeight {X : Type*} {k : ℕ}
  58     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ) : (Fin (k+1) → X) → ℝ :=
  59   weightedTransitionPath (d 0) (finiteIntervalTransition a d) 0 k

  61 theorem capacityPathWeight_nonneg {X : Type*} [Fintype X] {k : ℕ}
  62     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ)
  63     (ha : ∀ j x y, 0 ≤ a j x y) (hd : ∀ j x, 0 < d j x)
  64     (hrow : ∀ j x, ∑ y, a j x y=d j.castSucc x) :
  65     ∀ p, 0 ≤ capacityPathWeight a d p := by
  66   intro p
  67   exact mul_nonneg (hd _ _).le (transitionPathWeight_nonneg _
  68     (fun i => (finiteIntervalTransition_isTransition a d ha hd hrow i).1) 0 k p)

  70 theorem capacityPathWeight_edge {X : Type*} [Fintype X] {k : ℕ}
  71     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ)
  72     (ha : ∀ j x y, 0 ≤ a j x y) (hd : ∀ j x, 0 < d j x)
  73     (hrow : ∀ j x, ∑ y, a j x y=d j.castSucc x)
  74     (hcol : ∀ j y, ∑ x, a j x y=d j.succ y)
  75     (j : Fin k) (x y : X) : pathEdgeLoad (capacityPathWeight a d) j x y=a j x y := by
  76   unfold capacityPathWeight
  77   rw [weightedTransitionPath_edge _ _ (finiteIntervalTransition_isTransition a d ha hd hrow)]
  78   have hbal := transitionWalk_stationary_local (finiteIntervalTransition a d) (intervalDegreeExtend d)
  79     0 j.val (by
  80       intro i hi
  81       simpa only [Nat.zero_add] using intervalDegreeExtend_balance a d hd hcol i (by omega))
  82   have hzero : intervalDegreeExtend d 0=d 0 := by simp [intervalDegreeExtend]
  83   rw [hzero,Nat.zero_add] at hbal
  84   have hdj : intervalDegreeExtend d j.val=d j.castSucc := by
  85     unfold intervalDegreeExtend
  86     rw [dif_pos (show j.val < k+1 by omega)]
  87     rfl
  88   rw [hbal,hdj]
  89   simp only [Nat.zero_add,finiteIntervalTransition, dif_pos j.isLt]
  90   change d j.castSucc x*(a j x y/d j.castSucc x)=a j x y
  91   field_simp [(hd j.castSucc x).ne']

  93 theorem capacityPathWeight_boundary {X : Type*} [Fintype X] {k : ℕ}
  94     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ)
  95     (ha : ∀ j x y, 0 ≤ a j x y) (hd : ∀ j x, 0 < d j x)
  96     (hrow : ∀ j x, ∑ y, a j x y=d j.castSucc x)
  97     (hcol : ∀ j y, ∑ x, a j x y=d j.succ y) :
  98     (∀ x, pathVertexLoad (capacityPathWeight a d) 0 x=d 0 x) ∧
  99     (∀ y, pathVertexLoad (capacityPathWeight a d) (Fin.last k) y=d (Fin.last k) y) := by
 100   constructor
 101   · intro x
 102     exact weightedTransitionPath_first _ _ (finiteIntervalTransition_isTransition a d ha hd hrow) _ _ x
 103   · intro y
 104     rw [capacityPathWeight,weightedTransitionPath_last]
 105     have hbal := transitionWalk_stationary_local (finiteIntervalTransition a d) (intervalDegreeExtend d)
 106       0 k (by
 107         intro i hi
 108         simpa only [Nat.zero_add] using intervalDegreeExtend_balance a d hd hcol i hi)
 109     have hzero : intervalDegreeExtend d 0=d 0 := by simp [intervalDegreeExtend]
 110     rw [hzero,Nat.zero_add] at hbal
 111     rw [hbal]
 112     simp only [intervalDegreeExtend,dif_pos (Nat.lt_succ_self k)]
 113     rfl

 115 theorem transitionPathWeight_continuous {T X : Type*} [TopologicalSpace T]
 116     (R : T → ℕ → X → X → ℝ) (hR : ∀ i x y, Continuous (fun z => R z i x y))
 117     (i k : ℕ) (p : Fin (k+1) → X) : Continuous (fun z => transitionPathWeight (R z) i k p) := by
 118   induction k generalizing i with
 119   | zero => exact continuous_const
 120   | succ k ih => exact (hR i _ _).mul (ih (i+1) (Fin.tail p))

 122 theorem capacityPathWeight_continuous {T X : Type*} [TopologicalSpace T] {k : ℕ}
 123     (a : T → Fin k → X → X → ℝ) (d : T → Fin (k+1) → X → ℝ)
 124     (ha : ∀ j x y, Continuous (fun z => a z j x y))
 125     (hd : ∀ j x, Continuous (fun z => d z j x)) (hdpos : ∀ z j x, 0 < d z j x) :
 126     Continuous (fun z => capacityPathWeight (a z) (d z)) := by
 127   apply continuous_pi
 128   intro p
 129   apply (hd 0 (p 0)).mul
 130   apply transitionPathWeight_continuous
 131   intro i x y
 132   by_cases hi : i < k
 133   · simp only [finiteIntervalTransition,dif_pos hi]
 134     exact (ha _ _ _).div (hd _ _) (fun z => (hdpos z _ _).ne')
 135   · simp only [finiteIntervalTransition,dif_neg hi]
 136     exact continuous_const
