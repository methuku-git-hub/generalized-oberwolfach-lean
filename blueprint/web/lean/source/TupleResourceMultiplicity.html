  14 theorem tuple_resource_multiplicity (c : ℕ) (R : C → Finset V)
  15     (f : C → Fin c → V) (hf : Function.Injective f)
  16     (himage : ∀ C₀,R C₀=Finset.univ.image (f C₀))
  17     (hcard : ∀ C₀,(R C₀).card=c) (Y : Finset V) :
  18     Fintype.card {C₀ // R C₀=Y}≤c^c := by
  19   by_cases hY : ∃ C₀,R C₀=Y
  20   · obtain ⟨C₀,hC₀⟩ := hY
  21     have hYc : Y.card=c := by rw [← hC₀];exact hcard C₀
  22     let embed : {C₀ // R C₀=Y} → (Fin c → Y) := fun C₀ i =>
  23       ⟨f C₀.val i,by
  24         have hh : f C₀.val i∈R C₀.val := by
  25           rw [himage C₀.val]
  26           exact Finset.mem_image.mpr ⟨i,Finset.mem_univ _,rfl⟩
  27         exact (congrArg (fun S : Finset V => f C₀.val i∈S) C₀.property).mp hh⟩
  28     have hi : Function.Injective embed := by
  29       intro C₀ C₁ he
  30       apply Subtype.ext
  31       apply hf
  32       funext i
  33       exact congrArg Subtype.val (congrFun he i)
  34     have hh := Fintype.card_le_of_injective embed hi
  35     simpa only [Fintype.card_fun,Fintype.card_fin,Fintype.card_coe,hYc] using hh
  36   · have hempty : IsEmpty {C₀ // R C₀=Y} := ⟨fun C₀ => hY ⟨C₀.val,C₀.property⟩⟩
  37     letI := hempty
