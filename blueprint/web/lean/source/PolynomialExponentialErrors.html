   9 theorem two_polynomial_exponential_errors_le (n C D a b : ℝ) (i j K : ℕ)
  10     (hn : 1≤n) (hC : 0≤C) (hD : 0≤D) (hi : i≤K) (hj : j≤K) :
  11     C*n^i*Real.exp (-a*n)+D*n^j*Real.exp (-b*n)≤
  12       (C+D)*n^K*Real.exp (-min a b*n) := by
  13   have hn0 : 0≤n := by linarith
  14   have hpA : n^i≤n^K := pow_le_pow_right₀ hn hi
  15   have hpB : n^j≤n^K := pow_le_pow_right₀ hn hj
  16   have heA : Real.exp (-a*n)≤Real.exp (-min a b*n) := by
  17     apply Real.exp_le_exp.mpr
  18     have hh := mul_le_mul_of_nonneg_right (min_le_left a b) hn0
  19     linarith only [hh]
  20   have heB : Real.exp (-b*n)≤Real.exp (-min a b*n) := by
  21     apply Real.exp_le_exp.mpr
  22     have hh := mul_le_mul_of_nonneg_right (min_le_right a b) hn0
  23     linarith only [hh]
  24   have hA := mul_le_mul (mul_le_mul_of_nonneg_left hpA hC) heA (by positivity)
  25     (mul_nonneg hC (pow_nonneg hn0 K))
  26   have hB := mul_le_mul (mul_le_mul_of_nonneg_left hpB hD) heB (by positivity)
  27     (mul_nonneg hD (pow_nonneg hn0 K))
  28   nlinarith only [hA,hB]
