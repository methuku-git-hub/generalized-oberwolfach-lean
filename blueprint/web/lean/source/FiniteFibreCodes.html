  13 theorem finite_fibre_codes {A B : Type*} [Fintype A] [DecidableEq B]
  14     (f : A → B) (J : ℕ) (hJ : ∀ b,Fintype.card {a : A // f a=b} ≤ J) :
  15     ∃ code : A → Fin J,∀ a b,f a=f b → code a=code b → a=b := by
  16   have he (b : B) : Nonempty ({a : A // f a=b} ↪ Fin J) :=
  17     Function.Embedding.nonempty_of_card_le (by simpa only [Fintype.card_fin] using hJ b)
  18   let e := fun b => Classical.choice (he b)
  19   let code := fun a => e (f a) ⟨a,rfl⟩
  20   refine ⟨code,?_⟩
  21   intro a b hab hc
  22   have hb : f b=f a := hab.symm
  23   have hon (b₀ : B) (x : A) (hx : f x=b₀) : code x=e b₀ ⟨x,hx⟩ := by
  24     subst b₀
  25     rfl
  26   have hh : e (f a) ⟨a,rfl⟩=e (f a) ⟨b,hb⟩ := by
  27     rw [hon (f a) a rfl,hon (f a) b hb] at hc
  28     exact hc
  29   exact congrArg Subtype.val ((e (f a)).injective hh)
