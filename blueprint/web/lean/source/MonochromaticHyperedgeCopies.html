  14 abbrev MonochromaticHyperedge (R : E → Finset V) (σ : V → I) :=
  15   Σ i : I, {e : E // e∈hypergraphColorClass R σ i}

  17 def monochromaticProjection (R : E → Finset V) (σ : V → I) (e : MonochromaticHyperedge R σ) : E := e.2.val

  19 theorem monochromaticProjection_injective (R : E → Finset V) (σ : V → I)
  20     (hne : ∀ e,(R e).Nonempty) : Function.Injective (monochromaticProjection R σ) := by
  21   intro e f hef
  22   change e.2.val=f.2.val at hef
  23   obtain ⟨v,hv⟩ := hne e.2.val
  24   have hi : e.1=f.1 := by
  25     have he := (hypergraphColorClass_mem R σ e.1 e.2.val).mp e.2.property v hv
  26     have hf := (hypergraphColorClass_mem R σ f.1 f.2.val).mp f.2.property v (hef ▸ hv)
  27     exact he.symm.trans hf
  28   cases e with | mk i e =>
  29     cases f with | mk j f =>
  30       dsimp only at hi
  31       subst j
  32       exact congrArg (Sigma.mk i) (Subtype.ext hef)

  66 theorem monochromatic_copies_matching_projection (R : E → Finset V) (σ : V → I)
  67     (hne : ∀ e,(R e).Nonempty) (M : Finset (MonochromaticHyperedge R σ))
  68     (hM : ∀ e∈M,∀ f∈M,e≠f → Disjoint (R e.2.val) (R f.2.val)) :
