  25     ∃ n₀ : ℕ, 2 ≤ n₀ ∧ ∀ (V : Type*) [Fintype V] [DecidableEq V],

  27     (S : Fin N → Finset V) (family : Fin N → Set (SimpleGraph V)),
  28     (∀ i j, i ≠ j → ((S i ∩ S j).card : ℝ) ≤ p^2 * Fintype.card V) →
  29     (∀ v, ((Finset.univ.filter (fun i => v ∈ S i)).card : ℝ) ≤ p * Fintype.card V) →
  30     (∀ i, ∀ K : SimpleGraph V, K ≤ spanningRestriction G (S i) →
  31       (∀ v, (K.degree v : ℝ) ≤ D * (p * Real.sqrt p * Fintype.card V) +

  33       ∃ C : SimpleGraph V, C ∈ family i ∧ C ≤ spanningRestriction G (S i) ∧
