  14 theorem finiteWitness_coordinate_certificate
  15     {I J Ω : Type*} [Fintype I] [Fintype J]
  16     (P : J → (I → Ω) → Prop) (S : (I → Ω) → J → Finset I)
  17     {k : ℕ} {c M : ℝ} (hM : ∀ x,((Finset.univ.filter (fun j=>P j x)).card : ℝ)≤M)
  18     (hsize : ∀ x j,P j x → (S x j).card≤k)
  19     (hload : ∀ x i,((Finset.univ.filter (fun j=>P j x ∧ i∈S x j)).card : ℝ)≤c)
  20     (hwitness : ∀ x j,P j x → ∀ y,(∀ i∈S x j,x i=y i) → P j y) :
  21     HasCoordinateCertificateBound
  22       (fun x=>((Finset.univ.filter (fun j=>P j x)).card : ℝ)) c ((k : ℝ)*M) := by
  23   classical
  24   intro x
  25   let T := Finset.univ.filter (fun j=>P j x)
  26   let w : I → ℝ := fun i=>∑ j∈T,if i∈S x j then (1 : ℝ) else 0
  27   have hw0 (i : I) : 0≤w i := Finset.sum_nonneg (fun j _=>by split_ifs <;> norm_num)
  28   have hwle (i : I) : w i≤c := by
  29     have he : w i=((Finset.univ.filter (fun j=>P j x ∧ i∈S x j)).card : ℝ) := by
  30       unfold w T
  31       rw [Finset.sum_filter]
  32       simp only [← ite_and]
  33       rw [Finset.sum_boole]
  34     rw [he]
  35     exact hload x i
  36   refine ⟨w,fun i=>⟨hw0 i,hwle i⟩,?_,?_⟩
  37   · have he : (∑ i,w i)=∑ j∈T,((S x j).card : ℝ) := by
  38       unfold w
  39       rw [Finset.sum_comm]
  40       apply Finset.sum_congr rfl
  41       intro j _
  42       rw [Fintype.sum_ite_mem]
  43       simp
  44     rw [he]
  45     calc
  46       _ ≤ ∑ _j∈T,(k : ℝ) := Finset.sum_le_sum (fun j hj=>by
  47         exact_mod_cast hsize x j (Finset.mem_filter.mp hj).2)
  48       _ = (k : ℝ)*(T.card : ℝ) := by simp [mul_comm]
  49       _ ≤ _ := mul_le_mul_of_nonneg_left (hM x) (Nat.cast_nonneg _)
  50   · intro y
  51     let a : J → I → ℝ := fun j i=>if x i≠y i ∧ i∈S x j then 1 else 0
  52     have ha (j : J) (i : I) : 0≤a j i := by unfold a;split_ifs <;> norm_num
  53     have hone (j : J) (hj : j∈T) :
  54         (1 : ℝ)≤(if P j y then 1 else 0)+∑ i,a j i := by
  55       by_cases hy : P j y
  56       · simp only [hy,ite_true]
  57         have hn : 0≤∑ i,a j i := Finset.sum_nonneg (fun i _=>ha j i)
  58         linarith
  59       · have hex : ∃ i∈S x j,x i≠y i := by
  60           by_contra h
  61           push Not at h
  62           exact hy (hwitness x j (Finset.mem_filter.mp hj).2 y h)
  63         obtain ⟨i,hi,hne⟩ := hex
  64         have hh := Finset.single_le_sum (s:=Finset.univ) (f:=a j)
  65           (fun i _=>ha j i) (Finset.mem_univ i)
  66         simpa [a,hne,hi,hy] using hh
  67     have hs := Finset.sum_le_sum hone
  68     rw [Finset.sum_add_distrib] at hs
  69     have hsurv : (∑ j∈T,if P j y then (1 : ℝ) else 0)≤
  70         ((Finset.univ.filter (fun j=>P j y)).card : ℝ) := by
  71       calc
  72         _ ≤ ∑ j,if P j y then (1 : ℝ) else 0 :=
  73           Finset.sum_le_sum_of_subset_of_nonneg (Finset.subset_univ _)
  74             (fun j _ _=>by split_ifs <;> norm_num)
  75         _ = _ := Finset.sum_boole _ _
  76     have hchanges : (∑ j∈T,∑ i,a j i)=∑ i,if x i≠y i then w i else 0 := by
  77       rw [Finset.sum_comm]
  78       apply Finset.sum_congr rfl
  79       intro i _
  80       by_cases he : x i=y i
  81       · simp [a,he]
  82       · simp only [a,ne_eq,he,not_false_eq_true,true_and,ite_true]
  83         rfl
  84     rw [hchanges] at hs
  85     have hsum : (∑ _j∈T,(1 : ℝ))=(T.card : ℝ) := by simp
  86     rw [hsum] at hs
  87     change (T.card : ℝ)≤_
  88     linarith
