  16 theorem ordinary_fresh_comparison_embedding (Q F:SimpleGraph V) (H:V→U→Prop)
  17     (A W R:Finset V) (B:Finset U) (r k:ℕ) (hr:3≤r) (d c cH:ℝ) (hd:0≤d)
  18     (hF:∀ v,(F.degree v:ℝ)≤d) (hA:c≤(A.card:ℝ))
  19     (hcount:∀ v w,c≤((A∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  20     (hhub:∀ T:Finset V,T.card≤r→(∀ v∈T,∀ w∈T,¬F.Adj v w)→
  21       cH≤((common Finset.univ T (fun v=>Finset.univ.filter (H v))).card:ℝ))
  22     (hbudget:W.card+((R.card:ℝ)+k+k*r+r+6)*(d+1)<c)
  23     (hB:(B.card:ℝ)+k<cH) :
  24     ∃ (f:OrdinaryComparisonRows r k↪V) (g:OrdinaryComparisonHubs k↪U),
  25       (∀ a,f a∈A ∧ f a∉W ∧ f a∉R ∧ ∀ z∈R,¬F.Adj (f a) z) ∧
  26       (∀ a,g a∉B) ∧ (∀ a b,¬F.Adj (f a) (f b)) ∧
  27       (∀ a b,(ordinaryComparisonRim r k (by omega)).Adj a b→Q.Adj (f a) (f b)) ∧
  28       (∀ a b,ordinaryComparisonSpokes r k (by omega) a b→H (f a) (g b)) ∧
  29       HasResourceDecomposition (ordinaryWheelResources (P:=Q) (F:=F) (Hbase:=H) (r:=r))
  30         (ordinaryHostResources ((ordinaryComparisonRim r k (by omega)).map f)
  31           (ordinaryMappedSpokes (ordinaryComparisonSpokes r k (by omega)) f g)) := by
  32   have hd1:0≤d+1:=by linarith
  33   obtain ⟨φ,hφ,hφF,hφR⟩:=exists_separated_row_embedding (Fin k) F A W R d hd hF (by
  34     simp only [Fintype.card_fin]
  35     have hm:0≤((k:ℝ)*r+r+6)*(d+1):=by positivity
  36     nlinarith only [hbudget,hA,hm])
  37   let ψ:Empty↪U:=⟨Empty.elim,fun a=>a.elim⟩
  38   let R':=R∪Finset.univ.image φ
  39   have hRc:(R'.card:ℝ)≤R.card+k:=by
  40     have hh: (Finset.univ.image φ).card≤k:=Finset.card_image_le.trans (by simp)
  41     exact_mod_cast (Finset.card_union_le R (Finset.univ.image φ)).trans (Nat.add_le_add_left hh R.card)
  42   obtain ⟨f,g,hf,hg,hpriv,hnewhub,hprivF,hQ,hH⟩:=ordinary_relative_attachment_embedding Q F H A W R' B φ ψ
  43     ⊥ (fun (_:Fin k) (_:Empty)=>False) id
  44     (fun a=>Finset.mem_union_right _ (Finset.mem_image.mpr ⟨a,Finset.mem_univ _,rfl⟩))
  45     (fun a=>a.elim) (fun a b h=>h.elim) (fun a b h=>h.elim)
  46     r hr d c cH hd hF hcount hhub
  47     (by simp only [Fintype.card_fin];have hm:=mul_le_mul_of_nonneg_right hRc hd1;nlinarith only [hbudget,hm])
  48     (by simpa only [Fintype.card_fin] using hB)
  49   have hcross:∀ i w a,¬F.Adj (f (.inr (i,w))) (f (.inl a)):=by
  50     intro i w a
  51     rw [hf]
  52     exact (hpriv i w).2.2.2 (φ a) (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨a,Finset.mem_univ _,rfl⟩))
  53   have hind:∀ a b,¬F.Adj (f a) (f b):=by
  54     intro a b
  55     cases a with
  56     | inl a=>
  57       cases b with
  58       | inl b=>rw [hf,hf];exact hφF a b
  59       | inr p=>exact fun h=>hcross p.1 p.2 a h.symm
  60     | inr p=>
  61       cases b with
  62       | inl b=>exact hcross p.1 p.2 b
  63       | inr q=>exact hprivF p.1 q.1 p.2 q.2
  64   have hC:IsEmbeddedCycle (SimpleGraph.cycleGraph r) r:=by
  65     refine ⟨hr,Function.Embedding.refl _,?_⟩
  66     change SimpleGraph.cycleGraph r=(SimpleGraph.cycleGraph r).map id
  67     rw [SimpleGraph.map_id]
  68   have hfull:∀ w:Fin r,w∈(SimpleGraph.cycleGraph r).support:=by
  69     intro w;rw [cycleGraph_support_eq_univ hr];exact Set.mem_univ _
  70   have hdec:=ordinary_attachment_extra_host_decomposition Q F H (SimpleGraph.cycleGraph r) hC hfull
  71     ⟨0,by omega⟩ (id:Fin k→Fin k) f g hQ hH hcross hprivF
  72   refine ⟨f,g,?_,?_,hind,hQ,hH,hdec⟩
  73   · intro a
  74     cases a with
  75     | inl a=>rw [hf];exact ⟨(hφ a).1,(hφ a).2.1,(hφ a).2.2,hφR a⟩
  76     | inr p=>exact ⟨(hpriv p.1 p.2).1,(hpriv p.1 p.2).2.1,
  77         fun h=>(hpriv p.1 p.2).2.2.1 (Finset.mem_union_left _ h),
  78         fun z hz=>(hpriv p.1 p.2).2.2.2 z (Finset.mem_union_left _ hz)⟩
  79   · intro a
  80     cases a with
  81     | inl a=>exact a.elim
  82     | inr i=>exact hnewhub i
