  15 theorem sampling_probability_scaled_lower (p k t δ : ℝ) (h : ℕ)
  16     (ht : 0 < t) (hp : |(k/t)^h*p-1| ≤ δ) :
  17     (1-δ)*t^h ≤ p*k^h := by
  18   have hh : 1-δ ≤ (k/t)^h*p := by
  19     have hl := (abs_le.mp hp).1
  20     linarith
  21   have he : (k/t)^h*p*t^h=p*k^h := by
  22     rw [div_pow]
  23     field_simp
  24   calc
  25     _ ≤ (k/t)^h*p*t^h := mul_le_mul_of_nonneg_right hh (pow_nonneg ht.le _)
  26     _ = _ := he

  34     (hL : (Z.card*((c*c)*Fintype.card T*X.card^c*Fintype.card W^(c*c-1)*J^(2*c+1)):ℕ) ≤ L) :
  35     A-L ≤ (Fintype.card {D : LabelledAbsorberFamily R S X //
  36       Disjoint (labelledAbsorberFree R (S D.1) X D.2) Z}:ℝ) := by

  38   have hbad := labelled_absorber_free_deletion_count R S X Z J hJ
  39   have hpart : Fintype.card {D // P D}+Fintype.card {D // ¬P D}=
  40       Fintype.card (LabelledAbsorberFamily R S X) := by

  42     exact Nat.add_sub_of_le (Fintype.card_subtype_le P)
  43   have hpart' : (Fintype.card {D // P D}:ℝ)+Fintype.card {D // ¬P D}=
  44       Fintype.card (LabelledAbsorberFamily R S X) := by exact_mod_cast hpart

  53     {V : I → Type*} [∀ i,Fintype (V i)] [∀ i,DecidableEq (V i)]

  56     (hR : ∀ C₀ i,(Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈R C₀)).card=a i)
  57     (X Z : Finset (Σ i,V i)) (J : ℕ)
  58     (hJ : ∀ A : Finset (Σ i,V i),Fintype.card {C₀ : C // R C₀=A} ≤ J)
  59     (m : I → ℕ) [∀ i,Nonempty (FixedSubset (ProfileRemainingClass Z i) (m i))]
  60     (hm : ∀ i,m i ≤ Fintype.card (ProfileRemainingClass Z i))

  63     (hloss : (Z.card*((c*c)*Fintype.card T*X.card^c*Fintype.card (Σ i,V i)^(c*c-1)*J^(2*c+1)):ℕ) ≤
  64       α/4*k^(c*c))
  65     (hratio : |(k/t)^(c*c)*(∏ i,((m i).choose (c*a i):ℝ)/
  66       (Fintype.card (ProfileRemainingClass Z i)).choose (c*a i))-1| ≤ 1/4) :
  67     (α/2)*t^(c*c) ≤
  68       (FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (ProfileRemainingClass Z i) (m i)))).expectation
  69         (fun σ => absorberGroundCount R S X ((profileRootSet σ).map (profileComplementEmbedding Z))) := by
  70   let p := ∏ i,((m i).choose (c*a i):ℝ)/(Fintype.card (ProfileRemainingClass Z i)).choose (c*a i)
  71   have hp : 0 ≤ p := Finset.prod_nonneg (fun i _ => by positivity)
  72   have hgood := labelled_absorber_avoiding_lower R S X Z J hJ (α*k^(c*c)) (α/4*k^(c*c)) hcount hloss
  73   have hgood' : (3/4)*α*k^(c*c) ≤ (Fintype.card {D : LabelledAbsorberFamily R S X //
  74       Disjoint (labelledAbsorberFree R (S D.1) X D.2) Z}:ℝ) := by linarith
  75   have hprob := sampling_probability_scaled_lower p k t (1/4) (c*c) ht hratio
  76   rw [absorber_deletion_profile_expectation R S a hR X Z m hm]
  77   change _ ≤ p*_
  78   calc
  79     _ ≤ (3/4)*α*((1-1/4)*t^(c*c)) := by nlinarith [pow_nonneg ht.le (c*c)]
  80     _ ≤ (3/4)*α*(p*k^(c*c)) := mul_le_mul_of_nonneg_left hprob (by positivity)

  82     _ ≤ _ := mul_le_mul_of_nonneg_left hgood' hp
