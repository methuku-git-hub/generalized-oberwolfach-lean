  10 theorem assigned_factor_component_source_disjoint
  11     (F:U→SimpleGraph (Fin n)) (hub:A→U)
  12     (C:∀a,(F (hub a)).ConnectedComponent)
  13     (assign:A↪(Σu,(F u).ConnectedComponent))
  14     (hC:∀a,(⟨hub a,C a⟩:Σu,(F u).ConnectedComponent)=assign a):
  15     ∀a a',a≠a'→hub a=hub a'→Disjoint (C a).supp (C a').supp :=by
  16   intro a a' hne heq
  17   apply Set.disjoint_left.mpr
  18   intro x hx hx'
  19   have h1:(F (hub a)).connectedComponentMk x=C a:=hx
  20   have h2:(F (hub a')).connectedComponentMk x=C a':=hx'
  21   have hA:assign a=(⟨hub a,(F (hub a)).connectedComponentMk x⟩:Σu,(F u).ConnectedComponent):=by
  22     rw [←hC a,h1]
  23   have hB:assign a'=(⟨hub a',(F (hub a')).connectedComponentMk x⟩:Σu,(F u).ConnectedComponent):=by
  24     rw [←hC a',h2]
  25   have hsame:(⟨hub a,(F (hub a)).connectedComponentMk x⟩:Σu,(F u).ConnectedComponent)=
  26       ⟨hub a',(F (hub a')).connectedComponentMk x⟩:=by rw [heq]
  27   exact hne (assign.injective (hA.trans (hsame.trans hB.symm)))
  28 end Oberwolfach
