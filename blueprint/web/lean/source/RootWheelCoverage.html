  14 theorem root_wheel_family_exact_edges (C:I→OrdinaryWheel Q F H r) (x:V) (v w:I→V) (u:I→U)
  15     (hvw:∀ i,v i≠w i) (hhub:∀ i,ordinaryWheelHub (C i)=u i)
  16     (hxv:∀ i,(ordinaryWheelRim (C i)).Adj x (v i))
  17     (hxw:∀ i,(ordinaryWheelRim (C i)).Adj x (w i)) :
  18     (∀ a,(wheelFamilyRim C).Adj x a ↔ ∃ i,a=v i∨a=w i) ∧
  19       (∀ b,wheelFamilySpokes C x b ↔ ∃ i,u i=b) := by
  20   constructor
  21   · intro a
  22     rw [wheelFamilyRim,SimpleGraph.iSup_adj]
  23     constructor
  24     · rintro ⟨i,hi⟩
  25       have hh:=((ordinaryWheelRim (C i)).mem_neighborFinset x a).mpr hi
  26       rw [ordinaryWheel_root_neighbors (C i) x (v i) (w i) (hvw i) (hxv i) (hxw i)] at hh
  27       exact ⟨i,by simpa only [Finset.mem_insert,Finset.mem_singleton] using hh⟩
  28     · rintro ⟨i,he|he⟩
  29       · exact ⟨i,by rw [he];exact hxv i⟩
  30       · exact ⟨i,by rw [he];exact hxw i⟩
  31   · intro b
  32     constructor
  33     · rintro ⟨i,hi,hxi⟩
  34       exact ⟨i,(hhub i).symm.trans hi⟩
