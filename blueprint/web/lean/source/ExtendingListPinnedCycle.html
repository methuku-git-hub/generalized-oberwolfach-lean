  11 theorem extending_list_embed_pinned_cycle (D:V→V→Prop) (L:Finset V) (β:ℝ)
  12     (hext:IsExtendingList D L β) (F:SimpleGraph V) (M:ℝ) (hM:0≤M)
  13     (hdeg:∀ v,(F.degree v:ℝ)≤M) (B:Finset V) (x:V) (ℓ:ℕ) (hℓ:3≤ℓ)
  14     (hbudget:(B.card:ℝ)+ℓ+2+2*M<β*Fintype.card V):
  15     ∃ xs:List V,xs.length=ℓ-1 ∧ xs.Nodup ∧
  16       (∀ z∈xs,z∈L ∧ z∉B ∧ z≠x) ∧ DirectedPathThrough D F x x xs := by
  17   let B₀:=insert x B
  18   have hB₀:(B₀.card:ℝ)≤B.card+1:=by exact_mod_cast Finset.card_insert_le x B
  19   have hchoice:(B₀.card:ℝ)+F.degree x<β*Fintype.card V:=by
  20     have hx:=hdeg x
  21     nlinarith
  22   obtain ⟨y,hyL,hyB,hdxy,hFxy⟩:=extending_list_one_arc_choice D L B₀ β hext F x true hchoice
  23   have hyB':y∉B:=fun h=>hyB (by simp [B₀,h])
  24   have hyx:y≠x:=by intro h;subst y;apply hyB;simp [B₀]
  25   have hbudget':(B₀.card:ℝ)+((ℓ-3)+1:ℕ)+2+2*M<β*Fintype.card V:=by
  26     have hh:((ℓ-3)+1:ℕ)≤ℓ-2:=by omega
  27     have hhR:(((ℓ-3)+1:ℕ):ℝ)≤((ℓ-2:ℕ):ℝ):=by exact_mod_cast hh
  28     have hsub:((ℓ-2:ℕ):ℝ)+2=ℓ:=by exact_mod_cast Nat.sub_add_cancel (by omega:2≤ℓ)
  29     nlinarith
  30   obtain ⟨zs,hzslen,hzsnodup,hzs,hchain⟩:=extending_list_embed_internal_path D L β hext F M hM hdeg (ℓ-3) B₀ y x hyx hbudget'
  31   have hzy:y∉zs:=by intro hz;exact (hzs y hz).2.2.1 rfl
  32   refine ⟨y::zs,?_,List.nodup_cons.mpr ⟨hzy,hzsnodup⟩,?_,?_⟩
  33   · simp only [List.length_cons,hzslen];omega
  34   · intro z hz
  35     rcases List.mem_cons.mp hz with rfl | hz
  36     · exact ⟨hyL,hyB',hyx⟩
  37     obtain ⟨hzL,hzB₀,_,hzx⟩:=hzs z hz
  38     exact ⟨hzL,fun h=>hzB₀ (by simp [B₀,h]),hzx⟩
  39   · exact ⟨hdxy,hFxy,hchain⟩
