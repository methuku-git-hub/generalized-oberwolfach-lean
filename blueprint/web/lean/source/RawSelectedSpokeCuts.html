  18     ∃ N : ℕ,∀ (V U Γ Color : Type)
  19       [Fintype V] [Fintype U] [Fintype Γ] [Fintype Color] [DecidableEq V] [DecidableEq U],
  20       N≤Fintype.card V → ω*Fintype.card V≤(Fintype.card U:ℝ) → Fintype.card U≤Fintype.card V →
  21     ∀ (π : Equiv.Perm V)
  22       (q : U → Γ → ℝ) (hq : ∀ u g,0≤q u g) (hQ : ∀ u,∑ g,q u g≤1/8)
  23       (s : U → Color → ℝ) (hs : ∀ u j,0≤s u j)
  24       (hS : ∀ u,∑ j,s u j≤ intervalUncoveredProbability d (∑ g,q u g))
  25       (j : Γ⊕Color),(∀ x k,0<k → k<2*d+1 → (π^k) x≠x) →
  26     ∀ pH : ℝ,pstar≤pH → (∀ u,Sum.elim (q u) (s u) j=pH) →
  27     1-1/(Fintype.card V:ℝ)^q₀≤
  28       (FiniteLaw.pi (rawIntervalFamilyLaw (V:=V) d hd q hq hQ s hs hS)).probability
  29         (fun σ=>∀ (X : Finset V) (Y : Finset U),
  30           1-pstar/8<(X.card:ℝ)/Fintype.card V+(Y.card:ℝ)/Fintype.card U →
  31           pstar/8*min ((Fintype.card U:ℝ)*X.card) ((Fintype.card V:ℝ)*Y.card)≤
  32             (bipartiteMatrixCut (rawSelectedSpokeMatrix π d σ j) X Y:ℝ)) := by
  33   let ε : ℝ := pstar^2*ω/128
  34   have hε : 0<ε := by dsimp [ε];positivity
  35   obtain ⟨N₁,hN₁⟩ := rawSelectedSpokeMatrix_degree_event pstar ω hpstar hω d (q₀+1) hd
  36   obtain ⟨N₂,hN₂⟩ := rawSelectedCutCount_quadratic_event d hd ε hε (q₀+1)
  37   refine ⟨max 2 (max N₁ N₂),?_⟩
  38   intro V U Γ Color _ _ _ _ _ _ hnN hmlo hmhi π q hq hQ s hs hS j hnoshort pH hpH hj
  39   let n := Fintype.card V
  40   let m := Fintype.card U
  41   have hn : 0<n := by dsimp [n];omega
  42   have hn0 : (0:ℝ)<n := by exact_mod_cast hn
  43   have hm0 : (0:ℝ)<m := (mul_pos hω hn0).trans_le hmlo
  44   have hm : 0<m := by exact_mod_cast hm0
  45   let μ := FiniteLaw.pi (rawIntervalFamilyLaw (V:=V) d hd q hq hQ s hs hS)
  46   have hdegree := hN₁ V U Γ Color (by omega) hmlo hmhi π q hq hQ s hs hS j hnoshort pH hpH hj
  47   have hcuts := hN₂ V U Γ Color (by omega) hm hmhi π q hq hQ s hs hS j hnoshort pH hj
  48   have hjoint := finiteLaw_good_intersection μ
  49     (fun σ=>(∀ v : V,pstar*(Fintype.card U:ℝ)/2≤((∑ u,rawSelectedSpokeMatrix π d σ j v u:ℕ):ℝ)) ∧
  50         (∀ u : U,pstar*(Fintype.card V:ℝ)/2≤((∑ v,rawSelectedSpokeMatrix π d σ j v u:ℕ):ℝ)))
  51     _ (1/(n:ℝ)^(q₀+1)) hcuts
  52   have hbase := (sub_le_sub_right hdegree (1/(n:ℝ)^(q₀+1))).trans hjoint
  53   have hsum := scaled_inverse_power_le n q₀ hn 2 (by exact_mod_cast (show 2≤n by dsimp [n];omega))
  54   have hbound : 1-1/(n:ℝ)^q₀≤(1-1/(n:ℝ)^(q₀+1))-1/(n:ℝ)^(q₀+1) := by
  55     convert sub_le_sub_left hsum 1 using 1 <;> ring
  56   apply (hbound.trans hbase).trans
  57   apply finiteLaw_probability_mono
  58   intro σ hσ X Y hlarge
  59   have hquad (S : Finset V) (T : Finset U) :
  60       |(bipartiteMatrixCut (rawSelectedSpokeMatrix π d σ j) S T:ℝ)-pH*S.card*T.card|≤
  61         (pstar^2/128)*(n:ℝ)*m := by
  62     unfold bipartiteMatrixCut
  63     rw [rawSelectedSpokeMatrix_cut]
  64     have hh := (hσ.2 S T).le
  65     have hscale := mul_le_mul_of_nonneg_left hmlo (by positivity : (0:ℝ)≤pstar^2/128*n)
  66     dsimp only [ε,n,m] at hh ⊢
  67     nlinarith only [hh,hscale]
  68   have hcross := mul_lt_mul_of_pos_right hlarge (mul_pos hn0 hm0)
  69   have heq : ((X.card:ℝ)/n+(Y.card:ℝ)/m)*((n:ℝ)*m)=
  70       (m:ℝ)*X.card+(n:ℝ)*Y.card := by field_simp <;> ring
  71   change (1-pstar/8)*((n:ℝ)*m)<((X.card:ℝ)/n+(Y.card:ℝ)/m)*((n:ℝ)*m) at hcross
  72   rw [heq] at hcross

  76     (by convert hcross using 1 <;> ring)
