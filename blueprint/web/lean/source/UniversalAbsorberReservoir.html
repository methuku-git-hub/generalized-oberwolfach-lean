  22 theorem universal_absorber_reservoir (R : W → Finset E) (X : I → Finset E)
  23     (A : C → Finset E) (S : Finset C) (b : ℝ)
  24     (hAA : ∀ i∈S,∀ j∈S,i≠j → Disjoint (A i) (A j))
  25     (hA : ∀ j∈S,HasResourceDecomposition R (A j))
  26     (hchoices : ∀ i,b ≤ ((S.filter (fun j => HasResourceDecomposition R (A j∪X i))).card:ℝ)) :
  27     ∀ M : Finset I,(M.card:ℝ) ≤ b →
  28       (∀ i∈M,∀ j∈M,i≠j → Disjoint (X i) (X j)) →
  29       Disjoint (M.biUnion X) (S.biUnion A) →
  30       IsResourceAbsorber R (M.biUnion X) (S.biUnion A) := by
  31   intro M hM hXX hXA
  32   let P (i : M) (j : C) := j∈S ∧ HasResourceDecomposition R (A j∪X i.val)
  33   have hP (i : M) : Fintype.card M ≤ (Finset.univ.filter (P i)).card := by
  34     have he : Finset.univ.filter (P i)=S.filter (fun j => HasResourceDecomposition R (A j∪X i.val)) := by
  35       ext j;simp [P]
  36     rw [he,Fintype.card_coe]
  37     exact_mod_cast hM.trans (hchoices i.val)
  38   obtain ⟨f,hf,hfP⟩ := injective_assignment_of_many_choices P (by intro i;convert hP i using 1 <;> congr!)
  39   let f' : M → S := fun i => ⟨f i,(hfP i).1⟩
  40   have hf' : Function.Injective f' := by
  41     intro i j hij
  42     exact hf (congrArg Subtype.val hij)
  43   have hh := assigned_absorber_reservoir R (fun i : M => X i.val) (fun j : S => A j.val)
  44     (fun i j hij => hXX i.val i.property j.val j.property (fun h => hij (Subtype.ext h)))
  45     (fun i j hij => hAA i.val i.property j.val j.property (fun h => hij (Subtype.ext h)))
  46     (fun i j => hXA.mono (resource_subset_union X M i.val i.property)
  47       (resource_subset_union A S j.val j.property))
  48     (fun j => hA j.val j.property) f' hf' (fun i => (hfP i).2)
  49   rw [biUnion_finset_subtype,biUnion_finset_subtype] at hh
  50   exact hh
