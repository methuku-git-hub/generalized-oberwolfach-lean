  16     (tail head : E→V) (ktail khead : K→V) (π : Equiv.Perm V)
  17     (A : Finset (E⊕K)) (v : V) :
  18     (physicalEdgeGraph (A.image (markedPhysicalEdge tail head ktail khead π))).degree v ≤

  21   rw [Finset.filter_image]

  24 theorem finite_spoke_graph_degree_counts (S : Finset (V×U)) :
  25     (∀ v,(markedFiniteSpokeGraph S).degree (Sum.inl v)≤(S.filter (fun s=>s.1=v)).card) ∧
  26     (∀ u,(markedFiniteSpokeGraph S).degree (Sum.inr u)≤(S.filter (fun s=>s.2=u)).card) := by
  27   constructor
  28   · intro v
  29     rw [markedFiniteSpokeGraph_degree_row]
  30     apply Finset.card_le_card_of_injOn (fun u=>(v,u))
  31     · intro u hu
  32       exact Finset.mem_filter.mpr ⟨(Finset.mem_filter.mp hu).2,rfl⟩
  33     · intro a ha b hb h
  34       exact congrArg Prod.snd h
  35   · intro u
  36     rw [markedFiniteSpokeGraph_degree_col]
  37     apply Finset.card_le_card_of_injOn (fun v=>(v,u))
  38     · intro v hv
  39       exact Finset.mem_filter.mpr ⟨(Finset.mem_filter.mp hv).2,rfl⟩
  40     · intro a ha b hb h
  41       exact congrArg Prod.fst h

  43 theorem finite_physical_arc_graph_into_set
  44     (tail head : E→V) (ktail khead : K→V) (π : Equiv.Perm V)
  45     (A : Finset (E⊕K)) (X : Finset V) (hX : ∀ v,π v∈X↔v∈X) (v : V) :
  46     ((physicalEdgeGraph (A.image (markedPhysicalEdge tail head ktail khead π))).neighborFinset v∩X).card ≤
  47       (physicalEdgeGraph ((A.filter (fun a=>Sum.elim tail ktail a∈X ∨ Sum.elim head khead a∈X)).image
  48         (markedPhysicalEdge tail head ktail khead π))).degree v := by
  49   rw [←SimpleGraph.card_neighborFinset_eq_degree]
  50   apply Finset.card_le_card
  51   intro w hw
  52   obtain ⟨hadj,hwX⟩ := Finset.mem_inter.mp hw
  53   have hadj := (SimpleGraph.mem_neighborFinset _ _ _).mp hadj
  54   obtain ⟨a,ha,he⟩ := Finset.mem_image.mp hadj.2
  55   apply (SimpleGraph.mem_neighborFinset _ _ _).mpr
  56   refine ⟨hadj.1,Finset.mem_image.mpr ⟨a,Finset.mem_filter.mpr ⟨ha,?_⟩,he⟩⟩
  57   have hwA : w∈markedPhysicalEdge tail head ktail khead π a := by rw [he];simp
  58   cases a with
  59   | inl a=>
  60     simp only [Sum.elim_inl]
  61     have hw : w=tail a ∨ w=head a := by simpa only [markedPhysicalEdge,Sum.elim_inl,Sum.elim_inr,Finset.mem_insert,Finset.mem_singleton] using hwA
  62     rcases hw with h|h
  63     · exact Or.inl (h ▸ hwX)
  64     · exact Or.inr (h ▸ hwX)
  65   | inr a=>
  66     simp only [Sum.elim_inr]
  67     have hw : w=ktail a ∨ w=π (khead a) := by simpa only [markedPhysicalEdge,Sum.elim_inl,Sum.elim_inr,Finset.mem_insert,Finset.mem_singleton] using hwA
  68     rcases hw with h|h
  69     · exact Or.inl (h ▸ hwX)
  70     · exact Or.inr ((hX _).mp (h ▸ hwX))

  91 theorem marked_finite_deletion_graph_bounds
  92     (tail head : E→V) (ktail khead : K→V) (π : Equiv.Perm V) (H₀ HK : V→U→Prop)
  93     (A : Finset (E⊕K)) (S : Finset (V×U)) (B : Finset (Finset V⊕(V×U)))
  94     (X : Finset V) (Z : Finset U) (hX : ∀ v,π v∈X↔v∈X) (δ σ : ℝ)
  95     (hA : ∀ a∈A,Sum.inl (markedPhysicalEdge tail head ktail khead π a)∈B)
  96     (hS : ∀ s∈S,H₀ s.1 s.2 ∨ HK s.1 s.2) (hSB : ∀ s∈S,Sum.inr s∈B)
  97     (hB : ∀ physical z,(((markedIncidentEdges tail head ktail khead π H₀ HK physical z).filter
  98       (fun e=>markedActivePhysical tail head ktail khead π H₀ HK e∈B)).card : ℝ)≤δ)
  99     (hStar : ∀ physical z,(((markedIncidentEdges tail head ktail khead π H₀ HK physical z).filter
 100       (fun e=>markedActivePhysical tail head ktail khead π H₀ HK e∈B.filter (markedStarResource X Z))).card : ℝ)≤σ) :
 101     let DR := physicalEdgeGraph (A.image (markedPhysicalEdge tail head ktail khead π))
 102     let DS := markedFiniteSpokeGraph S;
 103     (∀ v,(DR.degree v:ℝ)≤δ) ∧ (∀ z,(DS.degree z:ℝ)≤δ) ∧
 104     (∀ v,((DR.neighborFinset v∩X).card:ℝ)≤σ) ∧
 105     (∀ v,((Z.filter (fun u=>DS.Adj (Sum.inl v) (Sum.inr u))).card:ℝ)≤σ) ∧
 106     (∀ u,((X.filter (fun v=>DS.Adj (Sum.inl v) (Sum.inr u))).card:ℝ)≤σ) := by
