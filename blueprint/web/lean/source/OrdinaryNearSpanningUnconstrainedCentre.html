  11 theorem ordinary_above_threshold_near_spanning_cycles
  12     (r:ℕ) (hr:3≤r) (α ε:ℝ) (hα:0<α) (hε:0<ε):
  13     ∃ η:ℝ,0<η ∧ ∃ N:ℕ,
  14     ∀ (V:Type) [Fintype V] [DecidableEq V] (G:SimpleGraph V),
  15       N≤Fintype.card V→
  16       (∀ v,(cycleThreshold r+α)*Fintype.card V≤(G.degree v:ℝ))→
  17       (∃ d:ℝ,∀ v,|(G.degree v:ℝ)-d|≤η*(Fintype.card V:ℝ))→
  18       ∃ M:Finset (OrdinaryCycle G r),
  19         (∀ C∈M,∀ C'∈M,C≠C'→Disjoint C.val.val.support.toFinset C'.val.val.support.toFinset) ∧
  20         (1-ε)*Fintype.card V≤(r:ℝ)*M.card := by
  21   obtain ⟨η₀,hη₀,N₀,hpack⟩:=ordinary_above_threshold_almost_regular_near_spanning_cycles r hr α ε hα hε
  22   let η:=min η₀ (α/4)
  23   have hη:0<η:=by dsimp [η];positivity
  24   have hη0:η≤η₀:=min_le_left _ _
  25   have hηα:η≤α/4:=min_le_right _ _
  26   refine ⟨η,hη,max 1 N₀,?_⟩
  27   intro V _ _ G hn hdeg ⟨d,hreg⟩
  28   have hn0:0<Fintype.card V:=by omega
  29   have hnpos:0<(Fintype.card V:ℝ):=by exact_mod_cast hn0
  30   letI:Nonempty V:=Fintype.card_pos_iff.mp hn0
  31   let v:V:=Classical.choice inferInstance
  32   have hhalf:(1/2:ℝ)≤cycleThreshold r:=cycleThreshold_half_le hr
  33   have hdeglo:(1/2+α)*(Fintype.card V:ℝ)≤(G.degree v:ℝ):=by
  34     have hh:=mul_le_mul_of_nonneg_right (show (1/2+α:ℝ)≤cycleThreshold r+α by linarith) hnpos.le
  35     exact hh.trans (hdeg v)
  36   have hregv:-(η*(Fintype.card V:ℝ))≤(G.degree v:ℝ)-d:=(abs_le.mp (hreg v)).1
  37   have hdiff:(G.degree v:ℝ)-d≤η*(Fintype.card V:ℝ):=(abs_le.mp (hreg v)).2
  38   have hηn:=mul_le_mul_of_nonneg_right hηα hnpos.le
  39   have hdlo:(Fintype.card V:ℝ)/2≤d:=by nlinarith
  40   let d':=min d (Fintype.card V:ℝ)
  41   have hd'lo:(Fintype.card V:ℝ)/2≤d' := by dsimp [d'];apply le_min hdlo;linarith
  42   have hd'hi:d'≤Fintype.card V := min_le_right _ _
  43   have hregular:∀ u,|(G.degree u:ℝ)-d'|≤η₀*(Fintype.card V:ℝ) := by
  44     intro u
  45     have hdegup:(G.degree u:ℝ)≤Fintype.card V := by
  46       exact_mod_cast (Nat.le_of_lt (G.degree_lt_card_verts u))
  47     have hreg0:|(G.degree u:ℝ)-d|≤η₀*(Fintype.card V:ℝ) :=
  48       (hreg u).trans (mul_le_mul_of_nonneg_right hη0 hnpos.le)
  49     by_cases hd:d≤Fintype.card V
  50     · simpa [d',min_eq_left hd] using hreg0
  51     · have hnd:(Fintype.card V:ℝ)≤d:=le_of_not_ge hd
  52       have h1:(G.degree u:ℝ)≤d:=hdegup.trans hnd
  53       dsimp [d']
  54       rw [min_eq_right hnd,abs_of_nonpos (sub_nonpos.mpr hdegup),abs_of_nonpos (sub_nonpos.mpr h1)] at *
  55       linarith
  56   exact hpack V G ((le_max_right 1 N₀).trans hn) d' hd'lo hd'hi hdeg hregular
