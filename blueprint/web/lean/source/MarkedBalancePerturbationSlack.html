  11     ∀ n I err:ℝ,0≤n→iota*n≤I→
  12       err≤(etaBal/2)*I → err≤(x/4)*n →
  13       err+(m+4:ℕ)*ζ*n≤etaBal*I ∧ err+(m+4:ℕ)*ζ*n≤(x/2)*n := by
  14   let a:ℝ:=(m+4:ℕ)
  15   have ha:0<a:=by dsimp [a];positivity
  16   let ζcap:=min (etaBal*iota/(2*(a+1))) (x/(4*(a+1)))
  17   have hcap:0<ζcap:=by
  18     dsimp [ζcap]
  19     exact lt_min (div_pos (mul_pos hη hi) (by linarith)) (div_pos hx (by linarith))
  20   refine ⟨ζcap,hcap,?_⟩

  22   have h1:ζ≤etaBal*iota/(2*(a+1)):=hζcap.trans (min_le_left _ _)

  24   have h1':ζ*(2*(a+1))≤etaBal*iota:=(le_div_iff₀ (by positivity)).mp h1
  25   have h2':ζ*(4*(a+1))≤x:=(le_div_iff₀ (by positivity)).mp h2

  30     nlinarith [mul_nonneg hζ hn,mul_nonneg ha.le (mul_nonneg hζ hn)]
