   9 theorem restricted_physical_packing_from_old_and_new
  10     {U V:Type*} [DecidableEq V]
  11     (remaining old new:U→Finset (Sym2 V)) (Pack:Finset (Sym2 V))
  12     (hcover:∀i,remaining i⊆old i∪new i)
  13     (hold:∀i j,i≠j→Disjoint (old i) (old j))
  14     (holdPack:∀i,old i⊆Pack)
  15     (hnew:∀i j,i≠j→Disjoint (new i) (new j))
  16     (hnewPack:∀i,Disjoint (new i) Pack):
  17     ∀i j,i≠j→Disjoint (remaining i) (remaining j):=by
  18   intro i j hij
  19   apply Finset.disjoint_left.mpr
  20   intro e hi hj
  21   rcases Finset.mem_union.mp (hcover i hi) with hiold | hinew
  22   · rcases Finset.mem_union.mp (hcover j hj) with hjold | hjnew
  23     · exact Finset.disjoint_left.mp (hold i j hij) hiold hjold
  24     · exact Finset.disjoint_left.mp (hnewPack j) hjnew (holdPack i hiold)
  25   · rcases Finset.mem_union.mp (hcover j hj) with hjold | hjnew
  26     · exact Finset.disjoint_left.mp (hnewPack i) hinew (holdPack j hjold)
  27     · exact Finset.disjoint_left.mp (hnew i j hij) hinew hjnew
