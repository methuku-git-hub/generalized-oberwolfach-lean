  12 def fixedSubsetComplEquiv {V : Type*} [Fintype V] {r : ℕ}
  13     (hr : r ≤ Fintype.card V) : FixedSubset V r ≃ FixedSubset V (Fintype.card V - r) where
  14   toFun S := ⟨S.1ᶜ, by rw [Finset.card_compl, S.2]⟩
  15   invFun T := ⟨T.1ᶜ, by simp only [Finset.card_compl, T.2]; omega⟩
  16   left_inv S := by apply Subtype.ext; exact compl_compl _
  17   right_inv T := by apply Subtype.ext; exact compl_compl _

  19 theorem fixedSubsetComplEquiv_map {V : Type*} [Fintype V] {r : ℕ}
  20     (hr : r ≤ Fintype.card V) (e : Equiv.Perm V) (S : FixedSubset V r) :
  21     fixedSubsetComplEquiv hr (fixedSubsetMap e S) =
  22       fixedSubsetMap e (fixedSubsetComplEquiv hr S) := by
  23   apply Subtype.ext
  24   ext x
  25   simp only [fixedSubsetComplEquiv, fixedSubsetMap, Equiv.coe_fn_mk,
  26     Finset.mem_compl, Finset.mem_map_equiv]

  28 theorem fixedSubset_mgf_min
  29     {r : ℕ} {V : Type*} [Fintype V]
  30     [Nonempty (FixedSubset V r)] (hr : r ≤ Fintype.card V)
  31     (f : FixedSubset V r → ℝ) {b : ℝ} (hb : 0 ≤ b)
  32     (hexchange : ∀ S u v, |f S - f (fixedSubsetMap (Equiv.swap u v) S)| ≤ b)
  33     (t : ℝ) :
  34     (FiniteLaw.uniform (FixedSubset V r)).expectation
  35       (fun S => Real.exp (t * (f S - (FiniteLaw.uniform (FixedSubset V r)).expectation f))) ≤
  36       Real.exp (t ^ 2 * (min r (Fintype.card V - r) : ℕ) * b ^ 2 / 8) := by
  37   by_cases hsmall : r ≤ Fintype.card V - r
  38   · simpa only [min_eq_left hsmall] using fixedSubset_mgf hr f hb hexchange t
  39   · rw [min_eq_right (le_of_not_ge hsmall)]
  40     let e := fixedSubsetComplEquiv (V := V) hr
  41     let : Nonempty (FixedSubset V (Fintype.card V - r)) :=
  42       fixedSubset_nonempty (Nat.sub_le _ _)
  43     have hs (S : FixedSubset V (Fintype.card V - r)) (u v : V) :
  44         e.symm (fixedSubsetMap (Equiv.swap u v) S) =
  45           fixedSubsetMap (Equiv.swap u v) (e.symm S) := by
  46       apply e.injective
  47       rw [e.apply_symm_apply, fixedSubsetComplEquiv_map, e.apply_symm_apply]
  48     have hchange : ∀ S u v, |f (e.symm S) - f (e.symm (fixedSubsetMap (Equiv.swap u v) S))| ≤ b := by
  49       intro S u v
  50       rw [hs]
  51       exact hexchange _ _ _
  52     have h := fixedSubset_mgf (Nat.sub_le (Fintype.card V) r) (fun S => f (e.symm S)) hb hchange t
  53     have he (Y : FixedSubset V r → ℝ) :
  54         (FiniteLaw.uniform (FixedSubset V (Fintype.card V - r))).expectation (fun S => Y (e.symm S)) =
  55         (FiniteLaw.uniform (FixedSubset V r)).expectation Y := by
  56       rw [← finiteLaw_map_expectation, finiteLaw_uniform_map_equiv e.symm]
  57     rw [he f, he (fun S => Real.exp (t * (f S -
  58       (FiniteLaw.uniform (FixedSubset V r)).expectation f)))] at h
  59     exact h

  61 theorem fixedSubset_absolute_tail
  62     {r : ℕ} {V : Type*} [Fintype V]
  63     [Nonempty (FixedSubset V r)] (hr : r ≤ Fintype.card V)
  64     (f : FixedSubset V r → ℝ) {b : ℝ} (hb : 0 ≤ b)
  65     (hexchange : ∀ S u v, |f S - f (fixedSubsetMap (Equiv.swap u v) S)| ≤ b)
  66     {t : ℝ} (ht : 0 ≤ t) :
  67     (FiniteLaw.uniform (FixedSubset V r)).probability
  68       (fun S => t ≤ |f S - (FiniteLaw.uniform (FixedSubset V r)).expectation f|) ≤
  69       2 * Real.exp (-2 * t ^ 2 / ((min r (Fintype.card V - r) : ℕ) * b ^ 2)) := by
  70   apply finiteLaw_mgf_absolute_tail _ _ (by positivity) _ ht
  71   intro u
  72   simpa only [mul_assoc] using fixedSubset_mgf_min hr f hb hexchange u
