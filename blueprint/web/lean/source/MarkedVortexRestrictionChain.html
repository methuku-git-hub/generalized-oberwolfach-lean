 361 /-- Resource layers of a decreasing finite-set sequence are disjoint. -/
 362 theorem decreasing_resource_layers_disjoint {X:Type} [DecidableEq X]
 363     (B R:ℕ→Finset X) (J:Finset ℕ)
 364     (hB:∀ i,B (i+1)⊆B i)
 365     (hR:∀ i∈J,R i⊆B i\B (i+1)):
 366     ∀ i∈J,∀ j∈J,i≠j→Disjoint (R i) (R j):=by
 367   have hanti:Antitone B:=antitone_nat_of_succ_le hB
 368   intro i hi j hj hne
 369   apply Finset.disjoint_left.mpr
 370   intro z hzi hzj
 371   by_cases hij:i<j
 372   · have hnext:=hanti (Nat.succ_le_of_lt hij)
 373     exact (Finset.mem_sdiff.mp (hR i hi hzi)).2 (hnext (Finset.mem_sdiff.mp (hR j hj hzj)).1)
 374   · have hji : j < i := by omega
 375     have hnext:=hanti (Nat.succ_le_of_lt hji)
 376     exact (Finset.mem_sdiff.mp (hR j hj hzj)).2 (hnext (Finset.mem_sdiff.mp (hR i hi hzi)).1)

 378 /-- Exact finite conservation: each covered layer and the final remainder
 379 partition the initial resource set. No infinite sequence hypothesis is needed. -/
 380 theorem finite_resource_layers_partition {X:Type} [DecidableEq X]
 381     (B R:ℕ→Finset X) (n:ℕ)
 382     (hB:∀ i,i<n→B (i+1)⊆B i)
 383     (hR:∀ i,i<n→R i=B i\B (i+1)):
 384     (Finset.range n).biUnion R ∪ B n=B 0:=by
 385   induction n with
 386   | zero=>simp
 387   | succ n ih=>
 388     have hBn:B (n+1)⊆B n:=hB n (by omega)
 389     have hRn:R n=B n\B (n+1):=hR n (by omega)
 390     have hprev:(Finset.range n).biUnion R ∪ B n=B 0:=
 391       ih (fun i hi=>hB i (by omega)) (fun i hi=>hR i (by omega))
 392     rw [Finset.range_add_one,Finset.biUnion_insert,hRn]
 393     calc
 394       (B n\B (n+1) ∪ (Finset.range n).biUnion R) ∪ B (n+1)
 395           = (Finset.range n).biUnion R ∪ ((B n\B (n+1)) ∪ B (n+1)):=by ac_rfl
 396       _ = (Finset.range n).biUnion R ∪ B n:=by rw [Finset.sdiff_union_of_subset hBn]
 397       _ = B 0:=hprev

 399 /-- Disjointness of the covered layers needs only the finite sequence of
 400 monotonicity and layer-containment facts used by the iteration. -/
 401 theorem finite_resource_layers_disjoint {X:Type} [DecidableEq X]
 402     (B R:ℕ→Finset X) (n:ℕ)
 403     (hB:∀ i,i<n→B (i+1)⊆B i)
 404     (hR:∀ i,i<n→R i⊆B i\B (i+1)):
 405     ∀ i<n,∀ j<n,i≠j→Disjoint (R i) (R j):=by
 406   have hmono:∀ i j,i≤j→j≤n→B j⊆B i:=by
 407     intro i j hij
 408     induction hij with
 409     | refl=>intro _;exact Finset.Subset.rfl
 410     | @step j hij ih=>
 411       intro hj
 412       exact (hB j (by omega)).trans (ih (by omega))
 413   intro i hi j hj hne
 414   apply Finset.disjoint_left.mpr
 415   intro z hzi hzj
 416   by_cases hij:i<j
 417   · have hnext:=hmono (i+1) j (by omega) (by omega)
 418     exact (Finset.mem_sdiff.mp (hR i hi hzi)).2 (hnext (Finset.mem_sdiff.mp (hR j hj hzj)).1)
 419   · have hji : j < i := by omega
 420     have hnext:=hmono (j+1) i (by omega) (by omega)
 421     exact (Finset.mem_sdiff.mp (hR j hj hzj)).2 (hnext (Finset.mem_sdiff.mp (hR i hi hzi)).1)

 423 /-- A finite prefix of successive deletions is decreasing. -/
 424 theorem finite_resource_prefix_subset {X:Type} [DecidableEq X]
 425     (B:ℕ→Finset X) (n:ℕ) (hB:∀ i,i<n→B (i+1)⊆B i)
 426     (i j:ℕ) (hij:i≤j) (hjn:j≤n):B j⊆B i:=by
 427   induction hij with
 428   | refl=>exact Finset.Subset.rfl
 429   | @step j hij ih=>
 430     exact (hB j (by omega)).trans (ih (by omega))

 432 /-- The final remainder is disjoint from all resources covered earlier. -/
 433 theorem finite_resource_layers_terminal_disjoint {X:Type} [DecidableEq X]
 434     (B R:ℕ→Finset X) (n:ℕ)
 435     (hB:∀ i,i<n→B (i+1)⊆B i)
 436     (hR:∀ i,i<n→R i⊆B i\B (i+1)):
 437     Disjoint ((Finset.range n).biUnion R) (B n):=by
 438   apply Finset.disjoint_left.mpr
 439   intro z hz hzn
 440   obtain ⟨i,hi,hzi⟩:=Finset.mem_biUnion.mp hz
 441   have hin:i<n:=Finset.mem_range.mp hi
 442   have hsub:=finite_resource_prefix_subset B n hB (i+1) n (by omega) (by omega)
 443   exact (Finset.mem_sdiff.mp (hR i hin hzi)).2 (hsub hzn)

 445 /-- Exact layer decompositions and a terminal decomposition combine into
 446 a decomposition of the initial resource set. -/
 447 theorem finite_resource_layer_decompositions {X W:Type} [DecidableEq X] [DecidableEq W]
 448     (F:W→Finset X) (B R:ℕ→Finset X) (n:ℕ)
 449     (hB:∀ i,i<n→B (i+1)⊆B i)
 450     (hR:∀ i,i<n→R i=B i\B (i+1))
 451     (hdec:∀ i,i<n→HasResourceDecomposition F (R i))
 452     (hterm:HasResourceDecomposition F (B n)):
 453     HasResourceDecomposition F (B 0):=by
 454   have hsub:∀ i,i<n→R i⊆B i\B (i+1):=by intro i hi;rw [hR i hi]
 455   have hdis:=finite_resource_layers_disjoint B R n hB hsub
 456   have hlayers:HasResourceDecomposition F ((Finset.range n).biUnion R):=by
 457     apply resource_decomposition_biUnion F R (Finset.range n)

 459     · intro i hi j hj hne;exact hdis i (Finset.mem_range.mp hi) j (Finset.mem_range.mp hj) hne
 460   have htermDis:=finite_resource_layers_terminal_disjoint B R n hB hsub
 461   have hfull:=resource_decomposition_union F ((Finset.range n).biUnion R) (B n) hlayers hterm htermDis
 462   rw [finite_resource_layers_partition B R n hB hR] at hfull
 463   exact hfull
