  13  theorem two_root_cycle_image (p:Fin n→V) (v w:V) (ρ:Equiv.Perm (Fin (n+2))):
  14     Finset.univ.image (fun i=>Fin.cons (α:=fun _=>V) v (Fin.cons w p) (ρ i))=
  15       insert v (insert w (Finset.univ.image p)):=by
  16   ext x
  17   simp only [Finset.mem_image,Finset.mem_univ,true_and,Finset.mem_insert]
  18   constructor
  19   · rintro ⟨i,hi⟩
  20     have hh:∀ j:Fin (n+2),Fin.cons (α:=fun _=>V) v (Fin.cons w p) j=x→
  21         x=v∨x=w∨∃ k,p k=x:=by
  22       intro j
  23       refine Fin.cases ?_ (fun j=>?_) j
  24       · intro h;exact Or.inl h.symm
  25       · refine Fin.cases ?_ (fun j=>?_) j
  26         · intro h;exact Or.inr (Or.inl h.symm)
  27         · intro h;exact Or.inr (Or.inr ⟨j,h⟩)
  28     exact hh (ρ i) hi
  29   · rintro (rfl|rfl|⟨i,hi⟩)
  30     · exact ⟨ρ.symm 0,by simp⟩
  31     · exact ⟨ρ.symm (0:Fin (n+1)).succ,by simp⟩
