  15 theorem conjunction_indicator_change {J : Type*} [Fintype J]
  16     (P Q D : J → Prop) (h : ∀ j,¬D j → (P j ↔ Q j)) :
  17     |(if ∀ j,Q j then (1:ℝ) else 0)-(if ∀ j,P j then 1 else 0)|≤
  18       ∑ j,if D j then (1:ℝ) else 0 := by
  19   classical
  20   by_cases he : ∃ j,D j
  21   · obtain ⟨j,hj⟩ := he
  22     have hsum := Finset.single_le_sum (s:=Finset.univ) (f:=fun j=>if D j then (1:ℝ) else 0)
  23       (fun _ _=>by split_ifs <;> norm_num) (Finset.mem_univ j)
  24     rw [if_pos hj] at hsum
  25     have hdiff : |(if ∀ j,Q j then (1:ℝ) else 0)-(if ∀ j,P j then 1 else 0)|≤1 := by
  26       split_ifs <;> norm_num
  27     exact hdiff.trans hsum
  28   · have heq : (∀ j,P j) ↔ (∀ j,Q j) := forall_congr' (fun j=>h j (fun hd=>he ⟨j,hd⟩))
  29     simp only [heq,sub_self,abs_zero]
  30     exact Finset.sum_nonneg (fun _ _=>by split_ifs <;> norm_num)

  38 theorem rootedPatternValue_change {I J X : Type*} [Fintype J]
  39     (active : J → (I → X) → Prop) (a b : J → X⊕I)
  40     (σ τ : J → Equiv.Perm X) (G G' D : J → X → X → Prop)
  41     (hchange : ∀ j x y,¬D j x y → (G j x y ↔ G' j x y)) (f : I → X) :
  42     |rootedPatternValue active a b σ τ G' f-rootedPatternValue active a b σ τ G f|≤
  43       ∑ j,if D j (σ j (rootedPositionValue f (a j))) (τ j (rootedPositionValue f (b j))) then (1:ℝ) else 0 := by
  44   have hh := conjunction_indicator_change
  45     (fun j=>active j f → G j (σ j (rootedPositionValue f (a j))) (τ j (rootedPositionValue f (b j))))
  46     (fun j=>active j f → G' j (σ j (rootedPositionValue f (a j))) (τ j (rootedPositionValue f (b j))))
  47     (fun j=>D j (σ j (rootedPositionValue f (a j))) (τ j (rootedPositionValue f (b j))))
  48     (by
  49       intro j hj
  50       constructor
  51       · intro hq ha;exact (hchange j _ _ hj).mp (hq ha)
  52       · intro hq ha;exact (hchange j _ _ hj).mpr (hq ha))
  53   convert hh using 1 <;> unfold rootedPatternValue <;> split_ifs <;> rfl

  55 theorem weighted_rooted_pattern_stability {I J X C : Type*}
  56     [Fintype I] [Fintype J] [Fintype X] [Fintype C] [DecidableEq I]
  57     (encode : C → (I → X)) (hinj : Function.Injective encode)
  58     (active : J → (I → X) → Prop) (a b : J → X⊕I)
  59     (hab : ∀ j,RootedEdgeHasFreeEndpoint (a j) (b j))
  60     (σ τ : J → Equiv.Perm X) (G G' D : J → X → X → Prop)
  61     (hchange : ∀ j x y,¬D j x y → (G j x y ↔ G' j x y))
  62     (d M : ℝ) (hM : 0≤M)
  63     (hout : ∀ j x,(∑ y,if D j x y then (1:ℝ) else 0)≤d)
  64     (hin : ∀ j y,(∑ x,if D j x y then (1:ℝ) else 0)≤d)
  65     (w : C → ℝ) (hw : ∀ c,0≤w c ∧ w c≤M) :
  66     |(∑ c,w c*rootedPatternValue active a b σ τ G' (encode c))-
  67       (∑ c,w c*rootedPatternValue active a b σ τ G (encode c))|≤
  68       M*Fintype.card J*(Fintype.card X:ℝ)^(Fintype.card I-1)*d := by
  69   classical
  70   let count (f : I → X) : ℝ :=
  71     ∑ j,if D j (σ j (rootedPositionValue f (a j))) (τ j (rootedPositionValue f (b j))) then 1 else 0
  72   have hcount (f : I → X) : 0≤count f := Finset.sum_nonneg (fun _ _=>by split_ifs <;> norm_num)
  73   have hdiff (c : C) :
  74       |w c*rootedPatternValue active a b σ τ G' (encode c)-w c*rootedPatternValue active a b σ τ G (encode c)|≤
  75       M*count (encode c) := by
  76     rw [← mul_sub,abs_mul,abs_of_nonneg (hw c).1]
  77     exact (mul_le_mul_of_nonneg_left (rootedPatternValue_change active a b σ τ G G' D hchange (encode c))
  78       (hw c).1).trans (mul_le_mul_of_nonneg_right (hw c).2 (hcount _))
  79   have htotal : (∑ f : I → X,count f)≤
  80       (Fintype.card J:ℝ)*(Fintype.card X:ℝ)^(Fintype.card I-1)*d := by
  81     unfold count
  82     rw [Finset.sum_comm]
  83     calc
  84       _ ≤ ∑ _j : J,(Fintype.card X:ℝ)^(Fintype.card I-1)*d :=
  85         Finset.sum_le_sum (fun j _=>rooted_edge_change_count (D j) d (hout j) (hin j) (a j) (b j) (hab j) (σ j) (τ j))
  86       _ = _ := by simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul];ring
  87   calc
  88     _ ≤ ∑ c,|w c*rootedPatternValue active a b σ τ G' (encode c)-w c*rootedPatternValue active a b σ τ G (encode c)| := by
  89       rw [← Finset.sum_sub_distrib]
  90       exact Finset.abs_sum_le_sum_abs _ _
  91     _ ≤ ∑ c,M*count (encode c) := Finset.sum_le_sum (fun c _=>hdiff c)
  92     _ = M*∑ c,count (encode c) := (Finset.mul_sum _ _ _).symm
  93     _ ≤ M*∑ f : I → X,count f := mul_le_mul_of_nonneg_left (finite_injective_sum_le encode hinj count hcount) hM
  94     _ ≤ M*((Fintype.card J:ℝ)*(Fintype.card X:ℝ)^(Fintype.card I-1)*d) := mul_le_mul_of_nonneg_left htotal hM
  95     _ = _ := by ring
