  13 theorem dependent_assigned_edges_injective {J V E:Type} {A B:J→Type}
  14     (x:J↪V) (f:∀ j,A j↪E) (e:∀ j,B j≃A j) (root:E→V)
  15     (hroot:∀ j a,root (f j a)=x j):
  16     Function.Injective (fun i:Σ j,B j=>f i.1 (e i.1 i.2)):=by
  17   rintro ⟨j,a⟩ ⟨k,b⟩ he
  18   have hx:x j=x k:=by simpa only [hroot] using congrArg root he
  19   have hj:=x.injective hx
  20   subst k
  21   exact congrArg (Sigma.mk j) ((e j).injective ((f j).injective he))
