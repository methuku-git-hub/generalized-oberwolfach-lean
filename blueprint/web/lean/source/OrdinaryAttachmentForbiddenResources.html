  19     (φ:V₀↪V) (ψ:U₀↪U) (f:V₀⊕(I×W)↪V) (g:U₀⊕I↪U)
  20     (hf:∀ a,f (.inl a)=φ a) (hg:∀ a,g (.inl a)=ψ a)
  21     (S:Finset V) (B:Finset U) (hfn:∀ i w,f (.inr (i,w))∉S) (hgn:∀ i,g (.inr i)∉B)
  22     (extra:Finset (Sym2 V⊕(V×U)))
  23     (hpart:ordinaryHostResources ((ordinaryAttachmentRim G C z ρ).map f)
  24         (ordinaryMappedSpokes (ordinaryAttachmentSpokes H z ρ) f g)=
  25       ordinaryHostResources (G.map φ) (ordinaryMappedSpokes H φ ψ)∪extra)
  26     (hdis:Disjoint (ordinaryHostResources (G.map φ) (ordinaryMappedSpokes H φ ψ)) extra) :
  27     Disjoint extra (ordinaryRootResources S B) := by
  28   apply Finset.disjoint_left.mpr
  29   intro e he he'
  30   have hfull:e∈ordinaryHostResources ((ordinaryAttachmentRim G C z ρ).map f)
  31       (ordinaryMappedSpokes (ordinaryAttachmentSpokes H z ρ) f g):=by
  32     rw [hpart];exact Finset.mem_union_right _ he
  33   cases e with
  34   | inl e=>
  35     induction e using Sym2.ind with
  36     | _ v w=>
  37       obtain ⟨hv,hw⟩:=(ordinaryRootResources_rim_iff S B v w).mp he'
  38       obtain ⟨a,b,hab,ha,hb⟩:=(SimpleGraph.map_adj _ _ _ _).mp ((ordinaryHostResources_rim _ _ _).mp hfull)
  39       cases a with
  40       | inr p=>exact hfn p.1 p.2 (by simpa only [ha] using hv)
  41       | inl a=>
  42         cases b with
  43         | inr p=>exact hfn p.1 p.2 (by simpa only [hb] using hw)
  44         | inl b=>
  45           have hold:Sum.inl s(v,w)∈ordinaryHostResources (G.map φ) (ordinaryMappedSpokes H φ ψ):=by
  46             apply (ordinaryHostResources_rim _ _ _).mpr
  47             exact (SimpleGraph.map_adj _ _ _ _).mpr ⟨a,b,hab,(hf a).symm.trans ha,(hf b).symm.trans hb⟩
  48           exact Finset.disjoint_left.mp hdis hold he
  49   | inr p=>
