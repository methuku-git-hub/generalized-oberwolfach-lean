  53       (Q.incidenceFinset v).image Sum.inl ∪
  54       (Finset.univ.filter (H v)).image (fun u=>Sum.inr (v,u)) := by

 135 theorem ordinary_packing_transport {P Q F : SimpleGraph V} {Hbase H : V → U → Prop} {r : ℕ}
 136     (hQ : Q≤P) (hH : ∀ v u,H v u → Hbase v u)
 137     (M : Finset (OrdinaryWheel Q F H r))
 138     (hM : ∀ W∈M,∀ Z∈M,W≠Z → Disjoint (ordinaryWheelResources W) (ordinaryWheelResources Z)) :
 139     let M' := M.image (ordinaryWheelToLarger (F:=F) (r:=r) hQ hH)
 140     (∀ W∈M',∀ Z∈M',W≠Z → Disjoint (ordinaryWheelResources W) (ordinaryWheelResources Z)) ∧
 141     M'.biUnion ordinaryWheelResources=M.biUnion ordinaryWheelResources := by
 142   constructor
 143   · intro W hW Z hZ hne
 144     obtain ⟨w,hw,rfl⟩ := Finset.mem_image.mp hW
 145     obtain ⟨z,hz,rfl⟩ := Finset.mem_image.mp hZ
 146     exact hM w hw z hz (by intro he;exact hne (congrArg _ he))
 147   · rw [Finset.image_biUnion]
