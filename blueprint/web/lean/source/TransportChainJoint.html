  24 def chainJoint {I X : Type*}
  25     (m : I → X → ℝ) (K : I → X → X → ℝ) (s : I → I) (i : I) :
  26     (j : ℕ) → (Fin (j + 1) → X) → ℝ
  27   | 0, w => m i (w 0)
  28   | j + 1, w =>
  29       K i (w 0) (w 1) / m (s i) (w 1) *
  30         chainJoint m K s (s i) j (Fin.tail w)

  32 theorem chainJoint_nonneg {I X : Type*}
  33     (m : I → X → ℝ) (K : I → X → X → ℝ) (s : I → I)
  34     (hm : ∀ i x, 0 ≤ m i x) (hK : ∀ i x y, 0 ≤ K i x y)
  35     (i : I) (j : ℕ) (w : Fin (j + 1) → X) :
  36     0 ≤ chainJoint m K s i j w := by
  37   induction j generalizing i with
  38   | zero => exact hm i (w 0)
  39   | succ j ih =>
  40     exact mul_nonneg (div_nonneg (hK i (w 0) (w 1)) (hm (s i) (w 1)))
  41       (ih (s i) (Fin.tail w))

  43 theorem chainJoint_endpoints {I X : Type*} [Fintype X]
  44     (m : I → X → ℝ) (K : I → X → X → ℝ) (s : I → I)
  45     (i : I) (j : ℕ) (f : X → X → ℝ) :
  46     (∑ w : Fin (j + 1) → X,
  47       chainJoint m K s i j w * f (w 0) (w (Fin.last j))) =
  48       ∑ x, ∑ z, walkTransport m K s i j x z * f x z := by
  49   induction j generalizing i f with
  50   | zero =>
  51     rw [sum_fin_tuple_one]
  52     simp [chainJoint, walkTransport, diagonalTransport, ite_mul]
  53   | succ j ih =>
  54     rw [sum_fin_tuple_cons]
  55     simp only [chainJoint, Fin.cons_zero, Fin.cons_one, Fin.tail_cons, Fin.cons_last]
  56     apply Finset.sum_congr rfl
  57     intro x hx
  58     have heq (v : Fin (j + 1) → X) :
  59         K i x (v 0) / m (s i) (v 0) * chainJoint m K s (s i) j v *
  60           f x (v (Fin.last j)) =
  61         chainJoint m K s (s i) j v *
  62           (K i x (v 0) / m (s i) (v 0) * f x (v (Fin.last j))) := by ring
  63     simp_rw [heq]
  64     rw [ih (s i) (fun y z => K i x y / m (s i) y * f x z)]
  65     change _ = ∑ z, composeTransport (m (s i)) (K i) (walkTransport m K s (s i) j) x z * f x z
  66     unfold composeTransport
  67     simp_rw [Finset.sum_mul]
  68     rw [Finset.sum_comm]
  69     apply Finset.sum_congr rfl
  70     intro y hy
  71     apply Finset.sum_congr rfl
  72     intro z hz
  73     ring
