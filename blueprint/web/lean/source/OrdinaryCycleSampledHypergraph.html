  27 theorem retained_ordinary_cycle_degree {V:Type} [Fintype V] [DecidableEq V]
  28     (G:SimpleGraph V) (r:ℕ) [DecidableEq (OrdinaryCycle G r)]
  29     (σ:OrdinaryCycle G r→Bool) (v:V):
  30     (Fintype.card (HypergraphEdgesAt (retainedOrdinaryCycleVertices (G:=G) (r:=r) (σ:=σ)) v):ℝ)=
  31       independentCycleCount (fun C=>v∈C.val.val.support) σ := by
  32   let f : HypergraphEdgesAt (retainedOrdinaryCycleVertices (G:=G) (r:=r) (σ:=σ)) v ≃
  33       {C:OrdinaryCycle G r // σ C=true ∧ v∈C.val.val.support} := {
  34     toFun := fun e =>⟨e.val.val,⟨e.val.property,by simpa [retainedOrdinaryCycleVertices] using e.property⟩⟩
  35     invFun := fun C =>⟨⟨C.val,C.property.1⟩,by simpa [retainedOrdinaryCycleVertices] using C.property.2⟩
  36     left_inv := fun e => by cases e with | mk e h => cases e; rfl
  37     right_inv := fun C => by cases C; rfl }
  38   rw [Fintype.card_congr f]
  39   simp only [Fintype.card_subtype,Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero,
  40     independentCycleCount]
  41   apply Finset.sum_congr rfl
  42   intro C _
  43   by_cases h1:σ C=true <;> by_cases h2:v∈C.val.val.support <;> simp [h1,h2]
