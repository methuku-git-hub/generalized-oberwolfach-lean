   8 theorem finiteLaw_prod_swap {A B:Type} [Fintype A] [Fintype B]
   9     (μ:FiniteLaw A) (ν:FiniteLaw B):
  10     (μ.prod ν).map Prod.swap=ν.prod μ:=by
  11   apply finiteLaw_ext
  12   rintro ⟨b,a⟩
  13   simp only [FiniteLaw.map,FiniteLaw.prod,Fintype.sum_prod_type,Prod.swap,Prod.mk.injEq]
  14   have heach (x:A):(∑ y:B,if y=b ∧ x=a then μ.mass x*ν.mass y else 0)=
  15       if x=a then μ.mass x*ν.mass b else 0:=by
  16     by_cases h:x=a <;> simp [h]

  18     _ = ∑ x:A,if x=a then μ.mass x*ν.mass b else 0:=by
