  14 theorem boundary_selection_supports (P G:SimpleGraph V) (A Z:Finset V)
  15     (τ κ ε:ℝ) (hτ:0≤τ) (hτ1:τ≤1) (hκ:0<κ) (hκ1:κ≤1)
  16     (hsize:κ*Fintype.card V≤(A.card:ℝ)) (herr:ε≤τ^2*A.card)
  17     (hrow:∀ x,|((A∩G.neighborFinset x).card:ℝ)-τ*(A∩P.neighborFinset x).card|≤ε)
  18     (hpair:∀ x y,x≠y→
  19       |(((A∩G.neighborFinset x)∩(A∩G.neighborFinset y)).card:ℝ)-
  20         τ^2*((A∩P.neighborFinset x)∩(A∩P.neighborFinset y)).card|≤ε)
  21     (hcol:∀ v∈A,|((Z.filter (fun x=>v∈A∩G.neighborFinset x)).card:ℝ)-
  22       τ*(Z∩P.neighborFinset v).card|≤ε):
  23     (∀ x,((A∩G.neighborFinset x).card:ℝ)≤2*τ*A.card) ∧
  24     (∀ x y,x≠y→(((A∩G.neighborFinset x)∩(A∩G.neighborFinset y)).card:ℝ)≤
  25       (3*τ/κ)^2*A.card) ∧
  26     (∀ v,((Z.filter (fun x=>v∈A∩G.neighborFinset x)).card:ℝ)≤(3*τ/κ)*A.card):=by
  27   have ha:0≤(A.card:ℝ):=Nat.cast_nonneg _
  28   have hτsq:τ^2≤τ:=by nlinarith
  29   have hετ:ε≤τ*A.card:=herr.trans (mul_le_mul_of_nonneg_right hτsq ha)
  30   have hdiv:τ≤3*τ/κ:=by apply (le_div_iff₀ hκ).mpr;nlinarith
  31   have hsq:2*τ^2≤(3*τ/κ)^2:=by
  32     have hh:3*τ≤3*τ/κ:=by apply (le_div_iff₀ hκ).mpr;nlinarith
  33     have hp:0≤3*τ/κ:=by positivity
  34     nlinarith [sq_nonneg τ]
  35   refine ⟨?_,?_,?_⟩
  36   · intro x
  37     have hcount:((A∩P.neighborFinset x).card:ℝ)≤A.card:=
  38       Nat.cast_le.mpr (Finset.card_le_card Finset.inter_subset_left)
  39     have hh:=(abs_le.mp (hrow x)).2
  40     have hm:=mul_le_mul_of_nonneg_left hcount hτ
  41     linarith
  42   · intro x y hxy
  43     have hcount:(((A∩P.neighborFinset x)∩(A∩P.neighborFinset y)).card:ℝ)≤A.card:=
  44       Nat.cast_le.mpr (Finset.card_le_card (Finset.inter_subset_left.trans Finset.inter_subset_left))
  45     have hh:=(abs_le.mp (hpair x y hxy)).2
  46     have hm:=mul_le_mul_of_nonneg_left hcount (sq_nonneg τ)
  47     have hn:=mul_le_mul_of_nonneg_right hsq ha
  48     linarith
  49   · intro v
  50     by_cases hv:v∈A
  51     · have hcount:((Z∩P.neighborFinset v).card:ℝ)≤Fintype.card V:=
  52         Nat.cast_le.mpr (Finset.card_le_univ _)
  53       have hh:=(abs_le.mp (hcol v hv)).2
  54       have hm:=mul_le_mul_of_nonneg_left hcount hτ
  55       have hsize':(Fintype.card V:ℝ)≤(A.card:ℝ)/κ:=by
  56         apply (le_div_iff₀ hκ).mpr;simpa only [mul_comm] using hsize
  57       have hs:=mul_le_mul_of_nonneg_left hsize' hτ
  58       have hthree:τ*((A.card:ℝ)/κ)+τ*A.card≤(3*τ/κ)*A.card:=by
  59         apply (mul_le_mul_iff_of_pos_right hκ).mp
  60         field_simp
  61         nlinarith [mul_nonneg hτ ha,mul_le_mul_of_nonneg_left hκ1 (mul_nonneg hτ ha)]
  62       linarith
  63     · have he:Z.filter (fun x=>v∈A∩G.neighborFinset x)=∅:=by
  64         ext x;simp [hv]
  65       rw [he];simp only [Finset.card_empty,Nat.cast_zero];positivity
