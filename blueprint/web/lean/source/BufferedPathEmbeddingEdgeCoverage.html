  10 theorem regular_two_buffered_embedding_edge_coverage
  11     {V W I:Type*} [Fintype V] [DecidableEq V] [DecidableEq I]
  12     (G:SimpleGraph V) (hG:G.IsRegularOfDegree 2)
  13     (n:I→ℕ) (ext:∀i,Fin (n i+5)↪V) (inner:∀i,Fin (n i+1)↪V)
  14     (hinner:∀i k,inner i k=ext i ⟨k.val+2,by omega⟩)
  15     (hwalk:∀i,∃p:G.Walk (ext i ⟨0,by omega⟩) (ext i ⟨n i+4,by omega⟩),
  16       p.length=n i+4 ∧ p.support=List.ofFn (ext i))
  17     (hdis:∀i j,i≠j→Disjoint (Finset.univ.image (ext i)) (Finset.univ.image (ext j)))
  18     (A:Finset I) (target:∀i,Fin (n i+1)→W)
  19     (φ:{x // x∈A.biUnion (fun i=>Finset.univ.image (inner i))}→W)
  20     (hmatch:∀i (hi:i∈A) (k:Fin (n i+1)),
  21       φ ⟨inner i k,Finset.mem_biUnion.mpr ⟨i,hi,Finset.mem_image.mpr ⟨k,Finset.mem_univ _,rfl⟩⟩⟩=target i k):
  22     ∀x y,G.Adj x.val y.val→∃i∈A,∃k:Fin (n i),
  23       (φ x=target i k.castSucc ∧ φ y=target i k.succ) ∨
  24       (φ y=target i k.castSucc ∧ φ x=target i k.succ):=by
  25   classical
  26   intro x y hxy
  27   obtain ⟨i,hi,hxi,hyi⟩:=regular_two_buffered_walk_edge_cover G hG n ext inner hinner hwalk hdis A
  28     x.val y.val x.property y.property hxy
  29   obtain ⟨p,_,hp⟩:=Finset.mem_image.mp hxi
  30   obtain ⟨q,_,hq⟩:=Finset.mem_image.mp hyi
  31   have hfx:φ x=target i p:=by simpa only [hp] using hmatch i hi p
  32   have hfy:φ y=target i q:=by simpa only [hq] using hmatch i hi q
  33   have hsource:G.Adj (inner i p) (inner i q):=by simpa only [hp,hq] using hxy
  34   obtain ⟨walk,hlen,hsupport⟩:=hwalk i
  35   have hstep:∀k:Fin (n i+4),G.Adj (ext i ⟨k.val,by omega⟩) (ext i ⟨k.val+1,by omega⟩):=by
  36     intro k
  37     exact walk_ofFn_support_adjacent G (n i+4) (ext i) walk hlen hsupport k
  38   have hindex:=regular_two_buffered_path_adjacent_indices G hG (n i) (ext i) (inner i) (hinner i) hstep p q hsource
  39   rcases hindex with hpq|hqp
  40   · have hpbound:p.val<n i:=by omega
  41     let k:Fin (n i):=⟨p.val,hpbound⟩
  42     have hp':k.castSucc=p:=Fin.ext rfl
  43     have hq':k.succ=q:=Fin.ext hpq
  44     refine ⟨i,hi,k,Or.inl ?_⟩
  45     simpa only [hp',hq'] using And.intro hfx hfy
  46   · have hqbound:q.val<n i:=by omega
  47     let k:Fin (n i):=⟨q.val,hqbound⟩
  48     have hq':k.castSucc=q:=Fin.ext rfl
  49     have hp':k.succ=p:=Fin.ext hqp
  50     refine ⟨i,hi,k,Or.inr ?_⟩
  51     simpa only [hq',hp'] using And.intro hfy hfx
