  16 theorem finiteLaw_singleton_patterns_tail_at_scale
  17     {E I : Type*} [Fintype E] [DecidableEq E]
  18     {State : E → Type*} [∀ e, Fintype (State e)]
  19     (μ : ∀ e, FiniteLaw (State e)) (C : Finset I) (e : I → E)
  20     (s : I → ∀ j, State j) (hinj : Set.InjOn e C)
  21     {n t : ℝ} (hcard : (C.card : ℝ) ≤ n) (ht : 0 ≤ t) :
  22     (FiniteLaw.pi μ).probability (fun σ =>
  23       t ≤ |(∑ i ∈ C, patternValue {e i} (s i) σ) - ∑ i ∈ C, (μ (e i)).mass (s i (e i))|) ≤
  24       2 * Real.exp (-2 * t ^ 2 / n) := by
  25   classical
  26   by_cases hC : C = ∅
  27   · subst C
  28     by_cases ht0 : t = 0
  29     · subst t
  30       norm_num [FiniteLaw.probability, (FiniteLaw.pi μ).total]
  31     · have htp : 0 < t := lt_of_le_of_ne ht (Ne.symm ht0)
  32       simp only [Finset.sum_empty, sub_self, abs_zero, not_le.mpr htp]
  33       rw [finiteLaw_probability_false]
  34       positivity
  35   · have hd : ∀ i ∈ C, ∀ j ∈ C, i ≠ j →
  36         Disjoint ({e i} : Finset E) {e j} := by
  37       intro i hi j hj hij
  38       simp only [Finset.disjoint_singleton_left, Finset.mem_singleton]
  39       exact fun he => hij (hinj hi hj he)
  40     have hh := finiteLaw_disjoint_patterns_tail μ C (fun i => {e i}) s hd ht
  41     simp only [Finset.prod_singleton, Finset.card_singleton, Nat.cast_one,
  42       Finset.sum_const, nsmul_eq_mul, mul_one] at hh
  43     apply hh.trans
  44     apply mul_le_mul_of_nonneg_left _ (by norm_num)
  45     apply Real.exp_le_exp.mpr
  46     have hpos : (0 : ℝ) < C.card := by exact_mod_cast Finset.card_pos.mpr (Finset.nonempty_iff_ne_empty.mpr hC)
  47     have hdiv := div_le_div_of_nonneg_left (show 0 ≤ (2 : ℝ) * t ^ 2 by positivity) hpos hcard
  48     simpa only [neg_div, neg_mul] using neg_le_neg hdiv
