  27 theorem divisibility_uniform_colour_sample (R : ℕ) (hR : 0<R)
  28     (ε ξ : ℝ) (hε : 0<ε) (hξ : 0<ξ) :
  29     ∃ N : ℕ,∀ (V : Type u) [Fintype V] [DecidableEq V],N≤Fintype.card V →
  30     ∀ (G : SimpleGraph V),(∀ v,(1/2+ε)*Fintype.card V≤(G.degree v:ℝ)) →
  31     ∀ b : Fin R,∃ σ : Sym2 V → Fin R,
  32       (∀ a v,|((edgeColourGraph G σ a).degree v:ℝ)-(G.degree v:ℝ)/R|<ξ*Fintype.card V/4) ∧
  33       (∀ a u v,u≠v → (ε/(R:ℝ)^2)*Fintype.card V≤(colouredTwoPathCount G σ u v a b:ℝ)) := by
  34   have hRr : (0:ℝ)<R := Nat.cast_pos.mpr hR
  35   let η := min (ξ/4) (ε/(R:ℝ)^2)
  36   have hη : 0<η := lt_min (by positivity) (by positivity)
  37   obtain ⟨N,hN⟩ := divisibility_colour_sample R η hη
  38   refine ⟨N,?_⟩
  39   intro V instV instDec hn G hdeg b
  40   let ρ : Fin R → ℝ := fun _ => 1/R
  41   have hρ : ∀ a,0≤ρ a := fun _ => by dsimp [ρ];positivity
  42   have hsum : ∑ a,ρ a=1 := by simp [ρ,ne_of_gt hRr]
  43   obtain ⟨σ,hσdeg,hσpath⟩ := hN V hn G ρ hρ hsum b
  44   refine ⟨σ,?_,?_⟩
  45   · intro a v
  46     have hh := hσdeg a v
  47     have hsmall := mul_le_mul_of_nonneg_right (min_le_left (ξ/4) (ε/(R:ℝ)^2))
  48       (Nat.cast_nonneg (α:=ℝ) (Fintype.card V))
  49     rw [edgeColourGraph_degree]
  50     have heq : ρ a*(G.degree v:ℝ)=(G.degree v:ℝ)/R := by dsimp [ρ];ring
  51     rw [heq] at hh
  52     exact hh.trans_le (by dsimp only [η] at hh;linarith)
  53   · intro a u v huv
  54     have hh := (abs_lt.mp (hσpath a u v huv)).1
  55     have hc := dense_graph_common_neighbours G ε hdeg u v
  56     have hmean := div_le_div_of_nonneg_right hc (sq_nonneg (R:ℝ))
  57     have heq : ((G.neighborFinset u∩G.neighborFinset v).card:ℝ)*ρ a*ρ b=
  58         ((G.neighborFinset u∩G.neighborFinset v).card:ℝ)/(R:ℝ)^2 := by dsimp [ρ];ring
  59     rw [heq] at hh
  60     have hsmall := mul_le_mul_of_nonneg_right (min_le_right (ξ/4) (ε/(R:ℝ)^2))
  61       (Nat.cast_nonneg (α:=ℝ) (Fintype.card V))
  62     have heq2 : 2*ε*(Fintype.card V:ℝ)/(R:ℝ)^2=2*((ε/(R:ℝ)^2)*Fintype.card V) := by ring
  63     rw [heq2] at hmean
  64     dsimp only [η] at hh
  65     calc
  66       (ε/(R:ℝ)^2)*Fintype.card V=2*((ε/(R:ℝ)^2)*Fintype.card V)-(ε/(R:ℝ)^2)*Fintype.card V := by ring
  67       _ ≤ ((G.neighborFinset u∩G.neighborFinset v).card:ℝ)/(R:ℝ)^2-
  68           min (ξ/4) (ε/(R:ℝ)^2)*Fintype.card V := sub_le_sub hmean hsmall
  69       _ ≤ (colouredTwoPathCount G σ u v a b:ℝ) := by
  70         have h := add_lt_add_right hh (((G.neighborFinset u∩G.neighborFinset v).card:ℝ)/(R:ℝ)^2)
  71         convert h.le using 1 <;> ring
