  16 theorem bipartite_cut_change_sum (P Q : X→Y→Prop) (S : Finset X) (T : Finset Y) :
  17     |(bipartiteMatrixCut (fun x y=>if Q x y then 1 else 0) S T:ℝ)-
  18       (bipartiteMatrixCut (fun x y=>if P x y then 1 else 0) S T:ℝ)|≤
  19     ∑ x∈S,∑ y∈T,if ¬(P x y ↔ Q x y) then (1:ℝ) else 0 := by
  20   simp only [bipartiteMatrixCut,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  21   rw [← Finset.sum_sub_distrib]
  22   apply (Finset.abs_sum_le_sum_abs _ _).trans
  23   apply Finset.sum_le_sum
  24   intro x hx
  25   rw [← Finset.sum_sub_distrib]
  26   apply (Finset.abs_sum_le_sum_abs _ _).trans
  27   apply Finset.sum_le_sum
  28   intro y hy
  29   by_cases hp : P x y <;> by_cases hq : Q x y <;> simp [hp,hq]
