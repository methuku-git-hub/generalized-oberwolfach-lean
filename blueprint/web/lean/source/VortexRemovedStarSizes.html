  10 theorem vortex_sdiff_size_bounds
  11     (V:ℕ→Type) [∀ i,Fintype (V i)] [∀ i,DecidableEq (V i)]
  12     (I T:∀ i,Finset (V i)) (L:ℕ) (θ η K m:ℝ) (X:ℕ→ℝ)
  13     (hθ0:0≤θ) (hθ1:θ<1) (hη:0≤η) (hK:0≤K)
  14     (hsub:∀ i,i≤L→T i⊆I i)
  15     (hsmall:∀ i,i≤L→((T i).card:ℝ)≤θ*(I i).card)
  16     (hstep:∀ i,i<L→((I (i+1)).card:ℝ)≤η*(I i).card)
  17     (hmin:∀ i,i≤L→m≤(1-θ)*((I i).card:ℝ))
  18     (hX:∀ i,i≤L→X i≤K*((I i).card:ℝ)):
  19     (∀ i,i≤L→m≤((I i\T i).card:ℝ)) ∧
  20     (∀ i,i≤L→X i≤(K/(1-θ))*((I i\T i).card:ℝ)):=by
  21   have hden:0<1-θ:=by linarith
  22   have hlo:=(vortex_sdiff_scales V I T L θ η hθ0 hθ1 hη hsub hsmall hstep).1
  23   constructor
  24   · intro i hi
  25     exact (hmin i hi).trans (hlo i hi)
  26   · intro i hi
  27     have hcur:=hlo i hi
  28     have hmul:0≤K*(((I i\T i).card:ℝ)-(1-θ)*((I i).card:ℝ)):=mul_nonneg hK (sub_nonneg.mpr hcur)
  29     have hbound:K*((I i).card:ℝ)≤(K/(1-θ))*((I i\T i).card:ℝ):=by
  30       rw [div_mul_eq_mul_div]
  31       apply (le_div_iff₀ hden).2
  32       nlinarith
  33     exact (hX i hi).trans hbound
  34 end Oberwolfach
