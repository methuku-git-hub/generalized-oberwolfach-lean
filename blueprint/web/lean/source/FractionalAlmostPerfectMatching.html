  14 theorem almost_perfect_matching_of_fractional_weights
  15     {V E : Type*} [Fintype V] [Fintype E] [DecidableEq V] [DecidableEq E]
  16     (R : E → Finset V) (r : ℕ) (hr : 2≤r) (hR : ∀ e, (R e).card=r)
  17     (ξ : ℝ) (hξ : 0<ξ) (hξ1 : ξ≤1)
  18     (w : E → ℝ) (hw : ∀ e, 0≤w e)
  19     (hload : ∀ v, 1-HadwigerLean.Theorem2.matchingMu r ξ/4 ≤
  20       (∑ e, if v∈R e then w e else 0) ∧ (∑ e, if v∈R e then w e else 0)≤1)
  21     (hpair : ∀ u v, u≠v → (∑ e, if u∈R e ∧ v∈R e then w e else 0)≤
  22       HadwigerLean.Theorem2.matchingMu r ξ/4) :
  23     ∃ M : Finset E,
  24       (M : Set E).Pairwise (fun e f => Disjoint (R e) (R f)) ∧
  25       (1-ξ)*Fintype.card V ≤ ((M.biUnion R).card : ℝ) := by
  26   let μ := HadwigerLean.Theorem2.matchingMu r ξ
  27   have hμ : 0<μ := by
  28     dsimp [μ,HadwigerLean.Theorem2.matchingMu]
  29     have hrR : (0 : ℝ)<r := by exact_mod_cast (show 0<r by omega)
  30     positivity
  31   obtain ⟨T,hT⟩ := exists_nat_gt (4*((Fintype.card E : ℝ)+1)/μ)
  32   have hTbig : 4*((Fintype.card E : ℝ)+1)<μ*T := by
  33     have hh := (div_lt_iff₀ hμ).mp hT
  34     nlinarith
  35   have hTR : (0 : ℝ)<T := lt_of_lt_of_le
  36     (div_pos (by positivity) hμ) hT.le
  37   have hTN : 0<T := Nat.cast_pos.mp hTR
  38   let m (e : E) := ⌊(T : ℝ)*w e⌋₊
  39   let H := copiedResourceHypergraph R m
  40   let B := ⌈μ*T/2⌉₊
  41   have hBmin : 1≤B := by
  42     have hlow := Nat.le_ceil (μ*T/2)
  43     have he0 : (0 : ℝ)≤Fintype.card E := Nat.cast_nonneg _
  44     have hpos : (0 : ℝ)<B := by dsimp [B];linarith
  45     exact Nat.cast_pos.mp hpos
  46   have hBmax : (B : ℝ)≤μ*T := by
  47     have hb := Nat.ceil_lt_add_one (show 0≤μ*T/2 by positivity)
  48     have he0 : (0 : ℝ)≤Fintype.card E := Nat.cast_nonneg _
  49     dsimp [B];linarith
  50   have hcap (v : V) (_ : v∈H.vertices) : H.degree v≤T := by
  51     rw [copiedResource_degree]
  52     have hh := (sum_floor_scaled_weight_bounds w hw (fun e => v∈R e) T).2
  53     have hb := mul_le_mul_of_nonneg_left (hload v).2 hTR.le
  54     have hgoal : ((∑ e, if v∈R e then m e else 0 : ℕ) : ℝ)≤T := by
  55       exact hh.trans (by simpa only [mul_one] using hb)
  56     exact_mod_cast hgoal
  57   have hmin (v : V) (_ : v∈H.vertices) : (1-μ)*(T : ℝ)≤(H.degree v : ℝ) := by
  58     rw [copiedResource_degree]
  59     have hh := (sum_floor_scaled_weight_bounds w hw (fun e => v∈R e) T).1
  60     have hb := mul_le_mul_of_nonneg_left (hload v).1 hTR.le
  61     change (1-μ)*T≤((∑ e, if v∈R e then ⌊(T : ℝ)*w e⌋₊ else 0 : ℕ) : ℝ)
  62     change (T : ℝ)*(1-μ/4)≤_ at hb
  63     have he0 : (0 : ℝ)≤Fintype.card E := Nat.cast_nonneg _
  64     have hh' : (T : ℝ)*(∑ e, if v∈R e then w e else 0)-Fintype.card E ≤
  65         ((∑ e, if v∈R e then ⌊(T : ℝ)*w e⌋₊ else 0 : ℕ) : ℝ) := hh
  66     have hb' : (T : ℝ)*(1-μ/4)-Fintype.card E ≤
  67         ((∑ e, if v∈R e then ⌊(T : ℝ)*w e⌋₊ else 0 : ℕ) : ℝ) :=
  68       (sub_le_sub_right hb _).trans hh'
  69     exact (show (1-μ)*(T : ℝ)≤(T : ℝ)*(1-μ/4)-Fintype.card E by nlinarith).trans hb'
  70   have hcodeg (u v : V) (huv : u≠v) : H.codegree u v≤B := by
  71     rw [copiedResource_codegree]
  72     have hh := (sum_floor_scaled_weight_bounds w hw (fun e => u∈R e ∧ v∈R e) T).2
  73     have hb := mul_le_mul_of_nonneg_left (hpair u v huv) hTR.le
  74     have hceil := Nat.le_ceil (μ*T/2)
  75     have hgoal : ((∑ e, if u∈R e ∧ v∈R e then m e else 0 : ℕ) : ℝ)≤B := by
  76       change (T : ℝ)*(∑ e, if u∈R e ∧ v∈R e then w e else 0)≤(T : ℝ)*(μ/4) at hb
  77       change μ*T/2≤(B : ℝ) at hceil
  78       exact hh.trans (hb.trans ((show (T : ℝ)*(μ/4)≤μ*T/2 by
  79         nlinarith [mul_pos hμ hTR]).trans hceil))
  80     exact_mod_cast hgoal
  81   have hmax : H.maxCodegree≤B := by
  82     apply Finset.sup_le
  83     intro u _
  84     apply Finset.sup_le
  85     intro v hv
  86     exact hcodeg u v (Ne.symm (Finset.mem_erase.mp hv).1)
  87   obtain ⟨M,hM,hcovered⟩ := HadwigerLean.Theorem2.exists_almostPerfectMatching_of_degree_bounds H
  88     hr hξ hξ1 hTN hBmin hBmax (fun e _ => hR e.1) hcap hmin hmax
  89   obtain ⟨hproj,heq⟩ := copiedResource_matching_projection R m M hM
  90   refine ⟨M.image Sigma.fst,hproj,?_⟩
  91   rw [heq]
  92   simpa only [H,copiedResourceHypergraph,Finset.card_univ] using hcovered
