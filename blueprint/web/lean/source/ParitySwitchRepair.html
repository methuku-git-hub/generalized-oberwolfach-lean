  25 theorem paired_colour_parity_repair {V : Type*} [Fintype V] [DecidableEq V]
  26     {q : ℕ} (hq : Even q) (G : SimpleGraph V) (σ : Sym2 V → Fin (q+1))
  27     (T : Fin q → Finset V) (m : Fin q → ℕ) (e : ∀ j,(Fin (m j)×Bool)≃T j)
  28     (hT : ∀ j v,v∈T j ↔ Odd ((G\edgeColourGraph G σ j.castSucc).degree v))
  29     (z : ((j : Fin q)×Fin (m j)) → V)
  30     (hdisj : ∀ i j,i≠j →
  31       Disjoint (twoPathEdges (e i.1 (i.2,false)).val (e i.1 (i.2,true)).val (z i))
  32         (twoPathEdges (e j.1 (j.2,false)).val (e j.1 (j.2,true)).val (z j)))
  33     (hxz : ∀ i,G.Adj (e i.1 (i.2,false)).val (z i))
  34     (hyz : ∀ i,G.Adj (z i) (e i.1 (i.2,true)).val)
  35     (hc0 : ∀ i,σ s((e i.1 (i.2,false)).val,z i)=i.1.castSucc)
  36     (hc1 : ∀ i,σ s(z i,(e i.1 (i.2,true)).val)=Fin.last q) :
  37     ∃ τ : Sym2 V → Fin (q+1),
  38       (∀ a,(edgeColourGraph G τ a).edgeFinset.card=(edgeColourGraph G σ a).edgeFinset.card) ∧
  39       (∀ a v,Even ((G\edgeColourGraph G τ a).degree v)) ∧
  40       (∀ a : Fin q,∀ v,|((edgeColourGraph G τ a.castSucc).degree v:ℝ)-
  41         ((edgeColourGraph G σ a.castSucc).degree v:ℝ)|≤1) ∧
  42       (∀ v,|((edgeColourGraph G τ (Fin.last q)).degree v:ℝ)-
  43         ((edgeColourGraph G σ (Fin.last q)).degree v:ℝ)|≤q) := by
  44   let x (i : (j : Fin q)×Fin (m j)) := (e i.1 (i.2,false)).val
  45   let y (i : (j : Fin q)×Fin (m j)) := (e i.1 (i.2,true)).val
  46   have hxy (i : (j : Fin q)×Fin (m j)) : x i≠y i := pairing_endpoints_distinct (e i.1) i.2
  47   let ι := twoPathEdgeEmbedding x y z hxy hdisj
  48   let τ := σ ∘ pairSwitchPermutation ι
  49   have hid (R : Type) [CommRing R] (a : Fin q) (v : V) :
  50       ((edgeColourGraph G τ a.castSucc).degree v:R)-((edgeColourGraph G σ a.castSucc).degree v:R)=
  51       ∑ i : Fin (m a),((if (e a (i,true)).val=v then (1:R) else 0)-
  52         (if (e a (i,false)).val=v then 1 else 0)) :=
  53     colour_path_switch_degree_identity m G σ x y z hxy hdisj hxz hyz hc0 hc1 a v
  54   have hbound (a : Fin q) (v : V) : |((edgeColourGraph G τ a.castSucc).degree v:ℝ)-
  55       ((edgeColourGraph G σ a.castSucc).degree v:ℝ)|≤1 := by
  56     rw [hid ℝ]
  57     exact pairing_endpoint_discrepancy_bound (e a) v
  58   have hmod (a : Fin q) (v : V) :
  59       ((edgeColourGraph G τ a.castSucc).degree v:ZMod 2)=(G.degree v:ZMod 2) := by
  60     have hi := hid (ZMod 2) a v
  61     rw [pairing_endpoint_discrepancy_mod_two] at hi
  62     simp only [hT,odd_indicator_zmod] at hi
  63     have hc := congrArg (fun n : ℕ => (n:ZMod 2))
  64       (degree_complement_add G (edgeColourGraph G σ a.castSucc) (edgeColourGraph_le G σ _) v)
  65     simp only [Nat.cast_add] at hc
  66     linear_combination hi+hc
  67   have hlast (R : Type) [CommRing R] (v : V) :
  68       (∑ a : Fin q,(((edgeColourGraph G τ a.castSucc).degree v:R)-
  69         ((edgeColourGraph G σ a.castSucc).degree v:R)))+
  70       (((edgeColourGraph G τ (Fin.last q)).degree v:R)-
  71         ((edgeColourGraph G σ (Fin.last q)).degree v:R))=0 := by
  72     have ht := congrArg (fun n : ℕ => (n:R)) (edgeColourGraph_degree_total G τ v)
  73     have hs := congrArg (fun n : ℕ => (n:R)) (edgeColourGraph_degree_total G σ v)
  74     simp only [Nat.cast_sum,Nat.cast_add,Fin.sum_univ_castSucc] at ht hs
  75     rw [Finset.sum_sub_distrib]
  76     linear_combination ht-hs
  77   have hmodall (a : Fin (q+1)) (v : V) :
  78       ((edgeColourGraph G τ a).degree v:ZMod 2)=(G.degree v:ZMod 2) := by
  79     refine Fin.lastCases ?_ (fun a => hmod a v) a
  80     have ht := congrArg (fun n : ℕ => (n:ZMod 2)) (edgeColourGraph_degree_total G τ v)
  81     simp only [Nat.cast_sum,Nat.cast_add,Fin.sum_univ_castSucc,hmod,Finset.sum_const,Finset.card_univ,
  82       Fintype.card_fin,nsmul_eq_mul,hq.natCast_zmod_two,zero_mul,zero_add] at ht
  83     exact ht
  84   refine ⟨τ,?_,?_,hbound,?_⟩
  85   · intro a
  86     exact twoPath_switch_preserves_card G σ x y z hxy hdisj hxz hyz a
  87   · intro a v
  88     apply ZMod.natCast_eq_zero_iff_even.mp
  89     have hc := congrArg (fun n : ℕ => (n:ZMod 2))
  90       (degree_complement_add G (edgeColourGraph G τ a) (edgeColourGraph_le G τ a) v)
  91     simp only [Nat.cast_add,hmodall] at hc
  92     exact add_right_cancel (hc.trans (zero_add _).symm)
  93   · intro v
  94     have hh := hlast ℝ v
  95     have he : ((edgeColourGraph G τ (Fin.last q)).degree v:ℝ)-
  96         ((edgeColourGraph G σ (Fin.last q)).degree v:ℝ)=
  97         -(∑ a : Fin q,(((edgeColourGraph G τ a.castSucc).degree v:ℝ)-
  98           ((edgeColourGraph G σ a.castSucc).degree v:ℝ))) := by linarith only [hh]
  99     rw [he,abs_neg]
 100     apply (Finset.abs_sum_le_sum_abs _ _).trans
 101     calc
 102       _ ≤ ∑ _ : Fin q,(1:ℝ) := Finset.sum_le_sum (fun a _ => hbound a v)
 103       _ = _ := by simp
