  15 theorem permutationSeparatedMatrix_row_loss (π : Equiv.Perm V) (b : ℕ)
  16     (M : V→V→ℕ) (hM : ∀ x y,M x y≤1) (x : V) :
  17     |((∑ y,permutationSeparatedMatrix π b M x y:ℕ):ℝ)-(∑ y,M x y:ℕ)|≤2*(b:ℝ)+1 := by
  18   have hh := permutationSeparatedMatrix_cut_loss π b M hM {x} Finset.univ
  19   have hn : 1≤Fintype.card V := Fintype.card_pos_iff.mpr ⟨x⟩
  20   have hn' : (1:ℝ)≤Fintype.card V := by exact_mod_cast hn
  21   simp only [matrixCut,Finset.sum_singleton,Finset.card_singleton,Finset.card_univ,Nat.cast_one,min_eq_left hn'] at hh
  22   have hsub : (∑ y,permutationSeparatedMatrix π b M x y)≤∑ y,M x y := by
  23     apply Finset.sum_le_sum
  24     intro y _
  25     simp only [permutationSeparatedMatrix]
  26     split_ifs <;> omega
  27   have hsub' : ((∑ y,permutationSeparatedMatrix π b M x y:ℕ):ℝ)≤(∑ y,M x y:ℕ) := by exact_mod_cast hsub
  28   rw [abs_sub_comm,abs_of_nonneg (sub_nonneg.mpr hsub')]
  29   linarith only [hh]
