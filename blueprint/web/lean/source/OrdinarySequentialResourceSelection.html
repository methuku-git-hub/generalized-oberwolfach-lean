   4 /-! Successive physical-edge selection from a bounded avoidance theorem.

  15 theorem ordinary_sequential_resource_selection (l f a b:ℕ) (γ:ℝ)
  16     (W:Finset V) (B:Finset U) (L:Finset (Sym2 V⊕(V×U)))
  17     (hW:W.card≤f) (hB:(B.card:ℝ)≤γ)
  18     (hrows:ordinaryResourceRows L⊆W) (hhubs:ordinaryResourceHubs L⊆B)
  19     (P:Fin l→Finset (Sym2 V⊕(V×U))→Prop)
  20     (hchoose:∀ (W':Finset V) (B':Finset U) (L':Finset (Sym2 V⊕(V×U))),
  21       W⊆W'→B⊆B'→W'.card≤f+l*a→(B'.card:ℝ)≤γ+l*b→
  22       ordinaryResourceRows L'⊆W'→ ordinaryResourceHubs L'⊆B'→
  23       ∀ i:Fin l,∃ K,P i K∧(ordinaryResourceRows K).card≤a∧
  24         (ordinaryResourceHubs K).card≤b∧Disjoint K L') :
  25     ∃ K:Fin l→Finset (Sym2 V⊕(V×U)),
  26       (∀ i,P i (K i)∧(ordinaryResourceRows (K i)).card≤a∧
  27         (ordinaryResourceHubs (K i)).card≤b∧Disjoint (K i) L)∧
  28       (∀ i j,i≠j→Disjoint (K i) (K j))∧
  29       (ordinaryResourceRows (Finset.univ.biUnion K)).card≤l*a∧
  30       (ordinaryResourceHubs (Finset.univ.biUnion K)).card≤l*b := by
  31   let allowed:=fun i K=>P i K∧(ordinaryResourceRows K).card≤a∧
  32     (ordinaryResourceHubs K).card≤b∧Disjoint K L
  33   obtain ⟨K,hK,hdis⟩:=sequential_select_prefix l allowed Disjoint ∅ (by
  34     intro i K hK
  35     let s:=Finset.univ.filter (fun j:Fin l=>j.val < i.val)
  36     let used:=s.biUnion K
  37     let W':=W∪ordinaryResourceRows used
  38     let B':=B∪ordinaryResourceHubs used
  39     have hsc:s.card≤l:=by simpa using Finset.card_le_univ s
  40     have hrow:(ordinaryResourceRows used).card≤l*a:=by
  41       rw [ordinaryResourceRows_biUnion]
  42       apply Finset.card_biUnion_le.trans
  43       calc
  44         _ ≤ ∑ _j∈s,a := Finset.sum_le_sum (fun j hj=>(hK.2.1 j (Finset.mem_filter.mp hj).2).2.1)
  45         _ = s.card*a := by simp
  46         _ ≤ l*a := Nat.mul_le_mul_right _ hsc
  47     have hhub:(ordinaryResourceHubs used).card≤l*b:=by
  48       rw [ordinaryResourceHubs_biUnion]
  49       apply Finset.card_biUnion_le.trans
  50       calc
  51         _ ≤ ∑ _j∈s,b := Finset.sum_le_sum (fun j hj=>(hK.2.1 j (Finset.mem_filter.mp hj).2).2.2.1)
  52         _ = s.card*b := by simp
  53         _ ≤ l*b := Nat.mul_le_mul_right _ hsc
  54     have hWc:W'.card≤f+l*a:=
  55       (Finset.card_union_le _ _).trans (Nat.add_le_add hW hrow)
  56     have hBc:(B'.card:ℝ)≤γ+l*b:=by
  57       have hc:(B'.card:ℝ)≤B.card+(ordinaryResourceHubs used).card:=by exact_mod_cast Finset.card_union_le B (ordinaryResourceHubs used)
  58       have hb':((ordinaryResourceHubs used).card:ℝ)≤l*b:=by exact_mod_cast hhub
  59       linarith only [hc,hb',hB]
  60     obtain ⟨C,hC,hCr,hCh,hCd⟩:=hchoose W' B' (L∪used)
  61       Finset.subset_union_left Finset.subset_union_left hWc hBc
  62       (by rw [ordinaryResourceRows_union];exact Finset.union_subset_union hrows (fun _ h=>h))
  63       (by rw [ordinaryResourceHubs_union];exact Finset.union_subset_union hhubs (fun _ h=>h)) i
  64     refine ⟨C,⟨hC,hCr,hCh,(Finset.disjoint_union_right.mp hCd).1⟩,?_⟩
  65     intro j hj
  66     exact ((Finset.disjoint_union_right.mp hCd).2.mono_right
  67       (Finset.subset_biUnion_of_mem K (Finset.mem_filter.mpr ⟨Finset.mem_univ _,hj⟩))).symm)
  68   refine ⟨K,hK,?_,?_,?_⟩
  69   · intro i j hij
  70     rcases lt_or_gt_of_ne hij with h|h
  71     · exact hdis i j h
  72     · exact (hdis j i h).symm
  73   · rw [ordinaryResourceRows_biUnion]
  74     apply Finset.card_biUnion_le.trans
  75     calc
  76       _ ≤ ∑ _i:Fin l,a := Finset.sum_le_sum (fun i _=>(hK i).2.1)
  77       _ = l*a := by simp
  78   · rw [ordinaryResourceHubs_biUnion]
  79     apply Finset.card_biUnion_le.trans
  80     calc
  81       _ ≤ ∑ _i:Fin l,b := Finset.sum_le_sum (fun i _=>(hK i).2.2.1)
  82       _ = l*b := by simp
