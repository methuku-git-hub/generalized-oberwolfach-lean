  14 def equalCellSizes (n c : ℕ) (a : Fin (c + 1)) : ℕ :=
  15   if a.val < c then n / c else n % c
  16 
  17 theorem equalCellSizes_castSucc (n c : ℕ) (i : Fin c) :
  18     equalCellSizes n c i.castSucc = n / c := by simp [equalCellSizes, i.isLt]
  19 
  20 theorem equalCellSizes_last (n c : ℕ) :
  21     equalCellSizes n c (Fin.last c) = n % c := by simp [equalCellSizes]
  22 
  23 theorem equalCellSizes_sum (n c : ℕ) : ∑ a, equalCellSizes n c a = n := by
  24   rw [Fin.sum_univ_castSucc]
  25   simp only [equalCellSizes_castSucc, equalCellSizes_last, Finset.sum_const,
  26     Finset.card_univ, Fintype.card_fin, smul_eq_mul]
  27   have h := Nat.mod_add_div n c
  28   omega
