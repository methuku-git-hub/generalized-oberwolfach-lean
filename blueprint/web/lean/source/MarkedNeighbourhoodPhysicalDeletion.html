  21 theorem marked_spoke_physical_deletion_column_change
  22     (H:Bool→V→U→Prop) (D:SimpleGraph (V⊕U)) (A:Finset V) (a:Bool) (u:U):
  23     (∑ x∈A,if ¬(H a x u↔markedDeletePhysicalSpokes H D a x u) then (1:ℝ) else 0)
  24       ≤D.degree (Sum.inr u):=by
  25   calc
  26     _ ≤∑ x∈A,if D.Adj (Sum.inl x) (Sum.inr u) then (1:ℝ) else 0:=by
  27       apply Finset.sum_le_sum
  28       intro x _
  29       by_cases h:H a x u <;> by_cases hd:D.Adj (Sum.inl x) (Sum.inr u) <;>
  30         simp [markedDeletePhysicalSpokes,h,hd]
  31     _ ≤∑ x:V,if D.Adj (Sum.inl x) (Sum.inr u) then (1:ℝ) else 0:=
  32       Finset.sum_le_sum_of_subset_of_nonneg (Finset.subset_univ A) (by intro x _ _;positivity)
  33     _ = ((Finset.univ.filter (fun x=>D.Adj (Sum.inl x) (Sum.inr u))).card:ℝ):=by
  34       simp only [Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
