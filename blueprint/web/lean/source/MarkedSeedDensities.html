  10 theorem marked_seed_density_bounds (m : ℕ) (hm : 1≤m)
  11     (ρ s ρstar sstar : ℝ) (hρstar : 0<ρstar) (hsstar : 0<sstar)
  12     (hρlo : ρstar≤ρ) (hρhi : ρ≤1) (hslo : sstar≤s) (hshi : s≤1/100) :
  13     let p := ρ*s/2
  14     let k := p/m
  15     let t := s/m
  16     let pstar := ρstar*sstar/(2*m)
  17     0<pstar ∧ pstar≤p ∧ pstar≤k ∧ pstar≤s ∧
  18       p≤1 ∧ k≤1 ∧ s≤1 ∧ 0<t ∧ 2*p+2*k≤1 := by
  19   intro p k t pstar
  20   have hm' : (1:ℝ)≤m := by exact_mod_cast hm
  21   have hm0 : (0:ℝ)<m := by linarith
  22   have hρ : 0<ρ := hρstar.trans_le hρlo
  23   have hs : 0<s := hsstar.trans_le hslo
  24   have hp0 : 0<p := by dsimp [p];positivity
  25   have hprod : ρstar*sstar≤ρ*s := mul_le_mul hρlo hslo hsstar.le hρ.le
  26   have hklo : pstar≤k := by
  27     dsimp [pstar,k,p]
  28     rw [div_div]
  29     exact div_le_div_of_nonneg_right hprod (by positivity)
  30   have hkp : k≤p := by dsimp [k];apply (div_le_iff₀ hm0).mpr;nlinarith
  31   have hps : p≤s := by dsimp [p];nlinarith [mul_le_mul_of_nonneg_right hρhi hs.le]
  32   have hpp : 0<pstar := by dsimp [pstar];positivity
  33   refine ⟨hpp,hklo.trans hkp,hklo,(hklo.trans hkp).trans hps,?_,?_,?_,?_,?_⟩
  34   · linarith
  35   · linarith
  36   · linarith
  37   · dsimp [t];positivity
  38   · linarith
