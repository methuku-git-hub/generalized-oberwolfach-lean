  16 theorem markedReferenceCutFamilies_fixed_cuts
  17     (P : SimpleGraph V) (π : Equiv.Perm V) (d : ℕ) (g : Γ) (j : Color)
  18     (α kstar tstar p k : ℝ) (hα : 0<α) (hkstar : 0<kstar) (htstar : 0<tstar)
  19     (hpmin : kstar≤p) (hkmin : kstar≤k) (hn : 0<Fintype.card V)
  20     (σ : RawMarkedOutcome V U Γ Color)
  21     (hcuts : MarkedReferenceCutFamilies P π d g j α p k tstar σ) :
  22     let δ := min (min α kstar) tstar/256
  23     let h := min (α*kstar) tstar/128
  24     (∀ b : Bool,∀ (X : Finset V) (Y : Finset U),
  25       1-2*δ<(X.card:ℝ)/Fintype.card V+(Y.card:ℝ)/Fintype.card U →
  26       16*h*min ((Fintype.card U:ℝ)*X.card) ((Fintype.card V:ℝ)*Y.card)≤
  27         (bipartiteMatrixCut (fun v u=>if rawMarkedPairSpokes π d σ g j b v u then 1 else 0) X Y:ℝ)) ∧
  28     (∀ b : Bool,∀ S T : Finset V,
  29       1-2*δ<(S.card:ℝ)/Fintype.card V+(T.card:ℝ)/Fintype.card V →
  30       16*h*min ((Fintype.card V:ℝ)*S.card) ((Fintype.card V:ℝ)*T.card)≤
  31         (matrixCut (markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) b) S T:ℝ)) := by
  32   intro δ h
  33   let n := Fintype.card V
  34   have hn0 : (0:ℝ)<n := by exact_mod_cast hn
  35   have hδα : δ≤α/32 := by
  36     have hh := (min_le_left (min α kstar) tstar).trans (min_le_left α kstar)
  37     dsimp [δ];linarith only [hh,hα]
  38   have hδk : δ≤kstar/32 := by
  39     have hh := (min_le_left (min α kstar) tstar).trans (min_le_right α kstar)
  40     dsimp [δ];linarith only [hh,hkstar]
  41   have hδt : 2*δ≤tstar/8 := by
  42     have hh := min_le_right (min α kstar) tstar
  43     dsimp [δ];linarith only [hh,htstar]
  44   constructor
  45   · intro b X Y hlarge
  46     change 1-2*δ<(X.card:ℝ)/Fintype.card V+(Y.card:ℝ)/Fintype.card U at hlarge
  47     have hh := hcuts.1 b X Y (by linarith only [hlarge,hδt])
  48     have hc : 16*h≤tstar/8 := by
  49       have hh := min_le_right (α*kstar) tstar
  50       dsimp [h];linarith only [hh]
  51     exact (mul_le_mul_of_nonneg_right hc (by positivity)).trans hh
  52   · intro b S T hlarge
  53     have hcoef : kstar≤(if b then k else p) := by cases b <;> assumption
  54     have hlarge' : (1-2*δ)*Fintype.card V<(S.card:ℝ)+T.card := by
  55       have hh := mul_lt_mul_of_pos_right hlarge hn0
  56       have heq : ((S.card:ℝ)/n+(T.card:ℝ)/n)*n=(S.card:ℝ)+T.card := by field_simp
  57       change (1-2*δ)*(n:ℝ)<((S.card:ℝ)/n+(T.card:ℝ)/n)*n at hh
  58       rw [heq] at hh
  59       exact hh
  60     have hh := hcuts.2 b δ hδα (by linarith only [hδk,hcoef]) S T hlarge'
  61     have hc : 16*h≤α*(if b then k else p)/4 := by
  62       have hh := min_le_left (α*kstar) tstar
  63       have hm := mul_le_mul_of_nonneg_left hcoef hα.le
  64       dsimp [h];nlinarith only [hh,hm,mul_pos hα hkstar]
  65     rw [← mul_min_of_nonneg _ _ hn0.le]
  66     change 16*h*((n:ℝ)*min (S.card:ℝ) T.card)≤_
  67     calc
  68       _ = (16*h*(n:ℝ))*min (S.card:ℝ) T.card := by ring
  69       _ ≤ (α*(if b then k else p)/4*(n:ℝ))*min (S.card:ℝ) T.card :=
  70         mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_right hc hn0.le)
  71           (le_min (Nat.cast_nonneg S.card) (Nat.cast_nonneg T.card))
  72       _ ≤ _ := hh
