  12 theorem uniform_root_load_endpoint_balance
  13     {V U:Type*} [Fintype V] [DecidableEq V] [Fintype U]
  14     (π:Equiv.Perm V) (Y B T:U→Finset V)
  15     (hrow:∀w,Y w∪B w=(Y w).image π∪T w)
  16     (hYB:∀w,Disjoint (Y w) (B w))
  17     (hπT:∀w,Disjoint ((Y w).image π) (T w))
  18     (huniform:∀v z,(Finset.univ.filter (fun w=>v∈Y w)).card=
  19       (Finset.univ.filter (fun w=>z∈Y w)).card):
  20     ∀v,(Finset.univ.filter (fun w=>v∈B w)).card=
  21       (Finset.univ.filter (fun w=>v∈T w)).card:=by
  22   classical
  23   intro v
  24   have hpoint (w:U):
  25       (if v∈Y w then 1 else 0)+(if v∈B w then 1 else 0)=
  26       (if π.symm v∈Y w then 1 else 0)+(if v∈T w then 1 else 0):=by
  27     have heq:(v∈Y w∨v∈B w)↔(v∈(Y w).image π∨v∈T w):=by
  28       have hh:=congrArg (fun s:Finset V=>v∈s) (hrow w)
  29       simp only [Finset.mem_union] at hh
  30       exact Iff.of_eq hh
  31     have hi:v∈(Y w).image π ↔ π.symm v∈Y w:=by
  32       constructor
  33       · intro h
  34         obtain ⟨a,ha,hav⟩:=Finset.mem_image.mp h
  35         rw [←hav,Equiv.symm_apply_apply]
  36         exact ha
  37       · intro h
  38         exact Finset.mem_image.mpr ⟨π.symm v,h,π.apply_symm_apply v⟩
  39     have hn1:¬(v∈Y w ∧ v∈B w):=fun h=>Finset.disjoint_left.mp (hYB w) h.1 h.2
  40     have hn2:¬(π.symm v∈Y w ∧ v∈T w):=fun h=>Finset.disjoint_left.mp (hπT w) (hi.mpr h.1) h.2
  41     rw [hi] at heq
  42     by_cases h₁:v∈Y w <;> by_cases h₂:v∈B w <;>
  43       by_cases h₃:π.symm v∈Y w <;> by_cases h₄:v∈T w <;>
  44       simp_all
  45   have hs:=Finset.sum_congr (s₁:=Finset.univ) (s₂:=Finset.univ) rfl (fun w _=>hpoint w)
  46   simp only [Finset.sum_add_distrib,Finset.sum_boole,Nat.cast_id] at hs
  47   have hu:=huniform v (π.symm v)
  48   omega
  49 end Oberwolfach
