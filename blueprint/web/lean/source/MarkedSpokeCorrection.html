  36     (herr : ∀ j v u,|markedSpokeResidual ψ j v u|≤η) :
  37     ∃ Φ : MarkedWheel tail head ktail khead m π b H₀ Hₖ → ℝ,
  38       (∀ e,markedWheelRimLoad Φ e=0) ∧
  39       (∀ j v u,markedWheelSpokeLoad Φ j v u=markedSpokeResidual ψ j v u) ∧
  40       (∀ W,|Φ W|≤markedSpokeCorrectionConstant m β L*η/(Fintype.card V:ℝ)^m) := by

  50       (∀ e,markedWheelRimLoad Φ e=0) ∧
  51       (∀ k v u,markedWheelSpokeLoad Φ k v u=if j=k then r j (v,u) else 0) ∧
  52       (∀ W,|Φ W|≤4*(C*A*η)/(β*(Fintype.card V:ℝ)^m)) := by

  54       (if j then Hₖ else H₀) (p j) (q j) L δ D (hdata j) hn hu hsmall (r j) (hres j)
