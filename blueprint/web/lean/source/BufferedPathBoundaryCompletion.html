  19     ∃i:I, x=inner i 0 ∨ x=inner i (Fin.last (n i)) := by

  24     intro hp

  27       Finset.mem_image.mpr ⟨p,Finset.mem_filter.mpr ⟨Finset.mem_univ _,hp⟩,rfl⟩⟩
