  10 theorem allowedThreshold_nonneg {S : Set ℕ} (hS : S.Nonempty) :
  11     0 ≤ allowedThreshold S := by
  12   obtain ⟨r, hr⟩ := hS
  13   apply (cycleThreshold_nonneg r).trans
  14   apply le_csSup (show BddAbove (cycleThreshold '' S) from ?_) ⟨r, hr, rfl⟩
  15   refine ⟨1, ?_⟩
  16   rintro y ⟨s, hs, rfl⟩
  17   exact cycleThreshold_le_one s

  19 theorem allowedThreshold_le_one {S : Set ℕ} (hS : S.Nonempty) :
  20     allowedThreshold S ≤ 1 := by
  21   apply csSup_le (hS.image cycleThreshold)
  22   rintro y ⟨r, hr, rfl⟩
  23   exact cycleThreshold_le_one r

  25 /-- The fixed-set theorem already implies the uniform statement. The hypothesis
  26 is an unproved target, not an axiom or an asserted decomposition result. -/
  27 theorem uniform_of_fixed_set_grid (hfixed : FixedSetTheorem) : MainTheorem := by
  28   classical
  29   intro α hα
  30   have hhalf : 0 < α / 2 := by linarith
  31   let M : ℕ := ⌈1 / (α / 2)⌉₊
  32   let T (q : Fin (M + 1)) : Set ℕ :=
  33     {r | 3 ≤ r ∧ cycleThreshold r ≤ (q.val : ℝ) * (α / 2)}
  34   have hf : ∀ q : Fin (M + 1), ∃ N : ℕ,
  35       (T q).Nonempty → ∀ n ≥ N, DecompositionAt (T q) (α / 2) n := by
  36     intro q
  37     by_cases hq : (T q).Nonempty
  38     · obtain ⟨N, hN⟩ := hfixed (T q) hq (fun r hr => hr.1) (α / 2) hhalf
  39       exact ⟨N, fun _ => hN⟩
  40     · exact ⟨0, fun h => (hq h).elim⟩
  41   choose bound hbound using hf
  42   refine ⟨Finset.univ.sup bound, ?_⟩
  43   intro S hSne hS n hn
  44   let θ := allowedThreshold S
  45   have hθ0 : 0 ≤ θ := allowedThreshold_nonneg hSne
  46   have hθ1 : θ ≤ 1 := allowedThreshold_le_one hSne
  47   let q : Fin (M + 1) := ⟨⌈θ / (α / 2)⌉₊, by
  48     have : ⌈θ / (α / 2)⌉₊ ≤ M :=
  49       Nat.ceil_mono (div_le_div_of_nonneg_right hθ1 hhalf.le)
  50     omega⟩
  51   have hround_lower : θ ≤ (q.val : ℝ) * (α / 2) := by
  52     exact (div_le_iff₀ hhalf).mp (Nat.le_ceil (θ / (α / 2)))
  53   have hround_upper : (q.val : ℝ) * (α / 2) ≤ θ + α / 2 := by
  54     have hc := Nat.ceil_lt_add_one (div_nonneg hθ0 hhalf.le)
  55     have := mul_lt_mul_of_pos_right hc hhalf
  56     dsimp only [q]
  57     calc
  58       (⌈θ / (α / 2)⌉₊ : ℝ) * (α / 2) ≤
  59           (θ / (α / 2) + 1) * (α / 2) := this.le
  60       _ = θ + α / 2 := by field_simp
  61   have hST : S ⊆ T q := by
  62     intro r hr
  63     refine ⟨hS r hr, ?_⟩
  64     apply le_trans ?_ hround_lower
  65     apply le_csSup (show BddAbove (cycleThreshold '' S) from ?_) ⟨r, hr, rfl⟩
  66     refine ⟨1, ?_⟩
  67     rintro y ⟨s, hs, rfl⟩
  68     exact cycleThreshold_le_one s
  69   have hTne := hSne.mono hST
  70   have hTupper : allowedThreshold (T q) ≤ (q.val : ℝ) * (α / 2) := by
  71     apply csSup_le (hTne.image cycleThreshold)
  72     rintro y ⟨r, hr, rfl⟩
  73     exact hr.2
  74   apply decompositionAt_transfer hST
  75     (show allowedThreshold (T q) + α / 2 ≤ allowedThreshold S + α by
  76       dsimp only [θ] at hround_upper
  77       linarith)
  78   exact hbound q hTne n ((Finset.le_sup (Finset.mem_univ q)).trans hn)
