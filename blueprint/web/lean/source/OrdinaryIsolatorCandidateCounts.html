  14 theorem ordinary_isolator_candidate_many (α β : ℝ) (hα : 0<α) (hα1 : α≤1) (hβ : 0<β) :
  15     ∃ K : ℕ,∀ k : ℕ,K≤k → ∃ c : ℝ,0<c ∧
  16     ∀ L : ℝ,0≤L → ∃ N : ℕ,∀ (V U : Type) [Fintype V] [DecidableEq V],N≤Fintype.card V →
  17     ∀ (P F : SimpleGraph V) (H : V → U → Prop) (A : Finset V) (x y : V) (u : U),x≠y →
  18     ¬F.Adj x y → β*Fintype.card V≤(A.card:ℝ) → (∀ z∈A,H z u) →
  19     ∀ θ : ℝ,0≤θ → θ≤1 →
  20     (∀ v∈A∪{x,y},(θ+α/2)*(A.card:ℝ)≤((P.neighborFinset v∩A).card:ℝ)) →
  21     (∀ v,(F.degree v:ℝ)≤L) →
  22     c*(Fintype.card V:ℝ)^k≤Fintype.card (OrdinaryIsolatorCandidate P F H (2+k) θ (α/4) x y u) := by
  23   obtain ⟨K,hK⟩ := external_rooted_dense_sets 2 α β hα hα1 hβ
  24   refine ⟨K,?_⟩
  25   intro k hk
  26   obtain ⟨c,hc,hcN⟩ := hK k hk
  27   refine ⟨c,hc,?_⟩
  28   intro L hL
  29   obtain ⟨N,hN⟩ := hcN L hL
  30   refine ⟨N,?_⟩
  31   intro V U _ _ hn P F H A x y u hxy hFxy hA hH θ hθ hθ1 hP hF
  32   have hrootcard : ({x,y}:Finset V).card=2 := by simp [hxy]
  33   have hind : ∀ z∈({x,y}:Finset V),∀ t∈({x,y}:Finset V),¬F.Adj z t := by
  34     intro z hz t ht
  35     simp only [Finset.mem_insert,Finset.mem_singleton] at hz ht
  36     rcases hz with rfl|rfl <;> rcases ht with rfl|rfl
  37     · exact F.loopless.irrefl _
  38     · exact hFxy
  39     · exact fun h=>hFxy h.symm
  40     · exact F.loopless.irrefl _
  41   have hh := hN V hn P F A {x,y} hrootcard hA θ hθ hθ1 hP hF hind
  42   let Good := {S : FixedSubset V (2+k) // ({x,y}:Finset V)⊆S.val ∧ S.val⊆A∪{x,y} ∧
  43     (∀ v∈S.val,(θ+α/4)*(2+k:ℝ)≤((P.neighborFinset v∩S.val).card:ℝ)) ∧
  44     (∀ z∈S.val,∀ t∈S.val,¬F.Adj z t)}
  45   let f (S : Good) : OrdinaryIsolatorCandidate P F H (2+k) θ (α/4) x y u :=
  46     ⟨S.val,⟨S.property.1 (by simp),S.property.1 (by simp)⟩,
  47       (by
  48         intro z hz hzx hzy
  49         have ha := S.property.2.1 hz
  50         have ha' : z∈A := by
  51           simpa only [Finset.mem_union,Finset.mem_insert,Finset.mem_singleton,hzx,hzy,
  52             false_or,or_false] using ha
  53         exact hH z ha'),
  54       S.property.2.2.2,(by simpa only [Nat.cast_add,Nat.cast_ofNat] using S.property.2.2.1)⟩
  55   have hinj : Function.Injective f := by
  56     intro S T he
  57     apply Subtype.ext
  58     exact congrArg (fun a : OrdinaryIsolatorCandidate P F H (2+k) θ (α/4) x y u=>a.val) he
  59   have hcard : (Fintype.card Good:ℝ)≤Fintype.card (OrdinaryIsolatorCandidate P F H (2+k) θ (α/4) x y u) := by
  60     exact_mod_cast Fintype.card_le_of_injective f hinj
  61   exact hh.trans hcard
