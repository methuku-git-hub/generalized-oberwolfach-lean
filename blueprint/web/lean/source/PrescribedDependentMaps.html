  13 def prescribedLiftedSet (A:∀ i,Finset (X i)) (i:I):Finset (Σ i,X i):=
  14   (A i).image (Sigma.mk i)

  16 def prescribedLiftedMap (v₀:V) (ψ:∀ i,X i→V) (i:I) (z:Σ i,X i):V:=
  17   if h:z.1=i then ψ i (h ▸ z.2) else v₀

  23 theorem prescribed_lifted_set_membership (A:∀ i,Finset (X i)) (i:I) (x:X i):
  24     Sigma.mk i x∈prescribedLiftedSet A i ↔ x∈A i := by
  25   simp only [prescribedLiftedSet,Finset.mem_image]
  26   constructor
  27   · rintro ⟨y,hy,he⟩
  28     have h:y=x:=sigma_mk_injective he
  29     simpa [h] using hy
  30   · intro hx;exact ⟨x,hx,rfl⟩
