  14 theorem marked_removed_typed_rim_from_pair_change
  15     (P D:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ)
  16     (σ τ:RawMarkedOutcome V U Γ Color)
  17     (hchange:∀ a, ¬(rawMarkedPairRims P π d σ a↔rawMarkedPairRims P π d τ a)→
  18       markedPairPhysical π a∈D.edgeFinset):
  19     ∀ x y,x≠y→markedRemovedPhysicalRim P π d σ τ s(x,y)→D.Adj x y :=by
  20   intro x y _ h
  21   obtain ⟨mark,a,b,hold,hnew,heq⟩:=h
  22   cases mark
  23   · have hdiff:¬(rawMarkedPairRims P π d σ (Sum.inl (a,b))↔
  24         rawMarkedPairRims P π d τ (Sum.inl (a,b))):=by
  25       simp only [rawMarkedPairRims,Sum.elim_inl]
  26       exact fun iff=>hnew (iff.mp hold)
  27     have hd:=hchange (Sum.inl (a,b)) hdiff
  28     simp only [markedPairPhysical,Sum.elim_inl] at hd
  29     simp only [Bool.false_eq_true,ite_false] at heq
  30     rw [←heq] at hd
  31     exact SimpleGraph.mem_edgeFinset.mp hd
  32   · have hdiff:¬(rawMarkedPairRims P π d σ (Sum.inr (a,b))↔
  33         rawMarkedPairRims P π d τ (Sum.inr (a,b))):=by
  34       simp only [rawMarkedPairRims,Sum.elim_inr]
  35       exact fun iff=>hnew (iff.mp hold)

  37     simp only [markedPairPhysical,Sum.elim_inr] at hd
  38     simp only [ite_true] at heq
