   9 theorem marked_seed_coefficient_bounds (m : ℕ) (hm : 1≤m)
  10     (n D p k s a B w pstar : ℝ)
  11     (hn : 0<n) (hDlo : n/2≤D) (hDhi : D≤2*n)
  12     (hpstar : 0<pstar) (hp : pstar≤p) (hk : pstar≤k) (hs : pstar≤s)
  13     (hp1 : p≤1) (hk1 : k≤1) (hs1 : s≤1) (ha : 0≤a) (hB : 0≤B)
  14     (hwlo : a/n^(m-1)≤w) (hwhi : w≤B/n^(m-1)) :
  15     (a/2)/n^m≤w/(D*p^m*k*s^m) ∧
  16       w/(D*p^m*k*s^m)≤(2*B/pstar^(2*m+1))/n^m := by
  17   have hp0 := hpstar.trans_le hp
  18   have hk0 := hpstar.trans_le hk
  19   have hs0 := hpstar.trans_le hs
  20   have hD0 : 0<D := (half_pos hn).trans_le hDlo
  21   have hw0 : 0≤w := (by positivity : 0≤a/n^(m-1)).trans hwlo
  22   have hpow : n^m=n^(m-1)*n := by rw [← pow_succ];congr 1;omega
  23   have hupper : p^m*k*s^m≤1 := by
  24     have hp' := pow_le_one₀ hp0.le hp1 (n:=m)
  25     have hs' := pow_le_one₀ hs0.le hs1 (n:=m)
  26     calc
  27       _ ≤ 1*1*1 := mul_le_mul (mul_le_mul hp' hk1 hk0.le (by norm_num)) hs'
  28         (pow_nonneg hs0.le _) (by norm_num)
  29       _ = 1 := by norm_num
  30   have hlower : pstar^(2*m+1)≤p^m*k*s^m := by
  31     calc
  32       _ = pstar^m*pstar*pstar^m := by rw [← pow_succ,← pow_add];congr 1;omega
  33       _ ≤ _ := mul_le_mul (mul_le_mul (pow_le_pow_left₀ hpstar.le hp m) hk hpstar.le
  34         (pow_nonneg hp0.le _)) (pow_le_pow_left₀ hpstar.le hs m)
  35         (pow_nonneg hpstar.le _) (mul_nonneg (pow_nonneg hp0.le _) hk0.le)
  36   have hdenlo : (n/2)*pstar^(2*m+1)≤D*p^m*k*s^m := by
  37     convert mul_le_mul hDlo hlower (by positivity) hD0.le using 1 <;> ring
  38   have hdenhi : D*p^m*k*s^m≤2*n := by
  39     calc
  40       _ = D*(p^m*k*s^m) := by ring
  41       _ ≤ D*1 := mul_le_mul_of_nonneg_left hupper hD0.le
  42       _ ≤ _ := by simpa using hDhi
  43   constructor
  44   · calc
  45       _ = (a/n^(m-1))/(2*n) := by rw [hpow];field_simp
  46       _ ≤ w/(D*p^m*k*s^m) := div_le_div₀ (by positivity) hwlo (by positivity) hdenhi
  47   · calc
  48       _ ≤ (B/n^(m-1))/((n/2)*pstar^(2*m+1)) :=
  49         div_le_div₀ (by positivity) hwhi (by positivity) hdenlo
  50       _ = _ := by rw [hpow];field_simp
  51 
  52 end Oberwolfach
