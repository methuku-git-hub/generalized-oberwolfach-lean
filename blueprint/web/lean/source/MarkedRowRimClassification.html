  18 theorem marked_row_raw_rim_classification (hn : 2≤n) (p : Fin (n+1) → V)
  19     (v : V) (jin jout : Bool) (htypes : ¬(jin=true ∧ jout=true))
  20     (X : Finset V) (hX : ∀ x, x∈X → π x∈X) (hp : ∀ i, p i∈X)
  21     (i : Fin (n+2)) :
  22     let k : Fin (n+2) := if jout then 0 else if jin then Fin.last (n+1) else 2
  23     let f := Fin.cons (α:=fun _=>V) v p
  24     let e : Finset V := if i=k then {f i,π (f (finRotate (n+2) i))} else {f i,f (finRotate (n+2) i)}
  25     e⊆X ∨ e∈markedRowRootRims π v (p 0) (p (Fin.last n)) jin jout := by
  26   dsimp only
  27   have h02 : (0 : Fin (n+2))≠2 := by
  28     intro h; have hh := congrArg Fin.val h
  29     change 0=2%(n+2) at hh
  30     rw [Nat.mod_eq_of_lt (by omega)] at hh; omega
  31   have h0last : (0 : Fin (n+2))≠Fin.last (n+1) := by
  32     intro h; have hh := congrArg Fin.val h; simp only [Fin.val_zero,Fin.val_last] at hh; omega
  33   have hl2 : (Fin.last (n+1) : Fin (n+2))≠2 := by
  34     intro h; have hh := congrArg Fin.val h
  35     change n+1=2%(n+2) at hh
  36     rw [Nat.mod_eq_of_lt (by omega)] at hh; omega
  37   refine Fin.cases ?_ (fun j => ?_) i
  38   · right
  39     have hr : finRotate (n+2) (0 : Fin (n+2))=(0 : Fin (n+1)).succ := finRotate_castSucc_eq_succ (n+1) 0
  40     cases jout <;> cases jin <;>
  41       simp [hr,markedRowRootRims,h02,h0last]
  42   · refine Fin.lastCases ?_ (fun j => ?_) j
  43     · right
  44       have hl : (Fin.last n).succ=Fin.last (n+1) := rfl
  45       rw [hl,finRotate_last]
  46       have hh : Fin.cons (α:=fun _=>V) v p (Fin.last (n+1))=p (Fin.last n) := rfl
  47       cases jout <;> cases jin <;> simp_all [markedRowRootRims,hh]
  48     · left

  50       rw [hi,finRotate_castSucc_eq_succ]

  56 theorem marked_assigned_row_rim_classification (hn : 2≤n) (p : Fin (n+1) → V)
  57     (v : V) (jin jout : Bool) (htypes : ¬(jin=true ∧ jout=true))
  58     (X : Finset V) (hX : ∀ x, x∈X → π x∈X) (hp : ∀ i, p i∈X)
  59     (C : MarkedWheel tail head ktail khead (n+1) π b H₀ HK)
  60     (hC : ∀ i, C.1.val i=Fin.cons (α:=fun _=>V) v p (markedAssignedRowOrder (n+1) jin jout i)) :
  61     ∀ e∈markedCyclePhysicalEdges tail head ktail khead (n+1) π b C.1,
  62       e⊆X ∨ e∈markedRowRootRims π v (p 0) (p (Fin.last n)) jin jout := by
  63   intro e he
  64   obtain ⟨i,hi,he⟩ := Finset.mem_image.mp he
  65   let k : Fin (n+2) := if jout then 0 else if jin then Fin.last (n+1) else 2
  66   let ρ := canonicalMarkedRotation k
  67   have hraw : ∀ i, C.1.val i=Fin.cons (α:=fun _=>V) v p (ρ i) := by
  68     intro i; rw [hC,markedAssignedRowOrder_canonical]
  69   have hr := marked_wheel_raw_rim_position C (Fin.cons v p) k hraw (ρ i)
  70   change markedWheelPositionResource C (Sum.inl (ρ.symm (ρ i))) = _ at hr
  71   rw [Equiv.symm_apply_apply] at hr
  72   have heq : e=(if ρ i=k then
  73       {Fin.cons (α:=fun _=>V) v p (ρ i),π (Fin.cons (α:=fun _=>V) v p (finRotate (n+2) (ρ i)))}
  74       else {Fin.cons (α:=fun _=>V) v p (ρ i),Fin.cons (α:=fun _=>V) v p (finRotate (n+2) (ρ i))}) := by
  75     apply Sum.inl_injective (β:=V×U)
  76     calc
  77       Sum.inl e = markedWheelPositionResource C (Sum.inl i) := congrArg Sum.inl he.symm
  78       _ = _ := hr
  79   rw [heq]
  80   exact marked_row_raw_rim_classification hn p v jin jout htypes X hX hp (ρ i)
