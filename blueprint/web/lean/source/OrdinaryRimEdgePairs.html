  15 theorem ordinary_rim_edge_pair (Q F:SimpleGraph V) (H:V→U→Prop)
  16     (A W Ω Z:Finset V) (B:Finset U) (hAΩ:A⊆Ω)
  17     (x y x' y':V) (hxy:Q.Adj x y) (hxy':Q.Adj x' y')
  18     (hx'x:x'≠x) (hx'y:x'≠y) (hy'x:y'≠x) (hy'y:y'≠y)
  19     (hroots:x∈Z∧y∈Z∧x'∈Z∧y'∈Z) (hZcard:Z.card≤8)
  20     (hΩ:x∈Ω∧y∈Ω∧x'∈Ω∧y'∈Ω)
  21     (hZ:∀ v∈Z,∀ w∈Z,¬F.Adj v w)
  22     (r:ℕ) (hr:4≤r) (L b bH:ℝ) (hL:0≤L) (hF:∀ v,(F.degree v:ℝ)≤L)
  23     (hcount:∀ v∈Ω,∀ w∈Ω,b≤((A∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  24     (hthree:r=4→∀ a∈A,b≤(((A∩Q.neighborFinset a)∩Q.neighborFinset y∩Q.neighborFinset y').card:ℝ))
  25     (hhubs:∀ S:Finset V,S.card≤r+6→(∀ v∈S,∀ w∈S,¬F.Adj v w)→
  26       bH≤((common (Finset.univ:Finset U) S (fun v=>Finset.univ.filter (H v))).card:ℝ))
  27     (hbudget:W.card+((Z.card:ℝ)+r+2)*(L+1)<b) (hhubBudget:(B.card:ℝ)<bH):
  28     ∃ a b',∃ p:Q.Walk a b',∃ u:U,∃ C C':OrdinaryWheel Q F H r,
  29       p.IsPath ∧ p.length=r-3 ∧ u∉B ∧ (∀ v∈Z,H v u) ∧
  30       (∀ v∈p.support,v∈A∧v∉W∧v∉Z) ∧
  31       ordinaryWheelHub C=u ∧ ordinaryWheelHub C'=u ∧
  32       (∀ v,v∈(ordinaryWheelRim C).support↔v=x∨v=y∨v∈p.support) ∧
  33       (∀ v,v∈(ordinaryWheelRim C').support↔v=x'∨v=y'∨v∈p.support) ∧
  34       (∀ e:Sym2 V,
  35         (if Sum.inl e∈ordinaryWheelResources C then (1:ℤ) else 0)-
  36           (if Sum.inl e∈ordinaryWheelResources C' then 1 else 0)=
  37         (if e=s(x,y) then 1 else 0)+(if e=s(x,a) then 1 else 0)+(if e=s(y,b') then 1 else 0)-
  38           (if e=s(x',y') then 1 else 0)-(if e=s(x',a) then 1 else 0)-(if e=s(y',b') then 1 else 0)) ∧
  39       (∀ v z,
  40         (if Sum.inr (v,z)∈ordinaryWheelResources C then (1:ℤ) else 0)-
  41           (if Sum.inr (v,z)∈ordinaryWheelResources C' then 1 else 0)=
  42         (if v=x∧z=u then 1 else 0)+(if v=y∧z=u then 1 else 0)-
  43           (if v=x'∧z=u then 1 else 0)-(if v=y'∧z=u then 1 else 0)):=by
  44   obtain ⟨a,c,p,hp,hlen,hxa,hx'a,hyc,hy'c,hclass,hind,hroot⟩:=
  45     transformer_rim_common_path Q F A W Ω Z hAΩ x y x' y' hroots hΩ hZ r hr L b hL hF
  46       hcount hthree hbudget
  47   let S:=Z∪p.support.toFinset
  48   have hSsize:S.card≤r+6:=by
  49     have hpCard:p.support.toFinset.card≤r-2:=by
  50       have hh:=List.toFinset_card_le p.support
  51       rw [p.length_support,hlen] at hh
  52       omega
  53     exact (Finset.card_union_le Z p.support.toFinset).trans (by omega)
  54   have hSF:∀ v∈S,∀ w∈S,¬F.Adj v w:=by
  55     intro v hv w hw
  56     rcases Finset.mem_union.mp hv with hv|hv <;> rcases Finset.mem_union.mp hw with hw|hw
  57     · exact hZ v hv w hw
  58     · exact fun h=>hroot w (List.mem_toFinset.mp hw) v hv h.symm
  59     · exact hroot v (List.mem_toFinset.mp hv) w hw
  60     · exact hind v (List.mem_toFinset.mp hv) w (List.mem_toFinset.mp hw)
  61   have hlarge:B.card<(common (Finset.univ:Finset U) S (fun v=>Finset.univ.filter (H v))).card:=by
  62     exact_mod_cast hhubBudget.trans_le (hhubs S hSsize hSF)
  63   obtain ⟨u,hu,huB⟩:=Finset.exists_mem_notMem_of_card_lt_card hlarge
  64   have hSH:∀ v∈S,H v u:=by
  65     simpa only [common,Finset.mem_filter,Finset.mem_univ,true_and] using hu
  66   have hrootSubset:({x,y,x',y'}:Finset V)⊆Z:=by
  67     intro v hv
  68     simp only [Finset.mem_insert,Finset.mem_singleton] at hv
  69     rcases hv with rfl|rfl|rfl|rfl
  70     · exact hroots.1
  71     · exact hroots.2.1
  72     · exact hroots.2.2.1
  73     · exact hroots.2.2.2
  74   have hsmall:insert x (insert y (insert x' (insert y' p.support.toFinset)))⊆S:=by
  75     intro v hv
  76     simp only [Finset.mem_insert] at hv
  77     rcases hv with hv|hv|hv|hv|hv
  78     · exact Finset.mem_union_left _ (hv.symm ▸ hroots.1)
  79     · exact Finset.mem_union_left _ (hv.symm ▸ hroots.2.1)
  80     · exact Finset.mem_union_left _ (hv.symm ▸ hroots.2.2.1)
  81     · exact Finset.mem_union_left _ (hv.symm ▸ hroots.2.2.2)
  82     · exact Finset.mem_union_right _ hv
  83   obtain ⟨C,C',hCu,hC'u,hCs,hC's,hrim,hspoke⟩:=ordinary_edge_path_wheels Q F H r (by omega)
  84     x y x' y' a c u p hp hlen hxy hxa hyc hxy' hx'a hy'c hx'x hx'y hy'x hy'y
  85     (fun v hv h=>(hclass v hv).2.2 (hrootSubset h))
  86     (fun v hv w hw=>hSF v (hsmall hv) w (hsmall hw)) (fun v hv=>hSH v (hsmall hv))
  87   exact ⟨a,c,p,u,C,C',hp,hlen,huB,(fun v hv=>hSH v (Finset.mem_union_left _ hv)),hclass,
  88     hCu,hC'u,hCs,hC's,hrim,hspoke⟩
