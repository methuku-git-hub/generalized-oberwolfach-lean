  15 theorem near_regular_cyclic_matrix_weights (β : ℝ) (hβ : 0 < β) :
  16     ∃ C η₀ : ℝ, 0 < C ∧ 0 < η₀ ∧ ∀ η : ℝ, 0 < η → η < η₀ →
  17       ∃ m₀ : ℕ, ∀ (k : ℕ) (X : Type) [Fintype X] [Nonempty X],
  18       m₀ ≤ Fintype.card X →
  19       ∀ (M : (Fin (k + 2) ⊕ Fin (k + 2)) → X → X → ℕ),
  20       (∀ i x y, M i x y ≤ 1) → ∀ d α : ℝ, 0 < α →
  21       (∀ i x, (1 / 2 + β) * Fintype.card X ≤ ((∑ y, M i x y : ℕ) : ℝ)) →
  22       (∀ i y, (1 / 2 + β) * Fintype.card X ≤ ((∑ x, M i x y : ℕ) : ℝ)) →
  23       (∀ i x, d - η * Fintype.card X ≤ ((∑ y, M i x y : ℕ) : ℝ) ∧
  24         ((∑ y, M i x y : ℕ) : ℝ) ≤ d + η * Fintype.card X) →
  25       (∀ i y, d - η * Fintype.card X ≤ ((∑ x, M i x y : ℕ) : ℝ) ∧
  26         ((∑ x, M i x y : ℕ) : ℝ) ≤ d + η * Fintype.card X) →
  27       let q := ⌊d - Real.sqrt η * Fintype.card X⌋₊
  28       ((k + 2) + (2 * (k + 2) - 1) * α) * Fintype.card X ≤
  29         (2 * (k + 2) - 1) * (q : ℝ) →
  30     ∃ w : (Fin (k + 2) → X × X) → ℝ,
  31       (∀ v, 0 ≤ w v) ∧
  32       (∀ v, 0 < w v → CyclicMatrixTupleValid M v) ∧
  33       (∀ i x y, (1 - C * Real.sqrt η) * M (Sum.inl i) x y ≤
  34         (∑ v, if v i = (x,y) then w v else 0) ∧
  35         (∑ v, if v i = (x,y) then w v else 0) ≤ M (Sum.inl i) x y) ∧
  36       (∀ i y z, (1 - C * Real.sqrt η) * M (Sum.inr i) y z ≤
  37         (∑ v, if ((v i).2, (v (finRotate (k + 2) i)).1) = (y,z) then w v else 0) ∧
  38         (∑ v, if ((v i).2, (v (finRotate (k + 2) i)).1) = (y,z) then w v else 0) ≤
  39           M (Sum.inr i) y z) ∧
  40       (∀ i x, (∑ v, if (v i).1 = x then w v else 0) = q ∧
  41         (∑ v, if (v i).2 = x then w v else 0) = q) ∧
  42       (∀ v, w v ≤ ((k + 2 : ℝ) * cyclicSectorUpper k α) /
  43         (Fintype.card X : ℝ) ^ (2 * k + 2)) ∧
  44       (∀ v, CyclicMatrixTupleValid M v →
  45         (1 - (2 * (k + 2 : ℝ)) * (C * Real.sqrt η)) *
  46           (((k + 2 : ℝ) * cyclicSectorLower k α) /
  47             (Fintype.card X : ℝ) ^ (2 * k + 2)) ≤ w v) := by
  48   obtain ⟨C, η₀, hC, hη₀, hall⟩ := interface_family_random_regularization β hβ
  49   refine ⟨C, η₀, hC, hη₀, ?_⟩
  50   intro η hη hηsmall
  51   obtain ⟨m₀, hm₀⟩ := hall η hη hηsmall
  52   refine ⟨m₀, ?_⟩
  53   intro k X _ _ hm M hM d α hα hl hr hnl hnr q hthreshold
  54   obtain ⟨ν, hregular, hsub, hprob⟩ := hm₀ (Fin (k + 2) ⊕ Fin (k + 2)) X hm
  55     M hM d hl hr hnl hnr
  56   let R (D : {D : ∀ i, BipartiteEdges (M i) → Bool // 0 < ν.mass D})
  57       (i : Fin (k + 2) ⊕ Fin (k + 2)) := retainedBipartiteMatrix (M i) (D.1 i)
  58   apply stochastic_cyclic_weights ν.support M hM R
  59     (fun D i x y => hsub D.1 i x y) q
  60     (fun D i x => (hregular D.1 D.2 i).1 x)
  61     (fun D i y => (hregular D.1 D.2 i).2 y) (C * Real.sqrt η) α hα _ hthreshold
  62   intro i x y hxy
  63   let e : BipartiteEdges (M i) := ⟨x,y,⟨0,hxy⟩⟩
  64   have h := hprob i e
  65   dsimp only [R]
  66   rw [finiteLaw_support_probability ν (fun D => retainedBipartiteMatrix (M i) (D i) x y = 0)]
  67   simpa only [e] using h
