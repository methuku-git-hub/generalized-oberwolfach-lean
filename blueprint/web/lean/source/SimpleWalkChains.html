  18 def toWalk : {a b:V} → SimpleWalkChain G a b → G.Walk a b
  19   | _,_,.nil _ => .nil
  20   | _,_,.cons p q => p.append q.toWalk

  22 def initials : {a b:V} → SimpleWalkChain G a b → Finset V
  23   | _,_,.nil _ => ∅
  24   | _,_,.cons p q => p.support.dropLast.toFinset ∪ q.initials

  26 def totalLength : {a b:V} → SimpleWalkChain G a b → ℕ
  27   | _,_,.nil _ => 0
  28   | _,_,.cons p q => p.length+q.totalLength

  30 theorem mem_toWalk_support {a b:V} (q:SimpleWalkChain G a b) (x:V):
  31     x∈q.toWalk.support ↔ x=b ∨ x∈q.initials := by
  32   induction q with
  33   | nil v => simp [toWalk,initials]
  34   | cons p q ih =>
  35     simp only [toWalk,SimpleGraph.Walk.support_append_eq_support_dropLast_append,List.mem_append,
  36       initials,Finset.mem_union,List.mem_toFinset,ih]
  37     tauto

  39 theorem toWalk_length {a b:V} (q:SimpleWalkChain G a b):q.toWalk.length=q.totalLength := by
  40   induction q with
  41   | nil v => rfl
  42   | cons p q ih => simp [toWalk,totalLength,ih]

  50 theorem toWalk_isPath {a b:V} {q:SimpleWalkChain G a b} (h:Compatible q):q.toWalk.IsPath := by
  51   induction h with
  52   | nil v => exact SimpleGraph.Walk.IsPath.nil
  53   | cons p q hp hq hdis ih =>
  54     apply (simple_walk_append_of_disjoint_initial_support p q.toWalk hp ih ?_).1
  55     intro x hx hmem
  56     rcases (mem_toWalk_support q x).mp hmem with hlast | hinitial
  57     · exact (hdis x hx).2 hlast
  58     · exact (hdis x hx).1 hinitial
