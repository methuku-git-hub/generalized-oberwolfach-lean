  14 theorem long_residual_system_positive {X : Type*} [Fintype X]
  15     (a : ℕ → X → X → ℝ) (d : ℕ → X → ℝ)
  16     (ha : ∀ i x y, 0 ≤ a i x y) (hd : ∀ i x, 0 < d i x)
  17     (hrow : ∀ i x, ∑ y, a i x y=d i x)
  18     (hcol : ∀ i y, ∑ x, a i x y=d (i+1) y)
  19     (s L : Fin 3 → ℕ) (hL : ∀ i, 0 < L i)
  20     (hbd : ∀ i x, d (s i+L i) x=d (s (finRotate 3 i)) x)
  21     (hT : ∀ i x z, 0 < transitionWalk (capacityTransition a d) (s i) (L i) x z)
  22     (hres : ∀ (Q : transportProduct (threeIntervalMarginal (intervalDegrees d s L)) (finRotate 3))
  23       i j x y, pathEdgeLoad (longCapacityReserve a d s L Q i) j x y ≤ (11/27)*a (s i+j.val) x y) :
  24     ∀ Q i x z, 0 < threeIntervalResidualSystem (intervalCapacities a s L) (intervalDegrees d s L)
  25       (longCapacityReserve a d s L) Q i x z := by
  26   let R := capacityTransition a d
  27   let A := intervalCapacities a s L
  28   let D := intervalDegrees d s L
  29   let ρ := longCapacityReserve a d s L
  30   have hR := capacityTransition_isTransition a d ha hd hrow
  31   have hD (i : Fin 3) (j : Fin (L i+1)) (x : X) : 0 < D i j x := hd _ _
  32   have hA (i : Fin 3) (j : Fin (L i)) (x y : X) : 0 ≤ A i j x y := ha _ _ _
  33   have hAr (i : Fin 3) (j : Fin (L i)) (x : X) : (∑ y, A i j x y)=D i j.castSucc x := hrow _ _
  34   have hAc (i : Fin 3) (j : Fin (L i)) (y : X) : (∑ x, A i j x y)=D i j.succ y := by
  35     simpa only [A,D,intervalCapacities,intervalDegrees,Fin.val_succ,Nat.add_assoc] using hcol (s i+j.val) y
  36   intro Q i x z
  37   have hQ : Q.val i ∈ transportMatrices (fun x => D i 0 x/3) (fun z => D i (Fin.last (L i)) z/3) := by
  38     have hh := Q.property i (Set.mem_univ i)
  39     change Q.val i ∈ transportMatrices (fun x => d (s i) x/3) (fun z => d (s (finRotate 3 i)) z/3) at hh
  40     simpa only [D,intervalDegrees,Fin.val_zero,Nat.add_zero,Fin.val_last,hbd] using hh
  41   have hρ0 (p : Fin (L i+1) → X) : 0 ≤ ρ Q i p :=
  42     reservedPathWeight_nonneg R hR (s i) (L i) (Q.val i) hQ.1 p
  43   have hρend (x z : X) : (∑ p, if p 0=x ∧ p (Fin.last (L i))=z then ρ Q i p else 0)=Q.val i x z :=
  44     reservedPathWeight_endpoints R (s i) (L i) (Q.val i) (hT i) x z
  45   have hpkg := halfResidualPathWeight_properties (hL i) (A i) (D i) (ρ Q i) (Q.val i)
  46     (hA i) (hD i) (hAr i) (hAc i) hρ0 (hres Q i) hρend hQ
  47   obtain ⟨p,hpx,hpz,hpath⟩ := positive_transition_path_exists R hR (s i) (L i) x z (hT i x z)
  48   have haPath (j : Fin (L i)) : 0 < A i j (p j.castSucc) (p j.succ) := by
  49     have hh := hpath j
  50     change 0 < a (s i+j.val) (p j.castSucc) (p j.succ) / d (s i+j.val) (p j.castSucc) at hh
  51     exact (div_pos_iff_of_pos_right (hd _ _)).mp hh
  52   have hpos := halfResidualPathWeight_pos (hL i) (A i) (D i) (ρ Q i)
  53     (hA i) (hD i) (hAr i) (hAc i) hρ0 (hres Q i) p haPath
  54   exact pathEndpointMatrix_pos_of_path _ hpkg.1 x z p ⟨hpx,hpz⟩ hpos
