  16 abbrev ExpansionRow (V E J W : Type*) := V ⊕ ((E ⊕ J) × W)
  17 abbrev ExpansionHub (U E J : Type*) := U ⊕ (E ⊕ J)

  19 abbrev expansionPrivateAdj (C : SimpleGraph W) (z o : W) :
  20     E ⊕ J → W → W → Prop
  21   | Sum.inl _,a,b => C.Adj a b ∧ ¬ ((a=z ∧ b=o) ∨ (a=o ∧ b=z))
  22   | Sum.inr _,a,b => C.Adj a b
  23 
  24 abbrev expansionCrossAdj (z o : W) (x y : E → V) : V → (E ⊕ J) × W → Prop
  25   | a,(Sum.inl e,w) => (a=x e ∧ w=o) ∨ (a=y e ∧ w=z)
  26   | _,(Sum.inr _,_) => False

  28 abbrev ordinaryExpansionRim (C : SimpleGraph W) (z o : W) (x y : E → V) :
  29     SimpleGraph (ExpansionRow V E J W) where
  30   Adj a b := match a,b with
  31     | Sum.inl _,Sum.inl _ => False
  32     | Sum.inl a,Sum.inr p => expansionCrossAdj z o x y a p
  33     | Sum.inr p,Sum.inl a => expansionCrossAdj z o x y a p
  34     | Sum.inr (i,w),Sum.inr (j,w') => i=j ∧ expansionPrivateAdj C z o i w w'
  35   symm := ⟨by
  36     intro a b hab
  37     cases a with
  38     | inl a => cases b <;> exact hab
  39     | inr p =>
  40       cases b with
  41       | inl b => exact hab
  42       | inr q =>
  43         rcases p with ⟨i,w⟩; rcases q with ⟨j,w'⟩
  44         obtain ⟨rfl,h⟩ := hab
  45         refine ⟨rfl,?_⟩
  46         cases i <;> simp only [expansionPrivateAdj] at h ⊢
  47         · exact ⟨h.1.symm,by tauto⟩
  48         · exact h.symm⟩
  49   loopless := ⟨by
  50     intro a
  51     cases a with
  52     | inl a => exact not_false
  53     | inr p =>
  54       rcases p with ⟨i,w⟩
  55       intro h
  56       cases i <;> simp only [expansionPrivateAdj] at h
  57       · exact h.2.1.ne rfl
  58       · exact h.2.ne rfl⟩

  82 abbrev ordinaryExpansionSpokes (z : W) (v : J → V) (u : J → U) :
  83     ExpansionRow V E J W → ExpansionHub U E J → Prop
  84   | Sum.inl _,Sum.inl _ => False
  85   | Sum.inl _,Sum.inr (Sum.inl _) => False
  86   | Sum.inl a,Sum.inr (Sum.inr j) => a=v j
  87   | Sum.inr (Sum.inl _,_),Sum.inl _ => False
  88   | Sum.inr (Sum.inr j,w),Sum.inl b => w=z ∧ b=u j
  89   | Sum.inr (Sum.inl e,_),Sum.inr i => i=Sum.inl e
  90   | Sum.inr (Sum.inr j,w),Sum.inr i => w≠z ∧ i=Sum.inr j

 102 def naturalIndicator (p : Prop) : ℕ := if p then 1 else 0

 104 theorem naturalIndicator_true : naturalIndicator True = 1 := by simp [naturalIndicator]
 105 theorem naturalIndicator_false : naturalIndicator False = 0 := by simp [naturalIndicator]

 107 theorem naturalIndicator_and (p q : Prop) :
 108     naturalIndicator (p ∧ q) = if p then naturalIndicator q else 0 := by
 109   by_cases hp : p <;> simp [naturalIndicator,hp]

 117 theorem graph_degree_indicator_sum {A : Type*} [Fintype A] (G : SimpleGraph A) (a : A) :
 118     G.degree a = ∑ b : A,naturalIndicator (G.Adj a b) := by
 119   rw [← SimpleGraph.card_neighborFinset_eq_degree]
 120   have h : G.neighborFinset a = Finset.univ.filter (G.Adj a) := by ext; simp
 121   rw [h,Finset.card_eq_sum_ones,Finset.sum_filter]
 122   rfl

 124 theorem ordinaryExpansionRim_degree_old (C : SimpleGraph W) (z o : W) (hzo : z≠o)
 125     (x y : E → V) (a : V) :
 126     (ordinaryExpansionRim (J:=J) C z o x y).degree (.inl a) =
 127       (Finset.univ.filter (fun e => a=x e)).card +
 128       (Finset.univ.filter (fun e => a=y e)).card := by
 129   rw [graph_degree_indicator_sum,Fintype.sum_sum_type]
 130   simp only [ordinaryExpansionRim,naturalIndicator_false,Finset.sum_const_zero,zero_add,Fintype.sum_prod_type]
 131   rw [Fintype.sum_sum_type]
 132   simp only [expansionCrossAdj,naturalIndicator_false,Finset.sum_const_zero,add_zero]
 133   have hi (e : E) (w : W) :
 134       naturalIndicator ((a=x e ∧ w=o) ∨ (a=y e ∧ w=z)) =
 135       (if a=x e then naturalIndicator (w=o) else 0)+
 136       (if a=y e then naturalIndicator (w=z) else 0) := by
 137     by_cases ha : a=x e <;> by_cases hb : a=y e <;> by_cases hw : w=o <;> by_cases hz : w=z
 138     all_goals simp_all [naturalIndicator]
 139   simp_rw [hi,Finset.sum_add_distrib]
 140   simp only [Finset.sum_ite_irrel,Finset.sum_const_zero,naturalIndicator_sum_eq]
 141   simp only [Finset.card_eq_sum_ones,Finset.sum_filter]

 143 theorem ordinaryExpansionRim_degree_private (C : SimpleGraph W) (z o : W)
 144     (hzo : C.Adj z o) (x y : E → V) (i : E ⊕ J) (w : W) :
 145     (ordinaryExpansionRim C z o x y).degree (.inr (i,w)) = C.degree w := by
 146   rw [graph_degree_indicator_sum,Fintype.sum_sum_type]
 147   simp only [ordinaryExpansionRim,Fintype.sum_prod_type]
 148   have hprivate : (∑ p : E ⊕ J,∑ w' : W,
 149       naturalIndicator (i=p ∧ expansionPrivateAdj C z o i w w')) =
 150       ∑ w' : W,naturalIndicator (expansionPrivateAdj C z o i w w') := by
 151     simp_rw [naturalIndicator_and]
 152     simp
 153   rw [hprivate,graph_degree_indicator_sum]
 154   cases i with
 155   | inr j => simp [expansionCrossAdj,expansionPrivateAdj,naturalIndicator_false]
 156   | inl e =>
 157     have he (a : V) :
 158         naturalIndicator (expansionCrossAdj (J:=J) z o x y a (Sum.inl e,w)) =
 159         (if w=o then naturalIndicator (a=x e) else 0)+
 160         (if w=z then naturalIndicator (a=y e) else 0) := by
 161       by_cases ha : a=x e <;> by_cases hb : a=y e <;> by_cases hw : w=o <;> by_cases hz : w=z
 162       all_goals simp_all [expansionCrossAdj,hzo.ne,naturalIndicator]
 163     simp_rw [he,Finset.sum_add_distrib]
 164     simp only [Finset.sum_ite_irrel,Finset.sum_const_zero,naturalIndicator_sum_eq]
 165     have hcount (w' : W) :
 166         naturalIndicator (expansionPrivateAdj C z o (Sum.inl e : E ⊕ J) w w')+
 167         naturalIndicator (w=o ∧ w'=z)+naturalIndicator (w=z ∧ w'=o)=
 168         naturalIndicator (C.Adj w w') := by
 169       by_cases hw : w=o <;> by_cases hz : w=z <;> by_cases hw' : w'=o <;> by_cases hz' : w'=z
 170       all_goals simp_all [expansionPrivateAdj,hzo.ne,hzo.symm,naturalIndicator]
 171     have hs := congrArg (fun f : W → ℕ => ∑ w',f w') (funext hcount)
 172     have hzsum : (∑ w' : W,naturalIndicator (w=o ∧ w'=z))=if w=o then 1 else 0 := by
 173       simp only [naturalIndicator_and,Finset.sum_ite_irrel,naturalIndicator_sum_eq,Finset.sum_const_zero]
 174     have hosum : (∑ w' : W,naturalIndicator (w=z ∧ w'=o))=if w=z then 1 else 0 := by
 175       simp only [naturalIndicator_and,Finset.sum_ite_irrel,naturalIndicator_sum_eq,Finset.sum_const_zero]
 176     rw [Finset.sum_add_distrib,Finset.sum_add_distrib,hzsum,hosum] at hs
 177     omega
