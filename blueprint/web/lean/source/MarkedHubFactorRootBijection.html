  11 theorem marked_hub_factor_root_bijection
  12     {V U E K:Type*} [Fintype V] [DecidableEq V]
  13     (tail head:E→V) (ktail khead:K→V) (m b:ℕ) (π:Equiv.Perm V)
  14     (H₀ Hₖ:V→U→Prop) (u:U)
  15     (M:Finset (HubMarkedCycle tail head ktail khead m π b H₀ Hₖ u))
  16     (hM:∀C∈M,∀D∈M,C≠D→Disjoint (hubMarkedCycleVertices u C) (hubMarkedCycleVertices u D))
  17     (S Y:Finset V) (hcover:M.biUnion (hubMarkedCycleVertices u)=S)
  18     (hYS:Y⊆S) (hmarked:∀x,Hₖ x u↔x∈Y) (hsep:∀x,Hₖ x u→¬H₀ x u):
  19     ∃select:{x // x∈Y}↪HubMarkedCycle tail head ktail khead m π b H₀ Hₖ u,
  20       (∀x,select x∈M ∧ (select x).val 0=x.val) ∧
  21       (∀C∈M,∃x:{x // x∈Y},select x=C) := by
  22   obtain ⟨select,hselected⟩:=marked_hub_factor_root_selection tail head ktail khead m b π H₀ Hₖ u M S Y hcover hYS
  23     (fun x hx=>(hmarked x).2 hx) hsep
  24   refine ⟨select,hselected,?_⟩
  25   intro C hCM
  26   have hx:C.val 0∈Y:=(hmarked (C.val 0)).1 C.property.2.1
  27   let x:{x // x∈Y}:=⟨C.val 0,hx⟩
  28   refine ⟨x,?_⟩
  29   by_contra hne
  30   have hdis:=hM (select x) (hselected x).1 C hCM hne
  31   have hroot:(select x).val 0=C.val 0:=by simpa [x] using (hselected x).2
  32   have ha:(select x).val 0∈hubMarkedCycleVertices u (select x):=Finset.mem_image.mpr ⟨0,Finset.mem_univ _,rfl⟩
  33   have hb:(select x).val 0∈hubMarkedCycleVertices u C:=by
  34     rw [hroot]
  35     exact Finset.mem_image.mpr ⟨0,Finset.mem_univ _,rfl⟩
  36   exact Finset.disjoint_left.mp hdis ha hb
  37 end Oberwolfach
