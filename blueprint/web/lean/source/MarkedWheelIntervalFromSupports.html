  12 theorem marked_wheel_interval_from_rim_supports
  13     {V:Type*} (G:SimpleGraph V) (v:ℕ→V) (c n start:ℕ)
  14     (C:ℕ→Fin c→V) (p:∀i:ℕ,G.Walk (v i) (v (i+1)))
  15     (hp:∀i,start ≤ i → i < start+n → (p i).IsPath)
  16     (hlen:∀i,start ≤ i → i < start+n → (p i).length=c)
  17     (hsupport:∀i,start ≤ i → i < start+n → (p i).support.dropLast=List.ofFn (C i))
  18     (hdis:∀i j,start ≤ i → i < start+n → start ≤ j → j < start+n → i≠j →
  19       ∀a b:Fin c,C i a≠C j b)
  20     (hend:∀i,start ≤ i → i < start+n → ∀a:Fin c,v (start+n)≠C i a):
  21     ∃q:G.Walk (v start) (v (start+n)),q.IsPath ∧ q.length=n*c ∧
  22       ∀x,x∈q.support ↔ x=v (start+n) ∨
  23         ∃i,start ≤ i ∧ i < start+n ∧ ∃a:Fin c,C i a=x := by
  24   have hd':∀i j,start ≤ i → i < start+n → start ≤ j → j < start+n → i≠j →
  25       ∀x∈(p i).support.dropLast,x∉(p j).support.dropLast:=by
  26     intro i j hi hj hk hl hij x hx hy
  27     rw [hsupport i hi hj] at hx
  28     rw [hsupport j hk hl] at hy
  29     obtain ⟨a,ha⟩:=List.mem_ofFn.mp hx
  30     obtain ⟨b,hb⟩:=List.mem_ofFn.mp hy
  31     exact hdis i j hi hj hk hl hij a b (ha.trans hb.symm)
  32   have he':∀i,start ≤ i → i < start+n → v (start+n)∉(p i).support.dropLast:=by
  33     intro i hi hj hx
  34     rw [hsupport i hi hj] at hx
  35     obtain ⟨a,ha⟩:=List.mem_ofFn.mp hx
  36     exact hend i hi hj a ha.symm
  37   obtain ⟨q,hq,hqlen,hqmem⟩:=simple_walk_interval_concatenation G v p c n start hp hlen hd' he'
  38   refine ⟨q,hq,hqlen,?_⟩
  39   intro x
  40   rw [hqmem]
  41   constructor
  42   · rintro (h|⟨i,hi,hj,hx⟩)
  43     · exact Or.inl h
  44     · right;refine ⟨i,hi,hj,?_⟩
  45       rw [hsupport i hi hj] at hx
  46       exact List.mem_ofFn.mp hx
  47   · rintro (h|⟨i,hi,hj,a,ha⟩)
  48     · exact Or.inl h
  49     · right;refine ⟨i,hi,hj,?_⟩
  50       rw [hsupport i hi hj]
  51       exact List.mem_ofFn.mpr ⟨a,ha⟩
