  13 def recolourCoordinates {I E C : Type*} [Fintype I] [DecidableEq E]
  14     (ι : I↪E) (σ : E → C) (c : I → C) (e : E) : C :=
  15   if he : e∈Set.range ι then c (ι.toEquivRange.symm ⟨e,he⟩) else σ e

  40 theorem disjoint_pattern_stability {E I : Type*} [DecidableEq E]
  41     {State : E → Type*} (C : Finset I) (F : I → Finset E)
  42     (s : I → ∀ e,State e) (D : Finset E) (σ τ : ∀ e,State e)
  43     (hdisj : ∀ i∈C,∀ j∈C,i≠j → Disjoint (F i) (F j))
  44     (h : ∀ e,e∉D → σ e=τ e) :
  45     |(∑ i∈C,patternValue (F i) (s i) τ)-(∑ i∈C,patternValue (F i) (s i) σ)|≤D.card := by
  46   have hh := weighted_product_stability C F s (fun _ => 1) D σ τ (by intros;norm_num)
  47     (fun i hi e he heD => h e heD)
  48   simp only [one_mul] at hh
  49   apply hh.trans
  50   have hi (e : E) : (∑ i∈C,if e∈F i then (1:ℝ) else 0)≤1 := by
  51     classical
  52     have hcard : (C.filter (fun i => e∈F i)).card≤1 := by
  53       apply Finset.card_le_one.mpr
  54       intro i hi j hj
  55       by_contra hij
  56       exact Finset.disjoint_left.mp (hdisj i (Finset.mem_filter.mp hi).1 j
  57         (Finset.mem_filter.mp hj).1 hij) (Finset.mem_filter.mp hi).2 (Finset.mem_filter.mp hj).2
  58     have heq : (∑ i∈C,if e∈F i then (1:ℝ) else 0)=((C.filter (fun i => e∈F i)).card:ℝ) := by
  59       simp only [Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  60     rw [heq]
  61     exact_mod_cast hcard
  62   calc
  63     _ ≤ ∑ _e∈D,(1:ℝ) := Finset.sum_le_sum (fun e _ => hi e)
  64     _ = _ := by simp
