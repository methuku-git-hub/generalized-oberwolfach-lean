  20     ∀(f:{x // x∈A}↪V),
  21     (∀x (hx:x∈A),ψ x=f ⟨x,hx⟩)→
  22     (∀z (k:Fin ((actualOldPieceWalk data z).length+1)),

  28     (∀j:{j // j∈data.t},R (ψ (data.root j.val)) (ψ (data.C j.val).snd)):=by

  33       Finset.mem_biUnion.mpr ⟨z,Finset.mem_univ _,List.mem_toFinset.mpr ((actualOldPieceWalk data z).getVert_mem_support _)⟩
