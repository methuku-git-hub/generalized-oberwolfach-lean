  10 theorem factor_class_pruning_bound {W:Type} [Fintype W] (K:ℕ)
  11     (classify:W→Option (Fin K)) (n κL κS:ℝ)
  12     (hn:0≤n) (hL:0≤κL) (hS:0≤κS):
  13     let fibres:=fun j:Option (Fin K)=>Finset.univ.filter (fun w=>classify w=j)
  14     let small:=fun j:Option (Fin K)=>
  15       ((fibres j).card:ℝ)<(if j=none then κL*n else κS*n)
  16     let removed:=Finset.univ.filter (fun w=>small (classify w));
  17     (removed.card:ℝ)≤(κL+(K:ℝ)*κS)*n ∧
  18     (∀ j,¬small j→(fibres j∩removed).card=0):=by
  19   intro fibres small removed
  20   have hcount (j:Option (Fin K)):
  21       (((removed.filter (fun w=>classify w=j)).card:ℝ))≤
  22         (if j=none then κL*n else κS*n):=by
  23     by_cases hj:small j
  24     · have hsub:removed.filter (fun w=>classify w=j)⊆fibres j:=by
  25         intro w hw
  26         exact Finset.mem_filter.mpr ⟨Finset.mem_univ w,(Finset.mem_filter.mp hw).2⟩
  27       have hle:(((removed.filter (fun w=>classify w=j)).card:ℝ))≤(fibres j).card:=by
  28         exact_mod_cast Finset.card_le_card hsub
  29       exact hle.trans (le_of_lt hj)
  30     · have he:removed.filter (fun w=>classify w=j)=∅:=by
  31         apply Finset.eq_empty_iff_forall_notMem.mpr
  32         intro w hw
  33         have hw':=Finset.mem_filter.mp hw
  34         have hs:small (classify w):=(Finset.mem_filter.mp hw'.1).2
  35         rw [hw'.2] at hs
  36         exact hj hs
  37       rw [he,Finset.card_empty,Nat.cast_zero]
  38       split <;> positivity
  39   have hpartition:(∑ j:Option (Fin K),((removed.filter (fun w=>classify w=j)).card:ℝ))=removed.card:=by
  40     have h:=Finset.sum_card_fiberwise_eq_card_filter removed (Finset.univ:Finset (Option (Fin K))) classify
  41     have hnat:(∑ j:Option (Fin K),(removed.filter (fun w=>classify w=j)).card)=removed.card:=by simpa using h
  42     exact_mod_cast hnat
  43   have hbound:(∑ j:Option (Fin K),((removed.filter (fun w=>classify w=j)).card:ℝ))≤
  44       (∑ j:Option (Fin K),if j=none then κL*n else κS*n):=
  45     Finset.sum_le_sum (fun j _=>hcount j)
  46   have hright:(∑ j:Option (Fin K),if j=none then κL*n else κS*n)=
  47       (κL+(K:ℝ)*κS)*n:=by simp only [Fintype.sum_option,Option.some_ne_none,ite_false,ite_true,Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul,Nat.cast_id];ring
  48   constructor
  49   · linarith
  50   · intro j hj
  51     apply Finset.card_eq_zero.mpr
  52     apply Finset.eq_empty_iff_forall_notMem.mpr
  53     intro w hw
  54     have hmem:=Finset.mem_inter.mp hw
  55     have hc:classify w=j:=(Finset.mem_filter.mp hmem.1).2
  56     have hs:small (classify w):=(Finset.mem_filter.mp hmem.2).2
  57     rw [hc] at hs
  58     exact hj hs

  60 /-- A class not removed by its size cutoff has the density required by its
  61 preparation proposition, since the remaining host degree is at most its order. -/
  62 theorem factor_retained_class_density {W:Type} [Fintype W] (K:ℕ)
  63     (classify:W→Option (Fin K)) (n D κL κS:ℝ)
  64     (hn:0≤n) (hD:0<D) (hDn:D≤n) (hL:0≤κL) (hS:0≤κS):
  65     let fibres:=fun j:Option (Fin K)=>Finset.univ.filter (fun w=>classify w=j)
  66     let cutoff:=fun j:Option (Fin K)=>if j=none then κL else κS;
  67     ∀ j, cutoff j*n≤(fibres j).card →
  68       2*cutoff j≤2*(fibres j).card/D:=by
  69   intro fibres cutoff j hj
  70   have hk:0≤cutoff j:=by dsimp [cutoff];split <;> assumption
  71   have hh:=mul_nonneg hk (sub_nonneg.mpr hDn)
  72   apply (le_div_iff₀ hD).2
  73   nlinarith
  74 end Oberwolfach
