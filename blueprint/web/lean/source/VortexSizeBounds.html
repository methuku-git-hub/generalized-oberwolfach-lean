  10 theorem floorOrbit_scaled_le {ε : ℝ} (he : 0 ≤ ε) {u n a : ℕ}
  11     (h : a * u ≤ n) (i : ℕ) :
  12     a * floorOrbit ε u i ≤ floorOrbit ε n i := by
  13   induction i with
  14   | zero => exact h
  15   | succ i ih =>
  16     rw [floorOrbit_succ, floorOrbit_succ]
  17     apply Nat.le_floor
  18     push_cast
  19     have hfloor := Nat.floor_le (show 0 ≤ ε * floorOrbit ε u i by positivity)
  20     have hcast : (a : ℝ) * floorOrbit ε u i ≤ floorOrbit ε n i := by exact_mod_cast ih
  21     have h₁ := mul_le_mul_of_nonneg_left hfloor (Nat.cast_nonneg a)
  22     have h₂ := mul_le_mul_of_nonneg_left hcast he
  23     nlinarith

  26     {D : ℕ} (hreg : G.IsRegularOfDegree D) (hn : 0 < Fintype.card V) :
  27     D ≤ Fintype.card V := by
  28   let v : V := Classical.choice (Fintype.card_pos_iff.mp hn)

  41 theorem Vortex.index_size_upper {V U : Type*} [Fintype V] [Fintype U]
  42     {G : SimpleGraph V} {D m : ℕ} {β ε : ℝ} (Q : Vortex U G D β ε m)
  43     (hreg : G.IsRegularOfDegree D) (hn : 0 < Fintype.card V)
  44     (hD : 0 < D) (he : 0 ≤ ε) (hρ : initialIndexDensity U D ≤ 1)
  45     {i : ℕ} (hi : i ≤ Q.length) :
  46     2 * (Q.indices i).card ≤ (Q.vertices i).card := by
  47   have hD' : (0 : ℝ) < D := by exact_mod_cast hD
  48   have hinit : 2 * Fintype.card U ≤ Fintype.card V := by
  49     have hh := (div_le_iff₀ hD').mp hρ
  50     have hd := regular_degree_le_card hreg hn
  51     have hUr : (2 : ℝ) * Fintype.card U ≤ D := by simpa [initialIndexDensity] using hh
  52     have hUn : 2 * Fintype.card U ≤ D := by exact_mod_cast hUr
  53     omega
  54   rw [Q.index_sizes, Q.sizes i hi]
  55   exact floorOrbit_scaled_le he hinit i

  57 theorem Vortex.index_size_lower {V U : Type*} [Fintype V] [Fintype U]
  58     {G : SimpleGraph V} {D m : ℕ} {β ε ρ₀ : ℝ} (Q : Vortex U G D β ε m)
  59     (hreg : G.IsRegularOfDegree D) (hn : 0 < Fintype.card V)
  60     (hD : (Fintype.card V : ℝ) / 2 ≤ D)
  61     (hρ₀ : 0 < ρ₀) (hρ : ρ₀ ≤ initialIndexDensity U D)
  62     (hb : 0 ≤ β) (hbq : β ≤ 1 / 4) {i : ℕ} (hi : i ≤ Q.length) :
  63     ρ₀ / 8 * (Q.vertices i).card ≤ ((Q.indices i).card : ℝ) := by
  64   have hp : 0 < initialIndexDensity U D := hρ₀.trans_le hρ
  65   have hn' : (0 : ℝ) < Fintype.card V := by exact_mod_cast hn
  66   have hd : (1 / 2 : ℝ) ≤ (D : ℝ) / Fintype.card V :=
  67     (le_div_iff₀ hn').mpr (by linarith)
  68   have hc := (abs_le.mp (Q.estimates_including_zero hreg hn hp hb i hi).1).1
  69   have hnc : (0 : ℝ) ≤ (Q.vertices i).card := Nat.cast_nonneg _
  70   have hmul := mul_le_mul_of_nonneg_right hd hnc
  71   have hbmul := mul_le_mul_of_nonneg_right hbq hnc
  72   have hquarter : ((Q.vertices i).card : ℝ) / 4 ≤
  73       2 * (Q.indices i).card / initialIndexDensity U D := by linarith
  74   have hh := (le_div_iff₀ hp).mp hquarter
  75   have hhρ := mul_le_mul_of_nonneg_left hρ (show (0 : ℝ) ≤ (Q.vertices i).card / 4 by positivity)
  76   nlinarith

  78 theorem Vortex.consecutive_size_bounds {V U : Type*} [Fintype V] [Fintype U]
  79     {G : SimpleGraph V} {D m : ℕ} {β ε : ℝ} (Q : Vortex U G D β ε m)
  80     (he : 0 < ε) (hm : 0 < m) {i : ℕ} (hi : i < Q.length) :
  81     ε * (Q.vertices i).card / 2 ≤ ((Q.vertices (i + 1)).card : ℝ) ∧
  82     ((Q.vertices (i + 1)).card : ℝ) ≤ ε * (Q.vertices i).card := by
  83   have hrec : (Q.vertices (i + 1)).card = ⌊ε * (Q.vertices i).card⌋₊ := by
  84     rw [Q.sizes (i + 1) (by omega), floorOrbit_succ, Q.sizes i hi.le]
  85   have hpositive : 0 < (Q.vertices (i + 1)).card := by
  86     have hh := Q.vertices_lower (by omega : i + 1 ≤ Q.length)
  87     have hm' : (0 : ℝ) < m := by exact_mod_cast hm
  88     have hh' : (0 : ℝ) < (Q.vertices (i + 1)).card := lt_of_lt_of_le (by positivity) hh
  89     exact_mod_cast hh'
  90   have hfloor := Nat.lt_floor_add_one (ε * (Q.vertices i).card)
  91   have hge : (1 : ℝ) ≤ (Q.vertices (i + 1)).card := by exact_mod_cast hpositive
  92   constructor
  93   · rw [← hrec] at hfloor
  94     linarith
  95   · rw [hrec]
  96     exact Nat.floor_le (by positivity)
