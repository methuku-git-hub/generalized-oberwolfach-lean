  10 theorem assigned_edge_orientation (G:SimpleGraph V) (D:V→V→Prop)
  11     (hD:∀x y,G.Adj x y→D x y ∨ D y x):
  12     ∃ orient:Sym2 V→V×V,∀e∈G.edgeFinset,
  13       prescribedArcPhysical (orient e)=e ∧ D (orient e).1 (orient e).2 := by
  14   have hex:∀e∈G.edgeFinset,∃a:V×V,prescribedArcPhysical a=e ∧ D a.1 a.2:=by
  15     intro e he
  16     induction e using Sym2.ind with
  17     | _ x y =>
  18       have hxy:G.Adj x y:=SimpleGraph.mem_edgeFinset.mp he
  19       rcases hD x y hxy with h|h
  20       · exact ⟨(x,y),rfl,h⟩
  21       · exact ⟨(y,x),by simp [prescribedArcPhysical,Sym2.eq_swap],h⟩
  22   let v:V:=Classical.choice inferInstance
  23   let orient:Sym2 V→V×V:=fun e=>if he:e∈G.edgeFinset then Classical.choose (hex e he) else (v,v)
  24   refine ⟨orient,?_⟩
  25   intro e he
  26   dsimp only [orient]
  27   rw [dif_pos he]
  28   exact Classical.choose_spec (hex e he)
  29 end Oberwolfach
