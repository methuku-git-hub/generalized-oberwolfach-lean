   8 theorem inner_reserved_graph_bounds
   9     {V:Type} [Fintype V] [DecidableEq V]
  10     (P Kbad:SimpleGraph V) (I:Finset V):
  11     let Jbad:=spanningRestriction P I\Kbad
  12     (∀v,(Jbad.degree v:ℝ)≤I.card) ∧ Disjoint Kbad.edgeFinset Jbad.edgeFinset:=by
  13   intro Jbad
  14   constructor
  15   · intro v
  16     have hsub:Jbad.neighborFinset v⊆I:=by
  17       intro w hw
  18       have hh:=(Jbad.mem_neighborFinset v w).mp hw
  19       have hi: (spanningRestriction P I).Adj v w:=hh.1
  20       exact ((spanningRestriction_adj _ _ _ _).mp hi).2.2
  21     have hh:=Finset.card_le_card hsub
  22     change Jbad.degree v≤I.card at hh
  23     exact_mod_cast hh
  24   · apply Finset.disjoint_left.mpr
  25     intro e he hj
  26     induction e using Sym2.ind with
  27     | _ x y =>
  28       have hk:Kbad.Adj x y:=by simpa only [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using he
  29       have hj':Jbad.Adj x y:=by simpa only [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using hj
  30       exact hj'.2 hk

  32 theorem inner_available_reserved_graph_bounds
  33     {V:Type} [Fintype V] [DecidableEq V]
  34     (P Used Kbad:SimpleGraph V) (I:Finset V):
  35     let Jbad:=spanningRestriction (P\Used) I\Kbad
  36     (∀v,(Jbad.degree v:ℝ)≤I.card) ∧ Disjoint Kbad.edgeFinset Jbad.edgeFinset ∧
  37       Disjoint Used.edgeFinset Jbad.edgeFinset:=by
  38   intro Jbad
  39   obtain ⟨hdegree,hK⟩:=inner_reserved_graph_bounds (P\Used) Kbad I
  40   refine ⟨hdegree,hK,?_⟩

  42   intro e he hj
  43   induction e using Sym2.ind with
