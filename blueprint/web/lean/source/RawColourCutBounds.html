  18     (hq : q ≤ α * ρ ^ 2 / 1024)

  21     (hrow : ∀ x, ρ * Fintype.card V / 4 - t ≤ ((∑ y, M x y : ℕ) : ℝ))

  23     (hcut : ∀ S T, |(matrixCut M S T : ℝ) - ρ / 2 * graphCut G S T| <

  26     (hlarge : (1 - 2 * δ) * Fintype.card V < (S.card : ℝ) + T.card) :
