  11 def RootedCycleFactorPair {V:Type} [Fintype V] [DecidableEq V]
  12     (G:SimpleGraph V) (T:Finset V) (r k:ℕ):Prop:=
  13   ∃ S:Finset V,S.card=k ∧ Disjoint S T ∧
  14     let R:=S∪T
  15     let J:=G.induce (R:Set V)
  16     let S':Finset (R:Set V):=S.subtype (fun v=>v∈R)
  17     ∃ M₁ M₂:Finset (OrdinaryCycle J r),
  18       (∀ C∈M₁,∀ E∈M₁,C≠E→Disjoint (ordinaryCycleVertices J r C) (ordinaryCycleVertices J r E)) ∧
  19       M₁.biUnion (ordinaryCycleVertices J r)=Finset.univ ∧
  20       (∀ C∈M₂,∀ E∈M₂,C≠E→Disjoint (ordinaryCycleVertices J r C) (ordinaryCycleVertices J r E)) ∧
  21       M₂.biUnion (ordinaryCycleVertices J r)=S' ∧
  22       (∀ C∈M₁,∀ E∈M₂,Disjoint C.val.val.edgeFinset E.val.val.edgeFinset) ∧
  23       (∀ C∈M₁,∀ v w,C.val.val.Adj v w→v∈S' ∨ w∈S')
