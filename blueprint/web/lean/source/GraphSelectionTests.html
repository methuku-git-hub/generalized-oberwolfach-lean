  16 def GraphSupportedOn (S : Finset V) (F : SimpleGraph V) : Prop :=
  17   ∀ v w,F.Adj v w → v∈S ∧ w∈S

  19 abbrev SelectionVertexTest (S : Fin N→Finset V) := {iv : Fin N×V // iv.2∈S iv.1}
  20 
  21 def graphSelectionActive (S : Fin N→Finset V) (j : Fin N) (x : SelectionVertexTest S) : Prop :=
  22   j.val < x.val.1.val ∧ x.val.2∈S j
  23 
  24 def graphSelectionHit (S : Fin N→Finset V) (x : SelectionVertexTest S) (F : SimpleGraph V) : Prop :=
  25   ∃ w∈S x.val.1,F.Adj x.val.2 w

  27 theorem graphSelectionHit_bot (S : Fin N→Finset V) (x : SelectionVertexTest S) :
  28     ¬graphSelectionHit S x (⊥ : SimpleGraph V) := by simp [graphSelectionHit]

  30 theorem selectionVertexTest_card (S : Fin N→Finset V) :
  31     Fintype.card (SelectionVertexTest S)=∑ v,(Finset.univ.filter (fun i=>v∈S i)).card := by
  32   simp only [SelectionVertexTest,Fintype.card_subtype,Finset.card_filter,Fintype.sum_prod_type]
  33   exact Finset.sum_comm

  35 theorem selectionVertexTest_card_le (S : Fin N→Finset V) (M : ℝ)
  36     (hM : ∀ v,((Finset.univ.filter (fun i=>v∈S i)).card:ℝ) ≤ M) :
  37     (Fintype.card (SelectionVertexTest S):ℝ) ≤ M*Fintype.card V := by
  38   rw [selectionVertexTest_card,Nat.cast_sum]
  39   have hh := Finset.sum_le_sum (fun v (_ : v∈Finset.univ)=>hM v)

  49 theorem disjoint_graphs_no_common_edge (F G : SimpleGraph V) (h : Disjoint F G)

  54 theorem graph_candidates_hit_bound (S T : Finset V) (v : V) (F : A→SimpleGraph V)
  55     (hsup : ∀ a,GraphSupportedOn S (F a))
  56     (hdisj : ∀ a b,a≠b → Disjoint (F a) (F b)) :
  57     (Finset.univ.filter (fun a=>∃ w∈T,(F a).Adj v w)).card ≤ (S∩T).card := by
  58   let Hit := {a : A // ∃ w∈T,(F a).Adj v w}
  59   let f : Hit→{w : V // w∈S∩T} := fun a=>⟨Classical.choose a.property,
  60     Finset.mem_inter.mpr ⟨(hsup a.val v _ (Classical.choose_spec a.property).2).2,
  61       (Classical.choose_spec a.property).1⟩⟩
  62   have hf : Function.Injective f := by
  63     intro a b he
  64     apply Subtype.ext
  65     by_contra hn
  66     have hv : Classical.choose a.property=Classical.choose b.property := congrArg Subtype.val he
  67     have ha := (Classical.choose_spec a.property).2
  68     have hb := (Classical.choose_spec b.property).2
  69     rw [← hv] at hb
  70     exact disjoint_graphs_no_common_edge (F a.val) (F b.val) (hdisj a.val b.val hn) v _ ha hb
  71   have hh := Fintype.card_le_of_injective f hf
  72   simpa only [Hit,Fintype.card_subtype,Fintype.card_coe,Finset.filter_mem_eq_inter,Finset.univ_inter] using hh
