  11 theorem successorCycleRatio_formula (k : ℕ) :
  12     successorCycleRatio k=1/2+1/(4*(k:ℝ)+6) := by
  13   unfold successorCycleRatio
  14   have hd : (2*((k:ℝ)+2)-1)≠0 := by linarith [Nat.cast_nonneg (α:=ℝ) k]
  15   have hd' : 4*(k:ℝ)+6≠0 := by positivity
  16   field_simp

  31 theorem marked_common_density_bounds (m : ℕ) (hm : 1≤m)
  32     (ρ s ρstar sstar : ℝ) (hρstar : 0<ρstar) (hsstar : 0<sstar)
  33     (hρlo : ρstar≤ρ) (hρhi : ρ≤1) (hslo : sstar≤s) (hshi : s≤1/100) :
  34     let p := ρ*s/2
  35     let k := p/m
  36     let pstar := min (ρstar*sstar/(2*m)) (sstar*sstar/m)
  37     let astar := (sstar/m)*(sstar/m)
  38     0<pstar ∧ 0<astar ∧ pstar≤p ∧ pstar≤k ∧ pstar≤s*s/m ∧ s*s/m≤1 ∧
  39       astar≤(s/m)*(s/m) ∧ astar≤s*s/m ∧ 0<p ∧ 0<k ∧ 2*p+2*k≤1 := by
  40   intro p k pstar astar
  41   have hmR : (1:ℝ)≤m := by exact_mod_cast hm
  42   have hm0 : (0:ℝ)<m := by linarith
  43   have hs0 : 0<s := hsstar.trans_le hslo
  44   have hρ0 : 0<ρ := hρstar.trans_le hρlo
  45   obtain ⟨hbase,hp,hk,hs,hp1,hk1,hs1,ht,hphys⟩ :=
  46     marked_seed_density_bounds m hm ρ s ρstar sstar hρstar hsstar hρlo hρhi hslo hshi
  47   have hpstar : 0<pstar := by dsimp [pstar];positivity
  48   have ha : 0<astar := by dsimp [astar];positivity
  49   have hsprod : sstar*sstar≤s*s := mul_le_mul hslo hslo hsstar.le hs0.le
  50   have hsbar : sstar*sstar/m≤s*s/m := div_le_div_of_nonneg_right hsprod hm0.le
  51   have hsq : (s/m)*(s/m)≤s*s/m := by
  52     have hdiv : s/m≤s := by apply (div_le_iff₀ hm0).mpr;nlinarith
  53     calc
  54       _ ≤ s*(s/m) := mul_le_mul_of_nonneg_right hdiv (by positivity)
  55       _ = _ := by ring
  56   have haraw : astar≤(s/m)*(s/m) := by
  57     dsimp [astar]
  58     have hh := div_le_div_of_nonneg_right hslo hm0.le
  59     exact mul_le_mul hh hh (by positivity) (by positivity)
  60   have hupper : s*s/m≤1 := by
  61     apply (div_le_iff₀ hm0).mpr
  62     nlinarith
  63   exact ⟨hpstar,ha,(min_le_left _ _).trans hp,(min_le_left _ _).trans hk,
  64     (min_le_right _ _).trans hsbar,hupper,haraw,haraw.trans hsq,
  65     hbase.trans_le hp,hbase.trans_le hk,hphys⟩
