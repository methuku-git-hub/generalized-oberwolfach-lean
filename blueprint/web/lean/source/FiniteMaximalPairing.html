  13 theorem finite_maximal_matching_on
  14     {V : Type*} [Fintype V] (G : SimpleGraph V) (D : Finset V) :
  15     ∃ M : G.Subgraph,M.IsMatching ∧ M.verts⊆(D : Set V) ∧
  16       ∀ u∈D,∀ v∈D,u∉M.verts → v∉M.verts → ¬G.Adj u v := by
  17   classical
  18   let P : G.Subgraph → Prop := fun M=>M.IsMatching ∧ M.verts⊆(D : Set V)
  19   have hbot : P ⊥ := by
  20     constructor
  21     · intro v hv
  22       exact False.elim hv
  23     · intro v hv
  24       exact False.elim hv
  25   obtain ⟨M,_,hmax⟩ := Finite.exists_le_maximal hbot
  26   refine ⟨M,hmax.1.1,hmax.1.2,?_⟩
  27   intro u hu v hv hmu hmv huv
  28   let E := G.subgraphOfAdj huv
  29   have hE : E.IsMatching := SimpleGraph.Subgraph.IsMatching.subgraphOfAdj huv
  30   have hdis : Disjoint M.support E.support := by
  31     apply Set.disjoint_left.mpr
  32     intro w hw hwe
  33     have hwM := M.support_subset_verts hw
  34     have hwE := E.support_subset_verts hwe
  35     change w∈({u,v} : Set V) at hwE
  36     rcases hwE with rfl|rfl
  37     · exact hmu hwM
  38     · exact hmv hwM
  39   have hnew : P (M⊔E) := by
  40     refine ⟨hmax.1.1.sup hE hdis,?_⟩
  41     intro w hw
  42     change w∈M.verts∪({u,v} : Set V) at hw
  43     rcases hw with hw|hw
  44     · exact hmax.1.2 hw
  45     · rcases hw with rfl|rfl
  46       · exact hu
  47       · exact hv
  48   have hle : M⊔E≤M := hmax.2 hnew le_sup_left
  49   have huE : u∈E.verts := by change u∈({u,v} : Set V);simp
  50   have huNew : u∈(M⊔E).verts := Or.inr huE
  51   exact hmu (hle.1 huNew)

  53 theorem finite_maximal_pairing
  54     {V : Type*} [Fintype V] (G : SimpleGraph V) (D : Finset V) :
  55     ∃ m : V → Option V,
  56       (∀ u v,m u=some v → u∈D ∧ v∈D ∧ G.Adj u v ∧ m v=some u) ∧
  57       (∀ u∈D,∀ v∈D,m u=none → m v=none → ¬G.Adj u v) := by
  58   classical
  59   obtain ⟨M,hM,hD,hind⟩ := finite_maximal_matching_on G D
  60   let m : V → Option V := fun v=>if hv : v∈M.verts then some ((hM hv).choose) else none
  61   have hnone (v : V) (hv : m v=none) : v∉M.verts := by
  62     intro hvm
  63     simp [m,hvm] at hv
  64   refine ⟨m,?_,fun u hu v hv hmu hmv=>hind u hu v hv (hnone u hmu) (hnone v hmv)⟩
  65   intro u v huv
  66   by_cases hu : u∈M.verts
  67   · have hvchoice : (hM hu).choose=v := by simpa [m,hu] using huv
  68     have hadj : M.Adj u v := hvchoice ▸ (hM hu).choose_spec.1
  69     have hv : v∈M.verts := hadj.snd_mem
  70     refine ⟨hD hu,hD hv,M.adj_sub hadj,?_⟩
  71     have huchoice : (hM hv).choose=u :=
  72       hM.eq_of_adj_left (hM hv).choose_spec.1 hadj.symm
  73     simp [m,hv,huchoice]
  74   · simp [m,hu] at huv

  76 end Oberwolfach
