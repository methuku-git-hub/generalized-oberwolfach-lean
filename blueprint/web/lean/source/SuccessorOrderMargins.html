  11 theorem eventually_reciprocal_small (t : ℝ) (ht : 0 < t) :
  12     ∃ N : ℕ, ∀ n : ℕ, N ≤ n → 0 < n ∧ 1 / (n : ℝ) ≤ t := by
  13   refine ⟨max 1 ⌈1/t⌉₊,?_⟩
  14   intro n hn
  15   have hnpos : 0 < n := by omega
  16   have hnr : (0 : ℝ) < n := by exact_mod_cast hnpos
  17   have hceil : (⌈1/t⌉₊ : ℝ) ≤ n := by exact_mod_cast ((le_max_right _ _).trans hn)
  18   have htN : 1/t ≤ (n : ℝ) := (Nat.le_ceil _).trans hceil
  19   refine ⟨hnpos,?_⟩
  20   apply (div_le_iff₀ hnr).mpr
  21   have hh := (div_le_iff₀ ht).mp htN
  22   nlinarith

  31 theorem successor_order_margins (c m₀ : ℕ) (hc : 0 < c)
  32     (η ε ξ : ℝ) (hη : 0 < η) (hε : 0 < ε) (hξ : 0 < ξ) :
  33     ∃ N : ℕ, ∀ n : ℕ, N ≤ n →
  34       2 ≤ n ∧ 0 < n/c ∧ m₀ ≤ n/c ∧ n ≤ 2*c*(n/c) ∧
  35       1 / ((n/c : ℕ) : ℝ) ≤ ξ ∧ 2 / (n : ℝ) ≤ ξ ∧
  36       partitionFailure c η n ≤ ε * successorPairProbability (n/c : ℕ) n ∧
  37       partitionFailure c η n ≤ (1/2) * (((n/c : ℕ) : ℝ) / n)^c := by
  38   have hcR : (0 : ℝ) < c := by exact_mod_cast hc
  39   have hc1 : (1 : ℝ) ≤ c := by exact_mod_cast hc
  40   let t := min (ξ/(2*c)) (min (ε*(1/(2*c))^2) ((1/2)*(1/(2*c))^c))
  41   have ht : 0 < t := by dsimp [t]; positivity
  42   obtain ⟨N₁,hN₁⟩ := partition_failure_eventually_small c hc η hη
  43   obtain ⟨N₂,hN₂⟩ := eventually_reciprocal_small t ht
  44   refine ⟨max N₁ (max N₂ (max 2 (c * max m₀ 1))),?_⟩
  45   intro n hn
  46   have hn₁ : N₁ ≤ n := (le_max_left _ _).trans hn
  47   have hn₂ : N₂ ≤ n := (le_max_left _ _).trans ((le_max_right _ _).trans hn)
  48   have hlarge : max 2 (c * max m₀ 1) ≤ n := (le_max_right _ _).trans ((le_max_right _ _).trans hn)
  49   have hn2 : 2 ≤ n := (le_max_left _ _).trans hlarge
  50   have hnr : (1 : ℝ) < n := by exact_mod_cast (show 1 < n by omega)
  51   have hn0 : (0 : ℝ) < n := by linarith
  52   obtain ⟨hmpos,hscale,hfail⟩ := hN₁ n hn₁
  53   have hmR : (0 : ℝ) < (n/c : ℕ) := by exact_mod_cast hmpos
  54   have hscaleR : (n : ℝ) ≤ 2*c*(n/c : ℕ) := by exact_mod_cast hscale
  55   have hrecip := (hN₂ n hn₂).2
  56   have hm₀ : m₀ ≤ n/c := (Nat.le_div_iff_mul_le hc).mpr (by
  57     have hh := (le_max_right _ _).trans hlarge
  58     nlinarith [le_max_left m₀ 1])
  59   have hξN : (2*c : ℝ) / n ≤ ξ := by
  60     have hh := hrecip.trans (min_le_left _ _)
  61     have hhh := mul_le_mul_of_nonneg_left hh (show (0 : ℝ) ≤ 2*c by positivity)
  62     have he : (2*c : ℝ) * (ξ/(2*c)) = ξ := by field_simp
  63     rw [he] at hhh
  64     simpa only [mul_one_div] using hhh
  65   have hmξ : (1 : ℝ) / (n/c : ℕ) ≤ ξ := by
  66     apply le_trans _ hξN
  67     apply (le_div_iff₀ hn0).mpr
  68     have hh : (n : ℝ) / (n/c : ℕ) ≤ 2*c := (div_le_iff₀ hmR).mpr hscaleR
  69     convert hh using 1 <;> ring
  70   have hnξ : (2 : ℝ) / n ≤ ξ :=
  71     (div_le_div_of_nonneg_right (by linarith : (2 : ℝ) ≤ 2*c) hn0.le).trans hξN
  72   have hratio : (1 : ℝ)/(2*c) ≤ ((n/c : ℕ) : ℝ)/n := by
  73     apply (le_div_iff₀ hn0).mpr
  74     have hh : (n : ℝ)/(2*c) ≤ (n/c : ℕ) := (div_le_iff₀ (by positivity)).mpr (by nlinarith)
  75     convert hh using 1 <;> ring
  76   have hp : ((1 : ℝ)/(2*c))^2 ≤ successorPairProbability (n/c : ℕ) n :=
  77     (pow_le_pow_left₀ (by positivity) hratio 2).trans (successor_pair_probability_lower hmR.le hnr)
  78   have htuple := pow_le_pow_left₀ (by positivity : (0 : ℝ) ≤ 1/(2*c)) hratio c
  79   refine ⟨hn2,hmpos,hm₀,hscale,hmξ,hnξ,?_,?_⟩
  80   · exact (hfail.trans (hrecip.trans ((min_le_right _ _).trans (min_le_left _ _)))).trans
  81       (mul_le_mul_of_nonneg_left hp hε.le)
  82   · exact (hfail.trans (hrecip.trans ((min_le_right _ _).trans (min_le_right _ _)))).trans
  83       (mul_le_mul_of_nonneg_left htuple (by norm_num))
