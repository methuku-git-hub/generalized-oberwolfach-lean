  28 theorem successor_cyclic_numeric_inputs (k : ℕ) (n m D β ε η : ℝ)
  29     (hn : 0 < n) (hm : 0 < m) (hβ : 0 < β)
  30     (hD : (successorCycleRatio k + β - ε)*n ≤ D)
  31     (hDhalf : n/2 ≤ D)
  32     (hη : ε+η ≤ β/4)
  33     (hsmall : Real.sqrt η ≤ 1/4)
  34     (hround : ε+Real.sqrt η+1/m ≤ β/2) :
  35     (1/2+β/4)*m ≤ D*(m/n)-η*m ∧
  36     0 ≤ D*(m/n)-Real.sqrt η*m ∧
  37     ((k+2)+(2*(k+2)-1)*(β/2))*m ≤
  38       (2*(k+2)-1)*(⌊D*(m/n)-Real.sqrt η*m⌋₊ : ℝ) := by
  39   have hden : (0 : ℝ) < 2*(k+2)-1 := by linarith [Nat.cast_nonneg (α := ℝ) k]
  40   have hd : (successorCycleRatio k+β-ε)*m ≤ D*(m/n) := by
  41     have hh := mul_le_mul_of_nonneg_right hD (div_nonneg hm.le hn.le)
  42     convert hh using 1 <;> field_simp
  43   have hhalf : m/2 ≤ D*(m/n) := by
  44     have hh := mul_le_mul_of_nonneg_right hDhalf (div_nonneg hm.le hn.le)
  45     convert hh using 1 <;> field_simp
  46   have hratio := successorCycleRatio_half_le k
  47   have hratioM := mul_le_mul_of_nonneg_right hratio hm.le
  48   have hηM := mul_le_mul_of_nonneg_right hη hm.le
  49   have hsM := mul_le_mul_of_nonneg_right hsmall hm.le
  50   refine ⟨by nlinarith,by nlinarith,?_⟩
  51   have hq := successor_regularized_degree_lower hn hm hD hround
  52   have ht := mul_le_mul_of_nonneg_left hq hden.le
  53   have he : (2*(k+2)-1) * successorCycleRatio k = (k+2 : ℝ) := by
  54     unfold successorCycleRatio
  55     field_simp
  56   have heq : ((k+2)+(2*(k+2)-1)*(β/2))*m =
  57       (2*(k+2)-1)*((successorCycleRatio k+β/2)*m) := by
  58     calc
  59       _ = ((2*(k+2)-1)*successorCycleRatio k + (2*(k+2)-1)*(β/2))*m := by rw [he]
  60       _ = _ := by ring

  62   simpa only [mul_div_assoc] using ht
