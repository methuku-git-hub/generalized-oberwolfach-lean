  14 theorem endpoint_parent_weight_family (P:SimpleGraph V) (A X:Finset V) (hn:32≤A.card)
  15     (hdeg:∀ x∈X,∀ v∈insert x A,(3/4:ℝ)*A.card≤((A∩P.neighborFinset v).card:ℝ)):
  16     ∃ weight:V→Sym2 V→ℝ,∀ x∈X,
  17       (∀ v∈A∩P.neighborFinset x,∀ w∈A∩P.neighborFinset x,
  18         0≤weight x s(v,w)∧weight x s(v,w)≤40/A.card) ∧
  19       (∀ v∈A∩P.neighborFinset x,∑ w∈A∩P.neighborFinset x,weight x s(v,w)=1) ∧
  20       (∀ v w,¬P.Adj v w→weight x s(v,w)=0) ∧
  21       (∀ F:SimpleGraph V,∀ e∈(endpointEligibleGraph P F (A∩P.neighborFinset x) x).edgeSet,
  22         (1/30:ℝ)/A.card≤weight x e∧weight x e≤40/A.card):=by
  23   have hex:=fun x:↥X=>endpoint_parent_matching_exists P A x.val hn (hdeg x.val x.property)
  24   choose w hw using hex
  25   let weight:V→Sym2 V→ℝ:=fun x=>if hx:x∈X then w ⟨x,hx⟩ else fun _=>0
  26   refine ⟨weight,?_⟩
  27   intro x hx
  28   simpa only [weight,dif_pos hx] using hw ⟨x,hx⟩
