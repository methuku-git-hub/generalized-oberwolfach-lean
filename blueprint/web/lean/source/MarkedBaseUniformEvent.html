  17 theorem markedBase_uniform_event
  18     (m b : ℕ) (hb : 1≤b) (hm : 2≤m) (α θ pstar : ℝ)
  19     (hα : 0<α) (hα1 : α≤1) (hθ : 0<θ) (hpstar : 0<pstar) (K q₀ : ℕ) :
  20     ∃ N : ℕ,∀ (V J : Type) [Fintype V] [LinearOrder V] [Fintype J],
  21       N≤Fintype.card V → Fintype.card J≤(Fintype.card V)^K →
  22     ∀ (P : SimpleGraph V) (π : Equiv.Perm V) (T₀ TK : J→Finset V),
  23       (∀ a,θ*Fintype.card V≤((T₀ a).card:ℝ)) →
  24       (∀ a,θ*Fintype.card V≤((TK a).card:ℝ)) →
  25       (∀ a x,(1/2+α/2)*(T₀ a).card≤((P.neighborFinset x∩T₀ a).card:ℝ)) →
  26       (∀ a x,(1/2+α/2)*(TK a).card≤((P.neighborFinset x∩TK a).card:ℝ)) →
  27     ∀ (p k : ℝ) (hp : 0≤p) (hk : 0≤k) (hsum : 2*p+2*k≤1),pstar≤p → pstar≤k →
  28     1-1/(Fintype.card V:ℝ)^q₀≤
  29       (FiniteLaw.pi (fun _ : Sym2 V=>markedPhysicalLaw p k hp hk hsum)).probability
  30         (fun σ=>∀ a : J,
  31           (pstar^(m+1)*(α*θ/4)^(m+1)/2)*(Fintype.card V:ℝ)^(m+1)≤
  32             (Fintype.card {C : DenseMarkedBaseFamily P π b m (T₀ a) (TK a) //
  33               MarkedBaseRims P π b m σ C.val}:ℝ)) := by
  34   let c := (α*θ/4)^(m+1)
  35   let β := pstar^(m+1)*c/2
  36   have hc : 0<c := by dsimp [c];positivity
  37   have hβ : 0<β := by dsimp [β];positivity
  38   obtain ⟨N₁,hN₁⟩ := denseMarkedBase_uniform_density m b (by omega) α θ hα hα1 hθ
  39   obtain ⟨N₂,hN₂⟩ := constant_deviation_failure_eventually_small 1 (2*((m:ℝ)+1)^2)
  40     β (by positivity) hβ K q₀
  41   refine ⟨max 1 (max N₁ N₂),?_⟩
  42   intro V J _ _ _ hnN hJ P π T₀ TK hs₀ hsK hd₀ hdK p k hp hk hsum hpmin hkmin
  43   let n := Fintype.card V
  44   have hn : 0<n := by dsimp [n];omega
  45   have hn0 : (0:ℝ)<n := by exact_mod_cast hn
  46   let μ := FiniteLaw.pi (fun _ : Sym2 V=>markedPhysicalLaw p k hp hk hsum)
  47   let failure := 2*Real.exp (-2*β^2/(2*((m:ℝ)+1)^2/n))
  48   have hmany (a : J) : c*(n:ℝ)^(m+1)≤Fintype.card (DenseMarkedBaseFamily P π b m (T₀ a) (TK a)) :=
  49     hN₁ V (by omega) P π (T₀ a) (TK a) (hs₀ a) (hsK a) (hd₀ a) (hdK a)
  50   have htail (a : J) : μ.probability (fun σ=>markedBaseSurvivalSum P π b m (T₀ a) (TK a) σ≤β)≤failure := by
  51     have hh := markedBaseSurvivalSum_lower_tail P π b m (T₀ a) (TK a) hn hb hm p k pstar c β hp hk hsum
  52       hpstar.le (by linarith only [hpmin,hsum,hk]) hpmin hkmin hβ (hmany a)
  53     have he : pstar^(m+1)*c-β=β := by dsimp [β];ring
  54     simpa only [he] using hh
  55   have hprob := finiteLaw_common_event_lower μ (Finset.univ : Finset J)
  56     (fun a σ=>markedBaseSurvivalSum P π b m (T₀ a) (TK a) σ≤β) (fun _=>failure)
  57     (fun a _=>htail a)
  58   simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul,Finset.mem_univ,forall_const,not_le] at hprob
  59   have hfail : (Fintype.card J:ℝ)*failure≤1/(n:ℝ)^q₀ :=
  60     hN₂ n (by dsimp [n];omega) (Fintype.card J:ℝ) (Nat.cast_nonneg _) (by
  61       simpa only [one_mul,Nat.cast_pow] using (show (Fintype.card J:ℝ)≤((Fintype.card V)^K:ℕ) by exact_mod_cast hJ))
  62   apply ((sub_le_sub_left hfail 1).trans hprob).trans
  63   apply finiteLaw_probability_mono
  64   intro σ hσ a
  65   have hh := (hσ a).le
  66   rw [markedBaseSurvivalSum_count P π b m (T₀ a) (TK a) hb hm] at hh
  67   exact (le_div_iff₀ (pow_pos hn0 _)).mp hh
