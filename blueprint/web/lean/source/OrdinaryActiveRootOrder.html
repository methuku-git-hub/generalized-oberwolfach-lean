  14 def ordinaryActiveVertexEmbedding (G : SimpleGraph V) (H : V → U → Prop) :
  15     (OrdinaryActiveRow G⊕OrdinaryActiveHub H) ↪ {x : V⊕U // x∈(ordinaryPhysicalGraph G H).support} where
  16   toFun := fun x=>match x with
  17     | .inl v=>⟨.inl v.val,by obtain ⟨w,hw⟩:=v.property;exact ⟨.inl w,hw⟩⟩
  18     | .inr u=>⟨.inr u.val,by obtain ⟨v,hv⟩:=u.property;exact ⟨.inl v,hv⟩⟩
  19   inj' := by
  20     intro x y he
  21     have he' := congrArg Subtype.val he
  22     cases x with
  23     | inl x=>
  24       cases y with
  25       | inl y=>exact congrArg Sum.inl (Subtype.ext (Sum.inl.inj he'))
  26       | inr y=>exact (Sum.inl_ne_inr he').elim
  27     | inr x=>
  28       cases y with
  29       | inl y=>exact (Sum.inr_ne_inl he').elim
  30       | inr y=>exact congrArg Sum.inr (Subtype.ext (Sum.inr.inj he'))

  32 theorem ordinary_active_root_order (G : SimpleGraph V) (H : V → U → Prop) :
  33     Fintype.card (OrdinaryActiveRow G)+Fintype.card (OrdinaryActiveHub H)≤
  34       (ordinaryPhysicalGraph G H).support.toFinset.card := by
  35   have hh := Fintype.card_le_of_injective _ (ordinaryActiveVertexEmbedding G H).injective
  36   simpa only [Fintype.card_sum,Set.toFinset_card] using hh
