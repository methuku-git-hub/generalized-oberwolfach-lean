  10 theorem cut_cycle_into_retained_paths (H:SimpleGraph V) (R:Finset V)
  11     (hisol:∀x,x∉R→∀y,¬H.Adj x y) (D:V→V→Prop)
  12     (a:V) (xs:List V) (hn:(a::xs).Nodup)
  13     (hchain:List.IsChain D ((a::xs)++[a]))
  14     (e:Sym2 V) (he:e∈prescribedOrderedPathEdges ((a::xs)++[a]))
  15     (hmissing:e∉H.edgeFinset):
  16     ∃ blocks:List (List V),
  17       blocks.Pairwise (fun p q=>Disjoint p.toFinset q.toFinset) ∧
  18       (∀ block∈blocks,block.Nodup ∧ List.IsChain H.Adj block ∧ List.IsChain D block) ∧
  19       (∀x,x∈blocks.flatten ↔ x∈a::xs ∧ x∈R) ∧
  20       blocks.toFinset.biUnion prescribedOrderedPathEdges =
  21         (prescribedOrderedPathEdges ((a::xs)++[a])).filter (fun f=>f∈H.edgeFinset) := by
  22   obtain ⟨v,u,ys,hn',hvertices,hchain',hphys,hdir,hlast,hnot⟩:=
  23     cycle_rotate_deleted_edge_to_closure H D a xs hn e he hmissing
  24   let blocks:=retainedPathBlocks H R (v::ys)
  25   have hbase:=split_path_graph_structure H (v::ys) hn'
  26   have hfilter:List.Sublist blocks (splitPathByGraph H (v::ys)) := List.filter_sublist
  27   have hpair:blocks.Pairwise (fun p q=>Disjoint p.toFinset q.toFinset) := hbase.2.2.sublist hfilter
  28   have hrot:List.IsChain D ((v::ys)++[v]) := hchain'.1 hchain
