  12 theorem ordered_cycle_representation_first_edge_injective {V:Type} [Fintype V] [DecidableEq V]
  13     {r:ℕ} (hr:3≤r) (f g:Fin r↪V)
  14     (heq:(SimpleGraph.cycleGraph r).map f=(SimpleGraph.cycleGraph r).map g)
  15     (h0:f ⟨0,by omega⟩=g ⟨0,by omega⟩)
  16     (h1:f ⟨1,by omega⟩=g ⟨1,by omega⟩):f=g:=by
  17   have hex:∀ i:Fin r,∃ j:Fin r,f j=g i:=by
  18     intro i
  19     have hg:g i∈((SimpleGraph.cycleGraph r).map g).support:=by
  20       rw [cycle_map_support hr g]
  21       exact ⟨i,rfl⟩
  22     rw [←heq,cycle_map_support hr f] at hg
  23     exact hg
  24   let σ:Fin r→Fin r:=fun i=>Classical.choose (hex i)
  25   have hs:∀ i,f (σ i)=g i:=fun i=>Classical.choose_spec (hex i)
  26   have hinj:Function.Injective σ:=by
  27     intro i j hij
  28     apply g.injective
  29     rw [←hs i,←hs j,hij]
  30   have hmap:∀ i j,(SimpleGraph.cycleGraph r).Adj i j→(SimpleGraph.cycleGraph r).Adj (σ i) (σ j):=by
  31     intro i j hij
  32     have hg:(SimpleGraph.cycleGraph r).map g |>.Adj (g i) (g j):=
  33       SimpleGraph.map_adj_apply.mpr hij
  34     rw [←heq,←hs i,←hs j] at hg
  35     exact SimpleGraph.map_adj_apply.mp hg
  36   have hz:σ ⟨0,by omega⟩=⟨0,by omega⟩:=by
  37     apply f.injective
  38     rw [hs,h0]
  39   have ho:σ ⟨1,by omega⟩=⟨1,by omega⟩:=by
  40     apply f.injective
  41     rw [hs,h1]
  42   have hfix:=cycleGraph_injective_hom_fix_zero_one hr σ hinj hmap hz ho
  43   apply Function.Embedding.ext
  44   intro i
  45   rw [←hs i,hfix]
  46 end Oberwolfach
