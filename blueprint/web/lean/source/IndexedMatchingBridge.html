  14 def copiedResourceHypergraph {V E : Type*} [Fintype V] [Fintype E]
  15     (R : E → Finset V) (m : E → ℕ) : HadwigerLean.IndexedHypergraph V (Σ e, Fin (m e)) where
  16   vertices := Finset.univ
  17   edges := Finset.univ
  18   edge := fun e => R e.1
  19   edge_subset := fun _ _ => Finset.subset_univ _

  21 theorem copiedResource_degree {V E : Type*} [Fintype V] [Fintype E] [DecidableEq V]
  22     (R : E → Finset V) (m : E → ℕ) (v : V) :
  23     (copiedResourceHypergraph R m).degree v = ∑ e, if v∈R e then m e else 0 := by
  24   simp only [HadwigerLean.IndexedHypergraph.degree,copiedResourceHypergraph,
  25     Finset.card_filter,Fintype.sum_sigma]
  26   apply Finset.sum_congr rfl
  27   intro e _
  28   by_cases h : v∈R e <;> simp [h]

  40 theorem copiedResource_matching_projection
  41     {V E : Type*} [Fintype V] [Fintype E] [DecidableEq V] [DecidableEq E]
  42     (R : E → Finset V) (m : E → ℕ) (M : Finset (Σ e, Fin (m e)))
  43     (hM : (copiedResourceHypergraph R m).IsMatching M) :
  44     (M.image Sigma.fst : Set E).Pairwise (fun e f => Disjoint (R e) (R f)) ∧
  45     (M.image Sigma.fst).biUnion R=(copiedResourceHypergraph R m).covered M := by
  46   constructor
  47   · intro e he f hf hef
  48     obtain ⟨e',he',rfl⟩ := Finset.mem_image.mp he
  49     obtain ⟨f',hf',rfl⟩ := Finset.mem_image.mp hf
  50     exact hM.2 he' hf' (fun h => hef (congrArg Sigma.fst h))
  51   · change (M.image Sigma.fst).biUnion R=M.biUnion (fun e => R e.1)
  52     ext v
  53     simp only [Finset.mem_biUnion,Finset.mem_image]
  54     constructor
  55     · rintro ⟨e,⟨f,hf,rfl⟩,hv⟩
  56       exact ⟨f,hf,hv⟩
  57     · rintro ⟨f,hf,hv⟩
  58       exact ⟨f.1,⟨f,hf,rfl⟩,hv⟩

  60 theorem sum_floor_scaled_weight_bounds {E : Type*} [Fintype E]
  61     (w : E → ℝ) (hw : ∀ e, 0≤w e) (P : E → Prop) [DecidablePred P] (T : ℕ) :
  62     (T : ℝ)*(∑ e, if P e then w e else 0)-Fintype.card E ≤
  63       ((∑ e, if P e then ⌊(T : ℝ)*w e⌋₊ else 0 : ℕ) : ℝ) ∧
  64     ((∑ e, if P e then ⌊(T : ℝ)*w e⌋₊ else 0 : ℕ) : ℝ) ≤
  65       (T : ℝ)*(∑ e, if P e then w e else 0) := by
  66   have hp (e : E) : 0≤(T : ℝ)*w e := mul_nonneg (Nat.cast_nonneg _) (hw e)
  67   constructor
  68   · have hh (e : E) : (T : ℝ)*(if P e then w e else 0)-1≤
  69         ((if P e then ⌊(T : ℝ)*w e⌋₊ else 0 : ℕ) : ℝ) := by
  70       by_cases h : P e
  71       · simp only [h,ite_true]
  72         have hf := Nat.lt_floor_add_one ((T : ℝ)*w e)
  73         linarith
  74       · simp [h]
  75     have hh' := Finset.sum_le_sum (fun e (_ : e∈Finset.univ) => hh e)
  76     simpa only [Finset.sum_sub_distrib,← Finset.mul_sum,Finset.sum_const,
  77       Finset.card_univ,nsmul_eq_mul,mul_one,← Nat.cast_sum] using hh'
  78   · have hh (e : E) : ((if P e then ⌊(T : ℝ)*w e⌋₊ else 0 : ℕ) : ℝ)≤
  79         (T : ℝ)*(if P e then w e else 0) := by
  80       by_cases h : P e
  81       · simpa only [h,ite_true] using Nat.floor_le (hp e)
  82       · simp [h]
  83     have hh' := Finset.sum_le_sum (fun e (_ : e∈Finset.univ) => hh e)
  84     simpa only [← Finset.mul_sum,← Nat.cast_sum] using hh'
