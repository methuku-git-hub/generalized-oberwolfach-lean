  17 theorem directed_two_step_deletion (R R' : V → V → Prop)
  18     (hsub : ∀ x y,R' x y → R x y) (x y : V) (d : ℝ)
  19     (hout : (∑ z,bipAdj (fun x z=>R x z ∧ ¬R' x z) x z)≤d)
  20     (hin : (∑ z,bipAdj (fun z y=>R z y ∧ ¬R' z y) z y)≤d) :
  21     |((Finset.univ.filter (fun z=>R' x z ∧ R' z y)).card:ℝ)-
  22       (Finset.univ.filter (fun z=>R x z ∧ R z y)).card|≤2*d := by
  23   have he (T : V → V → Prop) : ((Finset.univ.filter (fun z=>T x z ∧ T z y)).card:ℝ)=
  24       ∑ z,bipAdj T x z*bipAdj T z y := by
  25     rw [Finset.card_filter]
  26     push_cast
  27     apply Finset.sum_congr rfl
  28     intro z _
  29     by_cases h1 : T x z <;> by_cases h2 : T z y <;> simp [bipAdj,h1,h2]
  30   rw [he R,he R']
  31   have hp (z : V) := unit_product_deletion
  32     (bipAdj_nonneg R' x z) (bipAdj_subrelation R R' hsub x z) (bipAdj_le_one R x z)
  33     (bipAdj_nonneg R' z y) (bipAdj_subrelation R R' hsub z y) (bipAdj_le_one R z y)
  34   have hn : 0≤(∑ z,bipAdj R x z*bipAdj R z y)-(∑ z,bipAdj R' x z*bipAdj R' z y) := by
  35     rw [← Finset.sum_sub_distrib]
  36     exact Finset.sum_nonneg (fun z _=>(hp z).1)
  37   rw [abs_sub_comm,abs_of_nonneg hn,← Finset.sum_sub_distrib]
  38   calc
  39     _ ≤ ∑ z,((bipAdj R x z-bipAdj R' x z)+(bipAdj R z y-bipAdj R' z y)) :=
  40       Finset.sum_le_sum (fun z _=>(hp z).2)
  41     _ ≤ 2*d := by
  42       simp only [bipAdj_deletion R R' hsub,Finset.sum_add_distrib]
  43       linarith only [hout,hin]

  45 theorem represented_two_step_deletion [Fintype E] (tail head : E → V) (keep : E → Prop)
  46     (x y : V) (d : ℝ)
  47     (hout : (∑ e,if ¬keep e ∧ tail e=x then (1:ℝ) else 0)≤d)
  48     (hin : (∑ e,if ¬keep e ∧ head e=y then (1:ℝ) else 0)≤d) :
  49     |((Finset.univ.filter (fun z=>
  50         (∃ e : {e // keep e},tail e.val=x ∧ head e.val=z) ∧
  51         (∃ e : {e // keep e},tail e.val=z ∧ head e.val=y))).card:ℝ)-
  52       (Finset.univ.filter (fun z=>(∃ e,tail e=x ∧ head e=z) ∧ (∃ e,tail e=z ∧ head e=y))).card|≤2*d := by
  53   have hrow : (∑ z,bipAdj (fun x z=>(∃ e,tail e=x ∧ head e=z) ∧
  54       ¬(∃ e : {e // keep e},tail e.val=x ∧ head e.val=z)) x z)≤d := by
  55     convert (represented_relation_lost_outdegree tail head keep x).trans
  56       (by convert hout using 1 <;> congr!) using 1 <;> congr!
  57   have hcol : (∑ z,bipAdj (fun z y=>(∃ e,tail e=z ∧ head e=y) ∧
  58       ¬(∃ e : {e // keep e},tail e.val=z ∧ head e.val=y)) z y)≤d := by
  59     convert (represented_relation_lost_indegree tail head keep y).trans
  60       (by convert hin using 1 <;> congr!) using 1 <;> congr!
  61   have hh := directed_two_step_deletion
  62     (fun x y=>∃ e,tail e=x ∧ head e=y)
  63     (fun x y=>∃ e : {e // keep e},tail e.val=x ∧ head e.val=y)
  64     (by rintro a b ⟨e,he⟩;exact ⟨e.val,he⟩) x y d hrow hcol
  65   convert hh using 1 <;> congr!
