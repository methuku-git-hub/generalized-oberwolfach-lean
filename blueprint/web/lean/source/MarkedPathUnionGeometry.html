  13 def markedPathUnion (r s:ℕ) (pr:A→Fin r→V) (ps:S→Fin s→V):Finset V:=

  15     (Finset.univ.biUnion (fun e=>Finset.univ.image (ps e)))

  25 theorem markedPathUnion_card (r s:ℕ) (pr:A→Fin r→V) (ps:S→Fin s→V):
  26     (markedPathUnion r s pr ps).card≤Fintype.card A*r+Fintype.card S*s:=by
  27   have bound {J:Type} [Fintype J] (n:ℕ) (p:J→Fin n→V):
  28       (Finset.univ.biUnion (fun j=>Finset.univ.image (p j))).card≤Fintype.card J*n:=by
  29     calc
  30       _≤∑ j:J,(Finset.univ.image (p j)).card:=Finset.card_biUnion_le
  31       _≤∑ _:J,n:=Finset.sum_le_sum (fun j _=>by simpa using Finset.card_image_le (f:=p j) (s:=Finset.univ))
  32       _=Fintype.card J*n:=by simp
  33   exact (Finset.card_union_le _ _).trans (Nat.add_le_add (bound r pr) (bound s ps))

  64     (hr:∀ e f k z,pr e k≠pr f z→¬PermutationClose π b (pr e k) (pr f z))
