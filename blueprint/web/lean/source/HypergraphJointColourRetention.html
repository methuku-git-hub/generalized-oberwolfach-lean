  14 theorem hypergraph_joint_colour_retention
  15     {V E C J : Type*} [Fintype E] [Fintype C] [Fintype J]
  16     (R : E → Finset V) (L : E → Finset C) (hL : ∀ e,(L e).Nonempty)
  17     {l k B s : ℕ} (hl : 1<l) (hcard : ∀ e,(L e).card=l)
  18     (hsize : ∀ e,(R e).card≤k)
  19     (hcodeg : ∀ v w,v≠w → (Finset.univ.filter (fun e=>v∈R e ∧ w∈R e)).card≤B)
  20     (c : C) (hdegree : ∀ v,(hypergraphColourClique R L v c).card≤s) (hsl : s≤l)
  21     (z : J → V) (hz : Function.Injective z)
  22     (hsmall : 2*((Fintype.card J)^2*(k^2*B))≤l) :
  23     let G := hypergraphLineGraph R
  24     let μ := FiniteLaw.pi (fun e=>(FiniteLaw.uniformOn (L e) (hL e)).withCoin
  25       (equalizationCoin G L l (k*s) e)
  26       (equalizationCoin_bounds G L (by omega) (k*s) e))
  27     let r := ((1-1/(l : ℝ))^(k*s))/(l : ℝ)
  28     |μ.probability (fun σ=>∀ j,∃ e∈hypergraphColourClique R L (z j) c,
  29         coinRetainedColour G σ e=some c)-
  30       (∏ j,((hypergraphColourClique R L (z j) c).card : ℝ)*r)|≤
  31       (((Fintype.card J)^2*(2*k^2+k+2)*B : ℕ) : ℝ)/(l : ℝ) := by
  32   classical
  33   dsimp only
  34   let T := fun j=>hypergraphColourClique R L (z j) c
  35   have hinter (i j : J) (hij : i≠j) : (T i∩T j).card≤B :=
  36     hypergraphColourClique_intersection R L hcodeg (z i) (z j) (fun he=>hij (hz he)) c
  37   have hc (j : J) (e : E) (he : e∈T j) : c∈L e := (Finset.mem_filter.mp he).2.2
  38   have hh := overlapping_clique_joint_retention (hypergraphLineGraph R) L hL hl hcard T
  39     (fun j=>hypergraphColourClique_isClique R L (z j) c) hinter
  40     (fun j=>hdegree (z j)) hsl
  41     (fun i j hij e he=>restricted_hypergraph_clique_cross_degree R L z hsize hcodeg c i j hij e he)
  42     c hc (fun e _=>hypergraph_colour_neighbour_degree R L hsize c hdegree e)
  43     (hypergraph_colour_common_neighbours R L hsize hcodeg c) hsmall
  44   dsimp only at hh
