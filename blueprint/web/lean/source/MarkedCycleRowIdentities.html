  23 theorem markedCycleIncidence_outgoing {V E K : Type*} [Fintype E] [Fintype K] [DecidableEq V]
  24     (tail head : E → V) (ktail khead : K → V)
  25     (hinj : Function.Injective (fun e => (tail e,head e)))
  26     (hkinj : Function.Injective (fun e => (ktail e,khead e)))
  27     (m : ℕ) (π : Equiv.Perm V) (b : ℕ)
  28     (C : MarkedCycleTuple tail head ktail khead m π b) (v : V) :
  29     occurrenceSumLinear (Sum.elim tail ktail) (markedCycleIncidence tail head ktail khead m C.val) v=
  30       markedSpokeIndicator C.val false v+markedSpokeIndicator C.val true v := by
  31   change (∑ e : E⊕K,if Sum.elim tail ktail e=v then markedCycleIncidence tail head ktail khead m C.val e else 0)=_
  32   simp only [Fintype.sum_sum_type,markedCycleIncidence,Sum.elim_inl,Sum.elim_inr]
  33   have ho : (∑ e : E,if tail e=v then
  34       (∑ i : Fin m,directedEdgeIndicator tail head (C.val i.castSucc) (C.val i.succ) e) else 0)=
  35       ∑ i : Fin m,occurrenceSumLinear tail (directedEdgeIndicator tail head (C.val i.castSucc) (C.val i.succ)) v := by
  36     calc
  37       _ = ∑ e : E,∑ i : Fin m,if tail e=v then
  38           directedEdgeIndicator tail head (C.val i.castSucc) (C.val i.succ) e else 0 := by
  39         apply Finset.sum_congr rfl
  40         intro e _
  41         split_ifs <;> simp
  42       _ = _ := Finset.sum_comm
  43   rw [ho]
  44   change (∑ i : Fin m,occurrenceSumLinear tail (directedEdgeIndicator tail head (C.val i.castSucc) (C.val i.succ)) v)+
  45     occurrenceSumLinear ktail (directedEdgeIndicator ktail khead (C.val (Fin.last m)) (C.val 0)) v=_

  47     directedEdgeIndicator_tail ktail khead hkinj _ _ C.property.2.2]
  48   have hfirst := Fin.sum_univ_castSucc (fun i : Fin (m+1) => if C.val i=v then (1:ℝ) else 0)

  53 theorem marked_cycle_zero_on_close_arc {V E K : Type*} [DecidableEq V]
  54     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (hm : 1≤m)
  55     (π : Equiv.Perm V) (b : ℕ)
  56     (C : MarkedCycleTuple tail head ktail khead m π b) (e : E⊕K)
  57     (hc : PermutationClose π b (Sum.elim tail ktail e) (Sum.elim head khead e)) :
  58     markedCycleIncidence tail head ktail khead m C.val e=0 := by
  59   cases e with
  60   | inl e =>
  61     simp only [markedCycleIncidence,Sum.elim_inl]
  62     apply Finset.sum_eq_zero
  63     intro i _
  64     unfold directedEdgeIndicator
  65     apply if_neg
  66     intro hi
  67     have hne : i.castSucc≠i.succ := by
  68       intro he;have := congrArg Fin.val he;simp only [Fin.val_castSucc,Fin.val_succ] at this;omega
  69     exact C.property.1 i.castSucc i.succ hne (by simpa only [Sum.elim_inl,hi.1,hi.2] using hc)
  70   | inr e =>
  71     simp only [markedCycleIncidence,Sum.elim_inr,directedEdgeIndicator]
  72     apply if_neg
  73     intro hi
  74     have hne : (Fin.last m : Fin (m+1))≠0 := by
  75       intro he;have := congrArg Fin.val he;simp at this;omega
  76     exact C.property.1 (Fin.last m) 0 hne (by simpa only [Sum.elim_inr,hi.1,hi.2] using hc)
