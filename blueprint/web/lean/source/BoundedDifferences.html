  40 theorem finiteLaw_expectation_difference
  41     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X Y : Ω → ℝ) {c : ℝ}
  42     (h : ∀ ω, |X ω - Y ω| ≤ c) :
  43     |μ.expectation X - μ.expectation Y| ≤ c := by
  44   have heq : μ.expectation X - μ.expectation Y = μ.expectation (fun ω => X ω - Y ω) := by
  45     simp only [FiniteLaw.expectation, mul_sub, Finset.sum_sub_distrib]
  46   rw [heq, abs_le]
  47   constructor
  48   · have hh := finiteLaw_expectation_mono μ (fun _ => -c) (fun ω => X ω - Y ω)
  49       (fun ω => (abs_le.mp (h ω)).1)
  50     simpa only [finiteLaw_expectation_const] using hh
  51   · have hh := finiteLaw_expectation_mono μ (fun ω => X ω - Y ω) (fun _ => c)
  52       (fun ω => (abs_le.mp (h ω)).2)
  53     simpa only [finiteLaw_expectation_const] using hh

  55 theorem finiteLaw_mgf_of_diameter
  56     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X : Ω → ℝ) {c : ℝ}
  57     (hc : 0 ≤ c) (hdiam : ∀ x y, |X x - X y| ≤ c) (t : ℝ) :
  58     μ.expectation (fun ω => Real.exp (t * (X ω - μ.expectation X))) ≤
  59       Real.exp (t ^ 2 * c ^ 2 / 8) := by
  60   classical
  61   let : MeasurableSpace Ω := ⊤
  62   have : Nonempty Ω := finiteLaw_nonempty μ
  63   obtain ⟨x, _, hmin⟩ := Finset.univ.exists_min_image X Finset.univ_nonempty
  64   have hbound : ∀ ω, X ω ∈ Set.Icc (X x) (X x + c) := by
  65     intro ω
  66     refine ⟨hmin ω (Finset.mem_univ ω), ?_⟩
  67     have hh := (abs_le.mp (hdiam ω x)).2
  68     linarith
  69   have hsub := hasSubgaussianMGF_of_mem_Icc
  70     (μ := μ.toPMF.toMeasure) (measurable_of_finite X).aemeasurable
  71     (Filter.Eventually.of_forall hbound)
  72   have hh := hsub.mgf_le t
  73   simp only [mgf, finiteLaw_integral] at hh
  74   have he : (((‖X x + c - X x‖₊ / 2) ^ 2 : ℝ≥0) : ℝ) * t ^ 2 / 2 =
  75       t ^ 2 * c ^ 2 / 8 := by
  76     simp only [NNReal.coe_pow, NNReal.coe_div, NNReal.coe_ofNat,
  77       coe_nnnorm, Real.norm_eq_abs]
  78     rw [show X x + c - X x = c by ring, abs_of_nonneg hc]
  79     ring
  80   simpa only [he] using hh

 104 of the supplied categorical laws. Each coordinate has its own state type

 106 theorem finiteLaw_boundedDifferences_mgf
 107     {n : ℕ} {State : Fin n → Type*} [∀ i, Fintype (State i)]
 108     (μ : ∀ i, FiniteLaw (State i)) (X : (∀ i, State i) → ℝ)
 109     (c : Fin n → ℝ) (hc : ∀ i, 0 ≤ c i)
 110     (hchange : ∀ (i : Fin n) (σ τ : ∀ j, State j),
 111       (∀ j, j ≠ i → σ j = τ j) → |X σ - X τ| ≤ c i)
 112     (t : ℝ) :
 113     (FiniteLaw.pi μ).expectation (fun σ =>
 114       Real.exp (t * (X σ - (FiniteLaw.pi μ).expectation X))) ≤
 115       Real.exp (t ^ 2 * (∑ i, (c i) ^ 2) / 8) := by
 116   classical
 117   induction n with
 118   | zero =>
 119     let σ₀ : ∀ i : Fin 0, State i := fun i => Fin.elim0 i
 120     have hX : X = fun _ => X σ₀ := by
 121       funext σ
 122       exact congrArg X (Subsingleton.elim σ σ₀)
 123     rw [hX, finiteLaw_expectation_const]
 124     simp [finiteLaw_expectation_const]
 125   | succ n ih =>
 126     let ν := FiniteLaw.pi (fun i : Fin n => μ i.succ)
 127     let g (x : State 0) := ν.expectation (fun τ => X (Fin.cons x τ))
 128     have hmean : (FiniteLaw.pi μ).expectation X = (μ 0).expectation g :=
 129       finiteLaw_pi_expectation_succ μ X
 130     have htail (x : State 0) :
 131         ν.expectation (fun τ => Real.exp (t * (X (Fin.cons x τ) - g x))) ≤
 132           Real.exp (t ^ 2 * (∑ i : Fin n, (c i.succ) ^ 2) / 8) := by
 133       apply ih (fun i => μ i.succ) (fun τ => X (Fin.cons x τ))
 134         (fun i => c i.succ) (fun i => hc i.succ)
 135       intro i σ τ hagree
 136       apply hchange i.succ (Fin.cons x σ) (Fin.cons x τ)
 137       intro j hj
 138       cases j using Fin.cases with
 139       | zero => rfl
 140       | succ j =>
 141         simp only [Fin.cons_succ]
 142         exact hagree j (fun h => hj (congrArg Fin.succ h))
 143     have hdiam (x y : State 0) : |g x - g y| ≤ c 0 := by
 144       apply finiteLaw_expectation_difference ν
 145       intro τ
 146       apply hchange 0 (Fin.cons x τ) (Fin.cons y τ)
 147       intro j hj
 148       cases j using Fin.cases with
 149       | zero => exact (hj rfl).elim
 150       | succ j => rfl
 151     have hhead := finiteLaw_mgf_of_diameter (μ 0) g (hc 0) hdiam t
 152     let A := Real.exp (t ^ 2 * (∑ i : Fin n, (c i.succ) ^ 2) / 8)
 153     have hfactor (x : State 0) :
 154         ν.expectation (fun τ => Real.exp (t * (X (Fin.cons x τ) -
 155           (FiniteLaw.pi μ).expectation X))) =
 156         Real.exp (t * (g x - (μ 0).expectation g)) *
 157           ν.expectation (fun τ => Real.exp (t * (X (Fin.cons x τ) - g x))) := by
 158       rw [← finiteLaw_expectation_const_mul]
 159       apply congrArg ν.expectation
 160       funext τ
 161       rw [← Real.exp_add, hmean]
 162       congr 1
 163       ring
 164     rw [finiteLaw_pi_expectation_succ]
 165     change (μ 0).expectation (fun x =>
 166       ν.expectation (fun τ => Real.exp (t * (X (Fin.cons x τ) -
 167         (FiniteLaw.pi μ).expectation X)))) ≤ _
 168     simp_rw [hfactor]
 169     calc
 170       _ ≤ (μ 0).expectation (fun x => A *
 171           Real.exp (t * (g x - (μ 0).expectation g))) := by
 172         apply finiteLaw_expectation_mono
 173         intro x
 174         have hh := mul_le_mul_of_nonneg_left (htail x)
 175           (Real.exp_pos (t * (g x - (μ 0).expectation g))).le
 176         simpa only [mul_comm] using hh
 177       _ = A * (μ 0).expectation (fun x =>
 178           Real.exp (t * (g x - (μ 0).expectation g))) :=
 179         finiteLaw_expectation_const_mul _ _ _
 180       _ ≤ A * Real.exp (t ^ 2 * (c 0) ^ 2 / 8) :=
 181         mul_le_mul_of_nonneg_left hhead (Real.exp_pos _).le
 182       _ = _ := by
 183         dsimp [A]
 184         rw [← Real.exp_add, Fin.sum_univ_succ]
 185         congr 1
 186         ring

 189     {n : ℕ} {State : Fin n → Type*} [∀ i, Fintype (State i)]
 190     (μ : ∀ i, FiniteLaw (State i)) (X : (∀ i, State i) → ℝ)

 192     (hchange : ∀ (i : Fin n) (σ τ : ∀ j, State j),
 193       (∀ j, j ≠ i → σ j = τ j) → |X σ - X τ| ≤ c i)
 194     {t : ℝ} (ht : 0 ≤ t) :
 195     (FiniteLaw.pi μ).probability (fun σ =>
 196       t ≤ |X σ - (FiniteLaw.pi μ).expectation X|) ≤
 197       2 * Real.exp (-2 * t ^ 2 / (∑ i, (c i) ^ 2)) := by
 198   classical
 199   let : ∀ i, MeasurableSpace (State i) := fun _ => ⊤
 200   let q : ℝ≥0 := ⟨(∑ i, (c i) ^ 2) / 4,

 203   have hsub : HasSubgaussianMGF
 204       (fun σ => X σ - (FiniteLaw.pi μ).expectation X) q
 205       (FiniteLaw.pi μ).toPMF.toMeasure := by
 206     refine ⟨fun _ => Integrable.of_finite, ?_⟩
 207     intro u
 208     have hh := finiteLaw_boundedDifferences_mgf μ X c hc hchange u

 210     convert hh using 1
 211     rw [hq]
 212     congr 1
 213     ring
 214   have hh := subgaussian_absolute_tail hsub ht
 215   rw [finiteLaw_probability_measure] at hh
 216   have he : -t ^ 2 / (2 * (q : ℝ)) = -2 * t ^ 2 / (∑ i, (c i) ^ 2) := by
 217     rw [hq]

 259 /-- Bounded differences on any finite coordinate set. Reindexing transports
 260 the given probability law exactly; it does not replace that law by a more
 261 convenient sample. -/
 262 theorem finiteLaw_boundedDifferences
 263     {E : Type*} [Fintype E] [DecidableEq E]
 264     {State : E → Type*} [∀ e, Fintype (State e)]
 265     (μ : ∀ e, FiniteLaw (State e)) (X : (∀ e, State e) → ℝ)
 266     (c : E → ℝ) (hc : ∀ e, 0 ≤ c e)
 267     (hchange : ∀ (e : E) (σ τ : ∀ j, State j),
 268       (∀ j, j ≠ e → σ j = τ j) → |X σ - X τ| ≤ c e)
 269     {t : ℝ} (ht : 0 ≤ t) :
 270     (FiniteLaw.pi μ).probability (fun σ =>
 271       t ≤ |X σ - (FiniteLaw.pi μ).expectation X|) ≤
 272       2 * Real.exp (-2 * t ^ 2 / (∑ e, (c e) ^ 2)) := by
 273   classical
 274   let e := (Fintype.equivFin E).symm
 275   let P := Equiv.piCongrLeft State e
 276   have hch (i : Fin (Fintype.card E)) (σ τ : ∀ j, State (e j))
 277       (hagree : ∀ j, j ≠ i → σ j = τ j) :
 278       |X (P σ) - X (P τ)| ≤ c (e i) := by
 279     apply hchange (e i)
 280     intro j hj
 281     obtain ⟨k, rfl⟩ := e.surjective j
 282     change (Equiv.piCongrLeft State e σ) (e k) =
 283       (Equiv.piCongrLeft State e τ) (e k)
 284     simp only [Equiv.piCongrLeft_apply_apply]
 285     exact hagree k (fun h => hj (congrArg e h))
 286   have hh := finiteLaw_boundedDifferences_tail (fun i => μ (e i))
 287     (fun σ => X (P σ)) (fun i => c (e i)) (fun i => hc (e i)) hch ht
 288   rw [finiteLaw_pi_reindex_probability e μ, finiteLaw_pi_reindex_expectation e μ]
 289   have hsum : (∑ i : Fin (Fintype.card E), (c (e i)) ^ 2) = ∑ j, (c j) ^ 2 :=
 290     e.sum_comp (fun j => (c j) ^ 2)
 291   rw [hsum] at hh
 292   exact hh

 294 theorem coordinatewise_constant
 295     {E : Type*} [Fintype E] [DecidableEq E] {State : E → Type*}
 296     (X : (∀ e, State e) → ℝ)
 297     (hchange : ∀ (e : E) (σ τ : ∀ j, State j),
 298       (∀ j, j ≠ e → σ j = τ j) → X σ = X τ) :
 299     ∀ σ τ, X σ = X τ := by
 300   classical
 301   have hstep (S : Finset E) : ∀ (σ τ : ∀ j, State j),
 302       (∀ j, j ∉ S → σ j = τ j) → X σ = X τ := by
 303     induction S using Finset.induction_on with
 304     | empty =>
 305       intro σ τ hagree
 306       exact congrArg X (funext (fun j => hagree j (by simp)))
 307     | @insert e S he ih =>
 308       intro σ τ hagree
 309       let σ' := Function.update σ e (τ e)
 310       have hfirst : X σ = X σ' := by
 311         apply hchange e
 312         intro j hj
 313         exact (Function.update_of_ne hj _ _).symm
 314       apply hfirst.trans
 315       apply ih σ' τ
 316       intro j hj
 317       by_cases hje : j = e
 318       · subst j
 319         exact Function.update_self _ _ _
 320       · change Function.update σ e (τ e) j = τ j
 321         rw [Function.update_of_ne hje]
 322         exact hagree j (by simp [hje, hj])
 323   exact fun σ τ => hstep Finset.univ σ τ (fun j hj => (hj (Finset.mem_univ j)).elim)

 338 /-- Concentration for a fixed root-omitted weighted sum. The influence
 339 bound counts the supplied candidate occurrences separately and leaves the
 340 weights and omitted roots unchanged. -/
 341 theorem finiteLaw_omitted_weighted_tail
 342     {E : Type*} [Fintype E] [DecidableEq E]
 343     {State : E → Type*} [∀ e, Fintype (State e)]
 344     {I : Type*} (μ : ∀ e, FiniteLaw (State e))
 345     (C : Finset I) (F : I → Finset E) (R : Finset E)
 346     (s : I → ∀ e, State e) (w : I → ℝ) (b : E → ℝ)
 347     (hw : ∀ i ∈ C, 0 ≤ w i)
 348     (hload : ∀ e, (∑ i ∈ C, if e ∈ F i \ R then w i else 0) ≤ b e)
 349     {t : ℝ} (ht : 0 ≤ t) :
 350     (FiniteLaw.pi μ).probability (fun σ =>
 351       t ≤ |(∑ i ∈ C, w i * patternValue (F i \ R) (s i) σ) -
 352         (∑ i ∈ C, w i * ∏ e ∈ F i \ R, (μ e).mass (s i e))|) ≤
 353       2 * Real.exp (-2 * t ^ 2 / (∑ e, (b e) ^ 2)) := by
 354   classical
 355   let X (σ : ∀ e, State e) := ∑ i ∈ C, w i * patternValue (F i \ R) (s i) σ
 356   have hb (e : E) : 0 ≤ b e :=
 357     (Finset.sum_nonneg (fun i hi => by
 358       split_ifs
 359       · exact hw i hi
 360       · exact le_refl _)).trans (hload e)
 361   have hchange (e : E) (σ τ : ∀ j, State j)
 362       (hagree : ∀ j, j ≠ e → σ j = τ j) : |X σ - X τ| ≤ b e := by
 363     have hh := weighted_product_stability C (fun i => F i \ R) s w {e} τ σ hw
 364       (fun i _ j _ hj => (hagree j (by simpa only [Finset.mem_singleton] using hj)).symm)
 365     have hbase : |X σ - X τ| ≤ ∑ i ∈ C, if e ∈ F i \ R then w i else 0 := by
 366       simpa only [Finset.sum_singleton] using hh
 367     exact hbase.trans (hload e)
 368   have hh := finiteLaw_boundedDifferences μ X b hb hchange ht
 369   have hmean : (FiniteLaw.pi μ).expectation X =
 370       ∑ i ∈ C, w i * ∏ e ∈ F i \ R, (μ e).mass (s i e) :=
 371     finiteLaw_expectation_omitted_root_sum μ C F R s w
 372   simpa only [hmean, X] using hh
