  11 theorem balanced_pair_transfer_identity {I : Type*} [Fintype I] [DecidableEq I]
  12     (z : I → ℝ) (hz : ∑ i,z i=0) (h : I) :
  13     (∑ i,∑ j,z i*((if h=i then (1:ℝ) else 0)-(if h=j then 1 else 0)))/(Fintype.card I:ℝ)=z h := by
  14   have hn : (0:ℝ)<Fintype.card I := by
  15     letI : Nonempty I := ⟨h⟩
  16     exact Nat.cast_pos.mpr Fintype.card_pos
  17   have hi (i : I) :
  18       (∑ j,z i*((if h=i then (1:ℝ) else 0)-(if h=j then 1 else 0)))=
  19       (Fintype.card I:ℝ)*z i*(if h=i then 1 else 0)-z i := by
  20     simp only [mul_sub,Finset.sum_sub_distrib,Finset.sum_const,Finset.card_univ,nsmul_eq_mul]
  21     rw [← Finset.mul_sum]
  22     simp only [Finset.sum_ite_eq,Finset.mem_univ,ite_true,mul_one]
  23     ring
  24   simp only [hi,Finset.sum_sub_distrib,hz,sub_zero]
  25   have he : (∑ i,(Fintype.card I:ℝ)*z i*(if h=i then (1:ℝ) else 0))=
  26       (Fintype.card I:ℝ)*z h := by
  27     simp only [mul_ite,mul_one,mul_zero,Finset.sum_ite_eq,Finset.mem_univ,ite_true]
  28   rw [he]
