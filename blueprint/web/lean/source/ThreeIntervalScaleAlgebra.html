  26 theorem three_path_scale (n a b c d : ℝ) (hn : 0<n)
  27     (k l m : ℕ) (hk : 0<k) (hl : 0<l) (hm : 0<m) :
  28     ((a/n^(k-1))*(b/n^(l-1))*(c/n^(m-1)))/(d*n) =
  29       (a*b*c/d)/n^(k+l+m-2) := by
  30   have he : k+l+m-2=(k-1)+(l-1)+(m-1)+1 := by omega
  31   rw [he]
  32   simp only [pow_add,pow_one]
  33   field_simp [hn.ne']

  35 theorem three_path_scale_bounds
  36     (n u v w z a₀ a₁ b₀ b₁ c₀ c₁ d₀ d₁ : ℝ) (hn : 0<n)
  37     (k l m : ℕ) (hk : 0<k) (hl : 0<l) (hm : 0<m)
  38     (ha₀ : 0<a₀) (hb₀ : 0<b₀) (hc₀ : 0<c₀) (hd₀ : 0<d₀)
  39     (hu : a₀/n^(k-1) ≤ u ∧ u ≤ a₁/n^(k-1))
  40     (hv : b₀/n^(l-1) ≤ v ∧ v ≤ b₁/n^(l-1))
  41     (hw : c₀/n^(m-1) ≤ w ∧ w ≤ c₁/n^(m-1))
  42     (hz : d₀*n ≤ z ∧ z ≤ d₁*n) :
  43     (a₀*b₀*c₀/d₁)/n^(k+l+m-2) ≤ u*v*w/z ∧
  44       u*v*w/z ≤ (a₁*b₁*c₁/d₀)/n^(k+l+m-2) := by
  45   have hu0 : 0<u := (by positivity : 0<a₀/n^(k-1)).trans_le hu.1
  46   have hv0 : 0<v := (by positivity : 0<b₀/n^(l-1)).trans_le hv.1
  47   have hw0 : 0<w := (by positivity : 0<c₀/n^(m-1)).trans_le hw.1
  48   have hz0 : 0<z := (mul_pos hd₀ hn).trans_le hz.1
  49   have huv := mul_le_mul hu.1 hv.1 (by positivity) hu0.le
  50   have huvw := mul_le_mul huv hw.1 (by positivity) (mul_pos hu0 hv0).le
  51   have huv' := mul_le_mul hu.2 hv.2 hv0.le (hu0.le.trans hu.2)
  52   have huvw' := mul_le_mul huv' hw.2 hw0.le
  53     (mul_nonneg (hu0.le.trans hu.2) (hv0.le.trans hv.2))
  54   have hlo := div_le_div₀ (mul_pos (mul_pos hu0 hv0) hw0).le huvw hz0 hz.2
  55   have hup := div_le_div₀
  56     (mul_nonneg (mul_nonneg (hu0.le.trans hu.2) (hv0.le.trans hv.2)) (hw0.le.trans hw.2))
  57     huvw' (mul_pos hd₀ hn) hz.1
  58   rw [three_path_scale n a₀ b₀ c₀ d₁ hn k l m hk hl hm] at hlo
  59   rw [three_path_scale n a₁ b₁ c₁ d₀ hn k l m hk hl hm] at hup
  60   exact ⟨hlo,hup⟩
