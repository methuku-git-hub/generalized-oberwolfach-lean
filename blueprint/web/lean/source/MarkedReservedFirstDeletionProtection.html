  15 theorem marked_first_deletion_preserves_reserved_resources

  18     (hI:∀ x,π x∈I↔x∈I) (hC:∀ e∈C,¬markedInnerResource I J e):
  19     let E:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) false
  20     let K:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) true
  21     let tail:=fun e:E=>e.val.1
  22     let head:=fun e:E=>e.val.2
  23     let ktail:=fun e:K=>e.val.1
  24     let khead:=fun e:K=>e.val.2;
  25     ∀ (A:Finset (E⊕K)) (S:Finset (V×U))
  26       (M:Finset (MarkedWheel tail head ktail khead m π (3*d) (H false) (H true))),
  27     (∀ a∈A,Sum.elim tail ktail a∈I∧Sum.elim head khead a∈I)→
  28     (∀ z∈S,z.1∈I∧z.2∈J)→Disjoint C (M.biUnion markedWheelPhysicalResources)→
  29     let DR:=markedSampleArcSetGraph P π d σ (A∪M.biUnion markedWheelArcs)
  30     let DS:=markedFiniteSpokeGraph (S∪M.biUnion markedWheelSpokePairs)
  31     let σ':=rawMarkedDeleteRims σ DR.edgeFinset
  32     let H':=markedDeletePhysicalSpokes H DS
  33     let E':=MarkedSampleArc P π (3*d) (fun e=>σ' (Sum.inl e)) false
  34     let K':=MarkedSampleArc P π (3*d) (fun e=>σ' (Sum.inl e)) true;
  35     (∀ x y,Sum.inl ({x,y}:Finset V)∈C→¬DR.Adj x y) ∧
  36     (∀ v u,Sum.inr (v,u)∈C→¬DS.Adj (Sum.inl v) (Sum.inr u)) ∧
  37     (∀ e:E,Sum.inl ({e.val.1,e.val.2}:Finset V)∈C→∃ e':E',e'.val=e.val) ∧
  38     (∀ e:K,Sum.inl ({e.val.1,π e.val.2}:Finset V)∈C→∃ e':K',e'.val=e.val) ∧
  39     (∀ b v u,Sum.inr (v,u)∈C→H b v u→H' b v u) :=by
  40   intro E K tail head ktail khead A S M hA hS hM DR DS σ' H' E' K'
  41   have hR:∀ x y,Sum.inl ({x,y}:Finset V)∈C→¬DR.Adj x y:=by
  42     intro x y hc hxy
  43     obtain ⟨a,ha,he⟩:=Finset.mem_image.mp hxy.2
  44     rcases Finset.mem_union.mp ha with ha|ha
  45     · have hi:=hA a ha
  46       have hs:markedPhysicalEdge tail head ktail khead π a⊆I:=by
  47         cases a with
  48         | inl e=>
  49           simpa only [markedPhysicalEdge,Sum.elim_inl,Finset.insert_subset_iff,Finset.singleton_subset_iff] using hi
  50         | inr e=>
  51           simpa only [markedPhysicalEdge,Sum.elim_inr,Finset.insert_subset_iff,Finset.singleton_subset_iff,hI] using hi
  52       apply hC _ hc
  53       change ({x,y}:Finset V)⊆I
  54       rw [←he]
  55       exact hs
  56     · obtain ⟨W,hW,hWa⟩:=Finset.mem_biUnion.mp ha
  57       have hw:Sum.inl (markedPhysicalEdge tail head ktail khead π a)∈M.biUnion markedWheelPhysicalResources:=
  58         Finset.mem_biUnion.mpr ⟨W,hW,markedWheel_arc_resource W a hWa⟩
  59       rw [he] at hw
  60       exact Finset.disjoint_left.mp hM hc hw
  61   have hSp:∀ v u,Sum.inr (v,u)∈C→¬DS.Adj (Sum.inl v) (Sum.inr u):=by
  62     intro v u hc huv
  63     have hm: (v,u)∈S∪M.biUnion markedWheelSpokePairs:=(markedFiniteSpokeGraph_adj _ v u).mp huv
  64     rcases Finset.mem_union.mp hm with hs|hs
  65     · exact hC _ hc (hS (v,u) hs)
  66     · have hw:Sum.inr (v,u)∈M.biUnion markedWheelPhysicalResources:=by
  67         rw [←markedPackingResources_eq_finite tail head ktail khead π M]
  68         simp only [markedFiniteResources,Finset.mem_union,Finset.mem_image,Sum.inr.injEq]
  69         exact Or.inr ⟨(v,u),hs,rfl⟩
  70       exact Finset.disjoint_left.mp hM hc hw
