  32 abbrev RootFirstCommonMarkedPath (tail head : E → V) (ktail khead : K → V)
  33     (m : ℕ) (π : Equiv.Perm V) (b : ℕ) (H₀ Hₖ : V → U → Prop)
  34     (j : Bool) (v v' : V) (u u' : U) :=
  35   {z : Fin m → V //
  36     IsRootFirstMarkedCycle tail head ktail khead m π b
  37       (if j then Fin.last m else 2) (Fin.cases v z) ∧
  38     IsRootFirstMarkedCycle tail head ktail khead m π b
  39       (if j then Fin.last m else 2) (Fin.cases v' z) ∧
  40     ∀ l : Fin m,
  41       (if rootFirstSpokeType m j l.succ then Hₖ (z l) u else H₀ (z l) u) ∧
  42       (if rootFirstSpokeType m j l.succ then Hₖ (z l) u' else H₀ (z l) u')}
