  55 def longCloseRequiredGraph (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ)
  56     (D M:V→V→Prop):SimpleGraph V where
  57   Adj x y:=P.Adj x y ∧ PermutationClose π (3*d) x y ∧
  58     ((D x y ∧ ¬M x y) ∨ (D y x ∧ ¬M y x))
  59   symm:=⟨by
  60     intro x y h
  61     exact ⟨h.1.symm,(permutationClose_symm π (3*d) x y).mp h.2.1,Or.symm h.2.2⟩⟩
  62   loopless:=⟨by intro x h;exact P.loopless.irrefl x h.1⟩
