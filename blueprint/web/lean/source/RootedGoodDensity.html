  13 theorem fixedSubset_rooted_dense_probability {V : Type*} [Fintype V] [DecidableEq V]
  14     (T : Finset V) (m : ℕ) (hm : m≤Fintype.card V) (hTm : T.card+1<m)
  15     [Nonempty (FixedSubset V m)] [Nonempty {S : FixedSubset V m // T⊆S.val}]
  16     (A : V → Finset V) (δ α : ℝ) (hα : 0<α)
  17     (hA : ∀ x,(δ+α)*Fintype.card V≤((A x).card:ℝ))
  18     (hroots : (T.card:ℝ)+1≤α*m/4) :
  19     1-2*m*Real.exp (-α^2*m/8)≤
  20       (FiniteLaw.uniform {S : FixedSubset V m // T⊆S.val}).probability
  21         (fun S=>∀ x∈S.val.val,(δ+α/2)*m<((A x∩S.val.val).card:ℝ)) := by
  22   let Bad : FixedSubset V m → Prop := fun S=>∃ x∈S.val,((A x∩S.val).card:ℝ)≤(δ+α/2)*m
  23   have hh := fixedSubset_dense_neighbourhoods T m hm hTm A δ α hα hA hroots
  24   change (FiniteLaw.uniform (FixedSubset V m)).probability (fun S=>T⊆S.val ∧ Bad S)≤_ at hh
  25   rw [uniform_probability_intersection (fun S : FixedSubset V m=>T⊆S.val) Bad] at hh
  26   have hrootpos : 0<(FiniteLaw.uniform (FixedSubset V m)).probability (fun S=>T⊆S.val) := by
  27     rw [finiteLaw_uniform_probability]
  28     apply div_pos
  29     · have hh := Fintype.card_pos (α:={S : FixedSubset V m // T⊆S.val})
  30       simp only [← Nat.card_eq_fintype_card] at hh ⊢
  31       exact_mod_cast hh
  32     · exact_mod_cast Fintype.card_pos (α:=FixedSubset V m)
  33   have hbad : (FiniteLaw.uniform {S : FixedSubset V m // T⊆S.val}).probability (fun S=>Bad S.val)≤
  34       2*m*Real.exp (-α^2*m/8) := by
  35     have hh' :
  36         (FiniteLaw.uniform (FixedSubset V m)).probability (fun S=>T⊆S.val)*
  37           (FiniteLaw.uniform {S : FixedSubset V m // T⊆S.val}).probability (fun S=>Bad S.val)≤
  38         (FiniteLaw.uniform (FixedSubset V m)).probability (fun S=>T⊆S.val)*
  39           (2*m*Real.exp (-α^2*m/8)) := by
  40       convert hh using 1 <;> ring
  41     nlinarith only [hh',hrootpos]
  42   have hcomp := finiteLaw_probability_compl (FiniteLaw.uniform {S : FixedSubset V m // T⊆S.val}) (fun S=>Bad S.val)
  43   have he : (fun S : {S : FixedSubset V m // T⊆S.val}=>¬Bad S.val)=
  44       (fun S=>∀ x∈S.val.val,(δ+α/2)*m<((A x∩S.val.val).card:ℝ)) := by
  45     funext S
  46     apply propext
  47     simp only [Bad,not_exists,not_and,not_le]
  48   rw [he] at hcomp
  49   linarith only [hbad,hcomp]

  52 theorem fixedSubset_rooted_dense_card {V : Type*} [Fintype V] [DecidableEq V]
  53     (T : Finset V) (m : ℕ) (hm : m≤Fintype.card V) (hTm : T.card+1<m)
  54     (A : V → Finset V) (δ α : ℝ) (hα : 0<α)
  55     (hA : ∀ x,(δ+α)*Fintype.card V≤((A x).card:ℝ))
  56     (hroots : (T.card:ℝ)+1≤α*m/4) :
  57     (1-2*m*Real.exp (-α^2*m/8))*((Fintype.card V-T.card).choose (m-T.card):ℝ)≤
  58       Fintype.card {S : {S : FixedSubset V m // T⊆S.val} //
  59         ∀ x∈S.val.val,(δ+α/2)*m<((A x∩S.val.val).card:ℝ)} := by
  60   letI : Nonempty (FixedSubset V m) := fixedSubset_nonempty hm
  61   letI : Nonempty {S : FixedSubset V m // T⊆S.val} :=
  62     fixedSubset_containing_nonempty T m (by omega) hm
  63   have hh := fixedSubset_rooted_dense_probability T m hm hTm A δ α hα hA hroots
  64   rw [finiteLaw_uniform_probability] at hh
  65   have hpos : (0:ℝ)<Fintype.card {S : FixedSubset V m // T⊆S.val} := by
  66     exact_mod_cast Fintype.card_pos (α:={S : FixedSubset V m // T⊆S.val})
  67   have hc : (Fintype.card {S : FixedSubset V m // T⊆S.val}:ℝ)=
  68       ((Fintype.card V-T.card).choose (m-T.card):ℝ) := by
  69     have hh := fixedSubset_containing_card (r:=m) T (by omega)
  70     simp only [← Nat.card_eq_fintype_card] at hh ⊢
  71     exact_mod_cast hh
  72   have hh' := (le_div_iff₀ hpos).mp hh
  73   rw [hc] at hh'
  74   convert hh' using 1 <;> congr!
