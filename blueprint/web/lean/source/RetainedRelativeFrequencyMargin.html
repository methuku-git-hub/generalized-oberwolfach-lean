  10 theorem retained_target_lower_bound
  11     (n K r:ℕ) (θ f:ℝ) (hK:0<K) (hr:0<r) (hrK:r≤K) (hθ:0≤θ)
  12     (hf:θ/(100*(K:ℝ))≤f):
  13     θ*(n:ℝ)/(100*(K:ℝ)^2)≤f*(n:ℝ)/(r:ℝ) :=by
  14   have hK':0<(K:ℝ):=by exact_mod_cast hK
  15   have hr':0<(r:ℝ):=by exact_mod_cast hr
  16   have hrK':(r:ℝ)≤K:=by exact_mod_cast hrK
  17   have hf0:0≤f:=(by positivity : (0:ℝ)≤θ/(100*(K:ℝ))).trans hf
  18   have h₁:θ/(100*(K:ℝ))/(K:ℝ)≤f/(K:ℝ):=div_le_div_of_nonneg_right hf hK'.le
  19   have h₂:f/(K:ℝ)≤f/(r:ℝ):=div_le_div_of_nonneg_left hf0 hr' hrK'
  20   have h₃:=mul_le_mul_of_nonneg_right (h₁.trans h₂) (show (0:ℝ)≤n by positivity)
  21   have hleft:θ/(100*(K:ℝ))/(K:ℝ)*(n:ℝ)=θ*(n:ℝ)/(100*(K:ℝ)^2):=by field_simp <;> ring
  22   have hright:f/(r:ℝ)*(n:ℝ)=f*(n:ℝ)/(r:ℝ):=by ring
  23   rw [hleft,hright] at h₃
  24   exact h₃
