  14 lemma normalized_mass_error (τ c M S : ℝ) (hc : 0≤c)
  15     (hlo : (1-τ)*M≤S) (hhi : S≤M) (hone : c*M=1) : |c*S-1|≤τ := by
  16   have h1 := mul_le_mul_of_nonneg_left hlo hc
  17   have h2 := mul_le_mul_of_nonneg_left hhi hc
  18   have hτ : 0≤τ := by nlinarith [show c*((1-τ)*M)=(1-τ)*(c*M) by ring]
  19   rw [abs_le]
  20   constructor <;> nlinarith [show c*((1-τ)*M)=(1-τ)*(c*M) by ring]
