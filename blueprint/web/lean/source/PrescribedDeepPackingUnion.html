   5 /-! Exact recombination of the prescribed-inner-edge packing with a deep
   6 packing in its actual remainder. Protected intersections and covered stars
   7 are computed in the original physical graph. -/

  13 attribute [local instance] Classical.propDecidable
  14 variable {V U Z:Type} [Fintype V] [Fintype U] [Fintype Z] [DecidableEq V] [DecidableEq U]
  15   {Q F:SimpleGraph V} {H:V→U→Prop} {r nr nh:ℕ}

  17 theorem prescribed_deep_packing_union (C:Z→OrdinaryWheel Q F H r)
  18     (hC:∀ i j,i≠j→Disjoint (ordinaryWheelResources (C i)) (ordinaryWheelResources (C j)))
  19     (E:Finset (Sym2 V⊕(V×U))) (e:Z→Sym2 V⊕(V×U))
  20     (hinner:∀ i,ordinaryWheelResources (C i)∩E={e i})
  21     (M:Finset (OrdinaryWheel (Q\wheelFamilyRim C) F (fun v u=>H v u∧¬wheelFamilySpokes C v u) r))
  22     (hM:∀ A∈M,∀ B∈M,A≠B→Disjoint (ordinaryWheelResources A) (ordinaryWheelResources B))
  23     (hME:Disjoint (M.biUnion ordinaryWheelResources) E)
  24     (x:Fin nr→V) (u:Fin nh→U)
  25     (hrim:∀ i v,(ordinaryWheelRimGraph M).Adj (x i) v↔(Q\wheelFamilyRim C).Adj (x i) v)
  26     (hrow:∀ i v,ordinaryPackingSpokes M (x i) v↔H (x i) v∧¬wheelFamilySpokes C (x i) v)
  27     (hcol:∀ i v,ordinaryPackingSpokes M v (u i)↔H v (u i)∧¬wheelFamilySpokes C v (u i)):
  28     ∃ P:Finset (OrdinaryWheel Q F H r),
  29       (∀ A∈P,∀ B∈P,A≠B→Disjoint (ordinaryWheelResources A) (ordinaryWheelResources B)) ∧
  30       P.biUnion ordinaryWheelResources=ordinaryHostResources (wheelFamilyRim C) (wheelFamilySpokes C)∪M.biUnion ordinaryWheelResources ∧
  31       P.biUnion ordinaryWheelResources∩E=Finset.univ.image e ∧
  32       ordinaryWheelRimGraph P=wheelFamilyRim C⊔ordinaryWheelRimGraph M ∧
  33       (∀ v a,ordinaryPackingSpokes P v a↔wheelFamilySpokes C v a∨ordinaryPackingSpokes M v a) ∧
  34       (∀ i v,(ordinaryWheelRimGraph P).Adj (x i) v↔Q.Adj (x i) v) ∧
  35       (∀ i v,ordinaryPackingSpokes P (x i) v↔H (x i) v) ∧
  36       (∀ i v,ordinaryPackingSpokes P v (u i)↔H v (u i)):=by
  37   have hi:∀ A∈Finset.univ.image C,∀ B∈Finset.univ.image C,A≠B→Disjoint (ordinaryWheelResources A) (ordinaryWheelResources B):=by
  38     intro A hA B hB hne
  39     obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hA
  40     obtain ⟨j,_,rfl⟩:=Finset.mem_image.mp hB
  41     exact hC i j (fun he=>hne (congrArg C he))
  42   have hg:=ordinary_wheel_packing_glue (Finset.univ.image C)
  43   rw [ordinary_family_image_rim C,ordinary_family_image_spokes C] at hg
  44   obtain ⟨P,hP,hres,hPr,hPs⟩:=hg M hi hM
  45   have hfamily:(Finset.univ.image C).biUnion ordinaryWheelResources=
  46       ordinaryHostResources (wheelFamilyRim C) (wheelFamilySpokes C):=by
  47     rw [wheelFamily_resources,Finset.image_biUnion]
  48   rw [hfamily] at hres
  49   have hsub:=wheelFamily_subhost C
  50   refine ⟨P,hP,hres,?_,hPr,hPs,?_,?_,?_⟩
  51   · rw [hres,Finset.union_inter_distrib_right,Finset.disjoint_iff_inter_eq_empty.mp hME,Finset.union_empty]
  52     exact (prescribed_wheel_family_exact_union C E e hC hinner).2.1
  53   · intro i v
  54     rw [hPr,SimpleGraph.sup_adj,hrim]
  55     have hh: (wheelFamilyRim C).Adj (x i) v→Q.Adj (x i) v:=fun h=>hsub.1 h
  56     change _∨(Q.Adj (x i) v∧¬_)↔Q.Adj (x i) v
  57     tauto
  58   · intro i v
  59     rw [hPs,hrow]
  60     have hh:=hsub.2 (x i) v
  61     tauto
  62   · intro i v
  63     rw [hPs,hcol]
  64     have hh:=hsub.2 v (u i)
  65     tauto

  67 end Oberwolfach
