  15 def DirectedBoundaryEdge (tail head:A→W) (X:Finset W):Type _ :=
  16   {a:A // (tail a∉X∧head a∈X)∨(tail a∈X∧head a∉X)}

  23 def boundaryOutside (tail head:A→W) (X:Finset W) (a:DirectedBoundaryEdge tail head X):W :=
  24   if tail a.val∈X then head a.val else tail a.val

  42 theorem boundary_owner_label_fibre (tail head:A→W) (kind:A→T) (X:Finset W)
  43     (hinj:Function.Injective (fun a=>(kind a,tail a,head a))) (x y:W):
  44     (Finset.univ.filter (fun a:DirectedBoundaryEdge tail head X=>
  45       boundaryOutside tail head X a=x∧boundaryInside tail head X a=y)).card≤2*Fintype.card T := by
  46   let F:=Finset.univ.filter (fun a:DirectedBoundaryEdge tail head X=>
  47       boundaryOutside tail head X a=x∧boundaryInside tail head X a=y)
  48   let f:F→Bool×T:=fun a=>(decide (tail a.val.val∈X),kind a.val.val)
  49   have hf:Function.Injective f := by
  50     intro a b hab
  51     have ht:kind a.val.val=kind b.val.val:=congrArg Prod.snd hab
  52     have hd:decide (tail a.val.val∈X)=decide (tail b.val.val∈X):=congrArg Prod.fst hab
  53     have hm:(tail a.val.val∈X)↔(tail b.val.val∈X):=by
  54       simpa only [decide_eq_decide] using hd
  55     have ha:boundaryOutside tail head X a.val=x∧boundaryInside tail head X a.val=y:=
  56       (Finset.mem_filter.mp a.property).2
  57     have hb:boundaryOutside tail head X b.val=x∧boundaryInside tail head X b.val=y:=
  58       (Finset.mem_filter.mp b.property).2
  59     apply Subtype.ext
  60     apply Subtype.ext
  61     apply hinj
  62     have ea:=boundary_ordered_endpoints tail head X a.val
  63     have eb:=boundary_ordered_endpoints tail head X b.val
  64     rw [ha.1,ha.2] at ea
  65     rw [hb.1,hb.2] at eb
  66     by_cases h:tail a.val.val∈X
  67     · have h':tail b.val.val∈X:=hm.mp h
  68       simp only [h,h',ite_true] at ea eb
  69       exact Prod.ext ht (Prod.ext (ea.1.trans eb.1.symm) (ea.2.trans eb.2.symm))
  70     · have h':tail b.val.val∉X:=fun hb=>h (hm.mpr hb)
  71       simp only [h,h',ite_false] at ea eb
  72       exact Prod.ext ht (Prod.ext (ea.1.trans eb.1.symm) (ea.2.trans eb.2.symm))
  73   have hh:=Fintype.card_le_of_injective f hf
  74   simpa only [Fintype.card_coe,Fintype.card_prod,Fintype.card_bool,F] using hh
