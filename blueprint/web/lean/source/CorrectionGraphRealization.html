  14 theorem graph_of_suborientation {V : Type*} [Fintype V]
  15     (Q : SimpleGraph V) (A J : V → V → ℕ)
  16     (hA : ∀ x y,A x y+A y x=if Q.Adj x y then 1 else 0)
  17     (hJ : ∀ x y,J x y≤A x y) :
  18     ∃ P : SimpleGraph V,P≤Q ∧ ∀ v,P.degree v=(∑ w,J v w)+(∑ w,J w v) := by
  19   classical
  20   have hbound (x y : V) : J x y+J y x≤ if Q.Adj x y then 1 else 0 := by
  21     rw [← hA]
  22     exact Nat.add_le_add (hJ x y) (hJ y x)
  23   let P : SimpleGraph V := {
  24     Adj := fun x y=>0<J x y+J y x
  25     symm := ⟨by intro x y hx;simpa only [Nat.add_comm] using hx⟩
  26     loopless := ⟨by intro x hx;have hh:=hbound x x;simp only [SimpleGraph.irrefl,ite_false] at hh;omega⟩ }
  27   refine ⟨P,?_,?_⟩
  28   · intro x y hxy
  29     have hh := hbound x y
  30     change 0<J x y+J y x at hxy
  31     by_contra h
  32     simp only [h,ite_false] at hh
  33     omega
  34   · intro v
  35     rw [SimpleGraph.degree,SimpleGraph.neighborFinset_eq_filter,Finset.card_filter]
  36     rw [← Finset.sum_add_distrib]
  37     apply Finset.sum_congr rfl
  38     intro w _
  39     have hh : J v w+J w v≤1 := (hbound v w).trans (by split_ifs <;> omega)
  40     split_ifs with hp
  41     · change 0<J v w+J w v at hp
  42       omega
  43     · change ¬0<J v w+J w v at hp
  44       omega

  46 theorem relation_from_submatrix {V U : Type*} [Fintype V] [Fintype U]
  47     (H : V → U → Prop) (K : V → U → ℕ)
  48     (hK : ∀ v u,K v u≤ if H v u then 1 else 0) :
  49     ∃ J : V → U → Prop,(∀ v u,J v u → H v u) ∧
  50       (∀ v,(Finset.univ.filter (J v)).card=∑ u,K v u) ∧
  51       (∀ u,(Finset.univ.filter (fun v=>J v u)).card=∑ v,K v u) := by
  52   classical
  53   let J (v : V) (u : U) := 0<K v u
  54   have hind (v : V) (u : U) : (if J v u then 1 else 0 : ℕ)=K v u := by
  55     have hh : K v u≤1 := (hK v u).trans (by split_ifs <;> omega)
  56     dsimp [J]
  57     split_ifs <;> omega
  58   refine ⟨J,?_,?_,?_⟩
  59   · intro v u hvu
  60     have hh := hK v u
  61     change 0<K v u at hvu
  62     by_contra h
  63     simp only [h,ite_false] at hh
  64     omega
  65   · intro v
  66     rw [Finset.card_filter]
  67     apply Finset.sum_congr rfl
  68     intro u _
  69     split_ifs with hp
  70     · have hh := hind v u
  71       rw [if_pos hp] at hh
  72       exact hh
  73     · have hh := hind v u
  74       rw [if_neg hp] at hh
  75       exact hh
  76   · intro u
  77     rw [Finset.card_filter]
  78     apply Finset.sum_congr rfl
  79     intro v _
  80     split_ifs with hp
  81     · have hh := hind v u
  82       rw [if_pos hp] at hh
  83       exact hh
  84     · have hh := hind v u
  85       rw [if_neg hp] at hh
  86       exact hh

  88 theorem relation_deleted_row_degree {V U : Type*} [Fintype V] [Fintype U]
  89     (H J : V → U → Prop) (hJ : ∀ v u,J v u → H v u) (v : V) :
  90     (Finset.univ.filter (fun u=>H v u ∧ ¬J v u)).card=
  91       (Finset.univ.filter (H v)).card-(Finset.univ.filter (J v)).card := by
  92   classical
  93   have hs : Finset.univ.filter (J v)⊆Finset.univ.filter (H v) := by
  94     intro u hu
  95     simpa only [Finset.mem_filter,Finset.mem_univ,true_and] using hJ v u (by simpa using hu)
  96   have heq : Finset.univ.filter (fun u=>H v u ∧ ¬J v u)=
  97       Finset.univ.filter (H v) \ Finset.univ.filter (J v) := by ext u;simp
  98   rw [heq,Finset.card_sdiff_of_subset hs]
