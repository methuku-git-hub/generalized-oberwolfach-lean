  12 theorem prescribed_source_degree_from_components
  13     (Q:I→SimpleGraph X) (Dsrc:I→X→X→Prop) (Fsrc:I→SimpleGraph X)
  14     (ps:List (I×PrescribedSourcePiece X))
  15     (hpieces:∀ p∈ps,p.2.Valid (Dsrc p.1) (Fsrc p.1))
  16     (hapart:ps.Pairwise PrescribedPiecesApart)
  17     (hDQ:∀ i x y,Dsrc i x y→(Q i).Adj x y)
  18     (hcoverE:∀ i e,e∈(Q i).edgeFinset→∃ p∈ps,p.1=i ∧ e∈p.2.edges)
  19     (hcoverA:∀ i x y,Dsrc i x y→∃ p∈ps,p.1=i ∧ (x,y)∈p.2.arcs):
  20     ∀ i x,(Q i).degree x≤2 := by
  21   have huniq:=prescribed_source_family_unique_directions Dsrc Fsrc ps hpieces hapart hcoverA
  22   intro i
  23   apply prescribed_source_degree_from_directions (Q i) (Dsrc i) ?_ (huniq.1 i) (huniq.2 i)
  24   intro x y
  25   constructor
  26   · intro hxy
  27     have he:s(x,y)∈(Q i).edgeFinset:=SimpleGraph.mem_edgeFinset.mpr hxy
  28     obtain ⟨p,hp,hpi,hpE⟩:=hcoverE i s(x,y) he
  29     rw [prescribed_piece_edges_are_arc_images] at hpE
  30     obtain ⟨e,he,hphys⟩:=Finset.mem_image.mp hpE
  31     rcases e with ⟨u,v⟩
  32     have ha:Dsrc i u v:=by
  33       have hh:=prescribed_piece_arc_source p.2 (Dsrc p.1) (Fsrc p.1) (hpieces p hp) (u,v) he
  34       simpa [hpi] using hh
  35     change s(u,v)=s(x,y) at hphys
  36     rcases Sym2.eq_iff.mp hphys with ⟨hu,hv⟩|⟨hu,hv⟩
  37     · subst u;subst v;exact Or.inl ha
  38     · subst u;subst v;exact Or.inr ha
  39   · intro hxy
  40     rcases hxy with hxy|hyx
  41     · exact hDQ i x y hxy
  42     · exact (hDQ i y x hyx).symm
  43 end Oberwolfach
