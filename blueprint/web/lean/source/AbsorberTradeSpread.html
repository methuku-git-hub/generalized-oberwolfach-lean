  17 theorem labelled_family_trade_spread (hc : 1 ≤ c) (R : C → Finset V)

  20     (A : Finset (LabelledAbsorberFamily R S X))
  21     (a k : ℝ) (ha : 0 < a) (hk : 0 < k) (hn : (Fintype.card V:ℝ)=c*k)

  23     let K := (absorberTradeCountConstant c J (Fintype.card T):ℝ)*(c:ℝ)^(c*c)/a
  24     |labelledFamilyTrade R S X A C₀| ≤ K/k^c+
  25       (if (R C₀∩X).Nonempty then K/k^(c-1) else 0) := by
  26   let N := absorberTradeCountConstant c J (Fintype.card T)
  27   let K := (N:ℝ)*(c:ℝ)^(c*c)/a
  28   have hc0 : (0:ℝ)<c := by exact_mod_cast (show 0<c by omega)
  29   have hc1 : (1:ℝ) ≤ c := by exact_mod_cast hc
  30   have hh : c ≤ c*c := by nlinarith
  31   have hK : 0 ≤ K := by dsimp [K];positivity
  32   have step (s : ℕ) (hs : s ≤ c*c)
  33       (hcount : Fintype.card {D : A // ∃ o,(D.val.2.2 o).val=C₀} ≤ N*Fintype.card V^(c*c-s)) :
  34       |labelledFamilyTrade R S X A C₀| ≤ K/k^s := by
  35     have hz : (Fintype.card {D : A // ∃ o,(D.val.2.2 o).val=C₀}:ℝ) ≤ (N:ℝ)*((c:ℝ)*k)^(c*c-s) := by
  36       rw [← hn]
  37       exact_mod_cast hcount
  38     exact (labelled_family_trade_bound R S X A C₀).trans
  39       (power_count_ratio_bound (c*c) s hs a k c N A.card _ ha hk hc1 (Nat.cast_nonneg _) hA hz)

  42   · rw [if_pos hhit]
  43     have hcount := labelled_absorber_subfamily_occurrence_count R S X C₀
  44       (c^(2*c)*Fintype.card V^(c*c-c+1)*J^(2*c+1))
  45       (fun t o => labelled_absorber_occurrence_general hc R (S t) X hX J hJ o C₀ hC) A
  46     have hc' : Fintype.card {D : A // ∃ o,(D.val.2.2 o).val=C₀} ≤ N*Fintype.card V^(c*c-(c-1)) := by
  47       have he : c*c-(c-1)=c*c-c+1 := by omega
  48       rw [he]

  50     exact (step (c-1) (by omega) hc').trans (le_add_of_nonneg_left (by positivity))
  51   · rw [if_neg hhit,add_zero]

  53       intro v hv hx;exact hhit ⟨v,Finset.mem_inter.mpr ⟨hv,hx⟩⟩)
  54     have hcount := labelled_absorber_subfamily_occurrence_count R S X C₀
