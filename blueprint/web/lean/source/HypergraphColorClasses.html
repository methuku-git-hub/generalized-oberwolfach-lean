  16 def hypergraphColorClass (R : E → Finset V) (σ : V → C) (c : C) : Finset E :=
  17   Finset.univ.filter (fun e=>∀ v∈R e,σ v=c)

  19 theorem hypergraphColorClass_mem (R : E → Finset V) (σ : V → C) (c : C) (e : E) :
  20     e∈hypergraphColorClass R σ c ↔ ∀ v∈R e,σ v=c := by
  21   simp only [hypergraphColorClass,Finset.mem_filter,Finset.mem_univ,true_and]

  23 theorem hypergraphColorClass_disjoint_vertices (R : E → Finset V) (σ : V → C)
  24     (c d : C) (hcd : c≠d) (e f : E)
  25     (he : e∈hypergraphColorClass R σ c) (hf : f∈hypergraphColorClass R σ d) :
  26     Disjoint (R e) (R f) := by
  27   apply Finset.disjoint_left.mpr
  28   intro v hv hw
  29   exact hcd (((hypergraphColorClass_mem R σ c e).mp he v hv).symm.trans
  30     ((hypergraphColorClass_mem R σ d f).mp hf v hw))
