  12 theorem preparation_concentrated_balanced_sum {I:Type} [Fintype I]
  13     (τ a m x t:ℝ) (b z means errors:I→ℝ)
  14     (hbalance:a+∑ i,b i=1) (hm:(1-τ)*a≤m ∧ m≤(1+τ)*a)
  15     (hmeans:∀ i,(1-τ)*b i≤means i ∧ means i≤(1+τ)*b i)
  16     (hx:|x-m|<t) (hz:∀ i,|z i-means i|<errors i):
  17     |x+∑ i,z i-1|<t+(∑ i,errors i)+τ:=by
  18   have hmean:=preparation_balanced_means τ a m b means hbalance hm hmeans
  19   have hmeanerr:|m+∑ i,means i-1|≤τ:=abs_le.mpr ⟨by linarith [hmean.1],by linarith [hmean.2]⟩
  20   have hobs:|x+∑ i,z i-(m+∑ i,means i)|≤|x-m|+∑ i,|z i-means i|:=by
  21     calc
  22       _ = |(x-m)+∑ i,(z i-means i)|:=by rw [Finset.sum_sub_distrib];congr 1;ring
  23       _ ≤ |x-m|+|∑ i,(z i-means i)|:=abs_add_le _ _
  24       _ ≤ _:=by
  25         simpa only [add_comm] using add_le_add_left (Finset.abs_sum_le_sum_abs (fun i=>z i-means i) Finset.univ) |x-m|
  26   have hsum:(∑ i,|z i-means i|)≤∑ i,errors i:=Finset.sum_le_sum (fun i _=>(hz i).le)
  27   calc
  28     _ ≤ |x+∑ i,z i-(m+∑ i,means i)|+|m+∑ i,means i-1|:=abs_sub_le _ _ _
  29     _ ≤ |x-m|+(∑ i,|z i-means i|)+τ:=by linarith [hobs,hmeanerr]
  30     _ < t+(∑ i,errors i)+τ:=by linarith [hx,hsum]
  31 end Oberwolfach
