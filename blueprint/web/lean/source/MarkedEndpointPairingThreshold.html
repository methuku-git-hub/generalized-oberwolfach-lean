  12 theorem marked_endpoint_pairing_threshold
  13     (b M : ℕ) (κ lam ε : ℝ) (hκ : 0<κ) (hlam : 0<lam)
  14     (hε : 0<ε) (hεκ : ε<κ/8) (hepslam : ε<lam/8) :
  15     ∃ s₀ : ℕ, ∀ s : ℕ, s₀≤s →
  16     ∀ (A X Y V : Type) [Fintype A] [Fintype X] [Fintype Y] [Fintype V],
  17     ∀ (EA : A → Y → Prop) (EX : X → Y → Prop)
  18       (PA : A → Prop) (PX : X → Prop) (Q : Y → Prop),
  19       Fintype.card A=s → Fintype.card X=s → Fintype.card Y=s →
  20       Fintype.card {a // PA a}=Fintype.card {y // Q y} →
  21       ((Finset.univ.filter Q).card:ℝ)+(Finset.univ.filter PX).card≤(1-κ)*s →
  22       (∀ x y, EX x y → PX x → ¬Q y) →
  23     ∀ (π : Equiv.Perm V) (labelA : A → V) (labelX : X → V),
  24       (∀ v, Fintype.card {a // labelA a=v}≤M) →
  25       (∀ v, Fintype.card {x // labelX x=v}≤M) →
  26       BipartiteBlockBounds EA PA Q (2*lam*s) (ε*s) lam →
  27       BipartiteBlockBounds EA (fun x => ¬PA x) (fun y => ¬Q y) (2*lam*s) (ε*s) lam →
  28       BipartiteBlockBounds EX PX (fun y => ¬Q y) (2*lam*s) (ε*s) lam →
  29       BipartiteBlockBounds EX (fun x => ¬PX x) (fun y => ¬Q y) (2*lam*s) (ε*s) lam →
  30       BipartiteBlockBounds EX (fun x => ¬PX x) Q (2*lam*s) (ε*s) lam →
  31       ∃ f : A ≃ Y, ∃ g : X ≃ Y,
  32         (∀ x, EA x (f x)) ∧ (∀ x, EX x (g x)) ∧
  33         (∀ x, Q (f x) ↔ PA x) ∧ (∀ x, PX x → ¬Q (g x)) ∧
  34         ∀ y, ¬PermutationClose π b (labelA (f.symm y)) (labelX (g.symm y)) := by
  35   let C : ℝ := (M*(2*b+1):ℕ)
  36   obtain ⟨s₀,hs₀⟩ := exists_nat_gt (max 1 (max (C/lam) (C/(lam*ε))))
  37   refine ⟨s₀,?_⟩
  38   intro s hs A X Y V _ _ _ _ EA EX PA PX Q hA hX hY hclass hsize hallowed π labelA labelX hfA hfX B₁ B₂ B₃ B₄ B₅
  39   have hsr : (s₀:ℝ)≤s := Nat.cast_le.mpr hs
  40   have hspos : (0:ℝ)<s := by
  41     have hh := (le_max_left 1 (max (C/lam) (C/(lam*ε)))).trans_lt (hs₀.trans_le hsr)
  42     linarith
  43   have hC₁ : C<lam*s := by
  44     have hh := (le_max_left (C/lam) (C/(lam*ε))).trans (le_max_right 1 _)
  45     have hh' := (div_lt_iff₀ hlam).mp (hh.trans_lt (hs₀.trans_le hsr))
  46     nlinarith only [hh']
  47   have hC₂ : C<lam*(ε*s) := by
  48     have hh := (le_max_right (C/lam) (C/(lam*ε))).trans (le_max_right 1 _)
  49     have hh' := (div_lt_iff₀ (mul_pos hlam hε)).mp (hh.trans_lt (hs₀.trans_le hsr))
  50     nlinarith only [hh']
  51   have hdegree : 2*(ε*s)+C≤2*lam*s := by
  52     have hh := mul_lt_mul_of_pos_right hepslam hspos
  53     nlinarith only [hh,hC₁,mul_pos hε hspos]
  54   have hsplit : ((Finset.univ.filter Q).card:ℝ)+
  55       (Finset.univ.filter (fun y => ¬Q y)).card=s := by
  56     have hh := Finset.card_filter_add_card_filter_not (s:=(Finset.univ:Finset Y)) Q
  57     rw [Finset.card_univ,hY] at hh
  58     exact_mod_cast hh
  59   have hgap : ((Finset.univ.filter PX).card:ℝ)+2*(ε*s)≤
  60       (Finset.univ.filter (fun y => ¬Q y)).card := by
  61     have hh := mul_lt_mul_of_pos_right hεκ hspos
  62     nlinarith only [hsize,hsplit,hh,mul_pos hε hspos]
  63   exact marked_typed_endpoint_pairing_of_bounds EA EX PA PX Q
  64     (hA.trans hY.symm) (hX.trans hY.symm) hclass hallowed π b M labelA labelX hfA hfX
  65     (2*lam*s) (ε*s) lam (mul_pos hε hspos) hlam hdegree hC₂ hgap B₁ B₂ B₃ B₄ B₅
