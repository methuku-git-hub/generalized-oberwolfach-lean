  13 theorem signed_weights_from_capacities {I E : Type*} [Fintype I] [Fintype E]
  14     (A : I → E → ℝ) (S : Submodule ℝ (E → ℝ)) (w₀ : E → ℝ) (hw₀ : w₀∈S)
  15     (κ B : ℝ) (hκ : 0<κ) (hB : 0≤B)
  16     (hcap : ∀ w : E → ℝ,w∈S → ‖w-w₀‖≤κ →
  17       ∃ q : I → ℝ,(∀ i,0≤q i ∧ q i≤B) ∧ (∑ i,q i • A i)=w)
  18     (z : E → ℝ) (hz : z∈S) (η : ℝ) (hη : 0≤η) (hnorm : ‖z‖≤η) :
  19     ∃ q : I → ℝ,(∀ i,|q i|≤(2*B/κ)*η) ∧ (∑ i,q i • A i)=z := by
  20   by_cases hη0 : η=0
  21   · have hz0 : z=0 := norm_eq_zero.mp (le_antisymm (hη0 ▸ hnorm) (norm_nonneg _))
  22     refine ⟨fun _ => 0,?_,?_⟩
  23     · intro i;simp only [abs_zero,hη0,mul_zero,le_refl]
  24     · simp only [zero_smul,Finset.sum_const_zero,hz0]
  25   · have hηpos : 0<η := lt_of_le_of_ne hη (Ne.symm hη0)
  26     obtain ⟨q₀,hq₀,he₀⟩ := hcap w₀ hw₀ (by simpa using hκ.le)
  27     let w := w₀+(κ/η) • z
  28     have hw : w∈S := S.add_mem hw₀ (S.smul_mem _ hz)
  29     have hwclose : ‖w-w₀‖≤κ := by
  30       have he : w-w₀=(κ/η) • z := by dsimp [w];module
  31       rw [he,norm_smul,Real.norm_eq_abs,abs_of_pos (div_pos hκ hηpos)]
  32       calc
  33         _ ≤ (κ/η)*η := mul_le_mul_of_nonneg_left hnorm (by positivity)
  34         _ = κ := by field_simp
  35     obtain ⟨q₁,hq₁,he₁⟩ := hcap w hw hwclose
  36     let q (i : I) : ℝ := (η/κ)*(q₁ i-q₀ i)
  37     refine ⟨q,?_,?_⟩
  38     · intro i
  39       have hd : |q₁ i-q₀ i|≤2*B := by
  40         calc
  41           _ ≤ |q₁ i|+|q₀ i| := by simpa only [sub_eq_add_neg,abs_neg] using abs_add_le (q₁ i) (-q₀ i)
  42           _ ≤ B+B := by rw [abs_of_nonneg (hq₁ i).1,abs_of_nonneg (hq₀ i).1];exact add_le_add (hq₁ i).2 (hq₀ i).2
  43           _ = _ := by ring
  44       dsimp [q]
  45       rw [abs_mul,abs_of_pos (div_pos hηpos hκ)]
  46       calc
  47         _ ≤ (η/κ)*(2*B) := mul_le_mul_of_nonneg_left hd (by positivity)
  48         _ = _ := by ring
  49     · calc
  50         _ = (η/κ) • ((∑ i,q₁ i • A i)-(∑ i,q₀ i • A i)) := by
  51           simp only [q,mul_sub,smul_sub,Finset.smul_sum,Finset.sum_sub_distrib,smul_smul,sub_smul]
  52         _ = (η/κ) • (w-w₀) := by rw [he₁,he₀]
  53         _ = z := by
  54           have he : w-w₀=(κ/η) • z := by dsimp [w];module
  55           rw [he,smul_smul,show (η/κ)*(κ/η)=(1:ℝ) by field_simp,one_smul]
  56
