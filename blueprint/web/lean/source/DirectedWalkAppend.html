   8 theorem walk_append_preserves_successor_relation

  10     (hp:∀k,k<p.length→R (p.getVert k) (p.getVert (k+1)))
  11     (hq:∀k,k<q.length→R (q.getVert k) (q.getVert (k+1))):
  12     ∀k,k<(p.append q).length→R ((p.append q).getVert k) ((p.append q).getVert (k+1)):=by
  13   intro k hk
  14   rw [SimpleGraph.Walk.length_append] at hk
  15   by_cases h:k<p.length
  16   · rw [SimpleGraph.Walk.getVert_append' p q k,SimpleGraph.Walk.getVert_append' p q (k+1)]
  17     simp only [if_pos (show k≤p.length by omega),if_pos (show k+1≤p.length by omega)]
  18     exact hp k h
  19   · have hle:p.length≤k:=by omega
  20     rw [SimpleGraph.Walk.getVert_append p q k,SimpleGraph.Walk.getVert_append p q (k+1)]
  21     simp only [if_neg (show ¬k<p.length by omega),if_neg (show ¬k+1<p.length by omega)]
  22     have he:k+1-p.length=(k-p.length)+1:=by omega
  23     rw [he]
  24     exact hq (k-p.length) (by omega)
