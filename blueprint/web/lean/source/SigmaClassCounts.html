  14 theorem sigma_finset_class_card (A : Finset (Σ i,V i)) (i : I) :
  15     (A.filter (fun v => v.1=i)).card=
  16       (Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈A)).card := by
  17   let f : {v : V i // (⟨i,v⟩ : Σ i,V i)∈A} → A.filter (fun v => v.1=i) :=
  18     fun v => ⟨⟨i,v.val⟩,Finset.mem_filter.mpr ⟨v.property,rfl⟩⟩
  19   have hf : Function.Bijective f := by
  20     constructor
  21     · intro v w he
  22       apply Subtype.ext
  23       have hh : (⟨i,v.val⟩ : Σ i,V i)=⟨i,w.val⟩ := congrArg Subtype.val he
  24       exact eq_of_heq (Sigma.mk.inj_iff.mp hh).2
  25     · rintro ⟨⟨j,v⟩,hv⟩
  26       have he : j=i := (Finset.mem_filter.mp hv).2
  27       subst j
  28       exact ⟨⟨v,(Finset.mem_filter.mp hv).1⟩,rfl⟩
  29   have hh := Fintype.card_congr (Equiv.ofBijective f hf)
  30   simpa only [Fintype.card_subtype,Fintype.card_coe,Finset.filter_mem_eq_inter,Finset.univ_inter] using hh.symm
