  25 theorem triangle_edge_shifted_edge_injective (G:SimpleGraph V)
  26     (σ:∀ v,Equiv.Perm (G.neighborSet v)) (hσ:∀ v w,σ v w≠w):
  27     Function.Injective (fun d:Σ v,G.neighborSet v=>
  28       ((rimIncidenceEquiv G d).1,(shiftedIncidenceEquiv G σ d).1)):=by
  29   rintro ⟨v,w⟩ ⟨v',w'⟩ he
  30   have h1:=congrArg Subtype.val (Prod.mk.inj he).1
  31   have h2:=congrArg Subtype.val (Prod.mk.inj he).2
  32   rw [rimIncidenceEquiv_edge,rimIncidenceEquiv_edge] at h1
  33   rw [shiftedIncidenceEquiv_edge,shiftedIncidenceEquiv_edge] at h2
  34   change s(v,w.val)=s(v',w'.val) at h1
  35   change s(v,(σ v w).val)=s(v',(σ v' w').val) at h2
  36   rcases Sym2.eq_iff.mp h1 with ⟨hvv,hww⟩|⟨hvw,hwh⟩
  37   · subst v'
  38     have hw:w=w':=Subtype.ext hww
  39     subst w';rfl
  40   · rcases Sym2.eq_iff.mp h2 with ⟨hvv,_⟩|⟨_,hsv⟩
  41     · exact (w.property.ne (hvv.trans hwh.symm)).elim
  42     · have hfixed:σ v w=w:=Subtype.ext (hsv.trans hwh.symm)
  43       exact (hσ v w hfixed).elim
