  13 theorem endpointReserve_left_ratio {X Y Z : Type*} [Fintype Y] [Fintype Z]
  14     (L : X → Y → Prop) (R : Y → Z → Prop) (Q : X → Z → ℝ)
  15     (x : X) (y : Y) (D m : ℝ) (hD : 0 < D)
  16     (hQ : ∀ z, 0 ≤ Q x z) (hrow : ∑ z, Q x z = m)
  17     (hc : ∀ z, D ≤ (intervalCommon L R x z : ℝ)) :
  18     (∑ z, endpointReserve L R Q x y z) ≤ m / D := by
  19   calc
  20     (∑ z, endpointReserve L R Q x y z) ≤ ∑ z, Q x z / D := by
  21       apply Finset.sum_le_sum
  22       intro z _
  23       unfold endpointReserve
  24       split_ifs
  25       · exact div_le_div_of_nonneg_left (hQ z) hD (hc z)
  26       · exact div_nonneg (hQ z) hD.le
  27     _ = m / D := by rw [← Finset.sum_div, hrow]

  29 theorem endpointReserve_left_margin {X Y Z : Type*} [Fintype Y] [Fintype Z]
  30     (L : X → Y → Prop) (R : Y → Z → Prop) (Q : X → Z → ℝ)
  31     (k δ α : ℝ) (hk : 2 ≤ k) (hn : 0 < Fintype.card Y) (hα : 0 < α)
  32     (hthreshold : (k + (2 * k - 1) * α) * Fintype.card Y ≤ (2 * k - 1) * δ)
  33     (hdL : ∀ x, δ ≤ (Finset.univ.filter (L x)).card)
  34     (hdR : ∀ z, δ ≤ (Finset.univ.filter (fun y => R y z)).card)
  35     (hQ : ∀ x z, 0 ≤ Q x z)
  36     (hrow : ∀ x, ∑ z, Q x z = (Finset.univ.filter (L x)).card / k)
  37     (x : X) (y : Y) :
  38     (∑ z, endpointReserve L R Q x y z) ≤
  39       (1 - (2 * k - 1) * α / k) * interfaceCapacity L x y := by
  40   have hdn : ((Finset.univ.filter (L x)).card : ℝ) ≤ Fintype.card Y := by
  41     exact_mod_cast Finset.card_le_univ (Finset.univ.filter (L x))
  42   obtain ⟨_, _, hden, hcap⟩ := reserve_capacity_positive_margin hk
  43     (by exact_mod_cast hn) hα hthreshold (hdL x) hdn ((hdL x).trans hdn)
  44   by_cases hxy : L x y
  45   · simp only [interfaceCapacity, hxy, ite_true, mul_one]
  46     apply le_trans (endpointReserve_left_ratio L R Q x y
  47       ((Finset.univ.filter (L x)).card + δ - Fintype.card Y)
  48       ((Finset.univ.filter (L x)).card / k) hden (hQ x) (hrow x) ?_) hcap
  49     intro z
  50     have hc : ((Finset.univ.filter (L x)).card : ℝ) +
  51         (Finset.univ.filter (fun y => R y z)).card ≤
  52         intervalCommon L R x z + Fintype.card Y := by
  53       exact_mod_cast intervalCommon_lower L R x z
  54     linarith [hdR z]

  57 theorem endpointReserve_right_margin {X Y Z : Type*} [Fintype X] [Fintype Y]
  58     (L : X → Y → Prop) (R : Y → Z → Prop) (Q : X → Z → ℝ)
  59     (k δ α : ℝ) (hk : 2 ≤ k) (hn : 0 < Fintype.card Y) (hα : 0 < α)
  60     (hthreshold : (k + (2 * k - 1) * α) * Fintype.card Y ≤ (2 * k - 1) * δ)
  61     (hdL : ∀ x, δ ≤ (Finset.univ.filter (L x)).card)
  62     (hdR : ∀ z, δ ≤ (Finset.univ.filter (fun y => R y z)).card)
  63     (hQ : ∀ x z, 0 ≤ Q x z)
  64     (hcol : ∀ z, ∑ x, Q x z = (Finset.univ.filter (fun y => R y z)).card / k)
  65     (y : Y) (z : Z) :
  66     (∑ x, endpointReserve L R Q x y z) ≤
  67       (1 - (2 * k - 1) * α / k) * interfaceCapacity R y z := by
  68   have hh := endpointReserve_left_margin (fun z y => R y z) (fun y x => L x y)
  69     (fun z x => Q x z) k δ α hk hn hα hthreshold hdR hdL (fun z x => hQ x z) hcol z y
  70   have heq (x : X) :
  71       endpointReserve (fun z y => R y z) (fun y x => L x y) (fun z x => Q x z) z y x =
