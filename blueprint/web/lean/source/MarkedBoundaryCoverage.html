  16 theorem marked_boundary_coverage_from_incidence_lists

  18     (hroots:∀ v∉X,∃ j,root j=v) (hhubs:∀ u∉Z,∃ i,hub i=u)

  21     (hin:∀ j a,Sum.elim head khead a=root j→∃ w,incoming j w=a)
  22     (hout:∀ j a,Sum.elim tail ktail a=root j→∃ w,outgoing j w=a)
  23     (hrow:∀ j u,H₀ (root j) u∨HK (root j) u→u∈Y₀ j∪Y₁ j)
  24     (hcol:∀ i v,H₀ v (hub i)∨HK v (hub i)→v∈Y i)

  27     (hMo:∀ j a,Sum.inl (markedPhysicalEdge tail head ktail khead π (outgoing j a))∈M.biUnion markedWheelPhysicalResources)
  28     (hMr:∀ j u,u∈Y₀ j∪Y₁ j→Sum.inr (root j,u)∈M.biUnion markedWheelPhysicalResources)
  29     (hMh:∀ i v,v∈Y i→Sum.inr (v,hub i)∈M.biUnion markedWheelPhysicalResources):
  30     (∀ a:E⊕T,Sum.elim tail ktail a∉X∨Sum.elim head khead a∉X→
  31       Sum.inl (markedPhysicalEdge tail head ktail khead π a)∈M.biUnion markedWheelPhysicalResources)∧
