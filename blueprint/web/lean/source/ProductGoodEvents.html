  12 theorem finiteLaw_product_good_lower {Ω Γ : Type*} [Fintype Ω] [Fintype Γ]
  13     (μ : FiniteLaw Ω) (ν : FiniteLaw Γ) (Good : Ω → Prop) (A : Ω → Γ → Prop)
  14     (b c : ℝ) (hc : 0≤c) (hgood : 1-b≤μ.probability Good)
  15     (hsecond : ∀ ω,Good ω → 1-c≤ν.probability (A ω)) :
  16     1-b-c≤(μ.prod ν).probability (fun z=>Good z.1 ∧ A z.1 z.2) := by
  17   have hbad : ∀ ω,Good ω → ν.probability (fun γ=>¬A ω γ)≤c := by
  18     intro ω hω
  19     have hh := finiteLaw_probability_compl ν (A ω)
  20     linarith only [hh,hsecond ω hω]
  21   have hh := finiteLaw_product_rooted_failure μ ν (fun _=>True) (fun ω=>¬Good ω)
  22     (fun ω γ=>¬(Good ω ∧ A ω γ)) b c hc
  23     (by
  24       have hh := finiteLaw_probability_compl μ Good
  25       have htrue : μ.probability (fun _=>True)=1 := by simp [FiniteLaw.probability,μ.total]
  26       simpa only [true_and,htrue,mul_one] using (show μ.probability (fun ω=>¬Good ω)≤b by linarith only [hh,hgood]))
  27     (by
  28       intro ω hω
  29       have hg : Good ω := not_not.mp hω
  30       simpa only [hg,true_and] using hbad ω hg)
  31   simp only [true_and] at hh
  32   have htrue : μ.probability (fun _=>True)=1 := by simp [FiniteLaw.probability,μ.total]
  33   rw [htrue,mul_one] at hh
