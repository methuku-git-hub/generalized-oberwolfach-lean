  15 theorem prescribed_actual_forbidden_graph
  16     (Q:I→SimpleGraph X) (A A₀:I→Finset X) (f ψ:I→X→V) (T:Finset (I×X))
  17     (hinj:FamilyPartialInjections A f)
  18     (hdom:∀ i x,x∈A i ↔ x∈A₀ i ∨ (i,x)∈T)
  19     (hpres:∀ i x,x∈A₀ i→f i x=ψ i x)
  20     (hQ:∀ i x,(Q i).degree x≤2)
  21     (δ n t:ℝ)
  22     (hold:∀ v,((Finset.univ.filter (fun i=>∃ x,x∈A₀ i ∧ ψ i x=v)).card:ℝ)≤δ*n)
  23     (hfree:∀ v,((T.filter (fun p=>f p.1 p.2=v)).card:ℝ)≤t+1)
  24     (J:SimpleGraph V) (hJ:∀ v,(J.degree v:ℝ)≤δ*n):
  25     ∃ F:SimpleGraph V,(∀ v,(F.degree v:ℝ)≤3*δ*n+2*t+2) ∧
  26       J.edgeFinset⊆F.edgeFinset ∧
  27       ∀ i,((Q i).induce (↑(A i):Set X) |>.map (prescribedPartialEmbedding (A i) (f i) (hinj i))).edgeFinset⊆F.edgeFinset := by
  28   let X' : I→Type:=fun i=>{x // x∈A i}
  29   let Q' : ∀ i,SimpleGraph (X' i):=fun i=>(Q i).induce (↑(A i):Set X)
  30   let φ : ∀ i,X' i↪V:=fun i=>prescribedPartialEmbedding (A i) (f i) (hinj i)
  31   let H:I→SimpleGraph V:=fun i=>(Q' i).map (φ i)
  32   let old:I→V→Prop:=fun i v=>∃ x,x∈A₀ i ∧ ψ i x=v
  33   let fresh:I→V→Prop:=fun i v=>∃ x,(i,x)∈T ∧ f i x=v
  34   have hsource:∀ i,H i≤(Q' i).map (φ i):=fun _=>le_rfl
  35   have hdegree:∀ i x,(Q' i).degree x≤2:=by
  36     intro i x
  37     exact (prescribed_induced_degree_le (Q i) (A i) x).trans (hQ i x.1)
  38   have hocc:∀ i v,(∃ x,φ i x=v)→ old i v ∨ fresh i v:=by
  39     intro i v hex
  40     obtain ⟨x,hx⟩:=hex
  41     have hA: x.1∈A₀ i ∨ (i,x.1)∈T:=(hdom i x.1).mp x.2
  42     rcases hA with hA|hA
  43     · left
  44       refine ⟨x.1,hA,?_⟩
  45       exact (hpres i x.1 hA).symm.trans hx
  46     · exact Or.inr ⟨x.1,hA,hx⟩
  47   have hold':∀ v,((Finset.univ.filter (fun i=>old i v)).card:ℝ)≤δ*n:=by
  48     intro v
  49     have hsub:(Finset.univ.filter (fun i=>old i v))⊆
  50         (Finset.univ.filter (fun i=>∃ x,x∈A₀ i ∧ ψ i x=v)):=by
  51       intro i hi
  52       simp only [Finset.mem_filter,Finset.mem_univ,true_and] at hi ⊢
  53       exact hi
  54     have hc:((Finset.univ.filter (fun i=>old i v)).card:ℝ)≤
  55         ((Finset.univ.filter (fun i=>∃ x,x∈A₀ i ∧ ψ i x=v)).card:ℝ):=by exact_mod_cast Finset.card_le_card hsub
  56     exact hc.trans (hold v)
  57   have hfresh':∀ v,((Finset.univ.filter (fun i=>fresh i v)).card:ℝ)≤t+1:=by
  58     intro v
  59     have hsub:(Finset.univ.filter (fun i=>fresh i v))⊆
  60         (Finset.univ.filter (fun i=>∃ x,(i,x)∈T ∧ f i x=v)):=by
  61       intro i hi
  62       simp only [Finset.mem_filter,Finset.mem_univ,true_and] at hi ⊢
  63       exact hi
  64     have hc:((Finset.univ.filter (fun i=>fresh i v)).card:ℝ)≤
  65         ((Finset.univ.filter (fun i=>∃ x,(i,x)∈T ∧ f i x=v)).card:ℝ):=by exact_mod_cast Finset.card_le_card hsub
  66     have hsub':(Finset.univ.filter (fun i=>∃ x,(i,x)∈T ∧ f i x=v))⊆
  67         (T.filter (fun p=>f p.1 p.2=v)).image Prod.fst:=by
  68       intro i hi
  69       simp only [Finset.mem_filter,Finset.mem_univ,true_and] at hi
  70       obtain ⟨x,hx,hxv⟩:=hi
  71       exact Finset.mem_image.mpr ⟨(i,x),Finset.mem_filter.mpr ⟨hx,hxv⟩,rfl⟩
  72     have hnat':(Finset.univ.filter (fun i=>∃ x,(i,x)∈T ∧ f i x=v)).card≤
  73         (T.filter (fun p=>f p.1 p.2=v)).card:=
  74       (Finset.card_le_card hsub').trans Finset.card_image_le
  75     have hreal:((Finset.univ.filter (fun i=>∃ x,(i,x)∈T ∧ f i x=v)).card:ℝ)≤
  76         ((T.filter (fun p=>f p.1 p.2=v)).card:ℝ):=by exact_mod_cast hnat'
  77     exact hc.trans (hreal.trans (hfree v))
  78   let G:SimpleGraph V:=⨆ i,H i
  79   let F:SimpleGraph V:=G⊔J
  80   have hh:=prescribed_forbidden_graph_from_partial_images X' Q' φ H old fresh hsource hdegree hocc J δ n t (by
  81     intro v
  82     convert hold' v using 1
  83     congr 1
  84     congr 1
  85     ext i
  86     simp only [Finset.mem_filter,Finset.mem_univ,true_and]) (by
  87     intro v
  88     convert hfresh' v using 1
  89     congr 1
  90     congr 1
  91     ext i
  92     simp only [Finset.mem_filter,Finset.mem_univ,true_and]) hJ
  93   refine ⟨F,?_,?_,?_⟩
  94   · intro v
  95     exact (hh.1 v).2
  96   · intro e he
  97     simp only [SimpleGraph.mem_edgeFinset] at he ⊢
  98     have hJF:J≤F:=by dsimp [F];exact le_sup_right
  99     exact SimpleGraph.edgeSet_mono hJF he
 100   · intro i e he
 101     simp only [SimpleGraph.mem_edgeFinset] at he ⊢
 102     change e∈(H i).edgeSet at he
 103     have hHi:H i≤F:=by
 104       dsimp [F,G]
 105       exact (le_iSup H i).trans le_sup_left
 106     exact SimpleGraph.edgeSet_mono hHi he
