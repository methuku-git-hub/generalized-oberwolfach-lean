  14 theorem finiteLaw_complement_covariance {Ω : Type*} [Fintype Ω]
  15     (μ : FiniteLaw Ω) (E A : Ω → Prop) :
  16     |μ.probability (fun x=>¬E x ∧ A x)-(1-μ.probability E)*μ.probability A|=
  17       |μ.probability (fun x=>E x ∧ A x)-μ.probability E*μ.probability A| := by
  18   have hh := finiteLaw_probability_split μ A E
  19   have h₁ : μ.probability (fun x=>A x ∧ E x)=μ.probability (fun x=>E x ∧ A x) :=
  20     finiteLaw_probability_congr _ _ _ (fun _=>and_comm)
  21   have h₂ : μ.probability (fun x=>A x ∧ ¬E x)=μ.probability (fun x=>¬E x ∧ A x) :=
  22     finiteLaw_probability_congr _ _ _ (fun _=>and_comm)
  23   rw [h₁,h₂] at hh
  24   have he : μ.probability (fun x=>¬E x ∧ A x)-(1-μ.probability E)*μ.probability A=
  25       -(μ.probability (fun x=>E x ∧ A x)-μ.probability E*μ.probability A) := by linarith
  26   rw [he,abs_neg]
