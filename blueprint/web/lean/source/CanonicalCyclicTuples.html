  13 def cyclicTupleOfVertices {k : ℕ} {X Y : Type*}
  14     (v : Fin (k + 2) → X × Y) (i : Fin (k + 2)) : X × Y × X :=
  15   ((v i).1, (v i).2, (v (finRotate (k + 2) i)).1)

  17 def cyclicTupleVertices {k : ℕ} {X Y : Type*}
  18     (p : Fin (k + 2) → X × Y × X) (i : Fin (k + 2)) : X × Y :=
  19   ((p i).1, (p i).2.1)

  21 theorem cyclicTupleVertices_inverse {k : ℕ} {X Y : Type*}

  23     cyclicTupleVertices (cyclicTupleOfVertices v) = v := by

  25   rfl

  27 theorem cyclicTupleOfVertices_inverse {k : ℕ} {X Y : Type*}
  28     (p : Fin (k + 2) → X × Y × X)
  29     (hp : ∀ i, (p i).2.2 = (p (finRotate (k + 2) i)).1) :
  30     cyclicTupleOfVertices (cyclicTupleVertices p) = p := by
  31   funext i
  32   apply Prod.ext
  33   · rfl
  34   · apply Prod.ext
  35     · rfl
  36     · exact (hp i).symm

  38 theorem alternatingTupleGraph_determines_vertices {k : ℕ} {X Y : Type*}
  39     (p q : Fin (k + 2) → X × Y × X)
  40     (h : alternatingTupleGraph p = alternatingTupleGraph q) :
  41     cyclicTupleVertices p = cyclicTupleVertices q := by
  42   funext i
  43   have he : (alternatingTupleGraph p).Adj
  44       (Sum.inl (i, (p i).1)) (Sum.inr (i, (p i).2.1)) :=
  45     (alternatingTupleGraph_adj_cross p i i _ _).mpr ⟨Or.inl rfl, rfl, rfl⟩
  46   rw [h] at he
  47   obtain ⟨_, hx, hy⟩ := (alternatingTupleGraph_adj_cross q i i _ _).mp he
  48   exact Prod.ext hx.symm hy.symm

  50 theorem canonicalCycleGraph_injective {k : ℕ} {X Y : Type*} :
  51     Function.Injective (fun v : Fin (k + 2) → X × Y =>
  52       alternatingTupleGraph (cyclicTupleOfVertices v)) := by
  53   intro v w h
  54   simpa only [cyclicTupleVertices_inverse] using
  55     alternatingTupleGraph_determines_vertices _ _ h

  57 theorem compatible_cycle_tuple_injective {k : ℕ} {X Y : Type*}
  58     (p q : Fin (k + 2) → X × Y × X)
  59     (hp : ∀ i, (p i).2.2 = (p (finRotate (k + 2) i)).1)
  60     (hq : ∀ i, (q i).2.2 = (q (finRotate (k + 2) i)).1)
  61     (he : alternatingTupleGraph p = alternatingTupleGraph q) : p = q := by
  62   rw [← cyclicTupleOfVertices_inverse p hp, ← cyclicTupleOfVertices_inverse q hq]
  63   rw [alternatingTupleGraph_determines_vertices p q he]

  65 theorem cyclic_weight_of_canonical_graph {k : ℕ} {X Y : Type*}
  66     [Fintype X] [Fintype Y]
  67     (ω : (Fin (k + 2) → X × Y × X) → ℝ)
  68     (hω : ∀ p, 0 ≤ ω p)
  69     (hc : ∀ p, 0 < ω p → ∀ i, (p i).2.2 = (p (finRotate (k + 2) i)).1)
  70     (v : Fin (k + 2) → X × Y) :
  71     (∑ p, if alternatingTupleGraph p = alternatingTupleGraph (cyclicTupleOfVertices v)
  72       then ω p else 0) = ω (cyclicTupleOfVertices v) := by
  73   classical
  74   have hp : ∀ i, (cyclicTupleOfVertices v i).2.2 =
  75       (cyclicTupleOfVertices v (finRotate (k + 2) i)).1 := fun _ => rfl
  76   rw [Finset.sum_eq_single (cyclicTupleOfVertices v)]
  77   · simp
  78   · intro p _ hne
  79     by_cases he : alternatingTupleGraph p = alternatingTupleGraph (cyclicTupleOfVertices v)
  80     · simp only [he, ite_true]
  81       by_contra hz
  82       have hpos := lt_of_le_of_ne (hω p) (Ne.symm hz)
  83       exact hne (compatible_cycle_tuple_injective p _ (hc p hpos) hp he)
  84     · simp only [he, ite_false]
  85   · intro h
  86     exact False.elim (h (Finset.mem_univ _))
