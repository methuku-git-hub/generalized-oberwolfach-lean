  12 theorem ordinary_degree_normalized_seed_bounds {V:Type} [Fintype V] [DecidableEq V]
  13     (P F:SimpleGraph V) (r:ℕ) (hr:2≤r) (hn:0<Fintype.card V)
  14     (weight:OrdinaryCycle P r→ℝ) (hweight:∀ C,0≤weight C)
  15     (B:ℝ) (hB:0≤B) (hb:∀ C,weight C≤B/(Fintype.card V:ℝ)^(r-2))
  16     (D:ℝ) (hD:(Fintype.card V:ℝ)/2≤D):
  17     ∀ W:OrdinaryWheel P F (fun (_:V) (_:Unit)=>True) r,
  18       0≤ordinaryCycleWheelWeight P F r weight (D/2) W ∧
  19       ordinaryCycleWheelWeight P F r weight (D/2) W≤(4*B)/(Fintype.card V:ℝ)^(r-1) := by
  20   intro W
  21   have hn0:(0:ℝ)<Fintype.card V:=by exact_mod_cast hn
  22   have hD0:0<D/2:=by linarith
  23   have hDquarter:(Fintype.card V:ℝ)/4≤D/2:=by linarith
  24   have hinv:1/(D/2)≤4/(Fintype.card V:ℝ):=by
  25     apply (div_le_div_iff₀ hD0 hn0).mpr
  26     linarith only [hDquarter]
  27   have hw:=hb W.val.1
  28   have hpow:0<(Fintype.card V:ℝ)^(r-2):=by positivity
  29   have hscale:weight W.val.1*(1/(D/2))≤(B/(Fintype.card V:ℝ)^(r-2))*(4/(Fintype.card V:ℝ)):=
  30     mul_le_mul hw hinv (by positivity) (div_nonneg hB hpow.le)
  31   have heq:(B/(Fintype.card V:ℝ)^(r-2))*(4/(Fintype.card V:ℝ))=
  32       (4*B)/(Fintype.card V:ℝ)^(r-1):=by
  33     have hsub:(r-2)+1=r-1:=by omega
  34     rw [←hsub,pow_succ]
  35     field_simp
  36   refine ⟨?_,?_⟩
  37   · exact div_nonneg (hweight W.val.1) hD0.le
  38   · change weight W.val.1/(D/2)≤_
  39     rw [div_eq_mul_inv,←one_div]
  40     exact hscale.trans_eq heq
  41 end Oberwolfach
