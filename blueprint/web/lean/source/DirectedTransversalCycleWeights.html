  14 theorem directed_transversal_weight_restriction {c : ℕ} {X : Type*} [Fintype X]
  15     (a : Fin c → X → X → ℝ) (ha : ∀ i x y,0≤a i x y)
  16     (w : (Fin c → X) → ℝ)
  17     (hz : ∀ v,(∃ i,a i (v i) (v (finRotate c i))=0) → w v=0)
  18     (he : ∀ i x y,(∑ v,if v i=x ∧ v (finRotate c i)=y then w v else 0)=a i x y)
  19     (i : Fin c) (x y : X) :
  20     (∑ v : DirectedTransversalCycle a,if v.val i=x ∧ v.val (finRotate c i)=y then w v.val else 0)=a i x y := by
  21   classical
  22   let S : Set (Fin c → X) := {v | ∀ i,0<a i (v i) (v (finRotate c i))}
  23   have hh := Finset.sum_congr_set S
  24     (fun v => if v i=x ∧ v (finRotate c i)=y then w v else 0)
  25     (fun v : S => if v.val i=x ∧ v.val (finRotate c i)=y then w v.val else 0)
  26     (fun _ _ => rfl) (by
  27       intro v hv
  28       have hnot : ¬∀ i,0<a i (v i) (v (finRotate c i)) := hv
  29       push Not at hnot
  30       obtain ⟨j,hj⟩ := hnot
  31       have hzero := hz v ⟨j,le_antisymm hj (ha _ _ _)⟩
  32       simp only [hzero,ite_self])
  33   rw [he] at hh
  34   exact hh.symm
  35 
  36 end Oberwolfach
