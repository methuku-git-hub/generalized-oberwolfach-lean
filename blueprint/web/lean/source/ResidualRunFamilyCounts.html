  10 theorem maximal_run_family_counts (s:Finset I) (d:ℕ) (hd:0<d)
  11     (T:I→Finset ℕ) (hT:∀ i∈s,T i⊆Finset.range d):
  12     let missing:=∑ i∈s,(d-(T i).card)
  13     let runs:=∑ i∈s,(selectedSegmentRuns d (T i)).card
  14     let markers:=∑ i∈s,(survivingSegmentMarkers d (T i)).card;
  15     markers=s.card+missing+runs ∧ runs≤s.card+missing ∧ s.card≤runs+missing ∧
  16     missing=d*s.card-∑ i∈s,(T i).card := by
  17   dsimp only
  18   have hmarker:∀i∈s,(survivingSegmentMarkers d (T i)).card=
  19       1+(d-(T i).card)+(selectedSegmentRuns d (T i)).card := by
  20     intro i hi
  21     exact (maximal_selected_runs_and_markers_count d hd (T i) (hT i hi)).1
  22   have hb1:∀i∈s,(selectedSegmentRuns d (T i)).card≤(d-(T i).card)+1 := by
  23     intro i hi
  24     exact (maximal_selected_runs_and_markers_count d hd (T i) (hT i hi)).2.1
  25   have hb2:∀i∈s,1≤(selectedSegmentRuns d (T i)).card+(d-(T i).card) := by
  26     intro i hi
  27     exact (maximal_selected_runs_and_markers_count d hd (T i) (hT i hi)).2.2.1
  28   have hsum1:=Finset.sum_le_sum hb1
  29   have hsum2:=Finset.sum_le_sum hb2
  30   have hsumM : (∑i∈s,(survivingSegmentMarkers d (T i)).card)=
  31       ∑i∈s,(1+(d-(T i).card)+(selectedSegmentRuns d (T i)).card) := by
  32     apply Finset.sum_congr rfl
  33     exact hmarker
  34   have hsub:∀i∈s,(T i).card≤d := by
  35     intro i hi
  36     have hh:=Finset.card_le_card (hT i hi)
  37     simpa using hh
  38   have hsumsub:=Finset.sum_tsub_distrib s hsub
  39   simp only [Finset.sum_add_distrib,Finset.sum_const_zero,Finset.sum_const,
  40     nsmul_eq_mul,Finset.card_attach, mul_comm] at hsum1 hsum2 hsumM hsumsub ⊢
  41   simp only [Nat.cast_id, mul_one] at hsum1 hsum2 hsumM hsumsub
  42   omega
