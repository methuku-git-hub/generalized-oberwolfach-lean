  12 def ConflictFreeRootSeparatedCycleFactorPair {V:Type} [Fintype V] [DecidableEq V]
  13     (G F:SimpleGraph V) (T:Finset V) (r k:ℕ):Prop:=
  14   ∃ S:Finset V,S.card=k ∧ Disjoint S T ∧
  15     (∀ x∈S∪T,∀ y∈S∪T,¬F.Adj x y) ∧
  16     let R:=S∪T
  17     let J:=G.induce (R:Set V)
  18     let S':Finset (R:Set V):=S.subtype (fun v=>v∈R)
  19     ∃ M₁ M₂:Finset (OrdinaryCycle J r),
  20       (∀ C∈M₁,∀ E∈M₁,C≠E→Disjoint (ordinaryCycleVertices J r C) (ordinaryCycleVertices J r E)) ∧
  21       M₁.biUnion (ordinaryCycleVertices J r)=Finset.univ ∧
  22       (∀ C∈M₂,∀ E∈M₂,C≠E→Disjoint (ordinaryCycleVertices J r C) (ordinaryCycleVertices J r E)) ∧
  23       M₂.biUnion (ordinaryCycleVertices J r)=S' ∧
  24       (∀ C∈M₁,∀ E∈M₂,Disjoint C.val.val.edgeFinset E.val.val.edgeFinset) ∧
  25       (∀ C∈M₁,(ordinaryCycleVertices J r C∩(Finset.univ\S')).card≤1)
