  19 def retainedGraph {V : Type*} (G : SimpleGraph V) (σ : Sym2 V → Bool) : SimpleGraph V where
  20   Adj u v := G.Adj u v ∧ σ s(u,v)=true
  21   symm := ⟨by intro u v h;simpa only [Sym2.eq_swap] using (show G.Adj v u ∧ σ s(u,v)=true from ⟨h.1.symm,h.2⟩)⟩
  22   loopless := ⟨by intro v h;exact G.ne_of_adj h.1 rfl⟩
