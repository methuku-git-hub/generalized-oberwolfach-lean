   6 theorem ordinary_four_cycle_triple_margin (α pmin smin ω L:ℝ)
   7     (hα:0<α) (hp:0<pmin) (hs:0<smin) (hω:0<ω) (hL:0≤L):
   8     let c:ℝ:=(pmin^3*smin^3)*(3*(min α (1/3)));
   9     ∃ β row:ℝ,0<β ∧ 0<row ∧ β≤c/4 ∧ row≤c/4 ∧
  10       β≤α*pmin^2*smin^2 ∧ β≤ω*smin^10/4 ∧ row≤smin^10/2 ∧
  11       ∃ N:ℕ,∀ n:ℕ,(N:ℝ)≤n → β*n≤c*n-row*n-3*L := by
  12   intro c
  13   have hx:0<min α (1/3:ℝ):=lt_min hα (by norm_num)
  14   have hc:0<c:=by dsimp [c];positivity
  15   let β:ℝ:=min (c/4) (min (α*pmin^2*smin^2) (ω*smin^10/4))
  16   let row:ℝ:=min (c/4) (smin^10/2)
  17   have hβ:0<β:=by
  18     dsimp [β]
  19     exact lt_min (by positivity) (lt_min (by positivity) (by positivity))
  20   have hr:0<row:=by dsimp [row]; apply lt_min <;> positivity
  21   have hβc:β≤c/4:=min_le_left _ _
  22   have hβbase:β≤α*pmin^2*smin^2:=(min_le_right _ _).trans (min_le_left _ _)
  23   have hβhub:β≤ω*smin^10/4:=(min_le_right _ _).trans (min_le_right _ _)
  24   have hrc:row≤c/4:=min_le_left _ _
  25   have hrhub:row≤smin^10/2:=min_le_right _ _
  26   obtain ⟨N,hN⟩:=exists_nat_gt (6*L/c+1)
  27   refine ⟨β,row,hβ,hr,hβc,hrc,hβbase,hβhub,hrhub,N,?_⟩
  28   intro n hn
  29   have hn0:0≤(n:ℝ):=by positivity
  30   have hnL:6*L≤c*n:=by
  31     have hh:6*L/c≤(n:ℝ):=by linarith
  32     simpa only [mul_comm] using (div_le_iff₀ hc).mp hh
  33   have hβn:=mul_le_mul_of_nonneg_right hβc hn0
  34   have hrn:=mul_le_mul_of_nonneg_right hrc hn0
  35   nlinarith only [hnL,hβn,hrn]
  36 end Oberwolfach
