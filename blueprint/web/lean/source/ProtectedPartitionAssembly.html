  14 theorem protectedPartition_from_balancing
  15     {n k q : ℕ} (G F : SimpleGraph (Fin n)) (c : Sym2 (Fin n) → Fin k)
  16     (M₀ N : Fin n → Fin n → ℕ) (m : Fin k → ℕ) (δ h f : Fin k → ℝ)
  17     {t b e : ℝ} (hn : 0 < n) (he : 0 ≤ e) (heq : e < q)
  18     (hq : ∀ a, q ≤ m a) (hte : t + b ≤ e)
  19     (hphys₀ : ∀ x y, M₀ x y + M₀ y x = if G.Adj x y then 1 else 0)
  20     (hphys : ∀ x y, N x y + N y x = if G.Adj x y then 1 else 0)
  21     (hout : ∀ x, ∑ y, N x y = ∑ a, m a)
  22     (hin : ∀ y, ∑ x, N x y = ∑ a, m a)
  23     (hnearout : ∀ a x, |((∑ y, colourRestriction c M₀ a x y : ℕ) : ℝ) - m a| ≤ t)
  24     (hnearin : ∀ a y, |((∑ x, colourRestriction c M₀ a x y : ℕ) : ℝ) - m a| ≤ t)
  25     (hchange : ∀ x, ((Finset.univ.filter (fun y => M₀ x y ≠ N x y ∨ M₀ y x ≠ N y x)).card : ℝ) ≤ b)
  26     (hforbid : ∀ a x, (sampledColourDegree F c a x : ℝ) ≤ f a)
  27     (hδ : ∀ a, 0 ≤ δ a ∧ δ a < 1 / 4)
  28     (hmargin : ∀ a, 2 * e / ((q : ℝ) - e) ≤ δ a)
  29     (hcapacity : ∀ a, (q : ℝ) + e ≤ h a * n / 2)
  30     (hloss : ∀ a, f a + b ≤ h a * n)
  31     (hrawcut : ∀ a (S T : Finset (Fin n)),
  32       (1 - 2 * δ a) * n < (S.card : ℝ) + T.card →
  33       2 * h a * n * min (S.card : ℝ) T.card ≤
  34         (matrixCut (colourRestriction c M₀ a) S T : ℝ)) :
  35     ∃ H : Fin k → SimpleGraph (Fin n),
  36       EdgePartition G H ∧ (∀ a, (H a).IsRegularOfDegree (2 * m a)) ∧
  37       (∀ a x y, F.Adj x y → ((H a).Adj x y ↔ (edgeColourGraph G c a).Adj x y)) ∧
  38       (∀ a x, ((Finset.univ.filter (fun y =>
  39         Xor ((H a).Adj x y) ((edgeColourGraph G c a).Adj x y))).card : ℝ) ≤ 4 * q + 2 * e) := by
  40   let M := colourRestriction c N
  41   let L (a : Fin k) := excludeGraph F (M a)
  42   have hN (x y : Fin n) : N x y ≤ 1 := by
  43     have hh := hphys x y
  44     split_ifs at hh <;> omega
  45   have hM₀ (x y : Fin n) : M₀ x y ≤ 1 := by
  46     have hh := hphys₀ x y
  47     split_ifs at hh <;> omega
  48   have hL (a : Fin k) (x y : Fin n) : L a x y ≤ M a x y := by
  49     dsimp [L, excludeGraph]
  50     split_ifs <;> omega
  51   have hphysical (x y : Fin n) : (∑ a, M a x y) + (∑ a, M a y x) =
  52       if G.Adj x y then 1 else 0 := by
  53     simpa only [M, colourRestriction_sum] using hphys x y
  54   have hmout (x : Fin n) : ∑ a, ∑ y, M a x y = ∑ a, m a := by
  55     rw [Finset.sum_comm]
  56     simpa only [M, colourRestriction_sum] using hout x
  57   have hmin (y : Fin n) : ∑ a, ∑ x, M a x y = ∑ a, m a := by
  58     rw [Finset.sum_comm]
  59     simpa only [M, colourRestriction_sum] using hin y
  60   have hnear (a : Fin k) (x : Fin n) :
  61       (m a : ℝ) - e ≤ ((∑ y, M a x y : ℕ) : ℝ) ∧
  62       ((∑ y, M a x y : ℕ) : ℝ) ≤ m a + e := by
  63     have h₀ := abs_le.mp (hnearout a x)
  64     have hs := abs_le.mp ((colourRestriction_degree_stability c M₀ N hM₀ hN a x).trans (hchange x))
  65     dsimp [M]
  66     constructor <;> linarith [h₀.1, h₀.2, hs.1, hs.2]
  67   have hneari (a : Fin k) (x : Fin n) :
  68       (m a : ℝ) - e ≤ ((∑ y, M a y x : ℕ) : ℝ) ∧
  69       ((∑ y, M a y x : ℕ) : ℝ) ≤ m a + e := by
  70     have h₀ := abs_le.mp (hnearin a x)
  71     have hs := abs_le.mp ((colourRestriction_indegree_stability c M₀ N hM₀ hN a x).trans (hchange x))
  72     dsimp [M]
  73     constructor <;> linarith [h₀.1, h₀.2, hs.1, hs.2]
  74   have hlarge (a : Fin k) (S T : Finset (Fin n))
  75       (hST : 1 - 2 * δ a < (S.card : ℝ) / n + (T.card : ℝ) / n) :
  76       h a * n * n * min ((S.card : ℝ) / n) ((T.card : ℝ) / n) ≤
  77         ((∑ x, ∑ y, if x ∈ S ∧ y ∈ T then L a x y else 0 : ℕ) : ℝ) := by
  78     have hnpos : (0 : ℝ) < n := by exact_mod_cast hn
  79     have hST' : (1 - 2 * δ a) * n < (S.card : ℝ) + T.card := by
  80       rw [← add_div] at hST
  81       exact (lt_div_iff₀ hnpos).mp hST
  82     have hcut := hrawcut a S T hST'
  83     have hbound := colourReorientation_cut_loss c M₀ N F hM₀ hchange hforbid a S T
  84     have hl := mul_le_mul_of_nonneg_right (hloss a)
  85       (show (0 : ℝ) ≤ min (S.card : ℝ) T.card by positivity)
  86     have heq : h a * n * n * min ((S.card : ℝ) / n) ((T.card : ℝ) / n) =
  87         h a * n * min (S.card : ℝ) T.card := by
  88       rw [min_div_div_right hnpos.le]
  89       field_simp
  90     rw [heq]
  91     have hsum : (∑ x, ∑ y, if x ∈ S ∧ y ∈ T then L a x y else 0) = matrixCut (L a) S T := by
  92       simp only [matrixCut, ite_and, Finset.sum_ite_irrel, Finset.sum_const_zero,
  93         Finset.sum_ite_mem, Finset.univ_inter]
  94     rw [hsum]
  95     change _ ≤ (matrixCut (excludeGraph F (colourRestriction c N a)) S T : ℝ)
  96     nlinarith
  97   have hzero (a : Fin k) (x y : Fin n) (hf : F.Adj x y) : L a x y = 0 ∧ L a y x = 0 := by
  98     simp [L, excludeGraph, hf, hf.symm]

 104   · intro a x
