  12 theorem residue_increment (h M k : ℕ) (hh : 0<h) :
  13     ∃ a : ℕ,a<h ∧ (k+a)%h=M%h := by
  14   let a := (M%h+h-k%h)%h
  15   refine ⟨a,Nat.mod_lt _ hh,?_⟩
  16   have hk := Nat.mod_lt k hh
  17   have he : k%h+(M%h+h-k%h)=M%h+h := by omega
  18   have hr := congrArg (fun x => x%h) he
  19   simpa only [a,Nat.add_mod,Nat.mod_mod,Nat.mod_self,add_zero] using hr

  21 theorem edgeColourGraph_card_total {V : Type*} [Fintype V] {k : ℕ}
  22     (G : SimpleGraph V) (σ : Sym2 V → Fin k) :
  23     (∑ a,(edgeColourGraph G σ a).edgeFinset.card)=G.edgeFinset.card := by
  24   simp only [edgeColourGraph_edges,Finset.card_filter]
  25   rw [Finset.sum_comm]
  26   simp

  28 theorem edge_complement_card_add {V : Type*} [Fintype V] [DecidableEq V]
  29     (G J : SimpleGraph V) (hJ : J≤G) : (G\J).edgeFinset.card+J.edgeFinset.card=G.edgeFinset.card := by
  30   have hsub : J.edgeFinset⊆G.edgeFinset := by
  31     intro e he
  32     induction e using Sym2.ind with
  33     | _ u v => exact SimpleGraph.mem_edgeFinset.mpr (hJ (SimpleGraph.mem_edgeFinset.mp he))
  34   have he : (G\J).edgeFinset=G.edgeFinset\J.edgeFinset := by
  35     ext e
  36     induction e using Sym2.ind with
  37     | _ u v => simp only [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet,SimpleGraph.sdiff_adj,Finset.mem_sdiff]
  38   rw [he]
  39   exact Finset.card_sdiff_add_card_eq_card hsub

  41 theorem all_colour_residues_of_nonreserve {V : Type*} [Fintype V] {q h : ℕ}
  42     (G : SimpleGraph V) (σ : Sym2 V → Fin (q+1)) (hhq : h∣q)
  43     (hres : ∀ a : Fin q,((edgeColourGraph G σ a.castSucc).edgeFinset.card:ZMod h)=
  44       (G.edgeFinset.card:ZMod h)) :
  45     ∀ a : Fin (q+1),((edgeColourGraph G σ a).edgeFinset.card:ZMod h)=(G.edgeFinset.card:ZMod h) := by
  46   intro a
  47   refine Fin.lastCases ?_ hres a
  48   have ht := congrArg (fun n : ℕ => (n:ZMod h)) (edgeColourGraph_card_total G σ)
  49   have hq0 : (q:ZMod h)=0 := (ZMod.natCast_eq_zero_iff q h).mpr hhq
  50   simp only [Nat.cast_sum,Nat.cast_add,Fin.sum_univ_castSucc,hres,Finset.sum_const,
  51     Finset.card_univ,Fintype.card_fin,nsmul_eq_mul,hq0,zero_mul,zero_add] at ht
  52   exact ht

  54 theorem complement_edge_divisible_of_residue {V : Type*} [Fintype V] [DecidableEq V] (G J : SimpleGraph V)
  55     (hJ : J≤G) (h : ℕ) (hres : (J.edgeFinset.card:ZMod h)=(G.edgeFinset.card:ZMod h)) :
  56     h∣(G\J).edgeFinset.card := by
  57   apply (ZMod.natCast_eq_zero_iff _ h).mp
  58   have hc := congrArg (fun n : ℕ => (n:ZMod h)) (edge_complement_card_add G J hJ)
  59   simp only [Nat.cast_add,hres] at hc
  60   exact add_right_cancel (hc.trans (zero_add _).symm)
