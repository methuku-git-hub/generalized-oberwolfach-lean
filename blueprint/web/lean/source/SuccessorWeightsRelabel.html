  14     (G : SimpleGraph V) (π : Equiv.Perm V) (ρ : Equiv.Perm W)
  15     (hc : ∀ x, f (ρ x) = π (f x)) (z : Fin c → W) :
  16     SuccessorSequenceValid (G.comap f) ρ z ↔ SuccessorSequenceValid G π (f ∘ z) := by
  17   simp only [SuccessorSequenceValid, Function.comp_apply, SimpleGraph.comap_adj]
  18   apply forall_congr'
  19   intro j
  20   split_ifs <;> simp [hc]

  22 theorem successor_sequence_weights_relabel {c : ℕ} (f : W ≃ V)
  23     (G : SimpleGraph V) (π : Equiv.Perm V) (ρ : Equiv.Perm W)
  24     (hc : ∀ x, f (ρ x) = π (f x)) (b : ℕ) (D τ A B : ℝ)
  25     (w : (Fin c → V) → ℝ)
  26     (hw : SuccessorSequenceWeightConditions G π b D τ A B w) :
  27     SuccessorSequenceWeightConditions (G.comap f) ρ b D τ A B (fun z => w (f ∘ z)) := by
  28   letI : DecidableEq W := fun x y => Classical.propDecidable (x = y)
  29   letI : DecidableEq V := fun x y => Classical.propDecidable (x = y)
  30   let e : (Fin c → W) ≃ (Fin c → V) := Equiv.piCongrRight (fun _ => f)
  31   have he (z : Fin c → W) : e z = f ∘ z := rfl
  32   have hcard : Fintype.card W = Fintype.card V := Fintype.card_congr f
  33   refine ⟨fun z => hw.nonneg (f ∘ z), ?_, ?_, ?_, ?_, ?_⟩
  34   · intro z hz
  35     obtain ⟨hv,hs⟩ := hw.support (f ∘ z) hz
  36     exact ⟨(successor_sequence_valid_relabel f G π ρ hc z).mpr hv,
  37       (permutationSeparated_embedding_iff f.toEmbedding ρ π hc b z).mp hs⟩
  38   · intro z
  39     simpa only [hcard] using hw.upper (f ∘ z)
  40   · intro z hv hs
  41     have hv' := (successor_sequence_valid_relabel f G π ρ hc z).mp hv
  42     have hs' := (permutationSeparated_embedding_iff f.toEmbedding ρ π hc b z).mpr hs
  43     simpa only [hcard] using hw.lower (f ∘ z) hv' hs'
  44   · intro j x y hxy hadj
  45     have hxy' : ¬ PermutationClose π b (f x) (f y) := by
  46       intro h
  47       exact hxy ((permutationClose_embedding_iff f.toEmbedding ρ π hc b x y).mp h)
  48     have hadj' : G.Adj (f x) (if j.val+1=c then π (f y) else f y) := by
  49       change G.Adj (f x) (f (if j.val+1=c then ρ y else y)) at hadj
  50       split_ifs at hadj ⊢ <;> simp_all [hc]
  51     have hh := hw.edges j (f x) (f y) hxy' hadj'
  52     have hsum : (∑ z : Fin c → W, if z j=x ∧ z (cycleNext j)=y then w (f ∘ z) else 0) =
  53         ∑ z : Fin c → V, if z j=f x ∧ z (cycleNext j)=f y then w z else 0 := by
  54       rw [← e.sum_comp]
  55       apply Finset.sum_congr rfl
  56       intro z _
  57       simp only [he,Function.comp_apply,f.injective.eq_iff]
  58     change (1-τ ≤ ∑ z : Fin c → W, if z j=x ∧ z (cycleNext j)=y then w (f ∘ z) else 0) ∧
  59       (∑ z : Fin c → W, if z j=x ∧ z (cycleNext j)=y then w (f ∘ z) else 0) ≤ 1
  60     rw [hsum]
  61     exact hh
  62   · intro j x
  63     have hh := hw.positions j (f x)
  64     have hsum : (∑ z : Fin c → W, if z j=x then w (f ∘ z) else 0) =
  65         ∑ z : Fin c → V, if z j=f x then w z else 0 := by
  66       rw [← e.sum_comp]
  67       apply Finset.sum_congr rfl
  68       intro z _
  69       simp only [he,Function.comp_apply,f.injective.eq_iff]
  70     change ((1-τ)*D ≤ ∑ z : Fin c → W, if z j=x then w (f ∘ z) else 0) ∧
  71       (∑ z : Fin c → W, if z j=x then w (f ∘ z) else 0) ≤ D
  72     rw [hsum]
  73     exact hh
