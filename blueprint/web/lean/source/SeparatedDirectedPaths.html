  14  theorem separated_directed_path (F:SimpleGraph V)
  15     (D:ℕ→V→V→Prop) (A:ℕ→Finset V) (W Z:Finset V)
  16     (d c:ℝ) (hd:0≤d) (hF:∀ v,(F.degree v:ℝ)≤d)
  17     (l:ℕ) (x y:V) (hxy:x≠y) (hxyF:¬F.Adj x y)
  18     (hxA:x∈A 0) (hyA:y∈A (l+2)) (hxW:x∉W) (hyW:y∉W)
  19     (hxZ:x∉Z) (hyZ:y∉Z) (hxF:∀ z∈Z,¬F.Adj x z) (hyF:∀ z∈Z,¬F.Adj y z)
  20     (hcount:∀ i<l,∀ v∈A i,c≤(((A (i+1)).filter (D i v)).card:ℝ))
  21     (hcommon:∀ v∈A l,v≠y→c≤(((A (l+1)).filter (fun z=>D l v z∧D (l+1) z y)).card:ℝ))
  22     (hbudget:W.card+((Z.card:ℝ)+l+4)*(d+1)<c):
  23     ∃ p:Fin (l+3)↪V,p 0=x∧p (Fin.last (l+2))=y∧
  24       (∀ i,p i∈A i.val∧p i∉W∧p i∉Z∧∀ z∈Z,¬F.Adj (p i) z)∧
  25       (∀ i:Fin (l+2),D i.val (p i.castSucc) (p i.succ))∧
  26       ∀ i j,¬F.Adj (p i) (p j):=by
  27   have hins:((insert y Z).card:ℝ)≤Z.card+1:=by exact_mod_cast Finset.card_insert_le y Z
  28   have hbudget':W.card+(((insert y Z).card:ℝ)+l+1)*(d+1)<c:=by
  29     have hh:=mul_le_mul_of_nonneg_right hins (show 0≤d+1 by linarith)
  30     nlinarith only [hh,hbudget,hd]
  31   obtain ⟨p,hp0,hpA,hpD,hpF⟩:=separated_directed_prefix F D A W (insert y Z) d c hd hF l x hxA hxW
  32     (by simp [hxy,hxZ]) (by intro z hz;rcases Finset.mem_insert.mp hz with rfl|hz;exact hxyF;exact hxF z hz)
  33     hcount hbudget'
  34   let S:Finset V:=Finset.univ.image p
  35   have hS:S.card=l+1:=by
  36     dsimp only [S]
  37     rw [Finset.card_image_of_injective _ p.injective,Finset.card_univ,Fintype.card_fin]
  38   have hZS:(((insert y Z)∪S).card:ℝ)≤Z.card+l+2:=by
  39     have hh:(((insert y Z)∪S).card:ℝ)≤(insert y Z).card+S.card:=by exact_mod_cast Finset.card_union_le (insert y Z) S
  40     rw [hS] at hh
  41     push_cast at hh
  42     linarith
  43   have hchoice:W.card+(((insert y Z)∪S).card:ℝ)*(d+1)<c:=by
  44     have hh:=mul_le_mul_of_nonneg_right hZS (show 0≤d+1 by linarith)
  45     nlinarith only [hh,hbudget,hd]
  46   obtain ⟨z,hz,hzW,hzZS,hzF⟩:=candidate_vertex_avoiding F
  47     ((A (l+1)).filter (fun z=>D l (p (Fin.last l)) z∧D (l+1) z y)) W ((insert y Z)∪S)
  48     d c hF (hcommon _ (hpA (Fin.last l)).1
  49       (fun he=>(hpA (Fin.last l)).2.2.1 (by simp [he]))) hchoice
  50   have hzout:z∉Set.range p:=by
  51     rintro ⟨i,hi⟩
  52     exact hzZS (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨i,Finset.mem_univ _,hi⟩))
  53   let q:Fin (l+2)↪V:=⟨Fin.snoc p z,Fin.snoc_injective_of_injective p.injective hzout⟩
  54   have hqold (i:Fin (l+1)):q i.castSucc=p i:=by simp only [q,Function.Embedding.coeFn_mk,Fin.snoc_castSucc]
  55   have hqlast:q (Fin.last (l+1))=z:=by simp only [q,Function.Embedding.coeFn_mk,Fin.snoc_last]
  56   have hzy:z≠y:=fun h=>hzZS (Finset.mem_union_left _ (by simp [h]))
  57   have hyout:y∉Set.range q:=by
  58     rintro ⟨i,hi⟩
  59     revert hi
  60     refine Fin.lastCases ?_ (fun j=>?_) i
  61     · rw [hqlast];exact hzy
  62     · rw [hqold]
  63       intro h
  64       exact (hpA j).2.2.1 (by simp [h])
  65   let t:Fin (l+3)↪V:=⟨Fin.snoc q y,Fin.snoc_injective_of_injective q.injective hyout⟩
  66   have htold (i:Fin (l+2)):t i.castSucc=q i:=by simp only [t,Function.Embedding.coeFn_mk,Fin.snoc_castSucc]
  67   have htlast:t (Fin.last (l+2))=y:=by simp only [t,Function.Embedding.coeFn_mk,Fin.snoc_last]
  68   have hqA (i:Fin (l+2)):q i∈A i.val∧q i∉W∧q i∉Z∧∀ w∈Z,¬F.Adj (q i) w:=by
  69     refine Fin.lastCases ?_ (fun j=>?_) i
  70     · rw [hqlast]
  71       exact ⟨(Finset.mem_filter.mp hz).1,hzW,
  72         fun h=>hzZS (Finset.mem_union_left _ (Finset.mem_insert_of_mem h)),
  73         fun w hw=>hzF w (Finset.mem_union_left _ (Finset.mem_insert_of_mem hw))⟩
  74     · rw [hqold]
  75       exact ⟨(hpA j).1,(hpA j).2.1,fun h=>(hpA j).2.2.1 (Finset.mem_insert_of_mem h),
  76         fun w hw=>(hpA j).2.2.2 w (Finset.mem_insert_of_mem hw)⟩
  77   have hqD (i:Fin (l+1)):D i.val (q i.castSucc) (q i.succ):=by
  78     refine Fin.lastCases ?_ (fun j=>?_) i
  79     · rw [show (Fin.last l).succ=Fin.last (l+1) from rfl,hqold,hqlast]
  80       exact (Finset.mem_filter.mp hz).2.1
  81     · rw [hqold,show j.castSucc.succ=j.succ.castSucc from rfl,hqold]
  82       exact hpD j
  83   have hqF (i j:Fin (l+2)):¬F.Adj (q i) (q j):=by
  84     refine Fin.lastCases ?_ (fun a=>?_) i
  85     · rw [hqlast]
  86       refine Fin.lastCases ?_ (fun b=>?_) j
  87       · rw [hqlast];exact F.loopless.irrefl _
  88       · rw [hqold]
  89         exact hzF _ (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨b,Finset.mem_univ _,rfl⟩))
  90     · rw [hqold]
  91       refine Fin.lastCases ?_ (fun b=>?_) j
  92       · rw [hqlast]
  93         exact fun h=>hzF _ (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨a,Finset.mem_univ _,rfl⟩)) h.symm
  94       · rw [hqold];exact hpF a b
  95   have hqyF (i:Fin (l+2)):¬F.Adj (q i) y:=by
  96     refine Fin.lastCases ?_ (fun j=>?_) i
  97     · rw [hqlast];exact hzF y (Finset.mem_union_left _ (Finset.mem_insert_self _ _))
  98     · rw [hqold];exact (hpA j).2.2.2 y (Finset.mem_insert_self _ _)
  99   refine ⟨t,(htold 0).trans ((hqold 0).trans hp0),htlast,?_,?_,?_⟩
 100   · intro i
 101     refine Fin.lastCases ?_ (fun j=>?_) i
 102     · rw [htlast];exact ⟨hyA,hyW,hyZ,hyF⟩
 103     · rw [htold];exact hqA j
 104   · intro i
 105     refine Fin.lastCases ?_ (fun j=>?_) i
 106     · rw [show (Fin.last (l+1)).succ=Fin.last (l+2) from rfl,htold,htlast,hqlast]
 107       exact (Finset.mem_filter.mp hz).2.2
 108     · rw [htold,show j.castSucc.succ=j.succ.castSucc from rfl,htold]
 109       exact hqD j
 110   · intro i j
 111     refine Fin.lastCases ?_ (fun a=>?_) i
 112     · rw [htlast]
 113       refine Fin.lastCases ?_ (fun b=>?_) j
 114       · rw [htlast];exact F.loopless.irrefl _
 115       · rw [htold];exact fun h=>hqyF b h.symm
 116     · rw [htold]
 117       refine Fin.lastCases ?_ (fun b=>?_) j
 118       · rw [htlast];exact hqyF a
 119       · rw [htold];exact hqF a b
