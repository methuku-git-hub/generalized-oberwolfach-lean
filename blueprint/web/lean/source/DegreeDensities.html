   8 theorem regularDegree_colourDensities {k D : ℕ} (m : Fin k → ℕ)
   9     (hD : 0 < D) (heven : Even D) (hsum : ∑ a, m a = D / 2) :
  10     (∀ a, 0 ≤ 2 * (m a : ℝ) / D) ∧
  11     (∑ a, 2 * (m a : ℝ) / D = 1) ∧
  12     (∀ a, (2 * (m a : ℝ) / D) / 2 * D = (m a : ℝ)) := by
  13   have hDr : (0 : ℝ) < D := by exact_mod_cast hD
  14   have hhalf : 2 * (D / 2) = D := by obtain ⟨d, hd⟩ := heven; omega
