  17 def expansionResourceMap (f : V → V') (g : U → U') :
  18     (Sym2 (ExpansionRow V E J W) ⊕ (ExpansionRow V E J W × ExpansionHub U E J)) →
  19     (Sym2 (ExpansionRow V' E J W) ⊕ (ExpansionRow V' E J W × ExpansionHub U' E J)) :=
  20   Sum.map (Sym2.map (expansionRowMap f)) (Prod.map (expansionRowMap f) (expansionHubMap g))

  22 theorem ordinaryExpansion_resources_image (C : SimpleGraph W) (z o : W) (x y : E → V)
  23     (v : J → V) (u : J → U) (f : V → V') (g : U → U') :
  24     (ordinaryHostResources (ordinaryExpansionRim C z o x y) (ordinaryExpansionSpokes z v u)).image
  25       (expansionResourceMap f g)=
  26     ordinaryHostResources (ordinaryExpansionRim C z o (f ∘ x) (f ∘ y))
  27       (ordinaryExpansionSpokes z (f ∘ v) (g ∘ u)) := by
  28   ext e
  29   constructor
  30   · intro himage
  31     obtain ⟨e,he,rfl⟩ := Finset.mem_image.mp himage
  32     cases e with
  33     | inl e =>
  34       rw [ordinaryHostResources_rim] at he
  35       induction e using Sym2.ind with
  36       | _ a b =>
  37         apply (ordinaryHostResources_rim _ _ _).mpr
  38         exact ordinaryExpansion_rim_hom C z o x y f a b he
  39     | inr p =>
  40       rcases p with ⟨a,b⟩
  41       rw [ordinaryHostResources_spoke] at he
  42       exact (ordinaryHostResources_spoke _ _ _ _).mpr (ordinaryExpansion_spoke_hom z v u f g a b he)
  43   · intro he
  44     cases e with
  45     | inl e =>
  46       rw [ordinaryHostResources_rim] at he
  47       induction e using Sym2.ind with
  48       | _ a b =>
  49         obtain ⟨a',b',hab,ha,hb⟩ := ordinaryExpansion_rim_preimage C z o x y f a b he
  50         refine Finset.mem_image.mpr ⟨.inl s(a',b'),(ordinaryHostResources_rim _ _ _).mpr hab,?_⟩
  51         change Sum.inl s(expansionRowMap f a',expansionRowMap f b')=Sum.inl s(a,b)
  52         rw [ha,hb]
  53     | inr p =>
  54       rcases p with ⟨a,b⟩
  55       rw [ordinaryHostResources_spoke] at he
  56       obtain ⟨a',b',hab,ha,hb⟩ := ordinaryExpansion_spoke_preimage z v u f g a b he
  57       refine Finset.mem_image.mpr ⟨.inr (a',b'),(ordinaryHostResources_spoke _ _ _ _).mpr hab,?_⟩
  58       change Sum.inr (expansionRowMap f a',expansionHubMap g b')=Sum.inr (a,b)
  59       rw [ha,hb]

  61 theorem ordinaryExpansion_resources_injective (C : SimpleGraph W) (z o : W) (hzo : z≠o)
  62     (x y : E → V) (v : J → V) (u : J → U) (f : V → V') (g : U → U') :
  63     Set.InjOn (expansionResourceMap (E:=E) (J:=J) (W:=W) f g)
  64       (ordinaryHostResources (ordinaryExpansionRim C z o x y) (ordinaryExpansionSpokes z v u)) := by
  65   intro e he e' he' hh
  66   cases e with
  67   | inl e =>
  68     cases e' with
  69     | inr p => cases hh
  70     | inl e' =>
  71       rw [Finset.mem_coe,ordinaryHostResources_rim] at he he'
  72       exact congrArg Sum.inl (ordinaryExpansion_rim_edge_injective C z o hzo x y f he he' (Sum.inl.inj hh))
  73   | inr p =>
  74     cases e' with
  75     | inl e' => cases hh
  76     | inr q =>
  77       rcases p with ⟨a,b⟩;rcases q with ⟨c,d⟩
  78       rw [Finset.mem_coe,ordinaryHostResources_spoke] at he he'
  79       exact congrArg Sum.inr (ordinaryExpansion_spoke_injective z v u f g he he' (Sum.inr.inj hh))
