   8 theorem reserved_block_surplus_of_cutoff
   9     (n ell t K theta err : ℝ)
  10     (hn : 0 ≤ n) (ht : 0 < t) (hK : 0 < K) (hθ : 0 < theta)
  11     (hell : (1:ℝ)/2 ≤ ell)
  12     (hcut : 5/(t+5) + t/K ≤ theta/2)
  13     (herr : err < theta*n/(4*t)) :
  14     (1-theta)*ell*n/t + err < ell*n/(t+5) - ell*n/K := by
  15   have hgap : theta/2 ≤ theta - 5/(t+5) - t/K := by linarith
  16   have hbase : n/(2*t) ≤ ell*n/t := by
  17     have hmul : n/2 ≤ ell*n := by nlinarith
  18     have hd := div_le_div_of_nonneg_right hmul (le_of_lt ht)
  19     convert hd using 1 <;> ring
  20   have hprod : theta*n/(4*t) ≤ (ell*n/t)*(theta - 5/(t+5) - t/K) := by
  21     calc
  22       theta*n/(4*t) = (n/(2*t))*(theta/2) := by ring
  23       _ ≤ (ell*n/t)*(theta/2) := mul_le_mul_of_nonneg_right hbase (by positivity)
  24       _ ≤ (ell*n/t)*(theta - 5/(t+5) - t/K) :=
  25         mul_le_mul_of_nonneg_left hgap (by positivity)
  26   have hid : ell*n/(t+5) - ell*n/K - (1-theta)*ell*n/t =
  27       (ell*n/t)*(theta - 5/(t+5) - t/K) := by
  28     field_simp
  29     <;> ring
  30   linarith
  31 end Oberwolfach
