  11 theorem assigned_edges_avoid_old_images (G:SimpleGraph V)
  12     (f:{e:Sym2 V // e∈G.edgeFinset}→W) (orient:Sym2 V→V×V)
  13     (horient:∀e∈G.edgeFinset,prescribedArcPhysical (orient e)=e)
  14     (B:V→Finset W) (old:W→Finset V)
  15     (hOld:∀v i,v∈old i→i∈B v)
  16     (hB:∀e,∀v∈e.1.toFinset,f e∉B v):
  17     ∀i,Disjoint (old i) (prescribedMatchingEndpoints (prescribedAssignedArcs G (extendEdgeAssignment G f) orient i)) := by
  18   intro i
  19   apply Finset.disjoint_left.mpr
  20   intro v hvOld hvEnd
  21   obtain ⟨a,ha,hva⟩:=Finset.mem_biUnion.mp hvEnd
  22   obtain ⟨e,he,heq⟩:=Finset.mem_image.mp ha
  23   have heK:e∈G.edgeFinset:=(Finset.mem_filter.mp he).1
  24   have hei:extendEdgeAssignment G f e=i:=(Finset.mem_filter.mp he).2
  25   have hve:v∈e.toFinset:=by
  26     rw [←horient e heK]
  27     simp only [prescribedArcPhysical,Sym2.toFinset_mk_eq,Finset.mem_insert,Finset.mem_singleton]
  28     simpa [heq] using hva
  29   have hf:f ⟨e,heK⟩=i:=by simpa [extend_edge_assignment_value G f e heK] using hei
  30   have hb:=hB ⟨e,heK⟩ v hve
  31   exact hb (by rw [hf];exact hOld v i hvOld)

  33 theorem assigned_edges_avoid_inner_indices (G:SimpleGraph V)
  34     (f:{e:Sym2 V // e∈G.edgeFinset}→W) (orient:Sym2 V→V×V)
  35     (L:Sym2 V→Finset W) (inner:Finset W)
  36     (hAllowed:∀e,f e∈L e.1) (hOutside:∀e∈G.edgeFinset,Disjoint (L e) inner):
  37     ∀i,i∈inner→prescribedAssignedArcs G (extendEdgeAssignment G f) orient i=∅ := by
  38   intro i hi
  39   apply Finset.eq_empty_iff_forall_notMem.mpr
  40   intro a ha
  41   obtain ⟨e,he,heq⟩:=Finset.mem_image.mp ha
  42   have heK:e∈G.edgeFinset:=(Finset.mem_filter.mp he).1
  43   have hei:extendEdgeAssignment G f e=i:=(Finset.mem_filter.mp he).2
  44   have hf:f ⟨e,heK⟩=i:=by simpa [extend_edge_assignment_value G f e heK] using hei
  45   have hL:f ⟨e,heK⟩∈L e:=hAllowed ⟨e,heK⟩
  46   rw [hf] at hL
  47   exact (Finset.disjoint_left.mp (hOutside e heK)) hL hi
