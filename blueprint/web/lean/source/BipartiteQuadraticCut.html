  12 def bipartiteMatrixCut {A B : Type*} (M : A → B → ℕ) (S : Finset A) (T : Finset B) : ℕ :=
  13   ∑ x∈S,∑ y∈T,M x y

  15 theorem bipartiteMatrixCut_transpose {A B : Type*} (M : A → B → ℕ) (S : Finset A) (T : Finset B) :
  16     bipartiteMatrixCut M S T=bipartiteMatrixCut (fun y x=>M x y) T S := Finset.sum_comm

  18 theorem bipartiteMatrixCut_row_lower {A B : Type*} [Fintype B]
  19     (M : A → B → ℕ) (hM : ∀ x y,M x y≤1) (S : Finset A) (T : Finset B) (d : ℝ)
  20     (hrow : ∀ x∈S,d≤((∑ y,M x y:ℕ):ℝ)) :
  21     (S.card:ℝ)*(d+T.card-Fintype.card B)≤(bipartiteMatrixCut M S T:ℝ) := by
  22   have hpoint (x : A) (hx : x∈S) : d+T.card-Fintype.card B≤((∑ y∈T,M x y:ℕ):ℝ) := by
  23     have hc : ∑ y∈Finset.univ\T,M x y≤Fintype.card B-T.card := by
  24       calc
  25         _ ≤ ∑ _y∈Finset.univ\T,1 := Finset.sum_le_sum (fun y _=>hM x y)
  26         _ = _ := by simp [Finset.card_sdiff_of_subset (Finset.subset_univ T)]
  27     have hs := Finset.sum_sdiff (f:=fun y=>M x y) (Finset.subset_univ T)
  28     have ht := Finset.card_le_univ T
  29     have hcr : ((∑ y∈Finset.univ\T,M x y:ℕ):ℝ)≤(Fintype.card B:ℝ)-T.card := by exact_mod_cast hc
  30     have hsr : ((∑ y∈Finset.univ\T,M x y:ℕ):ℝ)+((∑ y∈T,M x y:ℕ):ℝ)=((∑ y,M x y:ℕ):ℝ) := by exact_mod_cast hs
  31     linarith only [hcr,hsr,hrow x hx]
  32   simpa only [bipartiteMatrixCut,Nat.cast_sum,Finset.sum_const,nsmul_eq_mul] using Finset.sum_le_sum hpoint

  34 theorem bipartite_large_cut_numeric
  35     (a b s t E p δ ε : ℝ) (ha : 0<a) (hb : 0<b) (hs : 0≤s) (ht : 0≤t)
  36     (hp : 0<p) (hp1 : p≤1) (hδ : δ≤p/16) (hε : ε≤p^2/128)
  37     (hlarge : (1-2*δ)*a*b<b*s+a*t)
  38     (hrow : s*(p*b/2+t-b)≤E) (hcol : t*(p*a/2+s-a)≤E)
  39     (hquad : p*s*t-ε*a*b≤E) :
  40     p/8*min (b*s) (a*t)≤E := by
  41   have hcase (a b s t : ℝ) (ha : 0<a) (hb : 0<b) (hs : 0≤s)
  42       (hlarge : (1-2*δ)*a*b<b*s+a*t) (hle : b*s≤a*t)
  43       (hrow : s*(p*b/2+t-b)≤E) (hquad : p*s*t-ε*a*b≤E) : p/8*(b*s)≤E := by
  44     have hab : 0<a*b := mul_pos ha hb
  45     have hδab := mul_le_mul_of_nonneg_right hδ hab.le
  46     have hεab := mul_le_mul_of_nonneg_right hε hab.le
  47     by_cases hsmall : s≤p*a/8
  48     · have hsb := mul_le_mul_of_nonneg_right hsmall hb.le
  49       have htlarge : p*b/4≤p*b/2+t-b := by
  50         apply (mul_le_mul_iff_right₀ ha).mp
  51         nlinarith only [hlarge,hδab,hsb]
  52       have hh := mul_le_mul_of_nonneg_left htlarge hs
  53       have hpos : 0≤p*b*s := mul_nonneg (mul_nonneg hp.le hb.le) hs
  54       nlinarith only [hh,hrow,hpos]
  55     · have hslarge : p*a/8≤s := (lt_of_not_ge hsmall).le
  56       have htlarge : b/4≤t := by
  57         apply (mul_le_mul_iff_right₀ ha).mp
  58         have hpab := mul_le_mul_of_nonneg_right hp1 hab.le
  59         nlinarith only [hlarge,hle,hδab,hpab,hab]
  60       have hmean := mul_le_mul_of_nonneg_left htlarge (mul_nonneg hp.le hs)
  61       have herror := mul_le_mul_of_nonneg_right hslarge (mul_nonneg hp.le hb.le)
  62       have hpos : 0≤p*b*s := mul_nonneg (mul_nonneg hp.le hb.le) hs
  63       nlinarith only [hmean,herror,hεab,hquad,hpos]
  64   rcases le_total (b*s) (a*t) with hle|hle
  65   · rw [min_eq_left hle]
  66     exact hcase a b s t ha hb hs hlarge hle hrow hquad
  67   · rw [min_eq_right hle]
  68     exact hcase b a t s hb ha ht (by nlinarith only [hlarge]) hle hcol (by nlinarith only [hquad])

  70 theorem bipartiteMatrixCut_large_of_degree_and_quadratic
  71     {A B : Type*} [Fintype A] [Fintype B]
  72     (M : A → B → ℕ) (hM : ∀ x y,M x y≤1)
  73     (ha : 0<Fintype.card A) (hb : 0<Fintype.card B)
  74     (p ρ δ ε : ℝ) (hp : 0<p) (hp1 : p≤1) (hρ : p≤ρ)
  75     (hδ : δ≤p/16) (hε : ε≤p^2/128)
  76     (hrow : ∀ x,p*(Fintype.card B:ℝ)/2≤((∑ y,M x y:ℕ):ℝ))
  77     (hcol : ∀ y,p*(Fintype.card A:ℝ)/2≤((∑ x,M x y:ℕ):ℝ))
  78     (hcut : ∀ S T,|(bipartiteMatrixCut M S T:ℝ)-ρ*S.card*T.card|≤ε*Fintype.card A*Fintype.card B)
  79     (S : Finset A) (T : Finset B)
  80     (hlarge : (1-2*δ)*(Fintype.card A:ℝ)*Fintype.card B<
  81       (Fintype.card B:ℝ)*S.card+(Fintype.card A:ℝ)*T.card) :
  82     p/8*min ((Fintype.card B:ℝ)*S.card) ((Fintype.card A:ℝ)*T.card)≤(bipartiteMatrixCut M S T:ℝ) := by
  83   have hR := bipartiteMatrixCut_row_lower M hM S T (p*(Fintype.card B:ℝ)/2) (fun x _=>hrow x)
  84   have hC := bipartiteMatrixCut_row_lower (fun y x=>M x y) (fun y x=>hM x y) T S
  85     (p*(Fintype.card A:ℝ)/2) (fun y _=>hcol y)
  86   rw [← bipartiteMatrixCut_transpose] at hC
  87   have hQ : p*S.card*T.card-ε*Fintype.card A*Fintype.card B≤(bipartiteMatrixCut M S T:ℝ) := by
  88     have hh := (abs_le.mp (hcut S T)).1
  89     have hmul := mul_le_mul_of_nonneg_right hρ (by positivity : (0:ℝ)≤(S.card:ℝ)*T.card)
  90     nlinarith only [hh,hmul]
  91   exact bipartite_large_cut_numeric (Fintype.card A) (Fintype.card B) S.card T.card
  92     (bipartiteMatrixCut M S T) p δ ε (by exact_mod_cast ha) (by exact_mod_cast hb)
  93     (by positivity) (by positivity) hp hp1 hδ hε hlarge hR hC hQ
