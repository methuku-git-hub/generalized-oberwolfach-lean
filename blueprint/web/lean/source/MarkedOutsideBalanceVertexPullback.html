  15 theorem marked_outside_balance_vertex_pullback
  16     (tail head:E→V) (ktail khead:K→V) (HK:V→U→Prop) (HK':V'→U'→Prop)
  17     (f:V↪V') (g:U↪U') (A:Finset (E⊕K)) (S:Finset (V×U)) (m:ℕ)
  18     (X:Finset V) (Z:Finset U) (I:Finset V') (J:Finset U')
  19     (hX:∀ x,f x∈I↔x∈X) (hZ:∀ u,g u∈J↔u∈Z)
  20     (hHK:∀ z∈S,HK' (f z.1) (g z.2)↔HK z.1 z.2)
  21     (h:MarkedOutsideBalance (f∘tail) (f∘head) (f∘ktail) (f∘khead) HK'
  22       A (S.image (Prod.map f g)) m I J):
  23     MarkedOutsideBalance tail head ktail khead HK A S m X Z:=by
  24   have hi:Function.Injective (Prod.map f g):=by
  25     intro a b hab
  26     exact Prod.ext (f.injective (congrArg Prod.fst hab)) (g.injective (congrArg Prod.snd hab))
  27   have ht (a:E⊕K):Sum.elim (f∘tail) (f∘ktail) a=f (Sum.elim tail ktail a):=by cases a <;> rfl
  28   have hh (a:E⊕K):Sum.elim (f∘head) (f∘khead) a=f (Sum.elim head khead a):=by cases a <;> rfl
  29   have hs (p:V'×U'→Prop):((S.image (Prod.map f g)).filter p).card=
  30       (S.filter (fun z=>p (f z.1,g z.2))).card:=by
  31     rw [Finset.filter_image,Finset.card_image_of_injective _ hi]
  32     rfl
  33   have htype (p:V×U→Prop) (b:Bool):
  34       S.filter (fun z=>p z∧(if b then HK' (f z.1) (g z.2) else ¬HK' (f z.1) (g z.2)))=
  35       S.filter (fun z=>p z∧(if b then HK z.1 z.2 else ¬HK z.1 z.2)):=by
  36     ext z
  37     simp only [Finset.mem_filter]
  38     apply and_congr_right
  39     intro hz
  40     rw [hHK z hz]
  41   simp only [←Set.ncard_coe_finset,Finset.coe_filter,Finset.coe_image] at hs
  42   constructor
  43   · intro v hv
  44     have hd:=h.directed (f v) (fun hv'=>hv ((hX v).mp hv'))
  45     simpa only [hh,ht,f.injective.eq_iff] using hd
  46   · intro v hv
  47     have hd:=h.row (f v) (fun hv'=>hv ((hX v).mp hv'))
  48     simp only [←Set.ncard_coe_finset,Finset.coe_filter,Finset.coe_image] at hd ⊢
  49     rw [hs] at hd
  50     simpa only [hh,f.injective.eq_iff] using hd
  51   · intro v hv
  52     have hd:=h.marked (f v) (fun hv'=>hv ((hX v).mp hv'))
  53     simp only [←Set.ncard_coe_finset,Finset.coe_filter,Finset.coe_image] at hd ⊢
  54     rw [hs] at hd
  55     simp only [hh,f.injective.eq_iff] at hd
  56     have he:=congrArg (fun T:Finset (V×U)=>(T:Set (V×U))) (htype (fun z=>z.1=v) true)
  57     simp only [Finset.coe_filter,ite_true] at he
  58     rw [he] at hd
  59     exact hd
  60   · intro u hu
  61     have hd:=h.hub (g u) (fun hu'=>hu ((hZ u).mp hu'))
  62     simp only [←Set.ncard_coe_finset,Finset.coe_filter,Finset.coe_image] at hd ⊢
  63     rw [hs,hs] at hd
  64     simp only [g.injective.eq_iff] at hd
  65     have he0:=congrArg (fun T:Finset (V×U)=>(T:Set (V×U))) (htype (fun z=>z.2=u) false)
  66     have he1:=congrArg (fun T:Finset (V×U)=>(T:Set (V×U))) (htype (fun z=>z.2=u) true)
  67     simp only [Finset.coe_filter,Bool.false_eq_true,ite_false] at he0
  68     simp only [Finset.coe_filter,ite_true] at he1
  69     rw [he0,he1] at hd
  70     exact hd
