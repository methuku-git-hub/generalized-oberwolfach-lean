  14 theorem minimum_degree_three_partition (α : ℝ) (hα : 0<α) :
  15     ∃ N : ℕ,∀ (V : Type) [Fintype V],N≤Fintype.card V → 3∣Fintype.card V →
  16       ∀ G : SimpleGraph V,(∀ v,(3/4+α)*Fintype.card V≤(G.degree v:ℝ)) →
  17       ∃ m : ℕ,∃ P : FixedPartition V 3 (fun _=>m),
  18         Fintype.card V=3*m ∧ ∀ v i,(3/4:ℝ)*m≤graphCellDegree G (Equiv.refl V) P v i := by
  19   let η := α/2
  20   have hη : 0<η := by dsimp [η];positivity
  21   obtain ⟨N,hN⟩ := partition_failure_eventually_small 3 (by norm_num) η hη
  22   refine ⟨max N 2,?_⟩
  23   intro V _ hn hdiv G hdeg
  24   have hn2 : 2≤Fintype.card V := (le_max_right N 2).trans hn
  25   have hnpos : (0:ℝ)<Fintype.card V := by exact_mod_cast (show 0<Fintype.card V by omega)
  26   letI : Nonempty V := Fintype.card_pos_iff.mp (by omega)
  27   obtain ⟨m,hnm⟩ := hdiv
  28   have hm : 0<m := by omega
  29   have hmR : (0:ℝ)<m := Nat.cast_pos.mpr hm
  30   have hnmR : (Fintype.card V:ℝ)=3*m := by exact_mod_cast hnm
  31   letI : Nonempty (FixedPartition V 3 (fun _=>m)) := fixedPartition_nonempty _ (by simpa using hnm.symm)
  32   let μ := FiniteLaw.uniform (FixedPartition V 3 (fun _=>m))
  33   let Bad := fun z : Fin 3×V=>fun P : FixedPartition V 3 (fun _=>m)=>
  34     η*m≤|(graphCellDegree G (Equiv.refl V) P z.2 z.1:ℝ)-(G.degree z.2)*(m/(Fintype.card V:ℝ))|
  35   let p := 2*Real.exp (-η^2*m/2)
  36   have htail (z : Fin 3×V) : μ.probability (Bad z)≤p := by
  37     exact graphPartition_degree_tail G (Equiv.refl V) z.2 z.1 m hm rfl _ 0 η hη
  38       (by positivity) (by simp)
  39   have hsum : ∑ _z : Fin 3×V,p<1 := by
  40     have hf := (hN _ ((le_max_left N 2).trans hn)).2.2
  41     have hd : Fintype.card V/3=m := by rw [hnm];omega
  42     rw [partitionFailure,hd] at hf
  43     have hinv : 1/(Fintype.card V:ℝ)<1 := (div_lt_one hnpos).mpr (by exact_mod_cast hn2)
  44     simp only [Finset.sum_const,Finset.card_univ,Fintype.card_prod,Fintype.card_fin,Nat.cast_mul,Nat.cast_ofNat,nsmul_eq_mul]
  45     dsimp [p]
  46     have he : 0≤Real.exp (-η^2*m/2) := (Real.exp_pos _).le
  47     nlinarith only [hf,hinv,mul_nonneg (Nat.cast_nonneg (Fintype.card V)) he]
  48   obtain ⟨P,_,hP⟩ := finiteLaw_common_event_exists μ Finset.univ Bad (fun _=>p)
  49     (fun z _=>htail z) hsum
  50   refine ⟨m,P,hnm,?_⟩
  51   intro v i
  52   have hg := lt_of_not_ge (hP (i,v) (Finset.mem_univ _))
  53   change |(graphCellDegree G (Equiv.refl V) P v i:ℝ)-(G.degree v)*(m/(Fintype.card V:ℝ))|<η*m at hg
  54   have hmean : (3/4+α)*m≤(G.degree v)*(m/(Fintype.card V:ℝ)) := by
  55     have hh := mul_le_mul_of_nonneg_right (hdeg v) (by positivity : (0:ℝ)≤m/(Fintype.card V:ℝ))
  56     have he : ((3/4+α)*Fintype.card V)*(m/(Fintype.card V:ℝ))=(3/4+α)*m := by field_simp
  57     rwa [he] at hh
  58   have hl := (abs_lt.mp hg).1
  59   dsimp [η] at hl
  60   nlinarith only [hl,hmean,mul_pos hα hmR]
  61 
  62 end Oberwolfach
