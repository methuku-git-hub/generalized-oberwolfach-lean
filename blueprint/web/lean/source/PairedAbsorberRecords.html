  17 abbrev PairedCycleRecord (R : C → Finset V) (X : Finset V) :=
  18   Σ f : RootedLabelMap V (c*c) c X,
  19     {C₀ : C // R C₀=(pairedBasePositions (c:=c)).image f.val} ×
  20     (∀ i : Fin c,{C₀ : C // R C₀=(pairedZeroPositions i).image f.val}) ×
  21     (∀ i : Fin c,{C₀ : C // R C₀=(pairedOnePositions i).image f.val})

  23 def pairedCycleRecordToLabelled (R : C → Finset V) (X : Finset V)
  24     (D : PairedCycleRecord (c:=c) R X) : LabelledCycleRecord R (pairedAbsorberTemplate c).val X := by
  25   refine ⟨D.1,?_⟩
  26   intro o
  27   rcases o with i | j
  28   · exact D.2.2.1 i
  29   · refine Fin.cases ?_ (fun i => ?_) j
  30     · exact D.2.1
  31     · exact D.2.2.2 i

  48 theorem pairedCycleRecordToLabelled_injective (R : C → Finset V) (X : Finset V) :
  49     Function.Injective (pairedCycleRecordToLabelled (c:=c) R X) := by
  50   intro D E he
  51   have hf : D.1=E.1 := congrArg (fun F : LabelledCycleRecord R (pairedAbsorberTemplate c).val X => F.1) he
  52   have hbase : D.2.1.val=E.2.1.val :=
  53     congrArg (fun F : LabelledCycleRecord R (pairedAbsorberTemplate c).val X => (F.2 (Sum.inr 0)).val) he
  54   have hz (i : Fin c) : (D.2.2.1 i).val=(E.2.2.1 i).val :=
  55     congrArg (fun F : LabelledCycleRecord R (pairedAbsorberTemplate c).val X => (F.2 (Sum.inl i)).val) he
  56   have ho (i : Fin c) : (D.2.2.2 i).val=(E.2.2.2 i).val :=
  57     congrArg (fun F : LabelledCycleRecord R (pairedAbsorberTemplate c).val X => (F.2 (Sum.inr i.succ)).val) he
  58   rcases D with ⟨f,d₀,dz,do'⟩
  59   rcases E with ⟨g,e₀,ez,eo⟩
  60   change f=g at hf
  61   subst g
  62   have hb : d₀=e₀ := Subtype.ext hbase
  63   have hzz : dz=ez := funext (fun i => Subtype.ext (hz i))
  64   have hoo : do'=eo := funext (fun i => Subtype.ext (ho i))
  65   subst e₀;subst ez;subst eo;rfl

  67 theorem paired_absorber_template_count (R : C → Finset V) (X : Finset V) :
  68     Fintype.card (PairedCycleRecord (c:=c) R X) ≤
  69       Fintype.card (LabelledCycleRecord R (pairedAbsorberTemplate c).val X) :=
