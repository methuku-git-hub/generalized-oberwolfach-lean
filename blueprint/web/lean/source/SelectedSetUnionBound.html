  12 theorem selected_set_probability_sum {V Ω : Type*} [Fintype V] [Fintype Ω]
  13     (μ : FiniteLaw Ω) (S : Ω → Finset V) (m : ℕ) (hm : ∀ ω,(S ω).card=m)
  14     (E : Ω → Prop) :
  15     (∑ x,μ.probability (fun ω => E ω ∧ x∈S ω))=(m:ℝ)*μ.probability E := by
  16   unfold FiniteLaw.probability
  17   rw [Finset.sum_comm,Finset.mul_sum]

  23       simp
  24     rw [he,hm]

  27 theorem selected_set_union_bound {V Ω : Type*} [Fintype V] [Fintype Ω]
  28     (μ : FiniteLaw Ω) (S : Ω → Finset V) (m : ℕ) (hm : ∀ ω,(S ω).card=m)
  29     (E : Ω → Prop) (Bad : V → Ω → Prop) (ε : ℝ)
  30     (hbad : ∀ x,μ.probability (fun ω => E ω ∧ x∈S ω ∧ Bad x ω)≤
  31       ε*μ.probability (fun ω => E ω ∧ x∈S ω)) :
  32     μ.probability (fun ω => E ω ∧ ∃ x∈S ω,Bad x ω)≤
  33       ε*m*μ.probability E := by
  34   have hu : μ.probability (fun ω => E ω ∧ ∃ x∈S ω,Bad x ω)≤
  35       ∑ x,μ.probability (fun ω => E ω ∧ x∈S ω ∧ Bad x ω) := by
  36     unfold FiniteLaw.probability
  37     rw [Finset.sum_comm]
  38     apply Finset.sum_le_sum
  39     intro ω _
  40     by_cases hb : E ω ∧ ∃ x∈S ω,Bad x ω
  41     · obtain ⟨hE,x,hx,hB⟩ := hb
  42       rw [if_pos ⟨hE,x,hx,hB⟩]
  43       have hh := Finset.single_le_sum (s:=Finset.univ)
  44         (f:=fun x => if E ω ∧ x∈S ω ∧ Bad x ω then μ.mass ω else 0)
  45         (fun x _ => by split_ifs;exact μ.nonneg ω;exact le_refl _) (Finset.mem_univ x)
  46       have hhE : E ω ∧ x∈S ω ∧ Bad x ω := ⟨hE,hx,hB⟩
  47       rw [if_pos hhE] at hh
  48       convert hh using 1
  49       apply Finset.sum_congr rfl
  50       intro y _
  51       split_ifs <;> rfl
  52     · rw [if_neg hb]
  53       exact Finset.sum_nonneg (fun x _ => by split_ifs;exact μ.nonneg ω;exact le_refl _)
  54   calc
  55     _ ≤ ∑ x,μ.probability (fun ω => E ω ∧ x∈S ω ∧ Bad x ω) := hu
  56     _ ≤ ∑ x,ε*μ.probability (fun ω => E ω ∧ x∈S ω) :=
  57       Finset.sum_le_sum (fun x _ => hbad x)
  58     _ = _ := by rw [← Finset.mul_sum,selected_set_probability_sum μ S m hm E];ring
