  11 theorem transition_two_step_support_lower {X : Type*} [Fintype X]
  12     (R S : X → X → ℝ) (A : X → X → Prop) [DecidableRel A]
  13     (a b : ℝ) (ha : 0≤a) (hb : 0≤b)
  14     (hR : ∀ x y,0≤R x y) (hS : ∀ x y,0≤S x y)
  15     (hRa : ∀ x y,A x y → a≤R x y) (hSb : ∀ x y,A x y → b≤S x y)
  16     (x y : X) :
  17     ((Finset.univ.filter (fun z => A x z ∧ A z y)).card : ℝ)*(a*b) ≤
  18       transitionCompose R S x y := by
  19   classical
  20   have he : ((Finset.univ.filter (fun z => A x z ∧ A z y)).card : ℝ)*(a*b)=
  21       ∑ z,if A x z ∧ A z y then a*b else 0 := by
  22     rw [← Finset.sum_filter]
  23     simp
  24   rw [he]
  25   apply Finset.sum_le_sum
  26   intro z _
  27   split_ifs with hz
  28   · exact mul_le_mul (hRa x z hz.1) (hSb z y hz.2) hb (hR x z)
  29   · exact mul_nonneg (hR x z) (hS z y)

  31 theorem transition_three_step_offdiagonal {X : Type*} [Fintype X]
  32     (R S T : X → X → ℝ) (hR : IsTransition R)
  33     (hS : ∀ x y,0≤S x y) (hT : ∀ x y,0≤T x y)
  34     (τ : ℝ) (hlower : ∀ z y,z≠y → τ≤transitionCompose S T z y) (x y : X) :
  35     τ*(1-R x y) ≤ transitionCompose R (transitionCompose S T) x y := by
  36   classical
  37   have hnonneg (z : X) : 0≤transitionCompose S T z y :=
  38     Finset.sum_nonneg fun u _ => mul_nonneg (hS z u) (hT u y)
  39   have he : (∑ z,if z=y then (0:ℝ) else R x z)=1-R x y := by
  40     have hi (z : X) : (if z=y then (0:ℝ) else R x z)=R x z-(if z=y then R x z else 0) := by
  41       split_ifs <;> ring
  42     simp only [hi,Finset.sum_sub_distrib,hR.2,Finset.sum_ite_eq',Finset.mem_univ,ite_true]
  43   calc
  44     τ*(1-R x y) = ∑ z,if z=y then 0 else R x z*τ := by
  45       rw [← he,Finset.mul_sum]
  46       apply Finset.sum_congr rfl
  47       intro z _
  48       split_ifs <;> ring
  49     _ ≤ _ := by
  50       apply Finset.sum_le_sum
  51       intro z _
  52       split_ifs with hz
  53       · exact mul_nonneg (hR.1 x z) (hnonneg z)
  54       · exact mul_le_mul_of_nonneg_left (hlower z y hz) (hR.1 x z)

  56 theorem ordinary_three_step_minorization {X : Type*} [Fintype X]
  57     (R S T : X → X → ℝ) (A : X → X → Prop) [DecidableRel A]
  58     (μ : X → ℝ) (α p p₀ n : ℝ)
  59     (hα : 0<α) (hp₀ : 0<p₀) (hp : p₀≤p) (hn : 0<n) (hsize : 10/p₀≤n)
  60     (hR : IsTransition R) (hS : ∀ x y,0≤S x y) (hT : ∀ x y,0≤T x y)
  61     (hSup : ∀ x y,A x y → 1/(6*p*n)≤S x y)
  62     (hTup : ∀ x y,A x y → 1/(6*p*n)≤T x y)
  63     (hRu : ∀ x y,R x y≤5/(p*n))
  64     (hcommon : ∀ x y,x≠y → α*p^2*n ≤
  65       ((Finset.univ.filter (fun z => A x z ∧ A z y)).card : ℝ))
  66     (hμ : ∀ y,μ y≤2/n) (x y : X) :
  67     (α/200)*μ y ≤ transitionCompose R (transitionCompose S T) x y := by
  68   have hpp : 0<p := hp₀.trans_le hp
  69   have ht (z y : X) (hzy : z≠y) : α/(36*n)≤transitionCompose S T z y := by
  70     have hh := transition_two_step_support_lower S T A (1/(6*p*n)) (1/(6*p*n))
  71       (by positivity) (by positivity) hS hT hSup hTup z y
  72     have hm := mul_le_mul_of_nonneg_right (hcommon z y hzy)
  73       (show 0≤(1/(6*p*n))*(1/(6*p*n)) by positivity)
  74     have he : (α*p^2*n)*((1/(6*p*n))*(1/(6*p*n)))=α/(36*n) := by
  75       field_simp
  76       <;> ring
  77     rw [he] at hm
  78     exact hm.trans hh
  79   have hsize' : 10≤p*n := by
  80     have hh : 10≤n*p₀ := (div_le_iff₀ hp₀).mp hsize
  81     nlinarith [mul_le_mul_of_nonneg_right hp hn.le]
  82   have hhalf : R x y≤1/2 := by
  83     apply (hRu x y).trans
  84     apply (div_le_iff₀ (mul_pos hpp hn)).mpr
  85     linarith
  86   have hm := transition_three_step_offdiagonal R S T hR hS hT (α/(36*n)) ht x y
  87   have hbase : α/(72*n)≤(α/(36*n))*(1-R x y) := by
  88     have hh := mul_le_mul_of_nonneg_left hhalf (show 0≤α/(36*n) by positivity)
  89     have hid : α/(36*n)/2=α/(72*n) := by ring
  90     linarith
  91   have hu := mul_le_mul_of_nonneg_left (hμ y) (show 0≤α/200 by positivity)
  92   have hnum : (α/200)*(2/n)≤α/(72*n) := by
  93     field_simp
  94     <;> nlinarith
  95   exact (hu.trans hnum).trans (hbase.trans hm)
