  15 theorem simpleGraph_edges_le_square (G : SimpleGraph V) : G.edgeFinset.card≤(Fintype.card V)^2 := by
  16   rw [←canonicalGraphPairs_card G]
  17   have hh := Finset.card_le_univ (canonicalGraphPairs G)
  18   simpa only [Fintype.card_prod,pow_two] using hh

  20 theorem actual_expansion_order_bound (G : SimpleGraph V) (H : V → U → Prop) (r m : ℕ)
  21     (hm : Fintype.card V+Fintype.card U≤m) :
  22     Fintype.card (ActualExpansionRow G H (Fin r))+Fintype.card (ActualExpansionHub G H)≤
  23       m+2*m^2*(r+1) := by
  24   have hV : Fintype.card V≤m := by omega
  25   have hU : Fintype.card U≤m := by omega
  26   have he : G.edgeFinset.card≤m^2 :=
  27     (simpleGraph_edges_le_square G).trans (Nat.pow_le_pow_left hV 2)
  28   have hs : Fintype.card (ActualSpoke H)≤m^2 := by
  29     have hh := Fintype.card_subtype_le (fun e:V×U=>H e.1 e.2)
  30     simp only [Fintype.card_prod] at hh
  31     exact hh.trans (by simpa only [pow_two] using Nat.mul_le_mul hV hU)
  32   have ho := ordinaryExpansion_order (V:=V) (U:=U) (E:=G.edgeFinset) (J:=ActualSpoke H) (W:=Fin r)
  33   simp only [Fintype.card_coe,Fintype.card_fin] at ho
  34   change Fintype.card (ExpansionRow V G.edgeFinset (ActualSpoke H) (Fin r))+
  35     Fintype.card (ExpansionHub U G.edgeFinset (ActualSpoke H))≤_
  36   rw [ho]
  37   have hsum : G.edgeFinset.card+Fintype.card (ActualSpoke H)≤2*m^2 := by omega
  38   exact Nat.add_le_add hm (Nat.mul_le_mul_right (r+1) hsum)

  41     (hV : Fintype.card V≤m) :
