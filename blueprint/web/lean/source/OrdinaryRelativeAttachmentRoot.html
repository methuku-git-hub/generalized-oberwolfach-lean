  15 theorem ordinary_relative_attachment_root (Q F : SimpleGraph V) (H : V → U → Prop)
  16     (A W R : Finset V) (B : Finset U) (x : V) (hx : x∈R) (r : ℕ) (hr : 3≤r) (d c cH : ℝ) (hd : 0≤d)
  17     (hF : ∀ v,(F.degree v:ℝ)≤d)
  18     (hcount : ∀ v w,c≤((A∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  19     (hbudget : W.card+((R.card:ℝ)+r+6)*(d+1)<c)
  20     (hhub : ∀ T:Finset V,T.card≤r → (∀ v∈T,∀ w∈T,¬F.Adj v w) →
  21       cH≤((common Finset.univ T (fun v => Finset.univ.filter (H v))).card:ℝ))
  22     (hB : (B.card:ℝ)<cH) :
  23     ∃ (f : Fin r ↪ V) (u : U),
  24       (∀ i j,i≠⟨0,by omega⟩ → j≠⟨0,by omega⟩ →
  25         (SimpleGraph.cycleGraph r).Adj i j → Q.Adj (f i) (f j)) ∧
  26       (∀ j,(SimpleGraph.cycleGraph r).Adj ⟨0,by omega⟩ j → Q.Adj x (f j)) ∧
  27       H x u ∧ (∀ j,j≠⟨0,by omega⟩ → H (f j) u) ∧ u∉B ∧
  28       (∀ i j,¬F.Adj (f i) (f j)) ∧
  29       ∀ j,f j∈A ∧ f j∉W ∧ f j∉R ∧ ∀ v∈R,¬F.Adj (f j) v := by
  30   have hd1 : 0≤d+1 := by linarith
  31   obtain ⟨g,hg0,hg,hind,havoid⟩ := sparse_rooted_cycle_embedding Q F A W R x hx r hr d c hd hF hcount
  32     (by nlinarith only [hbudget,hd1])
  33   let z : Fin r := ⟨0,by omega⟩
  34   let T := Finset.univ.image g
  35   have hTc : T.card≤r := Finset.card_image_le.trans (by simp)
  36   have hTF : ∀ v∈T,∀ w∈T,¬F.Adj v w := by
  37     intro v hv w hw
  38     obtain ⟨i,_,rfl⟩ := Finset.mem_image.mp hv
  39     obtain ⟨j,_,rfl⟩ := Finset.mem_image.mp hw
  40     exact hind i j
  41   obtain ⟨u,hu,hHu⟩ := common_hub_from_count H T B cH (hhub T hTc hTF) hB
  42   have hRT : ((R∪T).card:ℝ)≤R.card+r := by
  43     have hh := Finset.card_union_le R T
  44     exact_mod_cast hh.trans (Nat.add_le_add_left hTc R.card)
  45   obtain ⟨a,haA,haW,haRT,_hxa,_hxa',haF⟩ := sparse_common_neighbour_avoiding Q F A W (R∪T) x x d c hd hF
  46     (hcount x x) (by nlinarith only [hbudget,hRT,hd1])
  47   have hga : ∀ i,g i≠a := by
  48     intro i he
  49     exact haRT (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨i,Finset.mem_univ _,he⟩))
  50   let f := replaceEmbeddingVertex g z a hga
  51   have hf0 : f z=a := replaceEmbeddingVertex_at g z a hga
  52   have hfn (i : Fin r) (hi : i≠z) : f i=g i := replaceEmbeddingVertex_ne g z a hga i hi
  53   refine ⟨f,u,?_,?_,?_,?_,hu,?_,?_⟩
  54   · intro i j hi hj hij
  55     rw [hfn i hi,hfn j hj]
  56     exact hg i j hij
  57   · intro j hj
  58     rw [hfn j hj.ne.symm,←hg0]
  59     exact hg z j hj
  60   · exact hHu _ (Finset.mem_image.mpr ⟨z,Finset.mem_univ _,hg0⟩)
  61   · intro j hj
  62     rw [hfn j hj]
  63     exact hHu _ (Finset.mem_image.mpr ⟨j,Finset.mem_univ _,rfl⟩)
  64   · intro i j
  65     by_cases hi : i=z <;> by_cases hj : j=z
  66     · rw [hi,hj,hf0];exact F.loopless.irrefl a
  67     · rw [hi,hf0,hfn j hj]
  68       exact haF _ (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨j,Finset.mem_univ _,rfl⟩))
  69     · rw [hj,hf0,hfn i hi]
  70       exact fun h => haF _ (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨i,Finset.mem_univ _,rfl⟩)) h.symm
  71     · rw [hfn i hi,hfn j hj];exact hind i j
  72   · intro j
  73     by_cases hj : j=z
  74     · rw [hj,hf0]

  76         fun v hv => haF v (Finset.mem_union_left _ hv)⟩
  77     · rw [hfn j hj]
  78       exact havoid j hj
