  14 theorem marked_alternating_correction {V U E K : Type*} [Fintype V] [Fintype U]
  15     [DecidableEq V] [DecidableEq U]
  16     {tail head : E → V} {ktail khead : K → V} {m : ℕ} {π : Equiv.Perm V} {b : ℕ}
  17     {H₀ Hₖ : V → U → Prop}
  18     (i : Fin (m+1)) (k : Bool) (hik : markedPositionType i=k)
  19     (β M : ℝ) (hβ : 0<β) (hM : 0≤M) (hn : 0<Fintype.card V)
  20     (hcount : ∀ Q : BipartiteFourCycle (if k then Hₖ else H₀),
  21       β*(Fintype.card V:ℝ)^m≤
  22       (Fintype.card (MarkedReplacementPair tail head ktail khead m π b H₀ Hₖ i
  23         Q.val.1.1 Q.val.1.2 Q.val.2.1 Q.val.2.2):ℝ))
  24     (r : V×U → ℝ) (a : BipartiteFourCycle (if k then Hₖ else H₀) → ℝ)
  25     (hrep : ∀ e,r e=∑ Q,a Q*bipartiteFourCycleVector Q e)
  26     (hload : ∀ e,(∑ Q,if FourCycleIncident Q e then |a Q| else 0)≤M) :
  27     ∃ Φ : MarkedWheel tail head ktail khead m π b H₀ Hₖ → ℝ,
  28       (∀ e,markedWheelRimLoad Φ e=0) ∧
  29       (∀ j v u,markedWheelSpokeLoad Φ j v u=if k=j then r (v,u) else 0) ∧
  30       (∀ D,|Φ D|≤4*M/(β*(Fintype.card V:ℝ)^m)) := by
  31   have hn' : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr hn
  32   let T (Q : BipartiteFourCycle (if k then Hₖ else H₀)) :=
  33     markedReplacementTransfer (tail:=tail) (head:=head) (ktail:=ktail) (khead:=khead)
  34       (π:=π) (b:=b) (H₀:=H₀) (Hₖ:=Hₖ) i Q.val.1.1 Q.val.1.2 Q.val.2.1 Q.val.2.2
  35   have hcard (Q : BipartiteFourCycle (if k then Hₖ else H₀)) :
  36       0<Fintype.card (MarkedReplacementPair tail head ktail khead m π b H₀ Hₖ i
  37         Q.val.1.1 Q.val.1.2 Q.val.2.1 Q.val.2.2) := by
  38     have hh : (0:ℝ)<β*(Fintype.card V:ℝ)^m := by positivity
  39     exact_mod_cast hh.trans_le (hcount Q)
  40   have hspoke (Q : BipartiteFourCycle (if k then Hₖ else H₀)) (j : Bool) (v : V) (u : U) :
  41       markedWheelSpokeLoad (T Q) j v u=if k=j then bipartiteFourCycleVector Q (v,u) else 0 := by
  42     rw [markedReplacementTransfer_spoke i _ _ _ _ (hcard Q),hik]
  43     rfl
  44   let Φ (D : MarkedWheel tail head ktail khead m π b H₀ Hₖ) := ∑ Q,a Q*T Q D
  45   refine ⟨Φ,?_,?_,?_⟩
  46   · intro e
  47     unfold Φ
  48     rw [markedWheelRimLoad_combination]
  49     simp only [T,markedReplacementTransfer_rim,mul_zero,Finset.sum_const_zero]
  50   · intro j v u
  51     unfold Φ
  52     rw [markedWheelSpokeLoad_combination]
  53     simp only [hspoke,mul_ite,mul_zero,Finset.sum_ite_irrel,Finset.sum_const_zero]
  54     rw [← hrep (v,u)]
  55   · intro D
  56     let B := 4/(β*(Fintype.card V:ℝ)^m)
  57     have hB : 0≤B := by dsimp [B];positivity
  58     have hsum (Q : BipartiteFourCycle (if k then Hₖ else H₀)) :
  59         |a Q*T Q D|≤B*(if FourCycleIncident Q (D.1.val i,D.2.val) then |a Q| else 0) := by
  60       by_cases hi : FourCycleIncident Q (D.1.val i,D.2.val)
  61       · rw [if_pos hi,abs_mul]
  62         calc
  63           _ ≤ |a Q| *B := mul_le_mul_of_nonneg_left
  64             (markedReplacementTransfer_card_bound i _ _ _ _ β hβ hn (hcount Q) D) (abs_nonneg _)
  65           _ = _ := by ring
  66       · have ht : T Q D=0 := markedReplacementTransfer_support i _ _ _ _ D hi
  67         rw [if_neg hi,ht,mul_zero,abs_zero,mul_zero]
  68     calc
  69       _ ≤ ∑ Q,|a Q*T Q D| := Finset.abs_sum_le_sum_abs _ _
  70       _ ≤ ∑ Q,B*(if FourCycleIncident Q (D.1.val i,D.2.val) then |a Q| else 0) :=
  71         Finset.sum_le_sum (fun Q _=>hsum Q)
  72       _ = B*(∑ Q,if FourCycleIncident Q (D.1.val i,D.2.val) then |a Q| else 0) :=
  73         (Finset.mul_sum _ _ _).symm
  74       _ ≤ B*M := mul_le_mul_of_nonneg_left (hload _) hB
  75       _ = _ := by dsimp only [B];ring
