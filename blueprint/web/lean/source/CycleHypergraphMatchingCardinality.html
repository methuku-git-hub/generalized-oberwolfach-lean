  11 theorem cycle_hypergraph_matching_cardinality
  12     (k B:ℕ) (hk:2≤k) (K a γ:ℝ) (hK:0<K) (ha:0<a) (hγ:0<γ):
  13     ∃ n₀:ℕ,∀ n D:ℕ,n₀≤n→n≤D→D≤n^B→
  14     ∀ (V E:Type) [Fintype V] [Fintype E] [DecidableEq V],
  15     ∀ (R:E→Finset V),
  16     Fintype.card V≤n^B→
  17     (∀ e,(R e).card=k)→(∀ v,0<Fintype.card (HypergraphEdgesAt R v))→
  18     (∀ v,Fintype.card (HypergraphEdgesAt R v)≤D)→
  19     (∀ v z,v≠z→((Finset.univ.filter (fun e=>v∈R e ∧ z∈R e)).card:ℝ)≤K*D/n)→
  20     a*D*n≤(Fintype.card E:ℝ)→
  21     ∃ M:Finset E,(∀ e∈M,∀ f∈M,e≠f→Disjoint (R e) (R f)) ∧
  22       (1-γ)*(Fintype.card E:ℝ)/D≤M.card := by
  23   obtain ⟨N,hN⟩:=uniform_weighted_hypergraph_matching k B hk K a γ hK ha hγ
  24   refine ⟨max N 1,?_⟩
  25   intro n D hn hnD hDB V E _ _ _ R hV hsize hpositive hdegree hcodeg htotal
  26   have hnpos:0<n:=by omega
  27   have hpow:1≤n^B:=Nat.succ_le_iff.mpr (pow_pos hnpos B)
  28   obtain ⟨M,hM,hweight⟩:=hN n D ((le_max_left N 1).trans hn) hnD hDB V E Unit R
  29     (fun _ _=>1) (fun _=>1) (fun _=>(Fintype.card E:ℝ)) hV (by simpa using hpow)
  30     hsize hpositive hdegree hcodeg (by intro;norm_num) (by intro;norm_num)
  31     (by intro; simp) (by intro;simpa using htotal)
  32   refine ⟨M,hM,?_⟩
  33   have h:=hweight ()
  34   simp only [Finset.sum_const_zero,Finset.sum_const,Finset.card_univ, nsmul_eq_mul, mul_one] at h
  35   have hlo:-(γ*(Fintype.card E:ℝ)/D)<(M.card:ℝ)-(Fintype.card E:ℝ)/D:=(abs_lt.mp h).1
  36   have hDpos:(0:ℝ)<D:=by exact_mod_cast (lt_of_lt_of_le hnpos hnD)
  37   have hdiv:((1-γ)*(Fintype.card E:ℝ)/D)=
  38      (Fintype.card E:ℝ)/D-γ*(Fintype.card E:ℝ)/D:=by ring
  39   rw [hdiv];linarith
  40 end Oberwolfach
