  16 theorem rooted_label_maps_local_coordinate_count (X Y : Finset V) (j : Fin h) (v : V) :
  17     Fintype.card {f : RootedLabelMap V h c X //
  18       (∀ k : Fin h,f.val (Sum.inl k)∈Y) ∧ f.val (Sum.inl j)=v} ≤
  19       X.card^c*Y.card^(h-1) := by
  20   let A := {f : RootedLabelMap V h c X //
  21       (∀ k : Fin h,f.val (Sum.inl k)∈Y) ∧ f.val (Sum.inl j)=v}
  22   let enc (f : A) : (Fin c → X) × ({k : Fin h // k≠j} → Y) :=
  23     (fun i => ⟨f.val.val (Sum.inr i),rootedLabelMap_root_mem X f.val i⟩,
  24      fun k => ⟨f.val.val (Sum.inl k.val),f.property.1 k.val⟩)
  25   have hi : Function.Injective enc := by
  26     intro f g he
  27     apply Subtype.ext
  28     apply Subtype.ext
  29     funext k
  30     rcases k with k | k
  31     · by_cases hk : k=j
  32       · subst k;exact f.property.2.trans g.property.2.symm
  33       · exact congrArg (fun y : Y => y.val) (congrFun (congrArg Prod.snd he) ⟨k,hk⟩)
  34     · exact congrArg (fun x : X => x.val) (congrFun (congrArg Prod.fst he) k)
  35   have hn : Fintype.card {k : Fin h // k≠j}=h-1 := by
  36     rw [Fintype.card_subtype_compl]
  37     simp
  38   simpa only [Fintype.card_prod,Fintype.card_fun,Fintype.card_fin,Fintype.card_coe,hn] using
  39     Fintype.card_le_of_injective enc hi

  41 theorem labelled_cycle_records_local_coordinate_count (R : C → Finset V)
  42     (S : O → Finset (Fin h ⊕ Fin c)) (X Y : Finset V) (J : ℕ)
  43     (hJ : ∀ U : Finset V,Fintype.card {C₀ : C // R C₀=U} ≤ J) (j : Fin h) (v : V) :
  44     Fintype.card {D : LabelledCycleRecord R S X //
  45       (∀ k : Fin h,D.1.val (Sum.inl k)∈Y) ∧ D.1.val (Sum.inl j)=v} ≤
  46       X.card^c*Y.card^(h-1)*J^Fintype.card O := by
  47   let A := {D : LabelledCycleRecord R S X //
  48       (∀ k : Fin h,D.1.val (Sum.inl k)∈Y) ∧ D.1.val (Sum.inl j)=v}
  49   let F := {f : RootedLabelMap V h c X //
  50       (∀ k : Fin h,f.val (Sum.inl k)∈Y) ∧ f.val (Sum.inl j)=v}
  51   let B := Σ f : F,CycleIdentifierAssignments R S f.val.val
  52   let enc (D : A) : B := ⟨⟨D.val.1,D.property⟩,D.val.2⟩
  53   have hi : Function.Injective enc := by
  54     intro D E he
  55     apply Subtype.ext
  56     exact congrArg (fun b : B => (⟨b.1.val,b.2⟩ : LabelledCycleRecord R S X)) he
  57   have hb : Fintype.card B ≤ Fintype.card F*J^Fintype.card O := by
  58     rw [Fintype.card_sigma]
  59     calc
  60       _ ≤ ∑ _f : F,J^Fintype.card O := Finset.sum_le_sum
  61         (fun f _ => cycle_identifier_assignments_card R S J hJ f.val.val)
  62       _ = _ := by simp
  63   exact (Fintype.card_le_of_injective enc hi).trans (hb.trans
  64     (Nat.mul_le_mul_right _ (rooted_label_maps_local_coordinate_count X Y j v)))
