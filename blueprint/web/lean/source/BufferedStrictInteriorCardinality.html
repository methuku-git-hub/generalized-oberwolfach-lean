  14 theorem strict_path_index_card (n:ℕ) :
  15     (Finset.univ.filter (fun p:Fin (n+1)=>0<p.val ∧ p.val<n)).card=n-1 := by
  16   classical
  17   have heq:(Finset.univ.filter (fun p:Fin (n+1)=>0<p.val ∧ p.val<n))=
  18       Finset.univ.image (strictPathIndex n):=by
  19     ext p
  20     simp only [Finset.mem_filter,Finset.mem_univ,true_and,Finset.mem_image]
  21     constructor
  22     · intro hp
  23       let j:Fin (n-1):=⟨p.val-1,by omega⟩
  24       refine ⟨j,?_⟩
  25       apply Fin.ext
  26       change (p.val-1)+1=p.val
  27       omega
  28     · rintro ⟨j,rfl⟩
  29       change 0<j.val+1 ∧ j.val+1<n
  30       have hj:j.val<n-1:=j.isLt
  31       omega

  40     (bufferedPathStrictInterior n inner).card=∑i,(n i-1) := by

  42   have hsubset (i:I):(Finset.univ.filter (fun p:Fin (n i+1)=>0<p.val ∧ p.val<n i)).image (inner i)⊆
