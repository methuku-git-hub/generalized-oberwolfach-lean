  14     (F:U→SimpleGraph (Fin n)) (hF:∀u,(F u).IsRegularOfDegree 2)
  15     (A:Finset C) (key:C→U×Fin K) (θ lam:ℝ) (hlamθ:lam≤θ)

  18       |(((A.filter (fun c=>key c=(u,i))).card:ℝ)-(1-θ)*t)|≤lam*t):
  19     ∃h:{c // c∈A}↪(Σu,(F u).ConnectedComponent),
  20       ∀c,(h c).1=(key c.val).1 ∧ Fintype.card (h c).2=(key c.val).2.val+1 :=by

  22   have hcap:∀u r,(A.filter (fun c=>sizekey c=(u,r))).card≤
  23       (Finset.univ.filter (fun C:(F u).ConnectedComponent=>Fintype.card C=r)).card:=by
  24     intro u r
  25     by_cases hr:∃i:Fin K,i.val+1=r
  26     · obtain ⟨i,hi⟩:=hr
  27       have hf:A.filter (fun c=>sizekey c=(u,r))=A.filter (fun c=>key c=(u,i)):=by
  28         ext c

  31         · rintro ⟨hc,hu,hlen⟩
  32           refine ⟨hc,Prod.ext hu ?_⟩
