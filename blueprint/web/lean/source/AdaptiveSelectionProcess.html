  12 theorem adaptive_selection_process
  13     {I A State : Type*} [Fintype I] [Fintype A] [Nonempty A]
  14     (N B : ℕ) (active : Fin N→I→Prop) (q : ℝ) (hq : 0≤q)
  15     (counter : State→I→ℕ) (Good : ℕ→State→Prop) (initial : State)
  16     (hinit : Good 0 initial) (hzero : ∀ x,counter initial x=0)
  17     (hsmall : selectionPotential active q 0 (fun _=>0)<(2:ℝ)^B)
  18     (hstep : ∀ i : Fin N,∀ state,Good i.val state → (∀ x,counter state x<B) →
  19       ∃ next : A→State,∃ hit : A→I→Prop,
  20         (∀ a,Good (i.val+1) (next a)) ∧
  21         (∀ a x,counter (next a) x=counter state x+(if hit a x then 1 else 0)) ∧
  22         (∀ x,¬active i x → ∀ a,¬hit a x) ∧
  23         (∀ x,((Finset.univ.filter (fun a=>hit a x)).card:ℝ)≤q*Fintype.card A)) :
  24     ∃ state,Good N state ∧ ∀ x,counter state x<B := by
  25   have hex : ∀ k : ℕ,k≤N → ∃ state,Good k state ∧
  26       selectionPotential active q k (counter state)≤selectionPotential active q 0 (fun _=>0) := by
  27     intro k
  28     induction k with
  29     | zero=>
  30       intro hk
  31       refine ⟨initial,hinit,?_⟩
  32       have he : counter initial=(fun _=>0) := funext hzero
  33       rw [he]
  34     | succ k ih=>
  35       intro hk
  36       obtain ⟨state,hgood,hpot⟩ := ih (by omega)
  37       let i : Fin N := ⟨k,by omega⟩
  38       have hsafe := selectionPotential_counter_lt active q hq k B (counter state) (hpot.trans_lt hsmall)
  39       obtain ⟨next,hit,hnext,hcounter,hinactive,hcount⟩ := hstep i state hgood hsafe
  40       obtain ⟨a,ha⟩ := selectionPotential_step_choice active i q hq (counter state) hit hinactive hcount
  41       refine ⟨next a,hnext a,?_⟩
  42       have he : counter (next a)=(fun x=>counter state x+(if hit a x then 1 else 0)) := funext (hcounter a)
  43       rw [he]
  44       exact ha.trans hpot
  45   obtain ⟨state,hgood,hpot⟩ := hex N le_rfl
  46   exact ⟨state,hgood,selectionPotential_counter_lt active q hq N B (counter state) (hpot.trans_lt hsmall)⟩
