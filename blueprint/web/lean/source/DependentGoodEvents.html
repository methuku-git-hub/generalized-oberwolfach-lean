  10 theorem finiteLaw_twoStage_good {Ω : Type*} [Fintype Ω]
  11     {Γ : Ω→Type*} [∀ ω,Fintype (Γ ω)]
  12     (μ : FiniteLaw Ω) (ν : ∀ ω,FiniteLaw (Γ ω)) (E : Ω→Prop) (S : ∀ ω,Γ ω→Prop)
  13     (a b : ℝ) (ha : 0≤a) (hb : 0≤b)
  14     (hE : 1-a≤μ.probability E) (hS : ∀ ω,E ω → 1-b≤(ν ω).probability (S ω)) :
  15     1-a-b≤(μ.sigma ν).probability (fun z=>E z.1 ∧ S z.1 z.2) := by
  16   by_cases hb1 : b≤1
  17   · have hh := finiteLaw_twoStage_survival μ ν E S b hS
  18     have hlo := mul_le_mul_of_nonneg_left hE (show 0≤1-b by linarith)
  19     have hab := mul_nonneg ha hb
  20     nlinarith only [hh,hlo,hab]
  21   · exact (by linarith : 1-a-b≤0).trans (finiteLaw_probability_nonneg _ _)
