  12 theorem factor_oriented_physical_edge_injective {X:Type*} [Fintype X] [DecidableEq X]
  13     (G:SimpleGraph X) (hG:G.IsRegularOfDegree 2)
  14     (e:∀C:G.ConnectedComponent,SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph):
  15     Function.Injective (factorOrientedPhysicalEdge (factorCycleOrientation G e)):=by
  16   intro x y h
  17   change s(x,factorCycleOrientation G e x)=s(y,factorCycleOrientation G e y) at h
  18   rcases Sym2.eq_iff.mp h with ⟨hxy,_⟩ | ⟨hxy,hyx⟩
  19   · exact hxy
  20   · have htwo:factorCycleOrientation G e (factorCycleOrientation G e y)=y:=by rw [←hxy,hyx]
  21     exact (factor_orientation_no_two_cycles G hG e y htwo).elim

  33     · intro hxy
