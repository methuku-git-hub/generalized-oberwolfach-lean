  16 theorem ordinary_wheel_attachments (β : ℝ) (hβ : 0 < β) (r q N : ℕ) (hr : 3 ≤ r) :
  17     ∃ n₀ : ℕ,∀ n : ℕ,n₀ ≤ n →
  18     ∀ (V U : Type*) [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U],Fintype.card V ≤ n →
  19     ∀ (G : SimpleGraph V) (H : V → U → Prop),
  20     (∀ v,(1/2+β)*(n:ℝ) ≤ G.degree v) →
  21     (∀ v,(1-1/(2*(r:ℝ)))*(Fintype.card U:ℝ) ≤ (Finset.univ.filter (fun u => H v u)).card) →
  22     ∀ (R : Finset V) (B : Finset U) (x : Fin N → V),R.card ≤ q → (∀ i,x i∈R) →
  23     2*(B.card+N) < Fintype.card U →
  24     ∃ W : Fin N → OrdinaryWheel G ⊥ H r,
  25       (∀ i,x i∈(ordinaryWheelRim (W i)).support ∧ ordinaryWheelHub (W i)∉B ∧
  26         ∀ v∈(ordinaryWheelRim (W i)).support,v∈R → v=x i) ∧
  27       (∀ i j,i≠j → Disjoint (ordinaryWheelResources (W i)) (ordinaryWheelResources (W j)) ∧
  28         ordinaryWheelHub (W i)≠ordinaryWheelHub (W j) ∧
  29         ∀ v,v∈(ordinaryWheelRim (W i)).support → v∈(ordinaryWheelRim (W j)).support → v∈R) := by
  30   obtain ⟨n₀,hembed⟩ := ordinary_rooted_wheel_embedding β hβ r (q+N*r) hr
  31   refine ⟨n₀,?_⟩
  32   intro n hn V U _ _ _ _ hV G H hdeg hH R B x hR hx hB
  33   let allowed := fun i (o : Option (OrdinaryWheel G ⊥ H r)) =>
  34     ∃ W,o=some W ∧ x i∈(ordinaryWheelRim W).support ∧ ordinaryWheelHub W∉B ∧
  35       ∀ v∈(ordinaryWheelRim W).support,v∈R → v=x i
  36   let compatible := fun (o p : Option (OrdinaryWheel G ⊥ H r)) =>
  37     ∀ W Z,o=some W → p=some Z →
  38       Disjoint (ordinaryWheelResources W) (ordinaryWheelResources Z) ∧
  39       ordinaryWheelHub W≠ordinaryWheelHub Z ∧

  42     intro i F hprefix
  43     let J := Finset.univ.filter (fun j : Fin N => j.val < i.val)
  44     let A := R∪J.biUnion (fun j => optionWheelRimSupport (F j))
  45     let B' := B∪J.biUnion (fun j => optionWheelHubSet (F j))
  46     have hJ : J.card ≤ N := (Finset.card_le_univ _).trans_eq (Fintype.card_fin N)
  47     have hAc : A.card ≤ q+N*r := by
  48       have hsum : (J.biUnion (fun j => optionWheelRimSupport (F j))).card ≤ J.card*r := by
  49         calc
  50           _ ≤ ∑ j∈J,(optionWheelRimSupport (F j)).card := Finset.card_biUnion_le
  51           _ ≤ ∑ _j∈J,r := Finset.sum_le_sum (fun j _ => optionWheelRimSupport_card (F j))
  52           _ = _ := by simp
  53       have hu := Finset.card_union_le R (J.biUnion (fun j => optionWheelRimSupport (F j)))
  54       dsimp [A]
  55       nlinarith
  56     have hBc : B'.card ≤ B.card+N := by
  57       have hsum : (J.biUnion (fun j => optionWheelHubSet (F j))).card ≤ J.card := by
  58         calc
  59           _ ≤ ∑ j∈J,(optionWheelHubSet (F j)).card := Finset.card_biUnion_le
  60           _ ≤ ∑ _j∈J,1 := Finset.sum_le_sum (fun j _ => optionWheelHubSet_card (F j))
  61           _ = _ := by simp
  62       have hu := Finset.card_union_le B (J.biUnion (fun j => optionWheelHubSet (F j)))
  63       dsimp [B']
  64       omega
  65     obtain ⟨W,hxW,huW,havoid⟩ := hembed n hn V U hV G H hdeg hH (x i) (A.erase (x i)) B'
  66       ((Finset.card_erase_le).trans hAc) (Finset.notMem_erase _ _) (by omega)
  67     have hroot (v : V) (hv : v∈(ordinaryWheelRim W).support) (hva : v∈A) : v=x i := by
  68       by_contra hne
  69       exact havoid v hv (Finset.mem_erase.mpr ⟨hne,hva⟩)
  70     refine ⟨some W,?_,?_⟩
  71     · refine ⟨W,rfl,hxW,?_,?_⟩
  72       · exact fun h => huW (Finset.mem_union_left _ h)
  73       · intro v hv hvr
  74         exact hroot v hv (Finset.mem_union_left _ hvr)
  75     · intro j hj
  76       intro Z W' hZ hW'
  77       have hW'val : W=W' := Option.some.inj hW'
  78       subst W'
  79       have hjJ : j∈J := Finset.mem_filter.mpr ⟨Finset.mem_univ _,hj⟩
  80       have hcommon (v : V) (hvZ : v∈(ordinaryWheelRim Z).support)
  81           (hvW : v∈(ordinaryWheelRim W).support) : v=x i := by
  82         apply hroot v hvW
  83         apply Finset.mem_union_right
  84         apply Finset.mem_biUnion.mpr
  85         refine ⟨j,hjJ,?_⟩
  86         rw [hZ]
  87         exact Set.mem_toFinset.mpr hvZ
  88       have hhubs : ordinaryWheelHub Z≠ordinaryWheelHub W := by
  89         intro he
  90         apply huW
  91         apply Finset.mem_union_right
  92         apply Finset.mem_biUnion.mpr
  93         refine ⟨j,hjJ,?_⟩

  96       refine ⟨ordinaryWheels_disjoint_at_one_root Z W (x i) hcommon hhubs,hhubs,?_⟩
  97       intro v hvZ hvW
  98       rw [hcommon v hvZ hvW]
  99       exact hx i)
 100   choose W hW using hF
