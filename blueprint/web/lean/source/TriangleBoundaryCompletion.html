  14 theorem triangle_boundary_completion
  15     {V U I:Type} [Fintype V] [DecidableEq V] [Fintype U] [DecidableEq U] [Fintype I]
  16     (Q F:SimpleGraph V) (H:V→U→Prop) (A:Finset (V⊕U))
  17     (x:I→V⊕U) (T:I→Finset (V⊕U)) (rim:I→Finset V) (hub:I→U)
  18     (hx:∀ i,x i∉A) (hT:∀ i,T i⊆A) (hrim:∀ i,(rim i).card=3)
  19     (hvertices:∀ i,insert (x i) (T i)=(rim i).image Sum.inl∪{Sum.inr (hub i)})
  20     (hclique:∀ i,∀ v∈insert (x i) (T i),∀ w∈insert (x i) (T i),v≠w→
  21       (ordinaryPhysicalGraph Q H).Adj v w)
  22     (hF:∀ i,∀ v∈rim i,∀ w∈rim i,¬F.Adj v w)
  23     (hinner:∀ i j,i≠j→Disjoint (spanningRestriction (⊤:SimpleGraph (V⊕U)) (T i))
  24       (spanningRestriction ⊤ (T j)))
  25     (hsame:∀ i j,i≠j→x i=x j→Disjoint (T i) (T j)):
  26     ∃ W:I→OrdinaryWheel Q F H 3,
  27       (∀ i,(ordinaryWheelRim (W i)).support.toFinset=rim i ∧ ordinaryWheelHub (W i)=hub i) ∧
  28       (∀ i,ordinaryWheelPhysicalGraph (W i)=spanningRestriction ⊤ (insert (x i) (T i))) ∧
  29       (∀ i j,i≠j→Disjoint (ordinaryWheelResources (W i)) (ordinaryWheelResources (W j))) ∧
  30       HasOrdinaryWheelDecomposition (wheelFamilyRim W) (wheelFamilySpokes W) 3 ∧
  31       spanningRestriction (⨆ i,ordinaryWheelPhysicalGraph (W i)) A=
  32         (⨆ i,spanningRestriction (⊤:SimpleGraph (V⊕U)) (T i)) ∧
  33       (∀ z∉A,∀ v,(⨆ i,ordinaryWheelPhysicalGraph (W i)).Adj z v↔∃ i,x i=z ∧ v∈T i):=by
  34   have hex (i:I):∃ W:OrdinaryWheel Q F H 3,
  35       (ordinaryWheelRim W).support.toFinset=rim i ∧ ordinaryWheelHub W=hub i ∧
  36       ordinaryWheelPhysicalGraph W=spanningRestriction ⊤ (insert (x i) (T i)):=by
  37     have hh:=physical_triangle_wheel Q F H (rim i) (hub i) (hrim i)
  38       (by simpa only [←hvertices i] using hclique i) (hF i)
  39     simpa only [←hvertices i] using hh
  40   choose W hW using hex
  41   have hdis (i j:I) (hij:i≠j):Disjoint (ordinaryWheelResources (W i)) (ordinaryWheelResources (W j)):=by
  42     have hg:Disjoint (ordinaryWheelPhysicalGraph (W i)) (ordinaryWheelPhysicalGraph (W j)):=by
  43       rw [(hW i).2.2,(hW j).2.2]
  44       exact root_completed_cliques_disjoint A (T i) (T j) (x i) (x j)
  45         (hT i) (hT j) (hx i) (hx j) (hinner i j hij) (hsame i j hij)
  46     have he:=SimpleGraph.disjoint_edgeFinset.mpr hg
  47     rw [ordinaryWheelPhysicalGraph_edges,ordinaryWheelPhysicalGraph_edges] at he
  48     apply Finset.disjoint_left.mpr
  49     intro e hei hej
  50     exact Finset.disjoint_left.mp he (Finset.mem_image.mpr ⟨e,hei,rfl⟩) (Finset.mem_image.mpr ⟨e,hej,rfl⟩)
  51   have heq:(⨆ i,ordinaryWheelPhysicalGraph (W i))=
  52       (⨆ i,spanningRestriction (⊤:SimpleGraph (V⊕U)) (insert (x i) (T i))):=
  53     iSup_congr (fun i=>(hW i).2.2)
  54   refine ⟨W,fun i=>⟨(hW i).1,(hW i).2.1⟩,fun i=>(hW i).2.2,hdis,
  55     wheelFamily_decomposition W hdis,?_,?_⟩
  56   · rw [heq];exact root_completed_cliques_inner A x T hx hT
  57   · intro z hz v;rw [heq];exact root_completed_cliques_at_root A x T hx hT z v hz
