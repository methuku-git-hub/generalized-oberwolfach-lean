  17  theorem edge_vertex_wheels_disjoint (W C:OrdinaryWheel Q F H r) (hr:4≤r)
  18     (x a b:V) (hxa:(ordinaryWheelRim C).Adj x a) (hxb:(ordinaryWheelRim C).Adj x b)
  19     (hx:x∉(ordinaryWheelRim W).support)
  20     (hint:∀ v∈rootWheelInternal C x a b,v∉(ordinaryWheelRim W).support)
  21     (ha:Sum.inr (a,ordinaryWheelHub C)∉ordinaryWheelResources W)
  22     (hb:Sum.inr (b,ordinaryWheelHub C)∉ordinaryWheelResources W):
  23     Disjoint (ordinaryWheelResources W) (ordinaryWheelResources C):=by
  24   have hsplit (v:V) (hv:v∈(ordinaryWheelRim C).support)
  25       (hw:v∈(ordinaryWheelRim W).support):v=a∨v=b:=by
  26     by_contra hn
  27     have hn':v≠a∧v≠b:=by tauto
  28     have hvx:v≠x:=by intro he;exact hx (he ▸ hw)
  29     have hi:v∈rootWheelInternal C x a b:=by
  30       simp only [rootWheelInternal,Finset.mem_sdiff,Set.mem_toFinset,Finset.mem_insert,Finset.mem_singleton]
  31       exact ⟨hv,by tauto⟩
  32     exact hint v hi hw
  33   have hnab:¬(ordinaryWheelRim C).Adj a b:=
  34     fun hab=>(show IsEmbeddedCycle (ordinaryWheelRim C) r from C.val.1.val.property).no_triangle
  35       hr x a b hxa hab hxb.symm
  36   apply (ordinaryWheelResources_disjoint_iff W C).mpr
  37   constructor
  38   · apply disjoint_iff_inf_le.mpr
  39     intro v w h
  40     rcases hsplit v h.2.mem_support_left h.1.mem_support_left with hv|hv <;>
  41       rcases hsplit w h.2.mem_support_right h.1.mem_support_right with hw|hw
  42     · exact (h.2.ne (hv.trans hw.symm)).elim
  43     · exact (hnab (hv ▸ hw ▸ h.2)).elim
  44     · exact (hnab (hw ▸ hv ▸ h.2.symm)).elim
  45     · exact (h.2.ne (hv.trans hw.symm)).elim
  46   · intro he
  47     apply Finset.disjoint_left.mpr
  48     intro v hv hw
  49     have hv':v∈(ordinaryWheelRim W).support:=Set.mem_toFinset.mp hv
  50     have hw':v∈(ordinaryWheelRim C).support:=Set.mem_toFinset.mp hw
  51     have hs:Sum.inr (v,ordinaryWheelHub C)∈ordinaryWheelResources W:=
  52       (ordinaryWheelResources_spoke W v _).mpr ⟨he,hv'⟩
  53     rcases hsplit v hw' hv' with h|h
  54     · exact ha (h ▸ hs)
  55     · exact hb (h ▸ hs)

  61   ext v
