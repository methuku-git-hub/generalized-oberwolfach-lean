   3 /-! The original prescribed-edge assignment, stated as an actual function
   4 on the graph's edge set, including the empty graph case. -/
   5 set_option autoImplicit false
   6 noncomputable section
   7 namespace Oberwolfach
   8 attribute [local instance] Classical.propDecidable
   9 variable {V W:Type} [Fintype V] [Fintype W] [DecidableEq V] [DecidableEq W]

  11 theorem graph_edge_assignment_original (G:SimpleGraph V) (L:Sym2 V→Finset W) (B:V→Finset W)
  12     (a b:ℝ) (ha:0<a) (hb:0≤b) (hbsmall:b≤a/16) (hn:0<Fintype.card V)
  13     (hL:∀e∈G.edgeFinset,a*Fintype.card V≤((L e).card:ℝ))
  14     (hB:∀v,((B v).card:ℝ)≤b*Fintype.card V)
  15     (hG:∀v,(G.degree v:ℝ)≤b*Fintype.card V):
  16     let K:=⌈4*(G.edgeFinset.card:ℝ)/(a*Fintype.card V)⌉₊;
  17     ∃f:{e:Sym2 V // e∈G.edgeFinset}→W,
  18       (∀e,f e∈L e.1 ∧ ∀v∈e.1.toFinset,f e∉B v) ∧
  19       (∀e e',e≠e'→¬Disjoint e.1.toFinset e'.1.toFinset→f e≠f e') ∧
  20       (∀w,(Finset.univ.filter (fun e:{e:Sym2 V // e∈G.edgeFinset}=>f e=w)).card≤K) := by
  21   intro K
  22   obtain ⟨h,hallowed,hconf,hcap⟩:=graph_edge_assignment G L B a b ha hb hbsmall hn hL hB hG
  23   have hfull:∀e∈G.edgeFinset,∃w,h e=some w:=by
  24     intro e he
  25     obtain ⟨w,hw,_,_⟩:=hallowed e he
  26     exact ⟨w,hw⟩
  27   obtain ⟨f,hf,hcard⟩:=edge_assignment_option_to_function G.edgeFinset h hfull
  28   refine ⟨f,?_,?_,?_⟩
  29   · intro e
  30     obtain ⟨w,hw,hLw,hBw⟩:=hallowed e.1 e.2
  31     have heq:f e=w:=Option.some.inj ((hf e).symm.trans hw)
  32     simpa [heq] using And.intro hLw hBw
  33   · intro e e' hne hc heq
  34     have hvne:e.1≠e'.1:=by intro h;exact hne (Subtype.ext h)
  35     have hh:=hconf e.1 e.2 e'.1 e'.2 hvne hc
  36     apply hh
  37     rw [hf e,hf e',heq]
  38   · intro w
  39     rw [hcard w]
  40     exact hcap w
  41 end Oberwolfach
