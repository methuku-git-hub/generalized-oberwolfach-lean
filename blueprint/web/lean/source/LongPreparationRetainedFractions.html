  12 theorem long_preparation_retained_fractions (K:ℕ) (hK:0<K) (θ ℓ:ℝ) (hθ:0≤θ)
  13     (f:Fin K→ℝ) (hf:∀ i,0≤f i) (hsum:ℓ+∑ i,f i=1):
  14     let threshold:=θ/(100*(K:ℝ))
  15     let S:Finset (Fin K):=Finset.univ.filter (fun i=>threshold≤f i)
  16     let rare:=∑ i∈Finset.univ\S,f i;
  17     (∀ i∈S,threshold≤f i) ∧ 0≤rare ∧ rare≤θ/100 ∧
  18     ℓ+(∑ i∈S,f i)+rare=1:=by
  19   intro threshold S rare
  20   have hK':(0:ℝ)<K:=by exact_mod_cast hK
  21   have hden:(0:ℝ)<100*K:=by positivity
  22   have hthreshold:0≤threshold:=by dsimp [threshold];positivity
  23   have hsubset:S⊆Finset.univ:=Finset.subset_univ _
  24   have hrare0:0≤rare:=Finset.sum_nonneg (fun i _=>hf i)
  25   have hsbound:rare≤((Finset.univ\S).card:ℝ)*threshold:=by
  26     dsimp [rare]
  27     calc
  28       _ ≤ ∑ _i∈Finset.univ\S,threshold:=by
  29         apply Finset.sum_le_sum
  30         intro i hi
  31         have hn:i∉S:=(Finset.mem_sdiff.mp hi).2
  32         have hnot:¬threshold≤f i:=by
  33           intro h
  34           apply hn
  35           exact Finset.mem_filter.mpr ⟨Finset.mem_univ i,h⟩
  36         exact (lt_of_not_ge hnot).le
  37       _ = _:=by simp
  38   have hcard:((Finset.univ\S).card:ℝ)≤K:=by
  39     exact_mod_cast (by simpa using (Finset.card_le_univ (Finset.univ\S)))
  40   have htop:((Finset.univ\S).card:ℝ)*threshold≤(K:ℝ)*threshold:=mul_le_mul_of_nonneg_right hcard hthreshold
  41   have hscale:(K:ℝ)*threshold=θ/100:=by
  42     dsimp [threshold]
  43     field_simp
