  15 theorem finiteLaw_probability_congr_support {Ω : Type*} [Fintype Ω]
  16     (μ : FiniteLaw Ω) (A B : Ω → Prop)
  17     (h : ∀ x,0<μ.mass x → (A x ↔ B x)) : μ.probability A=μ.probability B := by
  18   unfold FiniteLaw.probability
  19   apply Finset.sum_congr rfl
  20   intro x _
  21   by_cases hx : 0<μ.mass x
  22   · simp only [h x hx]
  23   · have hz : μ.mass x=0 := le_antisymm (le_of_not_gt hx) (μ.nonneg x)
  24     simp [hz]

  26 theorem finiteLaw_pi_positive_coordinate
  27     {I : Type*} [Fintype I] {X : I → Type*} [∀ i,Fintype (X i)]
  28     (μ : ∀ i,FiniteLaw (X i)) (σ : ∀ i,X i)
  29     (hσ : 0<(FiniteLaw.pi μ).mass σ) (i : I) : 0<(μ i).mass (σ i) := by
  30   classical
  31   have hn : (μ i).mass (σ i)≠0 := by
  32     intro hz
  33     have he : (FiniteLaw.pi μ).mass σ=0 := by
  34       exact Finset.prod_eq_zero (Finset.mem_univ i) hz
  35     rw [he] at hσ
  36     exact (lt_irrefl 0) hσ
  37   exact lt_of_le_of_ne ((μ i).nonneg _) (Ne.symm hn)

  41     (L : E → Finset C) (hL : ∀ e,(L e).Nonempty)
  42     (a : E → C → ℝ) (ha : ∀ e c,0≤a e c ∧ a e c≤1)

  63 theorem resourceColour_avoidance_probability
  64     {V E C : Type*} [Fintype E] [Fintype C]
  65     (R : E → Finset V) (L : E → Finset C) (hL : ∀ e,(L e).Nonempty)
  66     (a : E → C → ℝ) (ha : ∀ e c,0≤a e c ∧ a e c≤1)
  67     (S : Finset V) (c : C) :
  68     let G := hypergraphLineGraph R
  69     let μ := FiniteLaw.pi (fun e=>(FiniteLaw.uniformOn (L e) (hL e)).withCoin (a e) (ha e))
  70     μ.probability (fun σ=>∀ v∈S,∀ e∈hypergraphVertexClique R v,
  71       coinRetainedColour G σ e≠some c)=
  72     μ.probability (fun σ=>∀ v∈S,∀ e∈hypergraphColourClique R L v c,
  73       coinRetainedColour G σ e≠some c) := by
  74   dsimp only
  75   apply finiteLaw_probability_congr_support
  76   intro σ hσ
  77   constructor
  78   · intro h v hv e he
  79     apply h v hv e
  80     exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,(Finset.mem_filter.mp he).2.1⟩
  81   · intro h v hv e he hc
  82     have hlist := retainedColour_mem_list_on_support (hypergraphLineGraph R) L hL a ha σ hσ e c hc
  83     exact h v hv e (Finset.mem_filter.mpr
  84       ⟨Finset.mem_univ _,(Finset.mem_filter.mp he).2,hlist⟩) hc

  86 theorem hypergraph_resource_colour_avoidance
  87     {V E C : Type*} [Fintype E] [Fintype C]
  88     (R : E → Finset V) (L : E → Finset C) (hL : ∀ e,(L e).Nonempty)
  89     {l k B s : ℕ} (hl : 1<l) (hcard : ∀ e,(L e).card=l)
  90     (hsize : ∀ e,(R e).card≤k)
  91     (hcodeg : ∀ v w,v≠w → (Finset.univ.filter (fun e=>v∈R e ∧ w∈R e)).card≤B)
  92     (c : C) (hdegree : ∀ v,(hypergraphColourClique R L v c).card≤s) (hsl : s≤l)
  93     (S : Finset V) (hsmall : 2*(S.card^2*(k^2*B))≤l) :
  94     let G := hypergraphLineGraph R
  95     let μ := FiniteLaw.pi (fun e=>(FiniteLaw.uniformOn (L e) (hL e)).withCoin
  96       (equalizationCoin G L l (k*s) e)
  97       (equalizationCoin_bounds G L (by omega) (k*s) e))
  98     let r := ((1-1/(l : ℝ))^(k*s))/(l : ℝ)
  99     |μ.probability (fun σ=>∀ v∈S,∀ e∈hypergraphVertexClique R v,
 100         coinRetainedColour G σ e≠some c)-
 101       (∏ v∈S,(1-((hypergraphColourClique R L v c).card : ℝ)*r))|≤
 102       (2 : ℝ)^S.card*((S.card^2*(2*k^2+k+2)*B : ℕ) : ℝ)/(l : ℝ) := by
 103   dsimp only
 104   rw [resourceColour_avoidance_probability]
