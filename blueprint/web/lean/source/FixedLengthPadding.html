  11 def padIndexedFunctions {I:Type} [Fintype I] [DecidableEq I] (S:Finset I) (X:I→Type)
  12     (w:∀ i:↥S,X i.val→ℝ) (i:I) (x:X i):ℝ:=
  13   if hi:i∈S then w ⟨i,hi⟩ x else 0
