  12 def predecessorGraph (B : ℕ → Finset ℕ) (hB : ∀ j,∀ i∈B j,i<j) : SimpleGraph ℕ where
  13   Adj i j := i∈B j ∨ j∈B i
  14   symm := ⟨fun _ _ h=>h.symm⟩
  15   loopless := ⟨fun i h=>by rcases h with h|h <;> exact (lt_irrefl i) (hB i i h)⟩
  16 
  17 theorem predecessorGraph_lower_adj (B : ℕ → Finset ℕ) (hB : ∀ j,∀ i∈B j,i<j)
  18     (i j : ℕ) (hij : i<j) : (predecessorGraph B hB).Adj i j ↔ i∈B j := by
  19   constructor
  20   · intro h
  21     rcases h with h|h
  22     · exact h
  23     · exact False.elim (Nat.lt_asymm hij (hB i j h))
  24   · intro h;exact Or.inl h
  25 
  26 theorem predecessorGraph_back_card (B : ℕ → Finset ℕ) (hB : ∀ j,∀ i∈B j,i<j) (n : ℕ) :
  27     (Finset.univ.filter (fun i : Fin (n+1)=>(predecessorGraph B hB).Adj (i:ℕ) (n+1))).card=(B (n+1)).card := by
  28   apply Finset.card_bij (fun (i : Fin (n+1)) _=>(i:ℕ))
  29   · intro i hi
  30     exact (predecessorGraph_lower_adj B hB _ _ i.isLt).mp (Finset.mem_filter.mp hi).2
  31   · intro i _ j _ he
  32     exact Fin.ext he
  33   · intro j hj
  34     refine ⟨⟨j,hB (n+1) j hj⟩,?_,rfl⟩
  35     exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,Or.inl hj⟩
