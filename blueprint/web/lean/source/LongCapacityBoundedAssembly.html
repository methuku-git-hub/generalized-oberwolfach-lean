  36 theorem long_capacity_bounded_interval_weights {X : Type*} [Fintype X]
  37     (a : ℕ → X → X → ℝ) (d : ℕ → X → ℝ)
  38     (ha : ∀ i x y, 0 ≤ a i x y) (hd : ∀ i x, 0 < d i x)
  39     (hrow : ∀ i x, ∑ y, a i x y=d i x)
  40     (hcol : ∀ i y, ∑ x, a i x y=d (i+1) y)
  41     (M : ℝ) (hM : 0 < M) (hms : ∀ i, ∑ x, d i x=M)
  42     (s L : Fin 3 → ℕ) (t : ℕ)
  43     (hbd : ∀ i x, d (s i+L i) x=d (s (finRotate 3 i)) x)
  44     (J : Finset ℕ) (hJ : J.card ≤ 1) (κ B : ℝ)
  45     (hκ : 0 < κ) (hκ1 : κ < 1) (hB : 1 ≤ B)
  46     (ht : 1+3*⌈Real.log (10*(B+1)) / (-Real.log (1-κ))⌉₊ ≤ t)
  47     (hL : ∀ i, 4*t+1 ≤ L i)
  48     (hupper : ∀ i j, s i ≤ j → j < s i+L i → j ∉ J → ∀ x y,
  49       capacityTransition a d j x y ≤ B*capacityProfile d M (j+1) y)
  50     (hminor : ∀ i j, s i ≤ j → j+3 ≤ s i+L i → j ∉ J → j+1 ∉ J → j+2 ∉ J →
  51       ∀ x y, κ*capacityProfile d M (j+3) y ≤ transitionWalk (capacityTransition a d) j 3 x y)
  52     (n a₀ a₁ d₀ d₁ : ℝ) (hn : (Fintype.card X : ℝ)=n) (hnpos : 0<n)
  53     (ha₀ : 0<a₀) (ha₁ : 0<a₁) (hd₀ : 0<d₀) (hd₁ : 0<d₁)
  54     (hdscale : ∀ i x, d₀*n ≤ d i x ∧ d i x ≤ d₁*n) :
  55     ∃ ω : (∀ i,Fin (L i+1) → X) → ℝ,
  56       (∀ p,0≤ω p) ∧ (∑ p,ω p)=M ∧
  57       (∀ (i : Fin 3) (j : Fin (L i)) (x y : X),
  58         (∑ p,if p i j.castSucc=x ∧ p i j.succ=y then ω p else 0)=a (s i+j.val) x y) ∧
  59       (∀ p,0<ω p → ∀ i,p i (Fin.last (L i))=p (finRotate 3 i) 0) ∧
  60       (∀ p,(∀ i,p i (Fin.last (L i))=p (finRotate 3 i) 0) →
  61         (∀ i (j : Fin (L i)),a₀ ≤ a (s i+j.val) (p i j.castSucc) (p i j.succ) ∧
  62           a (s i+j.val) (p i j.castSucc) (p i j.succ) ≤ a₁) →
  63         longCycleLower L a₀ d₀ d₁/n^((∑ i,L i)-2) ≤ ω p ∧
  64           ω p ≤ longCycleUpper L a₁ d₀ d₁/n^((∑ i,L i)-2)) := by
  65   obtain ⟨Q,ω,hω0,hωM,hωedge,hωsupp,hfixed,hformula⟩ :=
  66     long_capacity_interval_assembly a d ha hd hrow hcol M hM hms s L t hbd J hJ κ B hκ hκ1 hB ht hL hupper hminor
  67   obtain ⟨hmix,hres⟩ := long_capacity_mixing_reserve a d ha hd hrow hcol M hM hms s L t hbd J hJ κ B hκ hκ1 hB ht hL hupper hminor
  68   refine ⟨ω,hω0,hωM,hωedge,hωsupp,?_⟩
  69   intro p hp hap
  70   have hb (c : Fin 3) := long_capacity_sector_bounds a d ha hd hrow hcol M n a₀ a₁ d₀ d₁
  71     hn hnpos ha₀ ha₁ hd₀ hd₁ hms hdscale s L (fun i => by have := hL i;omega) hbd
  72     hmix hres Q hfixed p hp hap c
  73   have hlo := Finset.sum_le_sum (fun c (_ : c∈Finset.univ) => (hb c).1)
  74   have hup := Finset.sum_le_sum (fun c (_ : c∈Finset.univ) => (hb c).2)
  75   rw [← Finset.sum_div] at hlo hup
