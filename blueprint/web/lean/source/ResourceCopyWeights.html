  17 theorem resource_copy_matching_projection_weights
  18     {V E : Type*} [Fintype E] [DecidableEq V] [DecidableEq E]
  19     (R : E → Finset V) (m : E → ℕ) (hR : ∀ e,(R e).Nonempty)
  20     (M : Finset (Σ e,Fin (m e)))
  21     (hM : ∀ c∈M,∀ d∈M,c≠d → Disjoint (R c.1) (R d.1)) :
  22     (∀ e∈M.image Sigma.fst,∀ f∈M.image Sigma.fst,e≠f → Disjoint (R e) (R f)) ∧
  23     (∀ g : E → ℝ,(∑ e∈M.image Sigma.fst,g e)=∑ c∈M,g c.1) := by
  24   have hinj : Set.InjOn Sigma.fst (M : Set (Σ e,Fin (m e))) := by
  25     intro c hc d hd heq
  26     by_contra hne
  27     obtain ⟨v,hv⟩ := hR c.1
  28     have hdis := hM c hc d hd hne
  29     rw [← heq] at hdis
  30     exact Finset.disjoint_left.mp hdis hv hv
  31   constructor
  32   · intro e he f hf hef
  33     obtain ⟨c,hc,rfl⟩ := Finset.mem_image.mp he
  34     obtain ⟨d,hd,rfl⟩ := Finset.mem_image.mp hf
  35     exact hM c hc d hd (fun hh=>hef (congrArg Sigma.fst hh))
  36   · intro g
  37     exact Finset.sum_image hinj

  39 theorem sum_floor_weighted_support_bounds
  40     {E : Type*} [Fintype E] (w g : E → ℝ) (T : ℕ) (b : ℝ)
  41     (hw : ∀ e,0≤w e) (hb : 0≤b) (hg : ∀ e,0≤g e ∧ g e≤b) :
  42     (T : ℝ)*(∑ e,g e*w e)-(Finset.univ.filter (fun e=>0<w e)).card*b≤
  43       ∑ e,g e*⌊(T : ℝ)*w e⌋₊ ∧
  44     (∑ e,g e*⌊(T : ℝ)*w e⌋₊)≤(T : ℝ)*(∑ e,g e*w e) := by
  45   have hlow (e : E) : (T : ℝ)*(g e*w e)-(if 0<w e then b else 0)≤g e*⌊(T : ℝ)*w e⌋₊ := by
  46     by_cases he : 0<w e
  47     · simp only [he,ite_true]
  48       have hf := Nat.lt_floor_add_one ((T : ℝ)*w e)
  49       have hm := mul_le_mul_of_nonneg_left hf.le (hg e).1
  50       nlinarith [(hg e).2]
  51     · have hz : w e=0 := le_antisymm (le_of_not_gt he) (hw e)
  52       simp [hz]
  53   have hupp (e : E) : g e*⌊(T : ℝ)*w e⌋₊≤(T : ℝ)*(g e*w e) := by
  54     have hf := Nat.floor_le (mul_nonneg (Nat.cast_nonneg T) (hw e))
  55     have hm := mul_le_mul_of_nonneg_left hf (hg e).1
  56     nlinarith
  57   constructor
  58   · have hs := Finset.sum_le_sum (fun e (_ : e∈Finset.univ)=>hlow e)
  59     simpa only [Finset.sum_sub_distrib,← Finset.mul_sum,← Finset.sum_filter,Finset.sum_const,
  60       nsmul_eq_mul] using hs
  61   · simpa only [← Finset.mul_sum] using Finset.sum_le_sum (fun e (_ : e∈Finset.univ)=>hupp e)
