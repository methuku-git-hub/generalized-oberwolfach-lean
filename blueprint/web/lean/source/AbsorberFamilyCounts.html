  18 theorem labelled_absorber_family_occurrence_count (R : C → Finset V)
  19     (S : T → VertexAbsorberTemplate c) (X : Finset V) (C₀ : C) (B : ℕ)
  20     (hB : ∀ t o,Fintype.card {D : LabelledCycleRecord R (S t).val X // (D.2 o).val=C₀} ≤ B) :
  21     Fintype.card {D : LabelledAbsorberFamily R S X // ∃ o,(D.2.2 o).val=C₀} ≤
  22       (2*c+1)*Fintype.card T*B := by
  23   let F := LabelledAbsorberFamily R S X
  24   let P (o : Fin c ⊕ Fin (c+1)) := Finset.univ.filter (fun D : F => (D.2.2 o).val=C₀)
  25   have hp (o : Fin c ⊕ Fin (c+1)) : (P o).card ≤ Fintype.card T*B := by
  26     let e : {D : F // (D.2.2 o).val=C₀} ≃
  27         Σ t,{D : LabelledCycleRecord R (S t).val X // (D.2 o).val=C₀} := {
  28       toFun D := ⟨D.val.1,⟨D.val.2,D.property⟩⟩
  29       invFun D := ⟨⟨D.1,D.2.val⟩,D.2.property⟩
  30       left_inv := by rintro ⟨⟨t,D⟩,h⟩;rfl
  31       right_inv := by rintro ⟨t,⟨D,h⟩⟩;rfl }
  32     have he := Fintype.card_congr e
  33     calc
  34       _ = ∑ t,Fintype.card {D : LabelledCycleRecord R (S t).val X // (D.2 o).val=C₀} := by
  35         simpa only [Fintype.card_subtype,Fintype.card_sigma,P] using he
  36       _ ≤ ∑ _t : T,B := Finset.sum_le_sum (fun t _ => hB t o)
  37       _ = _ := by simp
  38   have he : Finset.univ.filter (fun D : F => ∃ o,(D.2.2 o).val=C₀)=Finset.univ.biUnion P := by
  39     ext D;simp [P]
  40   rw [Fintype.card_subtype,he]
  41   apply Finset.card_biUnion_le.trans
  42   calc
  43     _ ≤ ∑ _o : Fin c ⊕ Fin (c+1),Fintype.card T*B := Finset.sum_le_sum (fun o _ => hp o)
  44     _ = _ := by
  45       simp only [Finset.sum_const,Finset.card_univ,Fintype.card_sum,Fintype.card_fin,smul_eq_mul]
  46       ring

  48 theorem labelled_absorber_subfamily_occurrence_count (R : C → Finset V)
  49     (S : T → VertexAbsorberTemplate c) (X : Finset V) (C₀ : C) (B : ℕ)
  50     (hB : ∀ t o,Fintype.card {D : LabelledCycleRecord R (S t).val X // (D.2 o).val=C₀} ≤ B)
  51     (A : Finset (LabelledAbsorberFamily R S X)) :
  52     Fintype.card {D : A // ∃ o,(D.val.2.2 o).val=C₀} ≤ (2*c+1)*Fintype.card T*B := by
  53   let enc (D : {D : A // ∃ o,(D.val.2.2 o).val=C₀}) :
  54       {D : LabelledAbsorberFamily R S X // ∃ o,(D.2.2 o).val=C₀} := ⟨D.val.val,D.property⟩
  55   have hinj : Function.Injective enc := by
  56     intro D E he;apply Subtype.ext;apply Subtype.ext
  57     exact congrArg (fun z : {D : LabelledAbsorberFamily R S X // ∃ o,(D.2.2 o).val=C₀} => z.val) he
  58   exact (Fintype.card_le_of_injective enc hinj).trans
  59     (labelled_absorber_family_occurrence_count R S X C₀ B hB)
