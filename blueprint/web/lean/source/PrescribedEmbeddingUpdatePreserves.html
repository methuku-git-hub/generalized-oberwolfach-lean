   8 theorem prescribed_embedding_update_preserves {X V:Type} [DecidableEq X] [DecidableEq V]
   9     (A:Finset X) (f:X→V) (x:X) (z:V) (hx:x∉A):
  10     (∀ a∈A,Function.update f x z a=f a) ∧
  11     (insert x A).image (Function.update f x z)=insert z (A.image f) := by
  12   constructor
  13   · intro a ha
  14     have hax:a≠x:=by intro he;subst a;exact hx ha
  15     exact Function.update_of_ne hax z f
  16   · ext v
  17     simp only [Finset.mem_image,Finset.mem_insert]
  18     constructor
  19     · rintro ⟨a,ha,hav⟩
  20       rcases ha with rfl|ha
  21       · left;simpa using hav.symm
  22       · right
  23         have hax:a≠x:=by intro he;subst a;exact hx ha
  24         refine ⟨a,ha,?_⟩
  25         simpa [Function.update_of_ne hax] using hav
  26     · rintro (rfl|⟨a,ha,hav⟩)
  27       · exact ⟨x,Or.inl rfl,by simp⟩
  28       · have hax:a≠x:=by intro he;subst a;exact hx ha
  29         exact ⟨a,Or.inr ha,by simpa [Function.update_of_ne hax] using hav⟩
  30 end Oberwolfach
