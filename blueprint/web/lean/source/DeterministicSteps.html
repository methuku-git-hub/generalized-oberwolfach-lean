   8 /-- The local-scale conversion used in both cover-down inductions. -/
   9 theorem local_scale_error {δ C σ ε m n : ℝ}
  10     (hε : 0 < ε) (hC : 0 ≤ C) (hσ : 0 ≤ σ) (hn : 0 ≤ n)
  11     (hsize : ε * m / 2 ≤ n) (hmargin : 2 * C * σ / ε ≤ δ) :
  12     δ * n + C * σ * m ≤ 2 * δ * n := by
  13   have hstep := mul_le_mul_of_nonneg_left hsize (mul_nonneg hC hσ)
  14   have hb : 2 * C * σ ≤ δ * ε := (div_le_iff₀ hε).mp hmargin
  15   have hscaled := mul_le_mul_of_nonneg_right hb hn
  16   have : C * σ * m ≤ δ * n := by nlinarith
  17   linarith

  19 /-- Scalar calculation in `lem:pr:cut-stability`, using the manuscript's
  20 normalization min(a2*x,a1*y), rather than a product-density normalization.
  21 The graph-theoretic bound on the number of removed edges is an explicit input. -/
  22 theorem large_cut_stability_arithmetic {a1 a2 x y h b original remaining : ℝ}
  23     (ha1 : 0 < a1) (ha2 : 0 < a2) (hx : 0 ≤ x) (hy : 0 ≤ y)
  24     (hb : 0 ≤ b)
  25     (hcut : h * min (a2 * x) (a1 * y) ≤ original)
  26     (hdelete : original - b * min x y ≤ remaining) :
  27     (h - b / min a1 a2) * min (a2 * x) (a1 * y) ≤ remaining := by
  28   have hm : 0 < min a1 a2 := lt_min ha1 ha2
  29   have hscale : min a1 a2 * min x y ≤ min (a2 * x) (a1 * y) := by
  30     apply le_min
  31     · exact mul_le_mul (min_le_right _ _) (min_le_left _ _) (le_min hx hy) ha2.le
  32     · exact mul_le_mul (min_le_left _ _) (min_le_right _ _) (le_min hx hy) ha1.le
  33   have hscaled := mul_le_mul_of_nonneg_left hscale (div_nonneg hb hm.le)
  34   have hid : b / min a1 a2 * (min a1 a2 * min x y) = b * min x y := by
  35     field_simp
  36   rw [hid] at hscaled
  37   nlinarith

  39 /-- Universal estimates on one event remain valid for outcome-dependent choices.
  40 No intersection of separately chosen events occurs in this implication. -/
  41 theorem common_event_adaptive_choice {Ω X : Type*} (E : Set Ω)
  42     (admissible conclusion : Ω → X → Prop)
  43     (h : ∀ ω ∈ E, ∀ x, admissible ω x → conclusion ω x)
  44     (choose : Ω → X) (hc : ∀ ω ∈ E, admissible ω (choose ω)) :
  45     E ⊆ {ω | conclusion ω (choose ω)} := by
  46   intro ω hω
  47   exact h ω hω (choose ω) (hc ω hω)

  49 /-- Exact set identity behind the non-accumulation of deletion errors. -/
  50 theorem protected_restriction {V : Type*} (C E T Next : Set V)
  51     (hC : C ∩ T = Next) (hE : E ∩ T = ∅) : (C \ E) ∩ T = Next := by
  52   rw [← hC]
  53   ext x
  54   have he : x ∈ E → x ∉ T := by
  55     intro hx ht
  56     have : x ∈ E ∩ T := ⟨hx, ht⟩
  57     rw [hE] at this
  58     exact this
  59   simp only [Set.mem_inter_iff, Set.mem_sdiff]
  60   tauto
