   9 theorem permutation_maps_path_complement
  10     {X:Type*} [Fintype X] [DecidableEq X]
  11     (σ:Equiv.Perm X) (A Y B T:Finset X)
  12     (hA:Y∪B=A) (hshift:Y∪B=Y.image σ∪T)
  13     (hYB:Disjoint Y B) (hYT:Disjoint (Y.image σ) T):
  14     ((Finset.univ\A)∪B).image σ=(Finset.univ\A)∪T:=by
  15   classical
  16   have himage (S:Finset X) (x:X):x∈S.image σ↔σ.symm x∈S:=by
  17     constructor
  18     · intro h
  19       obtain ⟨z,hz,hzx⟩:=Finset.mem_image.mp h
  20       rw [←hzx,Equiv.symm_apply_apply]
  21       exact hz
  22     · intro h
  23       exact Finset.mem_image.mpr ⟨σ.symm x,h,σ.apply_symm_apply x⟩
  24   have ha (x:X):x∈A↔x∈Y∨x∈B:=by rw [←hA,Finset.mem_union]
  25   have hb (x:X):x∈A↔x∈Y.image σ∨x∈T:=by rw [←hA,hshift,Finset.mem_union]
  26   ext x
  27   rw [himage]
  28   simp only [Finset.mem_union,Finset.mem_sdiff,Finset.mem_univ,true_and]
  29   have he:=himage Y x
  30   have hd1:¬(σ.symm x∈Y ∧ σ.symm x∈B):=fun h=>Finset.disjoint_left.mp hYB h.1 h.2
  31   have hd2:¬(x∈Y.image σ ∧ x∈T):=fun h=>Finset.disjoint_left.mp hYT h.1 h.2
  32   have h1:=ha (σ.symm x)
  33   have h2:=hb x
  34   tauto
