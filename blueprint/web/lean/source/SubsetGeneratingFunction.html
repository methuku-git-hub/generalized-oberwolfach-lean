  11 theorem intersection_power_expansion {V : Type*} (A S : Finset V) (q : ℝ) :
  12     q ^ (A ∩ S).card = ∑ T ∈ A.powerset,
  13       (q - 1) ^ T.card * (if T ⊆ S then 1 else 0) := by
  14   have hp : (∏ x ∈ A, (1 + (q - 1) * (if x ∈ S then (1 : ℝ) else 0))) =
  15       q ^ (A ∩ S).card := by
  16     have he (x : V) : 1 + (q - 1) * (if x ∈ S then (1 : ℝ) else 0) =
  17         if x ∈ S then q else 1 := by split_ifs <;> ring
  18     simp_rw [he]
  19     simp
  20   rw [← hp, Finset.prod_one_add]
  21   apply Finset.sum_congr rfl
  22   intro T _
  23   rw [Finset.prod_mul_distrib, Finset.prod_const, Finset.prod_boole]
  24   rfl

  26 theorem finiteLaw_intersection_power_expectation
  27     {V : Type*} [Fintype V] {r : ℕ} (μ : FiniteLaw (FixedSubset V r))
  28     (A : Finset V) (q : ℝ) :
  29     μ.expectation (fun S => q ^ (A ∩ S.1).card) =
  30       ∑ T ∈ A.powerset, (q - 1) ^ T.card * μ.probability (fun S => T ⊆ S.1) := by
  31   simp only [FiniteLaw.expectation, intersection_power_expansion, Finset.mul_sum]
  32   rw [Finset.sum_comm]
  33   apply Finset.sum_congr rfl
  34   intro T _
  35   unfold FiniteLaw.probability
  36   rw [Finset.mul_sum]
  37   apply Finset.sum_congr rfl
  38   intro S _
  39   by_cases h : T ⊆ S.1 <;> simp [h, mul_comm]
