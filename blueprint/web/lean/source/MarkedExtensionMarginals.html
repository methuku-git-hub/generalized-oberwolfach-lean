   9 theorem marked_extension_marginal_lower (m : ℕ) (hm : 1≤m)
  10     (ρ s ρ₀ s₀ : ℝ) (hρ₀ : 0<ρ₀) (hs₀ : 0<s₀)
  11     (hρ : ρ₀≤ρ) (hρ1 : ρ≤1) (hs : s₀≤s) (hs1 : s≤1) :
  12     let pstar:=ρ₀*s₀/(2*m)
  13     let p:=ρ*s/2
  14     let k:=p/m
  15     let t:=s/m
  16     0<pstar ∧ pstar≤1 ∧ pstar≤p ∧ pstar≤k ∧ pstar≤s ∧ pstar≤t := by
  17   intro pstar p k t
  18   have hm1 : (1:ℝ)≤m := by exact_mod_cast hm
  19   have hmpos : (0:ℝ)<m := by linarith only [hm1]
  20   have hρpos : 0<ρ := hρ₀.trans_le hρ
  21   have hspos : 0<s := hs₀.trans_le hs
  22   have hpstar : 0<pstar := by dsimp [pstar];positivity
  23   have hp : 0<p := by dsimp [p];positivity
  24   have hprod:=mul_le_mul hρ hs hs₀.le hρpos.le
  25   have hklo : pstar≤k := by
  26     dsimp [pstar,k,p]
  27     rw [div_div]
  28     exact div_le_div_of_nonneg_right hprod (by positivity)
  29   have hkp : k≤p := by
  30     dsimp [k]
  31     apply (div_le_iff₀ hmpos).mpr
  32     nlinarith only [mul_le_mul_of_nonneg_left hm1 hp.le]
  33   have hps : p≤s := by
  34     dsimp [p]
  35     nlinarith only [mul_le_mul_of_nonneg_right hρ1 hspos.le,hspos.le]
  36   have hkt : k≤t := div_le_div_of_nonneg_right hps hmpos.le
  37   exact ⟨hpstar,(hklo.trans hkp).trans (hps.trans hs1),hklo.trans hkp,hklo,
  38     (hklo.trans hkp).trans hps,hklo.trans hkt⟩
