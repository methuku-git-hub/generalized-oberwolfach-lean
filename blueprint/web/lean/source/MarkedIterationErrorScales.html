  11 theorem marked_iteration_reference_scale
  12     (h : ℕ) (κ γ p k omega C : ℝ)
  13     (hκ : 0<κ) (hγ : 0<γ) (hp : 0<p) (hk : 0<k) (ho : 0<omega) (hC : 0≤C) :
  14     ∃ ζmax : ℝ,0<ζmax ∧ ∀ ε ζ : ℝ,ε≤κ/4→0≤ζ→ζ≤ζmax→
  15       ε+ζ≤κ ∧ ε+4*ζ≤κ ∧ ε+2*h*ζ≤κ ∧ ε+h*ζ≤κ ∧
  16       ζ≤(κ-ε)*omega ∧ ε≤κ ∧ 2*ζ≤(3/4)*γ*p^2 ∧ ζ≤γ*min p k/8 ∧ C*ζ≤κ/4 := by
  17   let ζmax := min (κ/8) (min (κ/(8*((h:ℝ)+1)))
  18     (min (κ*omega/8) (min (γ*p^2/8) (min (γ*min p k/16) (κ/(4*(C+1)))))))
  19   have hz : 0<ζmax := by dsimp [ζmax];positivity
  20   refine ⟨ζmax,hz,?_⟩
  21   intro ε ζ he hζ hzmax
  22   have hz1 : ζ≤κ/8 := hzmax.trans (min_le_left _ _)
  23   have hzt := hzmax.trans (min_le_right _ _)
  24   have hz2 : ζ≤κ/(8*((h:ℝ)+1)) := hzt.trans (min_le_left _ _)
  25   have hzt := hzt.trans (min_le_right _ _)
  26   have hz3 : ζ≤κ*omega/8 := hzt.trans (min_le_left _ _)
  27   have hzt := hzt.trans (min_le_right _ _)
  28   have hz4 : ζ≤γ*p^2/8 := hzt.trans (min_le_left _ _)
  29   have hzt := hzt.trans (min_le_right _ _)
  30   have hz5 : ζ≤γ*min p k/16 := hzt.trans (min_le_left _ _)
  31   have hz6 : ζ≤κ/(4*(C+1)) := hzt.trans (min_le_right _ _)
  32   have hz2' := (le_div_iff₀ (show 0<8*((h:ℝ)+1) by positivity)).mp hz2
  33   have hz6' := (le_div_iff₀ (show 0<4*(C+1) by positivity)).mp hz6
  34   have hω := mul_le_mul_of_nonneg_right he ho.le
  35   have hgpp : 0≤γ*p^2 := by positivity
  36   have hgpk : 0≤γ*min p k := by positivity
  37   refine ⟨by linarith only [he,hz1,hκ],by linarith only [he,hz1,hκ],
  38     by nlinarith only [he,hz2',hζ,hκ],by nlinarith only [he,hz2',hζ,hκ],
  39     by nlinarith only [hz3,hω,mul_pos hκ ho],by linarith only [he,hκ],
  40     by nlinarith only [hz4,hgpp],by linarith only [hz5,hgpk],?_⟩
  41   nlinarith only [hz6',hζ]

  43 theorem marked_iteration_star_scale
  44     (m : ℕ) (x eps C b theta : ℝ)
  45     (hx : 0<x) (heps : 0<eps) (hC : 0≤C) (hb : 0<b) (htheta : 0<theta) :
  46     ∃ sigmaMax : ℝ,0<sigmaMax ∧ ∀ sigma : ℝ,0≤sigma→sigma≤sigmaMax→
  47       (m+2)*sigma≤x/4 ∧ 4*C*sigma/eps≤x/4 ∧
  48       4*C*sigma/eps≤b/4 ∧ 4*C*sigma/eps≤theta/4 := by
  49   let sigmaMax := min (x/(4*((m:ℝ)+2))) (eps*min x (min b theta)/(16*(C+1)))
  50   refine ⟨sigmaMax,by dsimp [sigmaMax];positivity,?_⟩
  51   intro sigma hs hsmall
  52   have h1 := hsmall.trans (min_le_left _ _)
  53   have h2 := hsmall.trans (min_le_right _ _)
  54   have h1' := (le_div_iff₀ (show 0<4*((m:ℝ)+2) by positivity)).mp h1
  55   have h2' := (le_div_iff₀ (show 0<16*(C+1) by positivity)).mp h2
  56   have h2x := mul_le_mul_of_nonneg_left (min_le_left x (min b theta)) heps.le
  57   have h2b := mul_le_mul_of_nonneg_left ((min_le_right x (min b theta)).trans (min_le_left b theta)) heps.le
  58   have h2t := mul_le_mul_of_nonneg_left ((min_le_right x (min b theta)).trans (min_le_right b theta)) heps.le
  59   refine ⟨by nlinarith only [h1'],?_,?_,?_⟩
  60   all_goals apply (div_le_iff₀ heps).mpr
  61   · nlinarith only [h2',h2x,hs]
  62   · nlinarith only [h2',h2b,hs]
  63   · nlinarith only [h2',h2t,hs]

  65 theorem marked_iteration_local_normalization
  66     (C sigma eps n a budget : ℝ)
  67     (hC : 0≤C) (hs : 0≤sigma) (heps : 0<eps) (hn : 0≤n) (ha : 0<a)
  68     (hsize : eps*n/4≤a) (hbudget : 4*C*sigma/eps≤budget) :
  69     C*(sigma*n)/a≤budget := by
  70   apply le_trans _ hbudget
  71   apply (div_le_iff₀ ha).mpr
  72   have hm := mul_le_mul_of_nonneg_left hsize (show 0≤4*C*sigma/eps by positivity)
  73   have he : (4*C*sigma/eps)*(eps*n/4)=C*(sigma*n) := by field_simp
  74   rw [he] at hm
  75   exact hm
