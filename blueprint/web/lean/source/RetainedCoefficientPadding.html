  10 theorem retained_coefficient_padding {I U:Type} [Fintype I] [DecidableEq I] [Fintype U]
  11     (fraction:U→I→ℝ) (threshold θ:ℝ) (b:U→ℝ):
  12     let S:=Finset.univ.filter (fun i=>∃ u,threshold≤fraction u i)
  13     let f:=fun i u=>if threshold≤fraction u i then fraction u i else 0
  14     let ηS:=fun (i:↥S) u=>(1-θ)*f i.val u/b u;
  15     ∀ i u,padIndexedFunctions S (fun _=>U) ηS i u=(1-θ)*f i u/b u:=by
  16   intro S f ηS i u
  17   by_cases hi:i∈S
  18   · simpa [ηS] using pad_indexed_functions_on S (fun _=>U) ηS ⟨i,hi⟩ u
  19   · have hf:¬threshold≤fraction u i:=by
  20       intro hh
  21       apply hi
  22       exact Finset.mem_filter.mpr ⟨Finset.mem_univ i,u,hh⟩
  23     rw [pad_indexed_functions_off S (fun _=>U) ηS i hi u]
  24     simp [f,hf]
  25 end Oberwolfach
