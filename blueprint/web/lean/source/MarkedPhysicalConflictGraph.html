  14 def markedPhysicalConflictGraph (D:SimpleGraph V) (π:Equiv.Perm V) : SimpleGraph V where
  15   Adj x y:=x≠y ∧ (D.Adj x y ∨ D.Adj x (π y) ∨ D.Adj y (π x))
  16   symm:=⟨by
  17     intro x y h
  18     refine ⟨h.1.symm,?_⟩
  19     rcases h.2 with h|h|h
  20     · exact Or.inl h.symm
  21     · exact Or.inr (Or.inr h)
  22     · exact Or.inr (Or.inl h)⟩
  23   loopless:=⟨by intro x h;exact h.1 rfl⟩

  25 theorem marked_physical_conflict_degree (D:SimpleGraph V) (π:Equiv.Perm V) (x:V) :
  26     (markedPhysicalConflictGraph D π).degree x≤2*D.degree x+D.degree (π x) := by
  27   have hs : (markedPhysicalConflictGraph D π).neighborFinset x⊆
  28       (D.neighborFinset x)∪((D.neighborFinset x).image π.symm)∪D.neighborFinset (π x) := by
  29     intro y hy
  30     have hh:=(SimpleGraph.mem_neighborFinset _ _ _).mp hy
  31     rcases hh.2 with h|h|h
  32     · exact Finset.mem_union_left _ (Finset.mem_union_left _ ((SimpleGraph.mem_neighborFinset _ _ _).mpr h))
  33     · exact Finset.mem_union_left _ (Finset.mem_union_right _ (Finset.mem_image.mpr
  34         ⟨π y,(SimpleGraph.mem_neighborFinset _ _ _).mpr h,π.symm_apply_apply y⟩))
  35     · exact Finset.mem_union_right _ ((SimpleGraph.mem_neighborFinset _ _ _).mpr h.symm)
  36   rw [←SimpleGraph.card_neighborFinset_eq_degree]
  37   calc
  38     _ ≤ _ := Finset.card_le_card hs
  39     _ ≤ ((D.neighborFinset x)∪((D.neighborFinset x).image π.symm)).card+(D.neighborFinset (π x)).card := Finset.card_union_le _ _
  40     _ ≤ (D.neighborFinset x).card+((D.neighborFinset x).image π.symm).card+(D.neighborFinset (π x)).card := Nat.add_le_add_right (Finset.card_union_le _ _) _
  41     _ ≤ 2*D.degree x+D.degree (π x) := by
  42       have hh : ((D.neighborFinset x).image π.symm).card≤(D.neighborFinset x).card:=Finset.card_image_le
  43       simp only [SimpleGraph.card_neighborFinset_eq_degree] at hh ⊢
  44       omega

  46 theorem marked_physical_conflict_degree_bound (D:SimpleGraph V) (π:Equiv.Perm V) (δ:ℝ)
  47     (hD:∀ x,(D.degree x:ℝ)≤δ) (x:V) :
  48     ((markedPhysicalConflictGraph D π).degree x:ℝ)≤3*δ := by

  53 theorem no_adjacency_of_pair_equality (D:SimpleGraph V) {a b x y:V}
  54     (hab:¬D.Adj a b) (he:({a,b}:Finset V)={x,y}) : ¬D.Adj x y := by
  55   intro hxy
  56   have hx:x=a∨x=b:=by
  57     have h:x∈({a,b}:Finset V):=by rw [he];simp
  58     simpa only [Finset.mem_insert,Finset.mem_singleton] using h
  59   have hy:y=a∨y=b:=by
  60     have h:y∈({a,b}:Finset V):=by rw [he];simp
  61     simpa only [Finset.mem_insert,Finset.mem_singleton] using h
  62   rcases hx with rfl|rfl <;> rcases hy with rfl|rfl
  63   · exact D.irrefl hxy
  64   · exact hab hxy
  65   · exact hab hxy.symm
  66   · exact D.irrefl hxy
