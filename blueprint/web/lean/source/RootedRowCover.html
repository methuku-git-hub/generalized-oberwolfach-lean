  60 theorem rooted_ordinary_row_cover_family {n K:ℕ}

  62     (hx:Function.Injective x) (hxA:∀ i,x i∉A)
  63     (hM:∀ i,OrdinaryRowCover (M i) (x i) A K)
  64     (hcross:∀ i j,i≠j→Disjoint ((M i).biUnion ordinaryWheelResources) ((M j).biUnion ordinaryWheelResources)):
  65     ∃ N:Fin n→Finset (OrdinaryWheel Q F H r),
  66       (∀ i,N i⊆M i) ∧ (∀ i,OrdinaryRowCover (N i) (x i) A K) ∧
  67       (∀ i j,i≠j→Disjoint ((N i).biUnion ordinaryWheelResources) ((N j).biUnion ordinaryWheelResources)) ∧
  68       (∀ i,∀ W∈N i,∀ k,x k∈(ordinaryWheelRim W).support ↔ k=i) ∧
  69       (∀ a∈A,(ordinaryWheelRimGraph (Finset.univ.biUnion N)).degree a≤n*(2*(K+1))) ∧
  70       (∀ a∈A,(Finset.univ.filter (ordinaryPackingSpokes (Finset.univ.biUnion N) a)).card≤n*(K+1)) ∧
  71       (∀ u,(Finset.univ.filter (fun a=>ordinaryPackingSpokes (Finset.univ.biUnion N) a u)).card≤n*r):=by
  72   let N:=fun i=>(M i).filter (fun W=>x i∈(ordinaryWheelRim W).support)
  73   have hNM (i:Fin n):N i⊆M i:=Finset.filter_subset _ _
  74   have hc (i:Fin n):OrdinaryRowCover (N i) (x i) A K:=(hM i).filter_root
  75   refine ⟨N,hNM,hc,?_,?_,row_cover_union_degree_bounds N x A hxA hc⟩
