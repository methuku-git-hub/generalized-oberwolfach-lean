  13  theorem finite_subtype_image_filter_card {A B:Type} [DecidableEq A] [DecidableEq B]
  14     (S:Finset A) (f:S→B) (hf:Function.Injective f) (p:B→Prop) (q:A→Prop)
  15     (hp:∀ a:S,p (f a)↔q a.val):
  16     ((Finset.univ.image f).filter p).card=(S.filter q).card:=by
  17   rw [Finset.filter_image,Finset.card_image_of_injective _ hf]
  18   have he:(Finset.univ.filter (fun a:S=>p (f a)))=Finset.univ.filter (fun a:S=>q a.val):=by
  19     ext a
  20     simp only [Finset.mem_filter,Finset.mem_univ,true_and,hp]
  21   rw [he]
  22   exact (Fintype.card_subtype (fun a:S=>q a.val)).symm.trans (marked_finite_filter_card S q)
