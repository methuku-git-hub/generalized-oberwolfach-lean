  15 theorem fixed_domain_common_event (κ:ℝ) (hκ:0<κ) (q:ℕ):
  16     ∃ N:ℕ,∀ n:ℕ,N≤n→∀ (Ω J:Type) [Fintype Ω] [Fintype J],Fintype.card J≤n→
  17       ∀ (μ:FiniteLaw Ω) (s:J→ℕ) (A:J→Ω→Prop),
  18       (∀ j,κ*n≤(s j:ℝ))→
  19       (∀ j,1-1/(s j:ℝ)^(q+2)≤μ.probability (A j))→
  20       1-1/(n:ℝ)^q≤μ.probability (fun ω=>∀ j,A j ω):=by
  21   obtain ⟨N,hN⟩:=exists_nat_ge (1/κ^(q+2))
  22   refine ⟨max N 1,?_⟩
  23   intro n hn Ω J _ _ hJ μ s A hs hA
  24   have hn0:0<n:=by omega
  25   have hnR:(0:ℝ)<n:=Nat.cast_pos.mpr hn0
  26   have hκn:0<κ*n:=mul_pos hκ hnR
  27   have hbad (j:J):μ.probability (fun ω=>¬A j ω)≤1/(κ*n)^(q+2):=by
  28     have hcomp:=finiteLaw_probability_compl μ (A j)
  29     have hlocal:μ.probability (fun ω=>¬A j ω)≤1/(s j:ℝ)^(q+2):=by linarith only [hcomp,hA j]
  30     exact hlocal.trans (div_le_div_of_nonneg_left (by norm_num) (pow_pos hκn _)
  31       (pow_le_pow_left₀ hκn.le (hs j) _))
  32   have hevent:=finiteLaw_common_event_lower μ Finset.univ (fun j ω=>¬A j ω)
  33     (fun _=>1/(κ*n)^(q+2)) (fun j _=>hbad j)
  34   simp only [not_not,Finset.mem_univ,true_implies,Finset.sum_const,Finset.card_univ,nsmul_eq_mul] at hevent
  35   have hsize:(Fintype.card J:ℝ)*(1/(κ*n)^(q+2))≤(n:ℝ)*(1/(κ*n)^(q+2)):=
  36     mul_le_mul_of_nonneg_right (Nat.cast_le.mpr hJ) (by positivity)
  37   have he:(n:ℝ)*(1/(κ*n)^(q+2))=(1/κ^(q+2))/(n:ℝ)^(q+1):=by
  38     rw [mul_pow,show q+2=(q+1)+1 by omega,pow_succ (n:ℝ)]
  39     field_simp
  40   have hC:1/κ^(q+2)≤(n:ℝ):=hN.trans (by exact_mod_cast (show N≤n by omega))
  41   have hsmall:=hsize.trans (he.trans_le (scaled_inverse_power_le n q hn0 _ hC))
  42   exact (sub_le_sub_left hsmall 1).trans hevent
