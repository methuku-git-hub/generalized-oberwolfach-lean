   4 /-! Choosing an internal path vertex using an actual common-neighbour
   5 count. Vertex exclusions and forbidden graph neighbours are subtracted

  15 theorem sparse_common_neighbour_avoiding (Q F : SimpleGraph V) (A W Z : Finset V)
  16     (x y : V) (L b : ℝ) (hL : 0≤L) (hF : ∀ v,(F.degree v:ℝ)≤L)
  17     (hcount : b≤((A∩Q.neighborFinset x∩Q.neighborFinset y).card:ℝ))
  18     (hbudget : W.card+(Z.card:ℝ)*(L+1)<b) :
  19     ∃ v∈A,v∉W ∧ v∉Z ∧ Q.Adj x v ∧ Q.Adj y v ∧ ∀ z∈Z,¬F.Adj v z := by
  20   let D:=W∪(Z∪Z.biUnion (fun z=>F.neighborFinset z))
  21   have hZ:=forbidden_used_neighbours_card F Z L hF
  22   have hD : (D.card:ℝ)≤W.card+(Z.card:ℝ)*(L+1):=by
  23     have hh : (D.card:ℝ)≤W.card+(Z∪Z.biUnion (fun z=>F.neighborFinset z)).card:=by
  24       exact_mod_cast Finset.card_union_le W (Z∪Z.biUnion (fun z=>F.neighborFinset z))
  25     linarith only [hh,hZ]
  26   have hlt : D.card<(A∩Q.neighborFinset x∩Q.neighborFinset y).card:=by
  27     exact_mod_cast hD.trans_lt (hbudget.trans_le hcount)
  28   obtain ⟨v,hv,hvD⟩:=Finset.exists_mem_notMem_of_card_lt_card hlt
  29   obtain ⟨hvAx,hvy⟩:=Finset.mem_inter.mp hv
  30   obtain ⟨hvA,hvx⟩:=Finset.mem_inter.mp hvAx
  31   have hout : v∉W ∧ v∉Z ∧ v∉Z.biUnion (fun z=>F.neighborFinset z):=by
  32     simpa only [D,Finset.mem_union,not_or] using hvD
  33   refine ⟨v,hvA,hout.1,hout.2.1,(Q.mem_neighborFinset x v).mp hvx,
  34     (Q.mem_neighborFinset y v).mp hvy,?_⟩
  35   intro z hz he
  36   exact hout.2.2 (Finset.mem_biUnion.mpr ⟨z,hz,(F.mem_neighborFinset z v).mpr he.symm⟩)
  37
