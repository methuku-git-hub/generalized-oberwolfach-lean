  15 theorem copied_marked_support_iff {V E K : Type*} [Fintype E] [Fintype K] [DecidableEq V]
  16     (tail head : E → V) (ktail khead : K → V)
  17     (hinj : Function.Injective (fun e => (tail e,head e)))
  18     (hkinj : Function.Injective (fun e => (ktail e,khead e)))
  19     (m : ℕ) (f : Fin m → E → ℝ) (wK : K → ℝ)
  20     (hf : ∀ j e,0<f j e) (hwK : ∀ e,0<wK e) (C : Fin (m+1) → V) :
  21     (∀ i,0<copiedMarkedCapacity tail head ktail khead m f wK i (C i) (C (finRotate (m+1) i))) ↔
  22       (∀ j : Fin m,∃ e,tail e=C j.castSucc ∧ head e=C j.succ) ∧
  23         ∃ e,ktail e=C (Fin.last m) ∧ khead e=C 0 := by
  24   constructor
  25   · intro h
  26     constructor
  27     · intro j
  28       have hh := h j.castSucc
  29       rw [copiedMarkedCapacity_ordinary,finRotate_castSucc_eq_succ] at hh
  30       exact (occurrenceCapacityMatrix_pos_iff tail head hinj (f j) (hf j) _ _).mp hh
  31     · have hh := h (Fin.last m)
  32       rw [copiedMarkedCapacity_marked,finRotate_last] at hh
  33       exact (occurrenceCapacityMatrix_pos_iff ktail khead hkinj wK hwK _ _).mp hh
  34   · intro h i
  35     refine Fin.lastCases ?_ (fun j => ?_) i
  36     · rw [copiedMarkedCapacity_marked,finRotate_last]
  37       exact (occurrenceCapacityMatrix_pos_iff ktail khead hkinj wK hwK _ _).mpr h.2
  38     · rw [copiedMarkedCapacity_ordinary,finRotate_castSucc_eq_succ]
  39       exact (occurrenceCapacityMatrix_pos_iff tail head hinj (f j) (hf j) _ _).mpr (h.1 j)

  41 theorem marked_cycle_weight_restriction {V E K : Type*}
  42     [Fintype V] [Fintype E] [Fintype K] [DecidableEq V]
  43     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (π : Equiv.Perm V) (b : ℕ)
  44     (u : (Fin (m+1) → V) → ℝ) (hu : ∀ C,0≤u C)
  45     (hs : ∀ C,0<u C → IsMarkedCycleTuple tail head ktail khead m π b C) :
  46     (∑ C : MarkedCycleTuple tail head ktail khead m π b,
  47       u C.val • markedCycleIncidence tail head ktail khead m C.val)=
  48       ∑ C,u C • markedCycleIncidence tail head ktail khead m C := by
  49   let S : Set (Fin (m+1) → V) := {C | IsMarkedCycleTuple tail head ktail khead m π b C}
  50   symm
  51   apply Finset.sum_congr_set S
  52     (fun C => u C • markedCycleIncidence tail head ktail khead m C)
  53     (fun C : S => u C.val • markedCycleIncidence tail head ktail khead m C.val)
  54   · intro C hC
  55     rfl
  56   · intro C hC
  57     have hz : u C=0 := le_antisymm (le_of_not_gt (fun hp => hC (hs C hp))) (hu C)
  58     simp only [hz,zero_smul]
