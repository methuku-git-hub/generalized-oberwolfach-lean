  27       (∀ C₀,0 ≤ w C₀ ∧ w C₀ ≤ B/(k:ℝ)^(c-1)) →
  28       (∀ v,hypergraphWeightedDegree R w v=1) →

  32         ∀ i,(Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈Finset.univ\M.biUnion R)).card=
