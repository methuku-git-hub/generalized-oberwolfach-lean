   4 /-! Simultaneous construction of the rim paths in a bounded ordinary
   5 transformer. The forbidden set is enlarged by the actual earlier path
   6 supports, whose total size is bounded before each application. -/

  15  theorem simultaneous_transformer_rim_paths (Q F:SimpleGraph V)
  16     (A W Ω:Finset V) (Z:I→Finset V) (hAΩ:A⊆Ω)
  17     (x y x' y':I→V) (hroots:∀ i,x i∈Z i∧y i∈Z i∧x' i∈Z i∧y' i∈Z i)
  18     (hΩ:∀ i,x i∈Ω∧y i∈Ω∧x' i∈Ω∧y' i∈Ω)
  19     (hZ:∀ i,∀ v∈Z i,∀ w∈Z i,¬F.Adj v w)
  20     (r:ℕ) (hr:4≤r) (L b:ℝ) (hL:0≤L) (hF:∀ v,(F.degree v:ℝ)≤L)
  21     (hcount:∀ v∈Ω,∀ w∈Ω,b≤((A∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  22     (hthree:r=4→∀ i,∀ a∈A,b≤(((A∩Q.neighborFinset a)∩Q.neighborFinset (y i)∩Q.neighborFinset (y' i)).card:ℝ))
  23     (hbudget:∀ i,W.card+Fintype.card I*(r-2:ℕ)+(((Z i).card:ℝ)+r+2)*(L+1)<b):
  24     ∃ p:I→(Σ a:V,Σ c:V,Q.Walk a c),
  25       (∀ i,(p i).2.2.IsPath ∧ (p i).2.2.length=r-3 ∧
  26         Q.Adj (x i) (p i).1 ∧ Q.Adj (x' i) (p i).1 ∧
  27         Q.Adj (y i) (p i).2.1 ∧ Q.Adj (y' i) (p i).2.1 ∧
  28         (∀ v∈(p i).2.2.support,v∈A∧v∉W∧v∉Z i) ∧
  29         (∀ v∈(p i).2.2.support,∀ w∈(p i).2.2.support,¬F.Adj v w) ∧
  30         (∀ v∈(p i).2.2.support,∀ w∈Z i,¬F.Adj v w)) ∧
  31       ∀ i j,i≠j→Disjoint (p i).2.2.support.toFinset (p j).2.2.support.toFinset:=by
  32   let S:(Σ a:V,Σ c:V,Q.Walk a c)→Finset V:=fun p=>p.2.2.support.toFinset
  33   let Good:=fun (i:I) (p:Σ a:V,Σ c:V,Q.Walk a c)=>
  34     p.2.2.IsPath ∧ p.2.2.length=r-3 ∧ Q.Adj (x i) p.1 ∧ Q.Adj (x' i) p.1 ∧
  35       Q.Adj (y i) p.2.1 ∧ Q.Adj (y' i) p.2.1 ∧
  36       (∀ v∈p.2.2.support,v∈A∧v∉Z i) ∧
  37       (∀ v∈p.2.2.support,∀ w∈p.2.2.support,¬F.Adj v w) ∧
  38       (∀ v∈p.2.2.support,∀ w∈Z i,¬F.Adj v w)
  39   have hchoose:∀ i,∀ B:Finset V,B.card≤W.card+Fintype.card I*(r-2)→
  40       ∃ p,Good i p ∧ (S p).card≤r-2 ∧ Disjoint (S p) B:=by
  41     intro i B hB
  42     have hBr:(B.card:ℝ)≤W.card+Fintype.card I*(r-2:ℕ):=by exact_mod_cast hB
  43     obtain ⟨a,c,p,hp,hlen,hxa,hx'a,hyc,hy'c,hclass,hind,hroot⟩:=
  44       transformer_rim_common_path Q F A B Ω (Z i) hAΩ (x i) (y i) (x' i) (y' i)
  45         (hroots i) (hΩ i) (hZ i) r hr L b hL hF hcount (fun hr4=>hthree hr4 i)
  46         (by linarith only [hBr,hbudget i])
  47     refine ⟨⟨a,c,p⟩,⟨hp,hlen,hxa,hx'a,hyc,hy'c,?_,hind,hroot⟩,?_,?_⟩
  48     · intro v hv;exact ⟨(hclass v hv).1,(hclass v hv).2.2⟩
  49     · have hh:=List.toFinset_card_le p.support
  50       rw [p.length_support,hlen] at hh
  51       dsimp only [S]
  52       omega
  53     · apply Finset.disjoint_left.mpr
  54       intro v hv hBv
  55       exact (hclass v (List.mem_toFinset.mp hv)).2.1 hBv
  56   obtain ⟨p,hp,hdis⟩:=finite_disjoint_choices S Good W (r-2) hchoose
  57   refine ⟨p,?_,hdis⟩
  58   intro i
  59   obtain ⟨hi,hcard,hW⟩:=hp i
  60   rcases hi with ⟨hpath,hlen,hxa,hx'a,hyc,hy'c,hclass,hind,hroot⟩
  61   refine ⟨hpath,hlen,hxa,hx'a,hyc,hy'c,?_,hind,hroot⟩
  62   intro v hv
  63   exact ⟨(hclass v hv).1,fun hw=>Finset.disjoint_left.mp hW (List.mem_toFinset.mpr hv) hw,(hclass v hv).2⟩
