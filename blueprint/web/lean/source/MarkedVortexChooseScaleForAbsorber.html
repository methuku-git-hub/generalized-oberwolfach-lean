  11 theorem marked_vortex_choose_scale_for_absorber
  12     (β ε ρ₀ εAbs:ℝ) (hb:0<β) (he:0<ε) (heq:ε<1/4) (hρ:0<ρ₀)
  13     (heAbs:ε≤εAbs) (h required:ℕ):
  14     ∃ m size:ℕ,1≤m ∧ required≤m ∧ size=2*m ∧
  15       ∀ (V U:Type) [Fintype V] [Fintype U] [LinearOrder V] [DecidableEq U]
  16         (G:SimpleGraph V) (D:ℕ),G.IsRegularOfDegree D →
  17         (Fintype.card V:ℝ)/2≤D → ρ₀≤ initialIndexDensity U D → initialIndexDensity U D≤1 →
  18         m≤Fintype.card V →
  19         ∃ Q:Vortex U G D β ε m,∃ π:Equiv.Perm V,∃ L:ℕ,
  20           L+1=Q.length ∧
  21           (∀ i x,π x∈Q.extendedVertices i↔x∈Q.extendedVertices i) ∧
  22           (∀ x k,0<k→k<h→(π^k) x≠x) ∧
  23           ((markedVortexInnerVertices Q 0).card:ℝ)≤εAbs*Fintype.card (markedVortexVertexType Q 0) ∧
  24           ((markedVortexInnerHubs Q 0).card:ℝ)≤εAbs*Fintype.card (markedVortexVertexType Q 0) ∧
  25           (markedVortexInnerVertices Q L).card+(markedVortexInnerHubs Q L).card≤size := by
  26   obtain ⟨m₀,hm₀,hvortex⟩:=vortex_with_permutation_exists hb he heq hρ h
  27   let m:=max m₀ required
  28   have hm:1≤m:=hm₀.trans (le_max_left _ _)
  29   refine ⟨m,2*m,hm,le_max_right _ _,rfl,?_⟩
  30   intro V U _ _ _ _ G D hreg hD hρlo hρhi hmn
  31   obtain ⟨Q,π,hpres,_,hshort⟩:=hvortex V U G D m hreg hD hρlo hρhi ((le_max_left _ _)) hmn
  32   let L:=Q.length-1
  33   have hL:L+1=Q.length:=by dsimp [L];have hh:=Q.length_pos;omega
  34   have hn:0<Fintype.card V:=by omega
  35   have hDpos:0<D:=by
  36     have hn':(0:ℝ)<Fintype.card V:=by exact_mod_cast hn
  37     exact_mod_cast (show (0:ℝ)<D by linarith)
  38   have hf:=marked_vortex_inner_upper_size_inputs Q hreg hn hDpos hρhi he.le εAbs heAbs 0 (by have hh:=Q.length_pos;omega) 0 (by omega)
  39   have ht:=marked_vortex_terminal_pair_capacity Q hreg hn hDpos hρhi he.le L hL
  40   exact ⟨Q,π,L,hL,hpres,hshort,hf.1,hf.2,ht⟩
