  14 theorem symmetric_edge_isolator_sum {V : Type*} [Fintype V] [DecidableEq V]
  15     (z : V → V → ℝ) (hz : ∀ x y,z x y=z y x) (hdiag : ∀ x,z x x=0)
  16     (a b : V) :
  17     (1/2:ℝ)*(∑ p : V×V,z p.1 p.2*(if (SimpleGraph.edge p.1 p.2).Adj a b then 1 else 0))=z a b := by
  18   by_cases hab : a=b
  19   · subst b
  20     simp [hdiag]
  21   · have hterm (x y : V) :
  22         (if (SimpleGraph.edge x y).Adj a b then (1:ℝ) else 0)=
  23           (if a=x ∧ b=y then 1 else 0)+(if a=y ∧ b=x then 1 else 0) := by
  24       simp only [SimpleGraph.edge_adj]
  25       split_ifs <;> simp_all
  26     simp only [Fintype.sum_prod_type,hterm,mul_add,Finset.sum_add_distrib]
  27     have hfirst : (∑ x,∑ y,z x y*(if a=x ∧ b=y then (1:ℝ) else 0))=z a b := by
  28       simp [ite_and,eq_comm]
  29     have hsecond : (∑ x,∑ y,z x y*(if a=y ∧ b=x then (1:ℝ) else 0))=z b a := by
  30       simp [ite_and,eq_comm]
  31     rw [hfirst,hsecond,← hz a b]
  32     ring
