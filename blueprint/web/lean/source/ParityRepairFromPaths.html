  14 theorem parity_repair_from_coloured_paths {V : Type*} [Fintype V] [DecidableEq V] [Nonempty V]
  15     {q : ℕ} (hq : Even q) (G : SimpleGraph V) (σ : Sym2 V → Fin (q+1))
  16     (ρ : ℝ) (L : ℕ) (hL : 0<L)
  17     (hpath : ∀ a : Fin q,∀ u v,u≠v → ρ*Fintype.card V≤
  18       (colouredTwoPathCount G σ u v a.castSucc (Fin.last q):ℝ))
  19     (hlarge : (q:ℝ)*Fintype.card V/L+4*q+4*L<ρ*Fintype.card V) :
  20     ∃ τ : Sym2 V → Fin (q+1),
  21       (∀ a,(edgeColourGraph G τ a).edgeFinset.card=(edgeColourGraph G σ a).edgeFinset.card) ∧
  22       (∀ a v,Even ((G\edgeColourGraph G τ a).degree v)) ∧
  23       (∀ a v,|((edgeColourGraph G τ a).degree v:ℝ)-((edgeColourGraph G σ a).degree v:ℝ)|≤q+1) := by
  24   let T (j : Fin q) := Finset.univ.filter (fun v => Odd ((G\edgeColourGraph G σ j.castSucc).degree v))
  25   have hTeven (j : Fin q) : Even (T j).card := odd_complement_degree_set_even G (edgeColourGraph G σ j.castSucc)
  26   choose m hm he using fun j : Fin q => even_finset_pairing (T j) (hTeven j)
  27   let e (j : Fin q) : (Fin (m j)×Bool)≃T j := Classical.choice (he j)
  28   let I := (j : Fin q)×Fin (m j)
  29   let x (i : I) := (e i.1 (i.2,false)).val
  30   let y (i : I) := (e i.1 (i.2,true)).val
  31   have hxy (i : I) : x i≠y i := pairing_endpoints_distinct (e i.1) i.2
  32   let C (i : I) := (G.neighborFinset (x i)∩G.neighborFinset (y i)).filter
  33     (fun w => σ s(x i,w)=i.1.castSucc ∧ σ s(w,y i)=Fin.last q)
  34   have hI : Fintype.card I≤q*Fintype.card V := by
  35     have hmle (j : Fin q) : m j≤Fintype.card V := by
  36       have hh := Finset.card_le_univ (T j)
  37       rw [hm j] at hh
  38       omega
  39     simp only [I,Fintype.card_sigma,Fintype.card_fin]
  40     calc
  41       _ ≤ ∑ _ : Fin q,Fintype.card V := Finset.sum_le_sum (fun j _ => hmle j)
  42       _ = _ := by simp
  43   have hx (v : V) : ((Finset.univ.filter (fun i : I => x i=v)).card:ℝ)≤q := by
  44     dsimp only [I,x]
  45     have hh := paired_family_endpoint_bound T m e v false
  46     simp only [Fintype.card_fin] at hh
  47     exact_mod_cast hh
  48   have hy (v : V) : ((Finset.univ.filter (fun i : I => y i=v)).card:ℝ)≤q := by
  49     dsimp only [I,y]
  50     have hh := paired_family_endpoint_bound T m e v true
  51     simp only [Fintype.card_fin] at hh
  52     exact_mod_cast hh
  53   have hC (i : I) : (Fintype.card I:ℝ)/L+4*(q:ℝ)+4*L<(C i).card := by
  54     have hi : (Fintype.card I:ℝ)≤(q:ℝ)*Fintype.card V := by exact_mod_cast hI
  55     have hd := div_le_div_of_nonneg_right hi (Nat.cast_nonneg (α:=ℝ) L)
  56     have hp := hpath i.1 (x i) (y i) (hxy i)
  57     change ρ*Fintype.card V≤((C i).card:ℝ) at hp
  58     linarith only [hd,hlarge,hp]
  59   obtain ⟨z,hz,hd,_⟩ := undirectedTwoPathPacking_exists x y C L hL hx hy hC
  60   have hzx (i : I) : G.Adj (x i) (z i) :=
  61     (G.mem_neighborFinset _ _).mp (Finset.mem_inter.mp (Finset.mem_filter.mp (hz i)).1).1
  62   have hzy (i : I) : G.Adj (z i) (y i) :=
  63     ((G.mem_neighborFinset _ _).mp (Finset.mem_inter.mp (Finset.mem_filter.mp (hz i)).1).2).symm
  64   have hc0 (i : I) : σ s(x i,z i)=i.1.castSucc := (Finset.mem_filter.mp (hz i)).2.1
  65   have hc1 (i : I) : σ s(z i,y i)=Fin.last q := (Finset.mem_filter.mp (hz i)).2.2
  66   obtain ⟨τ,hcard,hparity,hdeg,hlast⟩ := paired_colour_parity_repair hq G σ T m e
  67     (fun j v => by simp only [T,Finset.mem_filter,Finset.mem_univ,true_and]) z hd hzx hzy hc0 hc1
  68   refine ⟨τ,hcard,hparity,?_⟩
  69   intro a v
  70   refine Fin.lastCases ?_ (fun j => ?_) a
  71   · exact (hlast v).trans (by linarith)
  72   · exact (hdeg j v).trans (by have hq0 : (0:ℝ)≤q := Nat.cast_nonneg _;linarith)
  73
