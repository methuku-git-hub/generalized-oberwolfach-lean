  11 theorem rooted_conflict_loss_le (n s k : ℕ) (hn : 0<n) (hns : 2*s≤n)
  12     (L : ℝ) (hL : 0≤L) :
  13     let q : ℝ := (k:ℝ)/(n-s:ℕ)
  14     L*(2*(s:ℝ)*q+n*q^2)≤L*(4*s*k+4*(k:ℝ)^2)/n := by
  15   intro q
  16   have hs : s≤n := by omega
  17   have hden : (0:ℝ)<(n-s:ℕ) := by exact_mod_cast (show 0<n-s by omega)
  18   have hnr : (0:ℝ)<n := Nat.cast_pos.mpr hn
  19   have hbase : (n:ℝ)≤2*(n-s:ℕ) := by exact_mod_cast (show n≤2*(n-s) by omega)
  20   have hq : q≤2*k/(n:ℝ) := by
  21     apply (div_le_div_iff₀ hden hnr).mpr
  22     nlinarith only [mul_le_mul_of_nonneg_left hbase (Nat.cast_nonneg (α:=ℝ) k)]
  23   have hq0 : 0≤q := by dsimp [q];positivity
  24   have hsq : q^2≤(2*k/(n:ℝ))^2 := (sq_le_sq₀ hq0 (by positivity)).mpr hq
  25   have hlin := mul_le_mul_of_nonneg_left hq (show (0:ℝ)≤2*s by positivity)
  26   have hquad := mul_le_mul_of_nonneg_left hsq hnr.le
  27   calc
  28     _ ≤ L*(2*(s:ℝ)*(2*k/n)+(n:ℝ)*(2*k/n)^2) :=
  29       mul_le_mul_of_nonneg_left (add_le_add hlin hquad) hL
  30     _ = _ := by field_simp;ring

  33 theorem rooted_dense_set_scales (s : ℕ) (α : ℝ) (hα : 0<α) :
  34     ∃ K : ℕ,∀ k : ℕ,K≤k → 2≤k ∧
  35       (s:ℝ)+1≤α*(s+k:ℝ)/4 ∧
  36       2*(s+k:ℝ)*Real.exp (-α^2*(s+k:ℝ)/8)≤1/4 ∧
  37       ∀ L : ℝ,0≤L → ∃ N : ℕ,∀ n : ℕ,N≤n → 0<n ∧ 2*(s+k)≤n ∧
  38         L*(2*(s:ℝ)*((k:ℝ)/(n-s:ℕ))+n*((k:ℝ)/(n-s:ℕ))^2)≤1/4 := by

  42   refine ⟨max (max K₁ K₂) 2,?_⟩
  43   intro k hk
  44   have ha := h₂ k (by omega)
  45   have hb := h₁ (s+k) (by omega)
