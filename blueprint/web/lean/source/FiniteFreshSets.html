  10 theorem finite_fresh_sets {J V:Type} [Fintype J] [DecidableEq J] [DecidableEq V]
  11     (s:ℕ) (P:J→Finset V→Prop)
  12     (hchoose:∀ i (B:Finset V),B.card≤Fintype.card J*s →
  13       ∃ S:Finset V,S.card=s ∧ P i S ∧ Disjoint S B):
  14     ∃ S:J→Finset V,(∀ i,(S i).card=s∧P i (S i)) ∧
  15       ∀ i j,i≠j→Disjoint (S i) (S j) := by
  16   have hstep:∀ T:Finset J,∃ S:J→Finset V,
  17       (∀ i∈T,(S i).card=s∧P i (S i)) ∧
  18       ∀ i∈T,∀ j∈T,i≠j→Disjoint (S i) (S j):=by
  19     intro T
  20     induction T using Finset.induction with
  21     | empty=>
  22       refine ⟨fun _=>∅,?_,?_⟩ <;> simp
  23     | @insert i T hi ih=>
  24       obtain ⟨S,hS,hdis⟩:=ih
  25       let B:=T.biUnion S
  26       have hB:B.card≤Fintype.card J*s:=by
  27         calc
  28           _ ≤ T.card*s:=Finset.card_biUnion_le_card_mul T S s (by intro j hj; exact (hS j hj).1.le)
  29           _ ≤ Fintype.card J*s:=Nat.mul_le_mul_right s (Finset.card_le_univ T)
  30       obtain ⟨A,hA,hP,hAB⟩:=hchoose i B hB
  31       let S':=Function.update S i A
  32       refine ⟨S',?_,?_⟩
  33       · intro j hj
  34         rcases Finset.mem_insert.mp hj with rfl|hj
  35         · simpa [S'] using And.intro hA hP
  36         · have hji:j≠i:=by intro he;subst j;exact hi hj
  37           simpa [S',Function.update_of_ne hji] using hS j hj
  38       · intro j hj k hk hjk
  39         by_cases hji:j=i
  40         · subst j
  41           have hki:k≠i:=hjk.symm
  42           have hkT:k∈T:=(Finset.mem_insert.mp hk).resolve_left hki
  43           have hkB:S k⊆B:=by intro v hv;exact Finset.mem_biUnion.mpr ⟨k,hkT,hv⟩
  44           simpa [S',Function.update_of_ne hki] using hAB.mono_right hkB
  45         · have hjT:j∈T:=(Finset.mem_insert.mp hj).resolve_left hji
  46           by_cases hki:k=i
  47           · subst k
  48             have hjB:S j⊆B:=by intro v hv;exact Finset.mem_biUnion.mpr ⟨j,hjT,hv⟩
  49             simpa [S',Function.update_of_ne hji] using (hAB.mono_right hjB).symm
  50           · have hkT:k∈T:=(Finset.mem_insert.mp hk).resolve_left hki
  51             simpa [S',Function.update_of_ne hji,Function.update_of_ne hki] using hdis j hjT k hkT hjk
  52 
  53   obtain ⟨S,hS,hdis⟩:=hstep Finset.univ
  54   exact ⟨S,fun i=>hS i (Finset.mem_univ i),fun i j=>hdis i (Finset.mem_univ i) j (Finset.mem_univ j)⟩
  55 end Oberwolfach
