  14 def ordinaryWheelCorrectionConstant (r : ℕ) (γ : ℝ) : ℝ :=
  15   (r+1)^2/γ+2*(1+1/γ)*(6*((r:ℝ)^2)^r/γ)*(r+1)

  17 theorem ordinaryWheelCorrectionConstant_pos (r : ℕ) (γ : ℝ) (hγ : 0<γ) :
  18     0<ordinaryWheelCorrectionConstant r γ := by
  19   unfold ordinaryWheelCorrectionConstant

  22 theorem ordinary_wheel_correction (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  23     ∃ N : ℕ,∀ (V U : Type) [Fintype V] [DecidableEq V] [Fintype U] [DecidableEq U],
  24       ∀ (P Q F : SimpleGraph V) (H : V → U → Prop) (k : ℕ),N≤k →
  25       ∀ θ : ℝ,cycleThreshold r≤θ → ∀ γ η : ℝ,0<γ → 0≤η →
  26       0<Fintype.card V → Q≤P →
  27       (∀ v,Q.degree v=2*(Finset.univ.filter (H v)).card) →
  28       (∀ x y,Q.Adj x y → γ*(Fintype.card V:ℝ)^(k-1)≤
  29         (Fintype.card (OrdinaryIsolatorConfiguration P Q F H k θ α x y):ℝ)) →
  30       (∀ v (i j : {u // H v u}),i.val≠j.val →
  31         γ*(Fintype.card V:ℝ)^(r^2)≤
  32           (Fintype.card (OrdinaryTransferConfiguration hr Q F H v i.val j.val):ℝ)) →
  33       ∀ ψ : OrdinaryWheel Q F H r → ℝ,
  34       (∀ x y,|ordinaryRimResidual ψ x y|≤η) →
  35       (∀ v u,|ordinarySpokeResidual ψ v u|≤η) →
  36       ∃ ω : OrdinaryWheel Q F H r → ℝ,
  37         (∀ x y,ordinaryRimLoad ω x y=if Q.Adj x y then 1 else 0) ∧
  38         (∀ v u,ordinarySpokeLoad ω v u=if H v u then 1 else 0) ∧
  39         (∀ D,|ω D-ψ D|≤ordinaryWheelCorrectionConstant r γ*η/(Fintype.card V:ℝ)^(r-1)) := by
  40   obtain ⟨N,hN⟩ := ordinary_rim_correction r hr α hα
  41   refine ⟨N,?_⟩
  42   intro V U _ _ _ _ P Q F H k hk θ hθ γ η hγ hη hn hQP hdeg hcount hT ψ hRerr hSerr
  43   obtain ⟨R,hR,hRb,hRs⟩ := hN V U P Q F H k hk θ hθ γ η hγ hη hn hQP hcount ψ hRerr
  44   let ψ₁ (D : OrdinaryWheel Q F H r) : ℝ := ψ D+R D
  45   have hR₁ (x y : V) : ordinaryRimLoad ψ₁ x y=if Q.Adj x y then 1 else 0 := by
  46     change ordinaryRimLoad (ψ+R) x y=if Q.Adj x y then 1 else 0
  47     rw [ordinaryRimLoad_add ψ R x y,hR,ordinaryRimResidual]
  48     ring
  49   have hS₁ (v : V) (u : U) : |ordinarySpokeResidual ψ₁ v u|≤η*(1+1/γ) := by
  50     have he : ordinarySpokeResidual ψ₁ v u=
  51         ordinarySpokeResidual ψ v u-ordinarySpokeLoad R v u := by
  52       change (if H v u then (1:ℝ) else 0)-ordinarySpokeLoad (ψ+R) v u=
  53         ((if H v u then (1:ℝ) else 0)-ordinarySpokeLoad ψ v u)-ordinarySpokeLoad R v u
  54       rw [ordinarySpokeLoad_add ψ R v u]
  55       ring
  56     rw [he]
  57     calc
  58       _ ≤ |ordinarySpokeResidual ψ v u|+|ordinarySpokeLoad R v u| := by
  59         simpa only [sub_eq_add_neg,abs_neg] using abs_add_le (ordinarySpokeResidual ψ v u) (-ordinarySpokeLoad R v u)
  60       _ ≤ η+η/γ := add_le_add (hSerr v u) (hRs v u)
  61       _ = _ := by ring
  62   obtain ⟨S,hSR,hSS,hSb⟩ := ordinary_spoke_correction hr Q F H γ (η*(1+1/γ))
  63     hγ (by positivity) hn ψ₁ hdeg (fun x y hxy => by simpa only [if_pos hxy] using hR₁ x y) hS₁ hT
  64   let ω (D : OrdinaryWheel Q F H r) : ℝ := ψ₁ D+S D
  65   refine ⟨ω,?_,?_,?_⟩
  66   · intro x y
  67     change ordinaryRimLoad (ψ₁+S) x y=if Q.Adj x y then 1 else 0
  68     rw [ordinaryRimLoad_add ψ₁ S x y,hSR,add_zero]
  69     exact hR₁ x y
  70   · intro v u
  71     change ordinarySpokeLoad (ψ₁+S) v u=if H v u then 1 else 0
  72     rw [ordinarySpokeLoad_add ψ₁ S v u,hSS,ordinarySpokeResidual]
  73     ring
  74   · intro D
  75     have he : ω D-ψ D=R D+S D := by dsimp [ω,ψ₁];ring
  76     rw [he]
  77     calc
  78       _ ≤ |R D|+|S D| := abs_add_le _ _
  79       _ ≤ η*(r+1)^2/(γ*(Fintype.card V:ℝ)^(r-1))+
  80           (2*(η*(1+1/γ))*(6*((r:ℝ)^2)^r/γ)*(r+1))/(Fintype.card V:ℝ)^(r-1) :=
  81         add_le_add (hRb D) (hSb D)
  82       _ = _ := by unfold ordinaryWheelCorrectionConstant;ring
