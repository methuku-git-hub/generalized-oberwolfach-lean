  11 theorem prescribed_cycle_two_boundary_split (a b:X) (xs:List X)
  12     (ha:a∉xs) (hxs:xs.Nodup) (hb:b∈xs):
  13     ∃ pre post:List X,xs=pre++b::post ∧
  14       (a::pre++[b]).Nodup ∧ (b::post++[a]).Nodup ∧
  15       (∀ x,x∈(a::pre++[b])→x∈(b::post++[a])→x=a ∨ x=b) ∧
  16       prescribedOrderedPathEdges ((a::xs)++[a]) =
  17         prescribedOrderedPathEdges (a::pre++[b]) ∪ prescribedOrderedPathEdges (b::post++[a]) ∧
  18       prescribedDirectedPathEdges ((a::xs)++[a]) =
  19         prescribedDirectedPathEdges (a::pre++[b]) ∪ prescribedDirectedPathEdges (b::post++[a]) := by
  20   have hsplit:∀ l:List X,b∈l→∃ pre post:List X,l=pre++b::post:=by
  21     intro l
  22     induction l with
  23     | nil => simp
  24     | cons x xs ih =>
  25       intro h
  26       rcases List.mem_cons.mp h with h|h
  27       · subst x;exact ⟨[],xs,rfl⟩
  28       · obtain ⟨pre,post,he⟩:=ih h
  29         exact ⟨x::pre,post,by simp [he]⟩
  30   obtain ⟨pre,post,hxsEq⟩:=hsplit xs hb
  31   subst xs
  32   have hnparts:=List.nodup_append'.mp hxs
  33   have hbpre:b∉pre:=by
  34     intro h;exact (List.disjoint_left.mp hnparts.2.2 h) (by simp)
  35   have hbpost:b∉post:=(List.nodup_cons.mp hnparts.2.1).1
  36   have hpost:post.Nodup:=(List.nodup_cons.mp hnparts.2.1).2
  37   have hapre:a∉pre:=by intro h;exact ha (List.mem_append_left _ h)
  38   have hab:a≠b:=by intro he;subst a;exact ha (by simp)
  39   have hapost:a∉post:=by intro h;exact ha (by simp [h])
  40   have hleft:(a::pre++[b]).Nodup:=by
  41     simp only [List.cons_append]
  42     rw [List.nodup_cons]
  43     constructor
  44     · simp [hapre,hab]
  45     · apply List.nodup_append'.mpr
  46       refine ⟨hnparts.1,by simp,?_⟩
  47       apply List.disjoint_left.mpr
  48       intro x hx hy
  49       simp only [List.mem_singleton] at hy
  50       subst x;exact hbpre hx
  51   have hright:(b::post++[a]).Nodup:=by
  52     simp only [List.cons_append]
  53     rw [List.nodup_cons]
  54     constructor
  55     · simp [hbpost,Ne.symm hab]
  56     · apply List.nodup_append'.mpr
  57       refine ⟨hpost,by simp,?_⟩
  58       apply List.disjoint_left.mpr
  59       intro x hx hy
  60       simp only [List.mem_singleton] at hy
  61       subst x;exact hapost hx
  62   refine ⟨pre,post,rfl,hleft,hright,?_,?_,?_⟩
  63   · intro x hx hy
  64     simp only [List.cons_append,List.mem_cons,List.mem_append,List.not_mem_nil,or_false] at hx hy
  65     rcases hx with hx|hx|hx
  66     · exact Or.inl hx
  67     · rcases hy with hy|hy|hy
  68       · exact Or.inr hy
  69       · exact ((List.disjoint_left.mp hnparts.2.2 hx) (by simp [hy])).elim
  70       · exact Or.inl hy
  71     · exact Or.inr hx
  72   · simpa [List.cons_append,List.append_assoc] using
  73       prescribed_path_edges_pivot b (post++[a]) (a::pre)
  74   · simpa [List.cons_append,List.append_assoc] using
  75       prescribed_directed_path_edges_pivot b (post++[a]) (a::pre)
