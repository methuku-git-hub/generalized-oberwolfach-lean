  10 def orientationDisagreementGraph {V:Type} [LinearOrder V]
  11     (G:SimpleGraph V) (bits:Sym2 V→Bool) (M:V→V→ℕ):SimpleGraph V where
  12   Adj x y:=G.Adj x y ∧ (M x y≠sampledOrientation G bits x y ∨ M y x≠sampledOrientation G bits y x)
  13   symm:=⟨by intro x y h;exact ⟨h.1.symm,Or.symm h.2⟩⟩
  14   loopless:=⟨by intro x h;exact G.loopless.irrefl x h.1⟩
