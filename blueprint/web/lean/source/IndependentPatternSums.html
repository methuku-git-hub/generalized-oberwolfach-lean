  16     {State : A → Type*} [∀ a,Fintype (State a)]
  17     (μ : ∀ a,FiniteLaw (State a)) (R : I → Finset A) (w : I → ℝ)
  18     (hw : ∀ i,0≤w i) (s : I → ∀ a,State a)
  19     (a q L : ℝ) (ha : 0<a) (hq : 0<q) (hL : 0<L)
  20     (hsize : ∀ i,((R i).card:ℝ)≤q) (htotal : (∑ i,w i)≤L)
  21     (hload : ∀ e,(∑ i,if e∈R i then w i else 0)≤a)
  22     (t : ℝ) (ht : 0<t) :

  24       |(∑ i,w i*patternValue (R i) (s i) σ)-
  25         (∑ i,w i*∏ e∈R i,(μ e).mass (s i e))|)≤
  26       2*Real.exp (-2*t^2/(a*q*L)) := by
  27   have hh := supportedPatternSum_tail μ R w hw s (fun e=>{e}) (fun σ=>σ)
  28     (by intro e σ τ h;exact h e (Finset.mem_singleton_self e))
  29     1 (a*q*L) (by norm_num) (by positivity)
  30     (by intro e;norm_num)
  31     (by
  32       intro e
  33       have he : Finset.univ.filter (fun b=>e∈({b}:Finset A))={e} := by ext b;simp [eq_comm]
  34       rw [he];norm_num)
  35     (finite_pattern_incidence_square_sum R w hw a q L ha.le hq.le hsize htotal hload) t ht
  36   have he := supportedPatternSum_expectation μ R w s (fun e=>{e}) (fun σ=>σ)
  37     (by intro e σ τ h;exact h e (Finset.mem_singleton_self e))
  38     (by intro i a ha b hb hab;simpa using hab)
