  39     (∑ f,resourceOmittedPairWeight R w e f)=
  40       ∑ i,if e∈R i then ((R i\{e}).card:ℝ)*w i else 0 := by
  41   unfold resourceOmittedPairWeight

  53 theorem resourceOmittedPairWeight_square_sum_le (R : I → Finset A) (w : I → ℝ)
  54     (hw : ∀ i,0≤w i) (e : A) (q L a : ℝ)
  55     (hq : 0≤q) (ha : 0≤a)
  56     (hsize : ∀ i,e∈R i → ((R i\{e}).card:ℝ)≤q)
  57     (hload : (∑ i,if e∈R i then w i else 0)≤L)
  58     (hpair : ∀ f,resourceOmittedPairWeight R w e f≤a) :
  59     (∑ f,(resourceOmittedPairWeight R w e f)^2)≤a*q*L := by
  60   have htotal : (∑ f,resourceOmittedPairWeight R w e f)≤q*L := by
  61     rw [resourceOmittedPairWeight_sum]
  62     calc
  63       _ ≤ ∑ i,q*(if e∈R i then w i else 0) := by
  64         apply Finset.sum_le_sum
  65         intro i _
  66         by_cases he : e∈R i
  67         · simp only [he,ite_true]
  68           exact mul_le_mul_of_nonneg_right (hsize i he) (hw i)
  69         · simp only [he,ite_false,mul_zero,le_refl]
  70       _ = q*∑ i,if e∈R i then w i else 0 := (Finset.mul_sum _ _ _).symm
  71       _ ≤ q*L := mul_le_mul_of_nonneg_left hload hq
  72   calc
  73     _ ≤ ∑ f,a*resourceOmittedPairWeight R w e f := by
  74       apply Finset.sum_le_sum
  75       intro f _
  76       simpa only [pow_two] using mul_le_mul_of_nonneg_right (hpair f)
  77         (resourceOmittedPairWeight_nonneg R w hw e f)
  78     _ = a*∑ f,resourceOmittedPairWeight R w e f := (Finset.mul_sum _ _ _).symm
  79     _ ≤ a*(q*L) := mul_le_mul_of_nonneg_left htotal ha
  80     _ = _ := by ring

 102 theorem finiteLaw_boundedDifferences_variance_bound
 103     {State : A → Type*} [∀ a,Fintype (State a)]
 104     (μ : ∀ a,FiniteLaw (State a)) (X : (∀ a,State a) → ℝ)
 105     (c : A → ℝ) (hc : ∀ a,0≤c a) (v : ℝ) (hv : 0<v)
 106     (hsq : (∑ a,(c a)^2)≤v)
 107     (hchange : ∀ a σ τ,(∀ b,b≠a → σ b=τ b) → |X σ-X τ|≤c a)
 108     (t : ℝ) (ht : 0<t) :
 109     (FiniteLaw.pi μ).probability (fun σ=>t≤|X σ-(FiniteLaw.pi μ).expectation X|)≤
 110       2*Real.exp (-2*t^2/v) := by
 111   have hs0 : 0≤∑ a,(c a)^2 := Finset.sum_nonneg (fun _ _=>sq_nonneg _)
 112   by_cases hs : (∑ a,(c a)^2)=0
 113   · have hc0 : ∀ a,c a=0 := by
 114       intro a
 115       have hh := Finset.single_le_sum (s:=Finset.univ) (f:=fun a=>(c a)^2)
 116         (fun _ _=>sq_nonneg _) (Finset.mem_univ a)
 117       rw [hs] at hh
 118       nlinarith only [hh,sq_nonneg (c a)]
 119     have hx := finiteLaw_zero_boundedDifferences μ X
 120       (fun a σ τ hagree=>by simpa only [hc0 a] using hchange a σ τ hagree)
 121     have hbad : ∀ σ,¬t≤|X σ-(FiniteLaw.pi μ).expectation X| := by
 122       intro σ;rw [hx σ,sub_self,abs_zero];exact not_le.mpr ht
 123     simp only [FiniteLaw.probability,hbad,ite_false,Finset.sum_const_zero]
 124     positivity
 125   · have hsp : 0<(∑ a,(c a)^2) := lt_of_le_of_ne hs0 (Ne.symm hs)
 126     apply (finiteLaw_boundedDifferences μ X c hc hchange ht.le).trans
 127     apply mul_le_mul_of_nonneg_left _ (by norm_num)
 128     apply Real.exp_le_exp.mpr
 129     exact (div_le_div_iff₀ hsp hv).mpr
 130       (mul_le_mul_of_nonpos_left hsq (by nlinarith only [sq_nonneg t]))

 172 /-- A sparse incidence relation between primitive coordinates and resource
 173 states costs the product of its two maximum incidence bounds. -/
 174 theorem locality_square_sum {E : Type*} [Fintype E] [DecidableEq E]
 175     (S : A → Finset E) (B : ℝ) (hB : 0≤B)
 176     (hread : ∀ a,((S a).card:ℝ)≤B)
 177     (haffect : ∀ e,((Finset.univ.filter (fun a=>e∈S a)).card:ℝ)≤B)
 178     (c : A → ℝ) :
 179     (∑ e,(∑ a,if e∈S a then c a else 0)^2)≤B^2*∑ a,(c a)^2 := by
 180   have hlocal (e : E) : (∑ a,if e∈S a then c a else 0)^2≤
 181       B*∑ a,if e∈S a then (c a)^2 else 0 := by
 182     rw [← Finset.sum_filter,← Finset.sum_filter]
 183     have hh := Finset.sum_mul_sq_le_sq_mul_sq
 184       (Finset.univ.filter (fun a=>e∈S a)) (fun _ : A=>(1:ℝ)) c
 185     simp only [one_mul,one_pow,Finset.sum_const,nsmul_eq_mul,mul_one] at hh
 186     exact hh.trans (mul_le_mul_of_nonneg_right (haffect e)
 187       (Finset.sum_nonneg (fun _ _=>sq_nonneg _)))
 188   calc
 189     _ ≤ ∑ e,B*∑ a,if e∈S a then (c a)^2 else 0 := Finset.sum_le_sum (fun e _=>hlocal e)
 190     _ = B*∑ a,((S a).card:ℝ)*(c a)^2 := by
 191       rw [← Finset.mul_sum,Finset.sum_comm]
 192       congr 1
 193       apply Finset.sum_congr rfl
 194       intro a _
 195       rw [← Finset.sum_filter]
 196       have hs : Finset.univ.filter (fun e=>e∈S a)=S a := by ext;simp
 197       rw [hs,Finset.sum_const,nsmul_eq_mul]
 198     _ ≤ B*∑ a,B*(c a)^2 := mul_le_mul_of_nonneg_left
 199       (Finset.sum_le_sum (fun a _=>mul_le_mul_of_nonneg_right (hread a) (sq_nonneg _))) hB
 200     _ = _ := by rw [← Finset.mul_sum];ring

 268 theorem finiteLaw_disjoint_supported_family
 269     {E J : Type*} [Fintype E] [DecidableEq E] [DecidableEq J]

 271     (μ : ∀ e,FiniteLaw (Base e)) (S : J → Finset E)
 272     (event : J → (∀ e,Base e) → Prop)
 273     (hs : ∀ j σ τ,(∀ e∈S j,σ e=τ e) → (event j σ↔event j τ))
 274     (T : Finset J)
 275     (hdisj : ∀ i∈T,∀ j∈T,i≠j → Disjoint (S i) (S j)) :
 276     (FiniteLaw.pi μ).probability (fun σ=>∀ j∈T,event j σ)=
 277       ∏ j∈T,(FiniteLaw.pi μ).probability (event j) := by

 283       intro e he ht
