  22 def markedFiniteBalance (tail head:E→V) (ktail khead:K→V) (HK:V→U→Prop)
  23     (A:Finset (E⊕K)) (S:Finset (V×U)) (m:ℕ):V⊕U→Fin 4→ℝ
  24   | Sum.inl v=>![((A.filter (fun a=>Sum.elim tail ktail a=v)).card:ℝ)-(S.filter (fun s=>s.1=v)).card,
  25       ((A.filter (fun a=>Sum.elim head khead a=v)).card:ℝ)-(S.filter (fun s=>s.1=v)).card,
  26       ((A.filter (fun a=>Sum.elim head khead a=v∧markedArcType a=true)).card:ℝ)-
  27         (S.filter (fun s=>s.1=v∧HK s.1 s.2)).card,0]
  28   | Sum.inr u=>![0,0,0,((S.filter (fun s=>s.2=u∧¬HK s.1 s.2)).card:ℝ)-
  29       m*(S.filter (fun s=>s.2=u∧HK s.1 s.2)).card]

  31 theorem markedFiniteBalance_sdiff
  32     (tail head:E→V) (ktail khead:K→V) (HK:V→U→Prop)
  33     (A B:Finset (E⊕K)) (S R:Finset (V×U)) (m:ℕ)
  34     (hA:B⊆A) (hS:R⊆S) (z:V⊕U) (i:Fin 4):
  35     markedFiniteBalance tail head ktail khead HK (A\B) (S\R) m z i=
  36       markedFiniteBalance tail head ktail khead HK A S m z i-
  37       markedFiniteBalance tail head ktail khead HK B R m z i:=by
  38   have ha:=real_filter_sdiff_card A B hA
  39   have hs:=real_filter_sdiff_card S R hS
  40   simp only [←Set.ncard_coe_finset,Finset.coe_filter,Finset.mem_sdiff] at ha hs
  41   cases z <;> fin_cases i <;>
  42     simp only [markedFiniteBalance,Matrix.cons_val_zero',Matrix.cons_val_succ'] <;>
  43     (try simp only [←Set.ncard_coe_finset,Finset.coe_filter,Finset.mem_sdiff]) <;>
  44     (try simp only [ha,hs]) <;> ring

  46 theorem markedFiniteBalance_eq_zero_iff
  47     (tail head:E→V) (ktail khead:K→V) (HK:V→U→Prop)
  48     (A:Finset (E⊕K)) (S:Finset (V×U)) (m:ℕ):
  49     (∀ z i,markedFiniteBalance tail head ktail khead HK A S m z i=0)↔
  50       MarkedRemainderDivisible tail head ktail khead HK A S m:=by
  51   constructor
  52   · intro h
  53     have ho (v:V):((A.filter (fun a=>Sum.elim tail ktail a=v)).card:ℝ)=
  54         (S.filter (fun s=>s.1=v)).card:=sub_eq_zero.mp (h (Sum.inl v) 0)
  55     have hi (v:V):((A.filter (fun a=>Sum.elim head khead a=v)).card:ℝ)=
  56         (S.filter (fun s=>s.1=v)).card:=sub_eq_zero.mp (h (Sum.inl v) 1)
  57     constructor
  58     · intro v
  59       exact_mod_cast (hi v).trans (ho v).symm
  60     · intro v
  61       exact_mod_cast hi v
  62     · intro v
  63       have hh:=sub_eq_zero.mp (h (Sum.inl v) 2)
  64       exact_mod_cast hh
  65     · intro u
  66       have hh:=sub_eq_zero.mp (h (Sum.inr u) 3)
  67       exact_mod_cast hh
  68   · intro h z i
  69     cases z with
  70     | inl v=>
  71       have ho:=h.directed v
  72       have hi:=h.row v
  73       have hk:=h.marked v
  74       fin_cases i <;> simp only [markedFiniteBalance,Matrix.cons_val_zero',Matrix.cons_val_succ'] <;>
  75         simp only [←Set.ncard_coe_finset,Finset.coe_filter] at ho hi hk ⊢
  76       · rw [←ho,hi,sub_self]
  77       · rw [hi,sub_self]
  78       · rw [hk,sub_self]
  79     | inr u=>
  80       have hh:=h.hub u
  81       fin_cases i <;> simp only [markedFiniteBalance,Matrix.cons_val_zero',Matrix.cons_val_succ'] <;>
  82         simp only [←Set.ncard_coe_finset,Finset.coe_filter] at hh ⊢
  83       rw [hh,Nat.cast_mul,sub_self]
