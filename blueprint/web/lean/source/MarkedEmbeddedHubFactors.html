  36 def hubFactorEmbedding (f:W↪V) (ρ:Equiv.Perm W) (π:Equiv.Perm V)
  37     (h:∀ x,f (ρ x)=π (f x)) (tail head:E→W) (ktail khead:K→W)
  38     (m b:ℕ) (H₀ HK:W→U→Prop) (u:U) (Y:Finset W)
  39     (P:MarkedHubFactorOn tail head ktail khead m ρ b H₀ HK u Y):
  40     MarkedHubFactorOn (f∘tail) (f∘head) (f∘ktail) (f∘khead) m π b
  41       (spokeRelationImage f H₀) (spokeRelationImage f HK) u (Y.image f) where
  42   cycles:=P.cycles.image (hubCycleEmbeddingEquiv f ρ π h tail head ktail khead m b H₀ HK u)
  43   disjoint:=by
  44     intro C hC D hD hCD
  45     obtain ⟨c,hc,hec⟩:=Finset.mem_image.mp hC
  46     obtain ⟨d,hd,hed⟩:=Finset.mem_image.mp hD
  47     subst C;subst D
  48     rw [hubCycleEmbedding_vertices,hubCycleEmbedding_vertices]
  49     apply Finset.disjoint_left.mpr
  50     intro v hv hv'
  51     obtain ⟨x,hx,hfx⟩:=Finset.mem_image.mp hv
  52     obtain ⟨y,hy,hfy⟩:=Finset.mem_image.mp hv'
  53     have hxy:x=y:=f.injective (hfx.trans hfy.symm)
  54     exact Finset.disjoint_left.mp (P.disjoint c hc d hd (fun he=>hCD (congrArg _ he))) hx (hxy.symm ▸ hy)
  55   covers:=by
  56     rw [Finset.image_biUnion]
  57     simp_rw [hubCycleEmbedding_vertices]
  58     rw [←Finset.biUnion_image,P.covers]
