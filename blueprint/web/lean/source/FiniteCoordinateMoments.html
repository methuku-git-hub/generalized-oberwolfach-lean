  15 theorem finiteLaw_two_coordinate_product {I : Type*} [Fintype I] [DecidableEq I]
  16     {State : I → Type*} [∀ i,Fintype (State i)]
  17     (μ : ∀ i,FiniteLaw (State i)) (i j : I) (hij : i≠j)
  18     (X : State i → ℝ) (Y : State j → ℝ) :
  19     (FiniteLaw.pi μ).expectation (fun σ => X (σ i)*Y (σ j))=
  20       (μ i).expectation X*(μ j).expectation Y := by
  21   let : ∀ e,MeasurableSpace (State e) := fun _ => ⊤
  22   let : ∀ e,MeasurableSingletonClass (State e) := fun _ => ⟨fun _ => trivial⟩
  23   have hcoord : iIndepFun (fun (e : I) (σ : ∀ k,State k) => σ e)
  24       (FiniteLaw.pi μ).toPMF.toMeasure := by
  25     rw [finiteLaw_pi_measure]
  26     exact iIndepFun_pi (fun _ => aemeasurable_id)
  27   have hXY := (hcoord.indepFun hij).comp (measurable_of_finite X) (measurable_of_finite Y)
  28   have hh := hXY.integral_mul_eq_mul_integral
  29     (measurable_of_finite (fun σ : ∀ k,State k => X (σ i))).aestronglyMeasurable
  30     (measurable_of_finite (fun σ : ∀ k,State k => Y (σ j))).aestronglyMeasurable
  31   change (∫ σ,X (σ i)*Y (σ j) ∂(FiniteLaw.pi μ).toPMF.toMeasure)=
  32     (∫ σ,X (σ i) ∂(FiniteLaw.pi μ).toPMF.toMeasure)*
  33     (∫ σ,Y (σ j) ∂(FiniteLaw.pi μ).toPMF.toMeasure) at hh
  34   simpa only [finiteLaw_integral,finiteLaw_coordinate_expectation] using hh
