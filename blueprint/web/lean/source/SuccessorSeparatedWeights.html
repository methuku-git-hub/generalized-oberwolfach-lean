  14 structure SuccessorSeparatedWeightConditions {V : Type*} [Fintype V] {k : ℕ}
  15     (G : SimpleGraph V) (σ : Fin (k+2) → Equiv.Perm V) (π : Equiv.Perm V) (b : ℕ)
  16     (D τ A B : ℝ) (w : (Fin (k+2) → V × V) → ℝ) : Prop where
  17   nonneg : ∀ t, 0 ≤ w t
  18   support : ∀ t, 0 < w t → TwistedCyclicTupleValid G σ t ∧ PermutationSeparated π b (pairTupleLabels t)
  19   upper : ∀ t, w t ≤ B / (Fintype.card V : ℝ) ^ (2*k+2)
  20   lower : ∀ t, TwistedCyclicTupleValid G σ t → PermutationSeparated π b (pairTupleLabels t) →
  21     A / (Fintype.card V : ℝ) ^ (2*k+2) ≤ w t
  22   left : ∀ i x y, ¬ PermutationClose π b x y → G.Adj x y →
  23     1-τ ≤ (∑ t, if t i = (x,y) then w t else 0) ∧ (∑ t, if t i = (x,y) then w t else 0) ≤ 1
  24   right : ∀ i x y, ¬ PermutationClose π b x y → G.Adj x (σ i y) →
  25     1-τ ≤ (∑ t, if ((t i).2,(t (finRotate (k+2) i)).1) = (x,y) then w t else 0) ∧
  26       (∑ t, if ((t i).2,(t (finRotate (k+2) i)).1) = (x,y) then w t else 0) ≤ 1
  27   positions : ∀ i v,
  28     ((1-τ)*D ≤ (∑ t, if (t i).1 = v then w t else 0) ∧
  29       (∑ t, if (t i).1 = v then w t else 0) ≤ D) ∧
  30     ((1-τ)*D ≤ (∑ t, if (t i).2 = v then w t else 0) ∧
  31       (∑ t, if (t i).2 = v then w t else 0) ≤ D)

  33 theorem successor_separation_step {V : Type*} [Fintype V] [Nonempty V] {k : ℕ}
  34     (G : SimpleGraph V) (σ : Fin (k+2) → Equiv.Perm V) (π : Equiv.Perm V) (b : ℕ)
  35     (D τ A B : ℝ) (hB : 0 ≤ B) (w : (Fin (k+2) → V × V) → ℝ)
  36     (hw : SuccessorPreweightConditions G σ D (τ/2) (τ/2) A B w)
  37     (hvertex : (2*(k+2 : ℝ))^2*(2*b+1)*B ≤ τ*D/2)
  38     (hedge : (2*(k+2 : ℝ))^2*(2*b+1)*B / Fintype.card V ≤ τ/2) :
  39     SuccessorSeparatedWeightConditions G σ π b D τ A B (separatedPairWeight π b w) := by
  40   have hcard : Fintype.card (Fin (k+2) ⊕ Fin (k+2)) = 2*(k+2) := by simp; omega
  41   have hcardR : (Fintype.card (Fin (k+2) ⊕ Fin (k+2)) : ℝ) = 2*(k+2 : ℝ) := by exact_mod_cast hcard
  42   have he : Fintype.card (Fin (k+2) ⊕ Fin (k+2))-2 = 2*k+2 := by omega
  43   have hupper : ∀ t, w t ≤ B / (Fintype.card V : ℝ) ^
  44       (Fintype.card (Fin (k+2) ⊕ Fin (k+2))-2) := by simpa only [he] using hw.upper
  45   have hpoint (t : Fin (k+2) → V × V) :
  46       0 ≤ separatedPairWeight π b w t ∧ separatedPairWeight π b w t ≤ w t := by
  47     unfold separatedPairWeight
  48     split_ifs
  49     · exact ⟨hw.nonneg t,le_rfl⟩
  50     · exact ⟨le_rfl,hw.nonneg t⟩
  51   refine ⟨fun t => (hpoint t).1,?_,fun t => (hpoint t).2.trans (hw.upper t),?_,?_,?_,?_⟩
  52   · intro t ht
  53     have hs : PermutationSeparated π b (pairTupleLabels t) := by
  54       by_contra hn
  55       simp only [separatedPairWeight,hn,ite_false] at ht
  56       exact (lt_irrefl 0 ht)
  57     refine ⟨?_,hs⟩
  58     by_contra hv
  59     have hz := hw.invalid_zero t hv
  60     linarith [(hpoint t).2]
  61   · intro t ht hs
  62     simp only [separatedPairWeight,hs,ite_true]
  63     exact hw.lower t ht (permutationSeparated_injective π b _ hs)
  64   · intro i x y hxy hG
  65     have h := separatedPairWeight_two_positions π b (by omega : 3 ≤ Fintype.card (Fin (k+2) ⊕ Fin (k+2)))
  66       (Sum.inl i) (Sum.inr i) (by intro h; cases h) x y hxy w hw.nonneg B hB hupper
  67     have heq (t : Fin (k+2) → V × V) : t i = (x,y) ↔ (t i).1=x ∧ (t i).2=y := Prod.ext_iff
  68     simp only [pairTupleLabels,← heq,hcardR] at h
  69     obtain ⟨hl,hu⟩ := hw.left i x y (fun he => hxy (Or.inl he)) hG
  70     exact ⟨by linarith [h.1],h.2.trans hu⟩
  71   · intro i x y hxy hG
  72     have h := separatedPairWeight_two_positions π b (by omega : 3 ≤ Fintype.card (Fin (k+2) ⊕ Fin (k+2)))
  73       (Sum.inr i) (Sum.inl (finRotate (k+2) i)) (by intro h; cases h) x y hxy w hw.nonneg B hB hupper
  74     have heq (t : Fin (k+2) → V × V) : ((t i).2,(t (finRotate (k+2) i)).1) = (x,y) ↔
  75         (t i).2=x ∧ (t (finRotate (k+2) i)).1=y := by simp only [Prod.mk.injEq]
  76     simp only [pairTupleLabels,← heq,hcardR] at h
  77     obtain ⟨hl,hu⟩ := hw.right i x y (fun he => hxy (Or.inl he)) hG
  78     exact ⟨by linarith [h.1],h.2.trans hu⟩
  79   · intro i v
  80     have h₁ := separatedPairWeight_one_position π b (by omega : 2 ≤ Fintype.card (Fin (k+2) ⊕ Fin (k+2)))
  81       (Sum.inl i) v w hw.nonneg B hB hupper
  82     have h₂ := separatedPairWeight_one_position π b (by omega : 2 ≤ Fintype.card (Fin (k+2) ⊕ Fin (k+2)))
  83       (Sum.inr i) v w hw.nonneg B hB hupper
  84     simp only [pairTupleLabels,hcardR] at h₁ h₂
  85     obtain ⟨⟨hl₁,hu₁⟩,⟨hl₂,hu₂⟩⟩ := hw.positions i v
  86     exact ⟨⟨by nlinarith [h₁.1],h₁.2.trans hu₁⟩,⟨by nlinarith [h₂.1],h₂.2.trans hu₂⟩⟩

  88 theorem successor_separated_weight_existence (k : ℕ) (β : ℝ) (hβ : 0 < β) :
  89     ∀ τ : ℝ, 0 < τ → ∃ ε₀ : ℝ, 0 < ε₀ ∧ ∀ b : ℕ, ∃ N : ℕ,
  90       ∀ (V : Type) [Fintype V] [Nonempty V], N ≤ Fintype.card V →
  91       ∀ (G : SimpleGraph V) (π : Equiv.Perm V) (mark : Fin (k+2) → Bool) (D : ℝ),
  92       (Fintype.card V : ℝ)/2 ≤ D → D ≤ 2*Fintype.card V →
  93       (∀ v, |(G.degree v : ℝ)-D| ≤ ε₀*Fintype.card V) →
  94       (∀ v, (successorCycleRatio k + β)*Fintype.card V ≤ (G.degree v : ℝ)) →
  95       ∃ w, SuccessorSeparatedWeightConditions G (fun i => choosePermutation π (mark i)) π b D τ
  96         (successorWeightLower k β) (successorWeightUpper k β) w := by
  97   intro τ hτ
  98   obtain ⟨ε₀,hε₀,N₀,hN₀⟩ := successor_preweight_existence k β hβ τ hτ
  99   refine ⟨ε₀,hε₀,?_⟩
 100   intro b
 101   let B := successorWeightUpper k β
 102   have hB : 0 < B := (successorWeightConstants_pos k β hβ).2
 103   let M := (2*(k+2 : ℝ))^2*(2*b+1)*B
 104   have hM : 0 < M := by dsimp [M]; positivity
 105   obtain ⟨N₁,hN₁⟩ := eventually_reciprocal_small (τ/(4*M)) (by positivity)
 106   refine ⟨max N₀ N₁,?_⟩
 107   intro V _ _ hn G π mark D hD hDup hnear hmin
 108   obtain ⟨w,hw⟩ := hN₀ V ((le_max_left _ _).trans hn) G π mark D hD hDup hnear hmin
 109   have hrec := (hN₁ (Fintype.card V) ((le_max_right _ _).trans hn)).2
 110   have hnR : (0 : ℝ) < Fintype.card V := by exact_mod_cast Fintype.card_pos
 111   have hMsmall : M / Fintype.card V ≤ τ/4 := by
 112     have hh := mul_le_mul_of_nonneg_left hrec hM.le
 113     have he : M * (τ/(4*M)) = τ/4 := by field_simp
 114     rw [he] at hh
 115     simpa only [mul_one_div] using hh
 116   have hMvertex : M ≤ τ*D/2 := by
 117     have hh := (div_le_iff₀ hnR).mp hMsmall
 118     have hd := mul_le_mul_of_nonneg_left hD hτ.le
 119     nlinarith
 120   exact ⟨separatedPairWeight π b w,successor_separation_step G _ π b D τ _ B hB.le w hw
 121     hMvertex (by dsimp [M] at hMsmall; linarith)⟩
