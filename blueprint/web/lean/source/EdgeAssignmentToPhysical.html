  11 def extendEdgeAssignment (G:SimpleGraph V) (f:{e:Sym2 V // e∈G.edgeFinset}→W):Sym2 V→W:=
  12   fun e=>if he:e∈G.edgeFinset then f ⟨e,he⟩ else Classical.choice inferInstance

  18 theorem extend_edge_assignment_fiber_card (G:SimpleGraph V) (f:{e:Sym2 V // e∈G.edgeFinset}→W) (w:W):
  19     (G.edgeFinset.filter (fun e=>extendEdgeAssignment G f e=w)).card=
  20       (Finset.univ.filter (fun e:{e:Sym2 V // e∈G.edgeFinset}=>f e=w)).card := by
  21   symm
  22   apply Finset.card_bij (fun e _=>e.1)
  23   · intro e he
  24     have he':f e=w:=(Finset.mem_filter.mp he).2
  25     exact Finset.mem_filter.mpr ⟨e.2,by simpa [extend_edge_assignment_value G f e.1 e.2] using he'⟩
  26   · intro a ha b hb hab;exact Subtype.ext hab
  27   · intro e he
  28     have hh:=Finset.mem_filter.mp he
  29     let x:{e:Sym2 V // e∈G.edgeFinset}:=⟨e,hh.1⟩
  30     have hx:f x=w:=by simpa [extend_edge_assignment_value G f e hh.1] using hh.2
  31     exact ⟨x,Finset.mem_filter.mpr ⟨Finset.mem_univ x,hx⟩,rfl⟩

  33 theorem extend_edge_assignment_matching (G:SimpleGraph V) (f:{e:Sym2 V // e∈G.edgeFinset}→W)
  34     (hmatch:∀e e',e≠e'→edgeAssignmentConflict e.1 e'.1→f e≠f e'):
  35     ∀i,∀e∈G.edgeFinset,∀e'∈G.edgeFinset,
  36       extendEdgeAssignment G f e=i→extendEdgeAssignment G f e'=i→e≠e'→Disjoint e.toFinset e'.toFinset := by
  37   intro i e he e' he' hi hi' hne
  38   by_contra hdis
  39   let x:{e:Sym2 V // e∈G.edgeFinset}:=⟨e,he⟩
  40   let y:{e:Sym2 V // e∈G.edgeFinset}:=⟨e',he'⟩
  41   have hxy:x≠y:=by intro h;exact hne (congrArg Subtype.val h)
  42   have hc:edgeAssignmentConflict x.1 y.1:=hdis
  43   have hf:f x=f y:=by
  44     have h1:f x=i:=by simpa [x,extend_edge_assignment_value G f e he] using hi
  45     have h2:f y=i:=by simpa [y,extend_edge_assignment_value G f e' he'] using hi'
  46     exact h1.trans h2.symm
  47   exact hmatch x y hxy hc hf
