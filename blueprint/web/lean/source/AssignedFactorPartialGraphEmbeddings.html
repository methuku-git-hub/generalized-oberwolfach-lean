  14 theorem assigned_factor_partial_graph_embeddings
  15     (F:U→SimpleGraph (Fin n)) (hub:A→U)
  16     (C:∀a,(F (hub a)).ConnectedComponent)
  17     (assign:A↪(Σu,(F u).ConnectedComponent))
  18     (hC:∀a,(⟨hub a,C a⟩:Σu,(F u).ConnectedComponent)=assign a)
  19     (f:∀a,C a↪Fin n)
  20     (hdis:∀a b,a≠b→hub a=hub b→
  21       Disjoint (Finset.univ.image (f a)) (Finset.univ.image (f b)))
  22     (R:A→Fin n→Fin n→Prop)
  23     (hR:∀a (x y:C a),(C a).toSimpleGraph.Adj x y→R a (f a x) (f a y)):
  24     ∀u,
  25     let B:Type:={a:A // hub a=u}
  26     let D:B→(F u).ConnectedComponent:=fun a=>
  27       factorComponentAtAssignedHub F ⟨hub a.val,C a.val⟩ u a.property
  28     let fB:∀a:B,D a↪Fin n:=fun a=>
  29       (factorComponentHubEquiv F ⟨hub a.val,C a.val⟩ u a.property).toEmbedding.trans (f a.val)
  30     ∃ψ:{v:Fin n // ∃a:B,v∈(D a).supp}↪Fin n,
  31       (∀a (x:D a),ψ ⟨x.val,⟨a,x.property⟩⟩=fB a x) ∧
  32       (∀x y,(F u).Adj x.val y.val→∃a:B,R a.val (ψ x) (ψ y)) :=by
  33   intro u B D fB
  34   have hD:Function.Injective D:=fixed_hub_assigned_components_injective F hub C assign hC u
  35   have hB:∀a b:B,a≠b→Disjoint (Finset.univ.image (fB a)) (Finset.univ.image (fB b)):=by
  36     intro a b hne
  37     have hne':a.val≠b.val:=by intro h;exact hne (Subtype.ext h)
  38     have hu:hub a.val=hub b.val:=a.property.trans b.property.symm
  39     apply Finset.disjoint_left.mpr
  40     intro v hv hv'
  41     obtain ⟨x,_,hx⟩:=Finset.mem_image.mp hv
  42     obtain ⟨y,_,hy⟩:=Finset.mem_image.mp hv'
  43     have h₁:v∈Finset.univ.image (f a.val):=by
  44       apply Finset.mem_image.mpr
  45       refine ⟨factorComponentHubEquiv F ⟨hub a.val,C a.val⟩ u a.property x,Finset.mem_univ _,?_⟩
  46       exact hx
  47     have h₂:v∈Finset.univ.image (f b.val):=by
  48       apply Finset.mem_image.mpr
  49       refine ⟨factorComponentHubEquiv F ⟨hub b.val,C b.val⟩ u b.property y,Finset.mem_univ _,?_⟩
  50       exact hy
  51     exact (Finset.disjoint_left.mp (hdis a.val b.val hne' hu)) h₁ h₂
  52   apply component_family_partial_graph_embedding (F u) D hD fB hB (fun a=>R a.val)
  53   intro a x y hxy
  54   apply hR a.val
  55   exact (factor_component_hub_equiv_adj F ⟨hub a.val,C a.val⟩ u a.property x y).mpr hxy
  56 end Oberwolfach
