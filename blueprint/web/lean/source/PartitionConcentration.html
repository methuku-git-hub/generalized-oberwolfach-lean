  13     {V : Type*} [Fintype V] {k : ℕ} {d : Fin k → ℕ}
  14     [Nonempty (FixedPartition V k d)]
  15     (f : FixedPartition V k d → ℝ) {b : ℝ} (hb : 0 ≤ b)
  16     (hexchange : ∀ P u v, |f P - f (fixedPartitionMap (Equiv.swap u v) P)| ≤ b)
  17     (t : ℝ) :
  18     (FiniteLaw.uniform (FixedPartition V k d)).expectation
  19       (fun P => Real.exp (t * (f P - (FiniteLaw.uniform (FixedPartition V k d)).expectation f))) ≤
  20       Real.exp (t ^ 2 * (Fintype.card V : ℝ) * b ^ 2 / 8) := by
  21   let P := Classical.choice (inferInstance : Nonempty (FixedPartition V k d))
  22   rw [← fixedPartition_uniform P, finiteLaw_map_expectation, finiteLaw_map_expectation]
  23   apply uniformPermutation_mgf _ hb
  24   intro σ u v
  25   rw [← fixedPartitionMap_trans]
  26   exact hexchange (fixedPartitionMap σ P) u v
