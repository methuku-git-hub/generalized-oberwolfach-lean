   9 theorem regularization_floor_argument {n d η s β a : ℝ}
  10     (hn : 0 < n) (hβ : 0 ≤ β) (hs : 0 ≤ s) (hs4 : s ≤ 1 / 4)
  11     (hη : η = s ^ 2) (hamin : (1 / 2 + β) * n ≤ a) (haup : a ≤ d + η * n) :
  12     0 ≤ d - s * n := by
  13   have hη16 : η ≤ 1 / 16 := by nlinarith
  14   have hsm := mul_le_mul_of_nonneg_right hs4 hn.le
  15   have hηm := mul_le_mul_of_nonneg_right hη16 hn.le
  16   have hβm := mul_nonneg hβ hn.le
  17   nlinarith

  19 theorem regularization_floor_demand {n d η s : ℝ} (a : ℕ)
  20     (hn : 0 < n) (harg : 0 ≤ d - s * n) (hηs : η ≤ s)
  21     (hnlarge : 1 ≤ η * n)
  22     (ha : d - η * n ≤ (a : ℝ) ∧ (a : ℝ) ≤ d + η * n) :
  23     let k := ⌊d - s * n⌋₊
  24     k ≤ a ∧ (s - η) * n ≤ ((a - k : ℕ) : ℝ) ∧
  25       ((a - k : ℕ) : ℝ) ≤ (s + 2 * η) * n := by
  26   dsimp only
  27   have hlo := Nat.floor_le harg
  28   have hhi := Nat.lt_floor_add_one (d - s * n)
  29   have hsn := mul_le_mul_of_nonneg_right hηs hn.le
  30   have hka : (⌊d - s * n⌋₊ : ℝ) ≤ a := by linarith [ha.1]
  31   have hka' : ⌊d - s * n⌋₊ ≤ a := by exact_mod_cast hka
  32   refine ⟨hka', ?_, ?_⟩ <;> rw [Nat.cast_sub hka'] <;> nlinarith [ha.1, ha.2]

  34 theorem regularization_relative_scale {s η δ : ℝ}
  35     (hs : 0 < s) (hη : η = s ^ 2) (hδ : 0 < δ) (hδ8 : δ ≤ 1 / 8)
  36     (hsmall : s ≤ δ / 8) :
  37     0 < s - η ∧ s + 2 * η ≤ (1 + δ) * (s - η) := by
  38   have hs1 : s < 1 := by linarith
  39   have hη0 : 0 ≤ η := by rw [hη]; positivity
  40   have hηs : η < s := by nlinarith
  41   have hδs := mul_le_mul_of_nonneg_right hsmall hs.le
  42   have hδη := mul_le_mul_of_nonneg_right hδ8 hη0
  43   constructor
  44   · exact sub_pos.mpr hηs
