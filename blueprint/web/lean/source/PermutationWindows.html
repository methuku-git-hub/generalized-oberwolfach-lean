  16 def permutationWindow (π : Equiv.Perm V) (left right : ℕ) (x : V) : Finset V :=
  17   Finset.univ.image (fun i : Fin (left+right+1)=>(π^i.val) ((π^left).symm x))

  19 theorem permutationWindow_card_le (π : Equiv.Perm V) (left right : ℕ) (x : V) :
  20     (permutationWindow π left right x).card≤left+right+1 := by
  21   exact Finset.card_image_le.trans (by simp)

  23 theorem permutationWindow_card (π : Equiv.Perm V) (left right : ℕ)
  24     (hnoshort : ∀ x k,0<k → k<left+right+1 → (π^k) x≠x) (x : V) :

  31 theorem permutation_power_cancel_image (π : Equiv.Perm V) (i j : ℕ) (x : V) :
  32     (π^i) ((π^j).symm x)=(π^j).symm ((π^i) x) := by
  33   apply (π^j).injective
  34   rw [Equiv.apply_symm_apply]
  35   rw [← Equiv.Perm.mul_apply,← pow_add,Nat.add_comm,pow_add,Equiv.Perm.mul_apply,
  36     Equiv.apply_symm_apply]

  38 theorem permutationWindow_mem_iff (π : Equiv.Perm V) (left right : ℕ) (x y : V) :
  39     y∈permutationWindow π left right x ↔
  40       ∃ i : Fin (left+right+1),(π^i.val) x=(π^left) y := by
  41   simp only [permutationWindow,Finset.mem_image,Finset.mem_univ,true_and]
  42   apply exists_congr
  43   intro i
  44   rw [permutation_power_cancel_image]
  45   exact (π^left).symm_apply_eq

  47 theorem permutationWindow_dual (π : Equiv.Perm V) (left right : ℕ) (x y : V) :
  48     y∈permutationWindow π left right x ↔ x∈permutationWindow π right left y := by
  49   rw [permutationWindow_mem_iff,permutationWindow_mem_iff]
  50   have hdir (l r : ℕ) (a b : V)
  51       (h : ∃ i : Fin (l+r+1),(π^i.val) a=(π^l) b) :
  52       ∃ j : Fin (r+l+1),(π^j.val) b=(π^r) a := by
  53     obtain ⟨i,hi⟩ := h
  54     refine ⟨⟨l+r-i.val,by omega⟩,?_⟩
  55     apply (π^l).injective
  56     calc
  57       _ = (π^(l+r-i.val)) ((π^l) b) := by
  58         rw [← Equiv.Perm.mul_apply,← pow_add,Nat.add_comm,pow_add,Equiv.Perm.mul_apply]
  59       _ = (π^(l+r-i.val)) ((π^i.val) a) := by rw [hi]
  60       _ = (π^(l+r)) a := by rw [← Equiv.Perm.mul_apply,← pow_add,Nat.sub_add_cancel (by omega)]
  61       _ = _ := by rw [pow_add,Equiv.Perm.mul_apply]
  62   exact ⟨hdir left right x y,hdir right left y x⟩

  64 theorem permutationWindow_reverse_card_le (π : Equiv.Perm V) (left right : ℕ) (y : V) :
  65     (Finset.univ.filter (fun x=>y∈permutationWindow π left right x)).card≤left+right+1 := by
  66   have he : Finset.univ.filter (fun x=>y∈permutationWindow π left right x)=
  67       permutationWindow π right left y := by
  68     ext x;simp only [Finset.mem_filter,Finset.mem_univ,true_and,permutationWindow_dual]
  69   rw [he]
  70   simpa only [Nat.add_comm] using permutationWindow_card_le π right left y

  72 theorem permutationClose_of_power_eq (π : Equiv.Perm V) (b i j : ℕ) (hi : i≤b) (hj : j≤b)
  73     (x y : V) (he : (π^i) x=(π^j) y) : PermutationClose π b x y := by
  74   by_cases hij : i=j
  75   · subst j;exact Or.inl ((π^i).injective he)
  76   rcases lt_or_gt_of_ne hij with hlt|hgt
  77   · refine Or.inr ⟨⟨j-i-1,by omega⟩,Or.inr ?_⟩
  78     apply (π^i).injective
  79     rw [← Equiv.Perm.mul_apply,← pow_add,show i+(j-i-1+1)=j by omega]
  80     exact he.symm
  81   · refine Or.inr ⟨⟨i-j-1,by omega⟩,Or.inl ?_⟩
  82     apply (π^j).injective
  83     rw [← Equiv.Perm.mul_apply,← pow_add,show j+(i-j-1+1)=i by omega]
  84     exact he

  86 theorem permutationWindow_disjoint (π : Equiv.Perm V) (left right : ℕ) (x y : V)
  87     (hfar : ¬PermutationClose π (left+right) x y) :
  88     Disjoint (permutationWindow π left right x) (permutationWindow π left right y) := by
  89   apply Finset.disjoint_left.mpr
  90   intro z hx hy
  91   obtain ⟨i,hi⟩ := (permutationWindow_mem_iff π left right x z).mp hx
  92   obtain ⟨j,hj⟩ := (permutationWindow_mem_iff π left right y z).mp hy
  93   exact hfar (permutationClose_of_power_eq π (left+right) i.val j.val (by omega) (by omega)
  94     x y (hi.trans hj.symm))

 102 theorem permutationWindow_comp_subset (π : Equiv.Perm V) (l₁ r₁ l₂ r₂ : ℕ)
 103     (x y z : V) (hy : y∈permutationWindow π l₁ r₁ x)
 104     (hz : z∈permutationWindow π l₂ r₂ y) :
 105     z∈permutationWindow π (l₁+l₂) (r₁+r₂) x := by
 106   obtain ⟨i,hi⟩ := (permutationWindow_mem_iff π l₁ r₁ x y).mp hy
 107   obtain ⟨j,hj⟩ := (permutationWindow_mem_iff π l₂ r₂ y z).mp hz
 108   apply (permutationWindow_mem_iff π (l₁+l₂) (r₁+r₂) x z).mpr
 109   refine ⟨⟨i.val+j.val,by omega⟩,?_⟩
 110   change (π^(i.val+j.val)) x=(π^(l₁+l₂)) z
 111   calc
 112     _ = (π^j.val) ((π^i.val) x) := by rw [Nat.add_comm,pow_add,Equiv.Perm.mul_apply]
 113     _ = (π^j.val) ((π^l₁) y) := by rw [hi]
 114     _ = (π^l₁) ((π^j.val) y) := by rw [← Equiv.Perm.mul_apply,← pow_add,Nat.add_comm,pow_add,Equiv.Perm.mul_apply]
 115     _ = (π^l₁) ((π^l₂) z) := by rw [hj]
 116     _ = _ := by rw [pow_add,Equiv.Perm.mul_apply]

 118 theorem permutationWindow_mono (π : Equiv.Perm V) {l₁ r₁ l₂ r₂ : ℕ}
 119     (hl : l₁≤l₂) (hr : r₁≤r₂) (x : V) :
 120     permutationWindow π l₁ r₁ x⊆permutationWindow π l₂ r₂ x := by
 121   intro y hy
 122   obtain ⟨i,hi⟩ := (permutationWindow_mem_iff π l₁ r₁ x y).mp hy
 123   apply (permutationWindow_mem_iff π l₂ r₂ x y).mpr
 124   refine ⟨⟨i.val+(l₂-l₁),by omega⟩,?_⟩
 125   change (π^(i.val+(l₂-l₁))) x=(π^l₂) y
 126   calc
 127     _ = (π^(l₂-l₁)) ((π^i.val) x) := by rw [Nat.add_comm,pow_add,Equiv.Perm.mul_apply]
 128     _ = (π^(l₂-l₁)) ((π^l₁) y) := by rw [hi]
 129     _ = _ := by rw [← Equiv.Perm.mul_apply,← pow_add,Nat.sub_add_cancel hl]
