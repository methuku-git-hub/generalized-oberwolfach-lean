   8 theorem factor_small_class_budget {W:Type} [Fintype W] (K:ℕ) (hK:0<K)
   9     (classify:W→Option (Fin K)) (n α κL κS:ℝ)
  10     (hn:0≤n) (hα:0<α) (hL0:0≤κL) (hS0:0≤κS)
  11     (hL:κL≤α/1000) (hS:κS≤α/(1000*(K:ℝ))):
  12     let fibres:=fun j:Option (Fin K)=>Finset.univ.filter (fun w=>classify w=j)
  13     let small:=fun j:Option (Fin K)=>
  14       ((fibres j).card:ℝ)<(if j=none then κL*n else κS*n)
  15     let removed:=Finset.univ.filter (fun w=>small (classify w));
  16     (removed.card:ℝ)≤α*n/500:=by
  17   intro fibres small removed
  18   have hK':(0:ℝ)<K:=by exact_mod_cast hK
  19   have hKS:(K:ℝ)*κS≤α/1000:=by
  20     have h:κS*(K:ℝ)≤α/1000:=by
  21       apply (le_div_iff₀ hK').mp
  22       have hh:=hS
  23       convert hh using 1 <;> ring
  24     nlinarith
  25   have hco:κL+(K:ℝ)*κS≤α/500:=by linarith
  26   have hsum:=factor_class_pruning_bound K classify n κL κS hn hL0 hS0
  27   have hraw:(removed.card:ℝ)≤(κL+(K:ℝ)*κS)*n:=by simpa [fibres,small,removed] using hsum.1
  28   calc
  29     _ ≤ (κL+(K:ℝ)*κS)*n:=hraw
  30     _ ≤ (α/500)*n:=mul_le_mul_of_nonneg_right hco hn
  31     _ = α*n/500:=by ring
  32 end Oberwolfach
