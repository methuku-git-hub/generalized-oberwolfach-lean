  18 theorem marked_removed_resources_empty_inner
  19     (P : SimpleGraph V) (π : Equiv.Perm V) (d : ℕ) (σ : RawMarkedOutcome V U Γ Color) :
  20     let E := MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) false
  21     let K := MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) true;
  22     ∀ (A : Finset (E⊕K)) (S : Finset (V×U)) (I : Finset V) (J : Finset U),
  23     (markedFiniteResources (fun e:E=>e.val.1) (fun e=>e.val.2)
  24       (fun e:K=>e.val.1) (fun e=>e.val.2) π A S).filter (markedInnerResource I J)=∅→
  25     (∀ x∈I,∀ y∈I,¬(markedSampleArcSetGraph P π d σ A).Adj x y) ∧
  26     (∀ x∈I,∀ u∈J,¬(markedFiniteSpokeGraph S).Adj (Sum.inl x) (Sum.inr u)) := by
  27   intro E K A S I J hEmpty
  28   let R := markedFiniteResources (fun e:E=>e.val.1) (fun e=>e.val.2)
  29     (fun e:K=>e.val.1) (fun e=>e.val.2) π A S
  30   have hNot (e : Finset V⊕(V×U)) (he : e∈R) : ¬markedInnerResource I J e := by
  31     intro hi
  32     have hh : e∈R.filter (markedInnerResource I J) := Finset.mem_filter.mpr ⟨he,hi⟩
  33     rw [hEmpty] at hh
  34     exact Finset.notMem_empty _ hh
  35   constructor
  36   · intro x hx y hy hxy
  37     obtain ⟨a,ha,he⟩ := Finset.mem_image.mp hxy.2
  38     have hr : Sum.inl ({x,y}:Finset V)∈R := by
  39       apply Finset.mem_union_left
  40       exact Finset.mem_image.mpr ⟨a,ha,congrArg Sum.inl he⟩
  41     apply hNot _ hr
  42     change ∀ z∈({x,y}:Finset V),z∈I
  43     intro z hz
  44     rcases Finset.mem_insert.mp hz with rfl|hz
  45     · exact hx
  46     · exact Finset.mem_singleton.mp hz ▸ hy
  47   · intro x hx u hu hxu
  48     have hs : (x,u)∈S := (markedFiniteSpokeGraph_adj S x u).mp hxu
  49     have hr : Sum.inr (x,u)∈R := Finset.mem_union_right _ (Finset.mem_image.mpr ⟨(x,u),hs,rfl⟩)
  50     exact hNot _ hr ⟨hx,hu⟩
