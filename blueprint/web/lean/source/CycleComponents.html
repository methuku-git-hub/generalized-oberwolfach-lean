  11 def internalEdges {V : Type*} [Fintype V] (G : SimpleGraph V) (A : Finset V) :
  12     Finset (Sym2 V) :=
  13   G.edgeFinset.filter (fun e => ∀ x ∈ e, x ∈ A)

  15 theorem mem_internalEdges {V : Type*} [Fintype V] (G : SimpleGraph V) (A : Finset V)
  16     (u v : V) : s(u, v) ∈ internalEdges G A ↔ G.Adj u v ∧ u ∈ A ∧ v ∈ A := by
  17   simp only [internalEdges, Finset.mem_filter, SimpleGraph.mem_edgeFinset,
  18     SimpleGraph.mem_edgeSet, Sym2.forall_mem_pair]

  20 theorem IsCycleCopy.internal_edge_count {n r : ℕ} {H : SimpleGraph (Fin n)}
  21     (hH : IsCycleCopy H r) (A : Finset (Fin n))
  22     (hclosed : ∀ u v, H.Adj u v → (u ∈ A ↔ v ∈ A)) :
  23     (internalEdges H A).card = r ∨ (internalEdges H A).card = 0 := by
  24   classical
  25   obtain ⟨hr, f, hf⟩ := hH
  26   let i₀ : Fin r := ⟨0, by omega⟩
  27   have hfc : ∀ i j : Fin r, f i ∈ A ↔ f j ∈ A := by
  28     intro i j
  29     obtain ⟨p⟩ := (SimpleGraph.cycleGraph_preconnected (n := r)) i j
  30     induction p with
  31     | nil => rfl
  32     | @cons u v w huv p ih =>
  33       apply Iff.trans ?_ ih
  34       apply hclosed
  35       apply (hf (f u) (f v)).mpr
  36       exact ⟨u, v, rfl, rfl, (cycleGraph_modular_adj (by omega : 2 ≤ r) u v).mp huv⟩
  37   by_cases hzero : f i₀ ∈ A
  38   · left
  39     have hall : internalEdges H A = H.edgeFinset := by
  40       ext e
  41       induction e using Sym2.ind with
  42       | _ u v =>
  43         rw [mem_internalEdges, SimpleGraph.mem_edgeFinset]
  44         refine ⟨fun he => he.1, fun he => ⟨he, ?_⟩⟩
  45         obtain ⟨i, j, rfl, rfl, hij⟩ := (hf u v).mp he
  46         exact ⟨(hfc i i₀).mpr hzero, (hfc j i₀).mpr hzero⟩
  47     rw [hall]
  48     exact IsCycleCopy.edge_card ⟨hr, f, hf⟩
  49   · right
  50     have hempty : internalEdges H A = ∅ := by
  51       ext e
  52       induction e using Sym2.ind with
  53       | _ u v =>
  54         rw [mem_internalEdges]
  55         simp only [Finset.notMem_empty, iff_false]
  56         rintro ⟨he, hu, hv⟩
  57         obtain ⟨i, j, rfl, rfl, hij⟩ := (hf u v).mp he
  58         exact hzero ((hfc i i₀).mp hu)
  59     rw [hempty, Finset.card_empty]

  61 theorem HasCycleDecomposition.closed_set_edge_divisible {n r : ℕ}
  62     {G : SimpleGraph (Fin n)} (hG : HasCycleDecomposition G r)
  63     (A : Finset (Fin n)) (hclosed : ∀ u v, G.Adj u v → (u ∈ A ↔ v ∈ A)) :
  64     r ∣ (internalEdges G A).card := by
  65   classical
  66   obtain ⟨q, H, hpart, hH⟩ := hG
  67   rw [internalEdges, hpart.filtered_edge_count]
  68   apply Finset.dvd_sum
  69   intro i hi
  70   have hci := (hH i).internal_edge_count A
  71     (fun u v huv => hclosed u v (hpart.piece_le i huv))
  72   rcases hci with he | he
  73   · change r ∣ (internalEdges (H i) A).card
  74     rw [he]
  75   · change r ∣ (internalEdges (H i) A).card
  76     rw [he]
  77     exact dvd_zero r
