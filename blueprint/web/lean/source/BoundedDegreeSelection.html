  22 theorem bounded_degree_selection (p : ℝ) (hp : 0 < p) :
  23     ∃ n₀ : ℕ, 2 ≤ n₀ ∧ ∀ (V : Type*) [Fintype V] [DecidableEq V],
  24     n₀ ≤ Fintype.card V → ∀ (N D : ℕ) (G : SimpleGraph V)
  25     (S : Fin N → Finset V) (family : Fin N → Set (SimpleGraph V)),
  26     (∀ i j, i ≠ j → ((S i ∩ S j).card : ℝ) ≤ p^2 * Fintype.card V) →
  27     (∀ v, ((Finset.univ.filter (fun i => v ∈ S i)).card : ℝ) ≤ p * Fintype.card V) →
  28     (∀ i, ∀ J : SimpleGraph V, J ≤ spanningRestriction G (S i) →
  29       (∀ v, (J.degree v : ℝ) ≤ D * (p * Real.sqrt p * Fintype.card V)) →
  30       ∃ C : Fin (Nat.ceil (4 * (p * Real.sqrt p * Fintype.card V))) → SimpleGraph V,
  31         (∀ a, C a ∈ family i ∧ C a ≤ spanningRestriction G (S i) ∧
  32           (∀ v, (C a).degree v ≤ D) ∧ Disjoint (C a) J) ∧
  33         (∀ a b, a ≠ b → Disjoint (C a) (C b))) →
  34     ∃ F : Fin N → SimpleGraph V,
  35       (∀ i, F i ∈ family i ∧ F i ≤ spanningRestriction G (S i) ∧ ∀ v, (F i).degree v ≤ D) ∧
  36       (∀ i j, i ≠ j → Disjoint (F i) (F j)) ∧
  37       ∀ v, ((⨆ i, F i).degree v : ℝ) ≤ D * (p * Fintype.card V) := by
  38   obtain ⟨n₀, hn₀, horder⟩ := bounded_degree_selection_order p hp
  39   refine ⟨n₀, hn₀, ?_⟩
  40   intro V _ _ hn N D G S family hinter hmem hcandidates
  41   let n := Fintype.card V
  42   let s := p * Real.sqrt p * n
  43   let t := Nat.ceil (4 * s)
  44   let B := Nat.floor s + 1
  45   have hnpos : (0 : ℝ) < n := by exact_mod_cast (show 0 < n from lt_of_lt_of_le (by omega) hn)
  46   have hs : 0 ≤ s := by dsimp [s]; positivity
  47   have hscale := selection_candidate_scale p n hp hnpos
  48   change 0 < t ∧ p^2 * n ≤ (Real.sqrt p / 4) * t ∧ (Nat.floor s : ℝ) ≤ s at hscale
  49   have hcount : (Fintype.card (SelectionVertexTest S) : ℝ) ≤ p * (n : ℝ)^2 := by
  50     have hh := selectionVertexTest_card_le S (p*n) hmem
  51     convert hh using 1 <;> dsimp [n] <;> ring
  52   have hremaining (x : SelectionVertexTest S) :
  53       (selectionRemaining (graphSelectionActive S) 0 x : ℝ) ≤ p*n := by
  54     have hh : (selectionRemaining (graphSelectionActive S) 0 x : ℝ) ≤
  55       ((Finset.univ.filter (fun i => x.val.2 ∈ S i)).card : ℝ) := by
  56       exact_mod_cast graphSelection_remaining_le S 0 x
  57     exact hh.trans (hmem x.val.2)
  58   have hsmall := horder n hn (SelectionVertexTest S) N (graphSelectionActive S) hcount hremaining
  59   exact bounded_degree_selection_finite N D t B hscale.1 (Real.sqrt p/4) s (p*n)
  60     (by positivity) hs (by simpa [B] using hscale.2.2) G S family
  61     (fun i j hij => (hinter i j hij).trans hscale.2.1) hmem hsmall hcandidates
