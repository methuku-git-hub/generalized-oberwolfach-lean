  14 def attachmentWheelEmbedding (z : W) (ρ : I → V) (i : I) : W ↪ V ⊕ (I×W) where
  15   toFun w := if w=z then .inl (ρ i) else .inr (i,w)
  16   inj' := by
  17     intro a b hab
  18     by_cases ha : a=z <;> by_cases hb : b=z
  19     all_goals simp only [ha,hb,ite_true,ite_false,Sum.inr.injEq,Prod.mk.injEq,true_and] at hab
  20     · exact ha.trans hb.symm
  21     · cases hab
  22     · cases hab
  23     · exact hab

  33 theorem attachment_cycle_no_old_edge (C : SimpleGraph W) (z : W) (ρ : I → V)
  34     (i : I) (a b : V) : ¬ (C.map (attachmentWheelEmbedding z ρ i)).Adj (.inl a) (.inl b) := by
  35   intro h
  36   obtain ⟨w,w',hww,ha,hb⟩ := (SimpleGraph.map_adj _ _ _ _).mp h
  37   have hw := (attachmentWheelEmbedding_old z ρ i w a).mp ha
  38   have hw' := (attachmentWheelEmbedding_old z ρ i w' b).mp hb
  39   exact hww.ne (hw.1.trans hw'.1.symm)

  41 theorem attachment_cycle_old_private (C : SimpleGraph W) (z : W) (ρ : I → V)
  42     (i j : I) (a : V) (w : W) :
  43     (C.map (attachmentWheelEmbedding z ρ i)).Adj (.inl a) (.inr (j,w)) ↔
  44       i=j ∧ ρ i=a ∧ w≠z ∧ C.Adj z w := by
  45   rw [SimpleGraph.map_adj]
  46   constructor
  47   · rintro ⟨w₀,w₁,hC,ha,hb⟩

  69 def ordinaryAttachmentRim (G : SimpleGraph V) (C : SimpleGraph W) (z : W) (ρ : I → V) :
  70     SimpleGraph (V ⊕ (I×W)) where
  71   Adj a b := match a,b with
  72     | .inl a,.inl b => G.Adj a b
  73     | .inl a,.inr (i,w) => ρ i=a ∧ w≠z ∧ C.Adj z w
  74     | .inr (i,w),.inl a => ρ i=a ∧ w≠z ∧ C.Adj z w
  75     | .inr (i,w),.inr (j,w') => i=j ∧ w≠z ∧ w'≠z ∧ C.Adj w w'
  76   symm := ⟨by
  77     intro a b h
  78     rcases a with a|⟨i,w⟩ <;> rcases b with b|⟨j,w'⟩
  79     · exact h.symm
  80     · exact h
  81     · exact h
  82     · exact ⟨h.1.symm,h.2.2.1,h.2.1,h.2.2.2.symm⟩⟩
  83   loopless := ⟨by
  84     intro a h
  85     rcases a with a|⟨i,w⟩
  86     · exact h.ne rfl
  87     · exact h.2.2.2.ne rfl⟩

  89 def ordinaryAttachmentSpokes {U : Type*} (H : V → U → Prop) (z : W) (ρ : I → V) :
  90     (V ⊕ (I×W)) → (U ⊕ I) → Prop
  91   | .inl a,.inl b => H a b
  92   | .inl a,.inr i => ρ i=a
  93   | .inr (i,w),.inl _ => False
  94   | .inr (i,w),.inr j => w≠z ∧ i=j
