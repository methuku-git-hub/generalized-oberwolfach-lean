  15 theorem markedSampleArc_predicate_multiplicity
  16     (P : SimpleGraph V) (π : Equiv.Perm V) (b : ℕ)
  17     (σ : Sym2 V→Option (Bool×Bool)) (mark : Bool) (Q : V→V→Prop) :
  18     occurrenceMultiplicity
  19       (fun e:{e:MarkedSampleArc P π b σ mark // Q e.val.1 e.val.2}=>e.val.val.1)
  20       (fun e=>e.val.val.2)=
  21     (fun x y=>if Q x y then markedPhysicalRim P π b σ mark x y else 0) := by
  22   rw [occurrenceMultiplicity_eq_relation _ _ (by
  23     intro a c he;apply Subtype.ext;exact Subtype.ext he)]
  24   funext x y
  25   have he : (∃ e:{e:MarkedSampleArc P π b σ mark // Q e.val.1 e.val.2},
  26       e.val.val.1=x∧e.val.val.2=y) ↔ markedPhysicalRim P π b σ mark x y=1∧Q x y := by
  27     constructor
  28     · rintro ⟨e,hx,hy⟩
  29       exact ⟨by simpa only [hx,hy] using e.val.property,by simpa only [hx,hy] using e.property⟩
  30     · rintro ⟨h,hq⟩
  31       exact ⟨⟨⟨(x,y),h⟩,hq⟩,rfl,rfl⟩
  32   simp only [relationMultiplicity,he]
  33   by_cases hq : Q x y
  34   · simp only [hq,and_true,ite_true,markedPhysicalRim_indicator]
  35   · simp only [hq,and_false,ite_false]

  37 theorem markedSampleArc_opposite_domain_degrees
  38     (P : SimpleGraph V) (π : Equiv.Perm V) (b : ℕ)
  39     (σ : Sym2 V→Option (Bool×Bool)) (mark : Bool) (x : V) (A : Finset V) :
  40     let E := {e:MarkedSampleArc P π b σ mark // (if e.val.1=x then e.val.2 else e.val.1)∈A};
  41     occurrenceDegree (fun e:E=>e.val.val.1) x=(markedRestrictedDegree P π b σ mark false x A:ℝ) ∧
  42     occurrenceDegree (fun e:E=>e.val.val.2) x=(markedRestrictedDegree P π b σ mark true x A:ℝ) := by
  43   intro E
  44   have he := markedSampleArc_predicate_multiplicity P π b σ mark
  45     (fun u v=>(if u=x then v else u)∈A)
  46   have he' : occurrenceMultiplicity (fun e:E=>e.val.val.1) (fun e=>e.val.val.2)=
  47       (fun u v=>if (if u=x then v else u)∈A then markedPhysicalRim P π b σ mark u v else 0) := by
  48     convert he using 1 <;> congr!
  49   constructor
  50   · rw [←occurrenceMultiplicity_row (fun e:E=>e.val.val.1) (fun e=>e.val.val.2),he']

  53     simp only [Finset.filter_univ_mem]
