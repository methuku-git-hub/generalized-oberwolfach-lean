  18 theorem endpoint_triangle_graph_eq {W:Type} [Fintype W] [DecidableEq W]
  19     (P F:SimpleGraph W) (A B:Finset W) (hAB:Disjoint A B) (C:EndpointTriangle P F A B):
  20     (endpointAsLocalCycle hAB C).val.val.val=spanningRestriction ⊤ C.val:=by
  21   ext v w
  22   rw [complete_on_adj]
  23   have ht:=(endpointAsLocalCycle hAB C).val.val.property.triangle_adj v w
  24   have hmem (a:W):a∈(endpointAsLocalCycle hAB C).val.val.val.support↔a∈C.val:=by
  25     rw [←Set.mem_toFinset]
  26     change a∈localOrdinaryVertices (endpointAsLocalCycle hAB C)↔a∈C.val
  27     rw [endpointAsLocalCycle_vertices];rfl
