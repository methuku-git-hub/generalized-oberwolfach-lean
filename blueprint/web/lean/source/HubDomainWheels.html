  87     (h:Disjoint (ordinaryWheelResources W) (ordinaryWheelResources Z)):

  89       (ordinaryWheelResources (ordinaryWheelOfHubDomain B Z)):=by
  90   apply Finset.disjoint_left.mpr
  91   intro e he hf

  93   | inl e=>

  95       ((ordinaryWheelResources_rim W e).mpr ((ordinaryWheelResources_rim (ordinaryWheelOfHubDomain B W) e).mp he))

  98     obtain ⟨hw,hv⟩:=(ordinaryWheelResources_spoke _ vu.1 vu.2).mp he
  99     obtain ⟨hz,hv'⟩:=(ordinaryWheelResources_spoke _ vu.1 vu.2).mp hf
