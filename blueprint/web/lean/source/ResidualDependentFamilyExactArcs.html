  11 theorem two_factor_family_dependent_models_exact_arcs
  12     (F H:∀i,SimpleGraph (X i)) (R:∀i,Finset (X i))
  13     (hF:∀i,(F i).IsRegularOfDegree 2) (hHF:∀i,H i≤F i)
  14     (hisol:∀i x,x∉R i→∀y,¬(H i).Adj x y):
  15     ∃ (D:∀i,X i→X i→Prop) (orig pieces:∀i,List (PrescribedSourcePiece (X i))),
  16       (∀i,(orig i).Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset)) ∧
  17       (∀i p,p∈orig i→∃a xs,p=.cycle a xs ∧ p.OriginalValid (D i)) ∧
  18       (∀i,sourcePieceVerticesUnion (orig i)=Finset.univ ∧ sourcePieceEdgesUnion (orig i)=(F i).edgeFinset) ∧
  19       (∀i,(pieces i).Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset)) ∧
  20       (∀i p,p∈pieces i→p.Shape ∧ p.OriginalValid (D i)) ∧
  21       (∀i,prescribedDependentVertices pieces i=R i) ∧
  22       (∀i,prescribedDependentGraph pieces i=H i) ∧
  23       (∀i x y,D i x y ↔ (x,y)∈sourcePieceArcsUnion (orig i)) ∧
  24       ∀i x y,prescribedDependentDigraph pieces i x y ↔ D i x y ∧ (H i).Adj x y := by
  25   choose D orig pieces hpair hcycles hvertices hedges hps hvalid hpsV hpsE hD hpsA using
  26     fun i=>two_factor_subgraph_dependent_model_exact_arcs (F i) (H i) (hF i) (hHF i) (R i) (hisol i)
  27   refine ⟨D,orig,pieces,hpair,hcycles,?_,hps,hvalid,?_,?_,hD,?_⟩
  28   · intro i;exact ⟨hvertices i,hedges i⟩
  29   · intro i;exact hpsV i
  30   · intro i;exact hpsE i
  31   · intro i x y;exact hpsA i x y
  32 end Oberwolfach
