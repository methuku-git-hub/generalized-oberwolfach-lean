  15 def vertexDifference {V : Type*} [DecidableEq V] (x y v : V) : ℝ :=
  16   (if x = v then 1 else 0) - (if y = v then 1 else 0)
  17 
  18 def flowDivergence {V E : Type*} [DecidableEq V] [Fintype E]
  19     (tail head : E → V) (f : E → ℝ) (v : V) : ℝ :=
  20   ∑ e, f e * vertexDifference (tail e) (head e) v
  21 
  22 def CapacityBounded {E : Type*} (c f : E → ℝ) : Prop :=
  23   ∀ e, 0 ≤ f e ∧ f e ≤ c e
  24 
  25 def IsNetworkFlow {V E : Type*} [DecidableEq V] [Fintype E]
  26     (tail head : E → V) (c : E → ℝ) (s t : V) (f : E → ℝ) : Prop :=
  27   CapacityBounded c f ∧ ∀ v, v ≠ s → v ≠ t → flowDivergence tail head f v = 0

  29 /-- A change of flow that transfers a positive amount from `x` to `y`.
  30 The base flow remains fixed in this relation. -/
  31 def FlowTransfer {V E : Type*} [DecidableEq V] [Fintype E]
  32     (tail head : E → V) (c f : E → ℝ) (x y : V) : Prop :=
  33   ∃ ε : ℝ, 0 < ε ∧ ∃ g : E → ℝ, CapacityBounded c g ∧
  34     ∀ v, flowDivergence tail head g v =
  35       flowDivergence tail head f v + ε * vertexDifference x y v

  37 theorem vertexDifference_trans {V : Type*} [DecidableEq V] (x y z v : V) :
  38     vertexDifference x z v + vertexDifference z y v = vertexDifference x y v := by
  39   unfold vertexDifference
  40   ring

  42 theorem flowDivergence_linear
  43     {V E : Type*} [DecidableEq V] [Fintype E]
  44     (tail head : E → V) (f g : E → ℝ) (a b : ℝ) (v : V) :
  45     flowDivergence tail head (fun e => a * f e + b * g e) v =
  46       a * flowDivergence tail head f v + b * flowDivergence tail head g v := by
  47   simp only [flowDivergence, add_mul, Finset.sum_add_distrib,
  48     mul_assoc, ← Finset.mul_sum]

  50 theorem capacityBounded_convex
  51     {E : Type*} (c f g : E → ℝ) (hf : CapacityBounded c f)
  52     (hg : CapacityBounded c g) {a b : ℝ}
  53     (ha : 0 ≤ a) (hb : 0 ≤ b) (hab : a + b = 1) :
  54     CapacityBounded c (fun e => a * f e + b * g e) := by
  55   intro e
  56   constructor
  57   · exact add_nonneg (mul_nonneg ha (hf e).1) (mul_nonneg hb (hg e).1)
  58   · calc
  59       a * f e + b * g e ≤ a * c e + b * c e :=
  60         add_le_add (mul_le_mul_of_nonneg_left (hf e).2 ha)
  61           (mul_le_mul_of_nonneg_left (hg e).2 hb)
  62       _ = c e := by rw [← add_mul, hab, one_mul]

  64 theorem flowTransfer_refl
  65     {V E : Type*} [DecidableEq V] [Fintype E]
  66     (tail head : E → V) (c f : E → ℝ) (hf : CapacityBounded c f) (x : V) :
  67     FlowTransfer tail head c f x x := by
  68   refine ⟨1, by norm_num, f, hf, ?_⟩
  69   intro v
  70   simp [vertexDifference]

  72 /-- Two transfers from the same base flow can be combined by a convex
  73 combination. The intermediate divergence cancels exactly. -/
  74 theorem flowTransfer_trans
  75     {V E : Type*} [DecidableEq V] [Fintype E]
  76     (tail head : E → V) (c f : E → ℝ) {x z y : V}
  77     (hx : FlowTransfer tail head c f x z)
  78     (hy : FlowTransfer tail head c f z y) :
  79     FlowTransfer tail head c f x y := by
  80   obtain ⟨u, hu, g, hg, hdivg⟩ := hx
  81   obtain ⟨w, hw, h, hh, hdivh⟩ := hy
  82   have hsum : 0 < u + w := add_pos hu hw
  83   let a := w / (u + w)
  84   let b := u / (u + w)
  85   have ha : 0 ≤ a := le_of_lt (div_pos hw hsum)
  86   have hb : 0 ≤ b := le_of_lt (div_pos hu hsum)
  87   have hab : a + b = 1 := by
  88     dsimp [a, b]
  89     field_simp
  90     ring
  91   refine ⟨u * w / (u + w), div_pos (mul_pos hu hw) hsum,
  92     (fun e => a * g e + b * h e), capacityBounded_convex c g h hg hh ha hb hab, ?_⟩
  93   intro v
  94   rw [flowDivergence_linear, hdivg, hdivh]
  95   have hd := vertexDifference_trans x y z v
  96   dsimp [a, b]
  97   rw [← hd]
  98   field_simp
  99   ring

 109 theorem flowTransfer_forward
 110     {V E : Type*} [DecidableEq V] [Fintype E] [DecidableEq E]
 111     (tail head : E → V) (c f : E → ℝ) (hf : CapacityBounded c f)
 112     (e : E) (he : f e < c e) :
 113     FlowTransfer tail head c f (tail e) (head e) := by
 114   refine ⟨c e - f e, sub_pos.mpr he, (fun j => f j + if j = e then c e - f e else 0),
 115     ?_, fun v => flowDivergence_add_single tail head f e (c e - f e) v⟩
 116   intro j
 117   by_cases hj : j = e
 118   · subst j
 119     simp only [ite_true, add_sub_cancel]
 120     exact ⟨(hf e).1.trans (hf e).2, le_refl _⟩
 121   · simpa [hj] using hf j

 123 theorem flowTransfer_backward
 124     {V E : Type*} [DecidableEq V] [Fintype E] [DecidableEq E]
 125     (tail head : E → V) (c f : E → ℝ) (hf : CapacityBounded c f)
 126     (e : E) (he : 0 < f e) :
 127     FlowTransfer tail head c f (head e) (tail e) := by
 128   refine ⟨f e, he, (fun j => f j + if j = e then -f e else 0), ?_, ?_⟩
 129   · intro j
 130     by_cases hj : j = e
 131     · subst j
 132       simp only [ite_true, add_neg_cancel]
 133       exact ⟨le_refl _, (hf e).1.trans (hf e).2⟩
 134     · simpa [hj] using hf j
 135   · intro v
 136     rw [flowDivergence_add_single]
 137     unfold vertexDifference
 138     ring

 140 theorem flowDivergence_sum_cut
 141     {V E : Type*} [DecidableEq V] [Fintype E]
 142     (tail head : E → V) (f : E → ℝ) (S : Finset V) :
 143     ∑ v ∈ S, flowDivergence tail head f v =
 144       ∑ e, f e * ((if tail e ∈ S then (1 : ℝ) else 0) -
 145         (if head e ∈ S then 1 else 0)) := by
 146   unfold flowDivergence
 147   rw [Finset.sum_comm]
 148   apply Finset.sum_congr rfl
 149   intro e _
 150   rw [← Finset.mul_sum]
 151   simp [vertexDifference, Finset.sum_sub_distrib]

 153 def networkCutCapacity
 154     {V E : Type*} [DecidableEq V] [Fintype E]
 155     (tail head : E → V) (c : E → ℝ) (S : Finset V) : ℝ :=
 156   ∑ e, if tail e ∈ S ∧ head e ∉ S then c e else 0

 158 theorem networkFlow_cut_bound
 159     {V E : Type*} [DecidableEq V] [Fintype E]
 160     (tail head : E → V) (c f : E → ℝ) (s t : V)
 161     (hf : IsNetworkFlow tail head c s t f)
 162     (S : Finset V) (hs : s ∈ S) (ht : t ∉ S) :
 163     flowDivergence tail head f s ≤ networkCutCapacity tail head c S := by
 164   have hdiv : ∑ v ∈ S, flowDivergence tail head f v =
 165       flowDivergence tail head f s := by
 166     apply Finset.sum_eq_single_of_mem s hs
 167     intro v hv hvs
 168     exact hf.2 v hvs (by intro hvt; subst v; exact ht hv)
 169   rw [← hdiv, flowDivergence_sum_cut]
 170   apply Finset.sum_le_sum
 171   intro e _
 172   by_cases hx : tail e ∈ S <;> by_cases hy : head e ∈ S <;>
 173     simp [hx, hy]
 174   · exact (hf.1 e).2
 175   · exact (hf.1 e).1

 177 theorem exists_maximum_real_flow
 178     {V E : Type*} [DecidableEq V] [Fintype E]
 179     (tail head : E → V) (c : E → ℝ) (hc : ∀ e, 0 ≤ c e) (s t : V) :
 180     ∃ f : E → ℝ, IsNetworkFlow tail head c s t f ∧
 181       ∀ g, IsNetworkFlow tail head c s t g →
 182         flowDivergence tail head g s ≤ flowDivergence tail head f s := by
 183   let K : Set (E → ℝ) := {f | IsNetworkFlow tail head c s t f}
 184   have hcont (v : V) : Continuous (fun f : E → ℝ => flowDivergence tail head f v) := by
 185     unfold flowDivergence
 186     fun_prop
 187   have hclosed : IsClosed K := by
 188     change IsClosed {f | (∀ e, 0 ≤ f e ∧ f e ≤ c e) ∧
 189       ∀ v, v ≠ s → v ≠ t → flowDivergence tail head f v = 0}
 190     simp only [Set.ofPred_and, Set.ofPred_forall]
 191     apply IsClosed.inter
 192     · exact isClosed_iInter fun e =>
 193         (isClosed_le continuous_const (continuous_apply e)).inter
 194           (isClosed_le (continuous_apply e) continuous_const)
 195     · exact isClosed_iInter fun v => isClosed_iInter fun _ =>
 196         isClosed_iInter fun _ => isClosed_eq (hcont v) continuous_const
 197   have hcompact : IsCompact K :=
 198     (isCompact_Icc : IsCompact (Set.Icc (fun _ : E => (0 : ℝ)) c)).of_isClosed_subset
 199       hclosed (by
 200         intro f hf
 201         exact ⟨fun e => (hf.1 e).1, fun e => (hf.1 e).2⟩)
 202   have hzero : (fun _ : E => (0 : ℝ)) ∈ K := by
 203     refine ⟨fun e => ⟨le_refl _, hc e⟩, ?_⟩
 204     intro v _ _
 205     simp [flowDivergence]
 206   obtain ⟨f, hf, hmax⟩ := hcompact.exists_isMaxOn ⟨_, hzero⟩ (hcont s).continuousOn
 207   exact ⟨f, hf, fun g hg => hmax hg⟩

 209 /-- A maximum real flow has a cut of equal capacity. There is no rationality
 210 assumption on the capacities and no termination claim for augmenting paths. -/
 211 theorem real_maxFlow_minCut
 212     {V E : Type*} [Fintype V] [DecidableEq V] [Fintype E] [DecidableEq E]
 213     (tail head : E → V) (c : E → ℝ) (hc : ∀ e, 0 ≤ c e)
 214     (s t : V) (hst : s ≠ t) :
 215     ∃ f : E → ℝ, IsNetworkFlow tail head c s t f ∧
 216       ∃ S : Finset V, s ∈ S ∧ t ∉ S ∧
 217         flowDivergence tail head f s = networkCutCapacity tail head c S := by
 218   classical
 219   obtain ⟨f, hf, hmax⟩ := exists_maximum_real_flow tail head c hc s t
 220   let S := Finset.univ.filter (FlowTransfer tail head c f s)
 221   have hs : s ∈ S := by
 222     simpa [S] using flowTransfer_refl tail head c f hf.1 s
 223   have ht : t ∉ S := by
 224     intro ht
 225     obtain ⟨ε, hε, g, hg, hd⟩ : FlowTransfer tail head c f s t := by
 226       simpa [S] using ht
 227     have hgf : IsNetworkFlow tail head c s t g := by
 228       refine ⟨hg, fun v hvs hvt => ?_⟩
 229       rw [hd, hf.2 v hvs hvt]
 230       simp [vertexDifference, Ne.symm hvs, Ne.symm hvt]
 231     have hle := hmax g hgf
 232     have hds := hd s
 233     simp only [vertexDifference, ite_true, Ne.symm hst, ite_false, sub_zero, mul_one] at hds
 234     linarith
 235   have hout (e : E) (hx : tail e ∈ S) (hy : head e ∉ S) : f e = c e := by
 236     apply le_antisymm (hf.1 e).2
 237     by_contra hn
 238     have hlt : f e < c e := lt_of_not_ge hn
 239     have hx' : FlowTransfer tail head c f s (tail e) := by simpa [S] using hx
 240     have hy' := flowTransfer_trans tail head c f hx'
 241       (flowTransfer_forward tail head c f hf.1 e hlt)
 242     exact hy (by simpa [S] using hy')
 243   have hin (e : E) (hx : tail e ∉ S) (hy : head e ∈ S) : f e = 0 := by
 244     apply le_antisymm _ (hf.1 e).1
 245     by_contra hn
 246     have hlt : 0 < f e := lt_of_not_ge hn
 247     have hy' : FlowTransfer tail head c f s (head e) := by simpa [S] using hy
 248     have hx' := flowTransfer_trans tail head c f hy'
 249       (flowTransfer_backward tail head c f hf.1 e hlt)
 250     exact hx (by simpa [S] using hx')
 251   refine ⟨f, hf, S, hs, ht, ?_⟩
 252   have hdiv : ∑ v ∈ S, flowDivergence tail head f v =
 253       flowDivergence tail head f s := by
 254     apply Finset.sum_eq_single_of_mem s hs
 255     intro v hv hvs
 256     exact hf.2 v hvs (by intro hvt; subst v; exact ht hv)
 257   rw [← hdiv, flowDivergence_sum_cut]
 258   apply Finset.sum_congr rfl
 259   intro e _
 260   by_cases hx : tail e ∈ S <;> by_cases hy : head e ∈ S <;> simp [hx, hy]
 261   · exact hout e hx hy
 262   · exact hin e hx hy
