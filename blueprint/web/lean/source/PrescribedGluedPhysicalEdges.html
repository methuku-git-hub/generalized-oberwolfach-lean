   2 /-! When factor edges are partitioned between the earlier part and a
   3 separately typed remainder, gluing their vertex maps preserves their exact
   4 physical edge images. -/

  11 theorem prescribed_glued_physical_edge_images (embed:X↪Y) (A:Finset Y)
  12     (Eold:Finset (Sym2 Y)) (Erem:Finset (Sym2 X)) (ψ h:Y→V) (Φ:X→V)
  13     (hold:∀x∈A,h x=ψ x) (hrem:∀x,h (embed x)=Φ x)
  14     (hEold:∀e∈Eold,∀v∈e.toFinset,v∈A):
  15     (Eold∪Erem.image (Sym2.map embed)).image (Sym2.map h)=
  16       Eold.image (Sym2.map ψ)∪Erem.image (Sym2.map Φ) := by
  17   rw [Finset.image_union,Finset.image_image]
  18   congr 1
  19   · apply Finset.image_congr
  20     intro e he
  21     induction e using Sym2.ind with
  22     | _ x y =>
  23       have hx:x∈A:=hEold _ he x (by simp [Sym2.toFinset_mk_eq])
  24       have hy:y∈A:=hEold _ he y (by simp [Sym2.toFinset_mk_eq])
  25       change s(h x,h y)=s(ψ x,ψ y)
  26       rw [hold x hx,hold y hy]
  27   · apply Finset.image_congr
  28     intro e he
  29     induction e using Sym2.ind with
  30     | _ x y =>
  31       change s(h (embed x),h (embed y))=s(Φ x,Φ y)
  32       rw [hrem x,hrem y]
  33 end Oberwolfach
