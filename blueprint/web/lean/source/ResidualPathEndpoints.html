  11 def pathEndpointMatrix {X : Type*} [Fintype X] {k : ℕ}
  12     (w : (Fin (k+1) → X) → ℝ) (x z : X) : ℝ :=
  13   ∑ p, if p 0=x ∧ p (Fin.last k)=z then w p else 0

  15 theorem pathEndpointMatrix_mem {X : Type*} [Fintype X] {k : ℕ}
  16     (w : (Fin (k+1) → X) → ℝ) (d e : X → ℝ)
  17     (hw : ∀ p, 0 ≤ w p)
  18     (hfirst : ∀ x, pathVertexLoad w 0 x=d x)
  19     (hlast : ∀ z, pathVertexLoad w (Fin.last k) z=e z) :
  20     pathEndpointMatrix w ∈ transportMatrices d e := by
  21   refine ⟨?_,?_,?_⟩
  22   · intro x z
  23     apply Finset.sum_nonneg
  24     intro p _
  25     split_ifs <;> first | exact hw p | exact le_refl _
  26   · intro x
  27     exact ((pathVertexLoad_endpoints w (pathEndpointMatrix w) (fun _ _ => rfl)).1 x).symm.trans (hfirst x)
  28   · intro z
  29     exact ((pathVertexLoad_endpoints w (pathEndpointMatrix w) (fun _ _ => rfl)).2 z).symm.trans (hlast z)

  59 theorem halfResidualPathWeight_continuous {T X : Type*} [TopologicalSpace T] [Fintype X] {k : ℕ}
  60     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ)
  61     (ρ : T → (Fin (k+1) → X) → ℝ) (hρ : ∀ p, Continuous (fun z => ρ z p))
  62     (hdpos : ∀ z j x, 0 < residualIntervalDegree d (ρ z) j x) :
  63     Continuous (fun z => halfResidualPathWeight a d (ρ z)) := by
  64   have ha : ∀ j x y, Continuous (fun z => residualIntervalCapacity a (ρ z) j x y) := by
  65     intro j x y
  66     exact continuous_const.sub (pathEdgeLoad_continuous ρ hρ j x y)
  67   have hd : ∀ j x, Continuous (fun z => residualIntervalDegree d (ρ z) j x) := by
  68     intro j x
  69     exact continuous_const.sub (pathVertexLoad_continuous ρ hρ j x)
  70   have hc := capacityPathWeight_continuous (fun z => residualIntervalCapacity a (ρ z))
  71     (fun z => residualIntervalDegree d (ρ z)) ha hd hdpos
  72   apply continuous_pi
  73   intro p
  74   exact ((continuous_apply p).comp hc).div_const 2

  76 theorem halfResidualEndpoint_mem {X : Type*} [Fintype X] {k : ℕ} (hk : 0 < k)
  77     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ)
  78     (ρ : (Fin (k+1) → X) → ℝ) (Q : X → X → ℝ)
  79     (ha : ∀ j x y, 0 ≤ a j x y) (hd : ∀ j x, 0 < d j x)
  80     (hrow : ∀ j x, ∑ y, a j x y=d j.castSucc x)
  81     (hcol : ∀ j y, ∑ x, a j x y=d j.succ y)
  82     (hρ : ∀ p, 0 ≤ ρ p)
  83     (hreserve : ∀ j x y, pathEdgeLoad ρ j x y ≤ (11/27)*a j x y)
  84     (hends : pathEndpointMatrix ρ=Q)
  85     (hQ : Q ∈ transportMatrices (fun x => d 0 x/3) (fun z => d (Fin.last k) z/3)) :
  86     pathEndpointMatrix (halfResidualPathWeight a d ρ) ∈
  87       transportMatrices (fun x => d 0 x/3) (fun z => d (Fin.last k) z/3) := by
  88   obtain ⟨h0,_,hfirst,hlast⟩ := halfResidualPathWeight_properties hk a d ρ Q ha hd hrow hcol hρ hreserve
  89     (fun x z => congrFun (congrFun hends x) z) hQ
  90   exact pathEndpointMatrix_mem _ _ _ h0 hfirst hlast
