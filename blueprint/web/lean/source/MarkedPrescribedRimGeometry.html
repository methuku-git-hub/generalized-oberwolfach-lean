  16     (C:MarkedWheel tail head ktail khead (n+2) π b H₀ HK)
  17     (hC:∀ a,C.1.val a=Fin.cons (α:=fun _=>V) v (Fin.cons w p) (ρ a)):
  18     markedWheelRows C=insert v (insert w (Finset.univ.image p)):=by
  19   ext x
  20   simp only [markedWheelRows,Finset.mem_image,Finset.mem_univ,true_and,Finset.mem_insert]

  22   · rintro ⟨a,ha⟩
  23     rw [hC] at ha
  24     generalize ρ a=i at ha
  25     revert ha
  26     refine Fin.cases ?_ (fun i=>?_) i
  27     · intro ha;exact Or.inl ha.symm
  28     · refine Fin.cases ?_ (fun i=>?_) i
  29       · intro ha;exact Or.inr (Or.inl ha.symm)

  66     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))

  70     (j:Bool) (v w:V) (hv:v∈R) (hw:w∈R)
  71     (hroot:markedArcPresent tail head ktail khead j v w)
  72     (hne:∀ i,old i≠(if j then {v,π w} else {v,w})):
  73     ¬(markedFamilyRim C).Adj v (if j then π w else w):=by
