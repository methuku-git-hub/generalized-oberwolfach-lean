  14 theorem successor_sequence_weight_families (k m : ℕ) (hlen : m+1=(k+2)*2)
  15     (β τ : ℝ) (hβ : 0<β) (hτ : 0<τ) :
  16     ∃ δ : ℝ,0<δ ∧ ∀ b : ℕ,∃ N : ℕ,1≤N ∧
  17     ∀ (W : ℕ→Type) [∀ i,Fintype (W i)]
  18       (G : ∀ i,SimpleGraph (W i)) (π : ∀ i,Equiv.Perm (W i)) (D : ℕ→ℝ) (L : ℕ),
  19       (∀ i≤L,N≤Fintype.card (W i)) →
  20       (∀ i≤L,(Fintype.card (W i):ℝ)/2≤D i) →
  21       (∀ i≤L,D i≤2*Fintype.card (W i)) →
  22       (∀ i≤L,∀ v,|((G i).degree v:ℝ)-D i|≤δ*Fintype.card (W i)) →
  23       (∀ i≤L,∀ v,(successorCycleRatio k+β)*Fintype.card (W i)≤((G i).degree v:ℝ)) →
  24       ∃ weight : ∀ i,(Fin (m+1)→W i)→ℝ,
  25         ∀ i≤L,SuccessorSequenceWeightConditions (G i) (π i) b (D i) τ
  26           (successorWeightLower k β) (successorWeightUpper k β) (weight i) := by
  27   obtain ⟨δ,hδ,hall⟩ := successor_sequence_weight_existence k β hβ τ hτ
  28   refine ⟨δ,hδ,?_⟩
  29   intro b
  30   obtain ⟨N,hN⟩ := hall b
  31   refine ⟨max 1 N,le_max_left _ _,?_⟩
  32   intro W _ G π D L hn hDlo hDhi hnear hmin
  33   have hex (i : ℕ) : ∃ w : (Fin (m+1)→W i)→ℝ,i≤L →
  34       SuccessorSequenceWeightConditions (G i) (π i) b (D i) τ
  35         (successorWeightLower k β) (successorWeightUpper k β) w := by
  36     by_cases hi : i≤L
  37     · have hnpos : 0<Fintype.card (W i) := by have hh:=hn i hi;omega
  38       letI : Nonempty (W i) := Fintype.card_pos_iff.mp hnpos
  39       have hh := hN (W i) ((le_max_right 1 N).trans (hn i hi)) (G i) (π i) (D i)
  40         (hDlo i hi) (hDhi i hi) (hnear i hi) (hmin i hi)
  41       rw [← hlen] at hh
  42       obtain ⟨w,hw⟩ := hh
  43       exact ⟨w,fun _=>hw⟩
  44     · exact ⟨fun _=>0,fun hi'=>False.elim (hi hi')⟩
  45   choose weight hw using hex
  46   exact ⟨weight,hw⟩
