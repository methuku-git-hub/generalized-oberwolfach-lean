  15 theorem uniform_retention_conditional_event
  16     (d C Q : ℕ) (hd : 0<d) (p ζ : ℝ) (hp : 0<p) (hζ : 0<ζ) :
  17     ∃ m N : ℕ,1≤m ∧ 1≤N ∧ ∀ (V U Γ Color I : Type)
  18       [Fintype V] [Fintype U] [Fintype Γ] [Fintype Color] [Fintype I] [Nonempty V]

  21     ∀ (π : Equiv.Perm V) (σ : RawIntervalFamilyOutcome V U Γ Color) (g : Γ) (μ : ℝ),
  22       p*Fintype.card V/d≤μ →
  23       (∀ a,|((rawStartHubSet π d σ g a).card:ℝ)-μ|≤(Fintype.card V:ℝ)^(3/5:ℝ)) →
  24     ∀ (S : I→Finset V) (T : I→Finset U),
  25       (∀ i x,π x∈S i ↔ x∈S i) → (∀ i,m≤(S i).card) → (∀ i,m≤(T i).card) →
  26     1-1/(2*(Fintype.card V:ℝ)^Q)≤
  27       (startRetentionLaw (rawStartHubSet π d σ g) (startRetentionMinimum (rawStartHubSet π d σ g))
  28         (startRetentionMinimum_le _)).probability
  29         (RetainedRestrictionGood π d σ g (startRetentionMinimum (rawStartHubSet π d σ g)) S T ζ) := by
  30   let K : ℝ := 4*d/p
  31   let a : ℝ := ζ/(d+1:ℕ)
  32   have hK : 0≤K := by dsimp [K];positivity
  33   have ha : 0<a := by dsimp [a];positivity
  34   obtain ⟨m,N₁,hm,hN₁1,hN₁⟩ := uniform_binomial_restriction_tail K a (2/5) hK ha (by norm_num) (C+Q+3)
  35   obtain ⟨N₂,hN₂1,hN₂⟩ := eventually_retention_error_small p hp d hd
  36   refine ⟨m,max N₁ (max N₂ (2*(d+2))),hm,hN₁1.trans (le_max_left _ _),?_⟩
  37   intro V U Γ Color I _ _ _ _ _ _ _ _ hn hu hI π σ g μ hμ hcounts S T hS hSc hTc
  38   let n := Fintype.card V
  39   have hnpos : 0<n := by dsimp [n];omega
  40   let A := rawStartHubSet π d σ g
  41   let p' := K*(n:ℝ)^(-2/5:ℝ)
  42   have hp' : 0≤p' := by dsimp [p'];positivity
  43   have hrate (v : V) : (((A v).card-startRetentionMinimum A:ℕ):ℝ)/(A v).card≤p' :=
  44     startRetentionMinimum_polynomial_rate A n d hnpos hd p μ hp hμ (hN₂ n (by dsimp [n];omega)) hcounts v
  45   have htail (M : ℕ) (hM : m≤M) :

  47     have hh := hN₁ n (by dsimp [n];omega) M hM p' hp' (by dsimp [p'];simp only [neg_div];exact le_refl _)
  48     have he : a*M=ζ*M/(d+1:ℕ) := by dsimp [a];ring
  49     simpa only [he] using hh
  50   have hh := retainedRestriction_common_event π d σ g (startRetentionMinimum A) (startRetentionMinimum_le A)
  51     S T hS ζ p' (1/(n:ℝ)^(C+Q+3)) hζ.le hp' hrate (fun i=>htail _ (hSc i)) (fun i=>htail _ (hTc i))
  52   have herror := retention_restriction_union_margin d C Q n (by dsimp [n];omega) (Fintype.card I) (Fintype.card U) hI hu
  53   exact (sub_le_sub_left herror 1).trans hh
