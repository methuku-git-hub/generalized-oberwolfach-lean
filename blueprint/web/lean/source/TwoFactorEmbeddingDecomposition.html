  61 theorem even_regular_two_factor_embeddings_give_decomposition {n D:ℕ}
  62     (hEven:Even D) (G:SimpleGraph (Fin n)) (hG:G.IsRegularOfDegree D)
  63     (F:Fin (D/2)→SimpleGraph (Fin n)) (hF:∀i,(F i).IsRegularOfDegree 2)
  64     (f:Fin (D/2)→Fin n→Fin n) (hinj:∀i,Function.Injective (f i))
  65     (hsub:∀i,((F i).edgeFinset).image (Sym2.map (f i))⊆G.edgeFinset)
  66     (hdis:PairwisePhysicalDisjoint (fun i=>((F i).edgeFinset).image (Sym2.map (f i)))):
  67     HasFactorDecomposition G F :=by
