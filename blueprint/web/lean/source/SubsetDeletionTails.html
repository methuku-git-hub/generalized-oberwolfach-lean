  16 theorem finiteLaw_set_count_tail {Ω V : Type*} [Fintype Ω] [DecidableEq V]
  17     (μ : FiniteLaw Ω) (X : Ω→Finset V) (T : Finset V) (k : ℕ) (p : ℝ)
  18     (hjoint : ∀ A∈T.powersetCard k,μ.probability (fun ω=>A⊆X ω)≤p^k) :
  19     μ.probability (fun ω=>k≤(T∩X ω).card)≤(T.card.choose k:ℝ)*p^k := by
  20   have hsub : ∀ ω,k≤(T∩X ω).card → ∃ A∈T.powersetCard k,A⊆X ω := by
  21     intro ω h
  22     obtain ⟨A,hA,hcard⟩ := Finset.exists_subset_card_eq h
  23     exact ⟨A,Finset.mem_powersetCard.mpr ⟨hA.trans Finset.inter_subset_left,hcard⟩,
  24       hA.trans Finset.inter_subset_right⟩
  25   have hu := finiteLaw_union_bound μ (T.powersetCard k) (fun A ω=>A⊆X ω)
  26   calc
  27     _ ≤ μ.probability (fun ω=>∃ A∈T.powersetCard k,A⊆X ω) := finiteLaw_probability_mono μ _ _ hsub
  28     _ ≤ ∑ A∈T.powersetCard k,μ.probability (fun ω=>A⊆X ω) := hu
  29     _ ≤ ∑ _A∈T.powersetCard k,p^k := Finset.sum_le_sum hjoint
  30     _ = _ := by simp

  33     [Nonempty (FixedSubset V r)] (hr : r≤Fintype.card V) (T : Finset V) :

  35       ((Fintype.card V-r:ℕ):ℝ)^T.card/(Fintype.card V:ℝ)^T.card := by
  36   letI := fixedSubset_nonempty (V:=V) (Nat.sub_le (Fintype.card V) r)
  37   have he := finiteLaw_uniform_map_equiv (fixedSubsetComplEquiv (V:=V) hr)
  38   have hp := fixedSubset_joint_inclusion_le (V:=V) (Nat.sub_le (Fintype.card V) r) T
  39   rw [← he,finiteLaw_map_probability] at hp
  40   simpa only [fixedSubsetComplEquiv,Equiv.coe_fn_mk,div_pow] using hp
