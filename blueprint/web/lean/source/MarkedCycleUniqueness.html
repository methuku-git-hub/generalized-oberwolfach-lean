  12 theorem markedCycleIncidence_ordinary_one_iff {V E K : Type*} [DecidableEq V]
  13     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (C : Fin (m+1) → V)
  14     (hC : Function.Injective C) (e : E) :
  15     markedCycleIncidence tail head ktail khead m C (Sum.inl e)=1 ↔
  16       ∃ j : Fin m,tail e=C j.castSucc ∧ head e=C j.succ := by
  17   classical
  18   have hnonneg (j : Fin m) : 0≤directedEdgeIndicator tail head (C j.castSucc) (C j.succ) e := by
  19     unfold directedEdgeIndicator
  20     split_ifs <;> norm_num
  21   constructor
  22   · intro h
  23     have hp : 0<∑ j : Fin m,directedEdgeIndicator tail head (C j.castSucc) (C j.succ) e := by
  24       change 0<markedCycleIncidence tail head ktail khead m C (Sum.inl e)
  25       rw [h];norm_num
  26     obtain ⟨j,_,hj⟩ := (Finset.sum_pos_iff_of_nonneg (fun j _ => hnonneg j)).mp hp
  27     refine ⟨j,?_⟩
  28     by_contra hn
  29     simp only [directedEdgeIndicator,hn,ite_false] at hj
  30     exact (lt_irrefl _ hj)
  31   · rintro ⟨j,hj⟩
  32     have hh := Finset.single_le_sum (s := Finset.univ)
  33       (f := fun l : Fin m => directedEdgeIndicator tail head (C l.castSucc) (C l.succ) e)
  34       (fun l _ => hnonneg l) (Finset.mem_univ j)
  35     have hid : directedEdgeIndicator tail head (C j.castSucc) (C j.succ) e=1 := by
  36       simp only [directedEdgeIndicator,hj.1,hj.2,and_self,ite_true]
  37     rw [hid] at hh
  38     change 1≤markedCycleIncidence tail head ktail khead m C (Sum.inl e) at hh
  39     rcases markedCycleIncidence_zero_one tail head ktail khead m C hC (Sum.inl e) with hz | ho
  40     · rw [hz] at hh;norm_num at hh
  41     · exact ho

  51 theorem markedCycleIncidence_injective {V E K : Type*} [DecidableEq V]
  52     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (π : Equiv.Perm V) (b : ℕ) :
  53     Function.Injective (fun C : MarkedCycleTuple tail head ktail khead m π b =>
  54       markedCycleIncidence tail head ktail khead m C.val) := by
  55   intro C C' heq
  56   change markedCycleIncidence tail head ktail khead m C.val=markedCycleIncidence tail head ktail khead m C'.val at heq
  57   have hC := permutationSeparated_injective π b C.val C.property.1
  58   have hC' := permutationSeparated_injective π b C'.val C'.property.1
  59   obtain ⟨e,he⟩ := C.property.2.2
  60   have hK := (markedCycleIncidence_marked_one_iff tail head ktail khead m C.val e).mpr he
  61   have hK' : markedCycleIncidence tail head ktail khead m C'.val (Sum.inr e)=1 := by
  62     rw [← heq];exact hK
  63   have hz : C.val 0=C'.val 0 := he.2.symm.trans
  64     ((markedCycleIncidence_marked_one_iff tail head ktail khead m C'.val e).mp hK').2
  65   have hall : ∀ j : ℕ,∀ hj : j≤m,C.val ⟨j,by omega⟩=C'.val ⟨j,by omega⟩ := by
  66     intro j
  67     induction j with
  68     | zero => intro hj;exact hz
  69     | succ j ih =>
  70       intro hj
  71       let s : Fin m := ⟨j,by omega⟩
  72       have hprev : C.val s.castSucc=C'.val s.castSucc := ih (by omega)
  73       obtain ⟨f,hf⟩ := C.property.2.1 s
  74       have hI := (markedCycleIncidence_ordinary_one_iff tail head ktail khead m C.val hC f).mpr ⟨s,hf⟩
  75       have hI' : markedCycleIncidence tail head ktail khead m C'.val (Sum.inl f)=1 := by
  76         rw [← heq];exact hI
  77       obtain ⟨l,hl⟩ := (markedCycleIncidence_ordinary_one_iff tail head ktail khead m C'.val hC' f).mp hI'
  78       have hls : l=s := Fin.castSucc_injective m (hC' (hl.1.symm.trans (hf.1.trans hprev)))
  79       subst l
  80       exact hf.2.symm.trans hl.2
  81   apply Subtype.ext
  82   funext i
  83   exact hall i.val (by omega)
