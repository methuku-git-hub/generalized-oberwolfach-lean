  11 theorem two_factor_residual_graph_after_disjoint_deletions
  12     (F:SimpleGraph V) (hF:F.IsRegularOfDegree 2)
  13     (s:Finset I) (u v:I→V) (P:∀ i,F.Walk (u i) (v i))
  14     (hp:∀i∈s,(P i).IsPath) (hne:∀i∈s,u i≠v i)
  15     (hP:(↑s:Set I).PairwiseDisjoint (fun i=>(P i).support.toFinset))
  16     (t:Finset J) (root:J→V) (C:∀ j,F.Walk (root j) (root j))
  17     (hc:∀j∈t,(C j).IsCycle)
  18     (hC:(↑t:Set J).PairwiseDisjoint (fun j=>(C j).support.toFinset))
  19     (hcross:∀i∈s,∀j∈t,Disjoint (P i).support.toFinset (C j).support.toFinset):
  20     let deletedV := (t.biUnion (fun j=>(C j).support.toFinset)) ∪
  21       (s.biUnion (fun i=>(P i).support.toFinset \ {u i,v i}))
  22     let deletedE := (t.biUnion (fun j=>(C j).edges.toFinset)) ∪
  23       (s.biUnion (fun i=>(P i).edges.toFinset))
  24     let Q:=F.deleteEdges (deletedE:Set (Sym2 V));
  25     Q.edgeFinset.card+s.card=(Finset.univ\deletedV).card ∧
  26       ∀ x∈deletedV,∀ y,¬Q.Adj x y := by
  27   dsimp only
  28   constructor
  29   · rw [SimpleGraph.edgeFinset_deleteEdges]
  30     exact two_factor_residual_count_after_disjoint_deletions F hF s u v P hp hne hP t root C hc hC hcross
  31   · intro x hx y hxy
  32     have hh:=SimpleGraph.deleteEdges_adj.mp hxy
  33     exact hh.2 (two_factor_deleted_vertex_incident F hF s u v P hp t root C hc x y hh.1 hx)
  34 end Oberwolfach
