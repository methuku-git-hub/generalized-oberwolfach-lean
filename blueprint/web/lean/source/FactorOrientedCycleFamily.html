  11 theorem factor_component_family_has_forward_cycles
  12     {X J:Type} [Fintype X] [DecidableEq X] [DecidableEq J]
  13     (G:SimpleGraph X) (hG:G.IsRegularOfDegree 2)
  14     (E:∀C:G.ConnectedComponent,SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph)
  15     (components:J↪G.ConnectedComponent):
  16     ∃ (root:J→X) (p:∀j,G.Walk (root j) (root j)),
  17       (∀j,(p j).IsCycle) ∧ (∀j,(p j).length=Fintype.card (components j)) ∧
  18       (∀j k,k<(p j).length→factorCycleOrientation G E ((p j).getVert k)=(p j).getVert (k+1)) ∧
  19       (∀j,(p j).toSubgraph.verts=(components j).supp) ∧
  20       (∀i j,i≠j→Disjoint (p i).support.toFinset (p j).support.toFinset):=by
  21   choose root p hp hlen hstep hsupp using fun j=>factor_component_has_forward_cycle G hG E (components j)
  22   refine ⟨root,p,hp,hlen,hstep,hsupp,?_⟩
  23   intro i j hij
  24   apply Finset.disjoint_left.mpr
  25   intro x hxi hxj
  26   have hxi':x∈(components i).supp:=by
  27     rw [←hsupp i]
  28     exact (p i).mem_verts_toSubgraph.mpr (List.mem_toFinset.mp hxi)
  29   have hxj':x∈(components j).supp:=by
  30     rw [←hsupp j]
  31     exact (p j).mem_verts_toSubgraph.mpr (List.mem_toFinset.mp hxj)
  32   have hcomp:components i≠components j:=fun h=>hij (components.injective h)
  33   exact Set.disjoint_left.mp (G.pairwise_disjoint_supp_connectedComponent hcomp) hxi' hxj'
  34 end Oberwolfach
