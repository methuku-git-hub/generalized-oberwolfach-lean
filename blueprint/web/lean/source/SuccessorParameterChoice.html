  10 theorem successor_small_parameters (c C η₀ β τ : ℝ)
  11     (hc : 0 < c) (hC : 0 < C) (hη₀ : 0 < η₀) (hβ : 0 < β) (hτ : 0 < τ) :
  12     ∃ η ε₀ : ℝ, 0 < η ∧ η < η₀ ∧ 0 < ε₀ ∧ ε₀ ≤ η/2 ∧
  13       Real.sqrt η ≤ 1/4 ∧ ε₀ + η ≤ β/4 ∧ ε₀ + Real.sqrt η ≤ β/4 ∧
  14       Real.sqrt η ≤ τ/32 ∧ C*Real.sqrt η ≤ τ/32 ∧ c*(C*Real.sqrt η) ≤ 1/4 := by
  15   let t := min (1/4 : ℝ) (min η₀ (min (β/64)
  16     (min (τ/64) (min (τ/(64*C)) (1/(8*c*C))))))
  17   have ht : 0 < t := by dsimp [t]; positivity
  18   obtain ⟨s,hs,hst⟩ := exists_between ht
  19   have hs1 : s < 1/4 := hst.trans_le (min_le_left _ _)
  20   have hsη : s < η₀ := hst.trans_le ((min_le_right _ _).trans (min_le_left _ _))
  21   have hsβ : s < β/64 := hst.trans_le ((min_le_right _ _).trans
  22     ((min_le_right _ _).trans (min_le_left _ _)))
  23   have hsτ : s < τ/64 := hst.trans_le ((min_le_right _ _).trans
  24     ((min_le_right _ _).trans ((min_le_right _ _).trans (min_le_left _ _))))
  25   have hsC : s < τ/(64*C) := hst.trans_le ((min_le_right _ _).trans
  26     ((min_le_right _ _).trans ((min_le_right _ _).trans
  27       ((min_le_right _ _).trans (min_le_left _ _)))))
  28   have hsc : s < 1/(8*c*C) := hst.trans_le ((min_le_right _ _).trans
  29     ((min_le_right _ _).trans ((min_le_right _ _).trans
  30       ((min_le_right _ _).trans (min_le_right _ _)))))
  31   have hs2 : s^2 ≤ s := by nlinarith
  32   have hsC' := (lt_div_iff₀ (show (0 : ℝ) < 64*C by positivity)).mp hsC
  33   have hsc' := (lt_div_iff₀ (show (0 : ℝ) < 8*c*C by positivity)).mp hsc
  34   refine ⟨s^2,s^2/4,by positivity,lt_of_le_of_lt hs2 hsη,by positivity,?_,?_,?_,?_,?_,?_,?_⟩
  35   all_goals try rw [Real.sqrt_sq hs.le]
