  16 theorem endpoint_parent_matching_exists (P:SimpleGraph V) (A:Finset V) (x:V)
  17     (hn:32≤A.card)
  18     (hdeg:∀ v∈insert x A,(3/4:ℝ)*A.card≤((A∩P.neighborFinset v).card:ℝ)) :
  19     ∃ weight:Sym2 V→ℝ,
  20       (∀ v∈A∩P.neighborFinset x,∀ w∈A∩P.neighborFinset x,
  21         0≤weight s(v,w) ∧ weight s(v,w)≤40/A.card) ∧
  22       (∀ v∈A∩P.neighborFinset x,∑ w∈A∩P.neighborFinset x,weight s(v,w)=1) ∧
  23       (∀ v w,¬P.Adj v w → weight s(v,w)=0) ∧
  24       (∀ F:SimpleGraph V,∀ e∈(endpointEligibleGraph P F (A∩P.neighborFinset x) x).edgeSet,
  25         (1/30:ℝ)/A.card≤weight e ∧ weight e≤40/A.card) := by
  26   let S:=A∩P.neighborFinset x
  27   have hS:S⊆A:=Finset.inter_subset_left
  28   have hsize:(3/4:ℝ)*A.card≤S.card:=hdeg x (Finset.mem_insert_self _ _)
  29   have hA:(0:ℝ)<A.card:=by exact_mod_cast (show 0<A.card by omega)
  30   have hSpos:(0:ℝ)<S.card:=(mul_pos (by norm_num) hA).trans_le hsize
  31   have hSA:(S.card:ℝ)≤A.card:=Nat.cast_le.mpr (Finset.card_le_card hS)
  32   have hlarge:4/(1/6:ℝ)≤S.card:=by
  33     have hh:(32:ℝ)≤A.card:=by exact_mod_cast hn
  34     linarith
  35   have hlocal (v:V) (hv:v∈S):(1/2+(1/6:ℝ))*S.card≤((S∩P.neighborFinset v).card:ℝ):=by
  36     have hc:=Finset.card_inter_add_card_union (A∩P.neighborFinset v) S
  37     have hu:(A∩P.neighborFinset v)∪S⊆A:=Finset.union_subset Finset.inter_subset_left hS
  38     have huc:=Finset.card_le_card hu
  39     have he:(A∩P.neighborFinset v)∩S=S∩P.neighborFinset v:=by
  40       ext z;simp only [Finset.mem_inter];constructor
  41       · exact fun h=>⟨h.2,h.1.2⟩
  42       · exact fun h=>⟨⟨hS h.1,h.2⟩,h.1⟩
  43     rw [he] at hc
  44     have hc':((S∩P.neighborFinset v).card:ℝ)+((A∩P.neighborFinset v)∪S).card=
  45         (A∩P.neighborFinset v).card+S.card:=by exact_mod_cast hc
  46     have huc':(((A∩P.neighborFinset v)∪S).card:ℝ)≤A.card:=by exact_mod_cast huc
  47     have hd:=hdeg v (Finset.mem_insert_of_mem (hS hv))
  48     linarith
  49   obtain ⟨weight,hw,hsum,hzero,hlo⟩:=bounded_fractional_matching_on_subset P S (1/6)
  50     (by norm_num) (by norm_num) hlarge hlocal
  51   have hupper:5/((1/6:ℝ)*S.card)≤40/A.card:=by
  52     apply (div_le_div_iff₀ (mul_pos (by norm_num) hSpos) hA).mpr
  53     nlinarith only [hsize]
  54   have hlower:(1/30:ℝ)/A.card≤(1/6:ℝ)/(5*S.card):=by
  55     apply (div_le_div_iff₀ hA (mul_pos (by norm_num) hSpos)).mpr
  56     nlinarith only [hSA]
  57   refine ⟨weight,(fun v hv w hw'=>⟨(hw v hv w hw').1,(hw v hv w hw').2.trans hupper⟩),hsum,hzero,?_⟩
  58   intro F e he
  59   induction e using Sym2.ind with
  60   | _ v w =>
  61     exact ⟨hlower.trans (hlo v he.1 w he.2.1 he.2.2.2.2.1),
  62       (hw v he.1 w he.2.1).2.trans hupper⟩
  63
