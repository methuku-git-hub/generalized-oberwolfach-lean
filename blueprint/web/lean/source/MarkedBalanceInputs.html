  11 theorem markedBalanceSubspace_inputs {V E K : Type*}
  12     [Fintype V] [Fintype E] [Fintype K] [DecidableEq V]
  13     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (w : E⊕K → ℝ)
  14     (hw : w∈markedBalanceSubspace tail head ktail khead m) :
  15     (∀ v,occurrenceSumLinear tail (fun e => w (Sum.inl e)) v+
  16       occurrenceSumLinear ktail (fun e => w (Sum.inr e)) v=
  17       occurrenceSumLinear head (fun e => w (Sum.inl e)) v+
  18       occurrenceSumLinear khead (fun e => w (Sum.inr e)) v) ∧
  19     (∑ e : E,w (Sum.inl e))=(m:ℝ)*(∑ e : K,w (Sum.inr e)) := by
  20   constructor
  21   · intro v
  22     have hh := congrFun hw.1 v
  23     change (∑ e : E⊕K,if Sum.elim tail ktail e=v then w e else 0)=
  24       (∑ e : E⊕K,if Sum.elim head khead e=v then w e else 0) at hh
  25     change (∑ e : E,if tail e=v then w (Sum.inl e) else 0)+(∑ e : K,if ktail e=v then w (Sum.inr e) else 0)=
  26       (∑ e : E,if head e=v then w (Sum.inl e) else 0)+(∑ e : K,if khead e=v then w (Sum.inr e) else 0)
  27     simp only [Fintype.sum_sum_type,Sum.elim_inl,Sum.elim_inr] at hh
  28     convert hh using 1 <;> congr 1 <;> apply Finset.sum_congr rfl <;>
  29       intro e _ <;> split_ifs <;> rfl
  30   · exact hw.2

  37 theorem graph_large_cut_monotone {V E : Type*} [Fintype V] [Fintype E] [DecidableEq V]
  38     (tail head : E → V) (δ δ' h h' : ℝ) (hδ : δ'≤δ) (hh : h'≤h)
  39     (hcut : ∀ S T : Finset V,(1-2*δ)*Fintype.card V<(S.card:ℝ)+T.card →
  40       h*Fintype.card V*min (S.card:ℝ) T.card≤
  41       ((Finset.univ.filter (fun e => tail e∈S ∧ head e∈T)).card:ℝ)) :
  42     ∀ S T : Finset V,(1-2*δ')*Fintype.card V<(S.card:ℝ)+T.card →
  43       h'*Fintype.card V*min (S.card:ℝ) T.card≤
  44       ((Finset.univ.filter (fun e => tail e∈S ∧ head e∈T)).card:ℝ) := by
  45   intro S T hST
  46   have hnp : (0:ℝ)≤Fintype.card V := Nat.cast_nonneg _
  47   have he : (1-2*δ)*Fintype.card V≤(1-2*δ')*Fintype.card V :=
  48     mul_le_mul_of_nonneg_right (by linarith) hnp
  49   have hmin : 0≤min (S.card:ℝ) T.card := by positivity
  50   exact (mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_right hh hnp) hmin).trans
  51     (hcut S T (he.trans_lt hST))
