  48 theorem ordinary_wheel_comparison_disjoint (M:Finset (OrdinaryWheel Q F H r)) (A:Finset V) (x:V)
  49     (hM:∀ W∈M,x∉(ordinaryWheelRim W).support)
  50     (C:OrdinaryWheel (Q\spanningRestriction (ordinaryWheelRimGraph M) A) F
  51       (fun v u=>H v u∧¬(v∈A∧ordinaryPackingSpokes M v u)) r)
  52     (hC:∀ a∈(ordinaryWheelRim C).support,a=x∨a∈A) :
  53     Disjoint (ordinaryWheelResources C) (M.biUnion ordinaryWheelResources) := by
  54   apply Finset.disjoint_left.mpr
  55   intro e he heM
  56   obtain ⟨W,hW,hew⟩:=Finset.mem_biUnion.mp heM
  57   cases e with
  58   | inl e =>
  59     rw [ordinaryWheelResources_rim] at he hew
  60     induction e using Sym2.inductionOn with
  61     | _ a b =>
  62       have ha:a∈A:=by
  63         rcases hC a he.mem_support_left with rfl|ha
  64         · exact (hM W hW hew.mem_support_left).elim
  65         · exact ha
  66       have hb:b∈A:=by
  67         rcases hC b he.mem_support_right with rfl|hb
  68         · exact (hM W hW hew.mem_support_right).elim
  69         · exact hb
  70       have hprev:(ordinaryWheelRimGraph M).Adj a b:=SimpleGraph.iSup_adj.mpr ⟨W,SimpleGraph.iSup_adj.mpr ⟨hW,hew⟩⟩
  71       have hd:(spanningRestriction (ordinaryWheelRimGraph M) A).Adj a b:=(spanningRestriction_adj _ A a b).mpr ⟨hprev,ha,hb⟩
  72       exact (C.val.1.property he).2 hd
  73   | inr vu =>
  74     rw [ordinaryWheelResources_spoke] at he hew
  75     have ha:vu.1∈A:=by
  76       rcases hC vu.1 he.2 with ha|ha
  77       · rw [ha] at hew;exact (hM W hW hew.2).elim
  78       · exact ha
  79     have hprev:ordinaryPackingSpokes M vu.1 (ordinaryWheelHub C):=

  81     exact (C.property.1 vu.1 he.2).2 ⟨ha,hprev⟩
