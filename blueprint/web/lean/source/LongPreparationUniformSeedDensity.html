   8 theorem long_preparation_uniform_seed_density (m:ℕ) (ρ₀ ρ:ℝ)
   9     (hρ₀:0<ρ₀) (hρ:ρ₀≤ρ):
  10     let c:ℝ:=(m:ℝ)+1
  11     let pstar:=min (1/4:ℝ) (min (ρ₀/8) (ρ₀/(6*c)));
  12     0<pstar ∧ pstar≤1/4 ∧ pstar≤ρ/8 ∧ pstar≤ρ/(6*c):=by
  13   intro c pstar
  14   have hc:0<c:=by dsimp [c];positivity
  15   have hp:0<pstar:=by dsimp [pstar];positivity
  16   have h₁:pstar≤1/4:=min_le_left _ _
  17   have h₂:pstar≤ρ₀/8:=le_trans (min_le_right _ _) (min_le_left _ _)
  18   have h₃:pstar≤ρ₀/(6*c):=le_trans (min_le_right _ _) (min_le_right _ _)
  19   refine ⟨hp,h₁,h₂.trans ?_,h₃.trans ?_⟩
  20   · exact div_le_div_of_nonneg_right hρ (by norm_num)
  21   · exact div_le_div_of_nonneg_right hρ (by positivity)
  22 end Oberwolfach
