  10 def physicalHubRims {C U V:Type} [Fintype V] [DecidableEq V]

  14 theorem physical_hub_rims_packed
  15     {C U V:Type} [Fintype V] [DecidableEq V]
  16     (S:Finset C) (hub:C→U) (R:C→Finset (Finset V⊕(V×U)))
  17     (hpack:∀c∈S,∀c'∈S,c≠c'→Disjoint (R c) (R c')):
  18     (∀w z,w≠z→Disjoint (physicalHubRims S hub R w) (physicalHubRims S hub R z)) ∧
  19     (∀c∈S,∀e,Sum.inl e∈R c→e∈physicalHubRims S hub R (hub c)):=by
