  11 theorem sampled_orientation_degree_stability
  12     {V:Type} [Fintype V] [LinearOrder V] (S T:SimpleGraph V) (bits:Sym2 V→Bool) (v:V):
  13     |((∑w,sampledOrientation S bits v w:ℕ):ℝ)-
  14       ((∑w,sampledOrientation T bits v w:ℕ):ℝ)|≤
  15       ((changed (S.neighborFinset v) (T.neighborFinset v)).card:ℝ):=by
  16   let A:=Finset.univ.filter (fun w=>0<sampledOrientation S bits v w)
  17   let B:=Finset.univ.filter (fun w=>0<sampledOrientation T bits v w)
  18   have hcardS:(A.card:ℕ)=∑w,sampledOrientation S bits v w:=by
  19     rw [Finset.card_filter]
  20     apply Finset.sum_congr rfl
  21     intro w _
  22     simp only [sampledOrientation]
  23     split_ifs <;> simp_all
  24   have hcardT:(B.card:ℕ)=∑w,sampledOrientation T bits v w:=by
  25     rw [Finset.card_filter]
  26     apply Finset.sum_congr rfl
  27     intro w _
  28     simp only [sampledOrientation]
  29     split_ifs <;> simp_all
  30   have hsub:changed A B⊆changed (S.neighborFinset v) (T.neighborFinset v):=by
  31     intro w hw
  32     simp only [changed,Finset.mem_union,Finset.mem_sdiff,SimpleGraph.mem_neighborFinset] at hw ⊢
  33     simp only [A,B,Finset.mem_filter,Finset.mem_univ,true_and] at hw
  34     by_cases hs:S.Adj v w <;> by_cases ht:T.Adj v w <;>
  35       simp [sampledOrientation,hs,ht] at hw ⊢
  36     all_goals (try split_ifs at hw) <;> simp_all
  37   have hc:((changed A B).card:ℝ)≤(changed (S.neighborFinset v) (T.neighborFinset v)).card:=by
  38     exact_mod_cast Finset.card_le_card hsub
  39   have hh:=card_difference_le_changed A B
  40   rw [hcardS,hcardT] at hh
  41   exact hh.trans hc
  42 end Oberwolfach
