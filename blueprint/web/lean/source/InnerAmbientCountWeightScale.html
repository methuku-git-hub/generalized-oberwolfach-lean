   8 theorem inner_ambient_weight_scale (m:ℕ) (n a κ b B w:ℝ)
   9     (hn:0<n) (ha:0<a) (hk:0<κ) (hb:0≤b) (hB:0≤B)
  10     (han:a≤n) (hka:κ*n≤a) (hw:b/a^m≤w∧w≤B/a^m):
  11     b/n^m≤w∧w≤(B/κ^m)/n^m:=by
  12   constructor
  13   · exact (div_le_div_of_nonneg_left hb (pow_pos ha _) (pow_le_pow_left₀ ha.le han _)).trans hw.1
  14   · apply hw.2.trans
  15     rw [div_div,←mul_pow]
  16     exact div_le_div_of_nonneg_left hB (pow_pos (mul_pos hk hn) _) (pow_le_pow_left₀ (mul_nonneg hk.le hn.le) hka _)

  18 theorem inner_ambient_count_scales (m:ℕ) (n a κ β:ℝ)
  19     (hn:0≤n) (ha:0≤a) (hk:0≤κ) (hk1:κ≤1) (hβ:0≤β) (hka:κ*n≤a):
  20     (β*κ^(m+1))*n^(m+1)≤β*a^(m+1)∧
  21     (β*κ^(m+1))*n^m≤β*a^m:=by
  22   have hm:κ^m*n^m≤a^m:=by
  23     rw [←mul_pow]
  24     exact pow_le_pow_left₀ (mul_nonneg hk hn) hka m
  25   have hs:κ^(m+1)*n^(m+1)≤a^(m+1):=by
  26     rw [←mul_pow]
  27     exact pow_le_pow_left₀ (mul_nonneg hk hn) hka (m+1)
  28   have hstep:κ^(m+1)≤κ^m:=by
  29     rw [pow_succ]
  30     exact (mul_le_mul_of_nonneg_left hk1 (pow_nonneg hk m)).trans_eq (mul_one _)
  31   constructor
  32   · nlinarith only [mul_le_mul_of_nonneg_left hs hβ]
  33   · have h':=mul_le_mul_of_nonneg_right hstep (pow_nonneg hn m)
  34     nlinarith only [mul_le_mul_of_nonneg_left (h'.trans hm) hβ]
