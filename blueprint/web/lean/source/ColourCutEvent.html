  15 theorem sampledColourCut_quadratic_event_pow (k : ℕ) {q : ℝ} (hq : 0 < q) (p : ℕ) :
  16     ∃ N₀ : ℕ, ∀ (V : Type u) [Fintype V] [LinearOrder V],
  17       let n := Fintype.card V
  18       N₀ ≤ n → ∀ (G : SimpleGraph V) (ρ : Fin k → ℝ)
  19         (hρ : ∀ a, 0 ≤ ρ a) (htotal : ∑ a, ρ a = 1),
  20       1 - 1 / (n : ℝ) ^ p ≤
  21         (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.categoricalDirection ρ hρ htotal)).probability
  22           (fun σ => ∀ a S T,
  23             |((∑ x ∈ S, ∑ y ∈ T, sampledColourOrientation G σ a x y : ℕ) : ℝ) -
  24               ρ a / 2 * graphCut G S T| < q * (n : ℝ) ^ 2) := by
  25   obtain ⟨N, hN⟩ := eventually_allCuts_failure_small
  26     (C := 2 * k) (c := q ^ 2 / 2) (by positivity) (by positivity) p
  27   refine ⟨max N 1, fun V _ _ => ?_⟩
  28   intro n hn G ρ hρ htotal
  29   have hnpos : 0 < n := by omega
  30   have hnreal : (0 : ℝ) < n := by exact_mod_cast hnpos
  31   have hh := sampledColourCut_all_cuts G hnpos ρ hρ htotal
  32     (fun _ => q * (n : ℝ) ^ 2) (fun _ => by positivity)
  33   have heq : -(q * (n : ℝ) ^ 2) ^ 2 / (2 * (n : ℝ) ^ 2) = -(q ^ 2 / 2) * (n : ℝ) ^ 2 := by
  34     field_simp
  35   change 1 - (4 : ℝ) ^ n *
  36       (∑ _a : Fin k, 2 * Real.exp (-(q * (n : ℝ) ^ 2) ^ 2 / (2 * (n : ℝ) ^ 2))) ≤ _ at hh
  37   simp only [heq, Finset.sum_const, Finset.card_univ, Fintype.card_fin, nsmul_eq_mul] at hh
  38   have hs := hN n (le_trans (le_max_left _ _) hn)
  39   have heq' : (4 : ℝ) ^ n * (k * (2 * Real.exp (-(q ^ 2 / 2) * (n : ℝ) ^ 2))) =
  40       2 * k * (4 : ℝ) ^ n * Real.exp (-(q ^ 2 / 2) * (n : ℝ) ^ 2) := by ring
  41   rw [heq'] at hh
  42   exact (sub_le_sub_left hs 1).trans hh
