  11 theorem interval_random_demands {V E : Type*}
  12     [Fintype V] [Fintype E] [DecidableEq V] [DecidableEq E]
  13     (tail head : E → V) (f g : V → ℕ) (a b δ h p : ℝ)
  14     (hn : 0 < Fintype.card V) (ha : 0 < a) (hp : 0 < p) (hp1 : p ≤ 1)
  15     (hδ : 0 ≤ δ) (hδsmall : δ < 1 / 4)
  16     (hrelative : b ≤ (1 + δ) * a)
  17     (hcapacity : b ≤ p * h * Fintype.card V / 2)
  18     (hf : ∀ x, a ≤ (f x : ℝ) ∧ (f x : ℝ) ≤ b)
  19     (hg : ∀ y, a ≤ (g y : ℝ) ∧ (g y : ℝ) ≤ b)
  20     (htotal : ∑ x, f x = ∑ y, g y)
  21     (hlarge : ∀ (S T : Finset V),
  22       1 - 2 * δ < (S.card : ℝ) / Fintype.card V + (T.card : ℝ) / Fintype.card V →
  23       h * Fintype.card V * Fintype.card V *
  24         min ((S.card : ℝ) / Fintype.card V) ((T.card : ℝ) / Fintype.card V) ≤
  25       ((Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∈ T)).card : ℝ)) :
  26     ∃ ν : FiniteLaw (E → Bool),
  27       (∀ D, 0 < ν.mass D →
  28         (∀ x, (∑ e, if tail e = x ∧ D e = true then (1 : ℕ) else 0) = f x) ∧
  29         (∀ y, (∑ e, if head e = y ∧ D e = true then (1 : ℕ) else 0) = g y)) ∧
  30       (∀ e, ν.probability (fun D => D e = true) ≤ p) := by
  31   let M : ℝ := ∑ x, (f x : ℝ)
  32   have hn' : (0 : ℝ) < Fintype.card V := by exact_mod_cast hn
  33   have hM : 0 ≤ M := Finset.sum_nonneg (fun x _ => Nat.cast_nonneg _)
  34   have hMlo : Fintype.card V * a ≤ M := by
  35     simpa only [Finset.sum_const, Finset.card_univ, nsmul_eq_mul] using
  36       Finset.sum_le_sum (s := Finset.univ) (fun x _ => (hf x).1)
  37   have hMhi : M ≤ Fintype.card V * b := by
  38     simpa only [Finset.sum_const, Finset.card_univ, nsmul_eq_mul] using
  39       Finset.sum_le_sum (s := Finset.univ) (fun x _ => (hf x).2)
  40   have hMC : M / p ≤ h * Fintype.card V * Fintype.card V / 2 := by
  41     apply (div_le_iff₀ hp).mpr
  42     calc
  43       M ≤ Fintype.card V * b := hMhi
  44       _ ≤ Fintype.card V * (p * h * Fintype.card V / 2) :=
  45         mul_le_mul_of_nonneg_left hcapacity hn'.le
  46       _ = _ := by ring
  47   have hbound (u : ℝ) (hu : u ≤ b) : u ≤ (1 + δ) * M / Fintype.card V := by
  48     apply (le_div_iff₀ hn').mpr
  49     calc
  50       u * Fintype.card V ≤ b * Fintype.card V := mul_le_mul_of_nonneg_right hu hn'.le
  51       _ ≤ ((1 + δ) * a) * Fintype.card V := mul_le_mul_of_nonneg_right hrelative hn'.le
  52       _ = (1 + δ) * (Fintype.card V * a) := by ring
  53       _ ≤ (1 + δ) * M := mul_le_mul_of_nonneg_left hMlo (by linarith)
  54   apply nearUniform_random_demands tail head f g δ M h p hp hp1 hn hn hδ hδsmall hM hMC
  55     rfl _ (fun x => hbound _ (hf x).2) (fun y => hbound _ (hg y).2) hlarge
  56   dsimp [M]
  57   exact_mod_cast htotal.symm
