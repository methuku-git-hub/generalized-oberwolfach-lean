  11 def transitionPathWeight {X : Type*} (R : ℕ → X → X → ℝ) (i : ℕ) :
  12     (k : ℕ) → (Fin (k+1) → X) → ℝ
  13   | 0, _ => 1
  14   | k+1, w => R i (w 0) (w 1) * transitionPathWeight R (i+1) k (Fin.tail w)
  15 
  16 theorem transitionPathWeight_nonneg {X : Type*}
  17     (R : ℕ → X → X → ℝ) (hR : ∀ i x y, 0 ≤ R i x y)
  18     (i k : ℕ) (w : Fin (k+1) → X) : 0 ≤ transitionPathWeight R i k w := by
  19   induction k generalizing i with
  20   | zero => exact zero_le_one
  21   | succ k ih => exact mul_nonneg (hR i _ _) (ih (i+1) (Fin.tail w))

  23 theorem transitionPathWeight_endpoints {X : Type*} [Fintype X]
  24     (R : ℕ → X → X → ℝ) (i k : ℕ) (f : X → X → ℝ) :
  25     (∑ w : Fin (k+1) → X, transitionPathWeight R i k w * f (w 0) (w (Fin.last k))) =
  26       ∑ x, ∑ z, transitionWalk R i k x z * f x z := by
  27   induction k generalizing i f with
  28   | zero =>
  29     rw [sum_fin_tuple_one]
  30     simp [transitionPathWeight,transitionWalk,identityTransition,ite_mul]
  31   | succ k ih =>
  32     rw [sum_fin_tuple_cons]
  33     simp only [transitionPathWeight,Fin.cons_zero,Fin.cons_one,Fin.tail_cons,Fin.cons_last]
  34     apply Finset.sum_congr rfl
  35     intro x _
  36     have hr (w : Fin (k+1) → X) : R i x (w 0)*transitionPathWeight R (i+1) k w*f x (w (Fin.last k)) =
  37       transitionPathWeight R (i+1) k w*(R i x (w 0)*f x (w (Fin.last k))) := by ring
  38     simp_rw [hr]
  39     rw [ih (i+1) (fun y z => R i x y * f x z)]
  40     simp only [transitionWalk,transitionCompose,Finset.sum_mul]
  41     rw [Finset.sum_comm]
  42     apply Finset.sum_congr rfl
  43     intro y _
  44     apply Finset.sum_congr rfl
  45     intro z _
  46     ring

  48 theorem transitionPathWeight_endpoint_marginal {X : Type*} [Fintype X]
  49     (R : ℕ → X → X → ℝ) (i k : ℕ) (x z : X) :
  50     (∑ w : Fin (k+1) → X, if w 0=x ∧ w (Fin.last k)=z then transitionPathWeight R i k w else 0) =
  51       transitionWalk R i k x z := by
  52   have hh := transitionPathWeight_endpoints R i k (fun a b => if a=x ∧ b=z then 1 else 0)
  53   simpa [mul_ite,ite_and] using hh

  55 def reservedPathWeight {X : Type*} [Fintype X] (R : ℕ → X → X → ℝ)
  56     (i k : ℕ) (Q : X → X → ℝ) (w : Fin (k+1) → X) : ℝ :=
  57   Q (w 0) (w (Fin.last k)) * transitionPathWeight R i k w /
  58     transitionWalk R i k (w 0) (w (Fin.last k))

  60 theorem reservedPathWeight_nonneg {X : Type*} [Fintype X]
  61     (R : ℕ → X → X → ℝ) (hR : ∀ i, IsTransition (R i))
  62     (i k : ℕ) (Q : X → X → ℝ) (hQ : ∀ x z, 0 ≤ Q x z) (w : Fin (k+1) → X) :
  63     0 ≤ reservedPathWeight R i k Q w := by
  64   exact div_nonneg (mul_nonneg (hQ _ _) (transitionPathWeight_nonneg R (fun i => (hR i).1) i k w))
  65     ((transitionWalk_isTransition R hR i k).1 _ _)

  67 theorem reservedPathWeight_endpoints {X : Type*} [Fintype X]
  68     (R : ℕ → X → X → ℝ) (i k : ℕ) (Q : X → X → ℝ)
  69     (hT : ∀ x z, 0 < transitionWalk R i k x z) (x z : X) :
  70     (∑ w : Fin (k+1) → X, if w 0=x ∧ w (Fin.last k)=z then reservedPathWeight R i k Q w else 0) = Q x z := by
  71   have hh := transitionPathWeight_endpoints R i k
  72     (fun a b => if a=x ∧ b=z then Q a b / transitionWalk R i k a b else 0)
  73   have hleft : (∑ w : Fin (k+1) → X,
  74       if w 0=x ∧ w (Fin.last k)=z then reservedPathWeight R i k Q w else 0) =
  75       ∑ w : Fin (k+1) → X, transitionPathWeight R i k w *
  76         (if w 0=x ∧ w (Fin.last k)=z then Q (w 0) (w (Fin.last k)) /
  77           transitionWalk R i k (w 0) (w (Fin.last k)) else 0) := by
  78     apply Finset.sum_congr rfl
  79     intro w _
  80     split_ifs <;> simp only [reservedPathWeight,mul_zero] <;> ring
  81   rw [hleft,hh]
  82   simp [mul_ite,ite_and]
  83   field_simp [(hT x z).ne']
