  13 /-- A spanning family partitions edges exactly once. The vertices are shared. -/
  14 def EdgePartition {n q : ℕ} (G : SimpleGraph (Fin n))
  15     (pieces : Fin q → SimpleGraph (Fin n)) : Prop :=
  16   ∀ u v, (G.Adj u v ↔ ∃ i, (pieces i).Adj u v) ∧
  17     ∀ i j, (pieces i).Adj u v → (pieces j).Adj u v → i = j

  19 /-- Decomposition into the prescribed list, with actual graph isomorphisms. -/
  20 def HasFactorDecomposition {n q : ℕ} (G : SimpleGraph (Fin n))
  21     (F : Fin q → SimpleGraph (Fin n)) : Prop :=
  22   ∃ pieces : Fin q → SimpleGraph (Fin n),
  23     EdgePartition G pieces ∧ ∀ i, Nonempty (pieces i ≃g F i)

  25 /-- One cycle of length r embedded in the ambient n-vertex set; other vertices
  26 are isolated. The two possible cyclic directions define undirected adjacency. -/
  27 def IsCycleCopy {n : ℕ} (H : SimpleGraph (Fin n)) (r : ℕ) : Prop :=
  28   3 ≤ r ∧ ∃ f : Fin r ↪ Fin n, ∀ u v,
  29     H.Adj u v ↔ ∃ i j : Fin r, f i = u ∧ f j = v ∧
  30       ((i.val + 1) % r = j.val ∨ (j.val + 1) % r = i.val)
  31 
  32 /-- Ordinary decompositions use small cycles, not spanning cycle factors. -/
  33 def HasCycleDecomposition {n : ℕ} (G : SimpleGraph (Fin n)) (r : ℕ) : Prop :=
  34   ∃ q : ℕ, ∃ pieces : Fin q → SimpleGraph (Fin n),
  35     EdgePartition G pieces ∧ ∀ i, IsCycleCopy (pieces i) r

  37 def CycleDivisible {n : ℕ} (G : SimpleGraph (Fin n)) (r : ℕ) : Prop := by
  38   classical
  39   exact (∀ v, Even (G.degree v)) ∧ r ∣ G.edgeFinset.card

  41 /-- The infimum in the manuscript's definition of the ordinary threshold. -/
  42 def AdmissibleCycleBound (r : ℕ) (d : ℝ) : Prop := by
  43   classical
  44   exact d ∈ Set.Icc (0 : ℝ) 1 ∧ ∃ N : ℕ, ∀ n ≥ N,
  45     ∀ G : SimpleGraph (Fin n), CycleDivisible G r →
  46       (∀ v, d * (n : ℝ) ≤ (G.degree v : ℝ)) → HasCycleDecomposition G r

  48 noncomputable def cycleThreshold (r : ℕ) : ℝ := sInf {d | AdmissibleCycleBound r d}
  49 noncomputable def allowedThreshold (S : Set ℕ) : ℝ := sSup (cycleThreshold '' S)

  51 def AllowedCycleLengths {n : ℕ} (F : SimpleGraph (Fin n)) (S : Set ℕ) : Prop :=
  52   ∀ v : Fin n, Nat.card {w : Fin n // F.Reachable v w} ∈ S

  54 /-- All inputs of Theorem 1.1 at a specified degree margin. -/

  56   classical

  62 /-- UNPROVED TARGET. In particular N is selected before S. -/
  63 def MainTheorem : Prop :=
