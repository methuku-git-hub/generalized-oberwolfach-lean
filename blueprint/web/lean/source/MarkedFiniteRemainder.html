  41     Fintype.card {x:T // p x.val}=(T.filter p).card:=by
  42   let e:{x:T // p x.val}≃(T.filter p):=
  43     { toFun:=fun x=>⟨x.val.val,Finset.mem_filter.mpr ⟨x.val.property,x.property⟩⟩
  44       invFun:=fun x=>⟨⟨x.val,(Finset.mem_filter.mp x.property).1⟩,(Finset.mem_filter.mp x.property).2⟩
  45       left_inv:=fun _=>rfl
  46       right_inv:=fun _=>rfl }
  47   exact (Fintype.card_congr e).trans (Fintype.card_coe _)

  58     convert h.directed v using 1 <;> congr 1 <;> ext a <;> simp
  59   · intro v

  63   · intro v

  65       Fintype.card {s:S // s.val.1=v∧decide (HK s.val.1 s.val.2)=true}
  66     simp only [decide_eq_true_eq]
  67     rw [marked_finite_filter_card A (fun a=>Sum.elim head khead a=v∧markedArcType a=true),marked_finite_filter_card S (fun s=>s.1=v∧HK s.1 s.2)]
