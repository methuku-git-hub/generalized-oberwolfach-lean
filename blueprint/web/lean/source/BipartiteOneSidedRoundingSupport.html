  15 theorem bipartite_one_sided_rounding_support {X Y E : Type*}
  16     [Fintype X] [Fintype Y] [Fintype E]
  17     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ) (z : E → ℝ)
  18     (hz : ∀ e, 0 ≤ z e ∧ z e ≤ 1)
  19     (hcol : ∀ y, (∑ e, if head e = y then z e else 0) = g y)
  20     (hslack : ∀ x, 0 ≤ (f x:ℝ) - (∑ e, if tail e = x then z e else 0) ∧
  21       (f x:ℝ) - (∑ e, if tail e = x then z e else 0) ≤ 1) :
  22     ∃ D : E → Bool,
  23       (∀ x, (∑ e, if tail e = x ∧ D e = true then (1:ℕ) else 0) ≤ f x) ∧
  24       (∀ y, (∑ e, if head e = y ∧ D e = true then (1:ℕ) else 0) = g y) ∧
  25       (∀ e, z e=0 → D e=false) := by
  26   letI : DecidableEq (Y⊕Unit) := fun a b => Classical.propDecidable (a=b)
  27   let row : X → ℝ := fun x => ∑ e, if tail e = x then z e else 0
  28   have hr : (∑ x, row x) = ∑ e, z e := by
  29     dsimp [row]
  30     rw [Finset.sum_comm]
  31     simp
  32   have hc : (∑ y, (g y:ℝ)) = ∑ e, z e := by
  33     simp_rw [← hcol]
  34     rw [Finset.sum_comm]
  35     simp
  36   have hsum : (∑ y, (g y:ℝ)) ≤ ∑ x, (f x:ℝ) := by
  37     rw [hc, ← hr]
  38     apply Finset.sum_le_sum
  39     intro x _
  40     have hx := (hslack x).1
  41     dsimp only [row]
  42     linarith
  43   have hn : (∑ y, g y) ≤ ∑ x, f x := by exact_mod_cast hsum
  44   let gap : ℕ := (∑ x, f x) - ∑ y, g y
  45   have hgap : (∑ x, ((f x:ℝ) - row x)) = (gap:ℝ) := by
  46     rw [Finset.sum_sub_distrib, hr, ← hc]
  47     dsimp [gap]
  48     rw [Nat.cast_sub hn, Nat.cast_sum, Nat.cast_sum]
  49   let T : E⊕X → X := Sum.elim tail id
  50   let H : E⊕X → Y⊕Unit := Sum.elim (fun e => Sum.inl (head e)) (fun _ => Sum.inr ())
  51   let weight : E⊕X → ℝ := Sum.elim z (fun x => (f x:ℝ)-row x)
  52   let target : Y⊕Unit → ℕ := Sum.elim g (fun _ => gap)
  53   have hw : ∀ e, 0 ≤ weight e ∧ weight e ≤ 1 := by
  54     intro e
  55     cases e with
  56     | inl e => exact hz e
  57     | inr x => exact hslack x
  58   have hrow : ∀ x, (∑ e, if T e = x then weight e else 0) = f x := by
  59     intro x
  60     rw [Fintype.sum_sum_type]
  61     simp only [T,weight,Sum.elim_inl,Sum.elim_inr,id_eq]
  62     have hsingle : (∑ x':X, if x'=x then (f x':ℝ)-row x' else 0) = (f x:ℝ)-row x := by
  63       classical
  64       simpa only [Finset.mem_univ, ite_true] using (Finset.sum_ite_eq' (s:=Finset.univ) (a:=x) (b:=fun x':X => (f x':ℝ)-row x'))
  65     rw [hsingle]
  66     dsimp only [row]
  67     ring
  68   have hhead : ∀ y, (∑ e, if H e = y then weight e else 0) = target y := by
  69     intro y
  70     cases y with
  71     | inl y =>
  72       rw [Fintype.sum_sum_type]
  73       simp only [H,weight,target,Sum.elim_inl,Sum.elim_inr,Sum.inl.injEq,Sum.inr.injEq,
  74         Sum.inr_ne_inl,if_false,Finset.sum_const_zero,add_zero]
  75       exact hcol y
  76     | inr y =>
  77       cases y
  78       rw [Fintype.sum_sum_type]
  79       simp only [H,weight,target,Sum.elim_inl,Sum.elim_inr,Sum.inl_ne_inr,if_false,
  80         Finset.sum_const_zero,zero_add,Sum.inr.injEq,if_true]
  81       exact hgap
  82   obtain ⟨μ,hμ,hMarg⟩ := bipartite_degree_rounding T H f target weight hw hrow hhead
  83   have hp : 0 < μ.probability (fun _ => True) := by
  84     rw [finiteLaw_probability_true]
  85     norm_num
  86   obtain ⟨D,hD,_⟩ := finiteLaw_positive_event_exists μ (fun _ => True) hp
  87   obtain ⟨hDrow,hDcol⟩ := hμ D hD
  88   refine ⟨fun e=>D (Sum.inl e), ?_, ?_, ?_⟩
  89   · intro x
  90     have hh := hDrow x
  91     rw [Fintype.sum_sum_type] at hh
  92     simp only [T,Sum.elim_inl,Sum.elim_inr,id_eq] at hh
  93     have hle : (∑ e:E, if tail e=x ∧ D (Sum.inl e)=true then (1:ℕ) else 0) ≤ f x := by
  94       calc
  95         _ ≤ _ + (∑ e:X, if e=x ∧ D (Sum.inr e)=true then (1:ℕ) else 0) := Nat.le_add_right _ _
  96         _ = f x := hh
  97     exact hle
  98   · intro y
  99     have hh := hDcol (Sum.inl y)
 100     rw [Fintype.sum_sum_type] at hh
 101     simpa only [H,target,Sum.elim_inl,Sum.elim_inr,Sum.inl.injEq,Sum.inr_ne_inl,
 102       false_and,if_false,Finset.sum_const_zero,add_zero] using hh
 103   · intro e he
 104     by_contra hfalse
 105     have htrue : D (Sum.inl e) = true := by cases h : D (Sum.inl e) <;> simp_all
 106     have hm := finiteLaw_probability_mono μ (fun A=>A=D) (fun A=>A (Sum.inl e)=true)
 107       (by intro A hA; subst A; exact htrue)
 108     rw [finiteLaw_probability_point, hMarg (Sum.inl e)] at hm
 109     change μ.mass D ≤ z e at hm
 110     linarith
 111 theorem bipartite_rounding_with_tail_ceilings_support {X Y E : Type*}
 112     [Fintype X] [Fintype Y] [Fintype E]
 113     (tail : E → X) (head : E → Y) (g : Y → ℕ) (z : E → ℝ)
 114     (hz : ∀ e, 0 ≤ z e ∧ z e ≤ 1)
 115     (hcol : ∀ y, (∑ e, if head e = y then z e else 0) = g y) :
 116     ∃ D : E → Bool,
 117       (∀ x, (∑ e, if tail e = x ∧ D e = true then (1:ℕ) else 0) ≤
 118         ⌈∑ e, if tail e = x then z e else 0⌉₊) ∧
 119       (∀ y, (∑ e, if head e = y ∧ D e = true then (1:ℕ) else 0) = g y) ∧
 120       (∀ e, z e=0 → D e=false) := by
 121   apply bipartite_one_sided_rounding_support tail head (fun x=>⌈∑ e, if tail e=x then z e else 0⌉₊) g z hz hcol
 122   intro x
 123   let a : ℝ := ∑ e, if tail e=x then z e else 0
 124   have ha : 0 ≤ a := by
 125     apply Finset.sum_nonneg
 126     intro e _
 127     split_ifs
 128     · exact (hz e).1
 129     · rfl
 130   have hlo := Nat.le_ceil a
 131   have hhi := Nat.ceil_lt_add_one ha
 132   change 0 ≤ ((⌈a⌉₊:ℕ):ℝ)-a ∧ ((⌈a⌉₊:ℕ):ℝ)-a ≤ 1
 133   constructor <;> linarith
 134 end Oberwolfach
