  13 def excludeGraph {V : Type*} (F : SimpleGraph V) (M : V → V → ℕ) (x y : V) : ℕ :=

  16 theorem sampledColourDegree_eq_sum {V : Type*} [Fintype V] {k : ℕ}
  17     (F : SimpleGraph V) (c : Sym2 V → Fin k) (a : Fin k) (x : V) :
  18     sampledColourDegree F c a x = ∑ y, if F.Adj x y ∧ c s(x,y) = a then 1 else 0 := by
  19   rw [sampledColourDegree, Finset.card_filter]
  20   have hh : F.neighborFinset x = Finset.univ.filter (F.Adj x) := by ext y; simp
  21   rw [hh, Finset.sum_filter]
  22   apply Finset.sum_congr rfl
  23   intro y _
  24   by_cases hf : F.Adj x y <;> by_cases hc : c s(x,y) = a <;> simp [hf, hc]

  26 theorem edgeColourGraph_degree {V : Type*} [Fintype V] {k : ℕ}
  27     (F : SimpleGraph V) (c : Sym2 V → Fin k) (a : Fin k) (x : V) :
  28     (edgeColourGraph F c a).degree x = sampledColourDegree F c a x := by

  34 theorem sampledColourDegree_sup_le {V : Type*} [Fintype V] {k : ℕ}
  35     (Z A : SimpleGraph V) (c : Sym2 V → Fin k) (a : Fin k) (x : V) :
  36     sampledColourDegree (Z ⊔ A) c a x ≤ sampledColourDegree Z c a x + A.degree x := by
  37   rw [sampledColourDegree_eq_sum, sampledColourDegree_eq_sum]
  38   have hdeg : (∑ y, if A.Adj x y then 1 else 0 : ℕ) = A.degree x := by
  39     rw [← Finset.card_filter, SimpleGraph.degree]
  40     congr 1
  41     ext y
  42     simp
  43   rw [← hdeg, ← Finset.sum_add_distrib]
  44   apply Finset.sum_le_sum
  45   intro y _
  46   by_cases hz : Z.Adj x y <;> by_cases ha : A.Adj x y <;>
  47     by_cases hc : c s(x,y) = a <;> simp [SimpleGraph.sup_adj, hz, ha, hc]

  54 theorem colourReorientation_cut_loss
  55     {V : Type*} [Fintype V] {k : ℕ}
  56     (c : Sym2 V → Fin k) (M N : V → V → ℕ) (F : SimpleGraph V)
  57     (hM : ∀ x y, M x y ≤ 1)
  58     {b : ℝ} {f : Fin k → ℝ}
  59     (hchange : ∀ x, ((Finset.univ.filter (fun y => M x y ≠ N x y ∨ M y x ≠ N y x)).card : ℝ) ≤ b)
  60     (hforbid : ∀ a x, (sampledColourDegree F c a x : ℝ) ≤ f a)
  61     (a : Fin k) (S T : Finset V) :
  62     (matrixCut (colourRestriction c M a) S T : ℝ) -
  63       (f a + b) * min (S.card : ℝ) T.card ≤
  64       (matrixCut (excludeGraph F (colourRestriction c N a)) S T : ℝ) := by
  65   let W (x y : V) :=
  66     (if F.Adj x y ∧ c s(x,y) = a then 1 else 0 : ℕ) +
  67     (if M x y ≠ N x y ∨ M y x ≠ N y x then 1 else 0 : ℕ)
  68   have hpoint (x y : V) : colourRestriction c M a x y ≤
  69       excludeGraph F (colourRestriction c N a) x y + W x y := by
  70     have hm := hM x y
  71     by_cases hc : c s(x,y) = a
  72     · by_cases hf : F.Adj x y
  73       · simp [colourRestriction, excludeGraph, W, hc, hf]; omega
  74       · by_cases he : M x y = N x y
  75         · simp [colourRestriction, excludeGraph, W, hc, hf, he]
  76         · simp [colourRestriction, excludeGraph, W, hc, hf, he]; omega
  77     · simp [colourRestriction, W, hc]
  78   have hrow (x : V) : ((∑ y, W x y : ℕ) : ℝ) ≤ f a + b := by
  79     dsimp [W]
  80     rw [Finset.sum_add_distrib, Nat.cast_add, ← sampledColourDegree_eq_sum,
  81       ← Finset.card_filter]
  82     exact add_le_add (hforbid a x) (hchange x)
  83   have hsym (x y : V) : W x y = W y x := by
  84     simp only [W, F.adj_comm, Sym2.eq_swap, or_comm]
  85   apply matrixCut_loss_bound _ _ W hpoint hrow
  86   intro y
  87   simpa only [hsym] using hrow y
