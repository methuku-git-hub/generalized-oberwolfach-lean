  17 theorem free_absorber_intersection_count (R : C → Finset V) (S : T → VertexAbsorberTemplate c)
  18     (J : ℕ) (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J)
  19     (F : FreeAbsorberFamily R S) :
  20     Fintype.card {G : FreeAbsorberFamily R S // ¬Disjoint (freeFamilyVertices R S F) (freeFamilyVertices R S G)} ≤
  21       (c*c)^2*Fintype.card T*Fintype.card V^(c*c-1)*J^c := by
  22   let A := freeFamilyVertices R S F
  23   have he (G : FreeAbsorberFamily R S) : ¬Disjoint A (freeFamilyVertices R S G) ↔
  24       ∃ v : A,v.val∈freeFamilyVertices R S G := by
  25     constructor
  26     · intro hd
  27       by_contra hn
  28       exact hd (Finset.disjoint_left.mpr (fun v hv hw => hn ⟨⟨v,hv⟩,hw⟩))
  29     · rintro ⟨v,hv⟩ hd;exact Finset.disjoint_left.mp hd v.property hv
  30   have hcard : Fintype.card {G : FreeAbsorberFamily R S // ¬Disjoint A (freeFamilyVertices R S G)}=
  31       Fintype.card {G : FreeAbsorberFamily R S // ∃ v : A,v.val∈freeFamilyVertices R S G} :=
  32     Fintype.card_congr (Equiv.subtypeEquivRight he)
  33   change Fintype.card {G : FreeAbsorberFamily R S // ¬Disjoint A (freeFamilyVertices R S G)} ≤ _
  34   rw [hcard]
  35   calc
  36     _ ≤ ∑ v : A,Fintype.card {G : FreeAbsorberFamily R S // v.val∈freeFamilyVertices R S G} := by
  37       convert finite_exists_card_le_sum (fun (v : A) G => v.val∈freeFamilyVertices R S G) using 1 <;> congr!
  38     _ ≤ ∑ _v : A,(c*c)*Fintype.card T*Fintype.card V^(c*c-1)*J^c :=
  39       Finset.sum_le_sum (fun v _ => free_absorber_family_vertex_count R S J hJ v.val)
  40     _ = _ := by
  41       simp only [Finset.sum_const,Finset.card_univ,Fintype.card_coe,smul_eq_mul]
  42       rw [show A.card=c*c from free_absorber_vertex_count R (S F.1) F.2]
  43       ring

  45 theorem free_absorber_intersecting_pairs_count (hc : 1 ≤ c) (R : C → Finset V)
  46     (S : T → VertexAbsorberTemplate c) (J : ℕ)
  47     (hJ : ∀ Y : Finset V,Fintype.card {C₀ : C // R C₀=Y} ≤ J) :
  48     Fintype.card {p : FreeAbsorberFamily R S × FreeAbsorberFamily R S //
  49       ¬Disjoint (freeFamilyVertices R S p.1) (freeFamilyVertices R S p.2)} ≤
  50       (Fintype.card T)^2*(c*c)^2*J^(2*c)*Fintype.card V^(2*(c*c)-1) := by
  51   let A := FreeAbsorberFamily R S
  52   let P := fun F G : A => ¬Disjoint (freeFamilyVertices R S F) (freeFamilyVertices R S G)
  53   let e : {p : A×A // P p.1 p.2} ≃ Σ F : A,{G : A // P F G} := {
  54     toFun p := ⟨p.val.1,⟨p.val.2,p.property⟩⟩
  55     invFun p := ⟨(p.1,p.2.val),p.2.property⟩
  56     left_inv := by rintro ⟨⟨F,G⟩,h⟩;rfl
  57     right_inv := by rintro ⟨F,⟨G,h⟩⟩;rfl }
  58   have he : Fintype.card {p : A×A // P p.1 p.2}=∑ F : A,Fintype.card {G : A // P F G} := by
  59     simpa only [Fintype.card_sigma] using Fintype.card_congr e
  60   change Fintype.card {p : A×A // P p.1 p.2} ≤ _
  61   rw [he]
  62   calc
  63     _ ≤ ∑ _F : A,(c*c)^2*Fintype.card T*Fintype.card V^(c*c-1)*J^c :=
  64       Finset.sum_le_sum (fun F _ => free_absorber_intersection_count R S J hJ F)
  65     _ = Fintype.card A*((c*c)^2*Fintype.card T*Fintype.card V^(c*c-1)*J^c) := by simp
  66     _ ≤ (Fintype.card T*Fintype.card V^(c*c)*J^c)*
  67         ((c*c)^2*Fintype.card T*Fintype.card V^(c*c-1)*J^c) :=
  68       Nat.mul_le_mul_right _ (free_absorber_family_card R S J hJ)
  69     _ = _ := by
  70       have hh : 1 ≤ c*c := by nlinarith
  71       have hn : Fintype.card V^(2*(c*c)-1)=Fintype.card V^(c*c)*Fintype.card V^(c*c-1) := by
  72         rw [← pow_add]
  73         congr 1;omega
  74       have hJ' : J^(2*c)=J^c*J^c := by rw [show 2*c=c+c by omega,pow_add]
  75       rw [hn,hJ'];ring
