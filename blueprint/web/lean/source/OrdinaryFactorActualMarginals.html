  11 theorem ordinary_factor_actual_marginals {n:ℕ} (hn:0<n)
  12     (F:SimpleGraph (Fin n)) (hF:F.IsRegularOfDegree 2) (K:ℕ) (hK:0<K)
  13     (i₀:Fin K) (s θ c:ℝ) (hs0:0≤s) (hs1:s≤1/100)
  14     (hθ0:0≤θ) (hθ1:θ≤1/100) (hc:8≤c)
  15     (hreserve:s≤((Finset.univ.filter (fun v=>longPreparationCycleClass F hF K v=some i₀)).card:ℝ)/n):
  16     let fraction:Option (Fin K)→ℝ:=fun j=>
  17       ((Finset.univ.filter (fun v=>longPreparationCycleClass F hF K v=j)).card:ℝ)/n
  18     let f:Fin K→ℝ:=fun i=>fraction (some i)-(if i=i₀ then s else 0)
  19     let long:=fraction none
  20     let cutoff:=θ/(100*(K:ℝ))
  21     let longKept:=if long<θ/100 then 0 else long
  22     let shortKept:=∑ i:Fin K,(if f i<cutoff then 0 else f i)
  23     let omitted:=(∑ i:Fin K,(if f i<cutoff then f i else 0))+
  24       (if long<θ/100 then long else 0)
  25     let a:=(1-θ)*longKept/c
  26     let b:=(c-1)*a+(1-θ)*shortKept
  27     let u:=1-s-a-b;
  28     0≤omitted ∧ omitted≤θ/50 ∧
  29     u=θ*(1-s)+(1-θ)*omitted ∧ θ/2≤u ∧ u≤2*θ ∧
  30     0≤a ∧ a≤1/c ∧ 1/2≤b ∧ s+a+b+u=1 := by
  31   intro fraction f long cutoff longKept shortKept omitted a b u
  32   have hpart:=ordinary_factor_frequency_partition hn F hF K hK i₀ s hs0 hreserve
  33   have hh:=ordinary_actual_frequency_marginals K hK s θ c long f hs0 hs1 hθ0 hθ1 hc hpart.2.1 hpart.1 hpart.2.2
  34   exact hh
  35 end Oberwolfach
