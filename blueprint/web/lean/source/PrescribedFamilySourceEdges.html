   1 import Oberwolfach.PrescribedSourceEdgeUpdate
   2 /-! Maintain the exact relation between tracked source edges and physical
   3 edges across a whole-family component update. -/
   4 set_option autoImplicit false
   5 noncomputable section
   6 namespace Oberwolfach
   7 attribute [local instance] Classical.propDecidable

   9 variable {I X V:Type} [Fintype X] [DecidableEq I] [DecidableEq X] [DecidableEq V]

  11 theorem prescribed_family_source_edge_update
  12     (A:I→Finset X) (f:I→X→V) (Q:I→SimpleGraph X)
  13     (R:I→Finset (Sym2 X)) (E:I→Finset (Sym2 V))
  14     (hrepr:∀ i,E i=(R i).image (Sym2.map (f i)))
  15     (hsource:∀ i,R i⊆(Q i).edgeFinset)
  16     (hends:∀ i e,e∈R i→∀ x∈e.toFinset,x∈A i)
  17     (k:I) (C:Finset X) (g:X→V) (D:Finset (Sym2 X))
  18     (hpres:∀ x∈A k,g x=f k x) (hnewsub:D⊆(Q k).edgeFinset)
  19     (hnewends:∀ e∈D,∀ x∈e.toFinset,x∈A k∪C):
  20     let A':=Function.update A k (A k∪C);
  21     let f':=Function.update f k g;
  22     let R':=Function.update R k (R k∪D);
  23     let E':=Function.update E k (E k∪D.image (Sym2.map g));
  24     (∀ i,E' i=(R' i).image (Sym2.map (f' i))) ∧
  25     (∀ i,R' i⊆(Q i).edgeFinset) ∧
  26     (∀ i e,e∈R' i→∀ x∈e.toFinset,x∈A' i) := by
  27   intro A' f' R' E'
  28   constructor
  29   · intro i
  30     by_cases hi:i=k
  31     · subst i
  32       simp only [E',R',f',Function.update_self]
  33       rw [hrepr k]
  34       exact (prescribed_source_edge_update (A k) (f k) g hpres (R k) D (hends k)).symm
  35     · simpa only [E',R',f',Function.update_of_ne hi] using hrepr i
  36   constructor
  37   · intro i
  38     by_cases hi:i=k
  39     · subst i
  40       simp only [R',Function.update_self]
  41       exact Finset.union_subset (hsource k) hnewsub
  42     · simpa only [R',Function.update_of_ne hi] using hsource i
  43   · intro i e he x hx
  44     by_cases hi:i=k
  45     · subst i
  46       have he':e∈R k∪D:=by simpa only [R',Function.update_self] using he
  47       have hx':x∈A k∪C:=by
  48         rcases Finset.mem_union.mp he' with he'|he'
  49         · exact Finset.mem_union_left C (hends k e he' x hx)
  50         · exact hnewends e he' x hx
  51       simpa only [A',Function.update_self] using hx'
  52     · have he':e∈R i:=by simpa only [R',Function.update_of_ne hi] using he
  53       have hx':x∈A i:=hends i e he' x hx
  54       simpa only [A',Function.update_of_ne hi] using hx'
  55 end Oberwolfach
