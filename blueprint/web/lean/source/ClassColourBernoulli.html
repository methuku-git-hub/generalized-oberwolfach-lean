  13 theorem categorical_class_membership_law {k:ℕ} (ρ:Fin k→ℝ)
  14     (hρ:∀ a,0≤ρ a) (htotal:∑ a,ρ a=1) (a:Fin k):
  15     (FiniteLaw.categorical ρ hρ htotal).map (classColourMembership a)=
  16       FiniteLaw.retentionCoin (ρ a) (hρ a)
  17         (by have h:=Finset.single_le_sum (fun b _=>hρ b) (Finset.mem_univ a);linarith):=by
  18   apply finiteLaw_ext
  19   intro b
  20   cases b with
  21   | false =>
  22     simp only [FiniteLaw.map,FiniteLaw.retentionCoin,Bool.false_eq_true,ite_false]
  23     have hsum:(∑ c:Fin k,((if c=a then ρ c else 0)+(if c=a then 0 else ρ c)))=1:=by
  24       calc
  25         _ = ∑ c:Fin k,ρ c:=by
  26           apply Finset.sum_congr rfl
  27           intro c _
  28           by_cases h:c=a <;> simp [h]
  29         _ = 1:=htotal
  30     have hs0:(∑ c:Fin k,if c=a then ρ c else 0)=ρ a:=by simp
  31     rw [Finset.sum_add_distrib,hs0] at hsum
  32     have hs:(∑ c:Fin k,if c=a then 0 else ρ c)=1-ρ a:=by linarith
  33     calc
  34       _ = ∑ c:Fin k,if c=a then 0 else ρ c:=by
  35         apply Finset.sum_congr rfl
  36         intro c _
  37         by_cases h:c=a <;> simp [classColourMembership,FiniteLaw.categorical,h]
  38       _ = _:=hs
  39   | true =>
