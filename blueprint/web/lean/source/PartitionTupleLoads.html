  14 theorem partitionTuple_position_loads {V : Type*} [Fintype V] {ℓ k : ℕ} {d : Fin k → ℕ}
  15     (P : FixedPartition V k d) (a : Fin (ℓ + 2) ⊕ Fin (ℓ + 2) → Fin k)
  16     (m : ℕ) (hm : ∀ s, d (a s) = m)
  17     (w : (Fin (ℓ + 2) → Fin m × Fin m) → ℝ) (q : ℝ)
  18     (hw : ∀ i x, (∑ s, if (s i).1 = x then w s else 0) = q ∧
  19       (∑ s, if (s i).2 = x then w s else 0) = q)
  20     (i : Fin (ℓ + 2)) (v : V) :
  21     (∑ t, if (t i).1 = v then embeddedWeight (partitionCyclicTupleEmbedding P a m hm) w t else 0) =
  22       (if P.1 v = a (Sum.inl i) then q else 0) ∧
  23     (∑ t, if (t i).2 = v then embeddedWeight (partitionCyclicTupleEmbedding P a m hm) w t else 0) =
  24       (if P.1 v = a (Sum.inr i) then q else 0) := by
  25   let f (j) := partitionCellEmbedding P (a (Sum.inl j)) m (hm _)
  26   let g (j) := partitionCellEmbedding P (a (Sum.inr j)) m (hm _)
  27   constructor
  28   · by_cases hv : P.1 v = a (Sum.inl i)
  29     · obtain ⟨x,hx⟩ := (partitionCellEmbedding_range P (a (Sum.inl i)) m (hm _) v).mpr hv
  30       rw [if_pos hv, ← hx]
  31       exact (pairTuple_weight_positions f g w i x).1.trans (hw i x).1
  32     · rw [if_neg hv]
  33       exact pairTuple_weight_position_outside f g w i v
  34         ((partitionCellEmbedding_range P (a (Sum.inl i)) m (hm _) v).not.mpr hv)
  35   · by_cases hv : P.1 v = a (Sum.inr i)
  36     · obtain ⟨x,hx⟩ := (partitionCellEmbedding_range P (a (Sum.inr i)) m (hm _) v).mpr hv
  37       rw [if_pos hv, ← hx]
  38       exact (pairTuple_weight_positions f g w i x).2.trans (hw i x).2
  39     · rw [if_neg hv, embeddedWeight_incidence]
  40       apply Finset.sum_eq_zero
  41       intro s _
  42       apply if_neg
  43       intro hs
  44       apply hv
  45       rw [← hs]
  46       exact partitionCellEmbedding_colour P (a (Sum.inr i)) m (hm _) (s i).2

 160 theorem partitionTuple_weight_support_and_bounds {V : Type*} [Fintype V]
 161     {ℓ k : ℕ} {d : Fin k → ℕ}
 162     (G : SimpleGraph V) (σ : Fin (ℓ + 2) → Equiv.Perm V)
 163     (P : FixedPartition V k d) (a : Fin (ℓ + 2) ⊕ Fin (ℓ + 2) → Fin k)
 164     (m : ℕ) (hm : ∀ s, d (a s) = m)
 165     (w : (Fin (ℓ + 2) → Fin m × Fin m) → ℝ) (A B : ℝ) (hB : 0 ≤ B)
 166     (hw : ∀ v, 0 ≤ w v)
 167     (hs : ∀ v, 0 < w v → CyclicMatrixTupleValid (partitionCyclicMatrices G σ P a m hm) v)
 168     (hlo : ∀ v, CyclicMatrixTupleValid (partitionCyclicMatrices G σ P a m hm) v → A ≤ w v)
 169     (hhi : ∀ v, w v ≤ B) :
 170     let W := embeddedWeight (partitionCyclicTupleEmbedding P a m hm) w
 171     (∀ t, 0 ≤ W t ∧ W t ≤ B) ∧
 172     (∀ t, 0 < W t → TwistedCyclicTupleValid G σ t ∧
 173       ∀ i, P.1 (t i).1 = a (Sum.inl i) ∧ P.1 (t i).2 = a (Sum.inr i)) ∧
 174     (∀ t, TwistedCyclicTupleValid G σ t →
 175       (∀ i, P.1 (t i).1 = a (Sum.inl i) ∧ P.1 (t i).2 = a (Sum.inr i)) → A ≤ W t) := by
 176   let e := partitionCyclicTupleEmbedding P a m hm
 177   refine ⟨fun t => ⟨embeddedWeight_nonneg e w hw t, embeddedWeight_bound e w B hB hhi t⟩, ?_, ?_⟩
 178   · intro t ht
 179     have hrange : t ∈ Set.range e := by
 180       by_contra hn
 181       have hz := embeddedWeight_not_in_range e w t hn
 182       linarith
 183     obtain ⟨v,rfl⟩ := hrange
 184     rw [embeddedWeight_apply] at ht
 185     exact ⟨(partitionCyclicMatrices_valid G σ P a m hm v).mp (hs v ht),
 186       (partitionCyclicTupleEmbedding_range P a m hm (e v)).mp ⟨v,rfl⟩⟩
 187   · intro t ht hplace
 188     obtain ⟨v,rfl⟩ := (partitionCyclicTupleEmbedding_range P a m hm t).mpr hplace
 189     rw [embeddedWeight_apply]
 190     exact hlo v ((partitionCyclicMatrices_valid G σ P a m hm v).mpr ht)
