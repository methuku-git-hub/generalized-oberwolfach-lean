  11 theorem long_preparation_factor_fractions {n:ℕ} (hn:0<n)
  12     (F:SimpleGraph (Fin n)) (hF:F.IsRegularOfDegree 2)
  13     (K:ℕ) (hK:0<K) (θ:ℝ) (hθ:0≤θ):
  14     let fraction:Option (Fin K)→ℝ:=fun j=>
  15       ((Finset.univ.filter (fun v=>longPreparationCycleClass F hF K v=j)).card:ℝ)/n
  16     let S:Finset (Fin K):=Finset.univ.filter (fun i=>θ/(100*(K:ℝ))≤fraction (some i))
  17     let rare:=∑ i∈Finset.univ\S,fraction (some i);
  18     (∀ v, longPreparationCycleClass F hF K v=none ↔ K<Fintype.card (F.connectedComponentMk v)) ∧
  19     (∀ v i, longPreparationCycleClass F hF K v=some i ↔ Fintype.card (F.connectedComponentMk v)=i.val+1) ∧
  20     (∀ j,0≤fraction j) ∧ (∀ i∈S,θ/(100*(K:ℝ))≤fraction (some i)) ∧
  21     0≤rare ∧ rare≤θ/100 ∧ fraction none+(∑ i∈S,fraction (some i))+rare=1:=by
  22   intro fraction S rare
  23   have hn':0<Fintype.card (Fin n):=by simpa using hn
  24   have hfreq:=long_preparation_vertex_fractions hn' K (longPreparationCycleClass F hF K)
  25   have hfreq':(∀ j,0≤fraction j) ∧ fraction none+(∑ i,fraction (some i))=1:=by
  26     simpa [fraction] using hfreq
  27   have hret:=long_preparation_retained_fractions K hK θ (fraction none) hθ
  28     (fun i=>fraction (some i)) (fun i=>hfreq'.1 _) hfreq'.2
  29   have hret':(∀ i∈S,θ/(100*(K:ℝ))≤fraction (some i)) ∧
  30       0≤rare ∧ rare≤θ/100 ∧ fraction none+(∑ i∈S,fraction (some i))+rare=1:=by
  31     simpa [S,rare] using hret
  32   exact ⟨fun v=>(long_preparation_cycle_class F hF K v).1,
  33     fun v=>(long_preparation_cycle_class F hF K v).2,hfreq'.1,hret'⟩
  34 end Oberwolfach
