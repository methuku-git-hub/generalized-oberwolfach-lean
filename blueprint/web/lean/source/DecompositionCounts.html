  16 theorem EdgePartition.filtered_edge_count {n q : ℕ} {G : SimpleGraph (Fin n)}
  17     {H : Fin q → SimpleGraph (Fin n)} (h : EdgePartition G H)
  18     (P : Sym2 (Fin n) → Prop) [DecidablePred P] :
  19     (G.edgeFinset.filter P).card = ∑ i, ((H i).edgeFinset.filter P).card := by
  20   classical
  21   have heq : G.edgeFinset.filter P =
  22       Finset.univ.biUnion (fun i : Fin q => (H i).edgeFinset.filter P) := by
  23     ext e
  24     induction e using Sym2.ind with
  25     | _ u v =>
  26       simp only [Finset.mem_filter, SimpleGraph.mem_edgeFinset, Finset.mem_biUnion,
  27         Finset.mem_univ, true_and]
  28       exact ⟨fun ⟨he, hp⟩ => by
  29         obtain ⟨i, hi⟩ := (h u v).1.mp he
  30         exact ⟨i, hi, hp⟩,
  31         fun ⟨i, hi, hp⟩ => ⟨(h u v).1.mpr ⟨i, hi⟩, hp⟩⟩
  32   rw [heq, Finset.card_biUnion]
  33   intro i hi j hj hij
  34   apply Finset.disjoint_left.mpr
  35   intro e hei hej
  36   have hei' := (Finset.mem_filter.mp hei).1
  37   have hej' := (Finset.mem_filter.mp hej).1
  38   induction e using Sym2.ind with
  39   | _ u v =>
  40     exact hij ((h u v).2 i j (SimpleGraph.mem_edgeFinset.mp hei')
  41       (SimpleGraph.mem_edgeFinset.mp hej'))
