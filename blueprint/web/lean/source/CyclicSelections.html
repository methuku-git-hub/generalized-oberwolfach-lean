  20   intro u v h
  21   cases u with

  53 theorem cyclicSelection_adj_forces {k : ℕ} {X : Type*}
  54     (f : Fin (k + 2) ⊕ Fin (k + 2) → X)
  55     (u v : (Fin (k + 2) × X) ⊕ (Fin (k + 2) × X))
  56     (h : (cyclicSelectionGraph f).Adj u v) :
  57     f (taggedClass u) = taggedValue u ∧ f (taggedClass v) = taggedValue v := by
  58   cases u with
  59   | inl a =>
  60     cases v with
  61     | inl b => exact False.elim ((alternatingTupleGraph_no_same _).1 a b h)

  63       have hh := (alternatingTupleGraph_adj_cross _ a.1 b.1 a.2 b.2).mp h
