  10 theorem cycle_walk_support_toFinset (F:SimpleGraph V) {v:V}
  11     (p:F.Walk v v) (hp:p.IsCycle):
  12     p.support.toFinset=p.support.dropLast.toFinset := by
  13   have hlen : p.support.dropLast.length = p.length := by
  14     simp [List.length_dropLast,p.length_support]
  15   have hne:p.support.dropLast≠[] := by
  16     apply List.ne_nil_of_length_pos
  17     rw [hlen]
  18     have hh:=hp.isCircuit.three_le_length
  19     omega
  20   have hv : v∈p.support.dropLast := by
  21     have hh:=List.head_mem hne
  22     simpa [List.head_dropLast,p.head_support] using hh
  23   have hclosed:p.support.dropLast++[v]=p.support := by
  24     simpa using List.dropLast_append_getLast p.support_ne_nil
  25   ext x
  26   simp only [List.mem_toFinset]
  27   have hmem : (x∈p.support.dropLast ∨ x=v) ↔ x∈p.support := by
  28     simpa only [List.mem_append,List.mem_singleton] using
  29       Iff.of_eq (congrArg (fun xs : List V => x∈xs) hclosed)
  30   constructor
  31   · intro hx
  32     rcases hmem.2 hx with h|rfl
  33     · exact h
  34     · exact hv
  35   · intro hx
  36     exact hmem.1 (Or.inl hx)
