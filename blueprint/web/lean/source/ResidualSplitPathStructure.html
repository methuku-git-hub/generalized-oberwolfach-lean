  11 theorem split_path_graph_structure (H:SimpleGraph V) (xs:List V) (hxs:xs.Nodup):
  12     (splitPathByGraph H xs).flatten=xs ∧
  13     (∀ block∈splitPathByGraph H xs,block.Nodup ∧ List.IsChain H.Adj block) ∧
  14     (splitPathByGraph H xs).Pairwise (fun p q=>Disjoint p.toFinset q.toFinset) := by
  15   have hflat:(splitPathByGraph H xs).flatten=xs := split_ordered_path_flatten _ xs
  16   have hn:(splitPathByGraph H xs).flatten.Nodup := by rw [hflat];exact hxs
  17   obtain ⟨hblocks,hpair⟩:=List.nodup_flatten.mp hn
  18   refine ⟨hflat,?_,?_⟩
  19   · intro block hblock
  20     refine ⟨hblocks block hblock,?_⟩
  21     apply (prescribed_directed_edges_chain H.Adj block).2
  22     intro e he
  23     rcases e with ⟨x,y⟩
  24     have hp:s(x,y)∈prescribedOrderedPathEdges block := directed_path_edge_is_physical block x y he
  25     have hu:s(x,y)∈splitPathEdgeUnion H xs := by
  26       exact Finset.mem_biUnion.mpr ⟨block,List.mem_toFinset.mpr hblock,hp⟩
  27     rw [split_path_edge_union] at hu
  28     simpa only [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using (Finset.mem_filter.mp hu).2
  29   · apply hpair.imp
  30     intro p q hd
  31     apply Finset.disjoint_left.mpr
  32     intro x hx hy
  33     exact List.disjoint_left.mp hd (List.mem_toFinset.mp hx) (List.mem_toFinset.mp hy)
