  15 def triangleAugmentationGraph (G D : SimpleGraph V) (S : Finset V) : SimpleGraph V where
  16   Adj v w := G.Adj v w ∧ (v∈S ∨ w∈S) ∧ ¬D.Adj v w
  17   symm := ⟨by intro v w h;exact ⟨h.1.symm,h.2.1.symm,fun hd=>h.2.2 hd.symm⟩⟩
  18   loopless := ⟨by intro v h;exact h.1.ne rfl⟩

  20 theorem triangleAugmentationGraph_degree (G D : SimpleGraph V) (S : Finset V)
  21     (hD : ∀ v,D.degree v≤2) (v : V) :
  22     ((G.neighborFinset v∩S).card:ℝ)-2≤(triangleAugmentationGraph G D S).degree v := by
  23   have hs : G.neighborFinset v∩S⊆
  24       (triangleAugmentationGraph G D S).neighborFinset v∪D.neighborFinset v := by
  25     intro w hw
  26     have hg : G.Adj v w := by simpa using (Finset.mem_inter.mp hw).1
  27     by_cases hd:D.Adj v w
  28     · apply Finset.mem_union_right;simpa using hd
  29     · apply Finset.mem_union_left
  30       have hh : (triangleAugmentationGraph G D S).Adj v w := ⟨hg,Or.inr (Finset.mem_inter.mp hw).2,hd⟩
  31       simpa using hh
  32   have hh:=(Finset.card_le_card hs).trans (Finset.card_union_le _ _)
  33   rw [SimpleGraph.card_neighborFinset_eq_degree,SimpleGraph.card_neighborFinset_eq_degree] at hh
  34   have hhR : ((G.neighborFinset v∩S).card:ℝ)≤
  35       (triangleAugmentationGraph G D S).degree v+(D.degree v:ℝ) := by exact_mod_cast hh
  36   have hdR : (D.degree v:ℝ)≤2 := by exact_mod_cast hD v
  37   linarith only [hhR,hdR]

  39 theorem graph_induced_degree_as_intersection (G : SimpleGraph V) (S : Finset V)
  40     (v : (S:Set V)) :
  41     (G.induce (S:Set V)).degree v=(G.neighborFinset v.val∩S).card := by

  48 theorem two_triangle_factors_at_hub (β : ℝ) (hβ : 0<β) (q : ℕ) :
  49     ∃ s₀ : ℕ,∀ (V : Type) [Fintype V] [DecidableEq V],∀ (G : SimpleGraph V) (S : Finset V),
  50       s₀≤S.card → 3∣S.card → 3∣Fintype.card V → Fintype.card V-S.card≤q →
  51       (∀ v,(3/4+β/4)*(S.card:ℝ)≤((G.neighborFinset v∩S).card:ℝ)) →
  52       ∃ M₁ M₂ : Finset (OrdinaryCycle G 3),
  53         (∀ C∈M₁,∀ D∈M₁,C≠D → Disjoint (ordinaryCycleVertices G 3 C) (ordinaryCycleVertices G 3 D)) ∧
  54         M₁.biUnion (ordinaryCycleVertices G 3)=Finset.univ ∧
  55         (∀ C∈M₂,∀ D∈M₂,C≠D → Disjoint (ordinaryCycleVertices G 3 C) (ordinaryCycleVertices G 3 D)) ∧
  56         M₂.biUnion (ordinaryCycleVertices G 3)=S ∧
  57         (∀ C∈M₁,∀ D∈M₂,Disjoint C.val.val.edgeFinset D.val.val.edgeFinset) ∧
  58         (∀ C∈M₁,∀ v w,C.val.val.Adj v w → v∈S ∨ w∈S) := by
  59   obtain ⟨N,hN⟩:=minimum_degree_triangle_factor (β/4) (by positivity)
  60   obtain ⟨K,hK⟩:=bounded_augmentation_triangle_factor β hβ q
  61   refine ⟨max N K,?_⟩
  62   intro V _ _ G S hs hSdiv hdiv hq hdeg
  63   have hS : HasResourceDecomposition (ordinaryCycleVertices G 3) S := by
  64     apply ordinary_factor_induced_lift S
  65     apply hN (S:Set V)
  66     · simpa using (le_max_left N K).trans hs
  67     · simpa using hSdiv
  68     · intro v
  69       rw [graph_induced_degree_as_intersection]
  70       simpa using hdeg v.val
  71   obtain ⟨M₂,hM₂,hcover₂⟩:=hS
  72   let D:=ordinaryFactorGraph M₂
  73   let Q:=triangleAugmentationGraph G D S
  74   have hD : ∀ v,D.degree v≤2 := ordinaryFactorGraph_degree_le_two M₂ hM₂
  75   have hQ : Q≤G := fun _ _ h=>h.1
  76   have hf : HasResourceDecomposition (ordinaryCycleVertices Q 3) Finset.univ := by
  77     apply hK S.card (Fintype.card V-S.card) ((le_max_right N K).trans hs) hq V
  78       (by have hh:=Finset.card_le_univ S;omega) hdiv Q
  79     intro v
  80     exact (sub_le_sub_right (hdeg v) 2).trans (triangleAugmentationGraph_degree G D S hD v)
  81   let lift : OrdinaryCycle Q 3→OrdinaryCycle G 3 := fun C=>⟨C.val,C.property.trans hQ⟩
  82   obtain ⟨M₁,hM₁,hcover₁⟩:=hf
  83   have hdec := resource_decomposition_map (ordinaryCycleVertices Q 3) (ordinaryCycleVertices G 3)
  84     lift (fun _=>rfl) Finset.univ ⟨M₁,hM₁,hcover₁⟩
  85   have hpack : ∀ C∈M₁.image lift,∀ E∈M₁.image lift,C≠E →
  86       Disjoint (ordinaryCycleVertices G 3 C) (ordinaryCycleVertices G 3 E) := by
  87     intro C hC E hE hne
  88     obtain ⟨C,hCM,rfl⟩:=Finset.mem_image.mp hC
  89     obtain ⟨E,hEM,rfl⟩:=Finset.mem_image.mp hE
  90     exact hM₁ C hCM E hEM (fun he=>hne (congrArg lift he))
  91   have hcover : (M₁.image lift).biUnion (ordinaryCycleVertices G 3)=Finset.univ := by
  92     rw [←hcover₁]
  93     ext v
  94     simp only [Finset.mem_biUnion,Finset.mem_image]
  95     constructor
  96     · rintro ⟨_,⟨C,hC,rfl⟩,hv⟩;exact ⟨C,hC,hv⟩
  97     · rintro ⟨C,hC,hv⟩;exact ⟨lift C,⟨C,hC,rfl⟩,hv⟩
  98   refine ⟨M₁.image lift,M₂,hpack,hcover,hM₂,hcover₂,?_,?_⟩
  99   · intro C hC E hE
 100     obtain ⟨C,hCM,rfl⟩:=Finset.mem_image.mp hC
 101     apply Finset.disjoint_left.mpr
 102     intro e he hf
 103     induction e using Sym2.ind with
 104     | _ v w =>
 105       have hc : C.val.val.Adj v w := by simpa using he
 106       have hd : E.val.val.Adj v w := by simpa using hf
 107       have hQedge:=C.property hc
 108       have hDedge : D.Adj v w := by
 109         change (⨆ C∈M₂,C.val.val).Adj v w

 111         exact ⟨E,hE,hd⟩
 112       exact hQedge.2.2 hDedge
