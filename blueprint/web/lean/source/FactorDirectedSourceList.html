  11 theorem two_factor_directed_source_cycles (F:SimpleGraph V) (hF:F.IsRegularOfDegree 2):
  12     ∃ (D:V→V→Prop) (orig:List (PrescribedSourcePiece V)),
  13       orig.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset) ∧
  14       (∀p∈orig,∃a xs,p=.cycle a xs ∧ p.OriginalValid D) ∧
  15       sourcePieceVerticesUnion orig=Finset.univ ∧ sourcePieceEdgesUnion orig=F.edgeFinset ∧
  16       ∀x y,D x y→F.Adj x y := by
  17   obtain ⟨root,tail,D,hvalid,hverts,hdisj,hcover,hsub,hedges⟩:=two_factor_components_as_directed_pieces F hF
  18   letI : Fintype F.ConnectedComponent := Fintype.ofFinite _
  19   let orig:List (PrescribedSourcePiece V):=Finset.univ.toList.map (fun c=>.cycle (root c) (tail c))
  20   refine ⟨D,orig,?_,?_,?_,?_,hsub⟩
  21   · dsimp [orig]
  22     rw [List.pairwise_map]
  23     apply List.Pairwise.imp ?_ (Finset.nodup_toList Finset.univ)
  24     intro c c' hne
  25     apply Finset.disjoint_left.mpr
  26     intro x hx hx'
  27     have hxc:x∈root c::tail c := by simpa [PrescribedSourcePiece.vertices] using hx
  28     have hxc':x∈root c'::tail c' := by simpa [PrescribedSourcePiece.vertices] using hx'
  29     exact hne (hdisj c c' x hxc hxc')
  30   · intro p hp
  31     obtain ⟨c,_,rfl⟩:=List.mem_map.mp hp
  32     have hh:=hvalid c
  33     refine ⟨root c,tail c,rfl,hh.1,hh.2.1,?_⟩
  34     have hc':= (directed_cycle_through_iff_chain D ⊥ (root c) (tail c)).1 hh.2.2
  35     simpa using hc'
  36   · ext x
  37     simp only [Finset.mem_univ,iff_true]
  38     obtain ⟨c,hc⟩:=hcover x
  39     exact Finset.mem_biUnion.mpr ⟨.cycle (root c) (tail c),
  40       List.mem_toFinset.mpr (List.mem_map.mpr ⟨c,by simp,rfl⟩),by simpa [PrescribedSourcePiece.vertices] using hc⟩
  41   · ext e
  42     induction e using Sym2.ind with
  43     | _ x y =>
  44       simp only [sourcePieceEdgesUnion,Finset.mem_biUnion,List.mem_toFinset,SimpleGraph.mem_edgeFinset,
  45         SimpleGraph.mem_edgeSet]
  46       constructor
  47       · rintro ⟨p,hp,he⟩
  48         obtain ⟨c,_,rfl⟩:=List.mem_map.mp hp
  49         have hx: x∈root c::tail c := (source_piece_physical_edge_endpoints (.cycle (root c) (tail c)) x y he).1
  50         exact (hedges x y).2 ⟨c,hx,he⟩
  51       · intro he
  52         obtain ⟨c,hx,he⟩:=(hedges x y).1 he
  53         exact ⟨.cycle (root c) (tail c),List.mem_map.mpr ⟨c,by simp,rfl⟩,he⟩
  54 end Oberwolfach
