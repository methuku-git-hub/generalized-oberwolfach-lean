  12  theorem endpoint_spoke_normalisation (m b c p p₀ ε e:ℝ)
  13     (hb:0≤b) (hp₀:0<p₀) (hp:p₀≤p) (hp1:p≤1) (hε:0≤ε)
  14     (he:0≤e) (hεsmall:ε≤p₀/2) (hεe:4*ε≤e*p₀)
  15     (hm:|m-p*b|≤ε*b) :
  16     p₀*b/2≤m ∧
  17     (|c-p^2*b|≤ε*b → (p-e)*m≤c) ∧
  18     (c≤p^3*b+ε*b → c≤(p^2+e)*m) := by
  19   have hp0:0≤p:=hp₀.le.trans hp
  20   have hp2:0≤p^2:=sq_nonneg p
  21   have hp21:p^2≤1:=by nlinarith only [hp0,hp1]
  22   have hmlo:= (abs_le.mp hm).1
  23   have hmhi:= (abs_le.mp hm).2
  24   have hpB:=mul_le_mul_of_nonneg_right hp hb
  25   have heB:=mul_le_mul_of_nonneg_right hεsmall hb
  26   have hm':p₀*b/2≤m:=by linarith only [hmlo,hpB,heB]
  27   have heM:=mul_le_mul_of_nonneg_left hm' he
  28   have hee:=mul_le_mul_of_nonneg_right hεe hb
  29   have hbudget:2*ε*b≤e*m:=by nlinarith only [heM,hee]
  30   refine ⟨hm',?_,?_⟩
  31   · intro hc
  32     have hmul:=mul_le_mul_of_nonneg_left hmhi hp0
  33     have hεp:=mul_le_mul_of_nonneg_right hp1 (mul_nonneg hε hb)
  34     nlinarith only [hmul,hεp,hbudget,(abs_le.mp hc).1]
  35   · intro hc
  36     have hmul:=mul_le_mul_of_nonneg_left hmlo hp2
  37     have hεp:=mul_le_mul_of_nonneg_right hp21 (mul_nonneg hε hb)
  38     nlinarith only [hmul,hεp,hbudget,hc]

  40 theorem endpoint_column_normalisation (m b c a q p ε:ℝ)
  41     (hp:0≤p) (hp1:p≤1) (hεa:0≤ε*a)
  42     (hsize:|2*m-q*b|≤ε*a) (hcount:|c-q*p*b|≤ε*a)
  43     (hbudget:2*ε*a≤p*m) : p*m≤c := by
  44   have hh:=mul_le_mul_of_nonneg_left (abs_le.mp hsize).2 hp
  45   have he:=mul_le_mul_of_nonneg_right hp1 hεa
  46   nlinarith only [hh,he,hbudget,(abs_le.mp hcount).1]

  50 theorem ordinary_endpoint_counts_normalised (P R F:SimpleGraph V) (H:V→U→Prop)
  51     (A:Finset V) (B:Finset U) (x:V) (q p p₀ ε e:ℝ)
  52     (hRP:R≤P) (hFx:∀ v,R.Adj x v → ¬F.Adj x v)
  53     (hp₀:0<p₀) (hp:p₀≤p) (hp1:p≤1) (hε:0≤ε) (he:0≤e)
  54     (hεsmall:ε≤p₀/2) (hεe:4*ε≤e*p₀)
  55     (hbal:(A∩R.neighborFinset x).card=2*(B.filter (H x)).card)
  56     (hbudget:2*ε*A.card≤p*(B.filter (H x)).card)
  57     (hrow:∀ S:Finset V,∀ J:Finset U,S.card≤1 → J.card≤1 →
  58       |((common (common A S (fun v=>R.neighborFinset v)) J
  59         (fun u=>Finset.univ.filter (fun v=>H v u))).card:ℝ)-
  60         q^S.card*p^J.card*(common A S (fun v=>P.neighborFinset v)).card|≤ε*A.card)
  61     (hindex:∀ S:Finset V,S.card≤3 → (∀ v∈S,∀ w∈S,¬F.Adj v w) →
  62       |((common B S (fun v=>Finset.univ.filter (H v))).card:ℝ)-p^S.card*B.card|≤ε*B.card) :
  63     let S:=A∩R.neighborFinset x
  64     let T:=B.filter (H x)
  65     p₀*B.card/2≤(T.card:ℝ) ∧
  66     (∀ v∈S,(p-e)*T.card≤((T.filter (H v)).card:ℝ)) ∧
  67     (∀ u∈B,p*T.card≤((S.filter (fun v=>H v u)).card:ℝ)) ∧
  68     (∀ v∈S,∀ w∈S,v≠w → ¬F.Adj v w →
  69       ((T.filter (fun u=>H v u∧H w u)).card:ℝ)≤(p^2+e)*T.card) := by
  70   dsimp only
  71   obtain ⟨hS,hT,hrows,hcols,hcodeg⟩:=ordinary_endpoint_counts_from_common P R F H A B x q p ε hRP hFx hrow hindex
  72   have hnorm (c:ℝ):=endpoint_spoke_normalisation (B.filter (H x)).card B.card c p p₀ ε e
  73     (Nat.cast_nonneg _) hp₀ hp hp1 hε he hεsmall hεe hT
  74   refine ⟨(hnorm 0).1,?_,?_,?_⟩
  75   · intro v hv
  76     exact (hnorm _).2.1 (hrows v hv)
  77   · intro u hu
  78     have hS':|2*(B.filter (H x)).card-q*(A∩P.neighborFinset x).card|≤ε*A.card:=by
  79       simpa only [hbal,Nat.cast_mul,Nat.cast_ofNat] using hS
  80     exact endpoint_column_normalisation _ _ _ _ q p ε (hp₀.le.trans hp) hp1 (by positivity)
  81       hS' (hcols u hu) hbudget
  82   · intro v hv w hw hvw hfw
  83     exact (hnorm _).2.2 (hcodeg v hv w hw hvw hfw)
