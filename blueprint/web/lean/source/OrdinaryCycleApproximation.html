  18 theorem ordinary_cycle_approximation_at_size (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  19     ∃ N : ℕ,∀ m≥N,4≤m → 3≤α*m/4 →
  20       ∀ (V : Type) [Fintype V] [DecidableEq V],m≤Fintype.card V →
  21       ∀ G : SimpleGraph V,
  22         (∀ v,(cycleThreshold r+α)*Fintype.card V≤(G.degree v:ℝ)) →
  23       ∀ κ : ℝ,0≤κ → κ≤1 → κ≤α/16 →
  24       ∀ w : G.edgeSet → ℝ,‖w-(fun _ => (1:ℝ))‖≤κ →
  25       ∃ a : OrdinaryCycle G r → ℝ,
  26         (∀ C,0≤a C ∧ a C≤4*(m:ℝ)^(r-2)/(Fintype.card V:ℝ)^(r-2)) ∧
  27         ‖(∑ C,a C • ordinaryCycleIncidence G r C)-w‖≤
  28           8*m*Real.exp (-α^2*m/32) := by
  29   obtain ⟨N,hN⟩ := sampled_induced_fractional r hr α hα
  30   refine ⟨N,?_⟩
  31   intro m hm hm4 hroots V _ _ hmn G hdeg κ hκ hκ1 hκα w hw
  32   letI : Nonempty (FixedSubset V m) := fixedSubset_nonempty hmn
  33   let μ := ordinaryRetentionLaw G w κ hκ hκ1 hw
  34   let ν := (FiniteLaw.uniform (FixedSubset V m)).prod (FiniteLaw.pi μ)
  35   let J := sampledInducedGraph (m:=m) G
  36   let good := sampledInducedDense (m:=m) G (cycleThreshold r) α
  37   let ρ := pairSampleProbability (Fintype.card V) m
  38   let t := ρ/(1+κ)
  39   have hρ : 0<ρ := pairSampleProbability_pos (by omega) hmn
  40   have hκ' : 0<1+κ := by linarith
  41   have ht : 0<t := div_pos hρ hκ'
  42   have hsub (z : FixedSubset V m × (Sym2 V → Bool)) : J z≤G :=
  43     (spanningRestriction_le _ _).trans (retainedGraph_le _ _)
  44   have hsupp (z : FixedSubset V m × (Sym2 V → Bool)) : (J z).support⊆z.1.val :=
  45     spanningRestriction_support _ _
  46   obtain ⟨a,ha,hs,hb,he,hloss⟩ := fractional_cycle_sample_average_bounded hmn ν Prod.fst
  47     (finiteLaw_prod_map_fst _ _) G J good hsub hsupp (fun z hz => hN V m hm G z hz) t ht
  48   refine ⟨fun C => a C.val.val,?_,?_⟩
  49   · intro C
  50     exact ⟨ha _,(hb _ C.val.property).trans (cycle_sample_coefficient_scale
  51       (by omega) hmn (by omega) κ hκ hκ1)⟩
  52   · apply (pi_norm_le_iff_of_nonneg (show 0≤8*(m:ℝ)*Real.exp (-α^2*m/32) by positivity)).mpr
  53     intro e
  54     rw [Pi.sub_apply,Finset.sum_apply]
  55     simp only [Pi.smul_apply,smul_eq_mul,Real.norm_eq_abs]
  56     rw [ordinaryCycle_load_from_graph_weights G r a ha hs e]
  57     rcases e with ⟨e,heG⟩
  58     induction e using Sym2.ind with
  59     | _ u v =>
  60       have huv : G.Adj u v := heG
  61       have hpair : (({u,v}:Finset V).card)=2 := by simp [huv.ne]
  62       have hfull : ν.probability (fun z => (J z).Adj u v)/t=w ⟨s(u,v),heG⟩ := by
  63         rw [sampledInducedGraph_edge_probability hmn G μ u v huv]
  64         change ρ*ordinaryRetentionProbability G w κ s(u,v)/(ρ/(1+κ))=_
  65         rw [ordinaryRetentionProbability,dif_pos heG]
  66         field_simp
  67       have hbad : ν.probability (fun z => ¬good z ∧ (J z).Adj u v)≤
  68           4*m*Real.exp (-α^2*m/32)*ρ := by
  69         calc
  70           _ ≤ ν.probability (fun z => ({u,v}:Finset V)⊆z.1.val ∧ ¬good z) := by
  71             apply finiteLaw_probability_mono
  72             intro z hz
  73             have hz' := (spanningRestriction_adj _ _ _ _).mp hz.2
  74             exact ⟨by simpa only [Finset.insert_subset_iff,Finset.singleton_subset_iff] using ⟨hz'.2.1,hz'.2.2⟩,hz.1⟩
  75           _ ≤ 4*m*Real.exp (-α^2*m/32)*
  76               (FiniteLaw.uniform (FixedSubset V m)).probability (fun S => ({u,v}:Finset V)⊆S.val) := by
  77             exact sampled_induced_dense_failure G {u,v} m hmn (by rw [hpair];omega) μ
  78               (cycleThreshold r) α hα hdeg (by rw [hpair];norm_num;exact hroots)
  79               (ordinaryRetentionLaw_lower G w κ α hκ hκ1 hκα hw)
  80           _ = _ := by
  81             congr 1
  82             simpa only [Finset.insert_subset_iff,Finset.singleton_subset_iff] using
  83               fixedSubset_pair_probability hmn u v huv.ne
  84       have hlower : 0≤ν.probability (fun z => ¬good z ∧ (J z).Adj u v)/t :=
  85         div_nonneg (finiteLaw_probability_nonneg _ _) ht.le
  86       have hupper : ν.probability (fun z => ¬good z ∧ (J z).Adj u v)/t≤
  87           8*m*Real.exp (-α^2*m/32) := by
  88         calc
  89           _ ≤ (4*m*Real.exp (-α^2*m/32)*ρ)/t := div_le_div_of_nonneg_right hbad ht.le
  90           _ = 4*m*Real.exp (-α^2*m/32)*(1+κ) := by dsimp [t];field_simp
  91           _ ≤ _ := by
  92             have hp : 0≤4*(m:ℝ)*Real.exp (-α^2*m/32) := by positivity
  93             have hh := mul_le_mul_of_nonneg_left hκ1 hp
  94             nlinarith only [hh]
  95       have hh := hloss u v
  96       rw [hfull] at hh
  97       simp only [SimpleGraph.mem_edgeSet]
  98       apply abs_le.mpr
  99       constructor <;> linarith
