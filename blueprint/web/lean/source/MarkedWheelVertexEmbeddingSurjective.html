  21     (hkt:∀ a,ktail' (k a)=f (ktail a)) (hkh:∀ a,khead' (k a)=f (khead a))
  22     (hH₀:∀ x u,H₀ x u↔H₀' (f x) (g u)) (hHK:∀ x u,HK x u↔HK' (f x) (g u))

  24     (hhubs:∀ W:MarkedWheel tail' head' ktail' khead' m π b H₀' HK',W.2.val∈Set.range g):
  25     Function.Surjective (markedWheelVertexEmbedding (m:=m) (b:=b) f g hcomm e k het heh hkt hkh
  26       (fun x u=>(hH₀ x u).mp) (fun x u=>(hHK x u).mp)):=by
  27   intro W
  28   let C:Fin (m+1)→V:=fun i=>Classical.choose (hrows W.1 i)
  29   have hC:∀ i,f (C i)=W.1.val i:=fun i=>Classical.choose_spec (hrows W.1 i)
  30   have hCs:f∘C=W.1.val:=funext hC
  31   let c:MarkedCycleTuple tail head ktail khead m ρ b:=⟨C,
  32     (permutationSeparated_embedding_iff f ρ π hcomm b C).mp (by rw [hCs];exact W.1.property.1),
  33     by
  34       intro i
  35       obtain ⟨a,ht,hh⟩:=W.1.property.2.1 i
  36       obtain ⟨a,rfl⟩:=hes a
  37       exact ⟨a,f.injective ((het a).symm.trans (ht.trans (hC i.castSucc).symm)),
  38         f.injective ((heh a).symm.trans (hh.trans (hC i.succ).symm))⟩,
  39     by
  40       obtain ⟨a,ht,hh⟩:=W.1.property.2.2
  41       obtain ⟨a,rfl⟩:=hks a
  42       exact ⟨a,f.injective ((hkt a).symm.trans (ht.trans (hC (Fin.last m)).symm)),
  43         f.injective ((hkh a).symm.trans (hh.trans (hC 0).symm))⟩⟩
  44   obtain ⟨u,hu⟩:=hhubs W
  45   have hcHub:MarkedHubCompatible H₀ HK c.val u:=by
  46     refine ⟨(hHK _ _).mpr ?_,?_⟩
  47     · simpa only [c,hC,hu] using W.2.property.1
  48     · intro i
  49       apply (hH₀ _ _).mpr
  50       simpa only [c,hC,hu] using W.2.property.2 i
  51   refine ⟨⟨c,⟨u,hcHub⟩⟩,?_⟩
  52   have hcycle:(markedWheelVertexEmbedding (m:=m) (b:=b) f g hcomm e k het heh hkt hkh
  53       (fun x u=>(hH₀ x u).mp) (fun x u=>(hHK x u).mp) ⟨c,⟨u,hcHub⟩⟩).1=W.1:=
  54     Subtype.ext hCs
  55   apply Sigma.ext hcycle
  56   apply (Subtype.heq_iff_coe_eq (fun z=>by rw [hcycle])).mpr
  57   exact hu
