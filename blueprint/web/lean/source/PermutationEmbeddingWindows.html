  17 theorem permutation_embedding_power (n : ℕ) (x : W) : f ((ρ^n) x)=(π^n) (f x) := by
  18   induction n with
  19   | zero=>rfl
  20   | succ n ih=>simpa only [pow_succ',Equiv.Perm.mul_apply,hcomm,ih]
  21 
  22 theorem permutation_embedding_inverse_power (n : ℕ) (x : W) :
  23     f ((ρ^n).symm x)=(π^n).symm (f x) := by
  24   apply (π^n).injective
  25   rw [← permutation_embedding_power f π ρ hcomm n,Equiv.apply_symm_apply,Equiv.apply_symm_apply]

  27 theorem permutationWindow_image (l r : ℕ) (x : W) :
  28     (permutationWindow ρ l r x).image f=permutationWindow π l r (f x) := by
  29   simp only [permutationWindow,Finset.image_image]
  30   congr 1
  31   funext i
  32   simp only [Function.comp_apply,permutation_embedding_power f π ρ hcomm,
  33     permutation_embedding_inverse_power f π ρ hcomm]

  35 theorem permutationWindow_embedding_mem (hf : Function.Injective f) (l r : ℕ) (x y : W) :
  36     f y∈permutationWindow π l r (f x) ↔ y∈permutationWindow ρ l r x := by
  37   rw [← permutationWindow_image f π ρ hcomm l r x,Finset.mem_image]
  38   constructor
  39   · rintro ⟨z,hz,hzy⟩;exact hf hzy ▸ hz
