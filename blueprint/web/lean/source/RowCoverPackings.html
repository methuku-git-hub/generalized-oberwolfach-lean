  14 structure OrdinaryRowCover (M:Finset (OrdinaryWheel Q F H r)) (x:V) (A:Finset V) (K:ℕ) : Prop where
  15   disjoint : ∀ W∈M,∀ Z∈M,W≠Z → Disjoint (ordinaryWheelResources W) (ordinaryWheelResources Z)
  16   support : ∀ W∈M,∀ a∈(ordinaryWheelRim W).support,a=x∨a∈A
  17   rims : ∀ a,(ordinaryWheelRimGraph M).Adj x a ↔ Q.Adj x a
  18   spokes : ∀ u,ordinaryPackingSpokes M x u ↔ H x u

  23 theorem ordinary_family_image_rim (C:I→OrdinaryWheel Q F H r) :
  24     ordinaryWheelRimGraph (Finset.univ.image C)=wheelFamilyRim C := by
  25   ext a b
  26   simp only [ordinaryWheelRimGraph,wheelFamilyRim,SimpleGraph.iSup_adj,Finset.mem_image,Finset.mem_univ,true_and]
  27   constructor
  28   · rintro ⟨W,⟨i,rfl⟩,hab⟩;exact ⟨i,hab⟩
  29   · rintro ⟨i,hab⟩;exact ⟨C i,⟨i,rfl⟩,hab⟩

  40 theorem ordinaryRowCover_of_family (C:I→OrdinaryWheel Q F H r) (x:V) (A:Finset V) (K:ℕ)
  41     (hsupport:∀ i,∀ a∈(ordinaryWheelRim (C i)).support,a=x∨a∈A)
  42     (hdis:∀ i j,i≠j → Disjoint (ordinaryWheelResources (C i)) (ordinaryWheelResources (C j)))
  43     (hrim:∀ a,(wheelFamilyRim C).Adj x a ↔ Q.Adj x a)
  44     (hspoke:∀ u,wheelFamilySpokes C x u ↔ H x u)
  45     (hdeg:∀ a,a≠x → (wheelFamilyRim C).degree a≤2*(K+1))
  46     (hrow:∀ a,a≠x → (Finset.univ.filter (fun u=>wheelFamilySpokes C a u)).card≤K+1)
  47     (hcol:∀ u,(Finset.univ.filter (fun a=>wheelFamilySpokes C a u)).card≤r) :
  48     OrdinaryRowCover (Finset.univ.image C) x A K := by
  49   refine ⟨?_,?_,?_,?_,?_,?_,?_⟩
  50   · intro W hW Z hZ hne
  51     obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hW
  52     obtain ⟨j,_,rfl⟩:=Finset.mem_image.mp hZ
  53     exact hdis i j (fun he=>hne (congrArg C he))
  54   · intro W hW a ha
  55     obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hW
  56     exact hsupport i a ha
  57   · simpa only [ordinary_family_image_rim] using hrim
  58   · simpa only [ordinary_family_image_spokes] using hspoke
  59   · intro a ha
  60     have he:=ordinary_family_image_rim C
  61     have hh:=hdeg a ha
  62     convert hh using 1 <;> rw [he]
  63   · simpa only [ordinary_family_image_spokes] using hrow
  64   · simpa only [ordinary_family_image_spokes] using hcol

  73 theorem relation_iUnion_degree_bound {X J:Type} [Fintype X] [Fintype J]
  74     (P:J→X→Prop) (D:ℕ) (hP:∀ j,(Finset.univ.filter (P j)).card≤D) :
  75     (Finset.univ.filter (fun x=>∃ j,P j x)).card≤Fintype.card J*D := by
  76   classical
  77   have he:Finset.univ.filter (fun x=>∃ j,P j x)=Finset.univ.biUnion (fun j=>Finset.univ.filter (P j)):=by
  78     ext x;simp
  79   rw [he]
  80   calc
  81     _ ≤ ∑ j,(Finset.univ.filter (P j)).card:=Finset.card_biUnion_le
  82     _ ≤ ∑ _j:J,D:=Finset.sum_le_sum (fun j _=>hP j)
  83     _ = _:=by simp
