  11 theorem wheel_decomposition_conflict_recovery {V U:Type} [Fintype V] [DecidableEq V]
  12     [Fintype U] [DecidableEq U] (Q F R:SimpleGraph V) (H S:V→U→Prop) (T:Finset V) (r:ℕ)
  13     (hR:R≤Q) (hS:∀ v u,S v u→H v u)
  14     (hsupp:∀ v w,R.Adj v w→v∈T∧w∈T) (hfree:∀ v∈T,∀ w∈T,¬F.Adj v w)
  15     (hdec:HasOrdinaryWheelDecomposition R S r):
  16     HasResourceDecomposition (ordinaryWheelResources (P:=Q) (F:=F) (Hbase:=H) (r:=r))
  17       (ordinaryHostResources R S):=by
  18   obtain ⟨M,hM,hcover⟩:=hdec
  19   apply ordinary_decomposition_recover_host Q F H (ordinaryHostResources R S)
  20     (ordinary_host_resources_monotone R Q S H hR hS) M ⟨hM,hcover⟩
  21   intro W hW v hv w hw
  22   have hsub:ordinaryWheelResources W⊆ordinaryHostResources R S:=by
  23     intro e he
  24     rw [←hcover]
  25     exact Finset.mem_biUnion.mpr ⟨W,hW,he⟩
  26   have hrim:(ordinaryWheelRim W)≤R:=((ordinaryWheelResources_subset_host W R S).mp hsub).1
  27   have hvR:v∈R.support:=SimpleGraph.support_mono hrim hv
  28   have hwR:w∈R.support:=SimpleGraph.support_mono hrim hw
  29   obtain ⟨v',hv'⟩:=R.mem_support.mp hvR
  30   obtain ⟨w',hw'⟩:=R.mem_support.mp hwR
  31   exact hfree v (hsupp v v' hv').1 w (hsupp w w' hw').1
