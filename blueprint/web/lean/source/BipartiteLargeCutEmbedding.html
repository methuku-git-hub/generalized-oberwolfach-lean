  31 theorem bipartiteLargeCut_embedding (f:A↪X) (g:B↪Y) (L:X→Y→ℕ)
  32     (hA:0<Fintype.card A) (hB:0<Fintype.card B)
  33     (κ δ δA h:ℝ) (hκ:0<κ) (hh:0≤h)
  34     (hAsize:κ*Fintype.card X≤(Fintype.card A:ℝ))
  35     (hBsize:κ*Fintype.card Y≤(Fintype.card B:ℝ))
  36     (hcutoff:1-2*δ≤κ*(1-2*δA)) (hcut:BipartiteLargeCut L δ h):
  37     BipartiteLargeCut (fun x y=>L (f x) (g y)) δA h:=by
  38   have hAX:Fintype.card A≤Fintype.card X:=Fintype.card_le_of_injective f f.injective
  39   have hBY:Fintype.card B≤Fintype.card Y:=Fintype.card_le_of_injective g g.injective
  40   have hX:0<Fintype.card X:=lt_of_lt_of_le hA hAX
  41   have hY:0<Fintype.card Y:=lt_of_lt_of_le hB hBY
  42   have ha:(0:ℝ)<Fintype.card A:=by exact_mod_cast hA
  43   have hb:(0:ℝ)<Fintype.card B:=by exact_mod_cast hB
  44   have hx:(0:ℝ)<Fintype.card X:=by exact_mod_cast hX
  45   have hy:(0:ℝ)<Fintype.card Y:=by exact_mod_cast hY
  46   intro S T hST
  47   have hSratio:κ*((S.card:ℝ)/Fintype.card A)≤(S.card:ℝ)/Fintype.card X:=by
  48     rw [←mul_div_assoc]
  49     apply (div_le_div_iff₀ ha hx).mpr
  50     nlinarith only [mul_le_mul_of_nonneg_left hAsize (Nat.cast_nonneg (α:=ℝ) S.card)]
  51   have hTratio:κ*((T.card:ℝ)/Fintype.card B)≤(T.card:ℝ)/Fintype.card Y:=by
  52     rw [←mul_div_assoc]
  53     apply (div_le_div_iff₀ hb hy).mpr
  54     nlinarith only [mul_le_mul_of_nonneg_left hBsize (Nat.cast_nonneg (α:=ℝ) T.card)]
  55   have hlarge:1-2*δ<((S.image f).card:ℝ)/Fintype.card X+((T.image g).card:ℝ)/Fintype.card Y:=by
  56     rw [Finset.card_image_of_injective _ f.injective,Finset.card_image_of_injective _ g.injective]
  57     have hscaled:=mul_lt_mul_of_pos_left hST hκ
  58     nlinarith only [hscaled,hcutoff,hSratio,hTratio]
  59   have hparent:=hcut (S.image f) (T.image g) hlarge
  60   rw [bipartiteLargeCut_sum_restrict,bipartiteMatrixCut_embedding,
  61     bipartite_normalized_cut_scale hX hY (S.image f) (T.image g) h,
  62     Finset.card_image_of_injective _ f.injective,Finset.card_image_of_injective _ g.injective] at hparent
  63   rw [bipartiteLargeCut_sum_restrict,bipartite_normalized_cut_scale hA hB S T h]
  64   apply le_trans _ hparent
