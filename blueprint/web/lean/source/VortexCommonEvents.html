  15 theorem vortex_inverse_sum_bound (ε M : ℝ) (hε : 0<ε) (hε1 : ε<1) (hM : 0<M)
  16     (n : ℕ→ℝ) (k : ℕ) (hn : ∀ i≤k,0<n i)
  17     (hstep : ∀ i<k,n (i+1)≤ε*n i) (hlast : M≤n k) :
  18     ∑ i∈Finset.range (k+1),1/n i≤1/((1-ε)*M) := by
  19   have hh := geometric_chain_sum_bound hε.le hε1 (fun i=>1/n i) k
  20     (fun i hi=>by have := hn i hi;positivity) (by
  21       intro i hi
  22       have hi0 := hn i (by omega)
  23       have hi1 := hn (i+1) (by omega)
  24       apply (div_le_iff₀ hi0).mpr
  25       have hs : 1≤ε*n i/n (i+1) := (le_div_iff₀ hi1).mpr (by simpa using hstep i hi)
  26       nlinarith only [hs,show ε*n i/n (i+1)=(ε*(1/n (i+1)))*n i by ring])
  27   calc
  28     _ ≤ (1/n k)/(1-ε) := hh
  29     _ ≤ (1/M)/(1-ε) := div_le_div_of_nonneg_right
  30       (one_div_le_one_div_of_le hM hlast) (by linarith)
  31     _ = _ := by rw [div_div];congr 1;ring

  33 theorem vortex_fixed_domain_common_event {Ω : Type*} [Fintype Ω]
  34     (μ : FiniteLaw Ω) (ε M : ℝ) (hε : 0<ε) (hε1 : ε<1) (hM : 0<M)
  35     (n : ℕ→ℝ) (k : ℕ) (hn : ∀ i≤k,0<n i)
  36     (hstep : ∀ i<k,n (i+1)≤ε*n i) (hlast : M≤n k)
  37     (Good : ℕ→Ω→Prop) (hGood : ∀ i≤k,1-1/n i≤μ.probability (Good i)) :
  38     1-1/((1-ε)*M)≤μ.probability (fun ω=>∀ i≤k,Good i ω) := by
  39   have hh := finiteLaw_common_event_lower μ (Finset.range (k+1)) (fun i ω=>¬Good i ω)
  40     (fun i=>1/n i) (by
  41       intro i hi
  42       have h := hGood i (by simpa only [Finset.mem_range,Nat.lt_succ_iff] using hi)
  43       have hc := finiteLaw_probability_compl μ (Good i)
  44       linarith)
  45   have he : (fun ω=>∀ i∈Finset.range (k+1),¬¬Good i ω)=(fun ω=>∀ i≤k,Good i ω) := by
  46     funext ω
  47     apply propext
  48     simp only [Finset.mem_range,not_not,Nat.lt_succ_iff]
