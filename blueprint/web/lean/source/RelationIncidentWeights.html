  14 theorem relation_separable_weight_bound {A B : Type*} [Fintype A] [Fintype B]
  15     (D : A → B → Prop) (d : ℝ)
  16     (hout : ∀ x,(∑ y,bipAdj D x y)≤d) (hin : ∀ y,(∑ x,bipAdj D x y)≤d)
  17     (a : A → ℝ) (b : B → ℝ) (c : ℝ)
  18     (ha : ∀ x,0≤a x) (hb : ∀ y,0≤b y) (hc : 0≤c) :
  19     (∑ x,∑ y,bipAdj D x y*(a x+b y+c))≤
  20       d*((∑ x,a x)+(∑ y,b y)+(Fintype.card A:ℝ)*c) := by
  21   have h1 : (∑ x,∑ y,bipAdj D x y*a x)≤d*∑ x,a x := by
  22     simp_rw [← Finset.sum_mul]
  23     calc
  24       _ ≤ ∑ x,d*a x := Finset.sum_le_sum (fun x _=>mul_le_mul_of_nonneg_right (hout x) (ha x))
  25       _ = _ := (Finset.mul_sum _ _ _).symm
  26   have h2 : (∑ x,∑ y,bipAdj D x y*b y)≤d*∑ y,b y := by
  27     rw [Finset.sum_comm]
  28     simp_rw [← Finset.sum_mul]
  29     calc
  30       _ ≤ ∑ y,d*b y := Finset.sum_le_sum (fun y _=>mul_le_mul_of_nonneg_right (hin y) (hb y))
  31       _ = _ := (Finset.mul_sum _ _ _).symm
  32   have h3 : (∑ x,∑ y,bipAdj D x y*c)≤(Fintype.card A:ℝ)*d*c := by
  33     calc
  34       _ = ∑ x,(∑ y,bipAdj D x y)*c := by
  35         apply Finset.sum_congr rfl
  36         intro x _
  37         rw [Finset.sum_mul]
  38       _ ≤ ∑ _x : A,d*c := Finset.sum_le_sum (fun x _=>mul_le_mul_of_nonneg_right (hout x) hc)
  39       _ = _ := by simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul];ring
  40   simp only [mul_add,Finset.sum_add_distrib]
  41   nlinarith

  43 theorem relation_root_incident_cost {A B : Type*} [Fintype A] [Fintype B]
  44     [DecidableEq A] [DecidableEq B]
  45     (D : A → B → Prop) (d : ℝ)
  46     (hout : ∀ x,(∑ y,bipAdj D x y)≤d) (hin : ∀ y,(∑ x,bipAdj D x y)≤d)
  47     (S : Finset A) (T : Finset B) (M N : ℝ) (hM : 0≤M) (hN : 0≤N) :
  48     (∑ x,∑ y,if D x y then (if x∈S ∨ y∈T then M else N) else 0)≤
  49       d*((S.card+T.card:ℝ)*M+(Fintype.card A:ℝ)*N) := by
  50   let a (x : A) : ℝ := if x∈S then M else 0
  51   let b (y : B) : ℝ := if y∈T then M else 0
  52   have hb := relation_separable_weight_bound D d hout hin a b N
  53     (fun x=>by dsimp only [a];split_ifs <;> positivity)
  54     (fun y=>by dsimp only [b];split_ifs <;> positivity) hN
  55   have hs : (∑ x,a x)=(S.card:ℝ)*M := by
  56     dsimp only [a]
  57     rw [← Finset.sum_filter]
  58     have he : Finset.univ.filter (fun x=>x∈S)=S := by ext;simp
  59     rw [he,Finset.sum_const,nsmul_eq_mul]
  60   have ht : (∑ y,b y)=(T.card:ℝ)*M := by
  61     dsimp only [b]
  62     rw [← Finset.sum_filter]
  63     have he : Finset.univ.filter (fun y=>y∈T)=T := by ext;simp
  64     rw [he,Finset.sum_const,nsmul_eq_mul]
  65   rw [hs,ht] at hb
  66   calc
  67     _ ≤ ∑ x,∑ y,bipAdj D x y*(a x+b y+N) := by
  68       apply Finset.sum_le_sum
  69       intro x _
  70       apply Finset.sum_le_sum
  71       intro y _
  72       dsimp only [a,b,bipAdj]
  73       split_ifs <;> simp_all <;> linarith
  74     _ ≤ _ := by convert hb using 1 <;> ring
