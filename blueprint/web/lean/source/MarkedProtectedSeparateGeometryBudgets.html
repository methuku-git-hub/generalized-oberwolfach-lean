   9 theorem marked_protected_inner_error_budget_separate_geometry
  10     (r h:ℕ) (rho omega t epsilon etaEnd a x n ambient nW nZ:ℝ)
  11     (hrho:0<rho) (ho:0≤ omega) (ht:0≤ t) (he:0≤ epsilon) (hEnd:0≤ etaEnd) (ha:0≤ a)
  12     (hx:0≤ x) (hn:0≤ n) (hambient:n/2≤ ambient)
  13     (hW:omega*n/2≤ nW) (hZ:omega*n/2≤ nZ)
  14     (hExt:(1+h*(24+3*(r+5:ℝ)))*x≤ epsilon*(omega/2))
  15     (hEndpoint:2*(1+h*(24+3*(r+5:ℝ)))*x≤ etaEnd*a/2)
  16     (hGap:(3+2/rho+h*(24+3*(r+5:ℝ)))*x≤ t*omega/16)
  17     (xiDeep:ℝ) (hDeepGap:(1+2/rho)*xiDeep≤ t*omega/32)
  18     (etaE errR0 errI0 errParent removedRows removedHubs exceptional:ℝ)
  19     (hEta:0≤ etaE) (hEtaMax:etaE≤ x)
  20     (hR:errR0≤ x*n) (hI:errI0≤ x*n) (hP:errParent≤ x*n)
  21     (hRows:removedRows≤ xiDeep*n) (hHubs:removedHubs≤ xiDeep*n)
  22     (hException:exceptional≤ t*omega*n/32):
  23     let errR:=errR0+h*(24*etaE*n+3*(r+5:ℝ)*etaE*n)
  24     let errI:=errI0+h*(6*etaE*n)
  25     let deltaParent:=errParent+removedRows+2*removedHubs/rho;
  26     errR≤ epsilon*nW∧ errI≤ epsilon*nZ∧ 
  27     2*errR≤ etaEnd*(a*ambient)∧ 2*errI≤ etaEnd*(a*ambient)∧ 
  28     errR+exceptional+deltaParent≤ t*omega*ambient/4∧ 
  29     errI≤ t*omega*ambient/4:=by
  30   intro errR errI deltaParent
  31   let C:ℝ:=1+h*(24+3*(r+5:ℝ))
  32   have hhn:0≤ (h:ℝ):=Nat.cast_nonneg _
  33   have hrn:0≤ (r:ℝ):=Nat.cast_nonneg _
  34   have hEn:=mul_le_mul_of_nonneg_right hEtaMax hn
  35   have hcoef:0≤ h*(24+3*(r+5:ℝ)):=by positivity
  36   have hmul:=mul_le_mul_of_nonneg_left hEn hcoef
  37   have hER:errR≤ C*x*n:=by dsimp [errR,C];nlinarith only [hR,hmul]
  38   have h6coef:6*(h:ℝ)≤ h*(24+3*(r+5:ℝ)):=by
  39     nlinarith only [hhn,mul_nonneg hhn hrn]
  40   have hmulI:=mul_le_mul_of_nonneg_left hEn (show 0≤ 6*(h:ℝ) by positivity)
  41   have hcompare:=mul_le_mul_of_nonneg_right h6coef (show 0≤ x*n by positivity)
  42   have hEI:errI≤ C*x*n:=by dsimp [errI,C];nlinarith only [hI,hmulI,hcompare]
  43   have hLoss:=mul_le_mul_of_nonneg_left hHubs (show 0≤ 2/rho by positivity)
  44   have hDP:deltaParent≤ x*n+(1+2/rho)*xiDeep*n:=by
  45     calc
  46       deltaParent=errParent+removedRows+(2/rho)*removedHubs:=by dsimp [deltaParent];ring
  47       _ ≤  x*n+xiDeep*n+(2/rho)*(xiDeep*n):=add_le_add (add_le_add hP hRows) hLoss
  48       _ = x*n+(1+2/rho)*xiDeep*n:=by ring
  49   have hExtn:=mul_le_mul_of_nonneg_right hExt hn
  50   have hEndn:=mul_le_mul_of_nonneg_right hEndpoint hn
  51   have hDeepGapn:=mul_le_mul_of_nonneg_right hDeepGap hn
  52   have hGapn:=mul_le_mul_of_nonneg_right hGap hn
  53   have hW':=mul_le_mul_of_nonneg_left hW he
  54   have hZ':=mul_le_mul_of_nonneg_left hZ he
  55   have hA':=mul_le_mul_of_nonneg_left hambient (show 0≤ etaEnd*a by positivity)
  56   have hG':=mul_le_mul_of_nonneg_left hambient (show 0≤ t*omega by positivity)
  57   have hPos:0≤ (1+2/rho)*x*n:=by positivity
  58   change C*x*n≤ epsilon*(omega/2)*n at hExtn
  59   change 2*C*x*n≤ (etaEnd*a/2)*n at hEndn
  60   have hGapn':(C+2+2/rho)*x*n≤ t*omega*n/16:=by
  61     dsimp [C];nlinarith only [hGapn]
  62   refine ⟨by nlinarith only [hER,hExtn,hW'],by nlinarith only [hEI,hExtn,hZ'],
  63     by nlinarith only [hER,hEndn,hA'],by nlinarith only [hEI,hEndn,hA'],
  64     by nlinarith only [hER,hDP,hException,hGapn',hG',hDeepGapn,hPos],?_⟩
  65   have htw:0≤ t*omega*n:=by positivity
  66   have hxn:0≤ x*n:=mul_nonneg hx hn
  67   nlinarith only [hEI,hGapn',hG',hPos,hxn,htw]
