  14 theorem exists_separated_row_set (F : SimpleGraph V) (A W R : Finset V)
  15     (d : ℝ) (hd : 0≤d) (hF : ∀ v,(F.degree v:ℝ)≤d) (n : ℕ)
  16     (hbudget : W.card+((R.card:ℝ)+n)*(d+1)<A.card) :
  17     ∃ S:Finset V,S.card=n ∧ S⊆A ∧ Disjoint S W ∧ Disjoint S R ∧
  18       (∀ v∈S,∀ w∈S,¬F.Adj v w) ∧ (∀ v∈S,∀ w∈R,¬F.Adj v w) := by
  19   induction n with
  20   | zero => exact ⟨∅,rfl,Finset.empty_subset _,by simp,by simp,by simp,by simp⟩
  21   | succ n ih =>
  22     have hd1 : 0≤d+1 := by linarith
  23     obtain ⟨S,hSc,hSA,hSW,hSR,hSS,hSF⟩ := ih (by push_cast at hbudget;nlinarith only [hbudget,hd1])
  24     let Z := R∪S
  25     let D := W∪(Z∪Z.biUnion (fun z=>F.neighborFinset z))
  26     have hZ : (Z.card:ℝ)≤R.card+n := by
  27       exact_mod_cast (Finset.card_union_le R S).trans_eq (by rw [hSc])
  28     have hD : (D.card:ℝ)≤W.card+((R.card:ℝ)+n)*(d+1) := by
  29       have hn := forbidden_used_neighbours_card F Z d hF
  30       have hc : (D.card:ℝ)≤W.card+(Z∪Z.biUnion (fun z=>F.neighborFinset z)).card := by
  31         exact_mod_cast Finset.card_union_le W (Z∪Z.biUnion (fun z=>F.neighborFinset z))
  32       have hm := mul_le_mul_of_nonneg_right hZ hd1
  33       linarith only [hn,hc,hm]
  34     have hDA : D.card<A.card := by
  35       have hh : (D.card:ℝ)<A.card := by push_cast at hbudget;nlinarith only [hbudget,hD,hd1]
  36       exact_mod_cast hh
  37     obtain ⟨v,hvA,hvD⟩ := Finset.exists_mem_notMem_of_card_lt_card hDA
  38     have hout : v∉W ∧ v∉Z ∧ v∉Z.biUnion (fun z=>F.neighborFinset z) := by
  39       simpa only [D,Finset.mem_union,not_or] using hvD
  40     have hvS : v∉S := fun h=>hout.2.1 (Finset.mem_union_right _ h)
  41     have hvR : v∉R := fun h=>hout.2.1 (Finset.mem_union_left _ h)
  42     have hvF : ∀ w∈Z,¬F.Adj v w := by
  43       intro w hw he
  44       exact hout.2.2 (Finset.mem_biUnion.mpr ⟨w,hw,(F.mem_neighborFinset _ _).mpr he.symm⟩)
  45     refine ⟨insert v S,by simp [hvS,hSc],Finset.insert_subset hvA hSA,
  46       Finset.disjoint_insert_left.mpr ⟨hout.1,hSW⟩,
  47       Finset.disjoint_insert_left.mpr ⟨hvR,hSR⟩,?_,?_⟩
  48     · intro a ha b hb
  49       simp only [Finset.mem_insert] at ha hb
  50       rcases ha with rfl|ha <;> rcases hb with rfl|hb
  51       · exact fun h=>h.ne rfl
  52       · exact hvF b (Finset.mem_union_right _ hb)
  53       · exact fun h=>hvF a (Finset.mem_union_right _ ha) h.symm
  54       · exact hSS a ha b hb
  55     · intro a ha b hb
  56       rcases Finset.mem_insert.mp ha with rfl|ha
  57       · exact hvF b (Finset.mem_union_left _ hb)
  58       · exact hSF a ha b hb

  60 theorem exists_separated_row_embedding (I : Type) [Fintype I]
  61     (F : SimpleGraph V) (A W R : Finset V) (d : ℝ) (hd : 0≤d)
  62     (hF : ∀ v,(F.degree v:ℝ)≤d)
  63     (hbudget : W.card+((R.card:ℝ)+Fintype.card I)*(d+1)<A.card) :
  64     ∃ f:I↪V,(∀ i,f i∈A ∧ f i∉W ∧ f i∉R) ∧
  65       (∀ i j,¬F.Adj (f i) (f j)) ∧ (∀ i v,v∈R→¬F.Adj (f i) v) := by
  66   obtain ⟨S,hS,hSA,hSW,hSR,hSS,hSF⟩ := exists_separated_row_set F A W R d hd hF _ hbudget
  67   let e : I≃S := Fintype.equivOfCardEq (by simpa only [Fintype.card_coe] using hS.symm)
  68   let f : I↪V := e.toEmbedding.trans (Function.Embedding.subtype _)
  69   exact ⟨f,(fun i=>⟨hSA (e i).property,fun h=>Finset.disjoint_left.mp hSW (e i).property h,
  70     fun h=>Finset.disjoint_left.mp hSR (e i).property h⟩),
  71     (fun i j=>hSS _ (e i).property _ (e j).property),fun i v hv=>hSF _ (e i).property v hv⟩

  73 theorem exists_fresh_hub_embedding (I : Type) [Fintype I] (B : Finset U)
  74     (hB : B.card+Fintype.card I≤Fintype.card U) :
  75     ∃ f:I↪U,∀ i,f i∉B := by
  76   let S := (Finset.univ:Finset U)\B
  77   have hS : Fintype.card I≤Fintype.card S := by
  78     have hc : S.card+B.card=Fintype.card U := by
  79       simpa only [S,Finset.card_univ] using Finset.card_sdiff_add_card_eq_card (Finset.subset_univ B)
  80     simpa only [Fintype.card_coe] using (show Fintype.card I≤S.card by omega)
  81   obtain ⟨f⟩ : Nonempty (I↪S) := Function.Embedding.nonempty_of_card_le hS
  82   exact ⟨f.trans (Function.Embedding.subtype _),fun i=>(Finset.mem_sdiff.mp (f i).property).2⟩
