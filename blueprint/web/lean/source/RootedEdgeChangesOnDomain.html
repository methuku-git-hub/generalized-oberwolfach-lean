  16 theorem rooted_edge_change_count_on_domain
  17     (e:X→V) (D:V→V→Prop) (delta:ℝ) (σ τ:Equiv.Perm V)
  18     (hout:∀ x,(∑ y:X,if D x (τ (e y)) then (1:ℝ) else 0)≤delta)
  19     (hin:∀ y,(∑ x:X,if D (σ (e x)) y then (1:ℝ) else 0)≤delta)
  20     (a b:V⊕I) (hab:RootedEdgeHasFreeEndpoint a b):
  21     (∑ f:I→X,if D (σ (rootedPositionValue (fun i=>e (f i)) a))
  22       (τ (rootedPositionValue (fun i=>e (f i)) b)) then (1:ℝ) else 0)≤
  23       (Fintype.card X:ℝ)^(Fintype.card I-1)*delta:=by
  24   cases a with
  25   | inl x=>
  26     cases b with
  27     | inl y=>exact hab.elim
  28     | inr j=>
  29       apply sum_bound_from_one_coordinate j
  30       intro g
  31       simp only [rootedPositionValue,Sum.elim_inl,Sum.elim_inr,id_eq,
  32         fillFreeCoordinate,dif_pos rfl,dite_true]
  33       exact hout (σ x)
  34   | inr i=>
  35     cases b with
  36     | inl y=>
  37       apply sum_bound_from_one_coordinate i
  38       intro g
  39       simp only [rootedPositionValue,Sum.elim_inl,Sum.elim_inr,id_eq,
  40         fillFreeCoordinate,dif_pos rfl,dite_true]
  41       exact hin (τ y)
  42     | inr j=>
  43       have hne:i≠j:=hab
  44       apply sum_bound_from_one_coordinate j
  45       intro g
  46       simp only [rootedPositionValue,Sum.elim_inr,fillFreeCoordinate,dif_pos rfl,dif_neg hne,dite_true]
  47       exact hout (σ (e (g ⟨i,hne⟩)))

  51 theorem fixedDomainTuple_edge_loss {J:Type*} [Fintype J]
  52     (A:Finset V) (P Q:(I→V)→Prop)
  53     (hP:∀ f,P f→∀ i,f i∈A) (hQ:∀ f,Q f→∀ i,f i∈A)
  54     (D:V→V→Prop) (delta:ℝ) (σ τ:J→Equiv.Perm V) (src dst:J→V⊕I)
  55     (hfree:∀ j,RootedEdgeHasFreeEndpoint (src j) (dst j))
  56     (hout:∀ j x,(∑ y:A,if D x (τ j y.val) then (1:ℝ) else 0)≤delta)
  57     (hin:∀ j y,(∑ x:A,if D (σ j x.val) y then (1:ℝ) else 0)≤delta)
  58     (hbad:∀ f,P f→¬Q f→∃ j,D (σ j (rootedPositionValue f (src j)))
  59       (τ j (rootedPositionValue f (dst j)))):
  60     (Fintype.card {f:I→V // P f}:ℝ)-Fintype.card {f:I→V // Q f}≤
  61       (Fintype.card J:ℝ)*delta*(A.card:ℝ)^(Fintype.card I-1):=by
  62   rw [Fintype.card_congr (fixedDomainTuplePredicateEquiv A P hP),
  63     Fintype.card_congr (fixedDomainTuplePredicateEquiv A Q hQ)]
  64   have hh:=finite_predicate_loss_union
  65     (fun f:I→A=>P (fun i=>(f i).val)) (fun f:I→A=>Q (fun i=>(f i).val))
  66     (fun j (f:I→A)=>D (σ j (rootedPositionValue (fun i=>(f i).val) (src j)))
  67       (τ j (rootedPositionValue (fun i=>(f i).val) (dst j))))
  68     (by intro f hp hq;exact hbad _ hp hq)
  69   apply hh.trans
  70   calc
  71     _ ≤∑ _j:J,(Fintype.card A:ℝ)^(Fintype.card I-1)*delta:=by
  72       apply Finset.sum_le_sum
  73       intro j _
  74       exact rooted_edge_change_count_on_domain (Subtype.val:A→V) D delta (σ j) (τ j)
  75         (hout j) (hin j) (src j) (dst j) (hfree j)
  76     _ = _:=by simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul,Fintype.card_coe];ring
