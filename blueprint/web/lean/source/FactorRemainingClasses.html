   9 theorem factor_remaining_class_not_small {W:Type} [Fintype W] (K:ℕ)
  10     (classify:W→Option (Fin K)) (n κL κS:ℝ):
  11     let fibres:=fun j:Option (Fin K)=>Finset.univ.filter (fun w=>classify w=j)
  12     let cutoff:=fun j:Option (Fin K)=>if j=none then κL else κS
  13     let small:=fun j:Option (Fin K)=>((fibres j).card:ℝ)<cutoff j*n
  14     let removed:=Finset.univ.filter (fun w=>small (classify w));
  15     ∀ j,(∃ w:{w:W // w∉removed},classify w.val=j)→¬small j:=by
  16   intro fibres cutoff small removed j ⟨w,hw⟩ hj
  17   apply w.property
  18   exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,by simpa [hw] using hj⟩
  19 end Oberwolfach
