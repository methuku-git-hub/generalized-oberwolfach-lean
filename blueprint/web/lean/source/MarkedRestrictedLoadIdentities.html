  35 theorem markedPotentialLoad_parent
  36     (w : MarkedWheel tail head ktail khead m π b H₀ Hₖ → ℝ) (e : (E⊕K)⊕(V×U)) :
  37     markedPotentialLoad w (fun _=>True) (fun v u=>H₀ v u ∨ Hₖ v u) e=
  38       ∑ W,if Sum.map (markedPhysicalEdge tail head ktail khead π) id e∈markedWheelPhysicalResources W
  39         then w W else 0 := by
  40   unfold markedPotentialLoad resourceRootOmittedLoad
  41   apply Finset.sum_congr rfl
  42   intro W _
  43   have hs : ∀ f∈markedOccurrenceResources W,
  44       Sum.elim (fun _ : E⊕K=>True) (fun vu=>H₀ vu.1 vu.2 ∨ Hₖ vu.1 vu.2) f := by
  45     intro f hf
  46     cases f with
  47     | inl a => trivial
  48     | inr vu =>
  49       have hh := (markedOccurrenceResources_physical W (Sum.inr vu)).mp hf
  50       exact markedWheelPhysicalResources_spoke_present W vu.1 vu.2 hh
  51   have hs' : ∀ f∈markedOccurrenceResources W,f≠e →
  52       Sum.elim (fun _ : E⊕K=>True) (fun vu=>H₀ vu.1 vu.2 ∨ Hₖ vu.1 vu.2) f :=
  53     fun f hf _=>hs f hf
  54   by_cases he : e∈markedOccurrenceResources W
  55   · rw [if_pos ⟨he,hs'⟩,if_pos ((markedOccurrenceResources_physical W e).mp he)]
  56   · rw [if_neg (fun h=>he h.1),if_neg (fun h=>he ((markedOccurrenceResources_physical W e).mpr h))]
