  13 theorem complete_on_adj (S:Finset V) (x y:V):
  14     (spanningRestriction (⊤:SimpleGraph V) S).Adj x y↔x∈S ∧ y∈S ∧ x≠y:=by
  15   rw [spanningRestriction_adj,SimpleGraph.top_adj];tauto

  17 theorem root_completed_cliques_disjoint (A S T:Finset V) (x y:V)
  18     (hSA:S⊆A) (hTA:T⊆A) (hx:x∉A) (hy:y∉A)
  19     (hinner:Disjoint (spanningRestriction (⊤:SimpleGraph V) S) (spanningRestriction ⊤ T))
  20     (hsame:x=y→Disjoint S T):
  21     Disjoint (spanningRestriction (⊤:SimpleGraph V) (insert x S))
  22       (spanningRestriction ⊤ (insert y T)):=by
  23   apply disjoint_iff_inf_le.mpr
  24   intro v w h
  25   obtain ⟨hv,hw,hvw⟩:=(complete_on_adj (insert x S) v w).mp h.1
  26   obtain ⟨hv',hw',_⟩:=(complete_on_adj (insert y T) v w).mp h.2
  27   simp only [Finset.mem_insert] at hv hw hv' hw'
  28   by_cases hva:v∈A
  29   · have hvs:v∈S:=hv.resolve_left (fun h=>hx (h ▸ hva))
  30     have hvt:v∈T:=hv'.resolve_left (fun h=>hy (h ▸ hva))
  31     by_cases hwa:w∈A
  32     · have hws:w∈S:=hw.resolve_left (fun h=>hx (h ▸ hwa))
  33       have hwt:w∈T:=hw'.resolve_left (fun h=>hy (h ▸ hwa))
  34       have hb:(⊥:SimpleGraph V).Adj v w:=(disjoint_iff_inf_le.mp hinner)
  35         ⟨(complete_on_adj S v w).mpr ⟨hvs,hws,hvw⟩,(complete_on_adj T v w).mpr ⟨hvt,hwt,hvw⟩⟩
  36       exact hb
  37     · have hwx:w=x:=hw.resolve_right (fun h=>hwa (hSA h))
  38       have hwy:w=y:=hw'.resolve_right (fun h=>hwa (hTA h))
  39       exact (Finset.disjoint_left.mp (hsame (hwx.symm.trans hwy)) hvs hvt).elim
  40   · have hvx:v=x:=hv.resolve_right (fun h=>hva (hSA h))
  41     have hvy:v=y:=hv'.resolve_right (fun h=>hva (hTA h))
  42     have hws:w∈S:=hw.resolve_left (fun h=>hvw (hvx.trans h.symm))
  43     have hwt:w∈T:=hw'.resolve_left (fun h=>hvw (hvy.trans h.symm))
  44     exact (Finset.disjoint_left.mp (hsame (hvx.symm.trans hvy)) hws hwt).elim

  46 theorem root_completed_cliques_inner (A:Finset V) (x:I→V) (T:I→Finset V)
  47     (hx:∀ i,x i∉A) (hT:∀ i,T i⊆A):
  48     spanningRestriction (⨆ i,spanningRestriction (⊤:SimpleGraph V) (insert (x i) (T i))) A=
  49       ⨆ i,spanningRestriction (⊤:SimpleGraph V) (T i):=by
  50   ext v w
  51   rw [spanningRestriction_adj,SimpleGraph.iSup_adj,SimpleGraph.iSup_adj]
  52   constructor
  53   · rintro ⟨⟨i,hi⟩,hv,hw⟩
  54     obtain ⟨hvi,hwi,hne⟩:=(complete_on_adj _ v w).mp hi
  55     simp only [Finset.mem_insert] at hvi hwi
  56     exact ⟨i,(complete_on_adj _ v w).mpr
  57       ⟨hvi.resolve_left (fun h=>hx i (h ▸ hv)),hwi.resolve_left (fun h=>hx i (h ▸ hw)),hne⟩⟩
  58   · rintro ⟨i,hi⟩
  59     obtain ⟨hvi,hwi,hne⟩:=(complete_on_adj _ v w).mp hi
  60     exact ⟨⟨i,(complete_on_adj _ v w).mpr
  61       ⟨Finset.mem_insert_of_mem hvi,Finset.mem_insert_of_mem hwi,hne⟩⟩,hT i hvi,hT i hwi⟩

  63 theorem root_completed_cliques_at_root (A:Finset V) (x:I→V) (T:I→Finset V)
  64     (hx:∀ i,x i∉A) (hT:∀ i,T i⊆A) (z v:V) (hz:z∉A):
  65     (⨆ i,spanningRestriction (⊤:SimpleGraph V) (insert (x i) (T i))).Adj z v↔
  66       ∃ i,x i=z ∧ v∈T i:=by
  67   rw [SimpleGraph.iSup_adj]
  68   constructor
  69   · rintro ⟨i,hi⟩
  70     obtain ⟨hzi,hvi,hne⟩:=(complete_on_adj _ z v).mp hi
  71     simp only [Finset.mem_insert] at hzi hvi
  72     have he:z=x i:=hzi.resolve_right (fun h=>hz (hT i h))
  73     exact ⟨i,he.symm,hvi.resolve_left (fun h=>hne (he.trans h.symm))⟩
  74   · rintro ⟨i,he,hv⟩
  75     exact ⟨i,(complete_on_adj _ z v).mpr
  76       ⟨by simp [he],Finset.mem_insert_of_mem hv,fun h=>hz (h ▸ hT i hv)⟩⟩

  78 theorem root_completed_cliques_subgraph (G:SimpleGraph V) (x:I→V) (T:I→Finset V)
  79     (hT:∀ i,∀ v∈T i,∀ w∈T i,v≠w→G.Adj v w)
  80     (hx:∀ i,∀ v∈T i,G.Adj (x i) v):
  81     (⨆ i,spanningRestriction (⊤:SimpleGraph V) (insert (x i) (T i)))≤G:=by
  82   intro v w h
  83   obtain ⟨i,hi⟩:=SimpleGraph.iSup_adj.mp h
  84   obtain ⟨hv,hw,hne⟩:=(complete_on_adj _ v w).mp hi
  85   rcases Finset.mem_insert.mp hv with rfl|hv <;> rcases Finset.mem_insert.mp hw with rfl|hw
  86   · exact (hne rfl).elim
  87   · exact hx i w hw
  88   · exact (hx i v hv).symm
  89   · exact hT i v hv w hw hne
