  10 theorem source_piece_lift_to_vertex_subtype (R:Finset V) (p:PrescribedSourcePiece V)
  11     (hR:p.vertices.toFinset⊆R):
  12     ∃ q:PrescribedSourcePiece {x // x∈R},q.map Subtype.val=p := by
  13   cases p with
  14   | path xs =>
  15     have hx:∀x∈xs,x∈R := by
  16       intro x hx
  17       exact hR (List.mem_toFinset.mpr hx)
  18     let ys:List {x // x∈R}:=xs.pmap (fun x hx=>⟨x,hx⟩) hx
  19     have hmap:ys.map Subtype.val=xs := by
  20       simp [ys,List.map_pmap,List.pmap_eq_map_attach]
  21     refine ⟨.path ys,?_⟩
  22     simpa [PrescribedSourcePiece.map] using hmap
  23   | cycle a xs =>
  24     have ha:a∈R := hR (by simp [PrescribedSourcePiece.vertices])
  25     have hx:∀x∈xs,x∈R := by
  26       intro x hx
  27       apply hR
  28       simp [PrescribedSourcePiece.vertices,hx]
  29     let ys:List {x // x∈R}:=xs.pmap (fun x hx=>⟨x,hx⟩) hx
  30     have hmap:ys.map Subtype.val=xs := by
  31       simp [ys,List.map_pmap,List.pmap_eq_map_attach]
  32     refine ⟨.cycle ⟨a,ha⟩ ys,?_⟩
  33     simpa [PrescribedSourcePiece.map] using hmap
  34 end Oberwolfach
