   9 /-- Degree at least n is impossible in a nonempty n-vertex simple graph.
  10 Thus 1 belongs to the set whose infimum defines the cycle threshold. -/
  11 theorem one_admissible (r : ℕ) : AdmissibleCycleBound r 1 := by
  12   classical
  13   refine ⟨by norm_num, 1, ?_⟩
  14   intro n hn G hdiv hdeg
  15   let v : Fin n := ⟨0, by omega⟩
  16   have hlt : G.degree v < n := by simpa using G.degree_lt_card_verts v
  17   have hge := hdeg v
  18   simp only [one_mul] at hge
  19   have hlt' : (G.degree v : ℝ) < n := by exact_mod_cast hlt
  20   exact False.elim (not_lt_of_ge hge hlt')
  21 
  22 theorem cycleThreshold_nonneg (r : ℕ) : 0 ≤ cycleThreshold r := by
  23   apply le_csInf (show {d | AdmissibleCycleBound r d}.Nonempty from ⟨1, one_admissible r⟩)
  24   intro d hd
  25   exact hd.1.1

  27 theorem cycleThreshold_le_one (r : ℕ) : cycleThreshold r ≤ 1 := by
  28   apply csInf_le (show BddBelow {d | AdmissibleCycleBound r d} from ?_) (one_admissible r)
  29   refine ⟨0, ?_⟩
  30   intro d hd
  31   exact hd.1.1

  33 /-- The definition of the infimum really implies the positive-margin statement. -/
  34 theorem cycleThreshold_positive_margin (r : ℕ) {α : ℝ} (hα : 0 < α) :
  35     ∃ N : ℕ, ∀ n ≥ N, ∀ G : SimpleGraph (Fin n), CycleDivisible G r →
  36       (∀ v, (cycleThreshold r + α) * (n : ℝ) ≤ (G.degree v : ℝ)) →
  37       HasCycleDecomposition G r := by
  38   classical
  39   have hn : {d | AdmissibleCycleBound r d}.Nonempty := ⟨1, one_admissible r⟩
  40   obtain ⟨d, hd, hdlt⟩ := exists_lt_of_csInf_lt hn (show cycleThreshold r < cycleThreshold r + α by linarith)
  41   obtain ⟨N, hN⟩ := hd.2
  42   refine ⟨N, ?_⟩
  43   intro n hn G hdiv hdeg
  44   apply hN n hn G hdiv
  45   intro v
  46   exact (mul_le_mul_of_nonneg_right hdlt.le (Nat.cast_nonneg n)).trans (hdeg v)
