  13 theorem resourceRootOmittedLoad_mono (R : I → Finset A) (w : I → ℝ)
  14     (hw : ∀ i,0≤w i) (keep keep' : A → Prop) (hsub : ∀ f,keep' f → keep f) (e : A) :
  15     resourceRootOmittedLoad R w keep' e≤resourceRootOmittedLoad R w keep e := by
  16   unfold resourceRootOmittedLoad
  17   apply Finset.sum_le_sum
  18   intro i _
  19   by_cases h : e∈R i ∧ (∀ f∈R i,f≠e → keep' f)
  20   · rw [if_pos h,if_pos ⟨h.1,fun f hf hne=>hsub f (h.2 f hf hne)⟩]
  21   · rw [if_neg h]
  22     split_ifs <;> first | exact hw i | rfl
