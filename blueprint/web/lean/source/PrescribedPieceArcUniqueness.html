   3 /-! In each ordered path or cycle, every vertex has at most one predecessor
   4 and at most one successor. -/
   5 set_option autoImplicit false
   6 noncomputable section

   8 attribute [local instance] Classical.propDecidable
   9 variable {X:Type} [DecidableEq X]

  11 theorem prescribed_piece_unique_successor (p:PrescribedSourcePiece X) (D:X→X→Prop) (F:SimpleGraph X)
  12     (hp:p.Valid D F) (x y z:X) (hy:(x,y)∈p.arcs) (hz:(x,z)∈p.arcs):y=z := by
  13   cases p with
  14   | path xs =>
  15     apply prescribed_path_unique_successor xs _ x y z hy hz
  16     exact hp.1.sublist (List.dropLast_sublist xs)
  17   | cycle a xs =>
  18     apply prescribed_path_unique_successor ((a::xs)++[a]) _ x y z hy hz
  19     change ((a::xs)++[a]).dropLast.Nodup
  20     rw [List.dropLast_concat]
  21     exact hp.1
  22 
  23 theorem prescribed_piece_unique_predecessor (p:PrescribedSourcePiece X) (D:X→X→Prop) (F:SimpleGraph X)
  24     (hp:p.Valid D F) (x y z:X) (hy:(y,x)∈p.arcs) (hz:(z,x)∈p.arcs):y=z := by
  25   cases p with
  26   | path xs =>
  27     apply prescribed_path_unique_predecessor xs _ x y z hy hz
  28     exact hp.1.sublist (List.tail_sublist xs)
  29   | cycle a xs =>
  30     apply prescribed_path_unique_predecessor ((a::xs)++[a]) _ x y z hy hz
  31     have hax:a∉xs:=(List.nodup_cons.mp hp.1).1
  32     have hxs:xs.Nodup:=(List.nodup_cons.mp hp.1).2
  33     have htail:(xs++[a]).Nodup:=by
  34       apply List.nodup_append'.mpr
  35       refine ⟨hxs,by simp,?_⟩
  36       apply List.disjoint_left.mpr
  37       intro x hx hy
  38       simp only [List.mem_singleton] at hy
  39       subst x;exact hax hx
  40     simpa [List.cons_append] using htail
  41 end Oberwolfach
