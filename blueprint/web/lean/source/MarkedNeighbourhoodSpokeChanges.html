  43 theorem markedRowCountWithSpokes_change (P : SimpleGraph V) (π : Equiv.Perm V) (d : ℕ)
  44     (σ : RawMarkedOutcome V U Γ Color) (H : Bool→V→U→Prop)
  45     (A C : Finset V) (hCA : C⊆A) (S : Finset (Bool×Bool×V)) (J : Finset (Bool×U))
  46     (g : Γ) (j : Color) (δ : ℝ)
  47     (hδ : ∀ a∈J,(∑ x∈A,if ¬(rawMarkedPairSpokes π d σ g j a.1 x a.2 ↔ H a.1 x a.2) then (1:ℝ) else 0)≤δ) :
  48     |markedRowCountWithSpokes P π d σ H C S J-rawMarkedRowCount P π d C S J g j σ|≤J.card*δ := by
  49   have hh := finiteConjunction_sum_change_at_outer A C hCA J
  50     (fun x=>∀ a∈S,rawMarkedPairRims P π d σ (markedRowRim a x))
  51     (fun a x=>rawMarkedPairSpokes π d σ g j a.1 x a.2) (fun a x=>H a.1 x a.2) δ hδ
  52   convert hh using 1
