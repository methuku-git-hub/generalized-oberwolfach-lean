  18 theorem HasFractionalDecomposition.map {V W : Type*}
  19     [Fintype V] [DecidableEq V] [Fintype W] [DecidableEq W]
  20     {G : SimpleGraph V} {P : SimpleGraph V → Prop}
  21     (h : HasFractionalDecomposition G P) (e : V ≃ W) :
  22     HasFractionalDecomposition (G.map e.toEmbedding)
  23       (fun K => ∃ H, P H ∧ K = H.map e.toEmbedding) := by
  24   obtain ⟨w, hw, hs, he⟩ := h
  25   apply fractionalDecomposition_of_indexed _ _ (fun H => H.map e.toEmbedding) w hw
  26   · intro H hH
  27     exact ⟨SimpleGraph.map_monotone _ (hs H hH).1, H, (hs H hH).2, rfl⟩
  28   · intro u v
  29     simp_rw [← SimpleGraph.comap_symm _ e]
  30     exact he (e.symm u) (e.symm v)
