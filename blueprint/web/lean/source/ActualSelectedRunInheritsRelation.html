  11 theorem walk_interval_inherits_successor_relation
  12     {X:Type} [Fintype X] [DecidableEq X] {G:SimpleGraph X} {u v:X} (p:G.Walk u v)
  13     (a b:ℕ) (hab:a≤b) (hb:b≤p.length) (R:X→X→Prop)
  14     (hp:∀j<p.length,R (p.getVert j) (p.getVert (j+1))):
  15     ∀j<(walkInterval p a b hab).length,
  16       R ((walkInterval p a b hab).getVert j) ((walkInterval p a b hab).getVert (j+1)):=by
  17   intro j hj
  18   rw [walk_interval_length p a b hab hb] at hj
  19   rw [walk_interval_getVert p a b hab j (by omega),walk_interval_getVert p a b hab (j+1) (by omega)]
  20   simpa only [Nat.add_assoc] using hp (a+j) (by omega)

  22 theorem actual_selected_run_inherits_parent_relation
  23     {X K J:Type} [Fintype X] [DecidableEq X] [DecidableEq K] [DecidableEq J]
  24     {G:SimpleGraph X} (data:ActualResidualData (K:=K) (J:=J) G) (R:X→X→Prop)
  25     (hparent:∀i∈data.s₂,∀k<(data.P i).length,R ((data.P i).getVert k) ((data.P i).getVert (k+1))):
  26     ∀r:SelectedRunIndex data.s₂ data.d data.T,∀k<(data.path (.inr r)).length,
  27       R ((data.path (.inr r)).getVert k) ((data.path (.inr r)).getVert (k+1)):=by
  28   intro r
  29   rcases r with ⟨i,r⟩
  30   have hr:=(selected_run_mem data.d (data.T i.val) (data.hT i.val) r.val.1 r.val.2).mp r.property
  31   have hrbound:r.val.2<data.d := by
  32     have hh:=data.hT i.val (hr.2.1 r.val.2 hr.1 le_rfl)
  33     simpa using hh
  34   have hright:2+data.c*(r.val.2+1)≤(data.P i.val).length:=by
  35     rw [data.hlen i.val (Finset.mem_union_right _ i.property)]
  36     have hm:=Nat.mul_le_mul_left data.c (show r.val.2+1≤data.d by omega)
  37     omega
  38   change ∀k<(selectedRunWalk (data.P i.val) data.c r.val.1 r.val.2 hr.1).length,
  39     R ((selectedRunWalk (data.P i.val) data.c r.val.1 r.val.2 hr.1).getVert k)
  40       ((selectedRunWalk (data.P i.val) data.c r.val.1 r.val.2 hr.1).getVert (k+1))
  41   unfold selectedRunWalk
  42   exact walk_interval_inherits_successor_relation (data.P i.val) _ _ _ hright R (hparent i.val i.property)
