  17 theorem profile_subset_inclusion_upper (m : I → ℕ)
  18     [∀ i,Nonempty (FixedSubset (V i) (m i))]
  19     (hm : ∀ i,m i ≤ Fintype.card (V i)) (ρ : ℝ) (hρ : 0 ≤ ρ)
  20     (hρm : ∀ i,(m i:ℝ)/Fintype.card (V i) ≤ ρ) (A : Finset (Σ i,V i)) :
  21     (FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (m i)))).probability
  22       (fun σ => A⊆profileRootSet σ) ≤ ρ^A.card := by
  23   rw [profile_subset_inclusion_probability m hm]
  24   let a := fun i => (Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈A)).card
  25   have ha : ∀ i,a i ≤ Fintype.card (V i) := fun i => Finset.card_le_univ _
  26   have hsum : ∑ i,a i=A.card := (sigma_finset_card A).symm
  27   calc
  28     _ ≤ ∏ i,ρ^(a i) := by
  29       apply Finset.prod_le_prod₀
  30       · intro i _;positivity
  31       · intro i _
  32         exact (choose_ratio_le_power (hm i) (ha i)).trans
  33           (pow_le_pow_left₀ (by positivity) (hρm i) _)
  34     _ = _ := by rw [Finset.prod_pow_eq_pow_sum,hsum]

  36 theorem profile_pair_inclusion_upper (m : I → ℕ)
  37     [∀ i,Nonempty (FixedSubset (V i) (m i))]
  38     (hm : ∀ i,m i ≤ Fintype.card (V i)) (ρ : ℝ) (hρ : 0 ≤ ρ)
  39     (hρm : ∀ i,(m i:ℝ)/Fintype.card (V i) ≤ ρ) (x y : Σ i,V i) (hxy : x≠y) :
  40     (FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (m i)))).probability
  41       (fun σ => x∈profileRootSet σ ∧ y∈profileRootSet σ) ≤ ρ^2 := by
  42   have hh := profile_subset_inclusion_upper m hm ρ hρ hρm {x,y}
  43   simpa only [Finset.insert_subset_iff,Finset.singleton_subset_iff,Finset.card_pair hxy] using hh

  45 theorem profile_forbidden_pairs_expectation (m : I → ℕ)
  46     [∀ i,Nonempty (FixedSubset (V i) (m i))]
  47     (hm : ∀ i,m i ≤ Fintype.card (V i)) (ρ : ℝ) (hρ : 0 ≤ ρ)
  48     (hρm : ∀ i,(m i:ℝ)/Fintype.card (V i) ≤ ρ)
  49     (E : Finset ((Σ i,V i)×(Σ i,V i))) (hE : ∀ e∈E,e.1≠e.2) :
  50     (FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (m i)))).expectation
  51       (fun σ => ((E.filter (fun e => e.1∈profileRootSet σ ∧ e.2∈profileRootSet σ)).card:ℝ)) ≤
  52       (E.card:ℝ)*ρ^2 := by
  53   calc
  54     _ = ∑ e∈E,(FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (m i)))).probability
  55         (fun σ => e.1∈profileRootSet σ ∧ e.2∈profileRootSet σ) := by
  56       convert finiteLaw_expectation_count
  57         (FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (m i)))) E
  58         (fun e σ => e.1∈profileRootSet σ ∧ e.2∈profileRootSet σ) using 1 <;> congr!
  59     _ ≤ ∑ _e∈E,ρ^2 := Finset.sum_le_sum (fun e he => profile_pair_inclusion_upper m hm ρ hρ hρm e.1 e.2 (hE e he))
  60     _ = _ := by simp
