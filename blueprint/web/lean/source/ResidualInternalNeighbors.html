  10 theorem two_factor_internal_path_edges (F:SimpleGraph V)
  11     (hF:F.IsRegularOfDegree 2) {u v:V} (p:F.Walk u v) (hp:p.IsPath)
  12     (x y:V) (hx:x∈p.support.toFinset \ {u,v}) (hxy:F.Adj x y):
  13     s(x,y)∈p.edges.toFinset := by
  14   have hx' := Finset.mem_sdiff.mp hx
  15   have hxu:x≠u := by
  16     intro h;subst x;exact hx'.2 (by simp)
  17   have hxv:x≠v := by
  18     intro h;subst x;exact hx'.2 (by simp)
  19   obtain ⟨i,hi,hile⟩:=SimpleGraph.Walk.mem_support_iff_exists_getVert.mp (List.mem_toFinset.mp hx'.1)
  20   have hi0:i≠0 := by
  21     intro h;subst i;apply hxu;simpa using hi.symm
  22   have hilt:i<p.length := by
  23     have hh:i≠p.length := by intro h;subst i;apply hxv;simpa using hi.symm
  24     omega
  25   have hsub:=p.toSubgraph.neighborSet_subset x
  26   have hsmall:(p.toSubgraph.neighborSet x).ncard=2 := by
  27     rw [←hi]
  28     exact hp.ncard_neighborSet_toSubgraph_internal_eq_two hi0 hilt
  29   have hbig:(F.neighborSet x).ncard=2 := by rw [F.ncard_neighborSet,hF x]
  30   have hsets:p.toSubgraph.neighborSet x=F.neighborSet x := by
  31     apply Set.eq_of_subset_of_ncard_le hsub
