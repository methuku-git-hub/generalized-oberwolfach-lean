  18 theorem relation_two_step_vertex_restriction (f:W↪V) (R:V→V→Prop) (x y:W):
  19     |((Finset.univ.filter (fun z=>R (f x) (f z)∧R (f z) (f y))).card:ℝ)-
  20       (Finset.univ.filter (fun z=>R (f x) z∧R z (f y))).card|≤
  21       (Finset.univ\Finset.univ.image f).card:=by
  22   have hh:=finite_bounded_sum_restriction (Finset.univ.image f) Finset.univ
  23     (Finset.subset_univ _) (fun z=>if R (f x) z∧R z (f y) then (1:ℝ) else 0)
  24     (by intro z _;split_ifs <;> norm_num)
  25   rw [Finset.sum_image (fun _ _ _ _ he=>f.injective he)] at hh
  26   simpa only [Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero] using hh

  28 theorem relation_two_step_changes (R Q:V→V→Prop) (x y:V) (δout δin:ℝ)
  29     (hout:(∑ z,if ¬(R x z↔Q x z) then (1:ℝ) else 0)≤δout)
  30     (hin:(∑ z,if ¬(R z y↔Q z y) then (1:ℝ) else 0)≤δin):
  31     |((Finset.univ.filter (fun z=>Q x z∧Q z y)).card:ℝ)-
  32       (Finset.univ.filter (fun z=>R x z∧R z y)).card|≤δout+δin:=by
  33   rw [relation_two_step_count_eq,relation_two_step_count_eq]
  34   apply (finset_indicator_sum_change Finset.univ _ _).trans
  35   calc
  36     _ ≤ ∑ z,((if ¬(R x z↔Q x z) then (1:ℝ) else 0)+(if ¬(R z y↔Q z y) then (1:ℝ) else 0)):=by
