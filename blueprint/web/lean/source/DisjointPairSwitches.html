   5 /-! Simultaneous switches on disjoint pairs are a single permutation of the
   6 original coordinates. Weighted sums change by the explicitly computed
   7 contribution of each pair. -/

  14 def pairSwitchPermutation {I E : Type*} [Fintype I] [DecidableEq E]
  15     (ι : (I×Bool)↪E) : Equiv.Perm E :=
  16   Equiv.Perm.viaFintypeEmbedding (Equiv.prodCongr (Equiv.refl I) Bool.involutive_not.toPerm) ι

  18 theorem pairSwitchPermutation_apply {I E : Type*} [Fintype I] [DecidableEq E]
  19     (ι : (I×Bool)↪E) (i : I) (b : Bool) :
  20     pairSwitchPermutation ι (ι (i,b))=ι (i,!b) := by
  21   simp only [pairSwitchPermutation,Equiv.Perm.viaFintypeEmbedding_apply_image]
  22   rfl

  24 theorem pairSwitchPermutation_fixed {I E : Type*} [Fintype I] [DecidableEq E]
  25     (ι : (I×Bool)↪E) (e : E) (he : e∉Set.range ι) : pairSwitchPermutation ι e=e :=
  26   Equiv.Perm.viaFintypeEmbedding_apply_notMem_range _ _ he

  28 theorem sum_supported_embedding {I E R : Type*} [Fintype I] [Fintype E]
  29     [AddCommMonoid R] (ι : I↪E) (f : E → R)
  30     (hf : ∀ e,e∉Set.range ι → f e=0) : (∑ e,f e)=∑ i,f (ι i) := by
  31   classical
  32   calc
  33     (∑ e,f e)=∑ e∈Finset.univ.image ι,f e := by
  34       symm
  35       apply Finset.sum_subset (Finset.subset_univ _)
  36       intro e _ he
  37       apply hf e
  38       simpa only [Finset.mem_image,Finset.mem_univ,true_and,Set.mem_range] using he
  39     _ = ∑ i,f (ι i) := Finset.sum_image (fun _ _ _ _ h => ι.injective h)

  41 theorem pair_switch_weighted_difference {I E C R : Type*}
  42     [Fintype I] [Fintype E] [DecidableEq E] [CommRing R]
  43     (ι : (I×Bool)↪E) (σ : E → C) (t : E → R) (g : C → R) :
  44     (∑ e,t e*g (σ (pairSwitchPermutation ι e)))-(∑ e,t e*g (σ e))=
  45       ∑ i,(t (ι (i,false))-t (ι (i,true)))*
  46         (g (σ (ι (i,true)))-g (σ (ι (i,false)))) := by
  47   rw [← Finset.sum_sub_distrib]
  48   rw [sum_supported_embedding ι _ (by
  49     intro e he
  50     rw [pairSwitchPermutation_fixed ι e he,sub_self])]
  51   rw [Fintype.sum_prod_type]
  52   apply Finset.sum_congr rfl
  53   intro i _
  54   simp only [Fintype.sum_bool,pairSwitchPermutation_apply,Bool.not_true,Bool.not_false]
  55   ring

  57 theorem pair_switch_preserves_weighted_sum {I E C R : Type*}
  58     [Fintype I] [Fintype E] [DecidableEq E] [CommRing R]
  59     (ι : (I×Bool)↪E) (σ : E → C) (t : E → R) (g : C → R)
  60     (ht : ∀ i,t (ι (i,false))=t (ι (i,true))) :
  61     (∑ e,t e*g (σ (pairSwitchPermutation ι e)))=(∑ e,t e*g (σ e)) := by
  62   apply sub_eq_zero.mp
  63   rw [pair_switch_weighted_difference]
  64   simp only [ht,sub_self,zero_mul,Finset.sum_const_zero]
