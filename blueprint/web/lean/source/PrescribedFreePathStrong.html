  13     (as:List X) (ha:as.Nodup) (hne:1≤as.length) (A:Finset X) (f:X→V)
  14     (hinj:Set.InjOn f A) (hsource:∀ u∈as,u∉A)

  17     (hbudget:((A.image f∪S).card:ℝ)+as.length+2+M<β*Fintype.card V):
  18     ∃ g:X→V,Set.InjOn g (↑(A∪as.toFinset):Set X) ∧ (∀ u∈A,g u=f u) ∧
  19       (∀ u∈as,g u∈L ∧ g u∉S) ∧ DirectedNonemptyPath D F true (as.map g) := by
  20   let B:=A.image f∪S
  21   obtain ⟨ys,hyslen,hysnodup,hys,hpath⟩:=extending_list_embed_free_path D L β hext F M hM hdeg true as.length hne B hbudget
  22   have hlen:as.length=ys.length:=hyslen.symm
  23   have htarget:∀ z∈ys,z∉A.image f:=by
  24     intro z hz hmem
  25     exact (hys z hz).2 (Finset.mem_union_left S hmem)
  26   have hgood:∀ z∈ys,z∈L ∧ z∉S:=by
  27     intro z hz
  28     exact ⟨(hys z hz).1,fun h=>(hys z hz).2 (Finset.mem_union_right _ h)⟩
  29   let g:=updatePairedImages f (as.zip ys)
  30   have hb:=prescribed_embedding_zip_update as ys hlen ha hysnodup A f hinj hsource htarget
  31   have hmap:=prescribed_embedding_update_map_list as ys hlen ha hysnodup A f hinj hsource htarget
  32   have hnew:=prescribed_embedding_zip_targets as ys hlen ha hysnodup A f hinj hsource htarget (fun z=>z∈L ∧ z∉S) hgood
  33   refine ⟨g,hb.1,hb.2.1,hnew,?_⟩
  34   rw [hmap]
