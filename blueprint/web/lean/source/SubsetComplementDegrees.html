   9 theorem subset_complement_degrees {V:Type} [Fintype V] [DecidableEq V]
  10     (G:SimpleGraph V) (S:Finset V) (v:V) (hv:v∈S):
  11     (G.neighborFinset v∩S).card+((Gᶜ).neighborFinset v∩S).card+1=S.card:=by
  12   have hpart:=Finset.card_filter_add_card_filter_not (s:=S) (fun w=>G.Adj v w)
  13   have heq:S.filter (fun w=>¬G.Adj v w)=((Gᶜ).neighborFinset v∩S)∪{v}:=by
  14     ext w
  15     simp only [Finset.mem_filter,Finset.mem_union,Finset.mem_inter,Finset.mem_singleton,
  16       SimpleGraph.mem_neighborFinset,SimpleGraph.compl_adj]
  17     constructor
  18     · rintro ⟨hw,hn⟩
  19       by_cases h:w=v
  20       · exact Or.inr h
  21       · exact Or.inl ⟨⟨Ne.symm h,hn⟩,hw⟩
  22     · rintro (⟨⟨hne,hn⟩,hw⟩|rfl)
  23       · exact ⟨hw,hn⟩
  24       · exact ⟨hv,by simp⟩
  25   have hd:Disjoint ((Gᶜ).neighborFinset v∩S) {v}:=by
  26     apply Finset.disjoint_singleton_right.mpr
  27     simp
  28   have heq2:S.filter (fun w=>G.Adj v w)=G.neighborFinset v∩S:=by ext w;simp [and_comm]
  29   rw [heq,heq2,Finset.card_union_of_disjoint hd,Finset.card_singleton] at hpart
  30   omega
  31 theorem subset_explicit_complement_degrees {V:Type} [Fintype V] [DecidableEq V]
  32     (G P:SimpleGraph V) (hP:P=Gᶜ) (S:Finset V) (v:V) (hv:v∈S):
  33     (G.neighborFinset v∩S).card+(P.neighborFinset v∩S).card+1=S.card:=by
  34   have hp:(P.neighborFinset v∩S)=((Gᶜ).neighborFinset v∩S):=by
  35     ext w
  36     simp only [Finset.mem_inter,SimpleGraph.mem_neighborFinset]
  37     rw [hP]
  38   rw [hp]
  39   exact subset_complement_degrees G S v hv
