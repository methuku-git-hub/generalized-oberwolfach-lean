  10 theorem physical_cover_gives_symmetric_old_edge_cover
  11     {X V K J:Type} [Fintype X] [DecidableEq X] [Fintype V] [DecidableEq V] [DecidableEq K] [DecidableEq J]
  12     {G:SimpleGraph X} (data:ActualResidualData (K:=K) (J:=J) G)
  13     (Final:Finset (Sym2 X)) (ψ:X→V) (D:X→X→Prop)
  14     (horient:∀x y,G.Adj x y→D x y ∨ D y x) (Pack:Finset (Finset V))
  15     (hcover:∀x y,s(x,y)∈data.deletedEdges\Final→D x y→{ψ x,ψ y}∈Pack):
  16     ∀e∈(data.deletedEdges\Final).image (Sym2.map ψ),e.toFinset∈Pack:=by
  17   intro e he
  18   obtain ⟨f,hf,rfl⟩:=Finset.mem_image.mp he
  19   induction f using Sym2.ind with
  20   | _ x y =>
  21     have hadj:G.Adj x y:=SimpleGraph.mem_edgeFinset.mp (data.deleted_edges_subset_factor (Finset.mem_sdiff.mp hf).1)
  22     rcases horient x y hadj with hxy|hyx
  23     · simpa only [Sym2.map_mk,Sym2.toFinset_mk_eq] using hcover x y hf hxy
  24     · have hf':s(y,x)∈data.deletedEdges\Final:=by simpa only [Sym2.eq_swap] using hf
  25       have hh:=hcover y x hf' hyx
  26       simpa only [Sym2.map_mk,Sym2.toFinset_mk_eq,Finset.pair_comm] using hh

  28 theorem physical_disjointness_gives_symmetric_packing
  29     {U V:Type} [Fintype U] [Fintype V] [DecidableEq V]
  30     (old:U→Finset (Sym2 V)) (physical:U→Finset (Finset V))
  31     (hcover:∀w e,e∈old w→e.toFinset∈physical w)
  32     (hdis:∀w z,w≠z→Disjoint (physical w) (physical z)):
  33     ∃Pack:Finset (Sym2 V),
  34       (∀w z,w≠z→Disjoint (old w) (old z)) ∧ (∀w,old w⊆Pack):=by
  35   let Pack:Finset (Sym2 V):=Finset.univ.filter (fun e=>e.toFinset∈Finset.univ.biUnion physical)
  36   refine ⟨Pack,?_,?_⟩
  37   · intro w z hne
  38     apply Finset.disjoint_left.mpr
  39     intro e he hf
  40     exact Finset.disjoint_left.mp (hdis w z hne) (hcover w e he) (hcover z e hf)
  41   · intro w e he
  42     apply Finset.mem_filter.mpr
  43     exact ⟨Finset.mem_univ _,Finset.mem_biUnion.mpr ⟨w,Finset.mem_univ _,hcover w e he⟩⟩
  44 end Oberwolfach
