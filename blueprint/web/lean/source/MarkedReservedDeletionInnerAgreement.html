  33     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ) (σ:RawMarkedOutcome V U Γ Color)

  37     let σ':=rawMarkedDeleteRims σ DR.edgeFinset;
  38     ∀ (mark:Bool) (e:MarkedSampleArc P π (3*d) (fun a=>σ (Sum.inl a)) mark),
  39       e.val.1∈I→e.val.2∈I→
  40       ∃ e':MarkedSampleArc P π (3*d) (fun a=>σ' (Sum.inl a)) mark,e'.val=e.val :=by
  41   intro DR σ' mark e h₁ h₂

  43   have hy:(if mark then π e.val.2 else e.val.2)∈I:=by
  44     cases mark <;> simp only [Bool.false_eq_true,ite_false,ite_true]
  45     · exact h₂
  46     · exact (hI _).mpr h₂
