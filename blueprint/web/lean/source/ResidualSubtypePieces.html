   9 theorem source_pieces_on_surviving_subtype
  10     (R:Finset V) (D:V→V→Prop) (ps:List (PrescribedSourcePiece V))
  11     (hp:ps.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset))
  12     (hv:∀p∈ps,p.Shape ∧ p.OriginalValid D)
  13     (hR:sourcePieceVerticesUnion ps=R):
  14     ∃ qs:List (PrescribedSourcePiece {x // x∈R}),
  15       qs.map (fun q=>q.map Subtype.val)=ps ∧
  16       qs.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset) ∧
  17       (∀q∈qs,q.Shape ∧ q.OriginalValid (fun x y=>D x.1 y.1)) ∧
  18       sourcePieceVerticesUnion qs=Finset.univ := by
  19   have hsub:∀p∈ps,p.vertices.toFinset⊆R:=by
  20     intro p hp x hx
  21     rw [←hR]
  22     exact source_piece_vertex_mem_union ps p hp x (List.mem_toFinset.mp hx)
  23   obtain ⟨qs,hqs⟩:=source_piece_list_lift_to_vertex_subtype R ps hsub
  24   refine ⟨qs,hqs,?_,?_,?_⟩
  25   · rw [←hqs,List.pairwise_map] at hp
  26     apply hp.imp
  27     intro p q hpq
  28     apply Finset.disjoint_left.mpr
  29     intro x hx hy
  30     have hx':x.1∈(p.map Subtype.val).vertices.toFinset:=by
  31       rw [prescribed_piece_map_vertices]
  32       exact List.mem_toFinset.mpr (List.mem_map.mpr ⟨x,List.mem_toFinset.mp hx,rfl⟩)
  33     have hy':x.1∈(q.map Subtype.val).vertices.toFinset:=by
  34       rw [prescribed_piece_map_vertices]
  35       exact List.mem_toFinset.mpr (List.mem_map.mpr ⟨x,List.mem_toFinset.mp hy,rfl⟩)
  36     exact (Finset.disjoint_left.mp hpq) hx' hy'
  37   · intro q hq
  38     apply subtype_piece_validity_reflected R q D
  39     apply (hv (q.map Subtype.val) ?_).2
  40     rw [←hqs]
  41     exact List.mem_map.mpr ⟨q,hq,rfl⟩
  42   · have himage:(sourcePieceVerticesUnion qs).image Subtype.val=R:=by
  43       rw [←source_piece_map_vertices_union,hqs,hR]
  44     apply Finset.eq_univ_iff_forall.mpr
  45     intro x
  46     have hx:x.1∈(sourcePieceVerticesUnion qs).image Subtype.val:=by rw [himage];exact x.2
  47     obtain ⟨y,hy,heq⟩:=Finset.mem_image.mp hx
  48     have :y=x:=Subtype.ext heq
  49     simpa [this] using hy
