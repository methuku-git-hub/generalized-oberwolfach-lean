   3 /-! Splitting an ordered path retains exactly the edges present in the

  11 def splitPathByGraph (H:SimpleGraph V) (xs:List V):List (List V) :=
  12   splitOrderedPath (fun x y=>decide (H.Adj x y)) xs

  14 def splitPathEdgeUnion (H:SimpleGraph V) (xs:List V):Finset (Sym2 V) :=
  15   (splitPathByGraph H xs).toFinset.biUnion prescribedOrderedPathEdges

  17 theorem split_path_edge_union (H:SimpleGraph V):
  18     ∀ xs:List V,splitPathEdgeUnion H xs =
  19       (prescribedOrderedPathEdges xs).filter (fun e=>e∈H.edgeFinset) := by
  20   intro xs
  21   induction xs with
  22   | nil => simp [splitPathEdgeUnion,splitPathByGraph,splitOrderedPath,prescribedOrderedPathEdges]
  23   | cons x xs ih =>
  24     cases xs with
  25     | nil => simp [splitPathEdgeUnion,splitPathByGraph,splitOrderedPath,prescribedOrderedPathEdges]
  26     | cons y ys =>
  27       obtain ⟨block,blocks,h⟩:=split_ordered_path_head (fun x y=>decide (H.Adj x y)) y ys
  28       have ih':( (y::block)::blocks).toFinset.biUnion prescribedOrderedPathEdges =
  29           (prescribedOrderedPathEdges (y::ys)).filter (fun e=>e∈H.edgeFinset) := by
  30         simpa [splitPathEdgeUnion,splitPathByGraph,h] using ih
  31       by_cases hxy:H.Adj x y
  32       · have he:s(x,y)∈H.edgeSet := hxy
  33         have h' := congrArg (fun E:Finset (Sym2 V)=>insert s(x,y) E) ih'
  34         simpa [splitPathEdgeUnion,splitPathByGraph,splitOrderedPath,h,hxy,
  35           prescribedOrderedPathEdges,SimpleGraph.mem_edgeFinset, Finset.filter_insert,he,
  36           Finset.insert_union] using h'
  37       · have he:s(x,y)∉H.edgeSet := hxy
  38         simpa [splitPathEdgeUnion,splitPathByGraph,splitOrderedPath,h,hxy,
  39           prescribedOrderedPathEdges,SimpleGraph.mem_edgeFinset,Finset.filter_insert,he] using ih'
  40 end Oberwolfach
