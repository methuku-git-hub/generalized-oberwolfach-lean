  13 def selectionCounter (active : Fin N→I→Prop) (hit : I→C→Prop) (F : Fin N→C) (x : I) : ℕ :=
  14   (Finset.univ.filter (fun j=>active j x ∧ hit x (F j))).card

  21 theorem selectionCounter_update (active : Fin N→I→Prop) (hit : I→C→Prop)
  22     (F : Fin N→C) (i : Fin N) (c : C) (x : I) (hbefore : ¬hit x (F i)) :
  23     selectionCounter active hit (Function.update F i c) x=
  24       selectionCounter active hit F x+(if active i x ∧ hit x c then 1 else 0) := by
  25   let S := Finset.univ.filter (fun j=>active j x ∧ hit x (F j))
  26   have hi : i∉S := by simp [S,hbefore]
  27   by_cases hnew : active i x ∧ hit x c
  28   · have he : Finset.univ.filter (fun j=>active j x ∧ hit x (Function.update F i c j))=insert i S := by
  29       ext j
  30       by_cases hj : j=i
  31       · subst j;simp [hnew]
  32       · simp [Function.update_of_ne hj,S,hj]
  33     rw [if_pos hnew]
  34     change _=S.card+1
  35     simpa only [selectionCounter,he,Finset.card_insert_of_notMem hi]
  36   · have he : Finset.univ.filter (fun j=>active j x ∧ hit x (Function.update F i c j))=S := by
  37       ext j
  38       by_cases hj : j=i
  39       · subst j;simp [S,hnew,hbefore]
  40       · simp [Function.update_of_ne hj,S,hj]
  41     simp only [selectionCounter,he,if_neg hnew,Nat.add_zero]
  42     rfl

  44 def SelectionPrefix (allowed : Fin N→C→Prop) (compatible : C→C→Prop)
  45     (empty : C) (k : ℕ) (F : Fin N→C) : Prop :=
  46   (∀ i,k ≤ i.val → F i=empty) ∧
  47   (∀ i,i.val < k → allowed i (F i)) ∧
  48   (∀ i j,i.val < j.val → j.val < k → compatible (F i) (F j))

  54 theorem selectionPrefix_update (allowed : Fin N→C→Prop) (compatible : C→C→Prop)
  55     (empty : C) (i : Fin N) (F : Fin N→C) (c : C)
  56     (hF : SelectionPrefix allowed compatible empty i.val F)
  57     (hc : allowed i c) (hcomp : ∀ j,j.val < i.val → compatible (F j) c) :
  58     SelectionPrefix allowed compatible empty (i.val+1) (Function.update F i c) := by
  59   refine ⟨?_,?_,?_⟩
  60   · intro j hj
  61     have hne : j≠i := by intro he;subst j;omega
  62     rw [Function.update_of_ne hne]
  63     exact hF.1 j (by omega)
  64   · intro j hj
  65     by_cases he : j=i
  66     · subst j;simpa using hc
  67     · rw [Function.update_of_ne he]
  68       have hne : j.val≠i.val := fun hh=>he (Fin.ext hh)
  69       exact hF.2.1 j (by omega)
  70   · intro a b hab hb
  71     have hai : a≠i := by intro he;subst a;omega
  72     rw [Function.update_of_ne hai]
  73     by_cases hbi : b=i
  74     · subst b;simp only [Function.update_self];exact hcomp a hab
  75     · rw [Function.update_of_ne hbi]
  76       have hne : b.val≠i.val := fun hh=>hbi (Fin.ext hh)
  77       exact hF.2.2 a b hab (by omega)
