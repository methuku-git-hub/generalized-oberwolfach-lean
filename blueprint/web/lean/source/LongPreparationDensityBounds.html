  11 theorem long_preparation_density_bounds {U:Type} [Fintype U]
  12     (D rho c:ℝ) (hD:0<D) (hrho:0<rho) (hc:0<c)
  13     (a b:U→ℝ) (ha:∀ u,1/(3*c)≤a u) (hb:∀ u,1/4≤b u)
  14     (hm:rho*D/2≤(Fintype.card U:ℝ)):
  15     rho/8≤(∑ u,b u)/D ∧ rho/(6*c)≤(∑ u,a u)/D:=by
  16   classical
  17   have hbs:(Fintype.card U:ℝ)/4≤∑ u,b u:=by
  18     have hh:∑ _u:U,(1/4:ℝ)≤∑ u,b u:=Finset.sum_le_sum (fun u _=>hb u)
  19     simpa [div_eq_mul_inv] using hh
  20   have has:(Fintype.card U:ℝ)/(3*c)≤∑ u,a u:=by
  21     have hh:∑ _u:U,(1/(3*c):ℝ)≤∑ u,a u:=Finset.sum_le_sum (fun u _=>ha u)
  22     simpa [div_eq_mul_inv] using hh
  23   constructor
  24   · apply (le_div_iff₀ hD).2
  25     nlinarith [hbs,hm]
  26   · apply (le_div_iff₀ hD).2
  27     have hden:0<3*c:=by positivity
  28     have hx:=div_le_div_of_nonneg_right hm hden.le
  29     have heq:(rho*D/2)/(3*c)=(rho/(6*c))*D:=by field_simp <;> ring
  30     rw [heq] at hx
  31     exact hx.trans has
