  11 theorem triangle_diagram_dense_geometry {V U:Type} [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U]

  13     (hdeg:∀ v,(3/4+β)*(Fintype.card V:ℝ)≤(Q.degree v:ℝ))
  14     (hnearV:∀ u,(((Finset.univ:Finset V).filter (fun v=>¬H v u)).card:ℝ)≤ε*(Fintype.card V:ℝ))
  15     (hnearU:∀ v,(((Finset.univ:Finset U).filter (fun u=>¬H v u)).card:ℝ)≤ε*(Fintype.card U:ℝ)):
  16     (∀ v w,β*(Fintype.card V:ℝ)≤((Q.neighborFinset v∩Q.neighborFinset w).card:ℝ)) ∧
  17     (∀ u u' v w,β*(Fintype.card V:ℝ)≤(((Finset.univ.filter (fun a=>H a u∧H a u'))∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ)) ∧
  18     (∀ T:Finset V,T.card≤3→(1/2:ℝ)*Fintype.card U≤
  19       ((common Finset.univ T (fun v=>Finset.univ.filter (H v))).card:ℝ)) :=by

  22   · intro v w
  23     have hh:=triangle_four_graph_neighbours Q β hdeg v w v w
  24     have he:(Finset.univ.filter (fun x=>Q.Adj v x∧Q.Adj w x∧Q.Adj v x∧Q.Adj w x))=
  25         Q.neighborFinset v∩Q.neighborFinset w:=by ext x;simp [SimpleGraph.mem_neighborFinset];tauto
  26     rw [he] at hh
  27     nlinarith [mul_nonneg hβ.le hn]
  28   constructor

  32         (Finset.univ.filter (fun x=>H x u∧H x u'))∩Q.neighborFinset v∩Q.neighborFinset w:=by
  33       ext x;simp [SimpleGraph.mem_neighborFinset, and_assoc];tauto
  34     rw [he] at hh
