  15 theorem neighbour_colour_witnesses
  16     {V C J : Type*} [Fintype V] (G : SimpleGraph V) (U : Finset V)
  17     (Q : V → Finset J) (T : J → Finset V) {k d : ℕ}
  18     (hQ : ∀ v∈U,(Q v).card≤k)
  19     (hlocal : ∀ (w : V) (S : Finset V),(∀ v∈S,G.Adj w v) →
  20       (∀ u∈S,∀ v∈S,u≠v → ¬G.Adj u v) → S.card≤k)
  21     (hinc : ∀ w,(U.filter (fun v=>∃ j∈Q v,w∈T j)).card≤d)
  22     (σ : V → C×Bool) (c : C) :
  23     ∃ W : V → Finset V,
  24       (∀ v∈U,(W v).card≤3*k) ∧
  25       (∀ i,(U.filter (fun v=>i∈W v)).card≤(k+3)*d) ∧
  26       (∀ v∈U,∀ τ,(∀ i∈W v,σ i=τ i) →
  27         (assignedSetIndices (Q v) T σ c).card≤(assignedSetIndices (Q v) T τ c).card ∧
  28         (rejectedSetIndices G (Q v) T σ c).card≤(rejectedSetIndices G (Q v) T τ c).card) := by
  29   classical
  30   let A := fun v=>assignedDirectVertices (Q v) T σ c
  31   let B := fun v=>rejectedDirectVertices G (Q v) T σ c
  32   let D := U.biUnion B
  33   have hBD (v : V) (hv : v∈U) : B v⊆D :=
  34     fun i hi=>Finset.mem_biUnion.mpr ⟨v,hv,hi⟩
  35   have hDcolour (i : V) (hi : i∈D) : (σ i).1=c := by
  36     obtain ⟨v,_hv,hi⟩ := Finset.mem_biUnion.mp hi
  37     exact (rejectedDirectVertices_spec G (Q v) T σ c hi).1
  38   have hDbad (i : V) (hi : i∈D) : ¬coinColourSurvives G σ i := by
  39     obtain ⟨v,_hv,hi⟩ := Finset.mem_biUnion.mp hi
  40     exact (rejectedDirectVertices_spec G (Q v) T σ c hi).2.1
  41   obtain ⟨f,hf,hfload⟩ := bounded_monochromatic_conflict_witnesses
  42     G D σ c hlocal hDcolour hDbad
  43   let W := fun v=>(A v∪B v)∪(B v).image f
  44   have hdirect (w : V) : (U.filter (fun v=>w∈A v∪B v)).card≤d := by
  45     have hsub : U.filter (fun v=>w∈A v∪B v)⊆
  46         U.filter (fun v=>∃ j∈Q v,w∈T j) := by
  47       intro v hv
  48       obtain ⟨hvU,hw⟩ := Finset.mem_filter.mp hv
  49       refine Finset.mem_filter.mpr ⟨hvU,?_⟩
  50       rcases Finset.mem_union.mp hw with hw|hw
  51       · exact (assignedDirectVertices_spec (Q v) T σ c hw).2
  52       · exact (rejectedDirectVertices_spec G (Q v) T σ c hw).2.2
  53     exact (Finset.card_le_card hsub).trans (hinc w)
  54   have hBload (w : V) : (U.filter (fun v=>w∈B v)).card≤d :=
  55     (Finset.card_le_card (show U.filter (fun v=>w∈B v)⊆
  56       U.filter (fun v=>w∈A v∪B v) from fun v hv=>
  57         Finset.mem_filter.mpr ⟨(Finset.mem_filter.mp hv).1,
  58           Finset.mem_union_right _ (Finset.mem_filter.mp hv).2⟩)).trans (hdirect w)
  59   have himage (w : V) : (U.filter (fun v=>w∈(B v).image f)).card≤(k+2)*d :=
  60     finite_witness_image_load U D B f hBD hBload hfload w
  61   refine ⟨W,?_,?_,?_⟩
  62   · intro v hv
  63     have hA := (assignedDirectVertices_card (Q v) T σ c).trans (hQ v hv)
  64     have hB := (rejectedDirectVertices_card G (Q v) T σ c).trans (hQ v hv)
  65     have hu₁ := Finset.card_union_le (A v) (B v)
  66     have hu₂ := Finset.card_union_le (A v∪B v) ((B v).image f)
  67     have hi : ((B v).image f).card≤(B v).card := Finset.card_image_le
  68     change ((A v∪B v)∪(B v).image f).card≤_
  69     change (A v).card≤k at hA
  70     change (B v).card≤k at hB
  71     omega
  72   · intro i
  73     have hsub : U.filter (fun v=>i∈W v)⊆
  74         (U.filter (fun v=>i∈A v∪B v))∪(U.filter (fun v=>i∈(B v).image f)) := by
  75       intro v hv
  76       obtain ⟨hvU,hi⟩ := Finset.mem_filter.mp hv
  77       rcases Finset.mem_union.mp hi with hi|hi
  78       · exact Finset.mem_union_left _ (Finset.mem_filter.mpr ⟨hvU,hi⟩)
  79       · exact Finset.mem_union_right _ (Finset.mem_filter.mpr ⟨hvU,hi⟩)
  80     have hcard := (Finset.card_le_card hsub).trans (Finset.card_union_le _ _)
  81     have h₁ := hdirect i
  82     have h₂ := himage i
  83     nlinarith
  84   · intro v hv τ he
  85     have hAeq (i : V) (hi : i∈A v) : σ i=τ i :=
  86       he i (Finset.mem_union_left _ (Finset.mem_union_left _ hi))
  87     have hBeq (i : V) (hi : i∈B v) : σ i=τ i :=
  88       he i (Finset.mem_union_left _ (Finset.mem_union_right _ hi))
  89     have hBbad (i : V) (hi : i∈B v) : ¬coinColourSurvives G τ i := by
  90       have heq := hBeq i hi
  91       have hfEq : σ (f i)=τ (f i) :=
  92         he (f i) (Finset.mem_union_right _ (Finset.mem_image.mpr ⟨i,hi,rfl⟩))
  93       obtain ⟨hcol,hcase⟩ := hf i (hBD v hv hi)
  94       intro hkeep
  95       rcases hcase with ⟨_hself,hcoin⟩|hadj
  96       · exact hcoin ((congrArg Prod.snd heq).trans hkeep.1)
  97       · apply hkeep.2 (f i) hadj
  98         exact (congrArg Prod.fst hfEq).symm.trans
  99           (hcol.trans ((hDcolour i (hBD v hv hi)).symm.trans (congrArg Prod.fst heq)))
 100     exact ⟨Finset.card_le_card (assignedDirectVertices_preserve (Q v) T σ τ c hAeq),
 101       Finset.card_le_card (rejectedDirectVertices_preserve G (Q v) T σ τ c hBeq hBbad)⟩
