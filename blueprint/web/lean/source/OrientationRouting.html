  16 def reversedMultiplicity {V : Type*} (M T : V → V → ℕ) (x y : V) : ℕ :=
  17   M x y - T x y + T y x

  19 theorem reversedMultiplicity_underlying
  20     {V : Type*} (M T : V → V → ℕ) (hT : ∀ x y, T x y ≤ M x y) (x y : V) :
  21     reversedMultiplicity M T x y + reversedMultiplicity M T y x = M x y + M y x := by
  22   have hxy := hT x y
  23   have hyx := hT y x
  24   unfold reversedMultiplicity
  25   omega

  27 theorem reversedMultiplicity_outdegree
  28     {V : Type*} [Fintype V] (M T : V → V → ℕ)
  29     (hT : ∀ x y, T x y ≤ M x y) (x : V) :
  30     (∑ y, reversedMultiplicity M T x y) + (∑ y, T x y) =
  31       (∑ y, M x y) + (∑ y, T y x) := by
  32   simp only [reversedMultiplicity, Finset.sum_add_distrib]
  33   rw [Finset.sum_tsub_distrib Finset.univ (fun y _ => hT x y)]
  34   have hh : ∑ y, T x y ≤ ∑ y, M x y := Finset.sum_le_sum (fun y _ => hT x y)
  35   omega

  37 theorem reversedMultiplicity_changed_degree
  38     {V : Type*} [Fintype V] (M T : V → V → ℕ) (x : V) :
  39     (Finset.univ.filter (fun y => reversedMultiplicity M T x y ≠ M x y ∨
  40       reversedMultiplicity M T y x ≠ M y x)).card ≤
  41       (∑ y, T x y) + (∑ y, T y x) := by
  42   classical
  43   rw [Finset.card_filter, ← Finset.sum_add_distrib]

  52 theorem orientationMatrix_total

  54     (hphysical : ∀ x y, M x y + M y x = if G.Adj x y then 1 else 0) :

  72 matrix represents an orientation because the physical-edge equality is
  73 preserved. Its changed-edge degree is measured on the physical vertices. -/
