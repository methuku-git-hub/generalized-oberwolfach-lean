  13 theorem nearUniform_random_demands {X Y E : Type*}
  14     [Fintype X] [Fintype Y] [Fintype E] [DecidableEq E] [DecidableEq X] [DecidableEq Y]
  15     (tail : E → X) (head : E → Y) (f : X → ℕ) (g : Y → ℕ)
  16     (δ M h p : ℝ) (hp : 0 < p) (hp1 : p ≤ 1)
  17     (ha : 0 < Fintype.card X) (hb : 0 < Fintype.card Y)
  18     (hδ : 0 ≤ δ) (hδsmall : δ < 1 / 4) (hM : 0 ≤ M)
  19     (hMC : M / p ≤ h * Fintype.card X * Fintype.card Y / 2)
  20     (hftotal : ∑ x, (f x : ℝ) = M) (hgtotal : ∑ y, (g y : ℝ) = M)
  21     (hf : ∀ x, (f x : ℝ) ≤ (1 + δ) * M / Fintype.card X)
  22     (hg : ∀ y, (g y : ℝ) ≤ (1 + δ) * M / Fintype.card Y)
  23     (hlarge : ∀ (S : Finset X) (T : Finset Y),
  24       1 - 2 * δ < (S.card : ℝ) / Fintype.card X + (T.card : ℝ) / Fintype.card Y →
  25       h * Fintype.card X * Fintype.card Y *
  26         min ((S.card : ℝ) / Fintype.card X) ((T.card : ℝ) / Fintype.card Y) ≤
  27       ((Finset.univ.filter (fun e => tail e ∈ S ∧ head e ∈ T)).card : ℝ)) :
  28     ∃ ν : FiniteLaw (E → Bool),
  29       (∀ D, 0 < ν.mass D →
  30         (∀ x, (∑ e, if tail e = x ∧ D e = true then (1 : ℕ) else 0) = f x) ∧
  31         (∀ y, (∑ e, if head e = y ∧ D e = true then (1 : ℕ) else 0) = g y)) ∧
  32       (∀ e, ν.probability (fun D => D e = true) ≤ p) := by
  33   obtain ⟨q, hq, hleft, hright⟩ := bipartite_nearUniformFractionalDegrees
  34     tail head (fun x => f x / p) (fun y => g y / p)
  35     ha hb hδ (le_refl δ) hδsmall (div_nonneg hM hp.le) hMC
  36     (by rw [← Finset.sum_div, hftotal]) (by rw [← Finset.sum_div, hgtotal])
  37     (fun x => div_nonneg (Nat.cast_nonneg _) hp.le)
  38     (fun y => div_nonneg (Nat.cast_nonneg _) hp.le)
  39     (by
  40       intro x
  41       convert div_le_div_of_nonneg_right (hf x) hp.le using 1 <;> ring)
  42     (by
  43       intro y
  44       convert div_le_div_of_nonneg_right (hg y) hp.le using 1 <;> ring)
  45     hlarge
  46   let z (e : E) := p * q e
  47   have hz (e : E) : 0 ≤ z e ∧ z e ≤ p := by
  48     refine ⟨mul_nonneg hp.le (hq e).1, ?_⟩
  49     calc
  50       p * q e ≤ p * 1 := mul_le_mul_of_nonneg_left (hq e).2 hp.le
  51       _ = p := mul_one p
  52   have hzl (x : X) : (∑ e, if tail e = x then z e else 0) = f x := by
  53     have hh : (∑ e, if tail e = x then z e else 0) =
  54         p * (∑ e, if tail e = x then q e else 0) := by
  55       simp only [Finset.mul_sum, mul_ite, mul_zero, z]
  56     rw [hh, hleft, mul_div_cancel₀ _ hp.ne']
  57   have hzr (y : Y) : (∑ e, if head e = y then z e else 0) = g y := by
  58     have hh : (∑ e, if head e = y then z e else 0) =
  59         p * (∑ e, if head e = y then q e else 0) := by
  60       simp only [Finset.mul_sum, mul_ite, mul_zero, z]
  61     rw [hh, hright, mul_div_cancel₀ _ hp.ne']
  62   obtain ⟨ν, hν, he⟩ := bipartite_degree_rounding tail head f g z
  63     (fun e => ⟨(hz e).1, (hz e).2.trans hp1⟩)
  64     (by
  65       intro x
  66       convert hzl x using 1
  67       apply Finset.sum_congr rfl
  68       intro e _
  69       split_ifs <;> rfl)
  70     (by
  71       intro y
  72       convert hzr y using 1
  73       apply Finset.sum_congr rfl
  74       intro e _
  75       split_ifs <;> rfl)
  76   refine ⟨ν, ?_, fun e => (he e).le.trans (hz e).2⟩
  77   intro D hD
  78   obtain ⟨hl, hr⟩ := hν D hD
  79   constructor
  80   · intro x
  81     convert hl x using 1
  82     apply Finset.sum_congr rfl
  83     intro e _
  84     split_ifs <;> rfl
  85   · intro y

  88     intro e _
