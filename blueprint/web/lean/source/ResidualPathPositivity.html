  12 theorem positive_transition_path_exists {X : Type*} [Fintype X]
  13     (R : ℕ → X → X → ℝ) (hR : ∀ i, IsTransition (R i))
  14     (i k : ℕ) (x z : X) (hT : 0 < transitionWalk R i k x z) :
  15     ∃ p : Fin (k+1) → X, p 0=x ∧ p (Fin.last k)=z ∧
  16       ∀ j : Fin k, 0 < R (i+j.val) (p j.castSucc) (p j.succ) := by
  17   rw [← transitionPathWeight_endpoint_marginal R i k x z] at hT
  18   have hn (p : Fin (k+1) → X) (_ : p ∈ Finset.univ) :
  19       0 ≤ (if p 0=x ∧ p (Fin.last k)=z then transitionPathWeight R i k p else 0) := by
  20     split_ifs
  21     · exact transitionPathWeight_nonneg R (fun i => (hR i).1) i k p
  22     · exact le_refl _
  23   obtain ⟨p,_,hp⟩ := (Finset.sum_pos_iff_of_nonneg hn).mp hT
  24   split_ifs at hp with he
  25   · exact ⟨p,he.1,he.2,(transitionPathWeight_pos_iff R (fun i => (hR i).1) i k p).mp hp⟩
  26   · exact False.elim (lt_irrefl 0 hp)

  28 theorem pathEndpointMatrix_pos_of_path {X : Type*} [Fintype X] {k : ℕ}
  29     (w : (Fin (k+1) → X) → ℝ) (hw : ∀ p, 0 ≤ w p)
  30     (x z : X) (p : Fin (k+1) → X) (hp : p 0=x ∧ p (Fin.last k)=z) (hwpos : 0 < w p) :
  31     0 < pathEndpointMatrix w x z := by
  32   have hsingle : w p ≤ pathEndpointMatrix w x z := by
  33     unfold pathEndpointMatrix
  34     have hh := Finset.single_le_sum (s := Finset.univ)
  35       (f := fun q => if q 0=x ∧ q (Fin.last k)=z then w q else 0)
  36       (by intro q _; split_ifs <;> first | exact hw q | exact le_refl _) (Finset.mem_univ p)
  37     simpa only [if_pos hp] using hh
  38   exact hwpos.trans_le hsingle

  40 theorem halfResidualPathWeight_pos {X : Type*} [Fintype X] {k : ℕ} (hk : 0 < k)
  41     (a : Fin k → X → X → ℝ) (d : Fin (k+1) → X → ℝ)
  42     (ρ : (Fin (k+1) → X) → ℝ)
  43     (ha : ∀ j x y, 0 ≤ a j x y) (hd : ∀ j x, 0 < d j x)
  44     (hrow : ∀ j x, ∑ y, a j x y=d j.castSucc x)
  45     (hcol : ∀ j y, ∑ x, a j x y=d j.succ y)
  46     (hρ : ∀ p, 0 ≤ ρ p)
  47     (hreserve : ∀ j x y, pathEdgeLoad ρ j x y ≤ (11/27)*a j x y)
  48     (p : Fin (k+1) → X) (hp : ∀ j, 0 < a j (p j.castSucc) (p j.succ)) :
  49     0 < halfResidualPathWeight a d ρ p := by
  50   have hrd : ∀ j x, 0 < residualIntervalDegree d ρ j x := by
  51     intro j x
  52     exact (half_pos (hd j x)).trans_le
  53       ((residualIntervalDegree_bounds hk a d ρ hρ hd hrow hcol hreserve j x).1)
  54   have hra : ∀ j, 0 < residualIntervalCapacity a ρ j (p j.castSucc) (p j.succ) := by
  55     intro j
  56     exact (half_pos (hp j)).trans_le
  57       ((residual_path_capacity_bounds a ρ ha hρ hreserve j _ _).1)
  58   exact div_pos (capacityPathWeight_pos _ _ p hrd hra) (by norm_num)
