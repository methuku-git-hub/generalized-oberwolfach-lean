  10 theorem finite_sequence_simple_walk
  11     {V:Type*} (G:SimpleGraph V) (n:ℕ) (f:Fin (n+1)↪V)
  12     (hadj:∀j:Fin n,G.Adj (f ⟨j.val,by omega⟩) (f ⟨j.val+1,by omega⟩)):
  13     ∃p:G.Walk (f ⟨0,by omega⟩) (f ⟨n,by omega⟩),
  14       p.IsPath ∧ p.length=n ∧ p.support=List.ofFn f := by
  15   induction n with
  16   | zero =>
  17     refine ⟨SimpleGraph.Walk.nil,SimpleGraph.Walk.IsPath.nil, rfl, ?_⟩
  18     simp [List.ofFn_succ]
  19   | succ n ih =>
  20     let tail:Fin (n+1)↪V := (Fin.succEmb (n+1)).trans f
  21     have ht:∀j:Fin n,G.Adj (tail ⟨j.val,by omega⟩) (tail ⟨j.val+1,by omega⟩):=by
  22       intro j
  23       have hh:=hadj ⟨j.val+1,by omega⟩
  24       simpa [tail,Fin.succEmb,Fin.succ, Nat.add_assoc, Nat.add_comm, Nat.add_left_comm] using hh
  25     obtain ⟨p,hp,hlen,hsupp⟩:=ih tail ht
  26     have h0:G.Adj (f ⟨0,by omega⟩) (tail ⟨0,by omega⟩):=by
  27       simpa [tail,Fin.succEmb,Fin.succ] using hadj ⟨0,by omega⟩
  28     have hnot:f ⟨0,by omega⟩∉p.support:=by
  29       rw [hsupp]
  30       have hnod: (List.ofFn f).Nodup:=List.nodup_ofFn_ofInjective f.injective
  31       have ht':List.ofFn f=f ⟨0,by omega⟩::List.ofFn tail:=by
  32         simp [List.ofFn_succ,tail,Fin.succEmb]
  33       rw [ht'] at hnod
  34       exact (List.nodup_cons.mp hnod).1
  35     refine ⟨p.cons h0, hp.cons hnot, ?_, ?_⟩
  36     · simpa [hlen]
  37     · rw [SimpleGraph.Walk.support_cons,hsupp]
  38       simp [List.ofFn_succ,tail,Fin.succEmb]
  39 end Oberwolfach
