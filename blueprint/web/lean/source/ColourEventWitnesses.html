  14 def colourAssignedIn {V C : Type*} (Q : Finset V) (σ : V → C×Bool) (c : C) : Prop :=

  21 theorem colourAssignedIn_witness {V C : Type*}
  22     (Q : Finset V) (σ : V → C×Bool) (c : C) (h : colourAssignedIn Q σ c) :
  23     ∃ W : Finset V,W.card≤1 ∧ (∀ i∈W,(σ i).1=c) ∧
  24       ∀ τ,(∀ i∈W,σ i=τ i) → colourAssignedIn Q τ c := by
  25   classical
  26   obtain ⟨v,hv,hc⟩ := h
  27   refine ⟨{v},by simp,?_,?_⟩
  28   · intro i hi
  29     simpa only [Finset.mem_singleton.mp hi] using hc
  30   · intro τ he
  31     exact ⟨v,hv,(congrArg Prod.fst (he v (Finset.mem_singleton_self _))).symm.trans hc⟩

  33 theorem colourRejectedIn_witness {V C : Type*}
  34     (G : SimpleGraph V) (Q : Finset V) (σ : V → C×Bool) (c : C)
  35     (h : colourRejectedIn G Q σ c) :
  36     ∃ W : Finset V,W.card≤2 ∧ (∀ i∈W,(σ i).1=c) ∧
  37       ∀ τ,(∀ i∈W,σ i=τ i) → colourRejectedIn G Q τ c := by
  38   classical
  39   obtain ⟨v,hv,hc,hbad⟩ := h
  40   by_cases hcoin : (σ v).2=true
  41   · have hex : ∃ u,G.Adj v u ∧ (σ u).1=(σ v).1 := by
  42       by_contra hn
  43       apply hbad
  44       refine ⟨hcoin,?_⟩
  45       intro u hu he
  46       exact hn ⟨u,hu,he⟩
  47     obtain ⟨u,hu,huc⟩ := hex
  48     refine ⟨{v,u},by simpa using Finset.card_insert_le v ({u} : Finset V),?_,?_⟩
  49     · intro i hi
  50       simp only [Finset.mem_insert,Finset.mem_singleton] at hi
  51       rcases hi with hi|hi
  52       · simpa only [hi] using hc
  53       · simpa only [hi] using huc.trans hc
  54     · intro τ he
  55       have hvτ := he v (Finset.mem_insert_self _ _)
  56       have huτ := he u (Finset.mem_insert_of_mem (Finset.mem_singleton_self _))

  58       intro hkeep
  59       exact hkeep.2 u hu ((congrArg Prod.fst huτ).symm.trans

  61   · refine ⟨{v},by simp,?_,?_⟩
  62     · intro i hi
  63       simpa only [Finset.mem_singleton.mp hi] using hc
  64     · intro τ he
  65       have hvτ := he v (Finset.mem_singleton_self _)
  66       refine ⟨v,hv,(congrArg Prod.fst hvτ).symm.trans hc,?_⟩
  67       intro hkeep

  70 theorem witnessed_count_threshold
  71     {I J Ω C : Type*} [Fintype J] [DecidableEq I]
  72     (P : J → (I → Ω) → Prop) (Q : Finset J) (colour : Ω → C)
  73     (x : I → Ω) (c : C) {k a : ℕ}
  74     (h : ∀ j∈Q,P j x →
  75       ∃ W : Finset I,W.card≤k ∧ (∀ i∈W,colour (x i)=c) ∧
  76         ∀ y,(∀ i∈W,x i=y i) → P j y)
  77     (ha : a≤(Q.filter (fun j=>P j x)).card) :
  78     ∃ W : Finset I,W.card≤k*Q.card ∧ (∀ i∈W,colour (x i)=c) ∧
  79       ∀ y,(∀ i∈W,x i=y i) → a≤(Q.filter (fun j=>P j y)).card := by
  80   classical
  81   let T := Q.filter (fun j=>P j x)
  82   let W : J → Finset I := fun j=>
  83     if hj : j∈T then Classical.choose (h j (Finset.mem_filter.mp hj).1
  84       (Finset.mem_filter.mp hj).2) else ∅
  85   have hW (j : J) (hj : j∈T) :
  86       (W j).card≤k ∧ (∀ i∈W j,colour (x i)=c) ∧
  87         ∀ y,(∀ i∈W j,x i=y i) → P j y := by
  88     dsimp only [W]
  89     rw [dif_pos hj]
  90     exact Classical.choose_spec (h j (Finset.mem_filter.mp hj).1
  91       (Finset.mem_filter.mp hj).2)
  92   refine ⟨T.biUnion W,?_,?_,?_⟩
  93   · calc
  94       _ ≤ ∑ j∈T,(W j).card := Finset.card_biUnion_le
  95       _ ≤ ∑ _j∈T,k := Finset.sum_le_sum (fun j hj=>(hW j hj).1)
  96       _ = k*T.card := by simp [mul_comm]
  97       _ ≤ k*Q.card := Nat.mul_le_mul_left _ (Finset.card_le_card (Finset.filter_subset _ _))
  98   · intro i hi
  99     obtain ⟨j,hj,hi⟩ := Finset.mem_biUnion.mp hi
 100     exact (hW j hj).2.1 i hi
 101   · intro y he
 102     apply ha.trans
 103     apply Finset.card_le_card
 104     intro j hj
 105     refine Finset.mem_filter.mpr ⟨(Finset.mem_filter.mp hj).1,?_⟩
 106     exact (hW j hj).2.2 y (fun i hi=>he i (Finset.mem_biUnion.mpr ⟨j,hj,hi⟩))
