   9 theorem finRotate_iterate_val (k j : ℕ) (i : Fin (k + 2)) :
  10     ((finRotate (k + 2))^[j] i).val = (i.val + j) % (k + 2) := by
  11   induction j with
  12   | zero => simp [Nat.mod_eq_of_lt i.isLt]
  13   | succ j ih =>
  14     rw [Function.iterate_succ_apply', finRotate_apply]
  15     simp only [Fin.val_add, Fin.val_one, ih]
  16     rw [Nat.mod_add_mod, Nat.add_assoc]

  18 def cyclicChainOrder {k : ℕ} (s : Fin (k + 2)) : Equiv.Perm (Fin (k + 2)) :=
  19   Equiv.addLeft (finRotate (k + 2) s)

  21 theorem cyclicChainOrder_zero {k : ℕ} (s : Fin (k + 2)) :
  22     cyclicChainOrder s 0 = finRotate (k + 2) s := by

  25 theorem cyclicChainOrder_last {k : ℕ} (s : Fin (k + 2)) :
  26     cyclicChainOrder s (Fin.last (k + 1)) = s := by
  27   change finRotate (k + 2) s + Fin.last (k + 1) = s
  28   rw [finRotate_apply, ← Fin.neg_last (k + 1), add_assoc, neg_add_cancel, add_zero]

  30 theorem cyclicChainOrder_succ {k : ℕ} (s : Fin (k + 2)) (i : Fin (k + 1)) :
  31     cyclicChainOrder s i.succ = finRotate (k + 2) (cyclicChainOrder s i.castSucc) := by
  32   change finRotate (k + 2) s + i.succ = finRotate (k + 2) (finRotate (k + 2) s + i.castSucc)
  33   rw [finRotate_apply (finRotate (k + 2) s + i.castSucc)]
  34   calc
  35     _ = finRotate (k + 2) s + (i.castSucc + 1) :=
  36       congrArg (fun a => finRotate (k + 2) s + a) (show i.castSucc + 1 = i.succ from Fin.coeSucc_eq_succ).symm
  37     _ = _ := (add_assoc _ _ _).symm

  39 theorem cyclicChainOrder_iterate {k : ℕ} (s i : Fin (k + 2)) :
  40     cyclicChainOrder s i = (finRotate (k + 2))^[i.val] (finRotate (k + 2) s) := by
  41   apply Fin.ext
  42   rw [finRotate_iterate_val]
  43   rfl

  45 theorem finRotate_complementary_closes (k : ℕ) (s : Fin (k + 2)) :

  47   simpa only [Fin.val_last, cyclicChainOrder_last] using
  48     (cyclicChainOrder_iterate s (Fin.last (k + 1))).symm

  50 theorem cyclicChainOrder_positions {k : ℕ} (s i : Fin (k + 2)) (hi : i ≠ s) :
  51     ∃ e : Fin (k + 1),
  52       (cyclicChainOrder s).symm i = e.castSucc ∧
  53       (cyclicChainOrder s).symm (finRotate (k + 2) i) = e.succ ∧
  54       (finRotate (k + 2))^[e.val] (finRotate (k + 2) s) = i := by
  55   let p := (cyclicChainOrder s).symm i
  56   have hlast : p ≠ Fin.last (k + 1) := by
  57     intro hp
  58     have hh := congrArg (cyclicChainOrder s) hp
  59     exact hi (by simpa only [p, Equiv.apply_symm_apply, cyclicChainOrder_last] using hh)
  60   have hp : p.val < k + 1 := by
  61     have hle := p.isLt
  62     have hn : p.val ≠ k + 1 := fun h => hlast (Fin.ext h)
  63     omega
  64   let e : Fin (k + 1) := ⟨p.val, hp⟩
  65   have hep : e.castSucc = p := Fin.ext rfl
  66   refine ⟨e, hep.symm, ?_, ?_⟩
  67   · apply (cyclicChainOrder s).injective
  68     rw [Equiv.apply_symm_apply, cyclicChainOrder_succ, hep]
  69     simp only [p, Equiv.apply_symm_apply]
  70   · calc
  71       _ = cyclicChainOrder s e.castSucc := (cyclicChainOrder_iterate s e.castSucc).symm
  72       _ = i := by
  73         rw [hep]
  74         exact (cyclicChainOrder s).apply_symm_apply i
