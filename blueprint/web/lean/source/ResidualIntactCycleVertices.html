   9 theorem intact_cycle_vertices_survive (H:SimpleGraph V) (R:Finset V)
  10     (hisol:∀x,x∉R→∀y,¬H.Adj x y) (a:V) (xs:List V) (hlen:2≤(a::xs).length)
  11     (hkeep:∀ e∈prescribedOrderedPathEdges ((a::xs)++[a]),e∈H.edgeFinset):
  12     (a::xs).toFinset⊆R := by
  13   intro x hx
  14   obtain ⟨y,hy⟩:=ordered_path_vertex_incident_edge (a::xs) x hlen (List.mem_toFinset.mp hx)
  15   let u:=(a::xs).getLast (by simp)
  16   have hlast:(a::xs).getLast?=some u := List.getLast?_eq_getLast_of_ne_nil (by simp)
  17   have hclose:=ordered_path_append_closing_edge (a::xs) a u hlast
  18   have he:s(x,y)∈prescribedOrderedPathEdges ((a::xs)++[a]) := by
  19     rw [hclose]
  20     exact Finset.mem_insert_of_mem hy
  21   have hadj:H.Adj x y := by
  22     have hh:=hkeep s(x,y) he
  23     simpa only [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using hh
  24   by_contra hn
  25   exact hisol x hn y hadj
