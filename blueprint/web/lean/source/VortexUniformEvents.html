  14 theorem vortex_exponential_common_event (C c ε χ : ℝ) (K : ℕ)
  15     (hc : 0<c) (hε : 0<ε) (hε1 : ε<1) (hχ : 0<χ) :
  16     ∃ M : ℕ,1≤M ∧ ∀ (Ω : Type) [Fintype Ω] (μ : FiniteLaw Ω)
  17       (n : ℕ→ℕ) (k : ℕ),
  18       (∀ i≤k,M≤n i) →
  19       (∀ i<k,(n (i+1):ℝ)≤ε*n i) →
  20       ∀ Good : ℕ→Ω→Prop,
  21       (∀ i≤k,1-C*(n i:ℝ)^K*Real.exp (-c*n i)≤μ.probability (Good i)) →
  22       1-χ≤μ.probability (fun ω=>∀ i≤k,Good i ω) := by
  23   obtain ⟨N,hN⟩ := eventually_stretchedExp_le_inv_pow C K hc (p:=1) (by norm_num) 1
  24   let M := max 1 (max N (Nat.ceil (1/(χ*(1-ε)))))
  25   refine ⟨M,le_max_left _ _,?_⟩
  26   intro Ω _ μ n k hn hstep Good hGood
  27   have hMnat : 1≤M := le_max_left _ _
  28   have hM : (0:ℝ)<M := by exact_mod_cast (show 0<M by omega)
  29   have hnn (i : ℕ) (hi : i≤k) : (0:ℝ)<n i := by exact_mod_cast (show 0<n i by have := hn i hi;omega)
  30   have hsum := vortex_fixed_domain_common_event μ ε M hε hε1 hM (fun i=>(n i:ℝ)) k hnn hstep
  31     (by exact_mod_cast hn k (le_refl _)) Good (by
  32       intro i hi
  33       have hnN : N≤n i := by have := hn i hi;dsimp [M] at *;omega
  34       have htail := hN (n i) hnN
  35       simp only [Real.rpow_natCast,Real.rpow_one,pow_one] at htail
  36       linarith only [hGood i hi,htail])
  37   have hceil : 1/(χ*(1-ε))≤(M:ℝ) :=
  38     (Nat.le_ceil _).trans (by exact_mod_cast (show Nat.ceil (1/(χ*(1-ε)))≤M by dsimp [M];omega))
  39   have hden : 0<χ*(1-ε) := by positivity
  40   have hh := (div_le_iff₀ hden).mp hceil
  41   have hbound : 1/((1-ε)*(M:ℝ))≤χ := by
  42     apply (div_le_iff₀ (by positivity : 0<(1-ε)*(M:ℝ))).mpr
  43     nlinarith only [hh]
  44   linarith only [hsum,hbound]
