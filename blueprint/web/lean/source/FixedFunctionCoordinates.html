  12 theorem finite_function_fixed_value_card {I V : Type*} [Fintype I] [DecidableEq I] [Fintype V]
  13     (i : I) (v : V) :
  14     Fintype.card {f : I → V // f i=v}=Fintype.card V^(Fintype.card I-1) := by
  15   let e : {f : I → V // f i=v} ≃ ({j : I // j≠i} → V) := {
  16     toFun f := fun j => f.val j.val
  17     invFun g := ⟨fun j => if h : j=i then v else g ⟨j,h⟩,by simp⟩
  18     left_inv := by
  19       intro f
  20       apply Subtype.ext
  21       funext j
  22       by_cases h : j=i
  23       · subst j;simp [f.property]
  24       · simp [h]
  25     right_inv := by intro g;funext j;simp [j.property] }
  26   simpa only [Fintype.card_fun,Fintype.card_subtype_compl,Fintype.card_unique] using Fintype.card_congr e
