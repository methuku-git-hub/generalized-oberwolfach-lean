  12 theorem actual_restricted_edge_image_in_host
  13     {X V K J:Type} [Fintype X] [Fintype V] [DecidableEq X] [DecidableEq V] [DecidableEq K] [DecidableEq J]
  14     (F:SimpleGraph X) (hF:F.IsRegularOfDegree 2) (H:SimpleGraph V)
  15     (a:ActualResidualData (K:=K) (J:=J) F)
  16     (hCycle:a.t.biUnion (fun j=>(a.C j).support.toFinset)⊆a.Cembed)
  17     (D:Finset X) (finalEdges:Finset (Sym2 X))
  18     (fC:{x // x∉D}↪V) (ψ:X→V) (Φ:{x // x∈a.remaining}→V)
  19     (hold:∀(x:{x // x∉D}),x.val∈a.oldDomain→fC x=ψ x.val)
  20     (hnew:∀(x:{x // x∉D}) (hx:x.val∈a.remaining),fC x=Φ ⟨x.val,hx⟩)
  21     (Pack:Finset (Sym2 V)) (hOld:(a.deletedEdges\finalEdges).image (Sym2.map ψ)⊆Pack)
  22     (hPack:Pack⊆H.edgeFinset)
  23     (Dsrc:X→X→Prop) (hOrient:∀x y,F.Adj x y→Dsrc x y ∨ Dsrc y x)
  24     (hArc:∀x y:{x // x∈a.remaining},Dsrc x.1 y.1→a.graph.Adj x.1 y.1→H.Adj (Φ x) (Φ y)):
  25     (((F.deleteEdges (↑finalEdges:Set (Sym2 X))).induce {x | x∉D}).edgeFinset).image (Sym2.map fC)⊆H.edgeFinset :=by
  26   intro e he
  27   have hcover:=actual_restricted_edge_image_cover F hF a hCycle D finalEdges fC ψ Φ hold hnew he
  28   rcases Finset.mem_union.mp hcover with hOld'|hNew
  29   · exact hPack (hOld hOld')
  30   · obtain ⟨e',he',rfl⟩:=Finset.mem_image.mp hNew
  31     induction e' using Sym2.ind with
  32     | _ x y =>
  33       have hadj:a.graph.Adj x.1 y.1:=SimpleGraph.mem_edgeFinset.mp he'
  34       have hFadj:F.Adj x.1 y.1:=(a.graph_facts hF).1 hadj
  35       rcases hOrient _ _ hFadj with hxy|hyx
  36       · exact SimpleGraph.mem_edgeFinset.mpr (hArc x y hxy hadj)
  37       · exact SimpleGraph.mem_edgeFinset.mpr ((hArc y x hyx hadj.symm).symm)
  38 end Oberwolfach
