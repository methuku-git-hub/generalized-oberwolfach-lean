  13 def ordinaryOldRim (G : SimpleGraph V) : SimpleGraph (V⊕X) :=
  14   G.map (⟨Sum.inl,Sum.inl_injective⟩ : V ↪ V⊕X)
  15 
  16 def ordinaryOldSpokes (H : V → U → Prop) : V⊕X → U⊕I → Prop
  17   | .inl a,.inl b => H a b
  18   | _,_ => False
  19 
  20 theorem ordinaryOldRim_old_old (G : SimpleGraph V) (a b : V) :
  21     (ordinaryOldRim (X:=X) G).Adj (.inl a) (.inl b) ↔ G.Adj a b := by
  22   exact SimpleGraph.map_adj_apply (G:=G) (f:=(⟨Sum.inl,Sum.inl_injective⟩ : V ↪ V⊕X))
  23 
  24 theorem ordinaryOldRim_old_private (G : SimpleGraph V) (a : V) (b : X) :
  25     (ordinaryOldRim G).Adj (.inl a) (.inr b) ↔ False := by
  26   simp [ordinaryOldRim,SimpleGraph.map_adj']
  27 
  28 theorem ordinaryOldRim_private_any (G : SimpleGraph V) (a : X) (b : V⊕X) :
  29     (ordinaryOldRim G).Adj (.inr a) b ↔ False := by
  30   simp [ordinaryOldRim,SimpleGraph.map_adj']

  36 theorem ordinaryOldRim_degree_old (G : SimpleGraph V) (a : V) :
  37     (ordinaryOldRim (X:=X) G).degree (.inl a)=G.degree a := by
  38   rw [graph_degree_indicator_sum,Fintype.sum_sum_type]
  39   simp only [ordinaryOldRim_old_old,ordinaryOldRim_old_private,naturalIndicator_false,Finset.sum_const_zero,add_zero]
  40   exact (graph_degree_indicator_sum G a).symm

  47 theorem ordinaryOldSpokes_row_old (H : V → U → Prop) (a : V) :
  48     (Finset.univ.filter (ordinaryOldSpokes (X:=X) (I:=I) H (.inl a))).card=

  54     (Finset.univ.filter (ordinaryOldSpokes (I:=I) H (.inr a))).card=0 := by
  55   rw [card_filter_naturalIndicator,Fintype.sum_sum_type]

  58 theorem ordinaryOldSpokes_hub_old (H : V → U → Prop) (b : U) :
  59     (Finset.univ.filter (fun a => ordinaryOldSpokes (X:=X) (I:=I) H a (.inl b))).card=
  60       (Finset.univ.filter (fun a => H a b)).card := by

  64 theorem ordinaryOldSpokes_hub_private (H : V → U → Prop) (b : I) :

  69 theorem ordinaryOldPair_balance (G : SimpleGraph V) (H : V → U → Prop)
  70     (h : ∀ a,G.degree a=2*(Finset.univ.filter (H a)).card) :
  71     ∀ a,(ordinaryOldRim (X:=X) G).degree a=2*(Finset.univ.filter (ordinaryOldSpokes (I:=I) H a)).card := by
  72   intro a
  73   cases a with
  74   | inl a => rw [ordinaryOldRim_degree_old,ordinaryOldSpokes_row_old,h]
  75   | inr a => rw [ordinaryOldRim_degree_private,ordinaryOldSpokes_row_private]
  76 
  77 theorem ordinaryOldPair_hub_divisibility (H : V → U → Prop) (r : ℕ)
  78     (h : ∀ b,r ∣ (Finset.univ.filter (fun a => H a b)).card) :
  79     ∀ b,r ∣ (Finset.univ.filter (fun a => ordinaryOldSpokes (X:=X) (I:=I) H a b)).card := by
  80   intro b
  81   cases b with
  82   | inl b => rw [ordinaryOldSpokes_hub_old];exact h b
  83   | inr b => rw [ordinaryOldSpokes_hub_private];exact dvd_zero r
