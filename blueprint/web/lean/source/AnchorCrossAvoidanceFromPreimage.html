   9 theorem anchor_cross_avoidance_from_piece_preimage
  10     {X V:Type} [DecidableEq X] (A P:Finset X)
  11     (f:{x // x∈A}↪V) (g:{x // x∈P}↪V)
  12     (hagree:∀x (hx:x∈A) (hy:x∈P),f ⟨x,hx⟩=g ⟨x,hy⟩)
  13     (hpreimage:∀y:{x // x∈A},f y∈Set.range g→y.val∈P):
  14     ∀x (hx:x∈P) (hxa:x∉A) (y:{x // x∈A}),g ⟨x,hx⟩≠f y:=by
  15   intro x hx hxa y he
  16   have hy:y.val∈P:=hpreimage y ⟨⟨x,hx⟩,he⟩
  17   have hh:g ⟨x,hx⟩=g ⟨y.val,hy⟩:=he.trans (hagree y.val y.property hy)
  18   have hxy:x=y.val:=congrArg Subtype.val (g.injective hh)
  19   exact hxa (hxy ▸ y.property)
  20 end Oberwolfach
