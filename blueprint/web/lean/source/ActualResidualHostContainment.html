  14 theorem prescribed_piece_images_in_host
  15     (F:∀i,SimpleGraph (X i)) (a:∀i,ActualResidualData (K:=K) (J:=J) (F i))
  16     (pieces:∀i,List (PrescribedSourcePiece {x // x∈(a i).remaining}))
  17     (Φ:∀i,{x // x∈(a i).remaining}→V) (Dstar:V→V→Prop)
  18     (harc:∀i x y,prescribedDependentDigraph pieces i x y→Dstar (Φ i x) (Φ i y))
  19     (G:SimpleGraph V) (hhost:∀x y,Dstar x y→G.Adj x y):
  20     ∀i,((prescribedDependentGraph pieces i).edgeFinset).image (Sym2.map (Φ i))⊆G.edgeFinset :=by
  21   intro i e he
  22   obtain ⟨s,hs,rfl⟩:=Finset.mem_image.mp he
  23   induction s using Sym2.ind with
  24   | _ x y =>
  25     have hxy:(prescribedDependentGraph pieces i).Adj x y:=by
  26       simpa only [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using hs
  27     obtain h|h:=prescribed_dependent_source_representation pieces i x y hxy
  28     · have hg:=hhost (Φ i x) (Φ i y) (harc i x y h)
  29       simpa only [Sym2.map_mk,SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using hg
  30     · have hg: G.Adj (Φ i x) (Φ i y):=(hhost (Φ i y) (Φ i x) (harc i y x h)).symm
  31       simpa only [Sym2.map_mk,SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using hg

  33 theorem actual_glued_factor_images_in_host
  34     (F:∀i,SimpleGraph (X i)) (hF:∀i,(F i).IsRegularOfDegree 2)
  35     (a:∀i,ActualResidualData (K:=K) (J:=J) (F i))
  36     (pieces:∀i,List (PrescribedSourcePiece {x // x∈(a i).remaining}))
  37     (hgraph:∀i,prescribedDependentGraph pieces i=((a i).graph).induce (↑(a i).remaining:Set (X i)))
  38     (hCycle:∀i,(a i).t.biUnion (fun j=>((a i).C j).support.toFinset)⊆(a i).Cembed)
  39     (ψ h:∀i,X i→V) (Φ:∀i,{x // x∈(a i).remaining}→V)
  40     (hold:∀i x,x∈(a i).oldDomain→h i x=ψ i x)
  41     (hrem:∀i (x:{x // x∈(a i).remaining}),h i x.1=Φ i x)
  42     (Dstar:V→V→Prop) (harc:∀i x y,prescribedDependentDigraph pieces i x y→Dstar (Φ i x) (Φ i y))
  43     (Pack:Finset (Sym2 V)) (hOldPack:∀i,((a i).deletedEdges).image (Sym2.map (ψ i))⊆Pack)
  44     (G:SimpleGraph V) (hPackHost:Pack⊆G.edgeFinset) (hhost:∀x y,Dstar x y→G.Adj x y):
  45     ∀i,((F i).edgeFinset).image (Sym2.map (h i))⊆G.edgeFinset :=by
  46   have hnew:=prescribed_piece_images_in_host F a pieces Φ Dstar harc G hhost
  47   intro i
  48   rw [(a i).glued_factor_edge_images (hF i) (hCycle i) (ψ i) (h i) (Φ i) (hold i) (hrem i)]
  49   apply Finset.union_subset
  50   · exact (hOldPack i).trans hPackHost
  51   · convert hnew i using 1
  52     congr 1
  53     rw [hgraph i]
  54     apply graph_edge_finset_instance_eq
