  18     (h : markedPhysicalRim P π b σ a x y=1) :
  19     if a then ∃ e : MarkedSampleArc P π b σ true,e.val.1=x ∧ e.val.2=y
  20       else ∃ e : MarkedSampleArc P π b σ false,e.val.1=x ∧ e.val.2=y := by

  23 theorem markedPathRims_rootCycle (hm : 2≤m) (z : Fin (m+1)→V) (side : Bool)
  24     (hsep : PermutationSeparated π b (Fin.cons (if side then v' else v) z))
  25     (hR : MarkedPathRims P π b m j v v' σ z) :
  26     IsRootFirstMarkedCycle
  27       (fun e : MarkedSampleArc P π b σ false=>e.val.1) (fun e=>e.val.2)
  28       (fun e : MarkedSampleArc P π b σ true=>e.val.1) (fun e=>e.val.2)
  29       (m+1) π b (if j then Fin.last (m+1) else 2) (Fin.cons (if side then v' else v) z) := by
  30   refine ⟨hsep,?_⟩
  31   intro t
  32   refine Fin.cases ?_ (fun l=>?_) t
  33   · have hs : (0:Fin (m+2))≠(if j then Fin.last (m+1) else 2) := by
  34       cases j <;> simp [Fin.ext_iff,Fin.val_natCast,Nat.mod_eq_of_lt (show 2<m+2 by omega)]
  35     rw [if_neg hs]
  36     have hrot : finRotate (m+2) (0:Fin (m+2))=(0:Fin (m+1)).succ :=
  37       finRotate_castSucc_eq_succ (m+1) 0
  38     rw [hrot]
  39     exact ⟨⟨(if side then v' else v,z 0),hR (Sum.inr (false,side))⟩,rfl,rfl⟩
  40   · refine Fin.lastCases ?_ (fun i=>?_) l
  41     · have hrot : finRotate (m+2) (Fin.last m).succ=0 := finRotate_last
  42       rw [hrot]
  43       have h := hR (Sum.inr (true,side))
  44       cases j
  45       · have hs : (Fin.last m).succ≠(2:Fin (m+2)) := by
  46           intro he
  47           have hh := congrArg Fin.val he
  48           change m+1=2%(m+2) at hh
  49           rw [Nat.mod_eq_of_lt (by omega)] at hh
  50           omega

  52         exact ⟨⟨(z (Fin.last m),if side then v' else v),h⟩,rfl,rfl⟩
  53       · simp only [ite_true,show (Fin.last m).succ=Fin.last (m+1) from rfl]
  54         exact ⟨⟨(z (Fin.last m),if side then v' else v),h⟩,rfl,rfl⟩
  55     · have hrot : finRotate (m+2) i.castSucc.succ=i.succ.succ :=
  56         finRotate_castSucc_eq_succ (m+1) i.succ
  57       have hmark : i.castSucc.succ=(if j then Fin.last (m+1) else (2:Fin (m+2))) ↔ j=false ∧ i.val=1 := by
  58         cases j
  59         · change (i.castSucc.succ=(2:Fin (m+2))) ↔ (false=false ∧ i.val=1)
  60           simp only [true_and]
  61           constructor
  62           · intro he

  64             change i.val+1=2%(m+2) at hh

  66             omega
  67           · intro he

  71             omega

  74           intro he
  75           have hh := congrArg Fin.val he
