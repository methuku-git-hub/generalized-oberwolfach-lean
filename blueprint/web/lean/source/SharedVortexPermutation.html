  13 theorem shared_vortex_with_permutation_exists {β ε ρ₀ : ℝ} (hb : 0 < β)
  14     (he : 0 < ε) (heq : ε < 1 / 4) (hp : 0 < ρ₀) (h : ℕ) :
  15     ∃ m₀ : ℕ, 1 ≤ m₀ ∧ ∀ (V : Type b) [Fintype V] (A : Type c) [Nonempty A]
  16       (U : A → Type d) [∀ a,Fintype (U a)] (G : SimpleGraph V) (D m : ℕ),
  17       G.IsRegularOfDegree D → (Fintype.card V : ℝ) / 2 ≤ D →
  18       (∀ a, ρ₀ ≤ initialIndexDensity (U a) D) →
  19       (∀ a, initialIndexDensity (U a) D ≤ 1) →
  20       m₀ ≤ m → m ≤ Fintype.card V →
  21       ∃ Q : ∀ a, Vortex (U a) G D β ε m, ∃ π : Equiv.Perm V,
  22         (∀ a i x, π x ∈ (Q a).extendedVertices i ↔ x ∈ (Q a).extendedVertices i) ∧
  23         (∀ a b, (Q a).toVertexVortex = (Q b).toVertexVortex) ∧
  24         (∀ x k, 0 < k → k < h → (π ^ k) x ≠ x) := by
  25   have he1 : ε < 1 := by linarith
  26   obtain ⟨N₁,hN₁,hv⟩ := vertexVortex_exists.{b} (η:=β/2) (by positivity) he he1
  27   obtain ⟨N₂,hN₂,hext⟩ := vortex_extend_fixed_vertices.{b,d} hb he heq hp
  28   let N := max N₁ (max N₂ ⌈8*(h:ℝ)/(3*ε)⌉₊)
  29   refine ⟨N,by dsimp [N]; omega,?_⟩
  30   intro V _ A _ U _ G D m hreg hD hρlo hρhi hm hmn
  31   have hN1 : N₁ ≤ m := by dsimp [N] at hm;omega
  32   have hN2 : N₂ ≤ m := by dsimp [N] at hm;omega
  33   obtain ⟨Qv⟩ := hv V G D m hreg hN1 hmn
  34   have hex : ∀ a, ∃ Q : Vortex (U a) G D β ε m, Q.toVertexVortex=Qv := by
  35     intro a
  36     exact hext V (U a) G D m hreg hD (hρlo a) (hρhi a) hN2 hmn Qv
  37   choose Q hQ using hex
  38   let a₀ : A := Classical.choice ‹Nonempty A›
  39   have hceil : 8 * (h : ℝ) / (3 * ε) ≤ (m : ℝ) :=
  40     (Nat.le_ceil _).trans (by exact_mod_cast (show ⌈8*(h:ℝ)/(3*ε)⌉₊ ≤ m from by dsimp [N] at hm;omega))
  41   have hsize : (h : ℝ) ≤ 3 * ε * m / 8 := by
  42     have hh := (div_le_iff₀ (show 0 < 3 * ε by positivity)).mp hceil
  43     nlinarith
  44   obtain ⟨π,hpres,hlevel,hshort⟩ := (Q a₀).preserving_permutation he.le heq.le h hsize
  45   refine ⟨Q,π,?_,?_,hshort⟩
  46   · intro a i x
  47     have hlen : (Q a).length = (Q a₀).length := by
  48       have ha:=congrArg VertexVortex.length (hQ a)
  49       have hb':=congrArg VertexVortex.length (hQ a₀)
  50       exact ha.trans hb'.symm
  51     have hsets : (Q a).vertices = (Q a₀).vertices := by
  52       have ha:=congrArg VertexVortex.vertices (hQ a)
  53       have hb':=congrArg VertexVortex.vertices (hQ a₀)
  54       exact ha.trans hb'.symm
  55     have hs : (Q a).extendedVertices i = (Q a₀).extendedVertices i := by
  56       simp only [Vortex.extendedVertices,hlen,hsets]
  57     simpa only [hs] using hpres i x
  58   · intro a b
  59     exact (hQ a).trans (hQ b).symm
  60 end Oberwolfach
