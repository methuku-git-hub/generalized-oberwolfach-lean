  23     (hpair:PairwisePhysicalDisjoint E) (hcover:∀ j,E j⊆F.edgeFinset)
  24     (hJ:J.edgeFinset⊆F.edgeFinset) (hedges:Disjoint added F.edgeFinset):
  25     let A':=Function.update A k (A k∪C);
  26     let f':=Function.update f k g;
  27     let T':=T∪({k}:Finset I).product C;

  29     (∀ i x,x∈A i→f' i x=f i x) ∧
  30     (∀ p∈T',p.2∈A' p.1) ∧
  31     (∀ v,((T'.filter (fun p=>familyVertexImage f' p=v)).card:ℝ)≤t+1) ∧
  32     PairwisePhysicalDisjoint (Function.update E k (E k∪added)) ∧ Disjoint added J.edgeFinset := by

  64     have hh:=Finset.mem_product.mp hp
  65     have hk:p.1=k:=by simpa using hh.1

  67     rw [hk]
