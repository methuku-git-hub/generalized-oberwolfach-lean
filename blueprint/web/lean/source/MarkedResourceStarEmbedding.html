  13 theorem markedResourceEmbedding_star_iff (f : V↪V') (g : U↪U')
  14     (A : Finset V) (B : Finset U) (I : Finset V') (J : Finset U')
  15     (hA : ∀ x,f x∈I↔x∈A) (hB : ∀ u,g u∈J↔u∈B)
  16     (e : Finset V⊕(V×U)) :
  17     markedStarResource I J (markedResourceEmbedding f g e)↔markedStarResource A B e := by
  18   cases e with
  19   | inl e =>
  20     change (∃ x∈e.image f,x∈I)↔∃ x∈e,x∈A
  21     constructor
  22     · rintro ⟨y,hy,hi⟩
  23       obtain ⟨x,hx,rfl⟩ := Finset.mem_image.mp hy
  24       exact ⟨x,hx,(hA x).mp hi⟩
  25     · rintro ⟨x,hx,ha⟩
  26       exact ⟨f x,Finset.mem_image.mpr ⟨x,hx,rfl⟩,(hA x).mpr ha⟩
  27   | inr z => exact or_congr (hA z.1) (hB z.2)
