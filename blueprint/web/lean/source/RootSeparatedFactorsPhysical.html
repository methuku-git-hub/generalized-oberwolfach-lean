  11 theorem root_separated_factors_physical {V:Type} [Fintype V] [DecidableEq V]
  12     (G F:SimpleGraph V) (T:Finset V) (r k:ℕ)
  13     (hpair:ConflictFreeRootSeparatedCycleFactorPair G F T r k):
  14     ∃ S:Finset V,S.card=k ∧ Disjoint S T ∧
  15       (∀ x∈S∪T,∀ y∈S∪T,¬F.Adj x y) ∧
  16       ∃ M₁ M₂:Finset (OrdinaryCycle G r),
  17         (∀ C∈M₁,∀ E∈M₁,C≠E→Disjoint (ordinaryCycleVertices G r C) (ordinaryCycleVertices G r E)) ∧
  18         M₁.biUnion (ordinaryCycleVertices G r)=S∪T ∧
  19         (∀ C∈M₂,∀ E∈M₂,C≠E→Disjoint (ordinaryCycleVertices G r C) (ordinaryCycleVertices G r E)) ∧
  20         M₂.biUnion (ordinaryCycleVertices G r)=S ∧
  21         (∀ C∈M₁,∀ E∈M₂,Disjoint C.val.val.edgeFinset E.val.val.edgeFinset) ∧
  22         (∀ C∈M₁,(ordinaryCycleVertices G r C∩T).card≤1):=by
  23   obtain ⟨S,hSc,hdis,hfree,M₁,M₂,hpack₁,hcover₁,hpack₂,hcover₂,hedge,hroot⟩:=hpair
  24   let R:=S∪T
  25   let W:Type:={v:V // v∈R}
  26   let J:SimpleGraph W:=G.induce (R:Set V)
  27   let S':Finset W:=S.subtype (fun v=>v∈R)
  28   let f:W↪V:=Function.Embedding.subtype _
  29   have hQ:J.map f≤G:=by
  30     intro v w h
  31     rw [SimpleGraph.map_adj] at h
  32     obtain ⟨a,b,hab,rfl,rfl⟩:=h
  33     exact hab
  34   let lift:=ordinaryCycleLift (r:=r) f hQ
  35   let M₁':=M₁.image lift
  36   let M₂':=M₂.image lift
  37   have hp₁:=ordinary_cycle_family_lift f hQ M₁ hpack₁
  38   have hp₂:=ordinary_cycle_family_lift f hQ M₂ hpack₂
  39   have huniv:(Finset.univ:Finset W).image f=R:=by
  40     ext v
  41     constructor
  42     · intro hv;obtain ⟨x,_,rfl⟩:=Finset.mem_image.mp hv;exact x.property
  43     · intro hv;exact Finset.mem_image.mpr ⟨⟨v,hv⟩,Finset.mem_univ _,rfl⟩
  44   have hSmap:S'.image f=S:=by
  45     ext v
  46     constructor
  47     · intro hv
  48       obtain ⟨x,hx,rfl⟩:=Finset.mem_image.mp hv
  49       exact Finset.mem_subtype.mp hx
  50     · intro hv
  51       have hvR:v∈R:=Finset.mem_union_left _ hv
  52       exact Finset.mem_image.mpr ⟨⟨v,hvR⟩,Finset.mem_subtype.mpr hv,rfl⟩
  53   have hcover₁':M₁'.biUnion (ordinaryCycleVertices G r)=S∪T:=by
  54     change (M₁.image lift).biUnion (ordinaryCycleVertices G r)=R
  55     rw [hp₁.2,hcover₁,huniv]
  56   have hcover₂':M₂'.biUnion (ordinaryCycleVertices G r)=S:=by
  57     change (M₂.image lift).biUnion (ordinaryCycleVertices G r)=S
  58     rw [hp₂.2,hcover₂,hSmap]
  59   refine ⟨S,hSc,hdis,hfree,M₁',M₂',hp₁.1,hcover₁',hp₂.1,hcover₂',?_,?_⟩
  60   · intro C hC E hE
  61     obtain ⟨C',hC',hCC⟩:=Finset.mem_image.mp hC
  62     obtain ⟨E',hE',hEE⟩:=Finset.mem_image.mp hE
  63     subst C;subst E
  64     exact ordinaryCycleLift_disjoint_edges f hQ C' E' (hedge C' hC' E' hE')
  65   · intro C hC
  66     obtain ⟨C',hC',hCC⟩:=Finset.mem_image.mp hC
  67     subst C
  68     apply Finset.card_le_one.mpr
  69     intro x hx y hy
  70     have hx':=Finset.mem_inter.mp hx
  71     have hy':=Finset.mem_inter.mp hy
  72     rw [ordinaryCycleLift_vertices] at hx' hy'
  73     obtain ⟨x',hx'C,hx⟩:=Finset.mem_image.mp hx'.1
  74     obtain ⟨y',hy'C,hy⟩:=Finset.mem_image.mp hy'.1
  75     have hxS':x'∉S':=by
  76       intro h
  77       have hxS:(f x')∈S:=Finset.mem_subtype.mp h
  78       have hxT:(f x')∈T:=by simpa [hx] using hx'.2
  79       exact (Finset.disjoint_left.mp hdis) hxS hxT
  80     have hyS':y'∉S':=by
  81       intro h
  82       have hyS:(f y')∈S:=Finset.mem_subtype.mp h
  83       have hyT:(f y')∈T:=by simpa [hy] using hy'.2
  84       exact (Finset.disjoint_left.mp hdis) hyS hyT
  85     have hxmem:x'∈ordinaryCycleVertices J r C'∩(Finset.univ\S'):=
  86       Finset.mem_inter.mpr ⟨hx'C,Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,hxS'⟩⟩
  87     have hymem:y'∈ordinaryCycleVertices J r C'∩(Finset.univ\S'):=
  88       Finset.mem_inter.mpr ⟨hy'C,Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,hyS'⟩⟩
  89     have hxy: x'=y':=Finset.card_le_one.mp (hroot C' hC') x' hxmem y' hymem
  90     calc
  91       x=f x':=hx.symm
  92       _=f y':=congrArg f hxy
  93       _=y:=hy
  94 end Oberwolfach
