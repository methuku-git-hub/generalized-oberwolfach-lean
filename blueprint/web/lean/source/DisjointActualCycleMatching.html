  12 theorem disjoint_cycle_families_match
  13     {X V I:Type} [Fintype X] [DecidableEq X] [Fintype V] [DecidableEq V] [Fintype I] [DecidableEq I]
  14     (G:SimpleGraph X) (H:SimpleGraph V) (a:I→X) (p:∀i,G.Walk (a i) (a i))
  15     (hp:∀i,(p i).IsCycle) (u:I→V) (q:∀i,H.Walk (u i) (u i))
  16     (hq:∀i,(q i).IsCycle) (hlen:∀i,(q i).length=(p i).length)
  17     (hpdis:∀i j,i≠j→Disjoint (p i).support.toFinset (p j).support.toFinset)
  18     (hqdis:∀i j,i≠j→Disjoint (q i).support.toFinset (q j).support.toFinset):
  19     ∃φ:{x // x∈Finset.univ.biUnion (fun i=>(p i).tail.support.toFinset)}↪V,
  20       ∀i (k:Fin ((p i).tail.length+1)),
  21         φ ⟨(p i).tail.getVert k.val,
  22           Finset.mem_biUnion.mpr ⟨i,Finset.mem_univ _,List.mem_toFinset.mpr ((p i).tail.getVert_mem_support _)⟩⟩=
  23           (q i).tail.getVert k.val:=by
  24   have hpt:∀i,(p i).tail.IsPath:=fun i=>(hp i).isPath_tail
  25   have hqt:∀i,(q i).tail.IsPath:=fun i=>(hq i).isPath_tail
  26   have hlent:∀i,(q i).tail.length=(p i).tail.length:=by
  27     intro i
  28     simp only [SimpleGraph.Walk.length_tail,hlen]
  29   have hpdt:∀i j,i≠j→Disjoint (p i).tail.support.toFinset (p j).tail.support.toFinset:=by
