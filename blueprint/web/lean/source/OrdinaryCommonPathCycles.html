  14 theorem ordinary_common_path_cycle (Q F : SimpleGraph V) (T : Finset V) (hm : 2 ≤ m)
  15     (x y z : V) (hz : z=x ∨ z=y) (hzT : z∈T) (f : Fin m→V)
  16     (hf : SparseGreedyGood Q F T {x,y} (ordinaryPathRequirements m x y) f) :
  17     ∃ C : LocalOrdinaryCycle Q F T (m+1),localOrdinaryVertices C=insert z (Finset.univ.image f) := by
  18   let g : Fin m→V := fun i=>f (commonPathOrder m hm i)
  19   have hginj : Function.Injective g := hf.1.comp (commonPathOrder m hm).injective
  20   have hout : z∉Set.range g := by
  21     rintro ⟨i,hi⟩
  22     have hh := (hf.2.1 (commonPathOrder m hm i)).2
  23     apply hh
  24     change f (commonPathOrder m hm i)=z at hi
  25     rw [hi]
  26     simpa only [Finset.mem_insert,Finset.mem_singleton] using hz
  27   let e : Fin (m+1) ↪ V := ⟨Fin.cons (α:=fun _=>V) z g,Fin.cons_injective_of_injective hout hginj⟩
  28   have e0 : e 0=z := rfl
  29   have epos (i : Fin (m+1)) (hi : 0 < i.val) :
  30       e i=f ⟨transferColumnOrder (m+1) i.val,by
  31         simpa only [Nat.add_sub_cancel] using transferColumnOrder_bound (m+1) i.val (by omega) hi i.isLt⟩ := by
  32     cases i using Fin.cases with
  33     | zero => simp at hi
  34     | succ j => rfl
  35   have hQ : ∀ i j,(SimpleGraph.cycleGraph (m+1)).Adj i j → Q.Adj (e i) (e j) := by
  36     have step (i j : Fin (m+1)) (hij : i.val+1=j.val) : Q.Adj (e i) (e j) := by
  37       by_cases hi0 : i.val=0
  38       · have hi : i=0 := Fin.ext hi0
  39         have hj1 : j.val=1 := by omega
  40         rw [hi,e0,epos j (by omega)]
  41         apply ordinaryCommonPath_endpoint Q F T x y f hf hm z hz
  42         left;simp only [transferColumnOrder,hj1,ite_true]
  43       · rw [epos i (by omega),epos j (by omega)]
  44         convert ordinaryCommonPath_step Q F T x y f hf hm i.val (by omega) (by have hj:=j.isLt;omega) using 1
  45         congr 2;rw [hij]
  46     have wrap (i j : Fin (m+1)) (hi : i.val=0) (hj : j.val=(m+1)-1) : Q.Adj (e i) (e j) := by
  47       rw [show i=0 from Fin.ext hi,e0,epos j (by omega)]
  48       apply ordinaryCommonPath_endpoint Q F T x y f hf hm z hz
  49       right;simp only [transferColumnOrder];split_ifs <;> omega
  50     intro i j hij
  51     rcases cycleGraph_adj_natural_cases (m+1) (by omega) i j hij with h|h|⟨hi,hj⟩|⟨hj,hi⟩
  52     · exact step i j h
  53     · exact (step j i h).symm
  54     · exact wrap i j hi hj
  55     · exact (wrap j i hj hi).symm
  56   have hT (i : Fin (m+1)) : e i∈T := by
  57     refine Fin.cases hzT (fun j=>?_) i
  58     exact (hf.2.1 (commonPathOrder m hm j)).1
  59   have hF (i j : Fin (m+1)) : ¬F.Adj (e i) (e j) := by
  60     refine Fin.cases ?_ (fun a=>?_) i
  61     · refine Fin.cases (F.loopless.irrefl z) (fun b=>?_) j
  62       exact fun h=>hf.2.2.2.1 (commonPathOrder m hm b) z
  63         (by simpa only [Finset.mem_insert,Finset.mem_singleton] using hz) h.symm
  64     · refine Fin.cases ?_ (fun b=>?_) j
  65       · exact hf.2.2.2.1 (commonPathOrder m hm a) z
  66           (by simpa only [Finset.mem_insert,Finset.mem_singleton] using hz)
  67       · exact hf.2.2.1 (commonPathOrder m hm a) (commonPathOrder m hm b)
  68   obtain ⟨C,hC⟩ := ordinary_cycle_from_embedding Q F T (by omega) e hT hF hQ
  69   refine ⟨C,hC.trans ?_⟩
  70   ext v
  71   simp only [Finset.mem_image,Finset.mem_univ,true_and,Finset.mem_insert]
  72   constructor
  73   · rintro ⟨i,hi⟩
  74     cases i using Fin.cases with
  75     | zero => exact Or.inl hi.symm
  76     | succ j => exact Or.inr ⟨commonPathOrder m hm j,hi⟩
  77   · rintro (rfl|⟨j,hj⟩)
  78     · exact ⟨0,rfl⟩
  79     · obtain ⟨i,hi⟩ := commonPathOrder_surjective m hm j
  80       exact ⟨i.succ,by change f (commonPathOrder m hm i)=v;rw [hi];exact hj⟩
