  11 theorem class_balanced_correction
  12     {V:Type} [Fintype V] [LinearOrder V]
  13     (S₀ S Protected:SimpleGraph V) (bits:Sym2 V→Bool) (m L:ℕ)
  14     (hregular:S.IsRegularOfDegree (2*m)) (hL:0<L) (hn:0<Fintype.card V)
  15     (a f β b:ℝ) (ha:0≤a) (hf:0≤f) (hβ:0<β)
  16     (hprotected:∀x y,Protected.Adj x y→(S.Adj x y↔S₀.Adj x y))
  17     (hedit:∀v,((changed (S₀.neighborFinset v) (S.neighborFinset v)).card:ℝ)≤b)
  18     (herr:∀v,|((∑w,sampledOrientation S bits v w:ℕ):ℝ)-m|≤a)
  19     (hforbid:∀v,(Protected.degree v:ℝ)≤f)
  20     (hpaths:∀x y,x≠y→β*Fintype.card V≤
  21       ((Finset.univ.filter (fun z=>0<sampledOrientation S bits x z ∧ 0<sampledOrientation S bits z y ∧
  22         S.Adj x z ∧ S.Adj z y)).card:ℝ))
  23     (hmargin:(Fintype.card V:ℝ)*a/L+2*a+4*L+2*f<β*Fintype.card V):
  24     ∃M:V→V→ℕ,
  25       (∀x y,M x y+M y x=if S.Adj x y then 1 else 0) ∧
  26       (∀v,(∑w,M v w)=m ∧ (∑w,M w v)=m) ∧
  27       (∀v,((orientationDisagreementGraph S₀ bits M).degree v:ℝ)≤b+a+2*L) ∧
  28       (∀x y,(orientationDisagreementGraph S₀ bits M).Adj x y→¬Protected.Adj x y):=by
  29   obtain ⟨M,hphys,hbalanced,havoid,hchange⟩:=class_orientation_routing S S Protected
  30     (sampledOrientation S bits) m L hregular hL hn a f β ha hf hβ
  31     (sampledOrientation_physical S bits) herr hforbid hpaths hmargin
  32   refine ⟨M,hphys,hbalanced,?_,?_⟩
  33   · intro v
  34     have hh:=class_orientation_correction_degree S₀ S bits M v
  35     linarith [hedit v,hchange v]
  36   · intro x y hDel hP
  37     have hS:S.Adj x y:=(hprotected x y hP).2 hDel.1
  38     have heq1:sampledOrientation S bits x y=sampledOrientation S₀ bits x y:=by
  39       simp [sampledOrientation,hS,hDel.1]
  40     have heq2:sampledOrientation S bits y x=sampledOrientation S₀ bits y x:=by
  41       simp [sampledOrientation,hS.symm,hDel.1.symm]
  42     have hchanged:M x y≠sampledOrientation S bits x y ∨ M y x≠sampledOrientation S bits y x:=by
  43       simpa only [heq1,heq2] using hDel.2
  44     exact (havoid x y hchanged).2 hP
  45 end Oberwolfach
