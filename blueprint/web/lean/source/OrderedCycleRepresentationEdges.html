  12 abbrev OrderedCycleRepresentations {V:Type} (H:SimpleGraph V) (r:ℕ):=
  13   {f:Fin r↪V // H=(SimpleGraph.cycleGraph r).map f}

  15 abbrev DirectedEdges {V:Type} (H:SimpleGraph V):={xy:V×V // H.Adj xy.1 xy.2}

  17 def orderedCycleFirstEdge {V:Type} [Fintype V] [DecidableEq V] (H:SimpleGraph V) (n:ℕ)
  18     (f:OrderedCycleRepresentations H (n+2)):DirectedEdges H:=by
  19   refine ⟨(f.val 0,f.val 1),?_⟩
  20   apply (congrArg (fun G:SimpleGraph V=>G.Adj (f.val 0) (f.val 1)) f.property).mpr
  21   apply SimpleGraph.map_adj_apply.mpr
  22   rw [SimpleGraph.cycleGraph_adj]
  23   right
  24   simp

  26 theorem orderedCycleFirstEdge_injective {V:Type} [Fintype V] [DecidableEq V] (H:SimpleGraph V)
  27     (n:ℕ) (hn:1≤n):Function.Injective (orderedCycleFirstEdge H n):=by
  28   intro f g h
  29   apply Subtype.ext
  30   apply ordered_cycle_representation_first_edge_injective (by omega)
  31   · exact f.property.symm.trans g.property
  32   · exact congrArg Prod.fst (congrArg Subtype.val h)
  33   · exact congrArg Prod.snd (congrArg Subtype.val h)

  35 theorem orderedCycleFirstEdge_surjective {V:Type} [Fintype V] [DecidableEq V] (H:SimpleGraph V)
  36     (n:ℕ) (hH:IsEmbeddedCycle H (n+2)):Function.Surjective (orderedCycleFirstEdge H n):=by
  37   obtain ⟨_,f,hf⟩:=hH
  38   intro xy
  39   obtain ⟨g,hg,h0,h1⟩:=ordered_cycle_representation_first_edge_exists n f xy.val.1 xy.val.2 (by simpa [hf] using xy.property)
  40   refine ⟨⟨g,hf.trans hg.symm⟩,?_⟩
  41   apply Subtype.ext
  42   exact Prod.ext h0 h1

  44 def orderedCycleFirstEdgeEquiv {V:Type} [Fintype V] [DecidableEq V] (H:SimpleGraph V)
  45     (n:ℕ) (hn:1≤n) (hH:IsEmbeddedCycle H (n+2)):
