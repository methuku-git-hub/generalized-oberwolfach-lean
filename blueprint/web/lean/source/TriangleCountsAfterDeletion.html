  14 theorem triangle_relative_counts_after_deletion (P R H F D E:SimpleGraph V) (A B Y:Finset V)
  15     (hAY:A⊆Y) (pR pH ε ε' L:ℝ) (hpR:0≤pR) (hpH:0≤pH) (hL:0≤L)
  16     (hrowbudget:ε*A.card+7*L≤ε'*A.card) (hindexbudget:ε*B.card+3*L≤ε'*B.card)
  17     (hF:∀ v,(F.degree v:ℝ)≤L) (hD:∀ v,(D.degree v:ℝ)≤L) (hE:∀ v,(E.degree v:ℝ)≤L)
  18     (hrow:∀ S J:Finset V,S⊆Y → J⊆B → S.card≤3 → J.card≤2 → S.card+J.card≤4 →
  19       |((common (common A S (fun v=>R.neighborFinset v)) J (fun u=>H.neighborFinset u)).card:ℝ)-
  20         pR^S.card*pH^J.card*(common A S (fun v=>P.neighborFinset v)).card|≤ε*A.card)
  21     (hindex:∀ S:Finset V,S⊆Y → S.card≤3 → (∀ v∈S,∀ w∈S,¬F.Adj v w) →
  22       |((common B S (fun v=>H.neighborFinset v)).card:ℝ)-pH^S.card*B.card|≤ε*B.card):
  23     (∀ S J:Finset V,S⊆Y → J⊆B → S.card≤3 → J.card≤2 → S.card+J.card≤4 →
  24       pR^S.card*pH^J.card*(common A S (fun v=>(P\spanningRestriction F Y).neighborFinset v)).card-ε'*A.card≤
  25         ((common (common A S (fun v=>((R\F)\D).neighborFinset v)) J (fun u=>(H\E).neighborFinset u)).card:ℝ)) ∧
  26     (∀ S:Finset V,S⊆Y → S.card≤3 → (∀ v∈S,∀ w∈S,¬(spanningRestriction F Y).Adj v w) →
  27       (pH^S.card-ε')*B.card≤((common B S (fun v=>(H\E).neighborFinset v)).card:ℝ)):=by
  28   constructor
  29   · intro S J hSY hJB hS hJ hSJ
  30     have h₂:=mixed_common_sdiff_bound (R\F) H D E A S J L hD hE
  31     have h₀:=hrow S J hSY hJB hS hJ hSJ
  32     have hfirst:|((common (common A S (fun v=>(R\F).neighborFinset v)) J (fun u=>H.neighborFinset u)).card:ℝ)-
  33         ((common (common A S (fun v=>R.neighborFinset v)) J (fun u=>H.neighborFinset u)).card:ℝ)|≤3*L:=by
  34       have hh:=ordinary_common_sdiff_bound R F (common A J (fun u=>H.neighborFinset u)) S L hF
  35       have hh':|((common (common A S (fun v=>(R\F).neighborFinset v)) J (fun u=>H.neighborFinset u)).card:ℝ)-
  36           ((common (common A S (fun v=>R.neighborFinset v)) J (fun u=>H.neighborFinset u)).card:ℝ)|≤S.card*L:=by
  37         convert hh using 2 <;> simp [common,Finset.filter_filter,and_comm,and_left_comm,and_assoc]
  38       exact hh'.trans (mul_le_mul_of_nonneg_right (show (S.card:ℝ)≤3 by exact_mod_cast hS) hL)
  39     have hsecond:=h₂.trans (mul_le_mul_of_nonneg_right (show ((S.card+J.card:ℕ):ℝ)≤4 by exact_mod_cast hSJ) hL)
  40     have hsub:common A S (fun v=>(P\spanningRestriction F Y).neighborFinset v)⊆
  41         common A S (fun v=>P.neighborFinset v):=by
  42       intro z hz
  43       have hh:=(mem_common_iff _ _ _ _).mp hz
  44       exact (mem_common_iff _ _ _ _).mpr ⟨hh.1,fun v hv=>(P.mem_neighborFinset v z).mpr (((P\spanningRestriction F Y).mem_neighborFinset v z).mp (hh.2 v hv)).1⟩
  45     have ht:=mul_le_mul_of_nonneg_left (Nat.cast_le.mpr (Finset.card_le_card hsub))
  46       (show 0≤pR^S.card*pH^J.card by positivity)
  47     have hsecond':|((common (common A S (fun v=>((R\F)\D).neighborFinset v)) J (fun u=>(H\E).neighborFinset u)).card:ℝ)-
  48         ((common (common A S (fun v=>(R\F).neighborFinset v)) J (fun u=>H.neighborFinset u)).card:ℝ)|≤4*L:=by
  49       convert hsecond using 1 <;> congr!
  50     have hlast:pR^S.card*pH^J.card*(common A S (fun v=>(P\spanningRestriction F Y).neighborFinset v)).card-ε'*A.card≤
  51         ((common (common A S (fun v=>((R\F)\D).neighborFinset v)) J (fun u=>(H\E).neighborFinset u)).card:ℝ):=by
  52       linarith only [(abs_le.mp hfirst).1,(abs_le.mp hsecond').1,(abs_le.mp h₀).1,ht,hrowbudget]
  53     convert hlast using 1 <;> congr!
  54   · intro S hSY hS hI
  55     have hI':∀ v∈S,∀ w∈S,¬F.Adj v w:=by
  56       intro v hv w hw hf
  57       exact hI v hv w hw ((spanningRestriction_adj F Y v w).mpr ⟨hf,hSY hv,hSY hw⟩)
  58     have h₀:=hindex S hSY hS hI'
  59     have hh:=(ordinary_common_sdiff_bound H E B S L hE).trans
  60       (mul_le_mul_of_nonneg_right (show (S.card:ℝ)≤3 by exact_mod_cast hS) hL)
  61     nlinarith only [(abs_le.mp h₀).1,(abs_le.mp hh).1,hindexbudget]
