  14 theorem separated_candidate_pairs (F:SimpleGraph V) (C:I→Finset V) (C':I→V→Finset V)
  15     (W R:Finset V) (d c:ℝ) (hd:0≤d) (hF:∀ v,(F.degree v:ℝ)≤d)
  16     (hC:∀ i,c≤((C i).card:ℝ)) (hC':∀ i,∀ v∈C i,c≤((C' i v).card:ℝ))
  17     (hbudget:W.card+((R.card:ℝ)+2*Fintype.card I+1)*(d+1)<c):
  18     ∃ a b:I→V,
  19       (∀ i,a i∈C i∧b i∈C' i (a i)∧a i≠b i∧¬F.Adj (a i) (b i))∧
  20       (∀ i,∀ v∈({a i,b i}:Finset V),v∉W∧v∉R∧∀ w∈R,¬F.Adj v w)∧
  21       (∀ i j,i≠j→Disjoint ({a i,b i}:Finset V) {a j,b j})∧
  22       ∀ i j,∀ v∈({a i,b i}:Finset V),∀ w∈({a j,b j}:Finset V),¬F.Adj v w:=by
  23   let Good:I→V×V→Prop:=fun i ab=>ab.1∈C i∧ab.2∈C' i ab.1∧ab.1≠ab.2∧¬F.Adj ab.1 ab.2∧
  24     ∀ v∈({ab.1,ab.2}:Finset V),v∉R∧∀ w∈R,¬F.Adj v w
  25   have hc:∀ i,∀ B:Finset V,(B.card:ℝ)≤W.card+Fintype.card I*2*(d+1)→
  26       ∃ ab:V×V,Good i ab∧({ab.1,ab.2}:Finset V).card≤2∧Disjoint ({ab.1,ab.2}:Finset V) B:=by
  27     intro i B hB
  28     have hbudget0:(B.card:ℝ)+R.card*(d+1)<c:=by nlinarith only [hB,hbudget,hd]
  29     obtain ⟨a,ha,haB,haR,haF⟩:=candidate_vertex_avoiding F (C i) B R d c hF (hC i) hbudget0
  30     have hRi:((insert a R).card:ℝ)≤R.card+1:=by exact_mod_cast Finset.card_insert_le a R
  31     have hbudget1:(B.card:ℝ)+(insert a R).card*(d+1)<c:=by
  32       have hh:=mul_le_mul_of_nonneg_right hRi (show 0≤d+1 by linarith)
  33       nlinarith only [hB,hbudget,hh]
  34     obtain ⟨b,hb,hbB,hbR,hbF⟩:=candidate_vertex_avoiding F (C' i a) B (insert a R) d c hF (hC' i a ha) hbudget1
  35     have hba:b≠a:=fun he=>hbR (by simp [he])
  36     have hfb:¬F.Adj b a:=hbF a (by simp)
  37     refine ⟨(a,b),⟨ha,hb,hba.symm,fun he=>hfb he.symm,?_⟩,?_,?_⟩
  38     · intro v hv
  39       rcases Finset.mem_insert.mp hv with rfl|hv
  40       · exact ⟨haR,haF⟩
  41       · have he:=Finset.mem_singleton.mp hv
  42         subst v
  43         exact ⟨fun he=>hbR (Finset.mem_insert_of_mem he),fun w hw=>hbF w (Finset.mem_insert_of_mem hw)⟩
  44     · simpa using Finset.card_insert_le a ({b}:Finset V)
  45     · apply Finset.disjoint_left.mpr
  46       intro v hv hvB
  47       rcases Finset.mem_insert.mp hv with rfl|hv
  48       · exact haB hvB
  49       · have he:v=b:=Finset.mem_singleton.mp hv
  50         exact hbB (he ▸ hvB)
  51   obtain ⟨ab,hab,hsep⟩:=finite_conflict_separated_choices (fun ab:V×V=>({ab.1,ab.2}:Finset V)) Good W 2 F d hd hF (by simpa only [Nat.cast_ofNat] using hc)
  52   refine ⟨fun i=>(ab i).1,fun i=>(ab i).2,?_,?_,fun i j hij=>(hsep i j hij).1,?_⟩
  53   · intro i
  54     exact ⟨(hab i).1.1,(hab i).1.2.1,(hab i).1.2.2.1,(hab i).1.2.2.2.1⟩
  55   · intro i v hv
  56     exact ⟨fun hw=>Finset.disjoint_left.mp (hab i).2.2 hv hw,((hab i).1.2.2.2.2 v hv).1,
  57       ((hab i).1.2.2.2.2 v hv).2⟩
  58   · intro i j v hv w hw
  59     by_cases hij:i=j
  60     · subst j
  61       have hno:¬F.Adj (ab i).1 (ab i).2:=(hab i).1.2.2.2.1
  62       simp only [Finset.mem_insert,Finset.mem_singleton] at hv hw
  63       rcases hv with rfl|rfl <;> rcases hw with rfl|rfl
  64       · exact F.loopless.irrefl _
  65       · exact hno
  66       · exact fun he=>hno he.symm
  67       · exact F.loopless.irrefl _
  68     · exact (hsep i j hij).2 v hv w hw
