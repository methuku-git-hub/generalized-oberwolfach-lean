  13 theorem binary_matrix_count {X : Type*} [Fintype X] (f : X → ℕ)
  14     (hf : ∀ x, f x ≤ 1) : (Finset.univ.filter (fun x => 0 < f x)).card = ∑ x, f x := by
  15   rw [Finset.card_filter]
  16   apply Finset.sum_congr rfl
  17   intro x _
  18   have hx := hf x
  19   split_ifs <;> omega

  21 def CyclicMatrixTupleValid {k : ℕ} {X : Type*}
  22     (M : (Fin (k + 2) ⊕ Fin (k + 2)) → X → X → ℕ)
  23     (v : Fin (k + 2) → X × X) : Prop :=
  24   ∀ i, 0 < M (Sum.inl i) (v i).1 (v i).2 ∧
  25     0 < M (Sum.inr i) (v i).2 (v (finRotate (k + 2) i)).1

  27 theorem regular_cyclic_matrix_weights {k : ℕ} {X : Type*}
  28     [Fintype X] [Nonempty X]
  29     (M : (Fin (k + 2) ⊕ Fin (k + 2)) → X → X → ℕ)
  30     (hM : ∀ i x y, M i x y ≤ 1) (q : ℕ)
  31     (hrow : ∀ i x, ∑ y, M i x y = q)
  32     (hcol : ∀ i y, ∑ x, M i x y = q)
  33     (α : ℝ) (hα : 0 < α)
  34     (hthreshold : ((k + 2) + (2 * (k + 2) - 1) * α) * Fintype.card X ≤
  35       (2 * (k + 2) - 1) * q) :
  36     ∃ w : (Fin (k + 2) → X × X) → ℝ,
  37       (∀ v, 0 ≤ w v) ∧
  38       (∀ v, 0 < w v → CyclicMatrixTupleValid M v) ∧
  39       (∀ i x y, (∑ v, if v i = (x,y) then w v else 0) = M (Sum.inl i) x y) ∧
  40       (∀ i y z, (∑ v, if ((v i).2, (v (finRotate (k + 2) i)).1) = (y,z)
  41         then w v else 0) = M (Sum.inr i) y z) ∧
  42       (∀ i x, (∑ v, if (v i).1 = x then w v else 0) = q ∧
  43         (∑ v, if (v i).2 = x then w v else 0) = q) ∧
  44       (∀ v, w v ≤ ((k + 2 : ℝ) * cyclicSectorUpper k α) /
  45         (Fintype.card X : ℝ) ^ (2 * k + 2)) ∧
  46       (∀ v, CyclicMatrixTupleValid M v →
  47         ((k + 2 : ℝ) * cyclicSectorLower k α) /
  48           (Fintype.card X : ℝ) ^ (2 * k + 2) ≤ w v) := by
  49   classical
  50   let L (i : Fin (k + 2)) (x y : X) := 0 < M (Sum.inl i) x y
  51   let R (i : Fin (k + 2)) (x y : X) := 0 < M (Sum.inr i) x y
  52   have hlrow (i) (x) : (Finset.univ.filter (L i x)).card = q :=
  53     (binary_matrix_count _ (hM (Sum.inl i) x)).trans (hrow _ _)
  54   have hlcol (i) (y) : (Finset.univ.filter (fun x => L i x y)).card = q :=
  55     (binary_matrix_count _ (fun x => hM (Sum.inl i) x y)).trans (hcol _ _)
  56   have hrrow (i) (x) : (Finset.univ.filter (R i x)).card = q :=
  57     (binary_matrix_count _ (hM (Sum.inr i) x)).trans (hrow _ _)
  58   have hrcol (i) (y) : (Finset.univ.filter (fun x => R i x y)).card = q :=
  59     (binary_matrix_count _ (fun x => hM (Sum.inr i) x y)).trans (hcol _ _)
  60   obtain ⟨w, hw, hs, hl, hr, hb, hlo⟩ := canonical_cyclic_tuple_weights
  61     L R q α Fintype.card_pos hα hthreshold
  62     (fun i x => by simpa only [L, Finset.filter_congr_decidable] using (show (q : ℝ) ≤ (Finset.univ.filter (L i x)).card by rw [hlrow]))
  63     (fun i y => by simpa only [L, Finset.filter_congr_decidable] using (show (q : ℝ) ≤ (Finset.univ.filter (fun x => L i x y)).card by rw [hlcol]))
  64     (fun i y => by simpa only [L, R, Finset.filter_congr_decidable] using (hlcol i y).trans (hrrow i y).symm)
  65     (fun i x => by simpa only [L, R, Finset.filter_congr_decidable] using (hrcol i x).trans (hlrow (finRotate (k + 2) i) x).symm)
  66   have hcap (i) (x y : X) :
  67       (if 0 < M i x y then (1 : ℝ) else 0) = M i x y := by
  68     have hh := hM i x y
  69     split_ifs with hp
  70     · have he : M i x y = 1 := by omega
  71       simp [he]
  72     · have he : M i x y = 0 := by omega
  73       simp [he]
  74   refine ⟨w, hw, hs, ?_, ?_, ?_, hb, hlo⟩
  75   · intro i x y
  76     rw [hl]
  77     convert hcap (Sum.inl i) x y using 1
  78     unfold interfaceCapacity L
  79     split_ifs <;> rfl
  80   · intro i y z
  81     rw [hr]
  82     convert hcap (Sum.inr i) y z using 1
  83     unfold interfaceCapacity R
  84     split_ifs <;> rfl
  85   · intro i x
  86     have h := canonical_cyclic_position_loads L w hl i x
  87     simp only [L, Finset.filter_congr_decidable] at h
  88     simpa only [binary_matrix_count _ (hM (Sum.inl i) x), binary_matrix_count _ (fun y => hM (Sum.inl i) y x), hrow, hcol] using h
