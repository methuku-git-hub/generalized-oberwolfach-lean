  18 theorem profile_trade_correction (a : I → ℕ) [∀ i,Nonempty (FixedSubset (V i) (a i))]
  19     (q : ℕ) (ha : ∀ i,1 ≤ a i) (hsize : ∑ i,a i=q+1)
  20     (k η K : ℝ) (hk : 2 ≤ k) (hη : 0 ≤ η) (hK : 0 ≤ K)
  21     (hn : ∀ i,(Fintype.card (V i):ℝ)=(a i:ℝ)*k)
  22     (R : C → Finset (Σ i,V i)) (w : C → ℝ)
  23     (hR : ∀ C₀ i,(Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈R C₀)).card=a i)
  24     (herr : ∀ i v,|1-hypergraphWeightedDegree R w ⟨i,v⟩| ≤ η)
  25     (T : (∀ i,FixedSubset (V i) (a i)) → C → ℝ)
  26     (hT : ∀ σ v,hypergraphWeightedDegree R (T σ) v=if v∈profileRootSet σ then 1 else 0)
  27     (hspread : ∀ σ C₀,|T σ C₀| ≤ K/k^(q+1)+
  28       (if (R C₀∩profileRootSet σ).Nonempty then K/k^q else 0)) :
  29     ∃ ψ : C → ℝ,(∀ v,hypergraphWeightedDegree R ψ v=1) ∧
  30       ∀ C₀,|ψ C₀-w C₀| ≤ (1+4*(∑ i,(a i:ℝ)))*K*(q+2)*η/k^q := by
  31   let μ := FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (a i)))
  32   let m := 1-(∑ C₀,w C₀)/k
  33   let y := fun i (v : V i) => 1-hypergraphWeightedDegree R w ⟨i,v⟩-m
  34   let t := profileDemandCoefficient a k m y
  35   let D := 1+4*(∑ i,(a i:ℝ))
  36   have hk0 : 0 < k := by linarith
  37   have hk1 : 1 < k := by linarith
  38   obtain ⟨hm,hy,hymag⟩ := profile_defect_centred_data R w a hR k η hk0 ha hn herr
  39   change |m| ≤ η at hm
  40   change ∀ i,∑ v,y i v=0 at hy
  41   change ∀ i v,|y i v| ≤ 2*η at hymag
  42   have ht : ∀ v,μ.expectation (fun σ => t σ*(if v∈profileRootSet σ then 1 else 0))=
  43       1-hypergraphWeightedDegree R w v := by
  44     rintro ⟨i,v⟩
  45     have hh := profileDemandCoefficient_vertex_identity a k m hk1 ha hn y hy i v
  46     have he : m+y i v=1-hypergraphWeightedDegree R w ⟨i,v⟩ := by dsimp [y];ring
  47     rw [he] at hh
  48     convert hh using 1
  49     apply congrArg μ.expectation
  50     funext σ
  51     simp only [profileRootSet_mem]
  52     dsimp [t]
  53     ring
  54   have htmag (σ : ∀ i,FixedSubset (V i) (a i)) : |t σ| ≤ D*η*k :=
  55     profileDemandCoefficient_bound a k m η hk hη ha hn hm y hymag σ
  56   have hcard (C₀ : C) : (R C₀).card=q+1 := by rw [sigma_finset_card];simp only [hR];exact hsize
  57   have hprob (C₀ : C) : μ.probability (fun σ => (R C₀∩profileRootSet σ).Nonempty) ≤ (q+1)/k := by
  58     have hh := profileRootSet_hit_probability a k hk1 ha hn (R C₀)
  59     simpa only [hcard,Nat.cast_add,Nat.cast_one] using hh
  60   have hTavg (C₀ : C) : μ.expectation (fun σ => |T σ C₀|) ≤ K*(q+2)/k^(q+1) := by
  61     have hh := finiteLaw_trade_bound_from_hit_probability μ T
  62       (fun C₀ σ => (R C₀∩profileRootSet σ).Nonempty) (K/k^(q+1)) (K/k^q) ((q+1)/k)
  63       (by positivity) (by intro σ C₀; convert hspread σ C₀ using 1 <;> congr!) hprob C₀
  64     have he : K/k^(q+1)+(K/k^q)*((q+1)/k)=K*(q+2)/k^(q+1) := by
  65       rw [pow_succ]
  66       field_simp
  67       <;> ring
  68     rwa [he] at hh
  69   refine ⟨correctedTradeWeight μ w t T,correctedTradeWeight_exact μ R profileRootSet w t T hT ht,?_⟩
  70   intro C₀
  71   have hh := correctedTradeWeight_difference μ w t T (D*η*k) (K*(q+2)/k^(q+1))
  72     (by dsimp [D];positivity) htmag hTavg C₀
  73   have he : (D*η*k)*(K*(q+2)/k^(q+1))=D*K*(q+2)*η/k^q := by
  74     rw [pow_succ]
  75     field_simp
  76     <;> ring
  77   rwa [he] at hh
