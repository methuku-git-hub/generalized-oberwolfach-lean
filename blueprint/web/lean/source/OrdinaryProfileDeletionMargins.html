  10 theorem ordinary_profile_deletion_margins (β a τ K L : ℝ)
  11     (hβ : 0<β) (ha : 0<a) (hτ : 0<τ) (hK : 0≤K) (hL : 0≤L) :
  12     ∃ ε : ℝ,0<ε ∧ ε≤β/2 ∧ ∀ f : ℕ,∃ N : ℕ,1≤N ∧
  13       ∀ n : ℕ,N≤n → ∀ z e : ℕ,(z:ℝ)≤ε*n → e≤f →
  14         ((z+e:ℕ):ℝ)*K/n≤τ/2 ∧
  15         ((z+e:ℕ):ℝ)*L≤(a/2)*n := by
  16   let ε := min (β/2) (min (τ/(4*(K+1))) (a/(4*(L+1))))
  17   have hε : 0<ε := by dsimp [ε];positivity
  18   have hεK : 2*ε*K≤τ/2 := by
  19     have hh : ε≤τ/(4*(K+1)) := (min_le_right _ _).trans (min_le_left _ _)
  20     have ht := (le_div_iff₀ (by positivity : (0:ℝ)<4*(K+1))).mp hh
  21     nlinarith only [ht,hε]
  22   have hεL : 2*ε*L≤a/2 := by
  23     have hh : ε≤a/(4*(L+1)) := (min_le_right _ _).trans (min_le_right _ _)
  24     have ht := (le_div_iff₀ (by positivity : (0:ℝ)<4*(L+1))).mp hh
  25     nlinarith only [ht,hε]
  26   refine ⟨ε,hε,min_le_left _ _,?_⟩
  27   intro f
  28   obtain ⟨N,hN⟩ := exists_nat_gt ((f:ℝ)/ε)
  29   refine ⟨max 1 N,le_max_left _ _,?_⟩
  30   intro n hn z e hz he
  31   have hn1 : 1≤n := (le_max_left 1 N).trans hn
  32   have hnR : (0:ℝ)<n := by exact_mod_cast (show 0<n by omega)
  33   have hNn : (N:ℝ)≤n := by exact_mod_cast (le_max_right 1 N).trans hn
  34   have hf : (f:ℝ)≤ε*n := by
  35     have hh := (div_lt_iff₀ hε).mp (lt_of_lt_of_le hN hNn)
  36     linarith
  37   have heR : (e:ℝ)≤ε*n := (Nat.cast_le.mpr he).trans hf
  38   have hsum : ((z+e:ℕ):ℝ)≤2*ε*n := by push_cast;linarith only [hz,heR]
  39   constructor
  40   · apply (div_le_iff₀ hnR).mpr
  41     have hh := mul_le_mul_of_nonneg_right hsum hK
  42     have ht := mul_le_mul_of_nonneg_right hεK hnR.le
  43     nlinarith only [hh,ht]
  44   · have hh := mul_le_mul_of_nonneg_right hsum hL
  45     have ht := mul_le_mul_of_nonneg_right hεL hnR.le
  46     nlinarith only [hh,ht]
