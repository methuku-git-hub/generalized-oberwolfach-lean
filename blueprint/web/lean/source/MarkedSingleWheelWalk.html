  10 theorem marked_single_wheel_physical_walk
  11     {V E K:Type*} (G:SimpleGraph V) (tail head:E→V) (ktail khead:K→V)
  12     (m b:ℕ) (π:Equiv.Perm V) (hb:1≤b)
  13     (C:MarkedCycleTuple tail head ktail khead m π b)
  14     (hroot:π (C.val 0)≠C.val 0)
  15     (hE:∀e:E,G.Adj (tail e) (head e))
  16     (hK:∀e:K,G.Adj (ktail e) (π (khead e))):
  17     ∃f:Fin ((m+1)+1)↪V,
  18       (∀i:Fin (m+1),f i.castSucc=C.val i) ∧ f (Fin.last (m+1))=π (C.val 0) ∧
  19       ∃p:G.Walk (C.val 0) (π (C.val 0)),
  20         p.IsPath ∧ p.length=m+1 ∧ p.support=List.ofFn f := by
  21   obtain ⟨f,hprefix,hlast,hadj⟩:=marked_single_wheel_physical_path G tail head ktail khead m b π hb C hroot hE hK
  22   have h0:(⟨0,by omega⟩:Fin ((m+1)+1))=(0:Fin ((m+1)+1)):=by apply Fin.ext; rfl
  23   have hfin:(⟨m+1,by omega⟩:Fin ((m+1)+1))=Fin.last (m+1):=by apply Fin.ext; rfl
  24   have hfirst:f (0:Fin ((m+1)+1))=C.val 0:=by simpa using hprefix 0
  25   obtain ⟨p,hp,hlen,hsupp⟩:=finite_sequence_simple_walk G (m+1) f hadj
  26   refine ⟨f,hprefix,hlast,?_⟩
  27   have hf0:f ⟨0,by omega⟩=C.val 0:=by rw [h0];exact hfirst
  28   have hfn:f ⟨m+1,by omega⟩=π (C.val 0):=by rw [hfin];exact hlast
  29   rw [←hfn,←hf0]
  30   exact ⟨p,hp,hlen,hsupp⟩
  31 end Oberwolfach
