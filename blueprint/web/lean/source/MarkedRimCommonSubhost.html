  14 theorem marked_rim_common_subhost_deletion
  15     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ)
  16     (σ τ:RawMarkedOutcome V U Γ Color) (I:Finset V)
  17     (hinner:∀ x y,x∈I→y∈I→σ (Sum.inl s(x,y))=τ (Sum.inl s(x,y))):
  18     ∃ D:SimpleGraph V,D≤P ∧
  19       (∀ x y,D.Adj x y↔P.Adj x y∧σ (Sum.inl s(x,y))≠τ (Sum.inl s(x,y))) ∧
  20       (∀ x∈I,∀ y∈I,¬D.Adj x y) ∧
  21       (∀ x,D.degree x=(Finset.univ.filter (fun y=>P.Adj x y∧
  22         σ (Sum.inl s(x,y))≠τ (Sum.inl s(x,y)))).card) ∧
  23       (∀ mark x y,markedPhysicalRim P π (3*d) (fun e=>rawMarkedDeleteRims σ D.edgeFinset (Sum.inl e)) mark x y=1 →
  24         markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) mark x y=1 ∧
  25         markedPhysicalRim P π (3*d) (fun e=>τ (Sum.inl e)) mark x y=1) :=by
  26   let D:SimpleGraph V := {
  27     Adj := fun x y => P.Adj x y∧σ (Sum.inl s(x,y))≠τ (Sum.inl s(x,y))
  28     symm := ⟨by
  29       intro x y h
  30       refine ⟨h.1.symm,?_⟩
  31       have he:s(y,x)=s(x,y):=Sym2.eq_swap
  32       rw [he]
  33       exact h.2⟩
  34     loopless := ⟨by
  35       intro x h
  36       exact P.loopless.irrefl x h.1⟩
  37   }
  38   letI : DecidableRel D.Adj := fun x y=>Classical.propDecidable (D.Adj x y)
  39   refine ⟨D,(fun _ _ h=>h.1),(fun _ _=>Iff.rfl),?_,?_,?_⟩
  40   · intro x hx y hy h
  41     exact h.2 (hinner x y hx hy)
  42   · intro x
  43     rw [←SimpleGraph.card_neighborFinset_eq_degree]
  44     congr 1
  45     ext y
  46     simp [D,SimpleGraph.mem_neighborFinset]

  48     let e:Sym2 V:=s(x,if mark then π y else y)
  49     have hh:=h
  50     rw [markedPhysicalRim_eq_one_iff] at hh
  51     have hp:=hh.2.1
  52     have hraw:=hh.2.2
  53     change rawMarkedDeleteRims σ D.edgeFinset (Sum.inl e)=_ at hraw
  54     have hnot:e∉D.edgeFinset:=by
  55       intro he
  56       simp only [rawMarkedDeleteRims,he,ite_true] at hraw
  57       cases hraw
  58     have heq:σ (Sum.inl e)=τ (Sum.inl e):=by
  59       by_contra hne
  60       have hd:D.Adj x (if mark then π y else y):=⟨hp,hne⟩
  61       exact hnot (by dsimp [e];exact SimpleGraph.mem_edgeFinset.mpr hd)
  62     have hσ:σ (Sum.inl e)=some (mark,orientationBit x (if mark then π y else y)):=by
  63       simpa only [rawMarkedDeleteRims,hnot,ite_false] using hraw
  64     constructor
  65     · rw [markedPhysicalRim_eq_one_iff]
  66       exact ⟨hh.1,hp,hσ⟩
  67     · rw [markedPhysicalRim_eq_one_iff]
