  13 theorem pair_fixed_label_count {I U : Type*} [Fintype I]
  14     (g : I ↪ U) (h : U) :
  15     (∑ p : I×I,if h=g p.1 ∨ h=g p.2 then (1:ℝ) else 0)≤2*Fintype.card I := by
  16   have he : (∑ i,if h=g i then (1:ℝ) else 0)=if h∈Set.range g then 1 else 0 := by
  17     have hh := injective_range_indicator g h
  18     simpa only [eq_comm] using hh
  19   calc
  20     _ ≤ ∑ p : I×I,((if h=g p.1 then (1:ℝ) else 0)+(if h=g p.2 then 1 else 0)) := by
  21       apply Finset.sum_le_sum
  22       intro p _
  23       split_ifs <;> simp_all
  24     _ = 2*(Fintype.card I:ℝ)*(if h∈Set.range g then 1 else 0) := by
  25       simp only [Fintype.sum_prod_type,Finset.sum_add_distrib]
  26       have hfirst : (∑ i : I,∑ j : I,if h=g i then (1:ℝ) else 0) =
  27           (Fintype.card I:ℝ)*(∑ i : I,if h=g i then (1:ℝ) else 0) := by
  28         simp_rw [Finset.sum_const,Finset.card_univ,nsmul_eq_mul]
  29         rw [Finset.mul_sum]
  30       have hsecond : (∑ i : I,∑ j : I,if h=g j then (1:ℝ) else 0) =
  31           (Fintype.card I:ℝ)*(∑ j : I,if h=g j then (1:ℝ) else 0) := by
  32         simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul]
  33       rw [hfirst,hsecond,he]
  34       ring
  35     _ ≤ _ := by
  36       split_ifs
  37       · simp
  38       · simp only [mul_zero]
  39         positivity

  41 theorem normalized_pair_transfer_bound {I U : Type*} [Fintype I]
  42     (g : I ↪ U) (h : U) (a : I → ℝ) (b : I → I → ℝ)
  43     (η B : ℝ) (hη : 0≤η) (hB : 0≤B)
  44     (ha : ∀ i,|a i|≤η) (hb : ∀ i j,|b i j|≤B)
  45     (hz : ∀ i j,h≠g i → h≠g j → b i j=0) :
  46     |(∑ p : I×I,a p.1*b p.1 p.2)/(Fintype.card I:ℝ)|≤2*η*B := by
  47   by_cases hI : Fintype.card I=0
  48   · simp only [hI,Nat.cast_zero,div_zero,abs_zero]
  49     positivity
  50   · have hn : (0:ℝ)<Fintype.card I := Nat.cast_pos.mpr (Nat.pos_of_ne_zero hI)
  51     have hterm (p : I×I) : |a p.1*b p.1 p.2|≤
  52         η*B*(if h=g p.1 ∨ h=g p.2 then (1:ℝ) else 0) := by
  53       by_cases hp : h=g p.1 ∨ h=g p.2
  54       · rw [if_pos hp,mul_one,abs_mul]
  55         exact mul_le_mul (ha _) (hb _ _) (abs_nonneg _) hη
  56       · have hi : h≠g p.1 := fun hi => hp (Or.inl hi)
  57         have hj : h≠g p.2 := fun hj => hp (Or.inr hj)
  58         rw [hz _ _ hi hj,if_neg hp,mul_zero,mul_zero,abs_zero]
  59     rw [abs_div,abs_of_nonneg hn.le]
  60     apply (div_le_iff₀ hn).mpr
  61     calc
  62       _ ≤ ∑ p : I×I,|a p.1*b p.1 p.2| := Finset.abs_sum_le_sum_abs _ _
  63       _ ≤ ∑ p : I×I,η*B*(if h=g p.1 ∨ h=g p.2 then (1:ℝ) else 0) :=
  64         Finset.sum_le_sum (fun p _ => hterm p)
  65       _ = η*B*∑ p : I×I,if h=g p.1 ∨ h=g p.2 then (1:ℝ) else 0 := by rw [Finset.mul_sum]
  66       _ ≤ η*B*(2*Fintype.card I) := mul_le_mul_of_nonneg_left (pair_fixed_label_count g h) (mul_nonneg hη hB)
  67       _ = _ := by ring
