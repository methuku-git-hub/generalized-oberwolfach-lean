   9 theorem walks_disjoint_from_separated_rim_labels
  10     {V I J A B:Type} [DecidableEq V]
  11     (G:SimpleGraph V) {a₁ b₁ a₂ b₂:V} (q₁:G.Walk a₁ b₁) (q₂:G.Walk a₂ b₂)
  12     (C:I→A→V) (D:J→B→V)
  13     (h₁:∀x,x∈q₁.support ↔ x=b₁ ∨ ∃i:I,∃a:A,C i a=x)
  14     (h₂:∀x,x∈q₂.support ↔ x=b₂ ∨ ∃j:J,∃b:B,D j b=x)
  15     (hend:b₁≠b₂) (h₁₂:∀j b,b₁≠D j b) (h₂₁:∀i a,b₂≠C i a)
  16     (hdis:∀i j a b,C i a≠D j b):
  17     Disjoint q₁.support.toFinset q₂.support.toFinset:=by
  18   apply Finset.disjoint_left.mpr
  19   intro x hx hy
  20   rcases (h₁ x).mp (List.mem_toFinset.mp hx) with he₁|⟨i,a,hi⟩
  21   · rcases (h₂ x).mp (List.mem_toFinset.mp hy) with he₂|⟨j,b,hj⟩
  22     · exact hend (he₁.symm.trans he₂)
  23     · exact h₁₂ j b (he₁.symm.trans hj.symm)
  24   · rcases (h₂ x).mp (List.mem_toFinset.mp hy) with he₂|⟨j,b,hj⟩
  25     · exact h₂₁ i a (he₂.symm.trans hi.symm)
  26     · exact hdis i j a b (hi.trans hj.symm)
  27 end Oberwolfach
