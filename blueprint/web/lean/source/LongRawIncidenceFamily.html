  15 def longRawResourceIncident {n:ℕ} {U:Type}:
  16     LongRawIncidenceIndex n U → LongRawHubResource n U → Prop
  17   | .inl x, .inl (.inl (v,w)) => v=x ∨ w=x
  18   | .inl x, .inl (.inr (v,w)) => v=x ∨ w=x
  19   | .inl _, .inr _ => False
  20   | .inr (typ,.inl x), .inr (j,v,_) => typ=j ∧ v=x
  21   | .inr (typ,.inr u), .inr (j,_,v) => typ=j ∧ v=u
  22   | .inr _, .inl _ => False
