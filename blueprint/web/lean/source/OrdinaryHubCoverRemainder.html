  16 theorem ordinaryHubResources (R : SimpleGraph V) (u : Fin n→U) (T : Fin n→Finset V) :
  17     ordinaryHostResources R (fun v w=>∃ i,u i=w ∧ v∈T i)=
  18       R.edgeFinset.image Sum.inl ∪ Finset.univ.biUnion (fun i=>(T i).image (fun v=>Sum.inr (v,u i))) := by
  19   ext e
  20   cases e with
  21   | inl e =>
  22     simp only [ordinaryHostResources_rim,Finset.mem_union,Finset.mem_image,Sum.inl.injEq,
  23       Sum.inr_ne_inl,and_false,exists_false,Finset.mem_biUnion,Finset.mem_univ,true_and,or_false,
  24       SimpleGraph.mem_edgeFinset,exists_eq_right]
  25   | inr vu =>
  26     rcases vu with ⟨v,w⟩
  27     simp only [ordinaryHostResources_spoke,Finset.mem_union,Finset.mem_image,Sum.inl_ne_inr,
  28       and_false,exists_false,false_or,Finset.mem_biUnion,Finset.mem_univ,true_and,
  29       Sum.inr.injEq,Prod.mk.injEq,exists_eq_right]
  30     constructor
  31     · rintro ⟨i,hi,hv⟩;exact ⟨i,v,hv,rfl,hi⟩
  32     · rintro ⟨i,z,hz,hzv,hi⟩;subst z;exact ⟨i,hi,hz⟩

  44 theorem ordinary_hub_cover_remainder (u : Fin n→U)
  45     (T : Fin n→Finset V) (hT : ∀ i v,v∈T i ↔ H v (u i))
  46     (M : Finset (OrdinaryWheel Q F H r))
  47     (hM : ∀ W∈M,∀ Z∈M,W≠Z → Disjoint (ordinaryWheelResources W) (ordinaryWheelResources Z))
  48     (hunion : M.biUnion ordinaryWheelResources=(ordinaryWheelRimGraph M).edgeFinset.image Sum.inl ∪
  49       Finset.univ.biUnion (fun i=>(T i).image (fun v=>Sum.inr (v,u i))))
  50     (hQ : ∀ v,Q.degree v=2*(Finset.univ.filter (H v)).card)
  51     (hU : ∀ w,r∣(Finset.univ.filter (fun v=>H v w)).card) :
  52     let H' : V→U→Prop:=fun v w=>H v w ∧ w∉Finset.univ.image u
  53     ordinaryHostResources (Q\ordinaryWheelRimGraph M) H'=
  54       ordinaryHostResources Q H\M.biUnion ordinaryWheelResources ∧
  55     (∀ v,(Q\ordinaryWheelRimGraph M).degree v=2*(Finset.univ.filter (H' v)).card) ∧
  56     (∀ w,r∣(Finset.univ.filter (fun v=>H' v w)).card) := by
  57   dsimp only
  58   let K : V→U→Prop:=fun v w=>∃ i,u i=w ∧ v∈T i
  59   have hK (v : V) (w : U) : K v w ↔ H v w ∧ w∈Finset.univ.image u := by
  60     constructor
  61     · rintro ⟨i,hi,hv⟩
  62       exact ⟨hi ▸ (hT i v).mp hv,Finset.mem_image.mpr ⟨i,Finset.mem_univ _,hi⟩⟩
  63     · rintro ⟨hv,hw⟩
  64       obtain ⟨i,_,hi⟩:=Finset.mem_image.mp hw
  65       exact ⟨i,hi,(hT i v).mpr (hi.symm ▸ hv)⟩
  66   have hrest : (fun v w=>H v w ∧ ¬K v w)=(fun v w=>H v w ∧ w∉Finset.univ.image u) := by
  67     funext v w;apply propext;rw [hK];tauto
  68   have hcover : M.biUnion ordinaryWheelResources=ordinaryHostResources (ordinaryWheelRimGraph M) K := by
  69     rw [ordinaryHubResources]
  70     exact hunion
  71   have hbal:=ordinary_decomposition_remainder_balance Q (ordinaryWheelRimGraph M) H K r
  72     (ordinaryWheelRimGraph_subhost M) (fun v w h=>(hK v w).mp h |>.1) hQ hU
  73     (ordinary_hub_packing_decomposition u T M hM hunion)
  74   have hrestpt (v : V) (w : U) : (H v w ∧ ¬K v w)=(H v w ∧ w∉Finset.univ.image u) :=
  75     congrFun (congrFun hrest v) w
  76   simp_rw [hrestpt] at hbal
  77   refine ⟨?_,hbal⟩
  78   rw [hcover,←ordinaryHostResources_sdiff,hrest]
