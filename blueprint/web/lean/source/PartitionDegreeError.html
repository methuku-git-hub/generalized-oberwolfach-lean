  13 theorem conditional_degree_tail_scale {r d m R η ε : ℝ}
  14     (hr : 0 ≤ r) (hrR : r ≤ R) (hd : 0 ≤ d) (hsum : r + d = m)
  15     (hm : 0 < m) (hη : 0 < η) (hη1 : η ≤ 1)
  16     (hroots : 2 * R ≤ η * m / 4) (hε : ε ≤ η / 4) :
  17     0 < d ∧ η * m / 2 + 2 * R + ε * m ≤ η * m ∧
  18       2 * Real.exp (-2 * (η * m / 2) ^ 2 / d) ≤
  19         2 * Real.exp (-η ^ 2 * m / 2) := by
  20   have hηm : η * m ≤ m := by nlinarith [mul_nonneg (sub_nonneg.mpr hη1) hm.le]
  21   have hdpos : 0 < d := by linarith
  22   have hεm := mul_le_mul_of_nonneg_right hε hm.le
  23   refine ⟨hdpos, by linarith, ?_⟩
  24   apply mul_le_mul_of_nonneg_left ?_ (by norm_num)
  25   apply Real.exp_le_exp.mpr
  26   apply (div_le_iff₀ hdpos).mpr
  27   have hdm : d ≤ m := by linarith
  28   have hh := mul_le_mul_of_nonneg_left hdm (sq_nonneg η)
  29   have hhm := mul_le_mul_of_nonneg_right hh hm.le
