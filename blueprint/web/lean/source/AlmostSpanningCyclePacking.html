  16 theorem ordinary_cycle_incidence_load_adjacency
  17     {V : Type*} [Fintype V] [DecidableEq V]
  18     (G : SimpleGraph V) (r : ℕ) (a : OrdinaryCycle G r → ℝ)
  19     (he : ∀ e : G.edgeSet, ∑ C, a C*ordinaryCycleIncidence G r C e=1) :
  20     ∀ x y, (∑ C, if C.val.val.Adj x y then a C else 0)=if G.Adj x y then 1 else 0 := by
  21   intro x y
  22   by_cases hxy : G.Adj x y
  23   · have hh := he ⟨s(x,y),hxy⟩
  24     simpa only [ordinaryCycleIncidence,SimpleGraph.mem_edgeSet,mul_ite,mul_one,mul_zero,
  25       hxy,ite_true] using hh
  26   · rw [if_neg hxy]
  27     apply Finset.sum_eq_zero
  28     intro C _
  29     have hn : ¬ C.val.val.Adj x y := fun h => hxy (C.property h)
  30     simp [hn]

  32 theorem almost_spanning_cycle_packing
  33     (r : ℕ) (hr : 3≤r) (α ξ : ℝ) (hα : 0<α) (hξ : 0<ξ) (hξ1 : ξ≤1) :
  34     ∃ η : ℝ, 0<η ∧ ∃ N : ℕ, 0<N ∧
  35       ∀ (V : Type) [Fintype V] [DecidableEq V], N≤Fintype.card V →
  36       ∀ G : SimpleGraph V,
  37         (∀ v, (cycleThreshold r+α)*Fintype.card V≤(G.degree v : ℝ)) →
  38       ∀ d : ℝ, (∀ v, (1-η)*d≤(G.degree v : ℝ) ∧ (G.degree v : ℝ)≤d) →
  39       ∃ M : Finset (OrdinaryCycle G r),
  40         (M : Set (OrdinaryCycle G r)).Pairwise (fun C D =>
  41           Disjoint (ordinaryCycleVertices G r C) (ordinaryCycleVertices G r D)) ∧
  42         (1-ξ)*Fintype.card V≤((M.biUnion (ordinaryCycleVertices G r)).card : ℝ) := by
  43   let μ := HadwigerLean.Theorem2.matchingMu r ξ
  44   have hμ : 0<μ := by
  45     dsimp [μ,HadwigerLean.Theorem2.matchingMu]
  46     have hrR : (0 : ℝ)<r := by exact_mod_cast (show 0<r by omega)
  47     positivity
  48   obtain ⟨c,C,hc,hC,N₀,hN₀,hweights⟩ := positive_ordinary_fractional_cycle_weights r hr α hα
  49   let N := max N₀ ⌈16*(r : ℝ)^2*C/μ⌉₊
  50   refine ⟨μ/4,by positivity,N,lt_of_lt_of_le hN₀ (le_max_left _ _),?_⟩
  51   intro V _ _ hn G hG d hd
  52   have hn₀ : N₀≤Fintype.card V := (le_max_left _ _).trans hn
  53   have hnpos : 0<Fintype.card V := hN₀.trans_le hn₀
  54   have hnR : (0 : ℝ)<Fintype.card V := Nat.cast_pos.mpr hnpos
  55   obtain ⟨a,ha,he⟩ := hweights V hn₀ G hG
  56   have ha0 (A : OrdinaryCycle G r) : 0≤a A :=
  57     (div_nonneg hc.le (by positivity)).trans (ha A).1
  58   have hnlarge : 16*(r : ℝ)^2*C/μ≤(Fintype.card V : ℝ) :=
  59     (Nat.le_ceil _).trans (by exact_mod_cast ((le_max_right _ _).trans hn : ⌈16*(r : ℝ)^2*C/μ⌉₊≤_))
  60   have hpairnum : 4*(r : ℝ)^2*C/(Fintype.card V : ℝ)≤μ/4 := by
  61     have hh := (div_le_iff₀ hμ).mp hnlarge
  62     apply (div_le_iff₀ hnR).mpr
  63     nlinarith
  64   have hdhalf : (Fintype.card V : ℝ)/2≤d := by
  65     obtain ⟨v⟩ := Fintype.card_pos_iff.mp hnpos
  66     have ht := cycleThreshold_half_le hr
  67     have hg := hG v
  68     have hh := (hd v).2
  69     have hn0 : (0 : ℝ)≤Fintype.card V := hnR.le
  70     nlinarith [mul_nonneg hα.le hn0,mul_nonneg (sub_nonneg.mpr ht) hn0]
  71   obtain ⟨b,hb,hload,hpair⟩ := ordinary_cycle_vertex_matching_weights G r (by omega) hnpos
  72     a ha0 C hC.le (fun A => (ha A).2)
  73     (ordinary_cycle_incidence_load_adjacency G r a he) hdhalf hd
  74   exact almost_perfect_matching_of_fractional_weights (ordinaryCycleVertices G r) r (by omega)
  75     (ordinaryCycleVertices_card G r) ξ hξ hξ1 b hb hload
  76     (fun x y hxy => (hpair x y hxy).trans hpairnum)
