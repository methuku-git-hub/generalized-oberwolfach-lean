  13 theorem graph_degree_disjoint_sup (G K : SimpleGraph V) (h : Disjoint G K) (a : V) :
  14     (G⊔K).degree a=G.degree a+K.degree a := by
  15   have hd : Disjoint (G.neighborFinset a) (K.neighborFinset a) := by
  16     apply Finset.disjoint_left.mpr
  17     intro b hb hk
  18     have hG := (SimpleGraph.mem_neighborFinset (G:=G) (v:=a) b).mp hb
  19     have hK := (SimpleGraph.mem_neighborFinset (G:=K) (v:=a) b).mp hk
  20     exact disjoint_iff_inf_le.mp h ⟨hG,hK⟩
  21   rw [←SimpleGraph.card_neighborFinset_eq_degree,SimpleGraph.neighborFinset_sup,
  22     Finset.card_union_of_disjoint hd,SimpleGraph.card_neighborFinset_eq_degree,
  23     SimpleGraph.card_neighborFinset_eq_degree]

  25 theorem card_filter_disjoint_or {A : Type*} [Fintype A] [DecidableEq A]
  26     (p q : A → Prop) [DecidablePred p] [DecidablePred q] (h : ∀ a,¬(p a ∧ q a)) :
  27     (Finset.univ.filter (fun a => p a ∨ q a)).card=
  28       (Finset.univ.filter p).card+(Finset.univ.filter q).card := by
  29   have he : Finset.univ.filter (fun a => p a ∨ q a)=(Finset.univ.filter p)∪(Finset.univ.filter q) := by ext;simp
  30   rw [he,Finset.card_union_of_disjoint]
  31   apply Finset.disjoint_left.mpr
  32   intro a hp hq
  33   exact h a ⟨(Finset.mem_filter.mp hp).2,(Finset.mem_filter.mp hq).2⟩

  35 theorem ordinary_balance_disjoint_union (G K : SimpleGraph V) (H J : V → U → Prop)
  36     (hGK : Disjoint G K) (hHJ : ∀ a b,¬(H a b ∧ J a b))
  37     (hG : ∀ a,G.degree a=2*(Finset.univ.filter (H a)).card)
  38     (hK : ∀ a,K.degree a=2*(Finset.univ.filter (J a)).card) :
  39     ∀ a,(G⊔K).degree a=2*(Finset.univ.filter (fun b => H a b ∨ J a b)).card := by
  40   intro a
  41   rw [graph_degree_disjoint_sup G K hGK,card_filter_disjoint_or (H a) (J a) (hHJ a),hG,hK]
  42   omega

  44 theorem ordinary_hub_divisibility_disjoint_union (H J : V → U → Prop) (r : ℕ)
  45     (hHJ : ∀ a b,¬(H a b ∧ J a b))
  46     (hH : ∀ b,r ∣ (Finset.univ.filter (fun a => H a b)).card)
  47     (hJ : ∀ b,r ∣ (Finset.univ.filter (fun a => J a b)).card) :
  48     ∀ b,r ∣ (Finset.univ.filter (fun a => H a b ∨ J a b)).card := by
  49   intro b
  50   rw [card_filter_disjoint_or (fun a => H a b) (fun a => J a b) (fun a => hHJ a b)]
  51   exact dvd_add (hH b) (hJ b)
