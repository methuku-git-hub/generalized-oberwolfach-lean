  24   first_disjoint : ∀ e f,e≠f→Disjoint (ordinaryWheelResources (first e)) (ordinaryWheelResources (first f))

  38  theorem ordinary_actual_spoke_wheels (G Q F:SimpleGraph V) (H₀ H:V→U→Prop)
  39     (φ:V→V) (ψ:U→U) (r:ℕ) (hr:3≤r)
  40     (p:(G.edgeFinset⊕{e:V×U // H₀ e.1 e.2})→(Σ a:V,Σ c:V,Q.Walk a c))
  41     (R:Finset V) (hroot:∀ e:{e:V×U // H₀ e.1 e.2},e.val.1∈R∧φ e.val.1∈R∧e.val.1≠φ e.val.1)
  42     (hpath:∀ e:{e:V×U // H₀ e.1 e.2},(p (.inr e)).2.2.IsPath∧(p (.inr e)).2.2.length=r-2)
  43     (hends:∀ e:{e:V×U // H₀ e.1 e.2},Q.Adj e.val.1 (p (.inr e)).1∧
  44       Q.Adj (φ e.val.1) (p (.inr e)).1∧Q.Adj e.val.1 (p (.inr e)).2.1∧Q.Adj (φ e.val.1) (p (.inr e)).2.1)
  45     (havoid:∀ i,∀ v∈(p i).2.2.support,v∉R)
  46     (hdis:∀ i j,i≠j→Disjoint (p i).2.2.support.toFinset (p j).2.2.support.toFinset)
  47     (hF:∀ e:{e:V×U // H₀ e.1 e.2},∀ v∈insert e.val.1 (insert (φ e.val.1) (p (.inr e)).2.2.support.toFinset),
  48       ∀ w∈insert e.val.1 (insert (φ e.val.1) (p (.inr e)).2.2.support.toFinset),¬F.Adj v w)
  49     (hH:∀ e:{e:V×U // H₀ e.1 e.2},H e.val.1 e.val.2∧H (φ e.val.1) (ψ e.val.2))
  50     (hHp:∀ e:{e:V×U // H₀ e.1 e.2},∀ v∈(p (.inr e)).2.2.support,H v e.val.2∧H v (ψ e.val.2))
  51     (hφedge:Function.Injective (fun e:{e:V×U // H₀ e.1 e.2}=>(φ e.val.1,ψ e.val.2))):
  52     Nonempty (OrdinaryActualSpokeWheels G Q F H₀ H φ ψ r p):=by
  53   obtain ⟨W,W',hh,hs,hs',hd,hd',hrim,hspoke⟩:=ordinary_spoke_wheel_families Q F H r hr R
  54     (fun e:{e:V×U // H₀ e.1 e.2}=>e.val.1) (fun e=>φ e.val.1) (fun e=>e.val.2) (fun e=>ψ e.val.2)
  55     (fun e=>p (.inr e)) (fun e=>⟨(hroot e).1,(hroot e).2.1⟩) (fun e=>(hroot e).2.2) hpath
  56     (fun e=>⟨(hends e).1,(hends e).2.2.1,(hends e).2.1,(hends e).2.2.2⟩)
  57     (fun e=>havoid (.inr e))
  58     (fun e f he=>hdis (.inr e) (.inr f) (fun hh=>he (Sum.inr.inj hh)))
  59     hF hH hHp (by intro e f he;exact Subtype.ext (Prod.ext (Prod.mk.inj he).1 (Prod.mk.inj he).2)) hφedge
  60   exact ⟨⟨W,W',fun e=>(hh e).1,fun e=>(hh e).2,hs,hs',hd,hd',hrim,hspoke⟩⟩
