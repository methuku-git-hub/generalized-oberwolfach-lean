  11 theorem actual_old_embedding_from_pieces_and_anchors
  12     {X V K J:Type} [Fintype X] [DecidableEq X] [DecidableEq V] [Nonempty V] [DecidableEq K] [DecidableEq J]
  13     {G:SimpleGraph X} (data:ActualResidualData (K:=K) (J:=J) G) (hG:G.IsRegularOfDegree 2)
  14     (hCycle:data.t.biUnion (fun j=>(data.C j).support.toFinset)⊆data.Cembed):
  15     let A:=Finset.univ.biUnion (fun z:actualOldPieceIndex data=>(actualOldPieceWalk data z).support.toFinset)
  16     ∀(f:{x // x∈A}↪V) (g:{x // x∈data.prescribed}↪V),
  17     (∀x (hx:x∈A) (hy:x∈data.prescribed),f ⟨x,hx⟩=g ⟨x,hy⟩)→
  18     (∀x (hx:x∈data.prescribed) (hxa:x∉A) (y:{x // x∈A}),g ⟨x,hx⟩≠f y)→
  19     ∃ψ:X→V,Set.InjOn ψ (↑(data.oldDomain\data.additionalSelectedInteriors):Set X) ∧
  20       (∀x (hx:x∈A),ψ x=f ⟨x,hx⟩) ∧
  21       (∀x (hx:x∈data.prescribed),ψ x=g ⟨x,hx⟩):=by
  22   classical
  23   intro A f g hagree hcross
  24   let ψ (x:X):V:=if hx:x∈A then f ⟨x,hx⟩ else if hy:x∈data.prescribed then g ⟨x,hy⟩ else Classical.choice inferInstance
  25   have hψA:∀x (hx:x∈A),ψ x=f ⟨x,hx⟩:=by intro x hx;simp [ψ,hx]
  26   have hψP:∀x (hx:x∈data.prescribed),ψ x=g ⟨x,hx⟩:=by
  27     intro x hx
  28     by_cases ha:x∈A
  29     · exact (hψA x ha).trans (hagree x ha hx)
  30     · simp [ψ,ha,hx]
  31   refine ⟨ψ,?_,hψA,hψP⟩
  32   intro x hx y hy hxy
  33   have hs:=actual_old_domain_without_final_interiors data hG hCycle
  34   have hx':x∈data.prescribed ∪ A:=by
  35     change x∈data.oldDomain\data.additionalSelectedInteriors at hx
  36     rw [hs] at hx
  37     exact hx
  38   have hy':y∈data.prescribed ∪ A:=by
  39     change y∈data.oldDomain\data.additionalSelectedInteriors at hy
  40     rw [hs] at hy
  41     exact hy
  42   by_cases hxa:x∈A
  43   · by_cases hya:y∈A
  44     · have hh:f ⟨x,hxa⟩=f ⟨y,hya⟩:=by rw [←hψA x hxa,←hψA y hya];exact hxy
  45       exact congrArg Subtype.val (f.injective hh)
  46     · have hyp:y∈data.prescribed:=((Finset.mem_union.mp hy').resolve_right hya)
  47       have hh:g ⟨y,hyp⟩=f ⟨x,hxa⟩:=by rw [←hψP y hyp,←hψA x hxa];exact hxy.symm
  48       exact False.elim (hcross y hyp hya ⟨x,hxa⟩ hh)
  49   · have hxp:x∈data.prescribed:=((Finset.mem_union.mp hx').resolve_right hxa)
  50     by_cases hya:y∈A
  51     · have hh:g ⟨x,hxp⟩=f ⟨y,hya⟩:=by rw [←hψP x hxp,←hψA y hya];exact hxy
  52       exact False.elim (hcross x hxp hxa ⟨y,hya⟩ hh)
  53     · have hyp:y∈data.prescribed:=((Finset.mem_union.mp hy').resolve_right hya)
  54       have hh:g ⟨x,hxp⟩=g ⟨y,hyp⟩:=by rw [←hψP x hxp,←hψP y hyp];exact hxy
  55       exact congrArg Subtype.val (g.injective hh)
  56 end Oberwolfach
