   9 theorem component_family_domain_card
  10     (F:SimpleGraph (Fin n)) (C:A→F.ConnectedComponent)
  11     (hC:Function.Injective C):
  12     Fintype.card {v:Fin n // ∃a,v∈(C a).supp} = ∑a:A,Fintype.card (C a) := by
  13   let f:(Σa:A,C a)→{v:Fin n // ∃a,v∈(C a).supp}:=fun z=>⟨z.2.val,⟨z.1,z.2.property⟩⟩
  14   have hinj:Function.Injective f:=by
  15     rintro ⟨a,x⟩ ⟨b,y⟩ heq
  16     have hv:x.val=y.val:=congrArg Subtype.val heq
  17     have hy:x.val∈(C b).supp:=by rw [hv];exact y.property
  18     have hc:C a=C b:=SimpleGraph.ConnectedComponent.eq_of_common_vertex x.property hy
  19     have hab:a=b:=hC hc
  20     subst b
  21     have hxy:x=y:=Subtype.ext hv
  22     subst y
  23     rfl
  24   have hsurj:Function.Surjective f:=by
  25     intro v
  26     obtain ⟨a,ha⟩:=v.property
  27     exact ⟨⟨a,⟨v.val,ha⟩⟩,Subtype.ext rfl⟩
  28   calc
  29     Fintype.card {v:Fin n // ∃a,v∈(C a).supp} = Fintype.card (Σa:A,C a):=
  30       (Fintype.card_congr (Equiv.ofBijective f ⟨hinj,hsurj⟩)).symm
  31     _ = ∑a:A,Fintype.card (C a):=Fintype.card_sigma
  32 end Oberwolfach
