   9 theorem disjoint_piece_embeddings_glue
  10     {X V J:Type*} [DecidableEq X] [DecidableEq J]
  11     (I:Finset J) (S:J→Finset X)
  12     (hS:∀i∈I,∀j∈I,i≠j→Disjoint (S i) (S j))
  13     (f:∀i,{x // x∈S i}↪V)
  14     (hf:∀i∈I,∀j∈I,i≠j→Disjoint (Set.range (f i)) (Set.range (f j))):
  15     ∃φ:{x // x∈I.biUnion S}↪V,
  16       ∀i (hi:i∈I) (x:X) (hx:x∈S i),
  17         φ ⟨x,Finset.mem_biUnion.mpr ⟨i,hi,hx⟩⟩=f i ⟨x,hx⟩ := by
  18   classical
  19   let idx (x:{x // x∈I.biUnion S}):J:=Classical.choose (Finset.mem_biUnion.mp x.property)
  20   have hidx (x:{x // x∈I.biUnion S}):idx x∈I ∧ x.val∈S (idx x):=
  21     Classical.choose_spec (Finset.mem_biUnion.mp x.property)
  22   have map_eq (i j:J) (hij:i=j) (x:X) (hxi:x∈S i) (hxj:x∈S j):
  23       f i ⟨x,hxi⟩=f j ⟨x,hxj⟩:=by
  24     subst j
  25     rfl
  26   let φ (x:{x // x∈I.biUnion S}):V:=f (idx x) ⟨x.val,(hidx x).2⟩
  27   have hinj:Function.Injective φ:=by
  28     intro x y hxy
  29     have hij:idx x=idx y:=by
  30       by_contra hne
  31       have hx:φ x∈Set.range (f (idx x)):=⟨⟨x.val,(hidx x).2⟩,rfl⟩
  32       have hy:φ y∈Set.range (f (idx y)):=⟨⟨y.val,(hidx y).2⟩,rfl⟩
  33       have hy':φ x∈Set.range (f (idx y)):=by rw [hxy];exact hy
  34       exact Set.disjoint_left.mp (hf (idx x) (hidx x).1 (idx y) (hidx y).1 hne) hx hy'
  35     apply Subtype.ext
  36     have hyx:y.val∈S (idx x):=by simpa only [hij] using (hidx y).2
  37     have hy':f (idx x) ⟨y.val,hyx⟩=f (idx y) ⟨y.val,(hidx y).2⟩:=
  38       map_eq (idx x) (idx y) hij y.val hyx (hidx y).2
  39     change f (idx x) ⟨x.val,(hidx x).2⟩=f (idx y) ⟨y.val,(hidx y).2⟩ at hxy
  40     have hsub: (⟨x.val,(hidx x).2⟩:{w:X // w∈S (idx x)})=⟨y.val,hyx⟩:=
  41       (f (idx x)).injective (hxy.trans hy'.symm)
  42     exact congrArg (fun (z:{w:X // w∈S (idx x)})=>z.val) hsub
  43   refine ⟨⟨φ,hinj⟩,?_⟩
  44   intro i hi x hx
  45   let z:{x // x∈I.biUnion S}:=⟨x,Finset.mem_biUnion.mpr ⟨i,hi,hx⟩⟩
  46   have heq:idx z=i:=by
  47     by_contra hne
  48     exact Finset.disjoint_left.mp (hS (idx z) (hidx z).1 i hi hne) (hidx z).2 hx
  49   change φ z=f i ⟨x,hx⟩
  50   exact map_eq (idx z) i heq x (hidx z).2 hx

  52 theorem disjoint_piece_graph_embeddings_glue
  53     {X V J:Type*} [DecidableEq X] [DecidableEq J]
  54     (I:Finset J) (S:J→Finset X)
  55     (hS:∀i∈I,∀j∈I,i≠j→Disjoint (S i) (S j))
  56     (f:∀i,{x // x∈S i}↪V)
  57     (hf:∀i∈I,∀j∈I,i≠j→Disjoint (Set.range (f i)) (Set.range (f j)))
  58     (G:SimpleGraph X) (H:SimpleGraph V)
  59     (hcover:∀x y,x∈I.biUnion S→y∈I.biUnion S→G.Adj x y→
  60       ∃i∈I,x∈S i ∧ y∈S i)
  61     (hlocal:∀i∈I,∀x y (hx:x∈S i) (hy:y∈S i),G.Adj x y→H.Adj (f i ⟨x,hx⟩) (f i ⟨y,hy⟩)):
  62     ∃φ:{x // x∈I.biUnion S}↪V,
  63       (∀i (hi:i∈I) (x:X) (hx:x∈S i),
  64         φ ⟨x,Finset.mem_biUnion.mpr ⟨i,hi,hx⟩⟩=f i ⟨x,hx⟩) ∧
  65       (∀x y,G.Adj x.val y.val→H.Adj (φ x) (φ y)) := by
  66   obtain ⟨φ,hφ⟩:=disjoint_piece_embeddings_glue I S hS f hf
  67   refine ⟨φ,hφ,?_⟩
  68   intro x y hxy
  69   obtain ⟨i,hi,hx,hy⟩:=hcover x.val y.val x.property y.property hxy
  70   have hx':φ x=f i ⟨x.val,hx⟩:=by simpa using hφ i hi x.val hx
  71   have hy':φ y=f i ⟨y.val,hy⟩:=by simpa using hφ i hi y.val hy
  72   rw [hx',hy']
  73   exact hlocal i hi x.val y.val hx hy hxy
  74 end Oberwolfach
