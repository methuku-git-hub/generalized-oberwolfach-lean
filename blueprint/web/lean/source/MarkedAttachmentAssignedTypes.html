  13  theorem marked_attachment_assigned_types {X Y S:Type*} [Fintype X] [Fintype Y] [Fintype S]

  15     (hXY:Fintype.card X=Fintype.card Y) (hXS:Fintype.card X=Fintype.card S)
  16     (hK:K.card=SK.card) (hpad:¬(Fintype.card X=1∧K.card=1)):
  17     ∃ τ:Equiv.Perm (X⊕Y),∃ a:X≃S,∃ d:Y≃S,
  18       (∀ x,τ x≠x)∧
  19       (∀ x:X,a x∈SK↔τ (Sum.inl x)∈K.image (Sum.inl (β:=Y)))∧
  20       ∀ y:Y,τ (Sum.inr y)∉K.image (Sum.inl (β:=Y)):=by
  21   by_cases hx:0<Fintype.card X
  22   · obtain ⟨σ,a,d,hσ,_,ha,hd⟩:=marked_attachment_pairings K SK hXY hXS hK hx hpad
  23     refine ⟨σ.symm,a,d,?_,ha,hd⟩
