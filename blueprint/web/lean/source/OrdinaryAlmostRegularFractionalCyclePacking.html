  12 theorem ordinary_almost_regular_fractional_cycle_packing
  13     (r:ℕ) (hr:3≤r) (B θ γ:ℝ) (hB:1≤B) (hθ:0<θ) (hθsmall:θ≤1/20)
  14     (hγ:0<γ) (hγ1:γ≤1):
  15     ∃ N:ℕ,∀ (V:Type) [Fintype V] [DecidableEq V] (G:SimpleGraph V)
  16       [DecidableEq (OrdinaryCycle G r)],N≤Fintype.card V→
  17       ∀ d:ℝ,(Fintype.card V:ℝ)/2≤d→d≤Fintype.card V→
  18       (∀ v,(Fintype.card V:ℝ)/2≤(G.degree v:ℝ))→
  19       (∀ v,|(G.degree v:ℝ)-d|≤θ*(Fintype.card V:ℝ)/4)→
  20       ∀ w:OrdinaryCycle G r→ℝ,
  21       (∀ e:G.edgeSet,∑ C,w C*ordinaryCycleIncidence G r C e=1)→
  22       (∀ C,0≤w C ∧ w C≤B/(Fintype.card V:ℝ)^(r-2))→
  23       ∃ M:Finset (OrdinaryCycle G r),
  24         (∀ C∈M,∀ C'∈M,C≠C'→Disjoint C.val.val.support.toFinset C'.val.val.support.toFinset) ∧
  25         (1-γ)*(1-10*θ)*Fintype.card V≤(r:ℝ)*M.card := by
  26   have hBpos:0<B := by linarith
  27   have hrpos:0<(r:ℝ) := by exact_mod_cast (show 0<r by omega)
  28   let a:ℝ:=(1-10*θ)/(2*r)
  29   let av:ℝ:=1/(8*B)
  30   let q:ℝ:=(r:ℝ)^2/2
  31   let c:ℝ:=Real.exp 1*(r:ℝ)^2
  32   let K:ℝ:=8*B*c
  33   have hslack:0<1-10*θ := by linarith
  34   have ha:0<a := by dsimp [a];positivity
  35   have hav:0<av := by dsimp [av];positivity
  36   have hq:0≤q := by dsimp [q];positivity
  37   have hc:0<c := by dsimp [c];positivity
  38   have hK:0<K := by dsimp [K];positivity
  39   have hε:0≤10*θ := by positivity
  40   have hmargin:a*r≤1-10*θ := by
  41     have hnon:0≤1-10*θ := by linarith
  42     dsimp [a]
  43     field_simp
  44     linarith
  45   have hscale:2*Real.exp 1*q≤c := by dsimp [q,c];ring_nf;rfl
  46   obtain ⟨Npack,hpack⟩:=ordinary_sampling_packing_from_means r 2 hr K a γ (10*θ) θ av q c
  47     hK ha hγ hγ1 hε hmargin hθ (by linarith) hav hq hc hscale
  48   obtain ⟨Nparam,hparam⟩:=cycle_sampling_almost_regular_parameters B θ c hB hθ hθsmall hc.le
  49   refine ⟨max 1 (max Npack Nparam),?_⟩
  50   intro V _ _ G _ hn d hdlo hdhi hmin hreg w hw hwb
  51   have hnpos:0<Fintype.card V:=by omega
  52   let scale:ℝ:=(Fintype.card V:ℝ)/(2*B)
  53   obtain ⟨D,hDlo,hDhi,hcode,hrows⟩:=hparam (Fintype.card V)
  54     ((le_max_right Npack Nparam).trans ((le_max_right 1 _).trans hn)) d hdlo hdhi
  55   have hp:=ordinary_cycle_sampling_scaled_probability G r hr hnpos B hBpos w hwb
  56   have hlo (v:V):av*(Fintype.card V:ℝ)^2≤scale*(G.degree v:ℝ)/2 := by
  57     have hh:=ordinary_cycle_sampling_vertex_mean_lower (Fintype.card V) (G.degree v) B (Nat.cast_nonneg _) hBpos (hmin v)
  58     dsimp [av,scale]
  59     convert hh using 1 <;> ring
  60   have hrows' (v:V):(1-10*θ)*D≤(1-θ)*(scale*(G.degree v:ℝ)/2) ∧
  61       (1+θ)*(scale*(G.degree v:ℝ)/2)≤D := hrows (G.degree v) (hreg v)
  62   have hpair:scale*((r:ℝ)^2*B)≤q*Fintype.card V := by
  63     have hh:=ordinary_cycle_sampling_pair_mean_scale (Fintype.card V) B r hBpos
  64     dsimp [scale,q]
  65     linarith
  66   have hcode':c*Fintype.card V≤K*D/Fintype.card V := by
  67     dsimp [K];exact hcode
  68   obtain ⟨σ,_,M,hdisj,hcover⟩:=hpack V G
  69     ((le_max_left Npack Nparam).trans ((le_max_right 1 _).trans hn)) D hDlo hDhi w hw B scale hBpos.le
  70     (by dsimp [scale];positivity) (fun C=>(hwb C).2) (fun C=>(hp C).1)
  71     (fun C=>le_trans (hp C).2 (by norm_num)) hlo hrows' hpair hcode'
  72   let f:RetainedOrdinaryCycle G r σ→OrdinaryCycle G r:=Subtype.val
  73   refine ⟨M.image f,?_,?_⟩
  74   · intro C hC C' hC' hne
  75     obtain ⟨x,hx,rfl⟩:=Finset.mem_image.mp hC
  76     obtain ⟨y,hy,rfl⟩:=Finset.mem_image.mp hC'
  77     have hxy:x≠y := fun hh=>hne (congrArg Subtype.val hh)
  78     simpa [retainedOrdinaryCycleVertices,f] using hdisj x hx y hy hxy
  79   · have hcard:(M.image f).card=M.card := Finset.card_image_of_injective _ Subtype.val_injective
  80     rw [hcard]
  81     exact hcover
  82 end Oberwolfach
