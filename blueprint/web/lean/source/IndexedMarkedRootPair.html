   4 /-! Construct the actual transformer input from two displayed realizations
   5 of one balanced indexed graph. Only the first vertex realization is
   6 injective; the second may identify vertices. -/

  12 attribute [local instance] Classical.propDecidable
  13 variable {V₀ U₀ V U E K A S:Type} [Fintype V₀] [Fintype U₀] [Fintype V] [Fintype U]
  14     [Fintype A] [Fintype S] [DecidableEq V₀] [DecidableEq U₀] [DecidableEq V] [DecidableEq U]
  15     {tail head:E→V} {ktail khead:K→V} {H₀ HK:V→U→Prop} {π:Equiv.Perm V} {b m:ℕ}

  17 def indexedMarkedRootPair (G:IndexedMarkedGraph V₀ U₀ A S) (hG:G.IsBalanced m)
  18     (f g:V₀→V) (h k:U₀→U) (hf:Function.Injective f) (hh:Function.Injective h)
  19     (R R':Finset V) (B:Finset U)
  20     (hfR:∀ v,f v∈R) (hgR:∀ v,g v∈R') (hhB:∀ u,h u∈B) (hkB:∀ u,k u∈B)
  21     (hRR:Disjoint R R') (hcross:∀ x∈R,∀ y∈R',¬PermutationClose π b x y)
  22     (hubs:∀ s,h (G.hub s)≠k (G.hub s))
  23     (hsep:∀ a,¬PermutationClose π b (f (G.src a)) (f (G.dst a)))
  24     (hsep':∀ a,¬PermutationClose π b (g (G.src a)) (g (G.dst a)))
  25     (hri:Function.Injective (fun a=>if G.jr a then ({f (G.src a),π (f (G.dst a))}:Finset V) else {f (G.src a),f (G.dst a)}))
  26     (hri':Function.Injective (fun a=>if G.jr a then ({g (G.src a),π (g (G.dst a))}:Finset V) else {g (G.src a),g (G.dst a)}))
  27     (hsi:Function.Injective (fun s=>(f (G.row s),h (G.hub s))))
  28     (hsi':Function.Injective (fun s=>(g (G.row s),k (G.hub s))))
  29     (ha:∀ a,if G.jr a then
  30       (∃ e,ktail e=f (G.src a)∧khead e=f (G.dst a))∧(∃ e,ktail e=g (G.src a)∧khead e=g (G.dst a))
  31       else (∃ e,tail e=f (G.src a)∧head e=f (G.dst a))∧(∃ e,tail e=g (G.src a)∧head e=g (G.dst a)))
  32     (hs:∀ s,if G.js s then HK (f (G.row s)) (h (G.hub s))∧HK (g (G.row s)) (k (G.hub s))
  33       else H₀ (f (G.row s)) (h (G.hub s))∧H₀ (g (G.row s)) (k (G.hub s))):
  34     MarkedRootPair tail head ktail khead H₀ HK π b m A S:=by
  35   let φ:=Function.extend f g id
  36   let ψ:=Function.extend h k id
  37   have hφ:∀ v,φ (f v)=g v:=fun v=>hf.extend_apply g id v
  38   have hψ:∀ u,ψ (h u)=k u:=fun u=>hh.extend_apply k id u
  39   have hbase:=G.map_balanced m hG f h
  40   refine {
  41     src:=f∘G.src,dst:=f∘G.dst,row:=f∘G.row,φ:=φ,ψ:=ψ,u:=h∘G.hub,u':=k∘G.hub,jr:=G.jr,js:=G.js,
  42     R:=R,R':=R',B:=B,disjointRows:=hRR,
  43     rimRoots:=?_,spokeRoots:=?_,hubRoots:=?_,hubMap:=?_,differentHubs:=hubs,
  44     sourceArcSeparation:=hsep,imageArcSeparation:=?_,fourRootSeparation:=?_,
  45     rowBalance:=fun v=>⟨hbase.directed v,hbase.row v⟩,markedRowBalance:=?_,hubBalance:=hbase.hub,
  46     sourceRimInjective:=hri,imageRimInjective:=?_,sourceSpokeInjective:=hsi,imageSpokeInjective:=?_,
  47     rimPresent:=?_,spokePresent:=?_}
  48   · intro a;simpa only [Function.comp_apply,hφ] using
  49       (show f (G.src a)∈R∧f (G.dst a)∈R∧g (G.src a)∈R'∧g (G.dst a)∈R' from ⟨hfR _,hfR _,hgR _,hgR _⟩)
  50   · intro s;simp only [Function.comp_apply,hφ];exact ⟨hfR _,hgR _⟩
  51   · intro s;exact ⟨hhB _,hkB _⟩
  52   · intro s;exact (hψ _).symm
  53   · simpa only [Function.comp_apply,hφ] using hsep'
  54   · intro a x hx y hy hxy
  55     simp only [Function.comp_apply,hφ,Finset.mem_insert,Finset.mem_singleton] at hx hy
  56     rcases hx with rfl|rfl|rfl|rfl <;> rcases hy with rfl|rfl|rfl|rfl
  57     all_goals first | exact False.elim (hxy rfl) | exact hsep a | exact hsep' a |
  58       exact (hsep a) ∘ (permutationClose_symm π b _ _).mp | exact (hsep' a) ∘ (permutationClose_symm π b _ _).mp |
  59       exact hcross _ (hfR _) _ (hgR _) | exact (hcross _ (hfR _) _ (hgR _)) ∘ (permutationClose_symm π b _ _).mp
  60   · intro v
  61     rw [finite_typed_fiber_filter_card,finite_typed_fiber_filter_card]
  62     exact hbase.marked v
  63   · simpa only [Function.comp_apply,hφ] using hri'
  64   · simpa only [Function.comp_apply,hφ] using hsi'
  65   · simpa only [Function.comp_apply,hφ] using ha
  66   · simpa only [Function.comp_apply,hφ] using hs
