  14 theorem ordinary_hub_packing_rim_support (M:Finset (OrdinaryWheel Q F H r))
  15     (T:Finset V) (u:U)
  16     (hunion:M.biUnion ordinaryWheelResources=(ordinaryWheelRimGraph M).edgeFinset.image Sum.inl ∪
  17       T.image (fun v=>Sum.inr (v,u))) (v w:V) (hvw:(ordinaryWheelRimGraph M).Adj v w) : v∈T := by
  18   obtain ⟨W,hW,hWvw⟩:=(ordinaryWheelRimGraph_edges M s(v,w)).mp hvw
  19   have hvW:v∈(ordinaryWheelRim W).support:=hWvw.mem_support_left
  20   have hs:Sum.inr (v,ordinaryWheelHub W)∈M.biUnion ordinaryWheelResources:=by
  21     apply Finset.mem_biUnion.mpr
  22     refine ⟨W,hW,?_⟩
  23     rw [ordinaryWheelResources_spoke]
  24     exact ⟨rfl,hvW⟩
  25   rw [hunion] at hs
  26   rcases Finset.mem_union.mp hs with h|h
  27   · obtain ⟨e,_,he⟩:=Finset.mem_image.mp h
  28     cases he
  29   · obtain ⟨a,ha,he⟩:=Finset.mem_image.mp h
  30     have hav:a=v:=(Prod.mk.inj (Sum.inr.inj he)).1
  31     exact hav ▸ ha

  33 theorem graph_family_degree_by_support {I:Type} [Fintype I]
  34     (G:I→SimpleGraph V) (T:I→Finset V) (D:ℕ) (v:V)
  35     (hdegree:∀ i,(G i).degree v≤D)
  36     (hsupport:∀ i,∀ w,(G i).Adj v w→v∈T i) :
  37     (⨆ i,G i).degree v≤D*(Finset.univ.filter (fun i=>v∈T i)).card := by
  38   have hh:=graph_iSup_local_degree_bound G Finset.univ v D hdegree
  39   simp only [Finset.inter_univ,SimpleGraph.card_neighborFinset_eq_degree] at hh
  40   apply hh.trans
  41   apply Nat.mul_le_mul_left D
  42   apply Finset.card_le_card
  43   intro i hi
  44   obtain ⟨w,_,hw⟩:=(Finset.mem_filter.mp hi).2
  45   exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,hsupport i w hw⟩
