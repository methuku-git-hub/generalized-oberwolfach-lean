  14 def markedSameDirectionSetRequests (j a:Bool) (X:Finset V):Finset (Bool×Bool×V):=
  15   X.image (fun x=>(j,a,x))
  16 
  17 def markedFixedEndpoint (π:Equiv.Perm V) (j a:Bool):Equiv.Perm V:=
  18   if j && a then π else Equiv.refl V
  19 
  20 def markedMovingEndpoint (π:Equiv.Perm V) (j a:Bool):Equiv.Perm V:=
  21   if j && !a then π else Equiv.refl V

  23 theorem marked_same_direction_eligible (P:SimpleGraph V) (π:Equiv.Perm V) (b:ℕ)

  26       ¬PermutationClose π b x z∧P.Adj (markedFixedEndpoint π j a x) (markedMovingEndpoint π j a z):=by
  27   cases j <;> cases a <;>
  28     simp [markedRowEligible,markedRowRim,markedFixedEndpoint,markedMovingEndpoint,P.adj_comm]

  38   intro e he f hf h

  46 theorem marked_same_direction_set_domain_lower (P:SimpleGraph V) (π:Equiv.Perm V) (b:ℕ)
