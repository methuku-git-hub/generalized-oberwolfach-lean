  10 theorem resource_root_omitted_load_relabel
  11     (ei : I ≃ J) (ea : A ≃ B) (R : I → Finset A) (S : J → Finset B)
  12     (hR : ∀ i, S (ei i) = (R i).image ea) (w : J → ℝ)
  13     (keep : A → Prop) (keep' : B → Prop) (hk : ∀ a, keep a ↔ keep' (ea a)) (root : A) :
  14     resourceRootOmittedLoad R (fun i => w (ei i)) keep root =
  15       resourceRootOmittedLoad S w keep' (ea root) := by
  16   unfold resourceRootOmittedLoad
  17   rw [← ei.sum_comp]
  18   apply Finset.sum_congr rfl
  19   intro i _
  20   have hroot : root ∈ R i ↔ ea root ∈ S (ei i) := by
  21     rw [hR,Finset.mem_image]
  22     constructor
  23     · intro h
  24       exact ⟨root,h,rfl⟩
  25     · rintro ⟨a,ha,he⟩
  26       exact ea.injective he ▸ ha
  27   have hkeep : (∀ a∈R i, a≠root → keep a) ↔
  28       (∀ b∈S (ei i), b≠ea root → keep' b) := by
  29     constructor
  30     · intro hh b hb hne
  31       rw [hR,Finset.mem_image] at hb
  32       obtain ⟨a,ha,rfl⟩ := hb
  33       apply (hk a).mp
  34       exact hh a ha (fun he => hne (congrArg ea he))
  35     · intro hh a ha hne
  36       apply (hk a).mpr
  37       apply hh (ea a)
  38       · rw [hR,Finset.mem_image]
  39         exact ⟨a,ha,rfl⟩
  40       · intro he
  41         exact hne (ea.injective he)
  42   exact if_congr (and_congr hroot hkeep) rfl rfl
