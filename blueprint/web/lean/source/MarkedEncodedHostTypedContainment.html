  15 theorem marked_encoded_host_typed_containment_iff
  16     (P:SimpleGraph V) (π:Equiv.Perm V) (d:ℕ)
  17     (tail head:E → V) (ktail khead:K → V)
  18     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  19     (hE:∀ e,¬PermutationClose π (3*d) (tail e) (head e) ∧ P.Adj (tail e) (head e))
  20     (hK:∀ e,¬PermutationClose π (3*d) (ktail e) (khead e) ∧ P.Adj (ktail e) (π (khead e)))
  21     (σ:RawMarkedOutcome V U Γ Color) (I:Finset V) (hI:∀ v,π v∈I↔v∈I)
  22     (hAgree:∀ mark x y,x∈I → y∈I →
  23       (markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) mark x y=1 ↔
  24         (if mark then ∃ e:K,ktail e=x∧khead e=y else ∃ e:E,tail e=x∧head e=y))) :
  25     (∀ mark x y,markedPhysicalRim P π (3*d)
  26        (fun e=>rawMarkedEncodeHost tail head ktail khead π σ I (Sum.inl e)) mark x y=1 →
  27        markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) mark x y=1) ↔
  28     ((∀ e:E,markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) false (tail e) (head e)=1) ∧
  29      (∀ e:K,markedPhysicalRim P π (3*d) (fun e=>σ (Sum.inl e)) true (ktail e) (khead e)=1)) := by
  30   constructor
  31   · intro hs
  32     constructor
  33     · intro e
  34       apply hs false (tail e) (head e)
  35       apply (rawMarkedEncodeHost_rims P π d tail head ktail khead hp hE hK σ I hI hAgree false (tail e) (head e)).mpr
  36       exact ⟨e,rfl,rfl⟩
  37     · intro e
  38       apply hs true (ktail e) (khead e)
  39       apply (rawMarkedEncodeHost_rims P π d tail head ktail khead hp hE hK σ I hI hAgree true (ktail e) (khead e)).mpr
  40       exact ⟨e,rfl,rfl⟩
  41   · rintro ⟨he,hk⟩ mark x y ht
  42     have hh := (rawMarkedEncodeHost_rims P π d tail head ktail khead hp hE hK σ I hI hAgree mark x y).mp ht
  43     cases mark with
  44     | false =>
  45       simp only [Bool.false_eq_true,↓reduceIte] at hh
  46       obtain ⟨e,rfl,rfl⟩ := hh
  47       exact he e
  48     | true =>
  49       simp only [↓reduceIte] at hh
