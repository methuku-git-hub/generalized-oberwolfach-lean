  15     (hfactor : ∀ i,∀ D : SimpleGraph V,(∀ v,D.degree v≤2*(Finset.univ.filter (fun i=>v∈T i)).card) →
  16       HasResourceDecomposition (localOrdinaryVertices (Q:=Q\D) (F:=F) (T:=T i) (r:=r)) (T i)) :
  17     ∃ M : Finset (OrdinaryWheel Q F H r),
  18       (∀ W∈M,∀ Z∈M,W≠Z → Disjoint (ordinaryWheelResources W) (ordinaryWheelResources Z)) ∧

  20         Finset.univ.biUnion (fun i=>(T i).image (fun v=>Sum.inr (v,u i))) ∧

  22       (∀ i,∀ v∈T i,∃! W : M,Sum.inr (v,u i)∈ordinaryWheelResources W.val) := by
  23   let allowed:=fun (i : Fin n) (M : Finset (OrdinaryWheel Q F H r))=>
  24     (∀ W∈M,∀ Z∈M,W≠Z → Disjoint (ordinaryWheelResources W) (ordinaryWheelResources Z)) ∧

  27     (∀ v,(ordinaryWheelRimGraph M).degree v≤2) ∧ (∀ W∈M,ordinaryWheelHub W=u i)
  28   let comp:=fun M N : Finset (OrdinaryWheel Q F H r)=>
  29     Disjoint (M.biUnion ordinaryWheelResources) (N.biUnion ordinaryWheelResources)
  30   obtain ⟨C,hC,hcross⟩:=sequential_select_prefix n allowed comp ∅ (by
  31     intro i A hA
  32     let D:=⨆ j,ordinaryWheelRimGraph (A j)
  33     have hdeg (j : Fin n) (v : V) : (ordinaryWheelRimGraph (A j)).degree v≤2 := by

  35       · exact (hA.2.1 j hj).2.2.1 v
  36       · rw [hA.1 j (by omega),ordinaryWheelRimGraph_empty];simp
  37     have hfuture (j : Fin n) (hj : i.val≤j.val) : ordinaryWheelRimGraph (A j)=⊥ := by
  38       rw [hA.1 j hj,ordinaryWheelRimGraph_empty]

  43       · exact ordinary_hub_packing_rim_support (A j) (T j) (u j) (hA.2.1 j hj).2.1 v w hjw
  44       · rw [hfuture j (by omega)] at hjw
  45         exact hjw.elim
  46     obtain ⟨M,hM,hunion,hdegM,hhub,havoid⟩:=ordinary_factor_wheels_avoiding D (T i) (u i) (hH i) (hfactor i D hD)
  47     refine ⟨M,⟨hM,hunion,hdegM,hhub⟩,?_⟩
  48     intro j hji
  49     apply ordinary_wheel_packings_disjoint (A j) M (u j) (u i)
  50       (fun h=>by have h':=congrArg Fin.val (hu h);omega) (hA.2.1 j hji).2.2.2 hhub
  51     exact (havoid.mono_right (le_iSup (fun j=>ordinaryWheelRimGraph (A j)) j)).symm)
  52   have hcross' (i j : Fin n) (hne : i≠j) :
  53       Disjoint ((C i).biUnion ordinaryWheelResources) ((C j).biUnion ordinaryWheelResources) := by
  54     rcases lt_or_gt_of_ne hne with hij|hji
  55     · exact hcross i j hij
  56     · exact (hcross j i hji).symm
  57   let M:=Finset.univ.biUnion C
  58   have hM : ∀ W∈M,∀ Z∈M,W≠Z → Disjoint (ordinaryWheelResources W) (ordinaryWheelResources Z) := by
  59     intro W hW Z hZ hne
  60     obtain ⟨i,_,hWi⟩:=Finset.mem_biUnion.mp hW
  61     obtain ⟨j,_,hZj⟩:=Finset.mem_biUnion.mp hZ
  62     by_cases hij : i=j
  63     · subst j;exact (hC i).1 W hWi Z hZj hne
  64     · exact (hcross' i j hij).mono
  65         (resource_subset_union ordinaryWheelResources (C i) W hWi)
  66         (resource_subset_union ordinaryWheelResources (C j) Z hZj)
  67   have hunion : M.biUnion ordinaryWheelResources=(ordinaryWheelRimGraph M).edgeFinset.image Sum.inl ∪
  68       Finset.univ.biUnion (fun i=>(T i).image (fun v=>Sum.inr (v,u i))) := by
  69     ext e
  70     cases e with
  71     | inl e =>
  72       have he:=(ordinaryWheelRimGraph_edges M e).symm
  73       simpa only [Finset.mem_biUnion,ordinaryWheelResources_rim,Finset.mem_union,
  74         Finset.mem_image,Sum.inl.injEq,Sum.inr_ne_inl,and_false,exists_false,or_false,
  75         SimpleGraph.mem_edgeFinset,exists_eq_right] using he
  76     | inr vu =>
  77       constructor
  78       · intro he
  79         obtain ⟨W,hW,he⟩:=Finset.mem_biUnion.mp he
  80         obtain ⟨i,_,hWi⟩:=Finset.mem_biUnion.mp hW

  82         rw [(hC i).2.1] at hi
  83         have hsp : Sum.inr (α:=Sym2 V) vu∈(T i).image (fun v=>Sum.inr (v,u i)) := by
  84           rcases Finset.mem_union.mp hi with hi|hi
  85           · obtain ⟨e,_,he⟩:=Finset.mem_image.mp hi;cases he
  86           · exact hi
  87         exact Finset.mem_union_right _ (Finset.mem_biUnion.mpr ⟨i,Finset.mem_univ _,hsp⟩)
  88       · intro he
  89         rcases Finset.mem_union.mp he with he|he
  90         · obtain ⟨e,_,he⟩:=Finset.mem_image.mp he;cases he
  91         · obtain ⟨i,_,hi⟩:=Finset.mem_biUnion.mp he
  92           have hi' : Sum.inr vu∈(C i).biUnion ordinaryWheelResources:=by
  93             rw [(hC i).2.1];exact Finset.mem_union_right _ hi
  94           obtain ⟨W,hWi,he⟩:=Finset.mem_biUnion.mp hi'
  95           exact Finset.mem_biUnion.mpr ⟨W,Finset.mem_biUnion.mpr ⟨i,Finset.mem_univ _,hWi⟩,he⟩
  96   refine ⟨M,hM,hunion,?_,?_⟩
  97   · intro v
