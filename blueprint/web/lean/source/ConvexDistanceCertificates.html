  14 theorem convexDistanceSq_certificate
  15     (A : Finset (I → Ω)) (hA : A.Nonempty) (x : I → Ω)
  16     (w : I → ℝ) {c D t : ℝ} (hc : 0≤c) (ht : 0≤t)
  17     (hw : ∀ i,0≤w i ∧ w i≤c) (hD : ∑ i,w i≤D)
  18     (hsep : ∀ y∈A,t≤∑ i,w i*disagreementVector x y i) :
  19     t^2≤c*D*convexDistanceSq A x := by
  20   obtain ⟨z,hz,he,_⟩ := finiteHullEnergy_attained (disagreementVector x) A hA
  21   have hlower : t≤∑ i,w i*z i := by
  22     have hsub : disagreementVector x '' (A : Set (I → Ω))⊆
  23         {z | t≤∑ i,w i*z i} := by
  24       rintro _ ⟨y,hy,rfl⟩
  25       exact hsep y hy
  26     have hconv : Convex ℝ {z : I → ℝ | t≤∑ i,w i*z i} := by
  27       intro u hu v hv a b ha hb hab
  28       change t≤∑ i,w i*(a*u i+b*v i)
  29       have heq : (∑ i,w i*(a*u i+b*v i))=a*(∑ i,w i*u i)+b*(∑ i,w i*v i) := by
  30         simp only [Finset.mul_sum,← Finset.sum_add_distrib]
  31         apply Finset.sum_congr rfl
  32         intro i _
  33         ring
  34       rw [heq]
  35       nlinarith [mul_le_mul_of_nonneg_left hu ha,mul_le_mul_of_nonneg_left hv hb]
  36     exact (convexHull_min hsub hconv) hz
  37   have hsq : (∑ i,(w i)^2)≤c*D := by
  38     calc
  39       _ ≤ ∑ i,c*w i := Finset.sum_le_sum (fun i _=>by
  40         nlinarith [(hw i).1,(hw i).2])
  41       _ = c*∑ i,w i := (Finset.mul_sum _ _ _).symm
  42       _ ≤ _ := mul_le_mul_of_nonneg_left hD hc
  43   have hCS := Finset.sum_mul_sq_le_sq_mul_sq Finset.univ w z
  44   have hz0 : 0≤∑ i,(z i)^2 := Finset.sum_nonneg (fun i _=>sq_nonneg _)
  45   have hm := mul_le_mul_of_nonneg_right hsq hz0
  46   change (∑ i,(z i)^2)=convexDistanceSq A x at he
  47   rw [he] at hCS hm
  48   nlinarith

  50 theorem convexDistanceSq_value_certificate
  51     (R : (I → Ω) → ℝ) (A : Finset (I → Ω)) (hA : A.Nonempty)
  52     (x : I → Ω) (w : I → ℝ) {c D m t : ℝ} (hc : 0≤c) (ht : 0≤t)
  53     (hw : ∀ i,0≤w i ∧ w i≤c) (hD : ∑ i,w i≤D)
  54     (hAvalue : ∀ y∈A,R y≤m) (hxvalue : m+t≤R x)
  55     (hcert : ∀ y,R x≤R y+∑ i,if x i≠y i then w i else 0) :
  56     t^2≤c*D*convexDistanceSq A x := by
  57   apply convexDistanceSq_certificate A hA x w hc ht hw hD
  58   intro y hy
  59   have heq : (∑ i,if x i≠y i then w i else 0)=
  60       ∑ i,w i*disagreementVector x y i := by
  61     apply Finset.sum_congr rfl
  62     intro i _
  63     by_cases h : x i=y i <;> simp [disagreementVector,h]
  64   have hh := hcert y
  65   rw [heq] at hh
  66   linarith [hAvalue y hy]
