  13 theorem disjoint_resource_tuple_injective (R : C → Finset V) (f : C → Fin c → V)
  14     (hf : ∀ C₀,Function.Injective (f C₀)) (hmem : ∀ C₀ i,f C₀ i∈R C₀)
  15     (M : Finset C) (hM : ∀ C₀∈M,∀ C₁∈M,C₀≠C₁ → Disjoint (R C₀) (R C₁)) :
  16     Function.Injective (fun p : M×Fin c => f p.1.val p.2) := by
  17   rintro ⟨C₀,i⟩ ⟨C₁,j⟩ he
  18   have hC : C₀=C₁ := by
  19     apply Subtype.ext
  20     by_contra hne
  21     have hv : f C₀.val i∈R C₁.val := by
  22       change f C₀.val i=f C₁.val j at he
  23       rw [he]
  24       exact hmem C₁.val j
  25     exact Finset.disjoint_left.mp (hM _ C₀.property _ C₁.property hne) (hmem _ _) hv
  26   subst C₁
  27   exact Prod.ext rfl (hf C₀.val he)

  30     (hM : ∀ C₀∈M,∀ C₁∈M,C₀≠C₁ → Disjoint (R C₀) (R C₁))
  31     (v : V) (hv : v∈M.biUnion R) :
