  12 theorem selected_pair_transfer_identity {U : Type*} [Fintype U] [DecidableEq U]
  13     (P : U → Prop) (z : U → ℝ)
  14     (hz : ∑ i : {u // P u},z i.val=0) (hout : ∀ u,¬P u → z u=0) (h : U) :
  15     (∑ i : {u // P u},∑ j : {u // P u},
  16       z i.val*((if h=i.val then (1:ℝ) else 0)-(if h=j.val then 1 else 0)))/
  17         (Fintype.card {u // P u}:ℝ)=z h := by
  18   by_cases hh : P h
  19   · let hP : {u // P u} := ⟨h,hh⟩
  20     have he := balanced_pair_transfer_identity (fun i : {u // P u} => z i.val) hz hP
  21     convert he using 1
  22     congr 1
  23     apply Finset.sum_congr rfl
  24     intro i _
  25     apply Finset.sum_congr rfl
  26     intro j _
  27     have hi : h=i.val ↔ hP=i := ⟨fun e => Subtype.ext e,fun e => congrArg Subtype.val e⟩
  28     have hj : h=j.val ↔ hP=j := ⟨fun e => Subtype.ext e,fun e => congrArg Subtype.val e⟩
  29     simp only [hi,hj]
  30   · rw [hout h hh]
  31     have hn (i : {u // P u}) : h≠i.val := by
  32       intro he
  33       exact hh (he ▸ i.property)
  34     simp only [hn,ite_false,sub_self,mul_zero,Finset.sum_const_zero,zero_div]
