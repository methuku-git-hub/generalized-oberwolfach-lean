  13 theorem finiteLaw_mass_le_map {P B : Type*} [Fintype P] [Fintype B]
  14     (μ : FiniteLaw P) (f : P → B) (p : P) :
  15     μ.mass p ≤ (μ.map f).mass (f p) := by
  16   have hh := Finset.single_le_sum (s := Finset.univ)
  17     (f := fun q => if f q = f p then μ.mass q else 0)
  18     (fun q _ => by split_ifs; exact μ.nonneg q; exact le_refl _) (Finset.mem_univ p)
  19   simpa only [ite_true, FiniteLaw.map] using hh

  21 def FiniteLaw.conditionOnMap {P B : Type*} [Fintype P] [Fintype B]
  22     (μ : FiniteLaw P) (f : P → B) (b : B) : FiniteLaw P := by
  23   classical
  24   by_cases hz : (μ.map f).mass b = 0
  25   · exact μ
  26   · exact {
  27       mass p := if f p = b then μ.mass p / (μ.map f).mass b else 0
  28       nonneg p := by
  29         split_ifs
  30         · exact div_nonneg (μ.nonneg p) ((μ.map f).nonneg b)
  31         · exact le_refl _
  32       total := by
  33         have hh (p : P) : (if f p = b then μ.mass p / (μ.map f).mass b else 0) =
  34             (if f p = b then μ.mass p else 0) / (μ.map f).mass b := by
  35           split_ifs <;> simp
  36         simp_rw [hh]
  37         rw [← Finset.sum_div]
  38         exact div_self hz }

  40 theorem finiteLaw_conditionOnMap_weight {P B : Type*} [Fintype P] [Fintype B]
  41     (μ : FiniteLaw P) (f : P → B) (b : B) (p : P) :
  42     (μ.map f).mass b * (μ.conditionOnMap f b).mass p =
  43       if f p = b then μ.mass p else 0 := by
  44   by_cases hz : (μ.map f).mass b = 0
  45   · simp only [hz, zero_mul]
  46     by_cases hp : f p = b
  47     · have hh := finiteLaw_mass_le_map μ f p
  48       rw [hp, hz] at hh
  49       simp only [hp, ite_true]
  50       exact (le_antisymm hh (μ.nonneg p)).symm
  51     · simp only [hp, ite_false]
  52   · simp only [FiniteLaw.conditionOnMap, hz, dite_false]
  53     by_cases hp : f p = b
  54     · simp only [hp, ite_true]
  55       exact mul_div_cancel₀ _ hz
  56     · simp only [hp, ite_false, mul_zero]

  58 theorem finiteLaw_conditionOnMap_reconstruct {T P B : Type*}
  59     [Fintype T] [Fintype P] [Fintype B]
  60     (ν : FiniteLaw T) (μ : FiniteLaw P) (g : T → B) (f : P → B)
  61     (h : ν.map g = μ.map f) (p : P) :
  62     (∑ t, ν.mass t * (μ.conditionOnMap f (g t)).mass p) = μ.mass p := by
  63   change ν.expectation (fun t => (μ.conditionOnMap f (g t)).mass p) = _
  64   rw [← finiteLaw_map_expectation ν g (fun b => (μ.conditionOnMap f b).mass p), h]
  65   unfold FiniteLaw.expectation
  66   simp_rw [finiteLaw_conditionOnMap_weight]
  67   simp

  78 theorem finiteLaw_conditional_realization_with_formula
  79     {T I : Type*} [Fintype T] [Fintype I] [DecidableEq I]
  80     {P B : I → Type*} [∀ i, Fintype (P i)] [∀ i, Fintype (B i)]
  81     (ν : FiniteLaw T) (μ : ∀ i, FiniteLaw (P i))
  82     (g : ∀ i, T → B i) (f : ∀ i, P i → B i)
  83     (h : ∀ i, ν.map (g i) = (μ i).map (f i)) :
  84     ∃ Ω : FiniteLaw (Σ _ : T, ∀ i, P i),
  85       (∀ t, (∑ σ, Ω.mass ⟨t, σ⟩) = ν.mass t) ∧
  86       (∀ i p, Ω.probability (fun z => z.2 i = p) = (μ i).mass p) ∧
  87       (∀ t σ, 0 < Ω.mass ⟨t, σ⟩ → ∀ i, f i (σ i) = g i t) ∧
  88       (∀ t σ, Ω.mass ⟨t, σ⟩ = ν.mass t *
  89         ∏ i, ((μ i).conditionOnMap (f i) (g i t)).mass (σ i)) := by
  90   let κ (t : T) (i : I) := (μ i).conditionOnMap (f i) (g i t)
  91   let Ω := ν.sigma (fun t => FiniteLaw.pi (κ t))
  92   refine ⟨Ω, ?_, ?_, ?_, ?_⟩
  93   · intro t
  94     change (∑ σ, ν.mass t * (FiniteLaw.pi (κ t)).mass σ) = _
  95     rw [← Finset.mul_sum, (FiniteLaw.pi (κ t)).total, mul_one]
  96   · intro i p
  97     change (∑ z : (Σ _ : T, ∀ i, P i), if z.2 i = p then
  98       ν.mass z.1 * (FiniteLaw.pi (κ z.1)).mass z.2 else 0) = _
  99     rw [Fintype.sum_sigma]
 100     have hh (t : T) :
 101         (∑ σ, if σ i = p then ν.mass t * (FiniteLaw.pi (κ t)).mass σ else 0) =
 102         ν.mass t * (κ t i).mass p := by
 103       rw [← finiteLaw_pi_coordinate_probability (κ t) i p]
 104       unfold FiniteLaw.probability
 105       rw [Finset.mul_sum]
 106       apply Finset.sum_congr rfl
 107       intro σ hσ
 108       split_ifs <;> simp
 109     simp_rw [hh]
 110     exact finiteLaw_conditionOnMap_reconstruct ν (μ i) (g i) (f i) (h i) p
 111   · intro t σ hpos i
 112     by_contra hbad
 113     have hv : 0 < ν.mass t := by
 114       have hh := (mul_pos_iff.mp hpos)
 115       rcases hh with ⟨hv, _⟩ | ⟨hv, _⟩
 116       · exact hv
 117       · exact False.elim ((not_lt_of_ge (ν.nonneg t)) hv)
 118     have hK : 0 < ((μ i).map (f i)).mass (g i t) := by
 119       rw [← h i]
 120       exact hv.trans_le (finiteLaw_mass_le_map ν (g i) t)
 121     have hz : (κ t i).mass (σ i) = 0 := by
 122       dsimp only [κ]
 123       simp only [FiniteLaw.conditionOnMap, hK.ne', dite_false, hbad, ite_false]
 124     have hprod : (FiniteLaw.pi (κ t)).mass σ = 0 := by
 125       exact Finset.prod_eq_zero (Finset.mem_univ i) hz
 126     change 0 < ν.mass t * (FiniteLaw.pi (κ t)).mass σ at hpos
 127     rw [hprod, mul_zero] at hpos
 128     exact (lt_irrefl 0) hpos
 129   · intro t σ
 130     rfl

 145 theorem weighted_conditional_realization_with_formula
 146     {T I : Type*} [Fintype T] [Fintype I] [DecidableEq I]
 147     {P E : I → Type*} [∀ i, Fintype (P i)] [∀ i, Fintype (E i)]
 148     (J : T → ℝ) (w : ∀ i, P i → ℝ) (B : ℝ) (hB : 0 < B)
 149     (hJ : ∀ t, 0 ≤ J t) (hw : ∀ i p, 0 ≤ w i p)
 150     (hJsum : ∑ t, J t = B) (hwsum : ∀ i, ∑ p, w i p = B)
 151     (g : ∀ i, T → E i) (f : ∀ i, P i → E i)
 152     (hmarg : ∀ i e, (∑ t, if g i t = e then J t else 0) =
 153       ∑ p, if f i p = e then w i p else 0) :
 154     ∃ W : (Σ _ : T, ∀ i, P i) → ℝ,
 155       (∀ z, 0 ≤ W z) ∧ (∑ z, W z) = B ∧
 156       (∀ t, (∑ σ, W ⟨t, σ⟩) = J t) ∧
 157       (∀ i p, (∑ z, if z.2 i = p then W z else 0) = w i p) ∧
 158       (∀ t σ, 0 < W ⟨t, σ⟩ → ∀ i, f i (σ i) = g i t) ∧
 159       (∀ t σ, W ⟨t, σ⟩ = J t * ∏ i,
 160         if f i (σ i) = g i t then
 161           w i (σ i) / (∑ p, if f i p = g i t then w i p else 0) else 0) := by
 162   let ν : FiniteLaw T := {
 163     mass t := J t / B
 164     nonneg t := div_nonneg (hJ t) hB.le
 165     total := by rw [← Finset.sum_div, hJsum, div_self hB.ne'] }
 166   let μ (i : I) : FiniteLaw (P i) := {
 167     mass p := w i p / B
 168     nonneg p := div_nonneg (hw i p) hB.le
 169     total := by rw [← Finset.sum_div, hwsum i, div_self hB.ne'] }
 170   have hm (i : I) : ν.map (g i) = (μ i).map (f i) := by
 171     apply finiteLaw_ext
 172     intro e
 173     change (∑ t, if g i t = e then J t / B else 0) =
 174       ∑ p, if f i p = e then w i p / B else 0
 175     have hid (a : ℝ) (b : Prop) [Decidable b] :
 176         (if b then a / B else 0) = (if b then a else 0) / B := by
 177       split_ifs <;> simp
 178     simp_rw [hid, ← Finset.sum_div]
 179     rw [hmarg i e]
 180   obtain ⟨Ω, hboundary, hpaths, hsupport, hformula⟩ :=
 181     finiteLaw_conditional_realization_with_formula ν μ g f hm
 182   refine ⟨fun z => B * Ω.mass z, ?_, ?_, ?_, ?_, ?_, ?_⟩
 183   · intro z
 184     exact mul_nonneg hB.le (Ω.nonneg z)
 185   · rw [← Finset.mul_sum, Ω.total, mul_one]
 186   · intro t
 187     rw [← Finset.mul_sum, hboundary]
 188     exact mul_div_cancel₀ _ hB.ne'
 189   · intro i p
 190     have heq (z : Σ _ : T, ∀ i, P i) :
 191         (if z.2 i = p then B * Ω.mass z else 0) =
 192         B * (if z.2 i = p then Ω.mass z else 0) := by
 193       split_ifs <;> simp
 194     simp_rw [heq]
 195     rw [← Finset.mul_sum]
 196     change B * Ω.probability (fun z => z.2 i = p) = _
 197     rw [hpaths i p]
 198     exact mul_div_cancel₀ _ hB.ne'
 199   · intro t σ hp
 200     exact hsupport t σ ((mul_pos_iff_of_pos_left hB).mp hp)
 201   · intro t σ
 202     change B * Ω.mass ⟨t, σ⟩ = _
 203     rw [hformula]
 204     change B * (J t / B * _) = _
 205     rw [← mul_assoc, mul_div_cancel₀ _ hB.ne']
 206     by_cases hzero : J t = 0
 207     · simp only [hzero, zero_mul]
 208     · apply congrArg (fun z => J t * z)
 209       apply Finset.prod_congr rfl
 210       intro i _
 211       have hJpos : 0 < J t := lt_of_le_of_ne (hJ t) (Ne.symm hzero)
 212       have hKpos : 0 < ((μ i).map (f i)).mass (g i t) := by
 213         rw [← hm i]
 214         exact (div_pos hJpos hB).trans_le (finiteLaw_mass_le_map ν (g i) t)
 215       have hKform : ((μ i).map (f i)).mass (g i t) =
 216           (∑ p, if f i p = g i t then w i p else 0) / B := by
 217         change (∑ p, if f i p = g i t then w i p / B else 0) = _
 218         rw [Finset.sum_div]
 219         apply Finset.sum_congr rfl
 220         intro p _
 221         split_ifs <;> simp
 222       have hc : ((μ i).conditionOnMap (f i) (g i t)).mass (σ i) =
 223           (if f i (σ i) = g i t then (μ i).mass (σ i) else 0) /
 224             ((μ i).map (f i)).mass (g i t) := by
 225         apply (eq_div_iff hKpos.ne').mpr
 226         simpa only [mul_comm] using finiteLaw_conditionOnMap_weight (μ i) (f i) (g i t) (σ i)
 227       rw [hc, hKform]
 228       by_cases he : f i (σ i) = g i t
 229       · simp only [he, ite_true]
 230         exact div_div_div_cancel_right₀ hB.ne' _ _
 231       · simp only [he, ite_false, zero_div]
