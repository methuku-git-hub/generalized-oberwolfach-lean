   9 theorem ordinary_position_capacity_from_displayed_bounds
  10     (n θ C ε lam invd inva A Z v tail:ℝ)
  11     (hn:0≤n) (hθ:0≤θ) (hC:1≤C)
  12     (hε:0≤ε) (hlam:0≤lam) (hinvd:0≤ invd) (hinva:0≤ inva)
  13     (hearly:C*(ε+ε^2*inva)≤1/100000)
  14     (hlate:C*invd+C*lam+C*lam*inva≤θ/100000)
  15     (hA:A≤(2*invd+lam)*n)
  16     (hZ:Z≤(2*θ*ε+2*lam)*n)
  17     (hv:v≤4*C*(θ*ε^2+lam*ε)*inva*n+tail)
  18     (smallTail:8*tail+10≤θ*n/16):
  19     8*A+8*v+4*Z+10≤θ*n/4 := by
  20   have hC0:0≤C:=by linarith
  21   have heps2:0≤C*ε^2*inva:=by positivity
  22   have heps:C*ε≤1/100000:=by nlinarith only [hearly,heps2]
  23   have hepsle:ε≤1:=by nlinarith only [heps,mul_nonneg (show 0≤C-1 by linarith) hε]
  24   have hquad:C*ε^2*inva≤1/100000:=by nlinarith only [hearly,heps, hC, hε]
  25   have hp:invd≤θ/100000:=by nlinarith only [hlate, hlam, hC0, mul_nonneg (show 0≤C-1 by linarith) hinvd, mul_nonneg (mul_nonneg hC0 hlam) hinva]
  26   have hl:lam≤θ/100000:=by nlinarith only [hlate, hinvd, hC0, mul_nonneg (show 0≤C-1 by linarith) hlam, mul_nonneg (mul_nonneg hC0 hlam) hinva]
  27   have hlcross:C*lam*inva≤θ/100000:=by nlinarith only [hlate,mul_nonneg hC0 hinvd,mul_nonneg hC0 hlam]
  28   have ha: C*lam*ε*inva≤θ/100000:=by
  29     nlinarith only [hlcross, mul_nonneg (mul_nonneg (mul_nonneg hC0 hlam) hinva) (show 0≤1-ε by linarith)]
  30   have hb:C*θ*ε^2*inva≤θ/100000:=by
  31     nlinarith only [mul_nonneg hθ (show 0≤1/100000-C*ε^2*inva by linarith)]
  32   have he:θ*ε≤θ/100000:=by
  33     have hh:ε≤1/100000:=by nlinarith only [heps,mul_nonneg (show 0≤C-1 by linarith) hε]
  34     nlinarith only [mul_nonneg hθ (show 0≤1/100000-ε by linarith)]
  35   have hcoef:16*invd+16*lam+32*C*(θ*ε^2+lam*ε)*inva+8*θ*ε≤θ/8:=by nlinarith only [hp,hl,ha,hb,he,hθ]
  36   have hcoefn:=mul_le_mul_of_nonneg_right hcoef hn
  37   nlinarith only [hA,hZ,hv,hcoefn,smallTail,hθ,hn]
  38 end Oberwolfach
