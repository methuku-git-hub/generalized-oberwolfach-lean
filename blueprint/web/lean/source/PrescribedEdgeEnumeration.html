  14 theorem finset_injective_enumeration {A:Type} [DecidableEq A] (S:Finset A):
  15     ∃ f:Fin S.card→A,Function.Injective f ∧ Finset.univ.image f=S:=by

  26 theorem prescribed_inner_edge_enumeration
  27     {V U:Type} [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U]
  28     (Q F:SimpleGraph V) (H:V→U→Prop) (I:Finset V) (J:Finset U)
  29     (E:Finset (Sym2 V⊕(V×U))) (D:ℕ)
  30     (hE:E⊆ordinaryHostResources (spanningRestriction Q I) (fun v u=>H v u∧v∈I∧u∈J))
  31     (hF:∀ v w,Sum.inl s(v,w)∈E→¬F.Adj v w)
  32     (hD:∀ t,(E.filter (ordinaryResourceIncident t)).card≤D):
  33     ∃ Nr Ns:ℕ,∃ x y:Fin Nr→V,∃ z:Fin Ns→V,∃ u:Fin Ns→U,
  34       Nr+Ns=E.card ∧
  35       (∀ i,x i∈I) ∧ (∀ i,y i∈I) ∧ (∀ i,Q.Adj (x i) (y i)) ∧ (∀ i,¬F.Adj (x i) (y i)) ∧
  36       (∀ i j,i≠j→¬((x i=x j∧y i=y j)∨(x i=y j∧y i=x j))) ∧
  37       (∀ v,(Finset.univ.filter (fun i=>v=x i∨v=y i)).card≤D) ∧
  38       (∀ i,z i∈I) ∧ (∀ i,u i∈J) ∧ (∀ i,H (z i) (u i)) ∧
  39       Function.Injective (fun i=>(z i,u i)) ∧
  40       (∀ v,(Finset.univ.filter (fun i=>v=z i)).card≤D) ∧
  41       (∀ a,(Finset.univ.filter (fun i=>u i=a)).card≤D) ∧
  42       Finset.univ.image (Sum.elim (fun i=>Sum.inl s(x i,y i)) (fun i=>Sum.inr (z i,u i)))=E:=by

  49     exact (Finset.mem_filter.mp hh).2
  50   have hgmem (i:Fin S.card):Sum.inr (g i)∈E:=by

  52     exact (Finset.mem_filter.mp hh).2
  53   let x:=fun i=>(f i).out.1
  54   let y:=fun i=>(f i).out.2
  55   let z:=fun i=>(g i).1
  56   let u:=fun i=>(g i).2

  58   have hr (i:Fin R.card):Q.Adj (x i) (y i)∧x i∈I∧y i∈I:=by
  59     have he:=hE (hfmem i)
  60     rw [ordinaryHostResources_rim,←hxy i] at he
