  11 theorem product_subtype_event_bound {A U : Type*} [Fintype A] [Fintype U] [DecidableEq U]
  12     (admissible : A×U → Prop) (w : {p // admissible p} → ℝ) (M : ℝ) (hM : 0≤M)
  13     (hw : ∀ p,w p≤M) (S : A → Prop) (T : Finset U)
  14     (P : {p // admissible p} → Prop) (hP : ∀ p,P p → S p.val.1 ∧ p.val.2∈T) :
  15     (∑ p,if P p then w p else 0)≤
  16       (T.card:ℝ)*(∑ a,if S a then M else 0) := by
  17   classical
  18   let g (p : A×U) : ℝ := if S p.1 ∧ p.2∈T then M else 0
  19   have hg (p : A×U) : 0≤g p := by dsimp only [g];split_ifs <;> positivity
  20   calc
  21     _ ≤ ∑ p : {p // admissible p},g p.val := by
  22       apply Finset.sum_le_sum
  23       intro p _
  24       by_cases hp : P p
  25       · simp only [if_pos hp,g,if_pos (hP p hp)]
  26         exact hw p
  27       · rw [if_neg hp];exact hg p.val
  28     _ ≤ ∑ p : A×U,g p :=
  29       finite_injective_sum_le (fun p : {p // admissible p}=>p.val) Subtype.val_injective g hg
  30     _ = _ := by
  31       rw [Fintype.sum_prod_type,Finset.sum_comm]
  32       have he (u : U) : (∑ a : A,g (a,u))=
  33           if u∈T then (∑ a : A,if S a then M else 0) else 0 := by
  34         by_cases hu : u∈T
  35         · simp only [g,hu,and_true,ite_true]
  36         · simp only [g,hu,and_false,ite_false,Finset.sum_const_zero]
  37       simp only [he]
  38       rw [← Finset.sum_filter]
  39       have ht : Finset.univ.filter (fun u=>u∈T)=T := by ext;simp
  40       rw [ht,Finset.sum_const,nsmul_eq_mul]
