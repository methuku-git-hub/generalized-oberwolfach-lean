  16  theorem marked_cycle_family_packing_no_old_arc
  17     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  18     (C:I→MarkedCycleTuple tail head ktail khead m π b) (R:Finset V) (P:I→Finset V)
  19     (hvertices:∀ i a,(C i).val a∈R∪P i)
  20     (hPP:∀ i j,i≠j→Disjoint (P i) (P j))
  21     (hroot:∀ i a,¬((C i).val a∈R∧(C i).val (finRotate (m+1) a)∈R)):
  22     ∀ i j,i≠j→Disjoint (markedCyclePhysicalEdges tail head ktail khead m π b (C i))
  23       (markedCyclePhysicalEdges tail head ktail khead m π b (C j)):=by
  24   intro i j hij
  25   apply Finset.disjoint_left.mpr
  26   intro e he hf
  27   obtain ⟨a,_,ha⟩:=Finset.mem_image.mp he
  28   obtain ⟨z,_,hz⟩:=Finset.mem_image.mp hf
  29   have hc:=marked_cycle_common_physical_endpoints hp (C i) (C j) a z (ha.trans hz.symm)
  30   have common (x:V) (hx:x∈R∪P i) (hy:x∈R∪P j):x∈R:=by
  31     rcases Finset.mem_union.mp hx with hx|hx
  32     · exact hx
  33     · rcases Finset.mem_union.mp hy with hy|hy
  34       · exact hy
  35       · exact (Finset.disjoint_left.mp (hPP i j hij) hx hy).elim
  36   exact hroot i a ⟨common _ (hvertices i a) (hc.1 ▸ hvertices j z),
  37     common _ (hvertices i _) (hc.2 ▸ hvertices j _)⟩

  39  theorem marked_spoke_wheel_family_packing
  40     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  41     (W:I→MarkedWheel tail head ktail khead m π b H₀ HK) (R:Finset V) (P:I→Finset V)
  42     (v:I→V) (hold:Function.Injective (fun i=>(v i,(W i).2.val)))
  43     (hvertices:∀ i a,(W i).1.val a∈R∪P i)
  44     (hPP:∀ i j,i≠j→Disjoint (P i) (P j))
  45     (hroot:∀ i a,(W i).1.val a∈R→(W i).1.val a=v i)
  46     (hnoarc:∀ i a,¬((W i).1.val a∈R∧(W i).1.val (finRotate (m+1) a)∈R)):
  47     ∀ i j,i≠j→Disjoint (markedWheelPhysicalResources (W i)) (markedWheelPhysicalResources (W j)):=by
  48   have hr:=marked_cycle_family_packing_no_old_arc hp (fun i=>(W i).1) R P hvertices hPP hnoarc
  49   intro i j hij
  50   apply Finset.disjoint_left.mpr
  51   intro e he hf
  52   cases e with
  53   | inl e=>
  54     exact Finset.disjoint_left.mp (hr i j hij) ((markedWheelPhysicalResources_rim _ _).mp he)
  55       ((markedWheelPhysicalResources_rim _ _).mp hf)
  56   | inr xu=>
  57     rw [markedWheelPhysicalResources_spoke] at he hf
  58     obtain ⟨a,_,ha⟩:=Finset.mem_image.mp he.2
  59     obtain ⟨z,_,hz⟩:=Finset.mem_image.mp hf.2
  60     have hx:xu.1∈R:=by
  61       have hx:=ha ▸ hvertices i a
  62       have hy:=hz ▸ hvertices j z
  63       rcases Finset.mem_union.mp hx with hx|hx
  64       · exact hx
  65       · rcases Finset.mem_union.mp hy with hy|hy
  66         · exact hy
  67         · exact (Finset.disjoint_left.mp (hPP i j hij) hx hy).elim
  68     have hv:v i=v j:=(hroot i a (ha ▸ hx)).symm.trans (ha.trans (hz.symm.trans (hroot j z (hz ▸ hx))))
  69     exact hij (hold (Prod.ext hv (he.1.trans hf.1.symm)))
