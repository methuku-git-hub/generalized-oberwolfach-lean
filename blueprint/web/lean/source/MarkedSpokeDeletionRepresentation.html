   4 /-! A typed spoke subhost of a type-disjoint original host is obtained by
   5 one physical spoke deletion. Its row and column degrees count the deleted
   6 physical pairs exactly. -/

  15 theorem marked_spoke_deletion_representation
  16     (H H':Bool→V→U→Prop)
  17     (hsub:∀ b v u,H' b v u→H b v u)
  18     (htypes:∀ v u,¬(H false v u∧H true v u)):
  19     ∃ D:SimpleGraph (V⊕U),
  20       H'=markedDeletePhysicalSpokes H D ∧
  21       (∀ v,D.degree (Sum.inl v)=(Finset.univ.filter (fun u=>∃ b,H b v u∧¬H' b v u)).card) ∧
  22       (∀ u,D.degree (Sum.inr u)=(Finset.univ.filter (fun v=>∃ b,H b v u∧¬H' b v u)).card) := by
  23   let S:Finset (V×U):=Finset.univ.filter (fun z=>∃ b,H b z.1 z.2∧¬H' b z.1 z.2)
  24   let D:=markedFiniteSpokeGraph S
  25   refine ⟨D,?_,?_,?_⟩
  26   · funext b v u
  27     apply propext
  28     simp only [markedDeletePhysicalSpokes,D,markedFiniteSpokeGraph_adj,S,Finset.mem_filter,Finset.mem_univ,true_and]
  29     have hs0:=hsub false v u
  30     have hs1:=hsub true v u
  31     have ht:=htypes v u
  32     cases b <;> simp only [Bool.exists_bool] <;> tauto
  33   · intro v
  34     change (markedFiniteSpokeGraph S).degree (Sum.inl v)=_
  35     rw [markedFiniteSpokeGraph_degree_row]
  36     congr 1
  37     ext u
  38     simp [S]
  39   · intro u
  40     change (markedFiniteSpokeGraph S).degree (Sum.inr u)=_
  41     rw [markedFiniteSpokeGraph_degree_col]
  42     congr 1
  43     ext v
  44     simp [S]
