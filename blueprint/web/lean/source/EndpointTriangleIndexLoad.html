  18   row_allowed : e∉F.edgeSet
  19   spokes : ∀ v∈e,Q.Adj u v
  20   spoke_allowed : ∀ v∈e,¬F.Adj u v

  22 theorem endpoint_triangle_index_coordinates_equiv (hST : Disjoint S T) (u : V) (hu : u∈T) :
  23     ∃ e : {C : EndpointTriangle Q F S T // u∈C.val}≃{edge : Sym2 V // EndpointIndexEdge Q F S u edge},
  24       ∀ C,endpointTriangleRowEdge C.val=(e C).val := by
  25   have hindex (C : {C : EndpointTriangle Q F S T // u∈C.val}) : endpointTriangleIndex C.val=u := by
  26     have hh : u∈C.val.val∩T:=Finset.mem_inter.mpr ⟨C.property,hu⟩
  27     rw [endpointTriangleIndex_spec] at hh
  28     exact (Finset.mem_singleton.mp hh).symm
  29   have hvalid (C : {C : EndpointTriangle Q F S T // u∈C.val}) :
  30       EndpointIndexEdge Q F S u (endpointTriangleRowEdge C.val) := by
  31     have hr:=endpointTriangleRowEdge_in_graph C.val
  32     refine ⟨fun v hv=>((endpointTriangleRowEdge_mem C.val v).mp hv).2,hr.1,hr.2,?_,?_⟩
  33     · intro v hv
  34       have hh:=(endpointTriangleRowEdge_mem C.val v).mp hv
  35       have hne : u≠v := by rintro rfl;exact Finset.disjoint_left.mp hST hh.2 hu
  36       exact C.val.property.2.2.2.1 u C.property v hh.1 hne
  37     · intro v hv
  38       exact C.val.property.2.2.2.2 u C.property v ((endpointTriangleRowEdge_mem C.val v).mp hv).1
  39   let f : {C : EndpointTriangle Q F S T // u∈C.val}→{edge : Sym2 V // EndpointIndexEdge Q F S u edge}:=
  40     fun C=>⟨endpointTriangleRowEdge C.val,hvalid C⟩
  41   have hf : Function.Bijective f := by
  42     constructor
  43     · intro C D h
  44       apply Subtype.ext
  45       apply endpointTriangle_coordinates_injective
  46       exact Prod.ext ((hindex C).trans (hindex D).symm) (congrArg Subtype.val h)
  47     · rintro ⟨edge,hedge⟩
  48       induction edge using Sym2.ind with
  49       | _ v w =>
  50         have hvS:=hedge.rows_mem v (by simp)
  51         have hwS:=hedge.rows_mem w (by simp)
  52         have hvwQ : Q.Adj v w:=hedge.row_edge
  53         have hvwF : ¬F.Adj v w:=hedge.row_allowed
  54         obtain ⟨C,hC⟩:=endpoint_triangle_from_vertices Q F S T hST u v w hu hvS hwS hvwQ.ne
  55           (hedge.spokes v (by simp)) (hedge.spokes w (by simp)) hvwQ
  56           (hedge.spoke_allowed v (by simp)) (hedge.spoke_allowed w (by simp)) hvwF
  57         change C.val={u,v,w} at hC
  58         let D : {C : EndpointTriangle Q F S T // u∈C.val}:=⟨C,by rw [hC];simp⟩
  59         refine ⟨D,Subtype.ext ?_⟩
  60         exact (endpointTriangle_coordinates_of_vertices hST C u v w hu hvS hwS hC).2
  61   exact ⟨Equiv.ofBijective f hf,fun _=>rfl⟩

  63 theorem endpoint_triangle_index_load (hST : Disjoint S T) (u : V) (hu : u∈T)
  64     (weight : Sym2 V→ℝ) :
  65     (∑ C : EndpointTriangle Q F S T,if u∈C.val then weight (endpointTriangleRowEdge C) else 0)=
  66       ∑ edge : Sym2 V,if EndpointIndexEdge Q F S u edge then weight edge else 0 := by
  67   obtain ⟨e,he⟩:=endpoint_triangle_index_coordinates_equiv hST u hu
  68   have hl : (∑ C : EndpointTriangle Q F S T,if u∈C.val then weight (endpointTriangleRowEdge C) else 0)=
  69       ∑ C : {C : EndpointTriangle Q F S T // u∈C.val},weight (endpointTriangleRowEdge C.val) := by
  70     rw [←Finset.sum_filter]
  71     exact Finset.sum_subtype _ (fun _=>by simp) _
  72   have hr : (∑ edge : Sym2 V,if EndpointIndexEdge Q F S u edge then weight edge else 0)=
  73       ∑ edge : {edge : Sym2 V // EndpointIndexEdge Q F S u edge},weight edge.val := by
  74     rw [←Finset.sum_filter]
  75     exact Finset.sum_subtype _ (fun _=>by simp) _
  76   rw [hl,hr,←e.sum_comp (fun edge=>weight edge.val)]
  77   apply Finset.sum_congr rfl
  78   intro C hC
  79   rw [he C]
