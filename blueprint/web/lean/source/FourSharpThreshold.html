  12 theorem four_barrier_choose_odd (k : ℕ) : Odd ((4 * k + 7).choose 2) := by
  13   have hh : 2 * (4 * k + 7).choose 2 = (4 * k + 7) * (4 * k + 6) := by
  14     rw [Nat.choose_two_right,
  15       Nat.mul_div_cancel' (Nat.even_mul_pred_self (4 * k + 7)).two_dvd]
  16     congr 1
  17   refine ⟨4 * k ^ 2 + 13 * k + 10, ?_⟩
  18   nlinarith

  27 theorem four_threshold_counterexample {d : ℝ} (hd : d < 2 / 3) (N : ℕ) :
  28     ∃ n ≥ N, ∃ G : SimpleGraph (Fin n), CycleDivisible G 4 ∧
  29       (∀ v, d * (n : ℝ) ≤ (G.degree v : ℝ)) ∧ ¬ HasCycleDecomposition G 4 := by
  30   have hm : 0 < 2 - 3 * d := by linarith
  31   let k := max (N + 1) ⌈(4 + |d|) / (2 - 3 * d)⌉₊
  32   let t := 4 * k + 7
  33   have ht : 6 ≤ t := by dsimp only [t]; omega
  34   have hkN : N + 1 ≤ k := le_max_left _ _
  35   have hkt : k ≤ t := by dsimp only [t]; omega
  36   have hceil : (4 + |d|) / (2 - 3 * d) ≤ (t : ℝ) :=
  37     (Nat.le_ceil _).trans (by exact_mod_cast
  38       (show ⌈(4 + |d|) / (2 - 3 * d)⌉₊ ≤ t from (le_max_right _ _).trans hkt))
  39   have hmargin := (div_le_iff₀ hm).mp hceil
  40   refine ⟨(t + t) + (t - 1), by omega, fourBarrierGraph ht, ?_, ?_, ?_⟩
  41   · constructor
  42     · intro v
  43       exact (fourBarrierGraph_degrees ht v).1
  44     · rw [fourBarrierGraph_edge_card]
  45       have heq : t - 1 = 4 * k + 6 := by dsimp only [t]; omega
  46       rw [heq]
  47       exact four_barrier_edge_divisible k
  48   · intro v
  49     have hdeg := (fourBarrierGraph_degrees ht v).2
  50     have hdeg' : (2 : ℝ) * t - 4 ≤ (fourBarrierGraph ht).degree v := by
  51       have hh : ((2 * t - 4 : ℕ) : ℝ) ≤ (fourBarrierGraph ht).degree v := by exact_mod_cast hdeg
  52       simpa only [Nat.cast_sub (by omega : 4 ≤ 2 * t), Nat.cast_mul, Nat.cast_ofNat] using hh
  53     rw [Nat.cast_add, Nat.cast_add, Nat.cast_sub (by omega : 1 ≤ t), Nat.cast_one]
  54     nlinarith [neg_le_abs d]
  55   · exact fourBarrierGraph_no_decomposition ht (four_barrier_choose_odd k)
