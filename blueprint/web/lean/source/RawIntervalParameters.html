  14 def intervalActivation (d : ℕ) : ℝ := 1/(2*(d:ℝ)+1)
  15 def isolatedStartProbability (d : ℕ) : ℝ :=
  16   intervalActivation d*(1-intervalActivation d)^(2*d)

  18 theorem intervalActivation_bounds (d : ℕ) (hd : 2≤d) :
  19     0 < intervalActivation d ∧ intervalActivation d<1 := by
  20   have hd' : (2:ℝ)≤d := by exact_mod_cast hd
  21   unfold intervalActivation
  22   constructor
  23   · positivity
  24   · exact (div_lt_one (by positivity)).mpr (by linarith)

  26 theorem isolatedStartProbability_pos (d : ℕ) (hd : 2≤d) :
  27     0 < isolatedStartProbability d :=
  28   mul_pos (intervalActivation_bounds d hd).1 (pow_pos (sub_pos.mpr (intervalActivation_bounds d hd).2) _)

  30 theorem isolatedStartProbability_scaled_lower (d : ℕ) (hd : 2≤d) :
  31     (2:ℝ)/15<(d:ℝ)*isolatedStartProbability d := by
  32   have hd' : (2:ℝ)≤d := by exact_mod_cast hd
  33   have hd0 : (0:ℝ)<d := by linarith
  34   have he : (1+(1/(2*(d:ℝ))))^(2*d)<3 := by
  35     have hh := (Real.one_add_inv_pow_le_exp (n:=2*d)).trans_lt Real.exp_one_lt_three
  36     simpa only [Nat.cast_mul,Nat.cast_ofNat,one_div] using hh
  37   have hbase : 1-intervalActivation d=(1+1/(2*(d:ℝ)))⁻¹ := by
  38     unfold intervalActivation
  39     field_simp
  40     <;> ring
  41   have hpos : 0<(1+1/(2*(d:ℝ)))^(2*d) := by positivity
  42   have hp : (1:ℝ)/3<(1-intervalActivation d)^(2*d) := by
  43     rw [hbase,inv_pow]
  44     simpa only [one_div] using inv_lt_inv₀ (by norm_num : (0:ℝ)<3) hpos |>.mpr he
  45   have hdfrac : (2:ℝ)/5≤(d:ℝ)/(2*d+1) := by
  46     apply (le_div_iff₀ (by positivity)).mpr
  47     nlinarith only [hd']
  48   have hmul := mul_lt_mul_of_pos_left hp (by positivity : 0<(d:ℝ)/(2*d+1))
  49   have hprod := mul_le_mul_of_nonneg_right hdfrac (by norm_num : (0:ℝ)≤1/3)
  50   have hh := hprod.trans_lt hmul
  51   calc
  52     (2:ℝ)/15 = (2/5)*(1/3) := by norm_num
  53     _ < (d:ℝ)/(2*d+1)*(1-intervalActivation d)^(2*d) := hh
  54     _ = (d:ℝ)*isolatedStartProbability d := by
  55       unfold isolatedStartProbability intervalActivation
  56       ring

  58 /-- A finite subprobability vector with the remaining mass at `none`. -/
  59 def FiniteLaw.withRemainder {Γ : Type*} [Fintype Γ]
  60     (q : Γ → ℝ) (hq : ∀ g,0≤q g) (hQ : ∑ g,q g≤1) : FiniteLaw (Option Γ) where
  61   mass g := match g with | none=>1-∑ g,q g | some g=>q g
  62   nonneg g := by cases g <;> simp only;linarith;exact hq _
  63   total := by rw [Fintype.sum_option];simp

  65 theorem rawInterval_start_probabilities {Γ : Type*} [Fintype Γ]
  66     (d : ℕ) (hd : 2≤d) (q : Γ → ℝ) (hq : ∀ g,0≤q g) (hQ : ∑ g,q g≤1/8) :
  67     (∀ g,0≤q g/((d:ℝ)*isolatedStartProbability d)) ∧
  68       (∑ g,q g/((d:ℝ)*isolatedStartProbability d))≤1 := by
  69   have hd0 : (0:ℝ)<d := by exact_mod_cast (show 0<d by omega)
  70   have hr := isolatedStartProbability_pos d hd
  71   refine ⟨fun g=>div_nonneg (hq g) (mul_pos hd0 hr).le,?_⟩
  72   rw [← Finset.sum_div]
  73   apply (div_le_one (mul_pos hd0 hr)).mpr
  74   have hh := isolatedStartProbability_scaled_lower d hd
  75   linarith
