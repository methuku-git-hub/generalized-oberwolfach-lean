  10 theorem choose_ratio_le_power {r n k : ℕ} (hr : r ≤ n) (hk : k ≤ n) :
  11     (r.choose k : ℝ) / n.choose k ≤ ((r : ℝ) / n) ^ k := by
  12   by_cases hn : n = 0
  13   · subst n
  14     have : k = 0 := by omega
  15     subst k
  16     simp
  17   have hnpos : (0 : ℝ) < n := by exact_mod_cast Nat.pos_of_ne_zero hn
  18   have hfac : (k.factorial : ℝ) ≠ 0 := by exact_mod_cast Nat.factorial_ne_zero k
  19   calc
  20     _ = ((k.factorial : ℝ) * r.choose k) / ((k.factorial : ℝ) * n.choose k) := by
  21       rw [mul_div_mul_left _ _ hfac]
  22     _ = (r.descFactorial k : ℝ) / n.descFactorial k := by
  23       simp only [Nat.descFactorial_eq_factorial_mul_choose, Nat.cast_mul]
  24     _ = ∏ i ∈ Finset.range k, ((r - i : ℕ) : ℝ) / (n - i : ℕ) := by
  25       simp only [Nat.descFactorial_eq_prod_range, Nat.cast_prod, Finset.prod_div_distrib]
  26     _ ≤ ∏ i ∈ Finset.range k, ((r : ℝ) / n) := by
  27       apply Finset.prod_le_prod₀
  28       · intro i _
  29         positivity
  30       · intro i hi
  31         have hin : i < n := lt_of_lt_of_le (Finset.mem_range.mp hi) hk
  32         have hd : (0 : ℝ) < (n - i : ℕ) := by exact_mod_cast Nat.sub_pos_of_lt hin
  33         by_cases hir : i ≤ r
  34         · rw [Nat.cast_sub hir, Nat.cast_sub hin.le]
  35           have hdiff : (0 : ℝ) < (n : ℝ) - i := by
  36             have : (i : ℝ) < n := by exact_mod_cast hin
  37             linarith
  38           apply (div_le_div_iff₀ hdiff hnpos).mpr
  39           have hri : (r : ℝ) ≤ n := by exact_mod_cast hr
  40           nlinarith [show (0 : ℝ) ≤ i by positivity]
  41         · have : r - i = 0 := Nat.sub_eq_zero_of_le (by omega)
  42           rw [this]
  43           simp only [Nat.cast_zero, zero_div]
  44           exact div_nonneg (by positivity) hnpos.le
  45     _ = _ := by simp [div_pow]
