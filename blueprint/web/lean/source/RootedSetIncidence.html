  16 theorem fixedSetFamily_root_count (S : I → FixedSubset V k) (hinj : Function.Injective S)
  17     (R : Finset V) (hR : R.card≤k) :
  18     (∑ i,if R⊆(S i).val then (1:ℝ) else 0)≤(Fintype.card V:ℝ)^(k-R.card) := by
  19   let A := {i : I // R⊆(S i).val}
  20   let B := {T : FixedSubset V k // R⊆T.val}
  21   let f (i : A) : B := ⟨S i.val,i.property⟩
  22   have hf : Function.Injective f := by
  23     intro i j hij
  24     apply Subtype.ext
  25     exact hinj (congrArg Subtype.val hij)
  26   have hc : Fintype.card A≤Fintype.card V^(k-R.card) := by
  27     calc
  28       _ ≤ Fintype.card B := Fintype.card_le_of_injective f hf
  29       _ = (Fintype.card V-R.card).choose (k-R.card) := by
  30         simpa only [← Nat.card_eq_fintype_card] using fixedSubset_containing_card R hR
  31       _ ≤ (Fintype.card V-R.card)^(k-R.card) := Nat.choose_le_pow _ _
  32       _ ≤ _ := Nat.pow_le_pow_left (Nat.sub_le _ _) _
  33   have he : (∑ i,if R⊆(S i).val then (1:ℝ) else 0)=(Fintype.card A:ℝ) := by
  34     simp only [← Finset.sum_filter,Finset.sum_const,nsmul_eq_mul,mul_one,Fintype.card_subtype,A]
  35   rw [he]
  36   exact_mod_cast hc

  39 theorem fixedSetFamily_total (S : I → FixedSubset V k) (hinj : Function.Injective S)
  40     (x y : V) (hxy : x≠y) (hk : 2≤k)
  41     (hroot : ∀ i,x∈(S i).val ∧ y∈(S i).val) :
  42     (Fintype.card I:ℝ)≤(Fintype.card V:ℝ)^(k-2) := by
  43   have hc : ({x,y}:Finset V).card=2 := by simp [hxy]
  44   have hh := fixedSetFamily_root_count S hinj {x,y} (by simpa only [hc] using hk)
  45   have he (i : I) : ({x,y}:Finset V)⊆(S i).val := by
  46     simpa only [Finset.insert_subset_iff,Finset.singleton_subset_iff] using hroot i
  47   simpa only [he,ite_true,Finset.sum_const,nsmul_eq_mul,mul_one,Finset.card_univ,hc] using hh

  50 theorem fixedSetFamily_distinct_edge_count (S : I → FixedSubset V k) (hinj : Function.Injective S)
  51     (x y : V) (hxy : x≠y) (hk : 4≤k) (hn : 0<Fintype.card V)
  52     (hroot : ∀ i,x∈(S i).val ∧ y∈(S i).val)
  53     (z t : V) (hzt : z≠t) (he : s(x,y)≠s(z,t)) :
  54     (∑ i,if z∈(S i).val ∧ t∈(S i).val then (1:ℝ) else 0)≤
  55       (Fintype.card V:ℝ)^(k-3) := by
  56   have hc := distinct_pair_union_card x y z t hxy hzt he
  57   have hh := fixedSetFamily_root_count S hinj {x,y,z,t} (hc.2.trans hk)
  58   have hpred (i : I) : (({x,y,z,t}:Finset V)⊆(S i).val)↔(z∈(S i).val ∧ t∈(S i).val) := by
  59     simp only [Finset.insert_subset_iff,Finset.singleton_subset_iff,(hroot i).1,(hroot i).2,true_and]
  60   simp only [hpred] at hh
  61   exact hh.trans (pow_le_pow_right₀ (by exact_mod_cast hn : (1:ℝ)≤Fintype.card V)
  62     (Nat.sub_le_sub_left hc.1 k))
