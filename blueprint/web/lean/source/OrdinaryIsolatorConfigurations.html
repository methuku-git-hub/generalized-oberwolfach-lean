  15 structure OrdinaryIsolatorConfiguration {V U : Type*} [Fintype V] [DecidableEq V]
  16     (P Q F : SimpleGraph V) (H : V → U → Prop) (k : ℕ) (θ α : ℝ) (x y : V) where
  17   hub : U
  18   vertices : FixedSubset V k
  19   endpoints : x∈vertices.val ∧ y∈vertices.val
  20   rim_edges : spanningRestriction P vertices.val≤Q
  21   spokes : ∀ z∈vertices.val,H z hub
  22   independent : ∀ z∈vertices.val,∀ w∈vertices.val,¬F.Adj z w
  23   density : ∀ z∈vertices.val,(θ+α)*k≤((P.neighborFinset z∩vertices.val).card:ℝ)

  42 theorem ordinary_isolator_configuration_weights (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  43     ∃ N : ℕ,∀ (V U : Type) [Fintype V] [DecidableEq V] [Fintype U] [DecidableEq U],
  44       ∀ (P Q F : SimpleGraph V) (H : V → U → Prop) (k : ℕ),N≤k →
  45       ∀ θ : ℝ,cycleThreshold r≤θ → ∀ x y,P.Adj x y →
  46       ∀ T : OrdinaryIsolatorConfiguration P Q F H k θ α x y,
  47       ∃ ψ : OrdinaryWheel Q F H r → ℝ,
  48         (∀ D,|ψ D|≤1) ∧
  49         (∀ D,ψ D≠0 → ordinaryWheelHub D=T.hub ∧ (ordinaryWheelRim D).support⊆(T.vertices.val:Set V)) ∧
  50         (∀ a b,ordinaryRimLoad ψ a b=if (SimpleGraph.edge x y).Adj a b then 1 else 0) ∧
  51         (∀ z h,ordinarySpokeLoad ψ z h=
  52           (if h=T.hub then (1:ℝ) else 0)*(1/2)*(if z=x ∨ z=y then 1 else 0)) := by
  53   obtain ⟨N,hN⟩ := signed_cycle_edge_isolator_on_subset r hr α hα
  54   refine ⟨N,?_⟩
  55   intro V U _ _ _ _ P Q F H k hk θ hθ x y hxy T
  56   obtain ⟨I,hs,hb,he,hv⟩ := hN V P T.vertices.val (by simpa only [T.vertices.property] using hk) (by
  57     intro z hz
  58     have hh := T.density z hz
  59     rw [T.vertices.property]
  60     nlinarith [mul_le_mul_of_nonneg_right hθ (Nat.cast_nonneg (α:=ℝ) k)])
  61     x y T.endpoints.1 T.endpoints.2 hxy
  62   exact signed_isolator_wheel_lift P Q F H r T.vertices.val T.hub
  63     T.rim_edges T.spokes T.independent x y I hs hb he hv
