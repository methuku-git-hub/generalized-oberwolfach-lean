  19 def disjointWheelSpokeEquiv (z : W) :
  20     (I×W) ≃ {e:(I⊕(I×W))×(Empty⊕I) // disjointWheelSpokes z e.1 e.2} := by
  21   let f : (I×W) → {e:(I⊕(I×W))×(Empty⊕I) // disjointWheelSpokes z e.1 e.2} := fun p =>
  22     ⟨(attachmentWheelEmbedding z id p.1 p.2,.inr p.1),by
  23       by_cases h : p.2=z <;> simp [attachmentWheelEmbedding,disjointWheelSpokes,ordinaryAttachmentSpokes,h]⟩
  24   apply Equiv.ofBijective f
  25   constructor
  26   · rintro ⟨i,w⟩ ⟨j,w'⟩ he
  27     have hij : i=j := Sum.inr.inj (congrArg (fun a => a.val.2) he)
  28     subst j
  29     have hww : w=w' := (attachmentWheelEmbedding z id i).injective (congrArg (fun a => a.val.1) he)
  30     exact Prod.ext rfl hww
  31   · rintro ⟨⟨a,b⟩,h⟩
  32     cases b with
  33     | inl b => exact b.elim
  34     | inr i =>
  35       cases a with
  36       | inl a =>
  37         change i=a at h
  38         subst a
  39         refine ⟨(i,z),?_⟩
  40         apply Subtype.ext
  41         simp [f,attachmentWheelEmbedding]
  42       | inr p =>
  43         rcases p with ⟨j,w⟩
  44         change w≠z ∧ j=i at h
  45         obtain ⟨hw,hji⟩ := h
  46         subst j
  47         refine ⟨(i,w),?_⟩
  48         apply Subtype.ext
  49         simp [f,attachmentWheelEmbedding,hw]

  55 theorem disjoint_wheel_comparison (C : SimpleGraph W) (r : ℕ) (hC : IsEmbeddedCycle C r)
  56     (z : W) (hfull : ∀ w,w∈C.support) (hW : Fintype.card W=r) :
  57     let G := ordinaryAttachmentRim (⊥:SimpleGraph I) C z id
  58     let H := disjointWheelSpokes (I:=I) z
  59     HasOrdinaryWheelDecomposition G H r ∧
  60     (∀ a,G.degree a=2*(Finset.univ.filter (H a)).card) ∧
  61     (∀ b,r∣(Finset.univ.filter (fun a => H a b)).card) ∧
  62     G.edgeFinset.card=Fintype.card I*r ∧
  63     Fintype.card {e:(I⊕(I×W))×(Empty⊕I) // H e.1 e.2}=Fintype.card I*r := by
  64   dsimp only
  65   have hdec := ordinaryAttachment_extra_wheel_decomposition (V:=I) (U:=Empty) C hC z id hfull
  66   have hbal := ordinary_wheel_decomposition_balance _ _ r hdec
  67   have hcard : Fintype.card {e:(I⊕(I×W))×(Empty⊕I) // disjointWheelSpokes z e.1 e.2}=Fintype.card I*r := by
  68     rw [disjointWheelSpokes_card,hW]
  69   exact ⟨hdec,hbal.1,hbal.2,(ordinary_balanced_total_edges _ _ hbal.1).trans hcard,hcard⟩
