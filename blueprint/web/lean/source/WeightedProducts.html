  11 noncomputable def patternValue {E : Type*} {State : E → Type*}
  12     (F : Finset E) (s ρ : ∀ e, State e) : ℝ := by
  13   classical
  14   exact ∏ e ∈ F, if ρ e = s e then (1 : ℝ) else 0

  28 theorem pattern_change_bound {E : Type*} [DecidableEq E] {State : E → Type*}
  29     (F D : Finset E) (s σ τ : ∀ e, State e)
  30     (h : ∀ e ∈ F, e ∉ D → σ e = τ e) :
  31     |patternValue F s τ - patternValue F s σ| ≤
  32       ∑ e ∈ D, if e ∈ F then (1 : ℝ) else 0 := by
  33   classical
  34   by_cases hhit : ∃ e ∈ D, e ∈ F
  35   · obtain ⟨e, heD, heF⟩ := hhit
  36     have hlo := (patternValue_mem_unit F s τ).1
  37     have hhi := (patternValue_mem_unit F s τ).2
  38     have hlo' := (patternValue_mem_unit F s σ).1
  39     have hhi' := (patternValue_mem_unit F s σ).2
  40     have hv : |patternValue F s τ - patternValue F s σ| ≤ 1 :=
  41       abs_le.mpr ⟨by linarith, by linarith⟩
  42     have hs : (1 : ℝ) ≤ ∑ e ∈ D, if e ∈ F then (1 : ℝ) else 0 := by
  43       have ht := Finset.single_le_sum
  44         (s := D) (f := fun e => if e ∈ F then (1 : ℝ) else 0)
  45         (fun e _ => by split_ifs <;> norm_num) heD
  46       simpa [heF] using ht
  47     exact hv.trans hs
  48   · have heq : patternValue F s σ = patternValue F s τ := by
  49       unfold patternValue
  50       apply Finset.prod_congr rfl
  51       intro e he
  52       rw [h e he (fun heD => hhit ⟨e, heD, he⟩)]
  53     rw [heq, sub_self, abs_zero]

  56 /-- Each occurrence contributes once for each candidate using it. Repeated
  57 physical images are not identified by this inequality. -/
  58 theorem weighted_product_stability {E I : Type*} [DecidableEq E]
  59     {State : E → Type*} (C : Finset I) (F : I → Finset E)
  60     (s : I → ∀ e, State e) (w : I → ℝ) (D : Finset E) (σ τ : ∀ e, State e)
  61     (hw : ∀ i ∈ C, 0 ≤ w i)
  62     (h : ∀ i ∈ C, ∀ e ∈ F i, e ∉ D → σ e = τ e) :
  63     |(∑ i ∈ C, w i * patternValue (F i) (s i) τ) -
  64       ∑ i ∈ C, w i * patternValue (F i) (s i) σ| ≤
  65     ∑ e ∈ D, ∑ i ∈ C, if e ∈ F i then w i else 0 := by
  66   classical
  67   rw [← Finset.sum_sub_distrib]
  68   calc
  69     _ ≤ ∑ i ∈ C, |w i * patternValue (F i) (s i) τ -
  70         w i * patternValue (F i) (s i) σ| := Finset.abs_sum_le_sum_abs _ _
  71     _ = ∑ i ∈ C, w i * |patternValue (F i) (s i) τ -
  72         patternValue (F i) (s i) σ| := by
  73       apply Finset.sum_congr rfl
  74       intro i hi
  75       rw [← mul_sub, abs_mul, abs_of_nonneg (hw i hi)]
  76     _ ≤ ∑ i ∈ C, w i * (∑ e ∈ D, if e ∈ F i then (1 : ℝ) else 0) := by
  77       apply Finset.sum_le_sum
  78       intro i hi
  79       exact mul_le_mul_of_nonneg_left (pattern_change_bound (F i) D (s i) σ τ (h i hi))
  80         (hw i hi)
  81     _ = ∑ e ∈ D, ∑ i ∈ C, if e ∈ F i then w i else 0 := by
  82       simp_rw [Finset.mul_sum]
  83       rw [Finset.sum_comm]
  84       apply Finset.sum_congr rfl
  85       intro e he
  86       apply Finset.sum_congr rfl
  87       intro i hi
  88       split_ifs <;> simp_all

  90 /-- Root occurrences are removed from each product. Their states may differ

  92 theorem omitted_root_stability {E I : Type*} [DecidableEq E]
  93     {State : E → Type*} (C : Finset I) (F : I → Finset E)
  94     (R D : Finset E) (s : I → ∀ e, State e) (w : I → ℝ)
  95     (σ τ : ∀ e, State e) (hw : ∀ i ∈ C, 0 ≤ w i)
  96     (h : ∀ i ∈ C, ∀ e ∈ F i, e ∉ R → e ∉ D → σ e = τ e) :
  97     |(∑ i ∈ C, w i * patternValue (F i \ R) (s i) τ) -
  98       ∑ i ∈ C, w i * patternValue (F i \ R) (s i) σ| ≤
  99     ∑ e ∈ D, ∑ i ∈ C, if e ∈ F i \ R then w i else 0 := by
