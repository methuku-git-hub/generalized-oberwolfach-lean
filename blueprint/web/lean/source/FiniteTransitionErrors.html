  10 def transitionApply {X Y : Type*} [Fintype X]
  11     (μ : X → ℝ) (R : X → Y → ℝ) (y : Y) : ℝ := ∑ x, μ x * R x y

  13 def transitionCompose {X Y Z : Type*} [Fintype Y]
  14     (R : X → Y → ℝ) (Q : Y → Z → ℝ) (x : X) (z : Z) : ℝ :=
  15   ∑ y, R x y * Q y z

  17 def IsTransition {X Y : Type*} [Fintype Y] (R : X → Y → ℝ) : Prop :=
  18   (∀ x y, 0 ≤ R x y) ∧ ∀ x, ∑ y, R x y = 1

  20 def RelativeError {X : Type*} (μ π : X → ℝ) (h : ℝ) : Prop :=
  21   ∀ x, |μ x - π x| ≤ h * π x

  23 theorem transitionApply_sub {X Y : Type*} [Fintype X]
  24     (μ π : X → ℝ) (R : X → Y → ℝ) (y : Y) :
  25     transitionApply μ R y - transitionApply π R y =
  26       ∑ x, (μ x - π x) * R x y := by
  27   simp only [transitionApply, sub_mul, Finset.sum_sub_distrib]

  29 theorem relative_error_preserved {X Y : Type*} [Fintype X]
  30     (μ π : X → ℝ) (R : X → Y → ℝ) (h : ℝ)
  31     (hR : ∀ x y, 0 ≤ R x y) (herror : RelativeError μ π h) :
  32     RelativeError (transitionApply μ R) (transitionApply π R) h := by
  33   intro y
  34   rw [transitionApply_sub]
  35   calc
  36     |∑ x, (μ x-π x)*R x y| ≤ ∑ x, |(μ x-π x)*R x y| := Finset.abs_sum_le_sum_abs _ _
  37     _ = ∑ x, |μ x-π x| *R x y := by simp_rw [abs_mul, abs_of_nonneg (hR _ _)]
  38     _ ≤ ∑ x, (h*π x)*R x y := Finset.sum_le_sum fun x _ =>
  39       mul_le_mul_of_nonneg_right (herror x) (hR x y)
  40     _ = h * transitionApply π R y := by simp [transitionApply, Finset.mul_sum, mul_assoc]

  42 theorem relative_error_contracted {X Y : Type*} [Fintype X]
  43     (μ π : X → ℝ) (R : X → Y → ℝ) (κ h : ℝ)
  44     (hμ : ∑ x, μ x = 1) (hπ : ∑ x, π x = 1)
  45     (hminor : ∀ x y, κ * transitionApply π R y ≤ R x y)
  46     (herror : RelativeError μ π h) :
  47     RelativeError (transitionApply μ R) (transitionApply π R) ((1-κ)*h) := by
  48   intro y
  49   let ν := transitionApply π R
  50   have hzero : ∑ x, (μ x-π x) = 0 := by rw [Finset.sum_sub_distrib,hμ,hπ,sub_self]
  51   have hrewrite : transitionApply μ R y - ν y =
  52       ∑ x, (μ x-π x)*(R x y-κ*ν y) := by
  53     rw [transitionApply_sub]
  54     simp only [mul_sub, Finset.sum_sub_distrib, ← Finset.sum_mul, hzero, zero_mul, sub_zero]
  55   have hmass : (∑ x, π x*(R x y-κ*ν y)) = (1-κ)*ν y := by
  56     simp only [mul_sub, Finset.sum_sub_distrib, ← Finset.sum_mul, hπ, one_mul]
  57     change ν y-κ*ν y=(1-κ)*ν y
  58     ring
  59   change |transitionApply μ R y-ν y| ≤ ((1-κ)*h)*ν y
  60   rw [hrewrite]
  61   calc
  62     |∑ x, (μ x-π x)*(R x y-κ*ν y)| ≤ ∑ x, |(μ x-π x)*(R x y-κ*ν y)| :=
  63       Finset.abs_sum_le_sum_abs _ _
  64     _ = ∑ x, |μ x-π x| *(R x y-κ*ν y) := by
  65       apply Finset.sum_congr rfl
  66       intro x _
  67       rw [abs_mul, abs_of_nonneg (sub_nonneg.mpr (hminor x y))]
  68     _ ≤ ∑ x, (h*π x)*(R x y-κ*ν y) := Finset.sum_le_sum fun x _ =>
  69       mul_le_mul_of_nonneg_right (herror x) (sub_nonneg.mpr (hminor x y))
  70     _ = h * ((1-κ)*ν y) := by simp_rw [mul_assoc]; rw [← Finset.mul_sum,hmass]
  71     _ = ((1-κ)*h)*ν y := by ring

  73 theorem transition_initial_relative_error {X : Type*}
  74     (μ π : X → ℝ) (B : ℝ) (hB : 0 ≤ B)
  75     (hμ : ∀ x, 0 ≤ μ x) (hπ : ∀ x, 0 ≤ π x) (hupper : ∀ x, μ x ≤ B*π x) :
  76     RelativeError μ π (B+1) := by
  77   intro x
  78   rw [abs_le]
  79   constructor <;> nlinarith [hμ x,hπ x,hupper x,mul_nonneg hB (hπ x)]

  81 theorem transitionApply_total {X Y : Type*} [Fintype X] [Fintype Y]
  82     (μ : X → ℝ) (R : X → Y → ℝ) (hR : IsTransition R) :
  83     (∑ y, transitionApply μ R y) = ∑ x, μ x := by
  84   unfold transitionApply
  85   rw [Finset.sum_comm]
  86   simp_rw [← Finset.mul_sum,hR.2,mul_one]

  88 theorem transitionApply_compose {X Y Z : Type*} [Fintype X] [Fintype Y]
  89     (μ : X → ℝ) (R : X → Y → ℝ) (Q : Y → Z → ℝ) :
  90     transitionApply (transitionApply μ R) Q = transitionApply μ (transitionCompose R Q) := by
  91   funext z
  92   simp only [transitionApply,transitionCompose,Finset.sum_mul,Finset.mul_sum,mul_assoc]
  93   exact Finset.sum_comm

  95 theorem transitionCompose_isTransition {X Y Z : Type*} [Fintype Y] [Fintype Z]
  96     (R : X → Y → ℝ) (Q : Y → Z → ℝ) (hR : IsTransition R) (hQ : IsTransition Q) :
  97     IsTransition (transitionCompose R Q) := by
  98   constructor
  99   · intro x z
 100     exact Finset.sum_nonneg fun y _ => mul_nonneg (hR.1 x y) (hQ.1 y z)
 101   · intro x
 102     exact (transitionApply_total (R x) Q hQ).trans (hR.2 x)

 104 theorem transitionCompose_assoc {W X Y Z : Type*} [Fintype X] [Fintype Y]
 105     (P : W → X → ℝ) (R : X → Y → ℝ) (Q : Y → Z → ℝ) :
 106     transitionCompose (transitionCompose P R) Q = transitionCompose P (transitionCompose R Q) := by
 107   funext w z
 108   exact congrFun (transitionApply_compose (P w) R Q) z

 110 theorem relative_error_averaged_rows {X Y : Type*} [Fintype X]
 111     (μ : X → ℝ) (R : X → Y → ℝ) (π : Y → ℝ) (h : ℝ)
 112     (hμ : ∀ x, 0 ≤ μ x) (hs : ∑ x, μ x = 1)
 113     (hrow : ∀ x, RelativeError (R x) π h) : RelativeError (transitionApply μ R) π h := by
 114   intro y
 115   have heq : transitionApply μ R y-π y = ∑ x, μ x*(R x y-π y) := by
 116     simp [transitionApply, mul_sub, Finset.sum_sub_distrib, ← Finset.sum_mul, hs]
 117   rw [heq]
 118   calc
 119     |∑ x, μ x*(R x y-π y)| ≤ ∑ x, |μ x*(R x y-π y)| := Finset.abs_sum_le_sum_abs _ _
 120     _ = ∑ x, μ x*|R x y-π y| := by simp_rw [abs_mul, abs_of_nonneg (hμ _)]
 121     _ ≤ ∑ x, μ x*(h*π y) := Finset.sum_le_sum fun x _ =>
 122       mul_le_mul_of_nonneg_left (hrow x y) (hμ x)
 123     _ = h*π y := by rw [← Finset.sum_mul,hs,one_mul]
