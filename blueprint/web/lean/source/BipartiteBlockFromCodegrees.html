  14 theorem bipartite_block_bounds_of_codegrees
  15     (E : X → Y → Prop) (P : X → Prop) (Q : Y → Prop)
  16     (F : SimpleGraph X) (b : ℕ) (p err eta d tau lam : ℝ)
  17     (hp : 0<p) (hp1 : p≤1) (herr : 0≤err) (heta : 0<eta)
  18     (hQ : 0<(Finset.univ.filter Q).card)
  19     (hF : ∀ x,F.degree x≤b)
  20     (hrow : ∀ x,P x → (p-err)*(Finset.univ.filter Q).card≤
  21       ((Finset.univ.filter (fun y=>Q y ∧ E x y)).card:ℝ))
  22     (hcol : ∀ y,Q y → (p-err)*(Finset.univ.filter P).card≤
  23       ((Finset.univ.filter (fun x=>P x ∧ E x y)).card:ℝ))
  24     (hcodeg : ∀ x,P x → ∀ z,P z → x≠z → ¬F.Adj x z →
  25       ((Finset.univ.filter (fun y=>Q y ∧ E x y ∧ E z y)).card:ℝ)≤
  26         (p^2+err)*(Finset.univ.filter Q).card)
  27     (hsmall : err≤p^2*eta/24)
  28     (hlarge : (b+1:ℝ)≤p^2*eta^2*(Finset.univ.filter Q).card/8)
  29     (hdQ : d≤(p-err)*(Finset.univ.filter Q).card)
  30     (hdP : d≤(p-err)*(Finset.univ.filter P).card)
  31     (htau : eta*(Finset.univ.filter Q).card≤tau) (hlam : lam≤p/2) :
  32     BipartiteBlockBounds E P Q d tau lam := by
  33   refine ⟨fun x hx=>hdQ.trans (hrow x hx),fun y hy=>hdP.trans (hcol y hy),?_⟩
  34   intro S T hS hT hSt hTt
  35   have hh := degree_codegree_cuts_on_domains E F (Finset.univ.filter P) (Finset.univ.filter Q)
  36     p err eta b hp hp1 herr heta hQ hF
  37     (by
  38       intro x hx
  39       simpa only [Finset.filter_filter] using hrow x (Finset.mem_filter.mp hx).2)
  40     (by
  41       intro x hx z hz hne hbad
  42       simpa only [Finset.filter_filter] using hcodeg x (Finset.mem_filter.mp hx).2 z
  43         (Finset.mem_filter.mp hz).2 hne hbad)
  44     hsmall hlarge S (by intro x hx;exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,hS x hx⟩)
  45     T (by intro y hy;exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,hT y hy⟩)
  46     (htau.trans hSt) (htau.trans hTt)
  47   exact (mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_right hlam (Nat.cast_nonneg _))
  48     (Nat.cast_nonneg _)).trans hh
