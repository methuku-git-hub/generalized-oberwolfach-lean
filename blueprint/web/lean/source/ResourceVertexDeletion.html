  18 theorem resource_vertex_deletion_degree (R : C → Finset V) (w : C → ℝ) (hw : ∀ C₀,0 ≤ w C₀)
  19     (K : ℝ) (hpair : ∀ u v,u≠v → (∑ C₀,if u∈R C₀ ∧ v∈R C₀ then w C₀ else 0) ≤ K)
  20     (Z : Finset V) (x : V) (hx : x∉Z) :
  21     0 ≤ hypergraphWeightedDegree R w x-hypergraphWeightedDegree R (vertexRestrictedWeight R w Z) x ∧
  22     hypergraphWeightedDegree R w x-hypergraphWeightedDegree R (vertexRestrictedWeight R w Z) x ≤ Z.card*K := by
  23   have hhit (C₀ : C) : ¬Disjoint (R C₀) Z ↔ ∃ z : Z,z.val∈R C₀ := by
  24     constructor
  25     · intro hd
  26       by_contra hn
  27       apply hd
  28       exact Finset.disjoint_left.mpr (fun v hv hz => hn ⟨⟨v,hz⟩,hv⟩)
  29     · rintro ⟨z,hz⟩ hd
  30       exact Finset.disjoint_left.mp hd hz z.property
  31   have he : hypergraphWeightedDegree R w x-hypergraphWeightedDegree R (vertexRestrictedWeight R w Z) x=
  32       ∑ C₀,if x∈R C₀ ∧ ∃ z : Z,z.val∈R C₀ then w C₀ else 0 := by
  33     unfold hypergraphWeightedDegree
  34     rw [← Finset.sum_sub_distrib]
  35     apply Finset.sum_congr rfl
  36     intro C₀ _
  37     by_cases h : x∈R C₀ <;> by_cases hd : Disjoint (R C₀) Z
  38     · have hh : ¬∃ z : Z,z.val∈R C₀ := fun hz => (hhit C₀).mpr hz hd
  39       simp [vertexRestrictedWeight,h,hd,hh]
  40     · have hh := (hhit C₀).mp hd
  41       simp [vertexRestrictedWeight,h,hd,hh]
  42     all_goals simp [vertexRestrictedWeight,h]
  43   rw [he]
  44   constructor
  45   · exact Finset.sum_nonneg (fun C₀ _ => by split_ifs <;> first | exact hw C₀ | exact le_refl _)
  46   · have hh := finite_weighted_union_bound w hw (fun C₀ => x∈R C₀) (fun C₀ (z : Z) => z.val∈R C₀)
  47     calc
  48       _ ≤ ∑ z : Z,∑ C₀,if x∈R C₀ ∧ z.val∈R C₀ then w C₀ else 0 := by
  49         convert hh using 1 <;> congr!
  50       _ ≤ ∑ _z : Z,K := Finset.sum_le_sum (fun z _ => hpair x z.val (by intro he;exact hx (he ▸ z.property)))
  51       _ = _ := by simp
