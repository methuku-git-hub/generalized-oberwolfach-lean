  15 theorem marked_index_tuple_estimate
  16     (π : Equiv.Perm V) (d h : ℕ) (B : Finset U)
  17     (q : U→Γ→ℝ) (s : U→Color→ℝ) (g : Γ) (j : Color) (err : ℝ)
  18     (σ : RawMarkedOutcome V U Γ Color) (H : Bool→V→U→Prop)
  19     (hindices : MarkedIndicesWithSpokes π d h B q s g j err σ H)
  20     (w : Bool→ℝ) (hq : ∀ u∈B,q u g=w true) (hs : ∀ u∈B,s u j=w false)
  21     (x : I→V) (t : I→Bool) (hx : Function.Injective x) (hsize : Fintype.card I≤h)
  22     (hsep : ∀ i k,i≠k → ¬PermutationClose π (3*d) (x i) (x k)) :
  23     |((B.filter (fun u=>∀ i,H (t i) (x i) u)).card:ℝ)-
  24       (∏ i,w (t i))*B.card|<err := by
  25   let f : I→Bool×V := fun i=>(t i,x i)
  26   have hf : Function.Injective f := by
  27     intro i k he; exact hx (congrArg Prod.snd he)
  28   let J := Finset.univ.image f
  29   have hJ : J.card≤h := (Finset.card_image_le).trans (by simpa using hsize)
  30   have hse : ∀ a∈J,∀ b∈J,a≠b → ¬PermutationClose π (3*d) a.2 b.2 := by
  31     intro a ha b hb hne
  32     obtain ⟨i,hi,rfl⟩ := Finset.mem_image.mp ha
  33     obtain ⟨k,hk,rfl⟩ := Finset.mem_image.mp hb
  34     exact hsep i k (fun he=>hne (congrArg f he))
  35   have he := hindices J hJ hse
  36   have hcount : markedIndexCountWithSpokes H B J=
  37       ((B.filter (fun u=>∀ i,H (t i) (x i) u)).card:ℝ) := by
  38     simp only [markedIndexCountWithSpokes,J,Finset.forall_mem_image,Finset.mem_univ,
  39       forall_const,f,Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  40   have htarget : (∑ u∈B,∏ a∈J,if a.1 then q u g else s u j)=(∏ i,w (t i))*B.card := by
  41     calc
  42       _ = ∑ _u∈B,∏ i,w (t i) := by
  43         apply Finset.sum_congr rfl
  44         intro u hu
  45         dsimp only [J]
  46         rw [Finset.prod_image hf.injOn]
  47         apply Finset.prod_congr rfl
  48         intro i hi
  49         dsimp only [f]
  50         by_cases ht : t i=true
  51         · simp [ht,hq u hu]
  52         · have ht' : t i=false := Bool.eq_false_iff.mpr ht
  53           simp [ht',hs u hu]
  54       _ = _ := by simp [mul_comm]
  55   rwa [hcount,htarget] at he
