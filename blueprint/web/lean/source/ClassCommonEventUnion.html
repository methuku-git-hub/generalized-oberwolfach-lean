  13 theorem class_common_event_union {I Ω C:Type} [Fintype I] [Fintype Ω] [Fintype C]
  14     {A:I→Type} [∀ i,Fintype (A i)]
  15     (global:FiniteLaw Ω) (colour:Ω→C) (colourLaw:FiniteLaw C)
  16     (view:∀ i,Ω→A i) (classLaw:∀ i,FiniteLaw (A i))
  17     (hcolour:global.map colour=colourLaw) (hview:∀ i,global.map (view i)=classLaw i)
  18     (Partition:C→Prop) (Good:∀ i,A i→Prop) (errPart:ℝ) (err:I→ℝ)
  19     (hpart:1-errPart≤colourLaw.probability Partition)
  20     (hgood:∀ i,1-err i≤(classLaw i).probability (Good i))
  21     (hbudget:errPart+∑ i,err i<1):
  22     ∃ ω,0<global.mass ω ∧ Partition (colour ω) ∧ ∀ i,Good i (view i ω):=by
  23   let P:=fun ω=>Partition (colour ω)
  24   let E:=fun i ω=>Good i (view i ω)
  25   have hp:1-errPart≤global.probability P:=by
  26     have hh:=finiteLaw_map_probability global colour Partition
  27     rw [hcolour] at hh
  28     exact hpart.trans_eq hh
  29   have hg (i:I):1-err i≤global.probability (E i):=by
  30     have hh:=finiteLaw_map_probability global (view i) (Good i)
  31     rw [hview i] at hh
  32     exact (hgood i).trans_eq hh
  33   let Bad:=fun ω=>∃ i,¬E i ω
  34   have hbad:global.probability Bad≤∑ i,err i:=by
  35     have hu:=finiteLaw_union_bound global (Finset.univ:Finset I) (fun i ω=>¬E i ω)
  36     calc
  37       _ ≤ ∑ i,global.probability (fun ω=>¬E i ω):=by simpa [Bad] using hu
  38       _ ≤ ∑ i,err i:=by
  39         apply Finset.sum_le_sum
  40         intro i _
  41         have hc:=finiteLaw_probability_compl global (E i)
  42         linarith [hg i]
  43   have hall:1-(∑ i,err i)≤global.probability (fun ω=>∀ i,E i ω):=by
  44     have hc:=finiteLaw_probability_compl global (fun ω=>∀ i,E i ω)
  45     have he:(fun ω=>¬∀ i,E i ω)=Bad:=by
  46       funext ω
  47       apply propext
  48       simp [Bad]
  49     rw [he] at hc
  50     linarith
  51   have hjoin:=finiteLaw_good_intersection global P (fun ω=>∀ i,E i ω) (∑ i,err i) hall
  52   have hpositive:0<global.probability (fun ω=>P ω ∧ ∀ i,E i ω):=by linarith
  53   unfold FiniteLaw.probability at hpositive
  54   have hnonneg:∀ ω∈(Finset.univ:Finset Ω),0≤(if P ω ∧ ∀ i,E i ω then global.mass ω else 0):=by
  55     intro ω _
  56     split_ifs
  57     · exact global.nonneg ω
  58     · norm_num
  59   have hpositive':0<∑ ω:Ω,if P ω ∧ ∀ i,E i ω then global.mass ω else 0:=by
  60     convert hpositive using 1
  61     apply Finset.sum_congr rfl
  62     intro ω _
  63     by_cases h:P ω ∧ ∀ i,E i ω <;> simp [h]
  64   obtain ⟨ω,_,hω⟩:=(Finset.sum_pos_iff_of_nonneg hnonneg).mp hpositive'
  65   by_cases h:P ω ∧ ∀ i,E i ω
  66   · exact ⟨ω,by simpa [h] using hω,h.1,h.2⟩
  67   · simp [h] at hω
