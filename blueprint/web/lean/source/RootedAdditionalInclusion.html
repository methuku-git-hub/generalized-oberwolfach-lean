  15 theorem rootedSubset_additional_inclusion {V : Type*} [Fintype V] [DecidableEq V]
  16     (T R : Finset V) (k : ℕ) (hk : T.card+k≤Fintype.card V)
  17     [Nonempty (RootedSubset V T k)] :
  18     (FiniteLaw.uniform (RootedSubset V T k)).probability (fun S=>R⊆S.val.val)≤
  19       ((k:ℝ)/(Fintype.card V-T.card:ℕ))^(R\T).card := by
  20   have hn : Fintype.card {x : V // x∉T}=Fintype.card V-T.card := by
  21     rw [Fintype.card_subtype_compl]
  22     simp
  23   have hkn : k≤Fintype.card {x : V // x∉T} := by rw [hn];omega
  24   letI : Nonempty (FixedSubset {x : V // x∉T} k) := fixedSubset_nonempty hkn
  25   let A := R.subtype (fun x=>x∉T)
  26   have hA : A.card=(R\T).card := by
  27     rw [Finset.card_subtype]
  28     congr 1
  29     ext x;simp
  30   have he (S : RootedSubset V T k) :
  31       (A⊆(rootedSubsetFree T S).val) ↔ R⊆S.val.val := by
  32     constructor
  33     · intro h x hx
  34       by_cases ht : x∈T
  35       · exact S.property ht
  36       · have hh := h (by simpa only [A,Finset.mem_subtype] using hx : (⟨x,ht⟩:{x : V // x∉T})∈A)
  37         simpa only [rootedSubsetFree,Finset.mem_subtype] using hh
  38     · intro h x hx
  39       have hh : x.val∈R := by simpa only [A,Finset.mem_subtype] using hx
  40       simpa only [rootedSubsetFree,Finset.mem_subtype] using h hh
  41   have hh := fixedSubset_joint_inclusion_le hkn A
  42   rw [← rootedSubset_uniform_free T k,finiteLaw_map_probability] at hh
  43   simpa only [he,hA,hn] using hh

  46 theorem rootedSubset_conflict_probability {V : Type*} [Fintype V] [DecidableEq V]
  47     (F : SimpleGraph V) (T : Finset V) (k : ℕ) (hk : T.card+k≤Fintype.card V)
  48     [Nonempty (RootedSubset V T k)] (L : ℝ)
  49     (hdeg : ∀ v,(F.degree v:ℝ)≤L)
  50     (hT : ∀ x∈T,∀ y∈T,¬F.Adj x y) :
  51     let q : ℝ := (k:ℝ)/(Fintype.card V-T.card:ℕ)
  52     (FiniteLaw.uniform (RootedSubset V T k)).probability
  53       (fun S=>∃ x∈S.val.val,∃ y∈S.val.val,F.Adj x y)≤
  54       L*(2*(T.card:ℝ)*q+Fintype.card V*q^2) := by
  55   intro q
  56   let μ := FiniteLaw.uniform (RootedSubset V T k)
  57   have hq : 0≤q := by dsimp [q];positivity
  58   have hpair (x y : V) (hxy : F.Adj x y) :
  59       μ.probability (fun S=>x∈S.val.val ∧ y∈S.val.val)≤
  60         if x∈T ∨ y∈T then q else q^2 := by
  61     have hh := rootedSubset_additional_inclusion T {x,y} k hk
  62     have hne : x≠y := hxy.ne
  63     have hnot : ¬(x∈T ∧ y∈T) := fun h=>hT x h.1 y h.2 hxy
  64     simp only [Finset.insert_subset_iff,Finset.singleton_subset_iff] at hh
  65     by_cases hx : x∈T <;> by_cases hy : y∈T
  66     · exact False.elim (hnot ⟨hx,hy⟩)
  67     · have hset : ({x,y}:Finset V)\T={y} := by
  68         ext z
  69         simp only [Finset.mem_sdiff,Finset.mem_insert,Finset.mem_singleton]
  70         aesop
  71       rw [hset] at hh
  72       simpa only [μ,q,Finset.card_singleton,pow_one,hx,true_or,ite_true] using hh
  73     · have hset : ({x,y}:Finset V)\T={x} := by
  74         ext z
  75         simp only [Finset.mem_sdiff,Finset.mem_insert,Finset.mem_singleton]
  76         aesop
  77       rw [hset] at hh
  78       simpa only [μ,q,Finset.card_singleton,pow_one,hy,or_true,ite_true] using hh
  79     · have hset : ({x,y}:Finset V)\T={x,y} := by
  80         ext z
  81         simp only [Finset.mem_sdiff,Finset.mem_insert,Finset.mem_singleton]
  82         aesop
  83       rw [hset] at hh
  84       have hcard : ({x,y}:Finset V).card=2 := by simp [hne]
  85       simpa only [μ,q,hcard,hx,hy,false_or,ite_false] using hh
  86   have hterm (a : V×V) :
  87       μ.probability (fun S=>a.1∈S.val.val ∧ a.2∈S.val.val ∧ F.Adj a.1 a.2)≤
  88         if F.Adj a.1 a.2 then (if a.1∈T ∨ a.2∈T then q else q^2) else 0 := by
  89     by_cases h : F.Adj a.1 a.2
  90     · simpa only [h,and_true,ite_true] using hpair a.1 a.2 h
  91     · simp only [h,and_false,FiniteLaw.probability,ite_false,Finset.sum_const_zero,le_refl]
  92   have hbad := finiteLaw_union_bound μ (Finset.univ : Finset (V×V))
  93     (fun a S=>a.1∈S.val.val ∧ a.2∈S.val.val ∧ F.Adj a.1 a.2)
  94   have hmono : μ.probability (fun S=>∃ x∈S.val.val,∃ y∈S.val.val,F.Adj x y)≤
  95       μ.probability (fun S=>∃ a∈(Finset.univ : Finset (V×V)),a.1∈S.val.val ∧ a.2∈S.val.val ∧ F.Adj a.1 a.2) := by
  96     apply finiteLaw_probability_mono
  97     rintro S ⟨x,hx,y,hy,hxy⟩
  98     exact ⟨(x,y),Finset.mem_univ _,hx,hy,hxy⟩
  99   have hout (x : V) : (∑ y,bipAdj F.Adj x y)≤L := by
 100     simpa only [bipAdj,graph_adjacency_row_sum] using hdeg x
 101   have hin (y : V) : (∑ x,bipAdj F.Adj x y)≤L := by
 102     have he : (∑ x,bipAdj F.Adj x y)=(∑ x,bipAdj F.Adj y x) := by
 103       apply Finset.sum_congr rfl
 104       intro x _
 105       simp only [bipAdj,F.adj_comm]
 106     rw [he]
 107     exact hout y
 108   calc
 109     _ ≤ ∑ a : V×V,μ.probability (fun S=>a.1∈S.val.val ∧ a.2∈S.val.val ∧ F.Adj a.1 a.2) := hmono.trans hbad
 110     _ ≤ ∑ a : V×V,if F.Adj a.1 a.2 then (if a.1∈T ∨ a.2∈T then q else q^2) else 0 := Finset.sum_le_sum (fun a _=>hterm a)
 111     _ ≤ _ := by
 112       rw [Fintype.sum_prod_type]
 113       convert relation_root_incident_cost F.Adj L hout hin T T q (q^2) hq (sq_nonneg q) using 1 <;> ring
