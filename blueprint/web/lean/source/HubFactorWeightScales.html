  10 theorem hub_factor_weight_scale (m : ℕ) (n k β b B t : ℝ)
  11     (hn : 0<n) (hk : 0<k) (hβ : 0<β) (hb : 0≤b) (hB : 0≤B)
  12     (hkn : k≤n) (hβn : β*n≤k) (ht : b/n^m≤t ∧ t≤B/n^m) :
  13     (b*β^m)/k^m≤t ∧ t≤B/k^m := by
  14   constructor
  15   · apply le_trans _ ht.1
  16     apply (div_le_div_iff₀ (pow_pos hk _) (pow_pos hn _)).mpr
  17     have hp := pow_le_pow_left₀ (mul_nonneg hβ.le hn.le) hβn m
  18     rw [mul_pow] at hp
  19     nlinarith only [mul_le_mul_of_nonneg_left hp hb]
  20   · exact ht.2.trans (div_le_div_of_nonneg_left hB (pow_pos hk _) (pow_le_pow_left₀ hk.le hkn _))
