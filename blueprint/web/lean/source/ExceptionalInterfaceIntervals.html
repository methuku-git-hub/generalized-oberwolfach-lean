   7 def HasCleanBlock (J : Finset ℕ) (a L t : ℕ) : Prop :=
   8   ∃ s : ℕ, a ≤ s ∧ s+t ≤ a+L ∧ ∀ j, s ≤ j → j < s+t → j ∉ J

  10 theorem two_blocks_one_exception (J : Finset ℕ) (hJ : J.card ≤ 1)
  11     (a L t : ℕ) (hL : 2*t ≤ L) : HasCleanBlock J a L t := by
  12   classical
  13   by_cases hfirst : ∀ j, a ≤ j → j < a+t → j ∉ J
  14   · exact ⟨a,le_rfl,by omega,hfirst⟩
  15   push Not at hfirst
  16   obtain ⟨e,hae,het,heJ⟩ := hfirst
  17   refine ⟨a+t,by omega,by omega,?_⟩
  18   intro j hj hjt hjJ
  19   have heq : e=j := (Finset.card_le_one.mp hJ) e heJ j hjJ
  20   omega

  22 theorem edge_has_clean_prefix_or_suffix (J : Finset ℕ) (hJ : J.card ≤ 1)
  23     (a L t e : ℕ) (hL : 4*t+1 ≤ L) (he : a ≤ e) (heL : e < a+L) :
