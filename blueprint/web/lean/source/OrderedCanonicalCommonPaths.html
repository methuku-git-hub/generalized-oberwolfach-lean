  16 theorem rootedCycleOrder_reinsert (σ : Equiv.Perm (Fin (m+1))) (v w : V) (z : Fin m → V) :
  17     Fin.insertNth (α:=fun _=>V) (σ.symm 0) w
  18       (fun k=>rootedCycleOrder σ v z ((σ.symm 0).succAbove k))=rootedCycleOrder σ w z := by
  19   funext a
  20   by_cases ha : a=σ.symm 0
  21   · subst a
  22     simp only [Fin.insertNth_apply_same,rootedCycleOrder,Equiv.apply_symm_apply,Fin.cases_zero]
  23   · obtain ⟨k,rfl⟩ := Fin.exists_succAbove_eq ha
  24     rw [Fin.insertNth_apply_succAbove]
  25     have hs : σ ((σ.symm 0).succAbove k)≠0 := by
  26       intro he
  27       have hh := congrArg σ.symm he
  28       rw [Equiv.symm_apply_apply] at hh
  29       exact Fin.succAbove_ne _ _ hh
  30     obtain ⟨l,hl⟩ := Fin.eq_succ_of_ne_zero hs
  31     simp only [rootedCycleOrder,hl,Fin.cases_succ]

  33 def orderedCommonMarkedPath_to_canonical
  34     (σ : Equiv.Perm (Fin (m+1))) (v v' : V) (u u' : U) :
  35     OrderedCommonMarkedPath tail head ktail khead m π b H₀ Hₖ σ v v' u u' ↪
  36       CanonicalCommonMarkedPath tail head ktail khead m π b H₀ Hₖ (σ.symm 0) v v' u u' where
  37   toFun z := by
  38     refine ⟨fun k=>rootedCycleOrder σ v z.val ((σ.symm 0).succAbove k),?_,?_,?_⟩
  39     · rw [rootedCycleOrder_reinsert]
  40       exact z.property.1
  41     · rw [rootedCycleOrder_reinsert]
  42       exact z.property.2.1
  43     · intro k
  44       have hs : σ ((σ.symm 0).succAbove k)≠0 := by
  45         intro he
  46         have hh := congrArg σ.symm he
  47         rw [Equiv.symm_apply_apply] at hh
  48         exact Fin.succAbove_ne _ _ hh
  49       obtain ⟨l,hl⟩ := Fin.eq_succ_of_ne_zero hs
  50       have hk : (σ.symm 0).succAbove k=σ.symm l.succ := by
  51         apply σ.injective
  52         simpa only [Equiv.apply_symm_apply] using hl
  53       simp only [hk,rootedCycleOrder,Equiv.apply_symm_apply,Fin.cases_succ]
  54       exact z.property.2.2 l
  55   inj' z w he := by
  56     apply Subtype.ext
  57     have hf := congrArg Subtype.val he
  58     have hi := congrArg (Fin.insertNth (α:=fun _=>V) (σ.symm 0) v) hf
  59     change Fin.insertNth (α:=fun _=>V) (σ.symm 0) v
  60         (fun k=>rootedCycleOrder σ v z.val ((σ.symm 0).succAbove k))=
  61       Fin.insertNth (α:=fun _=>V) (σ.symm 0) v
  62         (fun k=>rootedCycleOrder σ v w.val ((σ.symm 0).succAbove k)) at hi
  63     rw [rootedCycleOrder_reinsert,rootedCycleOrder_reinsert] at hi
  64     funext l
  65     have hh := congrFun hi (σ.symm l.succ)
  66     simpa only [rootedCycleOrder,Equiv.apply_symm_apply,Fin.cases_succ] using hh

  68 theorem orderedCommonMarkedPath_canonical_card_le [Fintype V]
  69     (σ : Equiv.Perm (Fin (m+1))) (v v' : V) (u u' : U) :
  70     Fintype.card (OrderedCommonMarkedPath tail head ktail khead m π b H₀ Hₖ σ v v' u u')≤
  71       Fintype.card (CanonicalCommonMarkedPath tail head ktail khead m π b H₀ Hₖ (σ.symm 0) v v' u u') :=
  72   Fintype.card_le_of_injective _ (orderedCommonMarkedPath_to_canonical σ v v' u u').injective
