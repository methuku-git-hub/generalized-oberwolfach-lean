  11 theorem finiteLaw_eq_uniform_of_mass_constant
  12     {Ω : Type*} [Fintype Ω] [Nonempty Ω] (μ : FiniteLaw Ω)
  13     (hc : ∀ x y, μ.mass x = μ.mass y) : μ = FiniteLaw.uniform Ω := by
  14   apply finiteLaw_ext
  15   intro x
  16   have hs : (Fintype.card Ω : ℝ) * μ.mass x = 1 := by
  17     have h := μ.total
  18     simp_rw [hc _ x] at h
  19     simpa only [Finset.sum_const, Finset.card_univ, nsmul_eq_mul] using h
  20   change μ.mass x = 1 / Fintype.card Ω
  21   have hn : (Fintype.card Ω : ℝ) ≠ 0 := by exact_mod_cast Fintype.card_ne_zero
  22   apply (eq_div_iff hn).2
  23   simpa only [mul_comm] using hs

  25 theorem finiteLaw_uniform_map_equal_fibres
  26     {Ω Γ : Type*} [Fintype Ω] [Fintype Γ] [Nonempty Ω] [Nonempty Γ]
  27     (f : Ω → Γ) (c : ℕ) (hc : ∀ γ, Fintype.card {ω // f ω = γ} = c) :
  28     (FiniteLaw.uniform Ω).map f = FiniteLaw.uniform Γ := by
  29   apply finiteLaw_eq_uniform_of_mass_constant
  30   intro x y
  31   have hm (γ : Γ) : ((FiniteLaw.uniform Ω).map f).mass γ =
  32       (Fintype.card {ω // f ω = γ} : ℝ) * (1 / Fintype.card Ω) := by
  33     simp [FiniteLaw.map, FiniteLaw.uniform, Fintype.card_subtype, ← Finset.sum_filter]
  34   rw [hm x, hm y, hc x, hc y]
