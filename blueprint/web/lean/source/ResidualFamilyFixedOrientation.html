   1 import Oberwolfach.ResidualFixedOrientation
   2 /-! Fix all factor orientations before choosing any of their residual
   3 subgraphs, then construct their exact dependent source models. -/
   4 set_option autoImplicit false
   5 noncomputable section
   6 namespace Oberwolfach
   7 attribute [local instance] Classical.propDecidable

  11 theorem two_factor_family_fixed_orientation_for_all_subgraphs
  12     (F:∀i,SimpleGraph (X i)) (hF:∀i,(F i).IsRegularOfDegree 2):
  13     ∃ (D:∀i,X i→X i→Prop) (orig:∀i,List (PrescribedSourcePiece (X i))),
  14       (∀i,(orig i).Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset)) ∧
  15       (∀i p,p∈orig i→∃a xs,p=.cycle a xs ∧ p.OriginalValid (D i)) ∧
  16       (∀i,sourcePieceVerticesUnion (orig i)=Finset.univ ∧ sourcePieceEdgesUnion (orig i)=(F i).edgeFinset) ∧
  17       (∀i x y,D i x y ↔ (x,y)∈sourcePieceArcsUnion (orig i)) ∧
  18       ∀ (H:∀i,SimpleGraph (X i)) (R:∀i,Finset (X i)),
  19         (∀i,H i≤F i)→(∀i x,x∉R i→∀y,¬(H i).Adj x y)→
  20         ∃ pieces:∀i,List (PrescribedSourcePiece (X i)),
  21           (∀i,(pieces i).Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset)) ∧
  22           (∀i p,p∈pieces i→p.Shape ∧ p.OriginalValid (D i)) ∧
  23           (∀i,prescribedDependentVertices pieces i=R i) ∧
  24           (∀i,prescribedDependentGraph pieces i=H i) ∧
  25           ∀i x y,prescribedDependentDigraph pieces i x y ↔ D i x y ∧ (H i).Adj x y := by
  26   choose D orig hpair hcycles hvertices hedges hD hsub using
  27     fun i=>two_factor_fixed_orientation_for_all_subgraphs (F i) (hF i)
  28   refine ⟨D,orig,hpair,hcycles,?_,hD,?_⟩
  29   · intro i;exact ⟨hvertices i,hedges i⟩
  30   · intro H R hHF hisol
  31     choose pieces hps hvalid hv he ha using
  32       fun i=>hsub i (H i) (R i) (hHF i) (hisol i)
  33     refine ⟨pieces,hps,hvalid,?_,?_,?_⟩
  34     · intro i;exact hv i
  35     · intro i;exact he i
  36     · intro i x y;exact ha i x y
  37 end Oberwolfach
