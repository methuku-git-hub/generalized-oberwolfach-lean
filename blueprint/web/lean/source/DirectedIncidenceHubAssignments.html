  34  theorem directed_incidence_assigned_hubs_injective
  35     (f:V→V') (h:E→U) (hh:Function.Injective h)
  36     (hloop:∀ e,f (head e)≠f (tail e))
  37     (σ:∀ v,Equiv.Perm (DirectedIncidence tail head v)):
  38     Function.Injective (fun i:Σ v,DirectedIncidence tail head v=>
  39       (f i.1,h (directedIncidenceEdge (σ i.1 i.2)))):=by
  40   rintro ⟨v,i⟩ ⟨w,j⟩ he
  41   have hfv:f v=f w:=congrArg Prod.fst he
  42   have hedge:directedIncidenceEdge (σ v i)=directedIncidenceEdge (σ w j):=hh (congrArg Prod.snd he)
  43   have hvw:v=w:=by
  44     have hv:=directedIncidence_endpoint (σ v i)
  45     have hw:=directedIncidence_endpoint (σ w j)
  46     rw [←hedge] at hw
  47     rcases hv with hv|hv <;> rcases hw with hw|hw
  48     · exact hv.trans hw.symm
  49     · exact False.elim (hloop _ ((congrArg f hv).symm.trans (hfv.trans (congrArg f hw))))
  50     · exact False.elim (hloop _ ((congrArg f hw).symm.trans (hfv.symm.trans (congrArg f hv))))
  51     · exact hv.trans hw.symm
  52   subst w
  53   have hloop':∀ e,head e≠tail e:=fun e he=>hloop e (congrArg f he)
  54   have hij:i=j:=(σ v).injective (directedIncidenceEdge_injective hloop' v hedge)
  55   subst j
  56   rfl
