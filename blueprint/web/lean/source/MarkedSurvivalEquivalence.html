  19 theorem markedCycle_restrict_iff (PE : E → Prop) (PK : K → Prop)
  20     (hp : Function.Injective (markedPhysicalEdge tail head ktail khead π))
  21     (C : MarkedCycleTuple tail head ktail khead m π b) :
  22     IsMarkedCycleTuple (fun e : {e // PE e}=>tail e.val) (fun e=>head e.val)
  23       (fun e : {e // PK e}=>ktail e.val) (fun e=>khead e.val) m π b C.val ↔
  24         ∀ i,Sum.elim PE PK (markedCycleTypedEdge tail head ktail khead m π b C i) := by
  25   have hinj := markedPhysicalEdge_endpoint_injections tail head ktail khead π hp
  26   constructor
  27   · intro h i
  28     refine Fin.lastCases ?_ (fun j=>?_) i
  29     · obtain ⟨e,he⟩ := h.2.2
  30       have hh := Classical.choose_spec C.property.2.2
  31       have heq : Classical.choose C.property.2.2=e.val := hinj.2 (Prod.ext (hh.1.trans he.1.symm) (hh.2.trans he.2.symm))
  32       simpa only [markedCycleTypedEdge,Fin.snoc_last,Sum.elim_inr,heq] using e.property
  33     · obtain ⟨e,he⟩ := h.2.1 j
  34       have hh := Classical.choose_spec (C.property.2.1 j)
  35       have heq : Classical.choose (C.property.2.1 j)=e.val := hinj.1 (Prod.ext (hh.1.trans he.1.symm) (hh.2.trans he.2.symm))
  36       simpa only [markedCycleTypedEdge,Fin.snoc_castSucc,Sum.elim_inl,heq] using e.property
  37   · intro h
  38     refine ⟨C.property.1,?_,?_⟩
  39     · intro j
  40       have hj := h j.castSucc
  41       simp only [markedCycleTypedEdge,Fin.snoc_castSucc,Sum.elim_inl] at hj
  42       exact ⟨⟨Classical.choose (C.property.2.1 j),hj⟩,Classical.choose_spec (C.property.2.1 j)⟩
  43     · have hj := h (Fin.last m)
  44       simp only [markedCycleTypedEdge,Fin.snoc_last,Sum.elim_inr] at hj
  45       exact ⟨⟨Classical.choose C.property.2.2,hj⟩,Classical.choose_spec C.property.2.2⟩

  47 theorem markedHub_restrict_iff
  48     (hH₀ : ∀ v u,H₀' v u → H₀ v u) (hHₖ : ∀ v u,Hₖ' v u → Hₖ v u)
  49     (hdisj : ∀ v u,¬(H₀ v u ∧ Hₖ v u))
  50     (W : MarkedWheel tail head ktail khead m π b H₀ Hₖ) :
  51     MarkedHubCompatible H₀' Hₖ' W.1.val W.2.val ↔
  52       ∀ v∈Set.range W.1.val,H₀' v W.2.val ∨ Hₖ' v W.2.val := by
  53   constructor
  54   · intro h v hv
  55     obtain ⟨i,rfl⟩ := hv
  56     refine Fin.cases ?_ (fun j=>?_) i
  57     · exact Or.inr h.1
  58     · exact Or.inl (h.2 j)
  59   · intro h
  60     constructor
  61     · rcases h (W.1.val 0) ⟨0,rfl⟩ with ho|hk
  62       · exact False.elim (hdisj _ _ ⟨hH₀ _ _ ho,W.2.property.1⟩)
  63       · exact hk
  64     · intro j
  65       rcases h (W.1.val j.succ) ⟨j.succ,rfl⟩ with ho|hk
  66       · exact ho
  67       · exact False.elim (hdisj _ _ ⟨W.2.property.2 j,hHₖ _ _ hk⟩)

  69 theorem markedWheelSurvives_iff_occurrences (PE : E → Prop) (PK : K → Prop)
  70     (hp : Function.Injective (markedPhysicalEdge tail head ktail khead π))
  71     (hH₀ : ∀ v u,H₀' v u → H₀ v u) (hHₖ : ∀ v u,Hₖ' v u → Hₖ v u)
  72     (hdisj : ∀ v u,¬(H₀ v u ∧ Hₖ v u))
  73     (W : MarkedWheel tail head ktail khead m π b H₀ Hₖ) :
  74     MarkedWheelSurvives PE PK H₀' Hₖ' W ↔
  75       ∀ e∈markedOccurrenceResources W,
  76         Sum.elim (Sum.elim PE PK) (fun vu=>H₀' vu.1 vu.2 ∨ Hₖ' vu.1 vu.2) e := by
  77   rw [MarkedWheelSurvives,markedCycle_restrict_iff PE PK hp,markedHub_restrict_iff hH₀ hHₖ hdisj]
  78   constructor
  79   · rintro ⟨hc,hh⟩ e he
  80     cases e with
  81     | inl e =>
  82       rw [markedOccurrenceResources_rim,markedCycleTypedEdge_mem_iff hp] at he
  83       obtain ⟨i,rfl⟩ := he
  84       exact hc i
  85     | inr vu =>
  86       rw [markedOccurrenceResources_spoke] at he
  87       dsimp only [Sum.elim_inr]
  88       rw [← he.1]
  89       exact hh vu.1 he.2
  90   · intro h
  91     constructor
  92     · intro i
  93       apply h (Sum.inl (markedCycleTypedEdge tail head ktail khead m π b W.1 i))
  94       rw [markedOccurrenceResources_rim,markedCycleTypedEdge_mem_iff hp]
  95       exact ⟨i,rfl⟩
  96     · intro v hv
  97       apply h (Sum.inr (v,W.2.val))
  98       rw [markedOccurrenceResources_spoke]
  99       exact ⟨rfl,hv⟩
