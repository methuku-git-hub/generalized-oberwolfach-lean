  13 theorem paired_fiber_card {A X Y : Type*} [Fintype A] [Fintype Y]
  14     (f : A → X) (g : A → Y) (hinj : Function.Injective (fun a=>(f a,g a))) (x : X) :
  15     (Finset.univ.filter (fun a=>f a=x)).card≤Fintype.card Y := by
  16   classical
  17   have hi : Set.InjOn g (↑(Finset.univ.filter (fun a=>f a=x)):Set A) := by
  18     intro a ha b hb he
  19     have ha' : f a=x := (Finset.mem_filter.mp ha).2
  20     have hb' : f b=x := (Finset.mem_filter.mp hb).2
  21     exact hinj (Prod.ext (ha'.trans hb'.symm) he)
  22   rw [← Finset.card_image_of_injOn hi]
  23   exact Finset.card_le_univ _

  25 theorem paired_incidence_card {A V : Type*} [Fintype A] [Fintype V]
  26     (f g : A → V) (hinj : Function.Injective (fun a=>(f a,g a))) (v : V) :
  27     (Finset.univ.filter (fun a=>f a=v ∨ g a=v)).card≤2*Fintype.card V := by
  28   classical
  29   rw [Finset.filter_or]
  30   have h1 := paired_fiber_card f g hinj v
  31   have h2 := paired_fiber_card g f (fun a b he=>hinj (Prod.ext (congrArg Prod.snd he) (congrArg Prod.fst he))) v
  32   have hh := Finset.card_union_le (Finset.univ.filter (fun a=>f a=v)) (Finset.univ.filter (fun a=>g a=v))
  33   omega

  35 theorem filter_card_sum_type {A B : Type*} [Fintype A] [Fintype B] (P : A⊕B → Prop) :
  36     (Finset.univ.filter P).card=(Finset.univ.filter (fun a=>P (Sum.inl a))).card+
  37       (Finset.univ.filter (fun b=>P (Sum.inr b))).card := by
  38   classical
  39   calc
  40     _ = ∑ x : A⊕B, if P x then (1:ℕ) else 0 := (Finset.sum_boole (R:=ℕ) P Finset.univ).symm
  41     _ = _ := by rw [Fintype.sum_sum_type]; simp only [Finset.sum_boole,Nat.cast_id]

  47 def markedResourceIncident (physical : Bool) : V⊕U → MarkedActiveEdge E K H₀ Hₖ → Prop
  48   | Sum.inl v,Sum.inl (Sum.inl e) => tail e=v ∨ head e=v
  49   | Sum.inl v,Sum.inl (Sum.inr e) => ktail e=v ∨ (if physical then π (khead e) else khead e)=v
  50   | Sum.inl v,Sum.inr vu => vu.val.1=v
  51   | Sum.inr _,Sum.inl _ => False
  52   | Sum.inr u,Sum.inr vu => vu.val.2=u
