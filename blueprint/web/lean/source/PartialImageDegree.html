   9 theorem partial_injective_image_degree {X V:Type} [Fintype X] [Fintype V]
  10     [DecidableEq V] (Q:SimpleGraph X) (f:X↪V) (H:SimpleGraph V)
  11     (hsub:H≤Q.map f) (hQ:∀ x,Q.degree x≤2) (v:V):
  12     H.degree v≤(if ∃ x,f x=v then 2 else 0) := by
  13   by_cases hv:∃ x,f x=v
  14   · rcases hv with ⟨x,rfl⟩
  15     have hx:∃ z,f z=f x:=⟨x,rfl⟩
  16     simp only [if_pos hx]
  17     have hN:H.neighborFinset (f x)⊆(Q.map f).neighborFinset (f x):=by
  18       intro y hy
  19       simp only [SimpleGraph.mem_neighborFinset] at hy ⊢
  20       exact hsub hy
  21     have hh:H.degree (f x)≤(Q.map f).degree (f x):=by
  22       simpa only [SimpleGraph.card_neighborFinset_eq_degree] using Finset.card_le_card hN
  23     exact hh.trans ((SimpleGraph.degree_map_apply Q f x).le.trans (hQ x))
  24   · simp only [if_neg hv]
  25     rw [←SimpleGraph.card_neighborFinset_eq_degree]
  26     apply Nat.le_zero.mpr
  27     apply Finset.card_eq_zero.mpr
  28     apply Finset.eq_empty_iff_forall_notMem.mpr
  29     intro y hy
  30     simp only [SimpleGraph.mem_neighborFinset] at hy
  31     have hh:(Q.map f).Adj v y:=hsub hy
  32     obtain ⟨x,z,hxz,hx,hz⟩:=(SimpleGraph.map_adj f Q v y).mp hh
  33     exact hv ⟨x,hx⟩
  34 end Oberwolfach
