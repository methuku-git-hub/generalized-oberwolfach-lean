  12 theorem finite_sequence_walk_edge_at
  13     {V:Type*} [DecidableEq V] (G:SimpleGraph V) (n:ℕ) (f:Fin (n+1)→V)
  14     {a b:V} (p:G.Walk a b) (hlen:p.length=n) (hsupp:p.support=List.ofFn f)
  15     (i:Fin n):
  16     (p.edges[ i.val ]' (by simpa [SimpleGraph.Walk.length_edges,hlen] using i.isLt)).toFinset=
  17       {f i.castSucc,f i.succ} := by
  18   have hv:∀j:ℕ,(hj:j≤n)→p.getVert j=f ⟨j,by omega⟩:=by
  19     intro j hj
  20     have hh:=p.getVert_eq_support_getElem? (n:=j) (by omega)
  21     rw [hsupp] at hh
  22     simpa only [List.getElem?_ofFn, dif_pos (show j<n+1 by omega), Option.some.injEq] using hh
  23   have hedge:i.val<p.edges.length:=by simpa [SimpleGraph.Walk.length_edges,hlen] using i.isLt
  24   rw [SimpleGraph.Walk.getElem_edges hedge]
  25   rw [hv i.val (by omega),hv (i.val+1) (by omega)]
  26   rw [Sym2.toFinset_mk_eq]
  27   have h1:(⟨i.val,by omega⟩:Fin (n+1))=i.castSucc:=by apply Fin.ext;rfl
  28   have h2:(⟨i.val+1,by omega⟩:Fin (n+1))=i.succ:=by apply Fin.ext;rfl
  29   rw [h1,h2]
  30 end Oberwolfach
