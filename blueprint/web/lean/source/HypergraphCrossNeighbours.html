  13 theorem hypergraph_neighbours_in_other_clique
  14     {V E : Type*} [Fintype E] (R : E → Finset V) {k B : ℕ}
  15     (hsize : ∀ e,(R e).card≤k)
  16     (hcodeg : ∀ v w,v≠w → (Finset.univ.filter (fun e=>v∈R e ∧ w∈R e)).card≤B)
  17     (e : E) (z : V) (hz : z∉R e) (T : Finset E) (hT : ∀ f∈T,z∈R f) :
  18     (T.filter (fun f=>(hypergraphLineGraph R).Adj e f)).card≤k*B := by
  19   classical
  20   let N := fun v=>Finset.univ.filter (fun f=>v∈R f ∧ z∈R f)
  21   have hsub : T.filter (fun f=>(hypergraphLineGraph R).Adj e f)⊆(R e).biUnion N := by
  22     intro f hf
  23     obtain ⟨hfT,_,hdis⟩ := Finset.mem_filter.mp hf
  24     obtain ⟨v,hve,hvf⟩ := Finset.not_disjoint_iff.mp hdis
  25     exact Finset.mem_biUnion.mpr ⟨v,hve,Finset.mem_filter.mpr
  26       ⟨Finset.mem_univ _,hvf,hT f hfT⟩⟩
  27   calc
  28     _ ≤ ((R e).biUnion N).card := Finset.card_le_card hsub
  29     _ ≤ (R e).card*B := Finset.card_biUnion_le_card_mul _ _ _ (fun v hv=>
  30       hcodeg v z (fun he=>hz (he ▸ hv)))
  31     _ ≤ _ := Nat.mul_le_mul_right _ (hsize e)
  32 
  33 end Oberwolfach
