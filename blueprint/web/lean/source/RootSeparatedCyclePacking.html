  12 theorem root_separated_cycle_packing (β:ℝ) (hβ:0<β) (r s:ℕ) (hr:3≤r):
  13     ∃ N:ℕ,∀ (V:Type) [Fintype V] [DecidableEq V],N≤Fintype.card V→
  14     ∀ (G:SimpleGraph V),(∀ v,(1/2+β)*(Fintype.card V:ℝ)≤G.degree v)→
  15     ∀ T:Finset V,T.card≤s→∃ M:Finset (OrdinaryCycle G r),
  16       (∀ C∈M,∀ E∈M,C≠E→Disjoint (ordinaryCycleVertices G r C) (ordinaryCycleVertices G r E)) ∧
  17       T⊆M.biUnion (ordinaryCycleVertices G r) ∧
  18       (∀ C∈M,∃ x∈T,x∈ordinaryCycleVertices G r C ∧
  19         ∀ y∈T,y∈ordinaryCycleVertices G r C→y=x) ∧ M.card≤T.card:=by
  20   obtain ⟨N,hN⟩:=fixed_cycle_through_vertex β hβ r (s+r*s) hr
  21   refine ⟨N,?_⟩
  22   intro V _ _ hn G hdeg T hT
  23   have build:∀ P:Finset V,P⊆T→∃ M:Finset (OrdinaryCycle G r),
  24       (∀ C∈M,∀ E∈M,C≠E→Disjoint (ordinaryCycleVertices G r C) (ordinaryCycleVertices G r E)) ∧
  25       P⊆M.biUnion (ordinaryCycleVertices G r) ∧
  26       (∀ C∈M,∃ x∈P,x∈ordinaryCycleVertices G r C ∧
  27         ∀ y∈T,y∈ordinaryCycleVertices G r C→y=x) ∧ M.card≤P.card:=by
  28     intro P
  29     induction P using Finset.induction_on with
  30     | empty =>
  31       intro _
  32       refine ⟨∅,?_,?_,?_,?_⟩ <;> simp
  33     | @insert x P hx ih =>
  34       intro hsub
  35       have hxT:x∈T:=hsub (Finset.mem_insert_self _ _)
  36       have hPT:P⊆T:=fun y hy=>hsub (Finset.mem_insert_of_mem hy)
  37       obtain ⟨M,hpack,hcover,hunique,hMcard⟩:=ih hPT
  38       let Used:=M.biUnion (ordinaryCycleVertices G r)
  39       have hUsed:Used.card≤r*P.card:=by
  40         have hh:=Finset.card_biUnion_le_card_mul M (ordinaryCycleVertices G r) r
  41           (by intro C hC;exact (ordinaryCycleVertices_card G r C).le)
  42         calc
  43           _ ≤ M.card*r := hh
  44           _ = r*M.card := Nat.mul_comm _ _
  45           _ ≤ r*P.card := Nat.mul_le_mul_left r hMcard
  46       have hxUsed:x∉Used:=by
  47         intro hxU
  48         obtain ⟨C,hCM,hxC⟩:=Finset.mem_biUnion.mp hxU
  49         obtain ⟨y,hyP,hyC,hyunique⟩:=hunique C hCM
  50         exact hx (by simpa [hyunique x hxT hxC] using hyP)
  51       let Forbidden:=(T.erase x)∪Used
  52       have hforbid:Forbidden.card≤s+r*s:=by
  53         change ((T.erase x)∪Used).card≤s+r*s
  54         have hh:=Finset.card_union_le (T.erase x) Used
  55         have he:(T.erase x).card≤s:=(Finset.card_erase_le (s:=T) (a:=x)).trans hT
  56         have hp:P.card≤s:=(Finset.card_le_card hPT).trans hT
  57         have hm:=Nat.mul_le_mul_left r hp
  58         omega
  59       have hxforbid:x∉Forbidden:=by simp [Forbidden,hxUsed]
  60       obtain ⟨C,hcycle,hCG,hxC,havoid⟩:=hN (Fintype.card V) hn V (le_refl _) G hdeg x Forbidden hforbid hxforbid
  61       let C':OrdinaryCycle G r:=⟨⟨C,hcycle⟩,hCG⟩
  62       have hxC':x∈ordinaryCycleVertices G r C':=by simpa only [ordinaryCycleVertices,Set.mem_toFinset] using hxC
  63       have hCavoid:∀ y∈ordinaryCycleVertices G r C',y∉Forbidden:=by
  64         intro y hy;apply havoid y;simpa only [ordinaryCycleVertices,Set.mem_toFinset] using hy
  65       have hCnew:C'∉M:=by
  66         intro hCM
  67         exact hxUsed (Finset.mem_biUnion.mpr ⟨C',hCM,hxC'⟩)
  68       refine ⟨insert C' M,?_,?_,?_,?_⟩
  69       · intro A hA E hE hne
  70         simp only [Finset.mem_insert] at hA hE
  71         rcases hA with rfl|hA <;> rcases hE with rfl|hE
  72         · exact (hne rfl).elim
  73         · apply Finset.disjoint_left.mpr
  74           intro y hyC hyE
  75           have hyU:y∈Used:=Finset.mem_biUnion.mpr ⟨E,hE,hyE⟩
  76           exact hCavoid y hyC (Finset.mem_union_right _ hyU)
  77         · apply Finset.disjoint_left.mpr
  78           intro y hyA hyC
  79           have hyU:y∈Used:=Finset.mem_biUnion.mpr ⟨A,hA,hyA⟩
  80           exact hCavoid y hyC (Finset.mem_union_right _ hyU)
  81         · exact hpack A hA E hE hne
  82       · intro y hy
  83         rcases Finset.mem_insert.mp hy with rfl|hy
  84         · exact Finset.mem_biUnion.mpr ⟨C',Finset.mem_insert_self _ _,hxC'⟩
  85         · obtain ⟨E,hE,hyE⟩:=Finset.mem_biUnion.mp (hcover hy)
  86           exact Finset.mem_biUnion.mpr ⟨E,Finset.mem_insert_of_mem hE,hyE⟩
  87       · intro E hE
  88         rcases Finset.mem_insert.mp hE with rfl|hE
  89         · refine ⟨x,Finset.mem_insert_self _ _,hxC',?_⟩
  90           intro y hyT hyC
  91           by_contra hne
  92           exact hCavoid y hyC (Finset.mem_union_left _ (Finset.mem_erase.mpr ⟨hne,hyT⟩))
  93         · obtain ⟨y,hyP,hyC,hyunique⟩:=hunique E hE
  94           exact ⟨y,Finset.mem_insert_of_mem hyP,hyC,hyunique⟩
  95       · rw [Finset.card_insert_of_notMem hCnew,Finset.card_insert_of_notMem hx]
  96         omega
  97   obtain ⟨M,hpack,hcover,hunique,hcard⟩:=build T (Finset.Subset.rfl)
  98   exact ⟨M,hpack,hcover,hunique,hcard⟩
  99 end Oberwolfach
