  10 theorem graph_degree_sdiff {V : Type*} [Fintype V] [DecidableEq V]
  11     (G H : SimpleGraph V) (h : H ≤ G) (x : V) :
  12     (G \ H).degree x = G.degree x - H.degree x := by
  13   have heq : (G \ H).neighborFinset x = G.neighborFinset x \ H.neighborFinset x := by
  14     ext y
  15     simp only [SimpleGraph.mem_neighborFinset, Finset.mem_sdiff, SimpleGraph.sdiff_adj]
  16   have hs : H.neighborFinset x ⊆ G.neighborFinset x := by
  17     intro y hy
  18     simp only [SimpleGraph.mem_neighborFinset] at hy ⊢
  19     exact h hy
  20   rw [← SimpleGraph.card_neighborFinset_eq_degree, heq, Finset.card_sdiff_of_subset hs,
  21     SimpleGraph.card_neighborFinset_eq_degree, SimpleGraph.card_neighborFinset_eq_degree]

  23 theorem graph_edge_count_sdiff {V : Type*} [Fintype V] [DecidableEq V]
  24     (G H : SimpleGraph V) (h : H ≤ G) :
  25     (G \ H).edgeFinset.card = G.edgeFinset.card - H.edgeFinset.card := by
  26   rw [SimpleGraph.edgeFinset_sdiff, Finset.card_sdiff_of_subset (SimpleGraph.edgeFinset_mono h)]

  28 theorem mapped_regular_degree_or_zero {V W : Type*} [Fintype V] [Fintype W]
  29     [DecidableEq V] [DecidableEq W] (G : SimpleGraph V) {d : ℕ}
  30     (hG : G.IsRegularOfDegree d) (f : V ↪ W) (x : W) :
  31     (G.map f).degree x = 0 ∨ (G.map f).degree x = d := by
  32   by_cases hx : x ∈ Set.range f
  33   · obtain ⟨v, rfl⟩ := hx
  34     right
  35     have hh := SimpleGraph.degree_map_apply G f v
  36     rw [hG v] at hh
  37     simpa only [← SimpleGraph.ncard_neighborSet] using hh
  38   · left
  39     have hiso : (G.map f).IsIsolated x := by
  40       intro y hxy
  41       rw [SimpleGraph.map_adj] at hxy
  42       obtain ⟨u, v, huv, hu, hv⟩ := hxy
  43       exact hx ⟨u, hu⟩
  44     exact hiso.degree_eq_zero

  46 theorem graph_sdiff_even_degrees {V : Type*} [Fintype V] [DecidableEq V]
  47     (G H : SimpleGraph V) (h : H ≤ G)
  48     (hG : ∀ x, Even (G.degree x)) (hH : ∀ x, Even (H.degree x)) :
  49     ∀ x, Even ((G \ H).degree x) := by
  50   intro x
  51   rw [graph_degree_sdiff G H h x]
  52   exact (hG x).tsub (hH x)

  54 theorem graph_filtered_edges_sdiff {V : Type*} [Fintype V] [DecidableEq V]
  55     (G H : SimpleGraph V) (P : Sym2 V → Prop) [DecidablePred P]
  56     (hP : ∀ e ∈ G.edgeFinset, P e → e ∉ H.edgeFinset) :
  57     (G \ H).edgeFinset.filter P = G.edgeFinset.filter P := by
  58   rw [SimpleGraph.edgeFinset_sdiff]
  59   ext e
  60   simp only [Finset.mem_filter, Finset.mem_sdiff]
  61   exact ⟨fun h => ⟨h.1.1, h.2⟩, fun h => ⟨⟨h.1, hP e h.1 h.2⟩, h.2⟩⟩
