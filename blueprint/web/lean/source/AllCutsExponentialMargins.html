  10 theorem eventually_allCuts_failure_exponential {c : ℝ} (hc : 0<c) :
  11     ∃ N : ℕ,1≤N ∧ ∀ n : ℕ,N≤n → ∀ C : ℝ,0≤C →
  12       C*(4:ℝ)^n*Real.exp (-c*(n:ℝ)^2)≤C*Real.exp (-(c/2)*n) := by
  13   let N := max 1 (Nat.ceil (2*Real.log 4/c))
  14   refine ⟨N,le_max_left _ _,?_⟩
  15   intro n hn C hC
  16   have hn1 : (1:ℝ)≤n := by exact_mod_cast (show 1≤n by dsimp [N] at hn;omega)
  17   have hn0 : (0:ℝ)≤n := Nat.cast_nonneg n
  18   have hceil : (Nat.ceil (2*Real.log 4/c):ℝ)≤n := by
  19     exact_mod_cast (show Nat.ceil (2*Real.log 4/c)≤n by dsimp [N] at hn;omega)
  20   have hh : 2*Real.log 4/c≤(n:ℝ) := (Nat.le_ceil _).trans hceil
  21   have hlog := (div_le_iff₀ hc).mp hh
  22   have hmul := mul_le_mul_of_nonneg_right hlog hn0
  23   have hexp : (n:ℝ)*Real.log 4-c*(n:ℝ)^2≤-(c/2)*(n:ℝ)^2 := by nlinarith only [hmul]
  24   have heq : (4:ℝ)^n=Real.exp ((n:ℝ)*Real.log 4) := by
  25     rw [Real.exp_nat_mul,Real.exp_log (by norm_num : (0:ℝ)<4)]
  26   calc
  27     _ = C*Real.exp ((n:ℝ)*Real.log 4-c*(n:ℝ)^2) := by rw [heq,Real.exp_sub];rw [div_eq_mul_inv,← Real.exp_neg];ring
  28     _ ≤ C*Real.exp (-(c/2)*(n:ℝ)^2) := mul_le_mul_of_nonneg_left (Real.exp_le_exp.mpr hexp) hC
  29     _ ≤ _ := by
  30       apply mul_le_mul_of_nonneg_left _ hC
  31       apply Real.exp_le_exp.mpr
  32       have hn2 : (n:ℝ)≤(n:ℝ)^2 := by nlinarith only [hn1]
  33       exact mul_le_mul_of_nonpos_left hn2 (by linarith)
