  13 theorem injective_fiber_sum_le {X Z : Type*} [Fintype X]
  14     (f : X → Z) (hf : Function.Injective f) (g : X → ℝ) {B : ℝ}
  15     (hB : 0≤B) (hg : ∀ x,g x≤B) (z : Z) :
  16     (∑ x,if f x=z then g x else 0)≤B := by
  17   classical
  18   by_cases hz : ∃ x,f x=z
  19   · obtain ⟨x,hx⟩ := hz
  20     calc
  21       _ = g x := by
  22         rw [Finset.sum_eq_single x]
  23         · simp only [hx,ite_true]
  24         · intro y _ hy
  25           have hh : f y≠z := fun he=>hy (hf (he.trans hx.symm))
  26           simp only [hh,ite_false]
  27         · simp
  28       _ ≤ B := hg x
  29   · have hz' : ∀ x,f x≠z := by simpa only [not_exists] using hz
  30     simpa only [hz',ite_false,Finset.sum_const_zero] using hB

  32 theorem typed_physical_incidence_bound {X Y Z : Type*} [Fintype X] [Fintype Y]
  33     (J : X → Y → ℕ) (f : X → Z) (g : Y → Z)
  34     (hf : Function.Injective f) (hg : Function.Injective g) {B : ℝ}
  35     (hB : 0≤B) (hrow : ∀ x,(∑ y,(J x y : ℝ))≤B)
  36     (hcol : ∀ y,(∑ x,(J x y : ℝ))≤B) (z : Z) :
  37     (∑ x,∑ y,(J x y : ℝ)*((if f x=z then 1 else 0)+(if g y=z then 1 else 0)))≤2*B := by
  38   have h1 := injective_fiber_sum_le f hf (fun x=>∑ y,(J x y : ℝ)) hB hrow z
  39   have h2 := injective_fiber_sum_le g hg (fun y=>∑ x,(J x y : ℝ)) hB hcol z
  40   have he : (∑ x,∑ y,(J x y : ℝ)*((if f x=z then 1 else 0)+(if g y=z then 1 else 0)))=
  41       (∑ x,if f x=z then (∑ y,(J x y : ℝ)) else 0)+
  42       (∑ y,if g y=z then (∑ x,(J x y : ℝ)) else 0) := by
  43     simp only [mul_add,Finset.sum_add_distrib]
  44     congr 1
  45     · apply Finset.sum_congr rfl
  46       intro x _
  47       by_cases hx : f x=z <;> simp only [hx,ite_true,ite_false,mul_one,mul_zero,Finset.sum_const_zero]
  48     · rw [Finset.sum_comm]
  49       apply Finset.sum_congr rfl
  50       intro y _
  51       by_cases hy : g y=z <;> simp only [hy,ite_true,ite_false,mul_one,mul_zero,Finset.sum_const_zero]
  52   rw [he]
  53   linarith only [h1,h2]
