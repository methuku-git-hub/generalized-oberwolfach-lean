   9 theorem regular_induced_degree_near_centre {V:Type} [Fintype V] [DecidableEq V]
  10     (G:SimpleGraph V) (D:ℕ) (hreg:G.IsRegularOfDegree D) (B:Finset V)
  11     (v:(B:Set V)):
  12     |(((G.induce (B:Set V)).degree v):ℝ)-
  13        ((D:ℝ)/Fintype.card V)*B.card|≤(Fintype.card V-B.card:ℕ):=by
  14   have hn:0<Fintype.card V:=Fintype.card_pos_iff.mpr ⟨v.val⟩
  15   have hnR:(0:ℝ)<Fintype.card V:=by exact_mod_cast hn
  16   have hD:D≤Fintype.card V:=by
  17     have hh:=G.degree_lt_card_verts v.val
  18     rw [hreg v.val] at hh
  19     omega
  20   have hDR:(D:ℝ)≤Fintype.card V:=by exact_mod_cast hD
  21   have hB:B.card≤Fintype.card V:=Finset.card_le_univ B
  22   have hBR:(B.card:ℝ)≤Fintype.card V:=by exact_mod_cast hB
  23   let θ:ℝ:=D/Fintype.card V
  24   have hθ0:0≤θ:=by dsimp [θ];positivity
  25   have hθ1:θ≤1:=by dsimp [θ];exact (div_le_one hnR).mpr hDR
  26   have hsub:G.neighborFinset v.val⊆(G.neighborFinset v.val∩B)∪(Finset.univ\B):=by
  27     intro w hw
  28     by_cases h:w∈B
  29     · exact Finset.mem_union_left _ (Finset.mem_inter.mpr ⟨hw,h⟩)
  30     · exact Finset.mem_union_right _ (Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,h⟩)
  31   have hl: D≤(G.neighborFinset v.val∩B).card+(Fintype.card V-B.card):=by
  32     have hh:=(Finset.card_le_card hsub).trans (Finset.card_union_le _ _)
  33     simpa only [SimpleGraph.card_neighborFinset_eq_degree,hreg v.val,Finset.card_sdiff,
  34       Finset.inter_univ,Finset.card_univ] using hh
  35   have hu:(G.neighborFinset v.val∩B).card≤D:=by
  36     have hh:=Finset.card_le_card (Finset.inter_subset_left (s₁:=G.neighborFinset v.val) (s₂:=B))
  37     simpa only [SimpleGraph.card_neighborFinset_eq_degree,hreg v.val] using hh
  38   have hlR:(D:ℝ)≤((G.neighborFinset v.val∩B).card:ℝ)+(Fintype.card V-B.card:ℕ):=by exact_mod_cast hl
  39   have huR:((G.neighborFinset v.val∩B).card:ℝ)≤D:=by exact_mod_cast hu
  40   have hmiss:((Fintype.card V-B.card:ℕ):ℝ)=(Fintype.card V:ℝ)-B.card:=by rw [Nat.cast_sub hB]
  41   have hcentre:(D:ℝ)=θ*Fintype.card V:=by dsimp [θ];field_simp
  42   have hmul0:0≤θ*((Fintype.card V:ℝ)-B.card):=mul_nonneg hθ0 (by linarith)
  43   have hmul1:θ*((Fintype.card V:ℝ)-B.card)≤(Fintype.card V:ℝ)-B.card:=by
  44     nlinarith [mul_nonneg (sub_nonneg.mpr hθ1) (sub_nonneg.mpr hBR)]
  45   rw [graph_induced_degree_as_intersection]
  46   change |((G.neighborFinset v.val∩B).card:ℝ)-θ*B.card|≤(Fintype.card V-B.card:ℕ)
  47   rw [hmiss]
  48   apply abs_le.mpr
  49   constructor <;> nlinarith
