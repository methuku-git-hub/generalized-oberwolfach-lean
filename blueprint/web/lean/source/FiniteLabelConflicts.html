  13 theorem label_predicate_card_bound {A V : Type*} [Fintype A] [Fintype V]
  14     (f : A → V) (P : V → Prop) (M : ℕ)
  15     (hf : ∀ v, Fintype.card {a // f a=v} ≤ M) :
  16     Fintype.card {a // P (f a)} ≤ Fintype.card {v // P v} * M := by
  17   classical
  18   let g : {a // P (f a)} → {v // P v} := fun a => ⟨f a.val,a.property⟩
  19   apply finite_fibre_card_bound g M
  20   intro v
  21   let enc : {a // g a=v} → {a // f a=v.val} :=
  22     fun a => ⟨a.val.val,congrArg Subtype.val a.property⟩
  23   have hi : Function.Injective enc := by
  24     intro a b h
  25     apply Subtype.ext; apply Subtype.ext
  26     exact congrArg (fun z : {a // f a=v.val} => z.val) h
  27   convert (Fintype.card_le_of_injective enc hi).trans (hf v.val) using 1 <;> congr!

  29 theorem label_permutation_close_card {A V : Type*} [Fintype A] [Fintype V]
  30     (f : A → V) (π : Equiv.Perm V) (b M : ℕ)
  31     (hf : ∀ v, Fintype.card {a // f a=v} ≤ M) (v : V) :
  32     Fintype.card {a // PermutationClose π b v (f a)} ≤ M*(2*b+1) := by
  33   calc
  34     _ ≤ Fintype.card {w // PermutationClose π b v w} * M :=
  35       label_predicate_card_bound f _ M hf
  36     _ ≤ (2*b+1)*M := Nat.mul_le_mul_right M (permutationClose_neighbour_card π b v)
  37     _ = _ := Nat.mul_comm _ _

  39 theorem label_fibre_bound_reindex {A Y V : Type*} [Fintype A] [Fintype Y]
  40     (f : A → V) (e : Y ≃ A) (M : ℕ)
  41     (hf : ∀ v, Fintype.card {a // f a=v} ≤ M) :
  42     ∀ v, Fintype.card {y // f (e y)=v} ≤ M := by
