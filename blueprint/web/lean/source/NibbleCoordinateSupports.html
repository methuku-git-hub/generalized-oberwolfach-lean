  18 theorem coinColourSurvives_congr
  19     {V C : Type*} (G : SimpleGraph V) (σ τ : V → C×Bool) (v : V)
  20     (hv : σ v=τ v) (hN : ∀ u,G.Adj v u → σ u=τ u) :
  21     coinColourSurvives G σ v ↔ coinColourSurvives G τ v := by
  22   unfold coinColourSurvives tentativeColourSurvives
  23   rw [hv]
  24   apply and_congr_right'
  25   apply forall_congr'
  26   intro u
  27   apply forall_congr'
  28   intro hu
  29   change (σ u).1≠(σ v).1 ↔ (τ u).1≠(τ v).1
  30   rw [hN u hu,hv]

  32 theorem coinConflictSet_congr
  33     {V C : Type*} [Fintype V] [DecidableEq V]
  34     (G : SimpleGraph V) (T : Finset V) (σ τ : V → C×Bool)
  35     (he : ∀ u∈closedSetNeighbours G T,σ u=τ u) :
  36     coinConflictSet G T σ=coinConflictSet G T τ := by
  37   ext v
  38   simp only [coinConflictSet,Finset.mem_filter]
  39   apply and_congr_right

  41   apply not_congr
  42   apply coinColourSurvives_congr G σ τ v

  44   · intro u hu
  45     exact he u (Finset.mem_union_right _ (Finset.mem_filter.mpr ⟨Finset.mem_univ _,v,hv,hu⟩))

  57 theorem colourRejectedIn_congr
  58     {V C : Type*} [Fintype V] [DecidableEq V] (G : SimpleGraph V)
  59     (T : Finset V) (σ τ : V → C×Bool)
  60     (he : ∀ v∈closedSetNeighbours G T,σ v=τ v) (c : C) :
  61     colourRejectedIn G T σ c ↔ colourRejectedIn G T τ c := by
  62   apply exists_congr
  63   intro v
  64   apply and_congr_right
  65   intro hv
  66   have hroot := he v (Finset.mem_union_left _ hv)
  67   have hsurv := coinColourSurvives_congr G σ τ v hroot (fun u hu=>
  68     he u (Finset.mem_union_right _ (Finset.mem_filter.mpr ⟨Finset.mem_univ _,v,hv,hu⟩)))
  69   change (σ v).1=c ∧ ¬coinColourSurvives G σ v ↔
  70     (τ v).1=c ∧ ¬coinColourSurvives G τ v
  71   rw [hroot,hsurv]

  74     {V : Type*} [Fintype V] [DecidableEq V] (G : SimpleGraph V)
  75     {S T : Finset V} (hST : S⊆T) : closedSetNeighbours G S⊆closedSetNeighbours G T := by
  76   intro v hv
  77   rcases Finset.mem_union.mp hv with hv|hv
  78   · exact Finset.mem_union_left _ (hST hv)

  80     exact Finset.mem_union_right _ (Finset.mem_filter.mpr ⟨Finset.mem_univ _,u,hST hu,huv⟩)

  82 theorem nibbleColourCount_support
  83     {V C J : Type*} [Fintype V] [DecidableEq V]
  84     (G : SimpleGraph V) (Q : Finset J) (T : J → Finset V)
  85     (L : Finset C) (a b : ℕ) (σ τ : V → C×Bool)
  86     (he : ∀ v∈closedSetNeighbours G (Q.biUnion T),σ v=τ v) :
  87     nibbleColourCount G Q T L a b σ=nibbleColourCount G Q T L a b τ := by
  88   have hclose (j : J) (hj : j∈Q) :
  89       ∀ v∈closedSetNeighbours G (T j),σ v=τ v :=
  90     fun v hv=>he v (closedSetNeighbours_mono G
  91       (fun v hv=>Finset.mem_biUnion.mpr ⟨j,hj,hv⟩) hv)
  92   have ha (c : C) : assignedSetIndices Q T σ c=assignedSetIndices Q T τ c := by
  93     ext j
  94     simp only [assignedSetIndices,Finset.mem_filter]
  95     apply and_congr_right
  96     intro hj
  97     exact colourAssignedIn_congr (T j) σ τ
  98       (fun v hv=>hclose j hj v (Finset.mem_union_left _ hv)) c
