   9 theorem hub_test_mass_margin
  10     (n s η f fmin θ τ err card r rmax mass:ℝ)
  11     (hn:0≤n) (hs:0<s) (hη:0≤η) (hf: fmin≤f) (hfmin:0≤fmin)
  12     (hθ:θ≤1/2) (hτ:τ≤1/2) (herr:err≤η/4)
  13     (hcount:s*n/2≤card) (hr:0<r) (hrmax:r≤rmax)
  14     (hbalance:η*s=(1-θ)*f)
  15     (hmass:((1-τ)*η-err)*card/r≤mass):
  16     fmin*n/(16*rmax)≤mass := by
  17   have hapos:η/4≤(1-τ)*η-err:=by nlinarith
  18   have hcard:0≤card:=le_trans (by positivity) hcount
  19   have hprod: (η/4)*(s*n/2)≤((1-τ)*η-err)*card :=
  20     mul_le_mul hapos hcount (by positivity) (le_trans (by positivity) hapos)
  21   have hθpos:(1/2:ℝ)≤1-θ:=by linarith
  22   have hfpos:0≤f:=le_trans hfmin hf
  23   have hbal:η*s/2≤η*s:=by nlinarith [mul_nonneg hη hs.le]
  24   have hfn:fmin*n/2≤(1-θ)*f*n:=by
  25     have h1:0≤(f-fmin)*n:=mul_nonneg (sub_nonneg.mpr hf) hn
  26     have h2:0≤((1-θ)-1/2)*(f*n):=mul_nonneg (sub_nonneg.mpr hθpos) (mul_nonneg hfpos hn)
  27     nlinarith
  28   have hrmaxpos:0<rmax:=lt_of_lt_of_le hr hrmax
  29   have hnum:fmin*n/16≤((1-τ)*η-err)*card:=by
  30     nlinarith [hprod,hfn,hbalance]
  31   have hdiv:fmin*n/(16*rmax)≤((1-τ)*η-err)*card/r := by
  32     apply (div_le_div_iff₀ (by positivity : 0<16*rmax) hr).2
  33     have hh:=mul_le_mul_of_nonneg_right hnum hrmaxpos.le
  34     have hx:=mul_le_mul_of_nonneg_left hrmax (by positivity : 0≤fmin*n)
  35     nlinarith
  36   exact hdiv.trans hmass
  37 end Oberwolfach
