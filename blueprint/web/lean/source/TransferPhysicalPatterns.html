  18 def transferRequiredEdges (T : OrdinaryTransferConfiguration hr P F H v u u') : Finset (Sym2 V) :=
  19   Finset.univ.biUnion (fun j : TransferCycleIndex r=>
  20     (transferCycleGraph (by omega) T.labels v T.omitted j).edgeFinset)

  22 theorem transferRequiredEdges_card (T : OrdinaryTransferConfiguration hr P F H v u u') :
  23     (transferRequiredEdges T).card≤(1+2*r)*r := by

  25     ∑ j : TransferCycleIndex r,(transferCycleGraph (by omega) T.labels v T.omitted j).edgeFinset.card)

  27   simp only [hc,Finset.sum_const,Finset.card_univ,Fintype.card_sum,Fintype.card_unique,
  28     Fintype.card_prod,Fintype.card_fin,Fintype.card_bool,smul_eq_mul,Nat.mul_comm] at hh

  31     congr!

  34 theorem transferRequiredEdges_subset (T : OrdinaryTransferConfiguration hr P F H v u u') :
  35     transferRequiredEdges T⊆P.edgeFinset := by
  36   intro e he
  37   obtain ⟨j,_,he⟩ := Finset.mem_biUnion.mp he

  39   exact SimpleGraph.edgeSet_mono (T.rim_edges j) he

  41 theorem transferRequiredEdges_endpoints (T : OrdinaryTransferConfiguration hr P F H v u u')
  42     (x y : V) (he : s(x,y)∈transferRequiredEdges T) :
  43     x≠y ∧ x∈insert v (Set.range T.labels) ∧ y∈insert v (Set.range T.labels) := by
  44   obtain ⟨j,_,he⟩ := Finset.mem_biUnion.mp he

  48   · exact transferCycleGraph_support_subset hr T.labels v T.omitted j ⟨x,he.symm⟩

  56 theorem transfer_required_edge_incidence (hn : 0<Fintype.card V) (e : Sym2 V) :

  63         fun T he=>(transferRequiredEdges_endpoints T x y he).1 hxy

  65       positivity

  77           split_ifs
  78           · exact hh.2.2
  79           · exact hh.2.1
  80         exact (Set.mem_insert_iff.mp hs).resolve_left hz
  81       have hh := ordinary_transfer_configuration_root_count (hr:=hr) (Q:=P) (F:=F) (H:=H) (v:=v) (u:=u) (u':=u') {z}

  83           (r:ℝ)^2*(Fintype.card V:ℝ)^(r^2-1) := by

  85         apply Finset.sum_le_sum
  86         intro T _
  87         by_cases he : s(x,y)∈transferRequiredEdges T
  88         · simp only [he,ite_true]
