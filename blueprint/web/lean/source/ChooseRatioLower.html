  13 theorem choose_ratio_product {r n k : ℕ} :
  14     (r.choose k:ℝ)/n.choose k=
  15       ∏ i∈Finset.range k,((r-i:ℕ):ℝ)/(n-i:ℕ) := by
  16   have hf : (k.factorial:ℝ)≠0 := by exact_mod_cast Nat.factorial_ne_zero k
  17   calc
  18     _ = ((k.factorial:ℝ)*r.choose k)/((k.factorial:ℝ)*n.choose k) := by rw [mul_div_mul_left _ _ hf]
  19     _ = (r.descFactorial k:ℝ)/n.descFactorial k := by
  20       simp only [Nat.descFactorial_eq_factorial_mul_choose,Nat.cast_mul]
  21     _ = _ := by simp only [Nat.descFactorial_eq_prod_range,Nat.cast_prod,Finset.prod_div_distrib]

  24     (((r-k:ℕ):ℝ)/n)^k ≤ (r.choose k:ℝ)/n.choose k := by
  25   rw [choose_ratio_product]

  27   calc
  28     _ = ∏ _i∈Finset.range k,((r-k:ℕ):ℝ)/n := by simp [div_pow]
  29     _ ≤ _ := by
  30       apply Finset.prod_le_prod₀
  31       · intro i _;positivity
  32       · intro i hi
  33         have hik := Finset.mem_range.mp hi
  34         have hin : i < n := by omega
  35         have hd : (0:ℝ)<(n-i:ℕ) := by exact_mod_cast Nat.sub_pos_of_lt hin
  36         have h₁ : ((r-k:ℕ):ℝ) ≤ (r-i:ℕ) := by exact_mod_cast (show r-k ≤ r-i by omega)

  39           _ ≤ ((r-i:ℕ):ℝ)/n := div_le_div_of_nonneg_right h₁ hnp.le
