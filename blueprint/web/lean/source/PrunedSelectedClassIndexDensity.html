  14     (D:ℕ) (hEven:Even D) (hsize:Fintype.card W=D/2) (hDn:D≤n)

  18     let fibres:=fun j:Option (Fin K)=>Finset.univ.filter (fun w=>classify w=j)
  19     let cutoff:=fun j:Option (Fin K)=>if j=none then κL else κS
  20     let small:=fun j:Option (Fin K)=>((fibres j).card:ℝ)<cutoff j*n
  21     let removed:=Finset.univ.filter (fun w=>small (classify w))
  22     let remaining:={w:W // w∉removed}
  23     let cf:=fun w:remaining=>classify w.val
  24     let D0:=D-2*removed.card
  25     let fibres0:=fun j:Option (Fin K)=>Finset.univ.filter (fun w:remaining=>cf w=j)
  26     let C:=Finset.univ.filter (fun j=>(fibres0 j).Nonempty)
  27     let k:=C.card
  28     let e:Fin k≃↥C:=(Fintype.equivFinOfCardEq (Fintype.card_coe C)).symm
  29     let m:=fun a:Fin k=>(fibres0 (e a).val).card
  30     let ρ:=fun a:Fin k=>2*(m a:ℝ)/(D0:ℝ);
  31     ∀a:Fin k, initialIndexDensity ↥(fibres0 (e a).val) D0=ρ a ∧
  32       ρ₀ ≤ initialIndexDensity ↥(fibres0 (e a).val) D0:=by
