  43     |rawMarkedRowCount P π d (markedRowDomain P π (3*d) A i.1.val i.2.1.val) i.1.val i.2.2.val g j σ-

  45         (markedRowDomain P π (3*d) A i.1.val i.2.1.val).card|
  46   let err := 2*Real.exp (-2*t^2/((8*(d:ℝ)+6)^2*(2*(2*(h:ℝ)+1)*A.card)))
  47   have htail (i : I) : (FiniteLaw.pi (rawMarkedWheelLaw p k hp hk hsum d hd q hq hQ s hs hS)).probability (Bad i)≤err := by
  48     have hh := rawMarkedRowCount_tail P π d hd p k hp hk hsum q hq hQ s hs hS hnoshort A hA
  49       i.1.val i.2.1.val i.2.2.val i.2.2.property.2 (2*h)

  55     exact hh
  56   have hcardJ : Fintype.card JJ≤(2*Fintype.card U+1)^h := by

  66       _=_ := by rw [show 2*h=h+h by omega,pow_add];ring
