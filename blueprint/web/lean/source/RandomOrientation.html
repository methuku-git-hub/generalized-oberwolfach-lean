   4 /-!
   5 Independent fair orientations of physical edges. A coordinate is an
   6 unordered pair of vertices, so opposite arcs share one random bit.
   7 Unused pairs have no effect on the orientation. The common event below is
   8 chosen before a forbidden graph or a prescribed outdegree vector is supplied.
   9 -/

  18 /-- The true bit directs the edge from its smaller to its larger endpoint. -/
  19 def orientationBit {V : Type*} [LinearOrder V] (x y : V) : Bool := decide (x < y)

  21 def sampledOrientation {V : Type*} [LinearOrder V] (G : SimpleGraph V)
  22     (σ : Sym2 V → Bool) (x y : V) : ℕ :=
  23   if G.Adj x y ∧ σ s(x, y) = orientationBit x y then 1 else 0

  25 theorem sampledOrientation_physical
  26     {V : Type*} [LinearOrder V] (G : SimpleGraph V) (σ : Sym2 V → Bool) (x y : V) :
  27     sampledOrientation G σ x y + sampledOrientation G σ y x =
  28       if G.Adj x y then 1 else 0 := by
  29   by_cases hxy : G.Adj x y
  30   · have hyx := hxy.symm
  31     rcases lt_or_gt_of_ne hxy.ne with hlt | hgt
  32     · have hrev : ¬ y < x := not_lt_of_gt hlt
  33       have hs : s(y, x) = s(x, y) := Sym2.eq_swap
  34       cases hb : σ s(x, y) <;> simp [sampledOrientation, hxy, hyx, hs,
  35         orientationBit, hlt, hrev, hb]
  36     · have hrev : ¬ x < y := not_lt_of_gt hgt
  37       have hs : s(y, x) = s(x, y) := Sym2.eq_swap
  38       cases hb : σ s(x, y) <;> simp [sampledOrientation, hxy, hyx, hs,
  39         orientationBit, hgt, hrev, hb]
  40   · have hyx : ¬ G.Adj y x := fun h => hxy h.symm
  41     simp [sampledOrientation, hxy, hyx]

  43 def twoEdgeCoordinates {V : Type*} [DecidableEq V] (x y z : V) : Finset (Sym2 V) :=
  44   {s(x, z), s(z, y)}

  46 def twoEdgeRequirements {V : Type*} [LinearOrder V] (x y z : V) (e : Sym2 V) : Bool :=
  47   if e = s(x, z) then orientationBit x z else orientationBit z y

  57 theorem twoEdgeCoordinates_disjoint
  58     {V : Type*} [DecidableEq V] {x y z w : V}
  59     (hxy : x ≠ y) (hzx : z ≠ x) (hzy : z ≠ y)
  60     (_hwx : w ≠ x) (_hwy : w ≠ y) (hzw : z ≠ w) :
  61     Disjoint (twoEdgeCoordinates x y z) (twoEdgeCoordinates x y w) := by
  62   apply Finset.disjoint_left.mpr
  63   intro e he hf
  64   simp only [twoEdgeCoordinates, Finset.mem_insert, Finset.mem_singleton] at he hf
  65   rcases he with rfl | rfl <;> rcases hf with hf | hf <;>
  66     simp only [Sym2.eq_iff] at hf <;> tauto

  72       ∑ y ∈ G.neighborFinset x,
  73         patternValue {s(x, y)} (fun _ => orientationBit x y) σ := by
  74   have hn : G.neighborFinset x = Finset.univ.filter (G.Adj x) := by

 103     {x y z : V} (hxy : x ≠ y) (hxz : G.Adj x z) (hzy : G.Adj z y) :

 105       if 0 < sampledOrientation G σ x z ∧ 0 < sampledOrientation G σ z y then 1 else 0 := by

 116 /-- Common neighbours whose two incident edges have the directions x to z
 117 and z to y. The eligibility graph E is fixed before the orientation. -/
 118 def eligibleTwoPathCount {V : Type*} [Fintype V] (E : SimpleGraph V)
 119     (M : V → V → ℕ) (x y : V) : ℕ :=
 120   ((E.neighborFinset x ∩ E.neighborFinset y).filter
 121     (fun z => 0 < M x z ∧ 0 < M z y)).card

 123 theorem sampledOrientation_twoPath_patterns
 124     {V : Type*} [Fintype V] [LinearOrder V] (G E : SimpleGraph V) (hEG : E ≤ G)
 125     (σ : Sym2 V → Bool) {x y : V} (hxy : x ≠ y) :
 126     (eligibleTwoPathCount E (sampledOrientation G σ) x y : ℝ) =
 127       ∑ z ∈ E.neighborFinset x ∩ E.neighborFinset y,
 128         patternValue (twoEdgeCoordinates x y z) (twoEdgeRequirements x y z) σ := by
 129   rw [eligibleTwoPathCount, Finset.card_filter, Nat.cast_sum]
 130   apply Finset.sum_congr rfl
 131   intro z hz
 132   have hxz : E.Adj x z := by simpa using (Finset.mem_inter.mp hz).1
 133   have hyz : E.Adj y z := by simpa using (Finset.mem_inter.mp hz).2
 134   rw [sampledOrientation_twoPath_pattern G σ hxy (hEG hxz) (hEG hyz.symm)]
 135   split_ifs <;> norm_num

 137 theorem sampledOrientation_twoPath_tail
 138     {V : Type*} [Fintype V] [LinearOrder V] (G E : SimpleGraph V) (hEG : E ≤ G)
 139     {x y : V} (hxy : x ≠ y) {t : ℝ} (ht : 0 ≤ t) :
 140     (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.fairCoin)).probability (fun σ =>
 141       t ≤ |(eligibleTwoPathCount E (sampledOrientation G σ) x y : ℝ) -
 142         ((E.neighborFinset x ∩ E.neighborFinset y).card : ℝ) / 4|) ≤
 143       2 * Real.exp (-2 * t ^ 2 /
 144         (((E.neighborFinset x ∩ E.neighborFinset y).card : ℝ) * 2)) := by
 145   let C := E.neighborFinset x ∩ E.neighborFinset y
 146   have hne (z : V) (hz : z ∈ C) : z ≠ x ∧ z ≠ y := by
 147     have hxz : E.Adj x z := by simpa using (Finset.mem_inter.mp hz).1
 148     have hyz : E.Adj y z := by simpa using (Finset.mem_inter.mp hz).2
 149     exact ⟨hxz.ne.symm, hyz.ne.symm⟩
 150   have hdisj : ∀ z ∈ C, ∀ w ∈ C, z ≠ w →
 151       Disjoint (twoEdgeCoordinates x y z) (twoEdgeCoordinates x y w) := by
 152     intro z hz w hw hzw
 153     exact twoEdgeCoordinates_disjoint hxy (hne z hz).1 (hne z hz).2
 154       (hne w hw).1 (hne w hw).2 hzw
 155   have hh := finiteLaw_fair_disjoint_patterns_tail C (twoEdgeCoordinates x y)
 156     (twoEdgeRequirements x y) 2 (fun _ _ => twoEdgeCoordinates_card hxy) hdisj ht
 157   have hcenter : (C.card : ℝ) * (1 / 2 : ℝ) ^ 2 = (C.card : ℝ) / 4 := by ring
 158   rw [hcenter] at hh
 159   simpa only [C, ← sampledOrientation_twoPath_patterns G E hEG _ hxy, Nat.cast_ofNat] using hh

 163     {x y : V} (hxy : x ≠ y) :

 187 /-- A single event controls all vertices and all distinct ordered pairs.
 188 The failure sum is explicit; no later forbidden graph enters the event. -/
 189 theorem sampledOrientation_common_event
 190     {V : Type*} [Fintype V] [LinearOrder V] (G E : SimpleGraph V) (hEG : E ≤ G)
 191     {t : ℝ} (ht : 0 ≤ t) :
 192     1 - ((∑ x : V, 2 * Real.exp (-2 * t ^ 2 / G.degree x)) +
 193       (∑ p : {p : V × V // p.1 ≠ p.2},
 194         2 * Real.exp (-((E.neighborFinset p.1.1 ∩ E.neighborFinset p.1.2).card : ℝ) / 64))) ≤
 195       (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.fairCoin)).probability (fun σ =>
 196         (∀ x, |((∑ y, sampledOrientation G σ x y : ℕ) : ℝ) - (G.degree x : ℝ) / 2| < t) ∧
 197         ∀ x y, x ≠ y → ((E.neighborFinset x ∩ E.neighborFinset y).card : ℝ) / 8 <
 198           (eligibleTwoPathCount E (sampledOrientation G σ) x y : ℝ)) := by
 199   let J := V ⊕ {p : V × V // p.1 ≠ p.2}
 200   let A : J → (Sym2 V → Bool) → Prop := Sum.elim
 201     (fun x σ => t ≤ |((∑ y, sampledOrientation G σ x y : ℕ) : ℝ) - (G.degree x : ℝ) / 2|)
 202     (fun p σ => (eligibleTwoPathCount E (sampledOrientation G σ) p.1.1 p.1.2 : ℝ) ≤
 203       ((E.neighborFinset p.1.1 ∩ E.neighborFinset p.1.2).card : ℝ) / 8)
 204   let b : J → ℝ := Sum.elim
 205     (fun x => 2 * Real.exp (-2 * t ^ 2 / G.degree x))
 206     (fun p => 2 * Real.exp (-((E.neighborFinset p.1.1 ∩ E.neighborFinset p.1.2).card : ℝ) / 64))
 207   have hb : ∀ j ∈ (Finset.univ : Finset J),
 208       (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.fairCoin)).probability (A j) ≤ b j := by
 209     intro j _
 210     cases j with
 211     | inl x => exact sampledOrientation_degree_tail G x ht
 212     | inr p => exact sampledOrientation_twoPath_lower_tail G E hEG p.2
 213   have hh := finiteLaw_common_event_lower
 214     (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.fairCoin)) Finset.univ A b hb
 215   simpa only [J, Fintype.sum_sum_type, b, A, Sum.elim_inl, Sum.elim_inr,
 216     Finset.mem_univ, forall_const, Sum.forall, not_le, Subtype.forall, Prod.forall] using hh

 218 theorem commonNeighbours_of_minDegree
 219     {V : Type*} [Fintype V] (G : SimpleGraph V) {α : ℝ}
 220     (hdeg : ∀ v, (1 / 2 + α) * Fintype.card V ≤ (G.degree v : ℝ)) (x y : V) :
 221     2 * α * Fintype.card V ≤ ((G.neighborFinset x ∩ G.neighborFinset y).card : ℝ) := by
 222   have hc := Finset.card_inter_add_card_union (G.neighborFinset x) (G.neighborFinset y)
 223   have hu : (G.neighborFinset x ∪ G.neighborFinset y).card ≤ Fintype.card V :=
 224     Finset.card_le_univ _
 225   have hr : ((G.neighborFinset x ∩ G.neighborFinset y).card : ℝ) +
 226       ((G.neighborFinset x ∪ G.neighborFinset y).card : ℝ) =

 255 /-- The routing conclusion holds for every later forbidden graph on one
 256 orientation event. The degree vector is exactly half the undirected degree;
 257 its total and the physical/auxiliary conversion are proved here. -/
 258 theorem sampledOrientation_balancing_event
 259     {V : Type*} [Fintype V] [LinearOrder V] (G E : SimpleGraph V) (hEG : E ≤ G)
 260     (d : V → ℕ) (hd : ∀ v, 2 * d v = G.degree v)
 261     (L : ℕ) (hL : 0 < L) (hn : 0 < Fintype.card V)
 262     {a f β : ℝ} (ha : 0 ≤ a) (hf : 0 ≤ f) (hβ : 0 < β)
 263     (hcommon : ∀ x y, x ≠ y → 8 * β * Fintype.card V ≤
 264       ((E.neighborFinset x ∩ E.neighborFinset y).card : ℝ))
 265     (hmargin : (Fintype.card V : ℝ) * a / L + 2 * a + 4 * L + 2 * f < β * Fintype.card V) :
 266     1 - ((∑ x : V, 2 * Real.exp (-2 * a ^ 2 / G.degree x)) +
 267       (∑ p : {p : V × V // p.1 ≠ p.2},
 268         2 * Real.exp (-((E.neighborFinset p.1.1 ∩ E.neighborFinset p.1.2).card : ℝ) / 64))) ≤
 269       (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.fairCoin)).probability (fun σ =>
 270         ∀ F : SimpleGraph V, (∀ v, (F.degree v : ℝ) ≤ f) →
 271         ∃ N : V → V → ℕ,
 272           (∀ x y, N x y + N y x = if G.Adj x y then 1 else 0) ∧
 273           (∀ v, ∑ w, N v w = d v) ∧
 274           (∀ x y, N x y ≠ sampledOrientation G σ x y ∨
 275             N y x ≠ sampledOrientation G σ y x → E.Adj x y ∧ ¬ F.Adj x y) ∧
 276           ∀ x, ((Finset.univ.filter (fun y => N x y ≠ sampledOrientation G σ x y ∨
 277             N y x ≠ sampledOrientation G σ y x)).card : ℝ) ≤ a + 2 * L) := by
 278   apply (sampledOrientation_common_event G E hEG ha).trans
 279   apply finiteLaw_probability_mono
 280   intro σ hσ F hF
 281   apply orientationRouting G E F (sampledOrientation G σ) d L hL hn ha hf hβ
 282     (sampledOrientation_physical G σ) (halfDegrees_total G d hd)
 283   · intro v
 284     have hdreal : (d v : ℝ) = (G.degree v : ℝ) / 2 := by
 285       have h := hd v
 286       have hr : (2 : ℝ) * d v = G.degree v := by exact_mod_cast h
 287       linarith
 288     rw [hdreal]
 289     exact (hσ.1 v).le
 290   · exact hF
 291   · intro x y hxy
 292     have hh := hσ.2 x y hxy
 293     have hc := hcommon x y hxy
 294     rw [eligibleTwoPathCount_as_filter] at hh
 295     linarith
 296   · exact hmargin
