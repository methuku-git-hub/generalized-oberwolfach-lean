   8     (L h:ℕ) (hL:0<L) (hlarge:L≤h):
   9     h≤2*L*(h/L) := by
  10   have hq:1≤h/L:=(Nat.le_div_iff_mul_le hL).2 (by simpa using hlarge)
  11   have hqL:L≤L*(h/L):=by
  12     simpa using Nat.mul_le_mul_left L hq
  13   have hmod:h%L<L:=Nat.mod_lt h hL
  14   have hsum:=Nat.mod_add_div h L

  17 theorem long_reserved_total_block_capacity
  18     {I:Type*} [Fintype I] (L:ℕ) (hL:0<L) (length:I→ℕ)
  19     (hlarge:∀i,L≤length i) (requests:ℕ)
  20     (hrequested:2*L*requests≤∑i,length i):
  21     requests≤∑i,length i/L := by
  22   have hsum:(∑i,length i)≤∑i,2*L*(length i/L):=by
  23     apply Finset.sum_le_sum
  24     intro i hi
  25     exact long_reserved_block_floor_lower L (length i) hL (hlarge i)
  26   rw [← Finset.mul_sum] at hsum
  27   have hp:0<2*L:=by omega
  28   exact Nat.le_of_mul_le_mul_left (hrequested.trans hsum) hp
