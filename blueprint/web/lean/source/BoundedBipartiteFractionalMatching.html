  16 theorem bounded_bipartite_fractional_matching
  17     {V : Type*} [Fintype V] [DecidableEq V]
  18     (M : V → V → ℕ) (hM : ∀ x y, M x y ≤ 1)
  19     (β : ℝ) (hβ : 0 < β) (hβhalf : β < 1/2)
  20     (hn : 0 < Fintype.card V)
  21     (hrow : ∀ x, (1/2+β)*Fintype.card V ≤ ((∑ y, M x y : ℕ) : ℝ))
  22     (hcol : ∀ y, (1/2+β)*Fintype.card V ≤ ((∑ x, M x y : ℕ) : ℝ)) :
  23     ∃ w : BipartiteEdges M → ℝ,
  24       (∀ e, β/(5*Fintype.card V) ≤ w e ∧ w e ≤ 5/(β*Fintype.card V)) ∧
  25       (∀ x, (∑ e, if e.1=x then w e else 0)=1) ∧
  26       (∀ y, (∑ e, if e.2.1=y then w e else 0)=1) := by
  27   let n : ℝ := Fintype.card V
  28   have hnR : 0<n := Nat.cast_pos.mpr hn
  29   let a : ℝ := β/(5*n)
  30   let p : ℝ := 4/(β*n)
  31   have ha : 0<a := div_pos hβ (by positivity)
  32   have hp : 0<p := div_pos (by norm_num) (mul_pos hβ hnR)
  33   have han : a*n=β/5 := by dsimp [a];field_simp <;> ring
  34   have hpn : p*(β*n)=4 := by dsimp [p];field_simp
  35   let f (x : V) : ℝ := 1-a*((∑ y, M x y : ℕ) : ℝ)
  36   let g (y : V) : ℝ := 1-a*((∑ x, M x y : ℕ) : ℝ)
  37   have hdegree (N : V → V → ℕ) (hN : ∀ x y, N x y ≤ 1) (x : V) :
  38       ((∑ y, N x y : ℕ) : ℝ) ≤ n := by
  39     dsimp [n]
  40     exact_mod_cast (show (∑ y, N x y) ≤ Fintype.card V from
  41       (Finset.sum_le_sum (fun y _ => hN x y)).trans_eq (by simp))
  42   have hf (x : V) : 1-β/5 ≤ f x ∧ f x ≤ 1 := by
  43     have hd := mul_le_mul_of_nonneg_left (hdegree M hM x) ha.le
  44     have hd0 : 0≤a*((∑ y, M x y : ℕ) : ℝ) := by positivity
  45     dsimp [f]; constructor <;> linarith
  46   have hg (y : V) : 1-β/5 ≤ g y ∧ g y ≤ 1 := by
  47     have hd := mul_le_mul_of_nonneg_left
  48       (hdegree (fun y x => M x y) (fun y x => hM x y) y) ha.le
  49     have hd0 : 0≤a*((∑ x, M x y : ℕ) : ℝ) := by positivity
  50     dsimp [g]; constructor <;> linarith
  51   have htotal : ∑ x, f x = ∑ y, g y := by
  52     simp only [f,g,Finset.sum_sub_distrib,← Finset.mul_sum,← Nat.cast_sum]
  53     rw [Finset.sum_comm (f := M)]
  54   let T : ℝ := ∑ x, f x
  55   have hTlo : n*(1-β/5)≤T := by
  56     simpa only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul] using
  57       Finset.sum_le_sum (fun x (_ : x∈Finset.univ) => (hf x).1)
  58   have hThi : T≤n := by
  59     simpa only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul,mul_one] using
  60       Finset.sum_le_sum (fun x (_ : x∈Finset.univ) => (hf x).2)
  61   have hT : 0≤T := le_trans (mul_nonneg hnR.le (by linarith : 0≤1-β/5)) hTlo
  62   have hrel : n ≤ (1+β/4)*T := by
  63     have he : 1≤(1+β/4)*(1-β/5) := by nlinarith
  64     have hh := mul_le_mul_of_nonneg_left hTlo (by linarith : 0≤1+β/4)
  65     nlinarith [mul_le_mul_of_nonneg_left he hnR.le]
  66   have hcap : T/p ≤ (3*β/4)*n*n/2 := by
  67     apply (div_le_iff₀ hp).mpr
  68     have hh : ((3*β/4)*n*n/2)*p=3*n/2 := by dsimp [p];field_simp <;> ring
  69     rw [hh]
  70     linarith
  71   have hupper (z : ℝ) (hz : z≤1) : z/p ≤ (1+β/4)*(T/p)/n := by
  72     have hz' : z*n≤n := mul_le_of_le_one_left hnR.le hz
  73     apply (le_div_iff₀ hnR).mpr
  74     rw [div_mul_eq_mul_div, mul_div_assoc]
  75     simpa only [mul_div_assoc] using div_le_div_of_nonneg_right (hz'.trans hrel) hp.le
  76   obtain ⟨q,hq,hl,hr⟩ := bipartite_nearUniformFractionalDegrees
  77     (fun e : BipartiteEdges M => e.1) (fun e => e.2.1)
  78     (fun x => f x/p) (fun y => g y/p)
  79     (δ := β/4) (δ₀ := β/4) (M := T/p) (h := 3*β/4)
  80     hn hn (by positivity) le_rfl (by linarith) (div_nonneg hT hp.le)
  81     hcap (by rw [← Finset.sum_div]) (by rw [← Finset.sum_div,← htotal])
  82     (fun x => div_nonneg (by linarith [(hf x).1]) hp.le)
  83     (fun y => div_nonneg (by linarith [(hg y).1]) hp.le)
  84     (fun x => hupper _ (hf x).2) (fun y => hupper _ (hg y).2)
  85     (fun S U hSU => matrixEdges_normalized_large_cuts M hM β (β/4) (3*β/4)
  86       hn (by linarith) hrow hcol S U hSU)
  87   refine ⟨fun e => a+p*q e, ?_, ?_, ?_⟩
  88   · intro e
  89     have hqe := hq e
  90     constructor
  91     · change a≤a+p*q e
  92       exact le_add_of_nonneg_right (mul_nonneg hp.le hqe.1)
  93     · have hb : a+p ≤ 5/(β*n) := by
  94         dsimp [a,p]
  95         apply (le_div_iff₀ (mul_pos hβ hnR)).mpr
  96         have he : (β/(5*n)+4/(β*n))*(β*n)=β^2/5+4 := by field_simp <;> ring
  97         rw [he]
  98         nlinarith
  99       have hh := mul_le_of_le_one_right hp.le hqe.2
 100       change a+p*q e ≤ 5/(β*n)
 101       linarith
 102   · intro x
 103     have hsum : (∑ e : BipartiteEdges M, if e.1=x then a+p*q e else 0)=
 104         a*((∑ y, M x y : ℕ) : ℝ)+p*(f x/p) := by
 105       rw [← hl x,Finset.mul_sum]
 106       simp only [mul_ite,mul_zero]
 107       have hc : (∑ e : BipartiteEdges M, if e.1=x then a else 0)=
 108           a*((∑ y, M x y : ℕ) : ℝ) := by
 109         have hh := congrArg (fun z : ℕ => a*(z : ℝ)) (bipartiteEdges_row_degree M x)
 110         simpa only [Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero,
 111           Finset.mul_sum,mul_ite,mul_one,mul_zero] using hh
 112       rw [← hc,← Finset.sum_add_distrib]
 113       apply Finset.sum_congr rfl
 114       intro e _
 115       split_ifs <;> ring
 116     rw [hsum,mul_div_cancel₀ _ hp.ne']
 117     dsimp [f];ring
 118   · intro y
 119     have hsum : (∑ e : BipartiteEdges M, if e.2.1=y then a+p*q e else 0)=
 120         a*((∑ x, M x y : ℕ) : ℝ)+p*(g y/p) := by
 121       rw [← hr y,Finset.mul_sum]
 122       simp only [mul_ite,mul_zero]
 123       have hc : (∑ e : BipartiteEdges M, if e.2.1=y then a else 0)=
 124           a*((∑ x, M x y : ℕ) : ℝ) := by
 125         have hh := congrArg (fun z : ℕ => a*(z : ℝ)) (bipartiteEdges_column_degree M y)
 126         simpa only [Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero,
 127           Finset.mul_sum,mul_ite,mul_one,mul_zero] using hh
 128       rw [← hc,← Finset.sum_add_distrib]
 129       apply Finset.sum_congr rfl
 130       intro e _
 131       split_ifs <;> ring
 132     rw [hsum,mul_div_cancel₀ _ hp.ne']
 133     dsimp [g];ring
