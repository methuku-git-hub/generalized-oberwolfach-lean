  13 theorem finiteConjunction_sum_change {X I : Type*} (A : Finset X) (J : Finset I)
  14     (F : X→Prop) (P Q : I→X→Prop) :
  15     |(∑ x∈A,if F x ∧ ∀ j∈J,Q j x then (1:ℝ) else 0)-
  16       (∑ x∈A,if F x ∧ ∀ j∈J,P j x then (1:ℝ) else 0)|≤
  17       ∑ j∈J,∑ x∈A,if ¬(P j x ↔ Q j x) then (1:ℝ) else 0 := by
  18   classical
  19   have hp (x : X) :
  20       |(if F x ∧ ∀ j∈J,Q j x then (1:ℝ) else 0)-
  21         (if F x ∧ ∀ j∈J,P j x then (1:ℝ) else 0)|≤
  22         ∑ j∈J,if ¬(P j x ↔ Q j x) then (1:ℝ) else 0 := by
  23     by_cases he : ∃ j∈J,¬(P j x ↔ Q j x)
  24     · obtain ⟨j,hj,hchange⟩ := he
  25       have hsum := Finset.single_le_sum (s:=J)
  26         (f:=fun j=>if ¬(P j x ↔ Q j x) then (1:ℝ) else 0)
  27         (fun _ _=>by split_ifs <;> norm_num) hj
  28       rw [if_pos hchange] at hsum
  29       apply le_trans (b:=1) ?_ hsum
  30       split_ifs <;> norm_num
  31     · have heq : (F x ∧ ∀ j∈J,P j x) ↔ (F x ∧ ∀ j∈J,Q j x) := by
  32         have hj : ∀ j∈J,P j x ↔ Q j x := by
  33           intro j hj
  34           by_contra hh
  35           exact he ⟨j,hj,hh⟩
  36         exact and_congr_right (fun _=>forall_congr' (fun j=>forall_congr' (fun h=>hj j h)))
  37       simp only [heq,sub_self,abs_zero]
  38       exact Finset.sum_nonneg (fun _ _=>by split_ifs <;> norm_num)
  39 
  40   calc
  41     _ ≤ ∑ x∈A,|(if F x ∧ ∀ j∈J,Q j x then (1:ℝ) else 0)-
  42           (if F x ∧ ∀ j∈J,P j x then (1:ℝ) else 0)| := by
  43       rw [← Finset.sum_sub_distrib]
  44       exact Finset.abs_sum_le_sum_abs _ _
  45     _ ≤ ∑ x∈A,∑ j∈J,if ¬(P j x ↔ Q j x) then (1:ℝ) else 0 :=
  46       Finset.sum_le_sum (fun x _=>hp x)
  47     _ = _ := Finset.sum_comm

  49 theorem finiteConjunction_sum_change_at_outer {X I : Type*} (A C : Finset X) (hCA : C⊆A)
  50     (J : Finset I) (F : X→Prop) (P Q : I→X→Prop) (δ : ℝ)
  51     (h : ∀ j∈J,(∑ x∈A,if ¬(P j x ↔ Q j x) then (1:ℝ) else 0)≤δ) :
  52     |(∑ x∈C,if F x ∧ ∀ j∈J,Q j x then (1:ℝ) else 0)-
  53       (∑ x∈C,if F x ∧ ∀ j∈J,P j x then (1:ℝ) else 0)|≤J.card*δ := by
  54   apply (finiteConjunction_sum_change C J F P Q).trans
  55   calc
  56     _ ≤ ∑ j∈J,δ := Finset.sum_le_sum (fun j hj=>
  57       (Finset.sum_le_sum_of_subset_of_nonneg hCA (by intro x hx hn;split_ifs <;> norm_num)).trans (h j hj))
  58     _ = _ := by simp only [Finset.sum_const,nsmul_eq_mul]
