  11  theorem finite_bijection_preserving_predicate {X Y:Type*} [Fintype X] [Fintype Y]
  12     (P:X→Prop) (Q:Y→Prop) [DecidablePred P] [DecidablePred Q]
  13     (hcard:Fintype.card X=Fintype.card Y)
  14     (hclass:Fintype.card {x // P x}=Fintype.card {y // Q y}):
  15     ∃ e:X≃Y,∀ x,Q (e x)↔P x:=by
  16   let e₁:{x // P x}≃{y // Q y}:=Fintype.equivOfCardEq hclass
  17   let e₂:{x // ¬P x}≃{y // ¬Q y}:=Fintype.equivOfCardEq
  18     (by rw [Fintype.card_subtype_compl,Fintype.card_subtype_compl,hcard,hclass])
  19   let e: X≃Y:=(Equiv.sumCompl P).symm.trans ((Equiv.sumCongr e₁ e₂).trans (Equiv.sumCompl Q))
  20   refine ⟨e,?_⟩
  21   intro x
  22   by_cases hx:P x
  23   · have he:Q (e x):=by
  24       change Q ((Equiv.sumCompl Q) ((Equiv.sumCongr e₁ e₂) ((Equiv.sumCompl P).symm x)))
  25       rw [Equiv.sumCompl_symm_apply_of_pos hx]
  26       exact (e₁ ⟨x,hx⟩).property
  27     exact ⟨fun _=>hx,fun _=>he⟩
  28   · have he:¬Q (e x):=by
  29       change ¬Q ((Equiv.sumCompl Q) ((Equiv.sumCongr e₁ e₂) ((Equiv.sumCompl P).symm x)))
  30       rw [Equiv.sumCompl_symm_apply_of_neg hx]
  31       exact (e₂ ⟨x,hx⟩).property
  32     exact ⟨fun h=>(he h).elim,fun h=>(hx h).elim⟩
