  22       (∀ i,(Fintype.card (V i):ℝ)=(a i:ℝ)*k) →
  23       ∀ (R : C → Finset (Σ i,V i)) (w : C → ℝ),
  24       (∀ C₀ i,(Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈R C₀)).card=a i) →
  25       (∀ Y : Finset (Σ i,V i),Fintype.card {C₀ : C // R C₀=Y} ≤ J) →
  26       (∀ C₀,b/k^(c-1) ≤ w C₀ ∧ w C₀ ≤ B/k^(c-1)) →
  27       (∀ i v,|1-hypergraphWeightedDegree R w ⟨i,v⟩| ≤ η) →
  28       (∀ σ : ∀ i,FixedSubset (V i) (a i),
  29         ∃ A : Finset (LabelledAbsorberFamily R S (profileRootSet σ)),Acoef*k^(c*c) ≤ A.card) →
  30       ∃ ψ : C → ℝ,(∀ v,hypergraphWeightedDegree R ψ v=1) ∧
  31         ∀ C₀,(b/2)/k^(c-1) ≤ ψ C₀ ∧ ψ C₀ ≤ (B+b/2)/k^(c-1) := by
  32   let K := (absorberTradeCountConstant c J (Fintype.card T):ℝ)*(c:ℝ)^(c*c)/Acoef

  35   have hD : 0 ≤ D := by dsimp [D];positivity

  48     rw [sigma_finset_card];simp only [hR];exact hsize

  50     simp only [Fintype.card_sigma,Nat.cast_sum,hn,← Finset.sum_mul]
  51     have he : (∑ i,(a i:ℝ))=c := by exact_mod_cast hsize

  55         if v∈profileRootSet σ then 1 else 0 := by

  62         (if (R C₀∩profileRootSet σ).Nonempty then K/k^(c-1) else 0) :=
  63     labelled_family_trade_spread hc R S (profileRootSet σ) (by rw [profileRootSet_card,hsize])
