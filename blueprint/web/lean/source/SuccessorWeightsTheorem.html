  13 theorem successorCycleRatio_standard (k : ℕ) :
  14     successorCycleRatio k = (((k+2)*2 : ℕ) : ℝ) / (2*(((k+2)*2 : ℕ) : ℝ)-2) := by
  15   have hd : (2*(k+2 : ℝ)-1) ≠ 0 := by linarith [Nat.cast_nonneg (α := ℝ) k]
  16   have hd' : (2*(((k+2)*2 : ℕ) : ℝ)-2) ≠ 0 := by push_cast; nlinarith [Nat.cast_nonneg (α := ℝ) k]
  17   unfold successorCycleRatio
  18   push_cast
  19   field_simp

  21 theorem successor_weights_theorem (c : ℕ) (hc : 4 ≤ c) (heven : Even c)
  22     (β : ℝ) (hβ : 0 < β) :
  23     ∃ A B : ℝ, 0 < A ∧ 0 < B ∧ ∀ τ : ℝ, 0 < τ →
  24       ∃ ε₀ : ℝ, 0 < ε₀ ∧ ∀ b : ℕ, 2 ≤ b → ∃ N : ℕ,
  25       ∀ (V : Type) [Fintype V], N ≤ Fintype.card V →
  26       ∀ (G : SimpleGraph V) (π : Equiv.Perm V) (D : ℝ),
  27       (Fintype.card V : ℝ)/2 ≤ D → D ≤ 2*Fintype.card V →
  28       (∀ v, |(G.degree v : ℝ)-D| ≤ ε₀*Fintype.card V) →
  29       (∀ v, ((c : ℝ)/(2*c-2)+β)*Fintype.card V ≤ (G.degree v : ℝ)) →
  30       ∃ w : (Fin c → V) → ℝ, SuccessorSequenceWeightConditions G π b D τ A B w := by
  31   obtain ⟨k,hk⟩ : ∃ k : ℕ, c=(k+2)*2 := by
  32     obtain ⟨r,hr⟩ := heven
  33     refine ⟨r-2,?_⟩
  34     omega
  35   subst c
  36   obtain ⟨hA,hB⟩ := successorWeightConstants_pos k β hβ
  37   refine ⟨successorWeightLower k β,successorWeightUpper k β,hA,hB,?_⟩
  38   intro τ hτ
  39   obtain ⟨ε₀,hε₀,hall⟩ := successor_sequence_weight_existence k β hβ τ hτ
  40   refine ⟨ε₀,hε₀,?_⟩
  41   intro b _
  42   obtain ⟨N,hN⟩ := hall b
  43   refine ⟨max N 1,?_⟩
  44   intro V _ hn G π D hD hDup hnear hmin
  45   have hv : 0 < Fintype.card V := by omega
  46   letI : Nonempty V := Fintype.card_pos_iff.mp hv
  47   exact hN V ((le_max_left _ _).trans hn) G π D hD hDup hnear
  48     (by simpa only [successorCycleRatio_standard] using hmin)
