   1 import Oberwolfach.ResidualPathPiecesWithArcs
   2 /-! Represent a retained part of an ordered source cycle by directed source
   3 pieces: the cycle itself if intact, and directed paths if cut. -/
   4 set_option autoImplicit false
   5 noncomputable section
   6 namespace Oberwolfach
   7 attribute [local instance] Classical.propDecidable
   8 variable {V:Type} [Fintype V] [DecidableEq V]
   9 
  10 theorem retained_cycle_source_pieces_with_arcs (H:SimpleGraph V) (R:Finset V)
  11     (hisol:∀x,x∉R→∀y,¬H.Adj x y) (D:V→V→Prop)
  12     (a:V) (xs:List V) (hn:(a::xs).Nodup) (hlen:3≤(a::xs).length)
  13     (hchain:List.IsChain D ((a::xs)++[a])):
  14     ∃ ps:List (PrescribedSourcePiece V),
  15       ps.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset) ∧
  16       (∀p∈ps,p.OriginalValid D) ∧
  17       (∀x,x∈sourcePieceVerticesUnion ps ↔ x∈a::xs ∧ x∈R) ∧
  18       sourcePieceEdgesUnion ps=(prescribedOrderedPathEdges ((a::xs)++[a])).filter (fun e=>e∈H.edgeFinset) ∧
  19       sourcePieceArcsUnion ps=(prescribedDirectedPathEdges ((a::xs)++[a])).filter (fun e=>H.Adj e.1 e.2) := by
  20   by_cases hkeep:∀e∈prescribedOrderedPathEdges ((a::xs)++[a]),e∈H.edgeFinset
  21   · have hsurv: (a::xs).toFinset⊆R := intact_cycle_vertices_survive H R hisol a xs (by omega) hkeep
  22     refine ⟨[.cycle a xs],by simp,?_,?_,?_,?_⟩
  23     · intro p hp
  24       have hp':p=.cycle a xs := by simpa using hp
  25       subst p
  26       exact ⟨hn,hlen,hchain⟩
  27     · intro x
  28       have heq:x∈sourcePieceVerticesUnion [.cycle a xs] ↔ x∈a::xs := by
  29         simp [sourcePieceVerticesUnion,PrescribedSourcePiece.vertices]
  30       rw [heq]
  31       constructor
  32       · intro hx
  33         exact ⟨hx,hsurv (List.mem_toFinset.mpr hx)⟩
  34       · exact And.left
  35     · have heq:sourcePieceEdgesUnion [.cycle a xs]=prescribedOrderedPathEdges ((a::xs)++[a]) := by
  36         simp [sourcePieceEdgesUnion,PrescribedSourcePiece.edges]
  37       rw [heq]
  38       exact (Finset.filter_true_of_mem hkeep).symm
  39     · have heq:sourcePieceArcsUnion [.cycle a xs]=prescribedDirectedPathEdges ((a::xs)++[a]) := by
  40         simp [sourcePieceArcsUnion,PrescribedSourcePiece.arcs]
  41       rw [heq]
  42       apply (Finset.filter_true_of_mem ?_).symm
  43       intro f hf
  44       rcases f with ⟨x,y⟩
  45       have hphys:=directed_path_edge_is_physical ((a::xs)++[a]) x y hf
  46       have hh:=hkeep s(x,y) hphys
  47       simpa only [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using hh
  48   · push_neg at hkeep
  49     obtain ⟨e,he,hmissing⟩:=hkeep
  50     obtain ⟨blocks,hpair,hblocks,hvertices,hedges,harcs⟩:=cut_cycle_into_retained_paths_with_arcs H R hisol D a xs hn hchain e he hmissing
  51     obtain ⟨ps,hp,hvalid,hpsverts,hpsedges,hpsarcs⟩:=path_blocks_as_source_pieces_with_arcs D blocks hpair
  52       (by intro block hb;exact ⟨(hblocks block hb).1,(hblocks block hb).2.2⟩)
  53     refine ⟨ps,hp,hvalid,?_,?_,?_⟩
  54     · intro x
  55       exact (hpsverts x).trans (hvertices x)
  56     · exact hpsedges.trans hedges
  57     · exact hpsarcs.trans harcs
  58 end Oberwolfach
