  14 theorem disjointPattern_influence_squares
  15     {E I : Type*} [Fintype E] [DecidableEq E]
  16     (C : Finset I) (F : I → Finset E)
  17     (hdisj : ∀ i ∈ C, ∀ j ∈ C, i ≠ j → Disjoint (F i) (F j)) :
  18     (∑ e : E, (∑ i ∈ C, if e ∈ F i then (1 : ℝ) else 0) ^ 2) =
  19       ∑ i ∈ C, ((F i).card : ℝ) := by
  20   classical
  21   have hsmall (e : E) : (C.filter (fun i => e ∈ F i)).card ≤ 1 := by
  22     apply Finset.card_le_one.mpr
  23     intro i hi j hj
  24     by_contra hij
  25     exact Finset.disjoint_left.mp
  26       (hdisj i (Finset.mem_filter.mp hi).1 j (Finset.mem_filter.mp hj).1 hij)
  27       (Finset.mem_filter.mp hi).2 (Finset.mem_filter.mp hj).2
  28   have hsum (e : E) : (∑ i ∈ C, if e ∈ F i then (1 : ℝ) else 0) =
  29       ((C.filter (fun i => e ∈ F i)).card : ℝ) := by
  30     simp only [Finset.card_filter, Nat.cast_sum, Nat.cast_ite, Nat.cast_one, Nat.cast_zero]
  31   have hsq (e : E) : (∑ i ∈ C, if e ∈ F i then (1 : ℝ) else 0) ^ 2 =
  32       ∑ i ∈ C, if e ∈ F i then (1 : ℝ) else 0 := by
  33     simp_rw [hsum]
  34     have hh := hsmall e
  35     have he : (C.filter (fun i => e ∈ F i)).card = 0 ∨
  36         (C.filter (fun i => e ∈ F i)).card = 1 := by omega
  37     rcases he with he | he <;> simp [he]
  38   simp_rw [hsq]
  39   rw [Finset.sum_comm]
  40   apply Finset.sum_congr rfl
  41   intro i _
  42   simp

  44 theorem finiteLaw_disjoint_patterns_tail
  45     {E I : Type*} [Fintype E] [DecidableEq E]
  46     {State : E → Type*} [∀ e, Fintype (State e)]
  47     (μ : ∀ e, FiniteLaw (State e))
  48     (C : Finset I) (F : I → Finset E) (s : I → ∀ e, State e)
  49     (hdisj : ∀ i ∈ C, ∀ j ∈ C, i ≠ j → Disjoint (F i) (F j))
  50     {t : ℝ} (ht : 0 ≤ t) :
  51     (FiniteLaw.pi μ).probability (fun σ =>
  52       t ≤ |(∑ i ∈ C, patternValue (F i) (s i) σ) -
  53         (∑ i ∈ C, ∏ e ∈ F i, (μ e).mass (s i e))|) ≤
  54       2 * Real.exp (-2 * t ^ 2 / (∑ i ∈ C, ((F i).card : ℝ))) := by
  55   classical
  56   have hh := finiteLaw_omitted_weighted_tail μ C F ∅ s (fun _ => 1)
  57     (fun e => ∑ i ∈ C, if e ∈ F i then (1 : ℝ) else 0)
  58     (fun _ _ => zero_le_one) (fun _ => by simp) ht
  59   simpa only [Finset.sdiff_empty, one_mul, disjointPattern_influence_squares C F hdisj] using hh

  61 theorem finiteLaw_fair_disjoint_patterns_tail
  62     {E I : Type*} [Fintype E] [DecidableEq E]
  63     (C : Finset I) (F : I → Finset E) (s : I → E → Bool) (k : ℕ)
  64     (hsize : ∀ i ∈ C, (F i).card = k)
  65     (hdisj : ∀ i ∈ C, ∀ j ∈ C, i ≠ j → Disjoint (F i) (F j))
  66     {t : ℝ} (ht : 0 ≤ t) :
  67     (FiniteLaw.pi (fun _ : E => FiniteLaw.fairCoin)).probability (fun σ =>
  68       t ≤ |(∑ i ∈ C, patternValue (F i) (s i) σ) -
  69         (C.card : ℝ) * (1 / 2 : ℝ) ^ k|) ≤
  70       2 * Real.exp (-2 * t ^ 2 / ((C.card : ℝ) * k)) := by
  71   have hh := finiteLaw_disjoint_patterns_tail
  72     (fun _ : E => FiniteLaw.fairCoin) C F s hdisj ht
  73   have hm : (∑ i ∈ C, ∏ e ∈ F i, (FiniteLaw.fairCoin).mass (s i e)) =
  74       (C.card : ℝ) * (1 / 2 : ℝ) ^ k := by
  75     simp only [FiniteLaw.fairCoin, Finset.prod_const]
  76     simp only [Finset.sum_congr rfl (fun i hi => congrArg (fun j => (1 / 2 : ℝ) ^ j) (hsize i hi)),
  77       Finset.sum_const, nsmul_eq_mul]
  78   have hv : (∑ i ∈ C, ((F i).card : ℝ)) = (C.card : ℝ) * k := by
  79     simp_rw [Finset.sum_congr rfl (fun i hi => congrArg (fun j : ℕ => (j : ℝ)) (hsize i hi))]
  80     simp
