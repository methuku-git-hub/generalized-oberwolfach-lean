   3 /-! The exact normalization used in the successor weighting. The tuple
   4 placement probability cancels the powers of the class size in the lower
   5 bound, so that its constant is independent of the approximation error. -/

  11 def successorPairProbability (m n : ℝ) : ℝ := m ^ 2 / (n * (n - 1))

  13 theorem successorPairProbability_pos {m n : ℝ} (hm : 0 < m) (hn : 1 < n) :
  14     0 < successorPairProbability m n := by
  15   unfold successorPairProbability
  16   positivity

  18 theorem successor_tuple_normalization {c : ℕ} (hc : 2 ≤ c)
  19     (a m n : ℝ) (hm : m ≠ 0) (hn : n ≠ 0) :
  20     (a / m ^ (c - 2)) * (m / n) ^ c / successorPairProbability m n =
  21       (a / n ^ (c - 2)) * ((n - 1) / n) := by
  22   have hmc : m ^ c = m ^ (c - 2) * m ^ 2 := by rw [← pow_add]; congr 1; omega
  23   have hnc : n ^ c = n ^ (c - 2) * n ^ 2 := by rw [← pow_add]; congr 1; omega
  24   unfold successorPairProbability
  25   rw [div_pow, hmc, hnc]
  26   field_simp
  27 
  28 theorem successor_tuple_normalization_lower {c : ℕ} (hc : 2 ≤ c)
  29     (a m n : ℝ) (ha : 0 ≤ a) (hm : 0 < m) (hn : 2 ≤ n) :
  30     a / (2 * n ^ (c - 2)) ≤
  31       (a / m ^ (c - 2)) * (m / n) ^ c / successorPairProbability m n := by
  32   have hn0 : 0 < n := by linarith
  33   rw [successor_tuple_normalization hc a m n hm.ne' hn0.ne']
  34   have hratio : (1 / 2 : ℝ) ≤ (n - 1) / n := by
  35     apply (le_div_iff₀ hn0).mpr
  36     linarith
  37   have h := mul_le_mul_of_nonneg_left hratio (div_nonneg ha (pow_pos hn0 (c - 2)).le)
  38   convert h using 1 <;> ring

  40 theorem successor_tuple_normalization_upper {c : ℕ} (hc : 2 ≤ c)
  41     (C m n K : ℝ) (hC : 0 ≤ C) (hm : 0 < m) (hn : 1 < n)
  42     (hK : n ≤ K * m) :
  43     (C / m ^ (c - 2)) / successorPairProbability m n ≤
  44       C * K ^ c / n ^ (c - 2) := by
  45   have hn0 : 0 < n := by linarith
  46   have hmc : m ^ c = m ^ (c - 2) * m ^ 2 := by rw [← pow_add]; congr 1; omega
  47   have hnc : n ^ c = n ^ (c - 2) * n ^ 2 := by rw [← pow_add]; congr 1; omega
  48   have hid : (C / m ^ (c - 2)) / successorPairProbability m n =
  49       (C * (n / m) ^ c / n ^ (c - 2)) * ((n - 1) / n) := by
  50     unfold successorPairProbability
  51     rw [div_pow, hnc, hmc]
  52     field_simp
  53   rw [hid]
  54   have hratio : 0 ≤ (n - 1) / n ∧ (n - 1) / n ≤ 1 := by
  55     constructor
  56     · positivity
  57     · apply (div_le_one hn0).mpr; linarith
  58   have hscale : n / m ≤ K := (div_le_iff₀ hm).mpr hK
  59   have hp : (n / m) ^ c ≤ K ^ c := pow_le_pow_left₀ (by positivity) hscale _
  60   calc
  61     _ ≤ C * (n / m) ^ c / n ^ (c - 2) :=
  62       mul_le_of_le_one_right (by positivity) hratio.2
  63     _ ≤ _ := div_le_div_of_nonneg_right (mul_le_mul_of_nonneg_left hp hC) (by positivity)
