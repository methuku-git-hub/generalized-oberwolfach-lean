  21 theorem monochromaticWeightedSum_expectation (R : E → Finset V) (w : E → ℝ)
  22     (k : ℕ) (hR : ∀ e,(R e).card=k) (c : C) :
  23     (FiniteLaw.pi (fun _ : V=>FiniteLaw.uniform C)).expectation (monochromaticWeightedSum R w c)=
  24       (∑ e,w e)/(Fintype.card C:ℝ)^k := by

  26   have hh := finiteLaw_expectation_omitted_root_sum (fun _ : V=>FiniteLaw.uniform C)
  27     Finset.univ R ∅ (fun _ _=>c) w

  29     ← Finset.sum_mul,mul_one_div,← Finset.sum_div,monochromaticWeightedSum] using hh

  31 theorem monochromaticWeightedSum_tail (R : E → Finset V) (w : E → ℝ)
  32     (hw : ∀ e,0≤w e) (k : ℕ) (hk : 0<k) (hR : ∀ e,(R e).card=k)
  33     (hW : 0<∑ e,w e) (D : ℝ) (hD : ∀ v,hypergraphWeightedDegree R w v≤D)
  34     (c : C) (t : ℝ) (ht : 0≤t) :
  35     (FiniteLaw.pi (fun _ : V=>FiniteLaw.uniform C)).probability
  36       (fun σ=>t≤|monochromaticWeightedSum R w c σ-(∑ e,w e)/(Fintype.card C:ℝ)^k|)≤
  37       2*Real.exp (-2*t^2/(D*k*∑ e,w e)) := by
  38   have hh := finiteLaw_omitted_weighted_tail (fun _ : V=>FiniteLaw.uniform C)
  39     Finset.univ R ∅ (fun _ _=>c) w (hypergraphWeightedDegree R w)
  40     (fun e _=>hw e) (by intro v;simp only [Finset.sdiff_empty];exact le_refl _) ht
  41   have hb : (FiniteLaw.pi (fun _ : V=>FiniteLaw.uniform C)).probability
  42       (fun σ=>t≤|monochromaticWeightedSum R w c σ-(∑ e,w e)/(Fintype.card C:ℝ)^k|)≤
  43       2*Real.exp (-2*t^2/(∑ v,(hypergraphWeightedDegree R w v)^2)) := by
  44     simpa only [Finset.sdiff_empty,FiniteLaw.uniform,Finset.prod_const,hR,one_div_pow,
  45       ← Finset.sum_mul,mul_one_div,← Finset.sum_div,monochromaticWeightedSum] using hh
  46   have hpos := hypergraphWeightedDegree_square_sum_pos R w hw k hk hR hW
  47   have hsq := hypergraphWeightedDegree_square_sum R w hw k hR D hD
  48   apply hb.trans
