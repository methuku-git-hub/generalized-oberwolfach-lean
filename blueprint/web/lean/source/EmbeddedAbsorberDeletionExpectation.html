  22 theorem embedded_absorber_deletion_expectation
  23     (e : (Σ i,W i) ↪ (Σ i,V i)) (he : ∀ v,(e v).1=v.1)
  24     (R : C → Finset (Σ i,V i)) (S : T → VertexAbsorberTemplate c) (a : I → ℕ)
  25     (hR : ∀ C₀ i,(Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈R C₀)).card=a i)
  26     (X Z : Finset (Σ i,V i)) (hrange : ∀ v,(∃ u,e u=v) ↔ v∉Z) (m : I → ℕ)
  27     [∀ i,Nonempty (FixedSubset (W i) (m i))]
  28     (hm : ∀ i,m i ≤ Fintype.card (W i)) :
  29     (FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (W i) (m i)))).expectation
  30       (fun σ => absorberGroundCount R S X ((profileRootSet σ).map e))=
  31       (∏ i,((m i).choose (c*a i):ℝ)/(Fintype.card (W i)).choose (c*a i))*
  32         Fintype.card {D : LabelledAbsorberFamily R S X // Disjoint (labelledAbsorberFree R (S D.1) X D.2) Z} := by
  33   let D := LabelledAbsorberFamily R S X
  34   let F := fun d : D => labelledAbsorberFree R (S d.1) X d.2
  35   let w := fun d : D => if Disjoint (F d) Z then (1:ℝ) else 0
  36   have hw (d : D) (hn : w d≠0) : Disjoint (F d) Z := by
  37     by_contra hd
  38     simp only [w,hd,ite_false,ne_eq,not_true_eq_false] at hn
  39   have hF : ∀ d : D,w d≠0 → F d⊆Finset.univ.map e := by
  40     intro d hn v hv
  41     have hz : v∉Z := fun hz => Finset.disjoint_left.mp (hw d hn) hv hz
  42     obtain ⟨u,hu⟩ := (hrange v).mpr hz
  43     exact Finset.mem_map.mpr ⟨u,Finset.mem_univ _,hu⟩
  44   have ha : ∀ d : D,w d≠0 → ∀ i,
  45       (Finset.univ.filter (fun u : W i => e ⟨i,u⟩∈F d)).card=c*a i := by
  46     intro d hn i
  47     have hcover : ∀ x∈F d,∃ u,e u=x := by
  48       intro x hx
  49       exact (hrange x).mpr (fun hz => Finset.disjoint_left.mp (hw d hn) hx hz)
  50     exact (profile_embedding_preimage_class_card e he (F d) hcover i).trans
  51       (labelled_absorber_free_class_card R (S d.1) X d.2 a hR i)
  52   have hex := embedded_profile_weighted_support_expectation e F w
  53     (fun i => c*a i) m hF ha hm
  54   have hsum (σ : ∀ i,FixedSubset (W i) (m i)) :
  55       absorberGroundCount R S X ((profileRootSet σ).map e)=
  56         ∑ d : D,if F d⊆(profileRootSet σ).map e then w d else 0 := by
  57     unfold absorberGroundCount
  58     simp only [Fintype.card_subtype]
  59     rw [← Finset.sum_boole (fun d : D => F d⊆(profileRootSet σ).map e) Finset.univ]
  60     apply Finset.sum_congr rfl
  61     intro d _
  62     by_cases hs : F d⊆(profileRootSet σ).map e
  63     · have hd : Disjoint (F d) Z := by
  64         apply Finset.disjoint_left.mpr
  65         intro v hv hz
  66         obtain ⟨u,_,hu⟩ := Finset.mem_map.mp (hs hv)
  67         exact ((hrange v).mp ⟨u,hu⟩) hz
  68       simp only [hs,ite_true,w,hd]
  69     · simp only [hs,ite_false]
  70   have hwcard : (∑ d : D,w d)=(Fintype.card {d : D // Disjoint (F d) Z}:ℝ) := by
  71     simp only [w,Fintype.card_subtype,Finset.sum_boole]
  72   simp_rw [hsum]
  73   rw [hex,hwcard]
