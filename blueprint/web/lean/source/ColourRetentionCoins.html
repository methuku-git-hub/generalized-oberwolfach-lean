  14 def FiniteLaw.withCoin {C : Type*} [Fintype C] (μ : FiniteLaw C)
  15     (a : C → ℝ) (ha : ∀ c,0≤a c ∧ a c≤1) : FiniteLaw (C×Bool) where
  16   mass x := μ.mass x.1*(if x.2 then a x.1 else 1-a x.1)
  17   nonneg x := mul_nonneg (μ.nonneg _) (by
  18     cases x.2 <;> simp only [Bool.false_eq_true,ite_false,ite_true]
  19     · exact sub_nonneg.mpr (ha _).2
  20     · exact (ha _).1)
  21   total := by
  22     rw [Fintype.sum_prod_type]
  23     have he (c : C) : (∑ b : Bool,μ.mass c*(if b then a c else 1-a c))=μ.mass c := by
  24       rw [Fintype.sum_bool]
  25       simp only [Bool.false_eq_true,ite_false,ite_true]
  26       ring
  27     simp_rw [he]
  28     exact μ.total

  31     (μ : FiniteLaw C) (a : C → ℝ) (ha : ∀ c,0≤a c ∧ a c≤1) (c : C) :
  32     (μ.withCoin a ha).probability (fun x=>x.1=c)=μ.mass c := by
  33   classical
  34   unfold FiniteLaw.probability FiniteLaw.withCoin
  35   rw [Fintype.sum_prod_type]
  36   simp only [Fintype.sum_bool,Bool.false_eq_true,ite_false,ite_true]
  37   have he (d : C) :
  38       (if d=c then μ.mass d*a d else 0)+
  39         (if d=c then μ.mass d*(1-a d) else 0)=if d=c then μ.mass d else 0 := by
  40     split_ifs <;> ring

  63 def coinColourSurvives {V C : Type*} (G : SimpleGraph V)
  64     (σ : V → C×Bool) (v : V) : Prop :=
  65   (σ v).2=true ∧ tentativeColourSurvives G (fun u=>(σ u).1) v

  67 def coinRetainedColour {V C : Type*} (G : SimpleGraph V)
  68     (σ : V → C×Bool) (v : V) : Option C :=
  69   if coinColourSurvives G σ v then some (σ v).1 else none

  78     (σ : V → C×Bool) {u v : V} (h : G.Adj u v) (c : C)
  79     (hu : coinRetainedColour G σ u=some c) : coinRetainedColour G σ v≠some c := by
