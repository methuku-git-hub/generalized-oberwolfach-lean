  15 theorem sequential_ordinary_row_covers (x:Fin n→V) (hx:Function.Injective x)
  16     (A:Finset V) (hxA:∀ i,x i∉A)
  17     (hstep:∀ i:Fin n,∀ M:Finset (OrdinaryWheel Q F H r),
  18       (∀ W∈M,x i∉(ordinaryWheelRim W).support) →
  19       (∀ a∈A,(ordinaryWheelRimGraph M).degree a≤n*(2*(K+1))) →
  20       (∀ a∈A,(Finset.univ.filter (ordinaryPackingSpokes M a)).card≤n*(K+1)) →
  21       (∀ u,(Finset.univ.filter (fun a=>ordinaryPackingSpokes M a u)).card≤n*r) →
  22       ∃ N:Finset (OrdinaryWheel Q F H r),OrdinaryRowCover N (x i) A K ∧
  23         Disjoint (N.biUnion ordinaryWheelResources) (M.biUnion ordinaryWheelResources)) :
  24     ∃ C:Fin n→Finset (OrdinaryWheel Q F H r),
  25       (∀ i,OrdinaryRowCover (C i) (x i) A K) ∧
  26       (∀ i j,i≠j → Disjoint ((C i).biUnion ordinaryWheelResources) ((C j).biUnion ordinaryWheelResources)) := by
  27   let allowed:=fun (i:Fin n) (M:Finset (OrdinaryWheel Q F H r))=>OrdinaryRowCover M (x i) A K
  28   let compatible:=fun M N:Finset (OrdinaryWheel Q F H r)=>
  29     Disjoint (M.biUnion ordinaryWheelResources) (N.biUnion ordinaryWheelResources)
  30   obtain ⟨C,hC,hcross⟩:=sequential_select_prefix n allowed compatible ∅ (by
  31     intro i C hC
  32     let M:=Finset.univ.biUnion C
  33     have hMr (a:V) (ha:a∈A):(ordinaryWheelRimGraph M).degree a≤n*(2*(K+1)):=by
  34       have hd (j:Fin n):(ordinaryWheelRimGraph (C j)).degree a≤2*(K+1):=by
  35         by_cases hj:j.val < i.val
  36         · exact (hC.2.1 j hj).rimDegree a (fun he=>hxA j (he ▸ ha))
  37         · rw [hC.1 j (by omega),ordinaryWheelRimGraph_empty];simp
  38       have hh:=graph_iSup_local_degree_bound (fun j=>ordinaryWheelRimGraph (C j)) Finset.univ a (2*(K+1)) hd
  39       simp only [Finset.inter_univ,SimpleGraph.card_neighborFinset_eq_degree] at hh
  40       have hcard:(Finset.univ.filter (fun j:Fin n=>∃ b∈(Finset.univ:Finset V),(ordinaryWheelRimGraph (C j)).Adj a b)).card≤n:=by
  41         simpa only [Finset.card_univ,Fintype.card_fin] using Finset.card_le_card (Finset.filter_subset (fun j:Fin n=>∃ b∈(Finset.univ:Finset V),(ordinaryWheelRimGraph (C j)).Adj a b) Finset.univ)
  42       have hh':(⨆ j,ordinaryWheelRimGraph (C j)).degree a≤n*(2*(K+1)):=by
  43         exact hh.trans (by simpa only [Nat.mul_comm] using Nat.mul_le_mul_left (2*(K+1)) hcard)
  44       convert hh' using 1
  45       rw [ordinaryWheelRimGraph_biUnion]
  46     have hMrow (a:V) (ha:a∈A):(Finset.univ.filter (ordinaryPackingSpokes M a)).card≤n*(K+1):=by
  47       have hj (j:Fin n):(Finset.univ.filter (ordinaryPackingSpokes (C j) a)).card≤K+1:=by
  48         by_cases hji:j.val < i.val
  49         · exact (hC.2.1 j hji).rowDegree a (fun he=>hxA j (he ▸ ha))
  50         · rw [hC.1 j (by omega)];simp [ordinaryPackingSpokes]
  51       have hh:=relation_iUnion_degree_bound (fun j=>ordinaryPackingSpokes (C j) a) (K+1) hj
  52       simpa only [Fintype.card_fin,←ordinaryPackingSpokes_biUnion,M] using hh
  53     have hMcol (u:U):(Finset.univ.filter (fun a=>ordinaryPackingSpokes M a u)).card≤n*r:=by
  54       have hj (j:Fin n):(Finset.univ.filter (fun a=>ordinaryPackingSpokes (C j) a u)).card≤r:=by
  55         by_cases hji:j.val < i.val
  56         · exact (hC.2.1 j hji).columnDegree u
  57         · rw [hC.1 j (by omega)];simp [ordinaryPackingSpokes]
  58       have hh:=relation_iUnion_degree_bound (fun j a=>ordinaryPackingSpokes (C j) a u) r hj
  59       simpa only [Fintype.card_fin,←ordinaryPackingSpokes_biUnion,M] using hh
  60     have hav:∀ W∈M,x i∉(ordinaryWheelRim W).support:=by
  61       intro W hW hix
  62       obtain ⟨j,_,hWj⟩:=Finset.mem_biUnion.mp hW
  63       have hji:j.val < i.val:=by
  64         by_contra hn
  65         rw [hC.1 j (by omega)] at hWj
  66         exact Finset.notMem_empty _ hWj
  67       rcases (hC.2.1 j hji).support W hWj (x i) hix with he|he
  68       · have hh:=congrArg Fin.val (hx he);omega
  69       · exact hxA i he
  70     obtain ⟨N,hN,hNM⟩:=hstep i M hav hMr hMrow hMcol
  71     refine ⟨N,hN,?_⟩
  72     intro j hj
  73     apply hNM.symm.mono_left
  74     intro e he
  75     obtain ⟨W,hW,heW⟩:=Finset.mem_biUnion.mp he
  76     exact Finset.mem_biUnion.mpr ⟨W,Finset.mem_biUnion.mpr ⟨j,Finset.mem_univ _,hW⟩,heW⟩)
  77   refine ⟨C,hC,?_⟩
  78   intro i j hij
  79   rcases lt_or_gt_of_ne hij with h|h
  80   · exact hcross i j h
  81   · exact (hcross j i h).symm
