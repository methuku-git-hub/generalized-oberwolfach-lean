  15  theorem actual_rim_attachment_sum (G Q:SimpleGraph V) (H₀:V→U→Prop)
  16     (p:(G.edgeFinset⊕{e:V×U // H₀ e.1 e.2})→(Σ a:V,Σ c:V,Q.Walk a c))
  17     (f:V→V→ℤ):
  18     (∑ e:G.edgeFinset,(f e.val.out.1 (p (.inl e)).1+f e.val.out.2 (p (.inl e)).2.1))=
  19       ∑ d:Σ v,G.neighborSet v,f d.1 (transformerRimEndpoint G Q H₀ p d):=by
  20   let g:G.edgeFinset×Fin 2→ℤ:=fun z=>
  21     f (if z.2=0 then z.1.val.out.1 else z.1.val.out.2) (graphPathEndpoint (p (.inl z.1)) z.2)
  22   have hh:=Equiv.sum_comp (rimIncidenceEquiv G) g
  23   have hleft:(∑ z:G.edgeFinset×Fin 2,g z)=
  24       ∑ e:G.edgeFinset,(f e.val.out.1 (p (.inl e)).1+f e.val.out.2 (p (.inl e)).2.1):=by
  25     rw [Fintype.sum_prod_type]
  26     simp [g,Fin.sum_univ_two,graphPathEndpoint]
  27   rw [hleft] at hh
  28   rw [←hh]
  29   apply Finset.sum_congr rfl
  30   intro d _
  31   dsimp only [g,transformerRimEndpoint]
  32   rw [←rimIncidenceEquiv_root G d]

  34  theorem actual_spoke_attachment_sum (G Q:SimpleGraph V) (H₀:V→U→Prop)
  35     (b:∀ v,G.neighborSet v≃({u:U // H₀ v u}×Fin 2))
  36     (p:(G.edgeFinset⊕{e:V×U // H₀ e.1 e.2})→(Σ a:V,Σ c:V,Q.Walk a c))
  37     (f:V→V→ℤ):
  38     (∑ e:{e:V×U // H₀ e.1 e.2},(f e.val.1 (p (.inr e)).1+f e.val.1 (p (.inr e)).2.1))=
  39       ∑ d:Σ v,G.neighborSet v,f d.1 (transformerSpokeEndpoint G Q H₀ b p d):=by
  40   let g:({e:V×U // H₀ e.1 e.2}×Fin 2)→ℤ:=fun z=>
  41     f z.1.val.1 (graphPathEndpoint (p (.inr z.1)) z.2)
  42   have hh:=Equiv.sum_comp (spokeEndpointEquiv G H₀ b) g
  43   have hleft:(∑ z,g z)=
  44       ∑ e:{e:V×U // H₀ e.1 e.2},(f e.val.1 (p (.inr e)).1+f e.val.1 (p (.inr e)).2.1):=by
  45     rw [Fintype.sum_prod_type]
  46     simp [g,Fin.sum_univ_two,graphPathEndpoint]
  47   rw [hleft] at hh
  48   rw [←hh]
  49   rfl

  51  theorem actual_shifted_hub_sum (G:SimpleGraph V)
  52     (σ:∀ v,Equiv.Perm (G.neighborSet v)) (h:G.edgeFinset↪U) (f:V→U→ℤ):
  53     (∑ e:G.edgeFinset,(f e.val.out.1 (h e)+f e.val.out.2 (h e)))=
  54       ∑ d:Σ v,G.neighborSet v,f d.1 (transformerAssignedHub G σ h d):=by
  55   let g:G.edgeFinset×Fin 2→ℤ:=fun z=>
  56     f (if z.2=0 then z.1.val.out.1 else z.1.val.out.2) (h z.1)
  57   have hh:=Equiv.sum_comp (shiftedIncidenceEquiv G σ) g
  58   have hleft:(∑ z,g z)=∑ e:G.edgeFinset,(f e.val.out.1 (h e)+f e.val.out.2 (h e)):=by
  59     rw [Fintype.sum_prod_type]
  60     simp [g,Fin.sum_univ_two]
  61   rw [hleft] at hh
  62   rw [←hh]
  63   apply Finset.sum_congr rfl
  64   intro d _
  65   dsimp only [g,transformerAssignedHub]
  66   have hroot:=rimIncidenceEquiv_root G (incidenceShiftEquiv G σ d)
  67   change d.1=(if (shiftedIncidenceEquiv G σ d).2=0 then
  68       (shiftedIncidenceEquiv G σ d).1.val.out.1 else (shiftedIncidenceEquiv G σ d).1.val.out.2) at hroot
  69   rw [←hroot]

  71  theorem actual_transformer_attachment_cancellation (G Q:SimpleGraph V) (H₀:V→U→Prop)
  72     (b:∀ v,G.neighborSet v≃({u:U // H₀ v u}×Fin 2))
  73     (σ:∀ v,Equiv.Perm (G.neighborSet v)) (h:G.edgeFinset↪U)
  74     (p:(G.edgeFinset⊕{e:V×U // H₀ e.1 e.2})→(Σ a:V,Σ c:V,Q.Walk a c))
  75     (rim:V→V→ℤ) (spoke:V→U→ℤ) (core:G.edgeFinset→ℤ)
  76     (cross load:{e:V×U // H₀ e.1 e.2}→ℤ):
  77     (∑ e:G.edgeFinset,(core e+rim e.val.out.1 (p (.inl e)).1+rim e.val.out.2 (p (.inl e)).2.1+
  78       spoke e.val.out.1 (h e)+spoke e.val.out.2 (h e)))+
  79     (∑ e:{e:V×U // H₀ e.1 e.2},(cross e+rim e.val.1 (p (.inr e)).1+
  80       rim e.val.1 (p (.inr e)).2.1+load e))-
  81     (∑ d:Σ v,G.neighborSet v,(rim d.1 (transformerRimEndpoint G Q H₀ p d)+
  82       rim d.1 (transformerSpokeEndpoint G Q H₀ b p d)+spoke d.1 (transformerAssignedHub G σ h d)))=
  83     (∑ e:G.edgeFinset,core e)+(∑ e:{e:V×U // H₀ e.1 e.2},cross e)+
  84       ∑ e:{e:V×U // H₀ e.1 e.2},load e:=by
  85   have hr:=actual_rim_attachment_sum G Q H₀ p rim
  86   have hz:=actual_spoke_attachment_sum G Q H₀ b p rim
  87   have hs:=actual_shifted_hub_sum G σ h spoke
  88   simp_rw [Finset.sum_add_distrib] at hr hz hs ⊢
  89   rw [←hr,←hz,←hs]
  90   abel
