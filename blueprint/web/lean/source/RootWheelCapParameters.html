  11 theorem root_wheel_cap_parameters (r:ℕ) (C β:ℝ) (hC:0≤C) (hβ:0<β) :
  12     ∃ K:ℕ,0<K ∧ ∃ ζ:ℝ,0<ζ ∧ ∃ n₀:ℕ,∀ n N:ℕ,n₀≤n →
  13       (N:ℝ)≤C*n → ∀ L:ℝ,0≤L → L≤ζ*n →
  14       (N*r:ℕ)/(K:ℝ)+((r:ℝ)+3)*(L+1)+4*(K+1)<β*n := by
  15   obtain ⟨K₀,hK₀⟩:=exists_nat_gt (4*C*r/β)
  16   let K:=K₀+1
  17   have hK:0<K:=by dsimp [K];omega
  18   have hKR:(0:ℝ)<K:=Nat.cast_pos.mpr hK
  19   have hKmargin:4*C*r<β*K:=by
  20     have hh:4*C*r/β<(K:ℝ):=hK₀.trans_le (by dsimp [K];push_cast;linarith)
  21     have hh':=(div_lt_iff₀ hβ).mp hh
  22     nlinarith only [hh']
  23   let ζ:=β/(4*((r:ℝ)+3))
  24   have hζ:0<ζ:=by dsimp [ζ];positivity
  25   obtain ⟨n₀,hn₀⟩:=exists_nat_gt (2*(((r:ℝ)+3)+4*(K+1))/β)
  26   refine ⟨K,hK,ζ,hζ,n₀,?_⟩
  27   intro n N hn hN L hL hLsmall
  28   have hnR:(n₀:ℝ)≤n:=Nat.cast_le.mpr hn
  29   have hconstant:2*(((r:ℝ)+3)+4*(K+1))<β*n:=by
  30     have hh:=(div_lt_iff₀ hβ).mp (hn₀.trans_le hnR)
  31     nlinarith only [hh]
  32   have hnpos:(0:ℝ)<n:=by
  33     have hh : 0<β*n:=lt_trans (by positivity) hconstant
  34     exact (mul_pos_iff.mp hh).resolve_right (fun h=>not_lt_of_ge hβ.le h.1) |>.2
  35   have hterm:(N*r:ℕ)/(K:ℝ)≤β*n/4:=by
  36     apply (div_le_iff₀ hKR).mpr
  37     push_cast
  38     have hh:=mul_le_mul_of_nonneg_right hN (Nat.cast_nonneg r)
  39     have hk':=mul_le_mul_of_nonneg_right hKmargin.le hnpos.le
  40     nlinarith only [hh,hk']
  41   have hLterm:((r:ℝ)+3)*L≤β*n/4:=by
  42     have hh:=mul_le_mul_of_nonneg_left hLsmall (show (0:ℝ)≤r+3 by positivity)
  43     have he:((r:ℝ)+3)*(ζ*n)=β*n/4:=by dsimp [ζ];field_simp
  44     rwa [he] at hh
  45   nlinarith only [hterm,hLterm,hconstant]
