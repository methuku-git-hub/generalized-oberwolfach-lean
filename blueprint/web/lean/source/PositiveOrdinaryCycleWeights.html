  13 theorem positive_ordinary_cycle_weights (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  14     ∃ ε c C : ℝ,0<ε ∧ 0<c ∧ 0<C ∧ ∃ N : ℕ,0<N ∧
  15       ∀ (V : Type) [Fintype V] [DecidableEq V],N≤Fintype.card V →
  16       ∀ G : SimpleGraph V,
  17         (∀ v,(cycleThreshold r+α)*Fintype.card V≤(G.degree v:ℝ)) →
  18       ∀ w : G.edgeSet → ℝ,‖w-(fun _ => (1:ℝ))‖≤ε →
  19       ∃ a : OrdinaryCycle G r → ℝ,
  20         (∀ D,c/(Fintype.card V:ℝ)^(r-2)≤a D ∧
  21           a D≤C/(Fintype.card V:ℝ)^(r-2)) ∧
  22         (∑ D,a D • ordinaryCycleIncidence G r D)=w := by
  23   let κ := min (1/2:ℝ) (α/32)
  24   have hκ : 0<κ := lt_min (by norm_num) (by positivity)
  25   have hκ1 : κ≤1 := (min_le_left _ _).trans (by norm_num)
  26   have hκα : κ≤α/16 := (min_le_right _ _).trans (by linarith)
  27   obtain ⟨N₀,hN₀⟩ := ordinary_cycle_approximation_at_size r hr α hα
  28   obtain ⟨m,hm₀,hm4,hmroots,hmerr⟩ := ordinary_positive_sample_size N₀ α κ hα hκ
  29   let c := κ/(4*(r:ℝ)^2)
  30   let C := c+4*(m:ℝ)^(r-2)
  31   have hr' : (0:ℝ)<r := Nat.cast_pos.mpr (by omega)
  32   have hc : 0<c := by dsimp [c];positivity
  33   have hC : 0<C := by dsimp [C];positivity
  34   refine ⟨κ/4,c,C,by positivity,hc,hC,m,by omega,?_⟩
  35   intro V _ _ hn G hdeg w hw
  36   obtain ⟨a,ha,he⟩ := exact_positive_ordinary_cycle_weights G r (by omega) (by omega)
  37     w κ (8*m*Real.exp (-α^2*m/32)) (4*(m:ℝ)^(r-2)) hκ hmerr hw
  38     (fun w' hw' => hN₀ m hm₀ hm4 hmroots V hn G hdeg κ hκ.le hκ1 hκα w' hw')
  39   exact ⟨a,fun D => ⟨(ha D).2.1,(ha D).2.2⟩,he⟩

  41 theorem positive_ordinary_fractional_cycle_weights (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  42     ∃ c C : ℝ,0<c ∧ 0<C ∧ ∃ N : ℕ,0<N ∧
  43       ∀ (V : Type) [Fintype V] [DecidableEq V],N≤Fintype.card V →
  44       ∀ G : SimpleGraph V,
  45         (∀ v,(cycleThreshold r+α)*Fintype.card V≤(G.degree v:ℝ)) →
  46       ∃ a : OrdinaryCycle G r → ℝ,
  47         (∀ D,c/(Fintype.card V:ℝ)^(r-2)≤a D ∧
  48           a D≤C/(Fintype.card V:ℝ)^(r-2)) ∧
  49         (∀ e : G.edgeSet,∑ D,a D*ordinaryCycleIncidence G r D e=1) := by
  50   obtain ⟨ε,c,C,hε,hc,hC,N,hN,hh⟩ := positive_ordinary_cycle_weights r hr α hα
  51   refine ⟨c,C,hc,hC,N,hN,?_⟩
  52   intro V _ _ hn G hdeg
  53   obtain ⟨a,ha,he⟩ := hh V hn G hdeg (fun _ => 1) (by simpa using hε.le)
  54   refine ⟨a,ha,?_⟩
  55   intro e
  56   have h := congrFun he e
  57   simpa only [Finset.sum_apply,Pi.smul_apply,smul_eq_mul] using h
