  23 theorem absorber_profile_count_exchange (e : (Σ i,V i) ↪ W)
  24     (R : C → Finset W) (S : T → VertexAbsorberTemplate c) (X : Finset W)
  25     (J : ℕ) (hJ : ∀ A : Finset W,Fintype.card {C₀ : C // R C₀=A} ≤ J)
  26     {m : Fin s → ℕ} (σ τ : ∀ i,FixedSubset (V i) (m i)) (i : Fin s)
  27     (hagree : ∀ j,j≠i → σ j=τ j) (x y : V i)
  28     (he : τ i=fixedSubsetMap (Equiv.swap x y) (σ i)) :
  29     |absorberGroundCount R S X ((profileRootSet σ).map e)-
  30       absorberGroundCount R S X ((profileRootSet τ).map e)| ≤
  31       2*((c*c)*Fintype.card T*X.card^c*J^(2*c+1):ℕ)*((∑ j,m j)+1:ℕ)^(c*c-1) := by
  32   let A := (profileRootSet σ).map e
  33   let B := (profileRootSet τ).map e
  34   let D := (c*c)*Fintype.card T*X.card^c*J^(2*c+1)
  35   let K := D*((∑ j,m j)+1)^(c*c-1)
  36   have hb := profileSample_exchange_card_bounds e (∅ : Finset W) σ τ i hagree x y he
  37   simp only [Finset.empty_union,Finset.card_empty,zero_add] at hb
  38   have hK (v : W) : Fintype.card {d : LabelledAbsorberFamily R S X //
  39       labelledAbsorberFree R (S d.1) X d.2⊆A∪B ∧ v∈labelledAbsorberFree R (S d.1) X d.2} ≤ K := by
  40     calc
  41       _ ≤ (c*c)*Fintype.card T*X.card^c*(A∪B).card^(c*c-1)*J^(2*c+1) :=
  42         labelled_absorber_local_free_vertex_count R S X (A∪B) J hJ v
  43       _ = D*(A∪B).card^(c*c-1) := by dsimp [D];ring
  44       _ ≤ K := Nat.mul_le_mul_left D (Nat.pow_le_pow_left hb.2.2 _)
  45   have hh := finite_support_count_difference
  46     (fun d : LabelledAbsorberFamily R S X => labelledAbsorberFree R (S d.1) X d.2) A B K hK
  47   have hsum : (A\B).card+(B\A).card ≤ 2 := by dsimp [A,B];omega
  48   have hsumR : (((A\B).card+(B\A).card:ℕ):ℝ) ≤ 2 := by exact_mod_cast hsum
  49   have hh' := hh.trans (mul_le_mul_of_nonneg_right hsumR (Nat.cast_nonneg K))
  50   simpa only [K,Nat.cast_mul,mul_assoc,absorberGroundCount,D,A,B,Nat.cast_pow] using hh'
