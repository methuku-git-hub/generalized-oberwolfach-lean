  14 theorem divisible_colour_repair {V : Type*} [Fintype V] [DecidableEq V] [Nonempty V]
  15     {q h : ℕ} (hq : Even q) (hh : 0<h) (hhq : h∣q)
  16     (G : SimpleGraph V) (σ : Sym2 V → Fin (q+1)) (ρ : ℝ) (L : ℕ) (hL : 0<L)
  17     (hreserve : q*h≤(edgeColourGraph G σ (Fin.last q)).edgeFinset.card)
  18     (hpath : ∀ a : Fin q,∀ u v,u≠v → ρ*Fintype.card V≤
  19       (colouredTwoPathCount G σ u v a.castSucc (Fin.last q):ℝ))
  20     (hloss : (q:ℝ)*h≤ρ*Fintype.card V/2)
  21     (hpack : (q:ℝ)*Fintype.card V/L+4*q+4*L<(ρ/2)*Fintype.card V) :
  22     ∃ τ : Sym2 V → Fin (q+1),
  23       (∀ a,(∀ v,Even ((G\edgeColourGraph G τ a).degree v)) ∧
  24         h∣(G\edgeColourGraph G τ a).edgeFinset.card) ∧
  25       (∀ a v,|((edgeColourGraph G τ a).degree v:ℝ)-((edgeColourGraph G σ a).degree v:ℝ)|≤q*h+q+1) := by
  26   obtain ⟨σ',hres,hdeg,hcand⟩ := colour_residue_repair hh hhq G σ hreserve
  27   have hpath' (a : Fin q) (u v : V) (huv : u≠v) :
  28       (ρ/2)*Fintype.card V≤(colouredTwoPathCount G σ' u v a.castSucc (Fin.last q):ℝ) := by
  29     have hh := (abs_le.mp (hcand a.castSucc u v huv)).1
  30     have hp := hpath a u v huv
  31     linarith only [hh,hp,hloss]
  32   obtain ⟨τ,hcard,hpar,hchg⟩ := parity_repair_from_coloured_paths hq G σ' (ρ/2) L hL hpath' hpack
  33   refine ⟨τ,?_,?_⟩
  34   · intro a
  35     refine ⟨hpar a,?_⟩
  36     apply complement_edge_divisible_of_residue G (edgeColourGraph G τ a) (edgeColourGraph_le G τ a) h
  37     rw [hcard]
  38     exact hres a
  39   · intro a v
  40     have hh := (abs_sub_le ((edgeColourGraph G τ a).degree v:ℝ)
  41       ((edgeColourGraph G σ' a).degree v:ℝ) ((edgeColourGraph G σ a).degree v:ℝ)).trans
  42       (add_le_add (hchg a v) (hdeg a v))
  43     linarith only [hh]
  44 
  45 end Oberwolfach
