   8 def IsSelectedSegmentRun (T:Finset ℕ) (a b:ℕ):Prop :=
   9   a≤b ∧ (∀ j,a≤j→j≤b→j∈T) ∧ (a=0 ∨ a-1∉T) ∧ b+1∉T

  24 theorem selected_run_exists_unique_from_start (d:ℕ) (T:Finset ℕ)
  25     (hT:T⊆Finset.range d) (a:ℕ) (ha:a∈selectedRunStarts T):
  26     ∃! b,IsSelectedSegmentRun T a b := by
  27   have ha':= (selected_run_start_mem T a).mp ha
  28   have had:a<d := by have hh:=hT ha'.1;simpa using hh
  29   have hex:∃ k,a≤k ∧ k∉T := by
  30     refine ⟨d,by omega,?_⟩
  31     intro hd
  32     have hh:=hT hd
  33     simp at hh
  34   let k:=Nat.find hex
  35   have hk:=Nat.find_spec hex
  36   change a≤k ∧ k∉T at hk
  37   have hak:a<k := by
  38     by_contra h
  39     have : k=a := by omega
