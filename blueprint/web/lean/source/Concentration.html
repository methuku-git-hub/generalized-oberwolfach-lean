   4 /-!
   5 Concentration for sums of independent bounded real random variables.
   6 Independence is imposed on the original random variables. In particular,
   7 different contributions of one physical edge must be combined into a single
   8 variable before this theorem is applied.
   9 -/

  15 theorem subgaussian_absolute_tail
  16     {Ω : Type*} [MeasurableSpace Ω] {μ : Measure Ω}
  17     {X : Ω → ℝ} {c : ℝ≥0} (hX : HasSubgaussianMGF X c μ)
  18     {t : ℝ} (ht : 0 ≤ t) :
  19     μ.real {ω | t ≤ |X ω|} ≤ 2 * Real.exp (-t ^ 2 / (2 * c)) := by
  20   have hevent : {ω | t ≤ |X ω|} =
  21       {ω | t ≤ X ω} ∪ {ω | t ≤ -X ω} := by
  22     ext ω
  23     simp only [Set.mem_ofPred_eq, Set.mem_union, le_abs]
  24   rw [hevent]
  25   have hupper := hX.measure_ge_le ht
  26   have hlower := hX.neg.measure_ge_le ht
  27   have hunion := measureReal_union_le (μ := μ) {ω | t ≤ X ω} {ω | t ≤ -X ω}
  28   simp only [Pi.neg_apply] at hlower
  29   exact hunion.trans (by linarith)

  31 /-- Hoeffding's two-sided bound, with the sum of squared interval lengths.
  32 The interval and independence assumptions are explicit; no independence
  33 between derived tests or between different colours is required. -/
  34 theorem boundedIndependentSum_tail
  35     {Ω I : Type*} [MeasurableSpace Ω] {μ : Measure Ω} [IsProbabilityMeasure μ]
  36     (X : I → Ω → ℝ) (a b : I → ℝ) (s : Finset I)
  37     (hindep : iIndepFun X μ)
  38     (hmeas : ∀ i ∈ s, AEMeasurable (X i) μ)
  39     (hbounded : ∀ i ∈ s, ∀ᵐ ω ∂μ, X i ω ∈ Set.Icc (a i) (b i))
  40     {t : ℝ} (ht : 0 ≤ t) :
  41     μ.real {ω | t ≤ |(∑ i ∈ s, X i ω) - ∑ i ∈ s, ∫ ω, X i ω ∂μ|} ≤
  42       2 * Real.exp (-2 * t ^ 2 / (∑ i ∈ s, (b i - a i) ^ 2)) := by
  43   classical
  44   let Y i ω := X i ω - ∫ z, X i z ∂μ
  45   let c (i : I) : ℝ≥0 := (‖b i - a i‖₊ / 2) ^ 2
  46   have hY : iIndepFun Y μ :=
  47     hindep.comp (fun i x => x - ∫ z, X i z ∂μ) (fun i => by fun_prop)
  48   have hsub (i : I) (hi : i ∈ s) : HasSubgaussianMGF (Y i) (c i) μ :=
  49     hasSubgaussianMGF_of_mem_Icc (hmeas i hi) (hbounded i hi)
  50   have hsum := HasSubgaussianMGF.sum_of_iIndepFun hY hsub
  51   have htail := subgaussian_absolute_tail hsum ht
  52   have hc : ((∑ i ∈ s, c i : ℝ≥0) : ℝ) =
  53       (∑ i ∈ s, (b i - a i) ^ 2) / 4 := by
  54     simp only [NNReal.coe_sum, c, NNReal.coe_pow, NNReal.coe_div, NNReal.coe_ofNat,
  55       coe_nnnorm, Real.norm_eq_abs]
  56     rw [Finset.sum_div]
  57     apply Finset.sum_congr rfl
  58     intro i _
  59     rw [div_pow, sq_abs]
  60     norm_num
  61   have heq : -t ^ 2 / (2 * ((∑ i ∈ s, c i : ℝ≥0) : ℝ)) =
  62       -2 * t ^ 2 / (∑ i ∈ s, (b i - a i) ^ 2) := by
  63     rw [hc]
  64     ring
  65   simpa only [Y, Finset.sum_sub_distrib, heq] using htail
