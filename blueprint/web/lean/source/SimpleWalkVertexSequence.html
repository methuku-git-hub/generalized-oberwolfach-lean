  10 def simpleWalkVertexEmbedding {V:Type*} (G:SimpleGraph V) {a b:V}
  11     (p:G.Walk a b) (hp:p.IsPath):Fin (p.length+1)↪V := by
  12   have hn:p.support.Nodup:=by simpa only [SimpleGraph.Walk.isPath_def] using hp
  13   refine ⟨fun i=>p.getVert i.val,?_⟩
  14   intro i j hij
  15   have hi:p.getVert i.val=p.support.get ⟨i.val,by simpa only [p.length_support] using i.isLt⟩:=by
  16     exact p.getVert_eq_support_getElem (by omega)
  17   have hj:p.getVert j.val=p.support.get ⟨j.val,by simpa only [p.length_support] using j.isLt⟩:=by
  18     exact p.getVert_eq_support_getElem (by omega)
  19   have hget:=hn.injective_get (hi.symm.trans (hij.trans hj))
  20   have hval:i.val=j.val:=congrArg (fun k:Fin p.support.length=>k.val) hget
  21   exact Fin.ext hval

  23 theorem simple_walk_vertex_embedding_adjacent {V:Type*} (G:SimpleGraph V) {a b:V}
  24     (p:G.Walk a b) (hp:p.IsPath):
  25     ∀k:Fin p.length,G.Adj (simpleWalkVertexEmbedding G p hp k.castSucc)
  26       (simpleWalkVertexEmbedding G p hp k.succ):=by
  27   intro k
  28   exact p.adj_getVert_succ k.isLt

  30 theorem simple_walk_vertex_embedding_support {V:Type*} [DecidableEq V]
  31     (G:SimpleGraph V) {a b:V} (p:G.Walk a b) (hp:p.IsPath):
  32     Finset.univ.image (simpleWalkVertexEmbedding G p hp)=p.support.toFinset:=by
  33   classical
  34   ext x
  35   simp only [Finset.mem_image,Finset.mem_univ,true_and,List.mem_toFinset]
  36   constructor
  37   · rintro ⟨i,rfl⟩
  38     exact p.getVert_mem_support i.val
  39   · intro hx
  40     obtain ⟨i,hi⟩:=List.mem_iff_get.mp hx
  41     let k:Fin (p.length+1):=⟨i.val,by simpa only [p.length_support] using i.isLt⟩
  42     refine ⟨k,?_⟩
  43     change p.getVert k.val=x
  44     rw [p.getVert_eq_support_getElem (by omega)]
  45     exact hi
  46 end Oberwolfach
