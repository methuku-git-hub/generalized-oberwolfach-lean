  28 theorem marked_supported_avoids_old
  29     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  30     {X Z:Finset V} (B:Finset U) (hd:Disjoint X Z)
  31     (e:Finset V⊕(V×U)) (he:MarkedResourceSupported tail head ktail khead π X e):
  32     ¬MarkedOldResource tail head ktail khead π Z B e:=by
  33   intro hold
  34   cases e with
  35   | inl e=>
  36     obtain ⟨a,ha,hat,hah⟩:=he
  37     obtain ⟨a',ha',hat',hah'⟩:=hold
  38     have h:a=a':=hp (ha.trans ha'.symm)
  39     exact Finset.disjoint_left.mp hd hat (h.symm ▸ hat')
  40   | inr vu=>exact Finset.disjoint_left.mp hd he hold.1
