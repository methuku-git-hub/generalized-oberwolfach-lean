  23 theorem buffered_residual_spoke_card
  24     {X V I:Type*} [Fintype X] [DecidableEq X] [Fintype V] [DecidableEq V] [Fintype I]
  25     (n:I→ℕ) (ext:∀i,Fin (n i+5)↪X) (inner:∀i,Fin (n i+1)↪X)
  26     (hinner:∀i p,inner i p=ext i ⟨p.val+2,by omega⟩)
  27     (hdis:∀i j,i≠j→Disjoint (Finset.univ.image (ext i)) (Finset.univ.image (ext j)))
  28     (fC:{x // x∉bufferedPathStrictInterior n inner}↪V) (hcard:Fintype.card X=Fintype.card V)
  29     (finish:I→V) (hend:∀i,fC (bufferedEndpointInComplement n ext inner hinner hdis i)=finish i)
  30     (R:Finset V)
  31     (hR:R=Finset.univ\(outsideBufferedPathImage n inner fC ∪ Finset.univ.image finish)) :
  32     R.card=∑i,n i := by
  33   classical
  34   let A:=bufferedPathFullDomain n inner
  35   let D:=bufferedPathStrictInterior n inner
  36   let E:=bufferedFinalEndpoints n inner
  37   let hDA:D⊆A:=buffered_strict_interior_subset_domain n inner
  38   have hEA:E⊆A:=by
  39     intro x hx
  40     obtain ⟨i,_,rfl⟩:=Finset.mem_image.mp hx
  41     exact buffered_path_vertex_mem_domain n inner i (Fin.last (n i))
  42   have hED:Disjoint E D:=buffered_final_endpoints_disjoint_strict n ext inner hinner hdis
  43   let Z:=outsideBufferedPathImage n inner fC
  44   let B:=Finset.univ.image finish
  45   have hZ:Z=Finset.univ.image ((complementDomainEmbedding hDA).trans fC):=by
  46     ext v
  47     constructor
  48     · intro hv
  49       obtain ⟨x,hx,heq⟩:=Finset.mem_image.mp hv
  50       have hxA:x.val∉A:=(Finset.mem_filter.mp hx).2
  51       refine Finset.mem_image.mpr ⟨⟨x.val,hxA⟩,Finset.mem_univ _,?_⟩
  52       change fC ⟨x.val,_⟩=v
  53       exact heq
  54     · intro hv
  55       obtain ⟨x,_,heq⟩:=Finset.mem_image.mp hv
  56       let y:{x // x∉D}:=complementDomainEmbedding hDA x
  57       refine Finset.mem_image.mpr ⟨y,Finset.mem_filter.mpr ⟨Finset.mem_univ _,?_⟩,?_⟩
  58       · exact x.property
  59       · simpa [y,complementDomainEmbedding] using heq
  60   have hB:B=Finset.univ.image ((endpointDomainEmbedding hED).trans fC):=by
  61     ext v
  62     constructor
  63     · intro hv
  64       obtain ⟨i,_,heq⟩:=Finset.mem_image.mp hv
  65       let x:{x // x∈E}:=⟨inner i (Fin.last (n i)),Finset.mem_image.mpr ⟨i,Finset.mem_univ _,rfl⟩⟩
  66       refine Finset.mem_image.mpr ⟨x,Finset.mem_univ _,?_⟩
  67       change fC (endpointDomainEmbedding hED x)=v
  68       rw [←heq,←hend i]
  69       congr 1
  70     · intro hv
  71       obtain ⟨x,_,heq⟩:=Finset.mem_image.mp hv
  72       obtain ⟨i,_,hix⟩:=Finset.mem_image.mp x.property
  73       refine Finset.mem_image.mpr ⟨i,Finset.mem_univ _,?_⟩
  74       have hxi:bufferedEndpointInComplement n ext inner hinner hdis i=endpointDomainEmbedding hED x:=by
  75         apply Subtype.ext
  76         exact hix
  77       calc
  78         finish i = fC (bufferedEndpointInComplement n ext inner hinner hdis i):=(hend i).symm
  79         _ = fC (endpointDomainEmbedding hED x):=congrArg fC hxi
  80         _ = v:=heq
  81   have hcardR:=residual_spoke_card_from_complement_embedding A D E hDA hEA hED fC hcard R Z B hZ hB hR
  82   have hA:A.card=∑i,(n i+1):=by
  83     simpa [A,bufferedPathFullDomain] using buffered_path_interior_domain_card n ext inner hinner hdis Finset.univ
  84   have hE:E.card=Fintype.card I:=buffered_final_endpoints_card n ext inner hinner hdis
  85   have hsum:(∑i,(n i+1))=(∑i,n i)+Fintype.card I:=by simp [Finset.sum_add_distrib]
  86   rw [hcardR,hA,hE,hsum]
  87   omega
  88 end Oberwolfach
