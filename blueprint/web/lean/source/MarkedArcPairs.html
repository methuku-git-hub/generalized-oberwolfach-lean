  13 theorem finRotate_twice_ne_self {c : ℕ} (hc : 3≤c) (i : Fin c) :
  14     finRotate c (finRotate c i)≠i := by
  15   letI : NeZero c := ⟨by omega⟩
  16   intro he
  17   simp only [finRotate_apply] at he
  18   have hh : (1 : Fin c)+1=0 := add_left_cancel (show i+(1+1)=i+0 by simpa only [add_assoc,add_zero] using he)
  19   have hv := congrArg Fin.val hh
  20   norm_num [Fin.val_add,Fin.val_one,Fin.val_zero,Nat.mod_eq_of_lt (show 1<c by omega),Nat.mod_eq_of_lt (show 2<c by omega)] at hv
