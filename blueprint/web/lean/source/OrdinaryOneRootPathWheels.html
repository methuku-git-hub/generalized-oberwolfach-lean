  16  theorem ordinary_one_root_path_wheels (Q F:SimpleGraph V) (H:V→U→Prop)
  17     (r:ℕ) (hr:3≤r) (x y a b:V) (u u':U) (p:Q.Walk a b)
  18     (hp:p.IsPath) (hlen:p.length=r-2) (hxy:x≠y)
  19     (hxa:Q.Adj x a) (hxb:Q.Adj x b) (hya:Q.Adj y a) (hyb:Q.Adj y b)
  20     (havoid:∀ v∈p.support,v≠x∧v≠y)
  21     (hF:∀ v∈insert x (insert y p.support.toFinset),
  22       ∀ w∈insert x (insert y p.support.toFinset),¬F.Adj v w)
  23     (hH:H x u∧H y u') (hHp:∀ v∈p.support,H v u∧H v u'):
  24     ∃ W W':OrdinaryWheel Q F H r,
  25       ordinaryWheelHub W=u ∧ ordinaryWheelHub W'=u' ∧
  26       (∀ v,v∈(ordinaryWheelRim W).support↔v=x∨v∈p.support) ∧
  27       (∀ v,v∈(ordinaryWheelRim W').support↔v=y∨v∈p.support) ∧
  28       (∀ e:Sym2 V,
  29         (if Sum.inl e∈ordinaryWheelResources W then (1:ℤ) else 0)-
  30           (if Sum.inl e∈ordinaryWheelResources W' then 1 else 0)=
  31         (if e=s(x,a) then 1 else 0)+(if e=s(x,b) then 1 else 0)-
  32           (if e=s(y,a) then 1 else 0)-(if e=s(y,b) then 1 else 0)) ∧
  33       (∀ v z,
  34         (if Sum.inr (v,z)∈ordinaryWheelResources W then (1:ℤ) else 0)-
  35           (if Sum.inr (v,z)∈ordinaryWheelResources W' then 1 else 0)=
  36         (if v=x∧z=u then 1 else 0)-(if v=y∧z=u' then 1 else 0)+
  37         (if v∈p.support∧z=u then 1 else 0)-(if v∈p.support∧z=u' then 1 else 0)):=by
  38   have hxp:x∉p.support:=fun h=>(havoid x h).1 rfl
  39   let q:Q.Walk x b:=p.cons hxa
  40   have hq:q.IsPath:=SimpleGraph.Walk.IsPath.cons hp hxp
  41   have hqLen:q.length=r-1:=by simp only [q,SimpleGraph.Walk.length_cons,hlen];omega
  42   obtain ⟨C,hC,hCQ,hCxb,hCs,hsteps⟩:=embedded_cycle_from_path_with_support Q r hr x b q hq hqLen hxb
  43   have hs (v:V):v∈C.support↔v=x∨v∈p.support:=by
  44     rw [hCs];simp only [q,SimpleGraph.Walk.support_cons,List.mem_cons]
  45   have hCxa:C.Adj x a:=by
  46     have hh:=hsteps 0 (by rw [hqLen];omega)
  47     simpa [q] using hh
  48   have hab:a≠b:=by
  49     intro he
  50     subst b
  51     have hz:=SimpleGraph.Walk.isPath_iff_eq_nil.mp hp
  52     rw [hz,SimpleGraph.Walk.length_nil] at hlen
  53     omega
  54   have hin (v:V) (hv:v∈C.support):v∈insert x (insert y p.support.toFinset):=by
  55     rcases (hs v).mp hv with rfl|hv <;> simp [hv]
  56   let W:OrdinaryWheel Q F H r:=⟨(⟨⟨C,hC⟩,hCQ⟩,u),
  57     (fun v hv=>by rcases (hs v).mp hv with rfl|hv;exact hH.1;exact (hHp v hv).1),
  58     (fun v hv w hw=>hF v (hin v hv) w (hin w hw))⟩
  59   have hyC:y∉C.support:=by
  60     intro hy
  61     rcases (hs y).mp hy with he|hy
  62     · exact hxy he.symm
  63     · exact (havoid y hy).2 rfl
  64   have hneigh:C.neighborFinset x={a,b}:=ordinaryWheel_root_neighbors W x a b hab hCxa hCxb
  65   have hnew:∀ v,C.Adj x v→Q.Adj y v:=by
  66     intro v hv
  67     have hm:v∈C.neighborFinset x:=(C.mem_neighborFinset x v).mpr hv
  68     rw [hneigh] at hm
  69     rcases Finset.mem_insert.mp hm with rfl|hm
  70     · exact hya
  71     · exact Finset.mem_singleton.mp hm ▸ hyb
  72   obtain ⟨W',hWr,hWu,hWs,hrim,hsp⟩:=ordinary_wheel_root_replacement W x y u'
  73     hCxa.mem_support_left hyC hnew (fun v hv _=>hF y (by simp) v (hin v hv)) hH.2
  74     (fun v hv hvx=>by
  75       have hpv: v∈p.support:=((hs v).mp hv).resolve_left hvx
  76       exact (hHp v hpv).2)
  77   have hs' (v:V):v∈(ordinaryWheelRim W').support↔v=y∨v∈p.support:=by
  78     rw [hWs,show ordinaryWheelRim W=C from rfl,hs]
  79     constructor
  80     · rintro (he|⟨he|hv,hvx⟩)
  81       · exact Or.inl he
  82       · exact (hvx he).elim
  83       · exact Or.inr hv
  84     · rintro (he|hv)
  85       · exact Or.inl he
  86       · exact Or.inr ⟨Or.inr hv,(havoid v hv).1⟩
  87   refine ⟨W,W',rfl,hWu,hs,hs',?_,?_⟩
  88   · intro e
  89     have hge (z:V):graphRootEdges C x z={s(z,a),s(z,b)}:=by
  90       unfold graphRootEdges;rw [hneigh];simp
  91     have hh:=hrim e
  92     change (if Sum.inl e∈ordinaryWheelResources W then (1:ℤ) else 0)-
  93       (if Sum.inl e∈ordinaryWheelResources W' then 1 else 0)=
  94       (if e∈graphRootEdges C x x then 1 else 0)-(if e∈graphRootEdges C x y then 1 else 0) at hh
  95     rw [hge,hge,int_indicator_pair e _ _ (fun h=>hab (sym2_fixed_endpoint_injective x h)),
  96       int_indicator_pair e _ _ (fun h=>hab (sym2_fixed_endpoint_injective y h))] at hh
  97     linarith only [hh]
  98   · intro v z
  99     have he:(v∈C.support∧v≠x)↔v∈p.support:=by
 100       rw [hs]
 101       constructor
 102       · rintro ⟨he|hv,hvx⟩
 103         · exact (hvx he).elim
 104         · exact hv
 105       · intro hv
 106         exact ⟨Or.inr hv,(havoid v hv).1⟩
 107     simpa only [show ordinaryWheelRim W=C from rfl,show ordinaryWheelHub W=u from rfl,
 108       and_assoc,←and_assoc,he] using hsp v z
