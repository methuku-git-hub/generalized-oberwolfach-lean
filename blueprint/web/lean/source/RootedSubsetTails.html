  13 theorem fixedSubset_intersection_tail_at_size {V : Type*} [Fintype V] [DecidableEq V]
  14     {k : ℕ} [Nonempty (FixedSubset V k)] (hk : k≤Fintype.card V)
  15     (A : Finset V) (t : ℝ) (ht : 0≤t) :
  16     (FiniteLaw.uniform (FixedSubset V k)).probability
  17       (fun S => t≤|((A∩S.val).card:ℝ)-(A.card:ℝ)*(k:ℝ)/Fintype.card V|)≤
  18         2*Real.exp (-2*t^2/k) := by
  19   let f (S : FixedSubset V k) := ((A∩S.val).card:ℝ)
  20   have hm (s : ℝ) : (FiniteLaw.uniform (FixedSubset V k)).expectation
  21       (fun S => Real.exp (s*(f S-(FiniteLaw.uniform (FixedSubset V k)).expectation f)))≤
  22         Real.exp (s^2*(k:ℝ)/8) := by
  23     have hh := fixedSubset_mgf hk f (b:=1) (by norm_num)
  24       (fun S u v => by
  25         simpa only [f,fixedSubsetMap,← Set.ncard_coe_finset,Finset.coe_inter]
  26           using swap_subset_intersection_change A S.val u v) s
  27     simpa only [one_pow,mul_one] using hh
  28   have hh := finiteLaw_mgf_absolute_tail (FiniteLaw.uniform (FixedSubset V k)) f
  29     (show (0:ℝ)≤k by positivity) hm ht
  30   have he : (FiniteLaw.uniform (FixedSubset V k)).expectation f=
  31       (A.card:ℝ)*((k:ℝ)/Fintype.card V) := by
  32     convert fixedSubset_intersection_mean hk A using 1
  33     unfold FiniteLaw.expectation
  34     apply Finset.sum_congr rfl
  35     intro S _
  36     congr 1
  37     simp only [f,← Set.ncard_coe_finset,Finset.coe_inter]
  38   rw [he] at hh
  39   simpa only [f,mul_div_assoc] using hh

  41 theorem rooted_subset_mean_error {n t k a q : ℝ}
  42     (hn : 0<n) (ht : 0≤t) (htn : t<n) (hk : 0≤k) (hkt : k≤n-t)
  43     (ha : 0≤a) (han : a≤n) (hq : 0≤q) (hqt : q≤t) :
  44     |q+(a-q)*k/(n-t)-a*(t+k)/n|≤t := by
  45   have hn' : 0<n-t := sub_pos.mpr htn
  46   have hfactor : 0≤1-k/(n-t) := by
  47     have hh := (div_le_one hn').mpr hkt
  48     linarith
  49   have hfactor' : 1-k/(n-t)≤1 := by linarith [div_nonneg hk hn'.le]
  50   have hmid : 0≤a*t/n := by positivity
  51   have hmid' : a*t/n≤t := by
  52     apply (div_le_iff₀ hn).mpr
  53     nlinarith
  54   have hb : |q-a*t/n|≤t := abs_le.mpr ⟨by linarith,by linarith⟩
  55   have he : q+(a-q)*k/(n-t)-a*(t+k)/n=(1-k/(n-t))*(q-a*t/n) := by
  56     field_simp
  57     ring
  58   rw [he,abs_mul,abs_of_nonneg hfactor]
  59   calc
  60     _ ≤ (1-k/(n-t))*t := mul_le_mul_of_nonneg_left hb hfactor
  61     _ ≤ t := by nlinarith

  63 theorem rootedSubset_intersection_tail {V : Type*} [Fintype V] [DecidableEq V]
  64     (T A : Finset V) (k : ℕ) (hk : T.card+k≤Fintype.card V) (hkpos : 0<k)
  65     [Nonempty (RootedSubset V T k)] [Nonempty (FixedSubset {x : V // x∉T} k)]
  66     (s : ℝ) (hs : 0≤s) :
  67     (FiniteLaw.uniform (RootedSubset V T k)).probability
  68       (fun S => s+T.card≤|((A∩S.val.val).card:ℝ)-
  69         (A.card:ℝ)*(T.card+k:ℝ)/Fintype.card V|)≤
  70       2*Real.exp (-2*s^2/k) := by
  71   let Y := {x : V // x∉T}
  72   let B : Finset Y := A.subtype (fun x => x∉T)
  73   have hY : Fintype.card Y=Fintype.card V-T.card := by
  74     rw [Fintype.card_subtype_compl]
  75     simp
  76   have hky : k≤Fintype.card Y := by rw [hY];omega
  77   have hsum : (A∩T).card+B.card=A.card := by
  78     dsimp only [B]
  79     rw [Finset.card_subtype]
  80     have he : A.filter (fun x => x∉T)=A\T := by ext x;simp
  81     rw [he]
  82     exact Finset.card_inter_add_card_sdiff A T
  83   have hqr : ((A∩T).card:ℝ)+(B.card:ℝ)=A.card := by exact_mod_cast hsum
  84   have hn : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr (by omega)
  85   have htn : (T.card:ℝ)<Fintype.card V := by exact_mod_cast (by omega : T.card<Fintype.card V)
  86   have hkt : (k:ℝ)≤Fintype.card V-T.card := by
  87     have hh : (T.card:ℝ)+k≤Fintype.card V := by exact_mod_cast hk
  88     linarith
  89   have hnr : (Fintype.card Y:ℝ)=Fintype.card V-T.card := by
  90     rw [hY,Nat.cast_sub (by omega)]
  91   have hmean : |((A∩T).card:ℝ)+(B.card:ℝ)*k/Fintype.card Y-
  92       (A.card:ℝ)*(T.card+k:ℝ)/Fintype.card V|≤T.card := by
  93     have hh := rooted_subset_mean_error (a:=(A.card:ℝ)) (q:=((A∩T).card:ℝ)) hn
  94       (Nat.cast_nonneg _) htn (Nat.cast_nonneg k) hkt (Nat.cast_nonneg _)
  95       (by exact_mod_cast A.card_le_univ) (Nat.cast_nonneg _)
  96       (by exact_mod_cast Finset.card_le_card (Finset.inter_subset_right (s₁:=A) (s₂:=T)))
  97     rw [hnr]
  98     have hb : (B.card:ℝ)=(A.card:ℝ)-(A∩T).card := by linarith
  99     simpa only [hb] using hh
 100   rw [← finiteLaw_uniform_map_equiv (rootedSubsetEquiv T k).symm,finiteLaw_map_probability]
 101   apply le_trans (finiteLaw_probability_mono _ _ _ ?_)
 102     (fixedSubset_intersection_tail_at_size hky B s hs)
 103   intro S hS
 104   change s+T.card≤|((A∩(rootedSubsetUnion T S).val.val).card:ℝ)-
 105     (A.card:ℝ)*(T.card+k:ℝ)/Fintype.card V| at hS
 106   rw [rootedSubset_intersection_count T A S,Nat.cast_add] at hS
 107   have hb := abs_sub_le (((A∩T).card:ℝ)+((B∩S.val).card:ℝ))
 108     (((A∩T).card:ℝ)+(B.card:ℝ)*k/Fintype.card Y)
 109     ((A.card:ℝ)*(T.card+k:ℝ)/Fintype.card V)
 110   have he : ((A∩T).card:ℝ)+((B∩S.val).card:ℝ)-
 111       (((A∩T).card:ℝ)+(B.card:ℝ)*k/Fintype.card Y)=
 112       ((B∩S.val).card:ℝ)-(B.card:ℝ)*k/Fintype.card Y := by ring
 113   rw [he] at hb
 114   change s≤|((B∩S.val).card:ℝ)-(B.card:ℝ)*k/Fintype.card Y|
 115   change s+T.card≤|((A∩T).card:ℝ)+((B∩S.val).card:ℝ)-
