  13 def distinctCellEmbeddingEquiv {I V : Type*} [Fintype V] {k : ℕ}
  14     {d : Fin k → ℕ} (P : FixedPartition V k d) (a : I ↪ Fin k) :
  15     {f : I ↪ V // ∀ i, P.1 (f i) = a i} ≃ (∀ i, {v : V // P.1 v = a i}) where
  16   toFun f i := ⟨f.1 i, f.2 i⟩
  17   invFun f := ⟨⟨fun i => (f i).1, by
  18     intro i j h
  19     apply a.injective
  20     calc
  21       a i = P.1 (f i).1 := (f i).2.symm
  22       _ = P.1 (f j).1 := congrArg P.1 h
  23       _ = a j := (f j).2⟩, fun i => (f i).2⟩
  24   left_inv f := by apply Subtype.ext; ext i; rfl
  25   right_inv f := by funext i; rfl

  28     {k : ℕ} {d : Fin k → ℕ} (P : FixedPartition V k d) (a : I ↪ Fin k) :
  29     Fintype.card {f : I ↪ V // ∀ i, P.1 (f i) = a i} = ∏ i, d (a i) := by

  33 theorem uniformEmbedding_cell_probability {I V : Type*} [Fintype I] [Fintype V]
  34     [Nonempty (I ↪ V)] {k : ℕ} {d : Fin k → ℕ}
  35     (P : FixedPartition V k d) (a : I ↪ Fin k) :
  36     (FiniteLaw.uniform (I ↪ V)).probability (fun f => ∀ i, P.1 (f i) = a i) =
  37       ((∏ i, d (a i) : ℕ) : ℝ) / (Fintype.card V).descFactorial (Fintype.card I) := by
  38   rw [finiteLaw_uniform_probability, Fintype.card_embedding_eq]
  39   congr 1
  40   norm_cast
  41   convert! distinctCell_embedding_count P a using 1
  42   congr 1
  43   exact Subsingleton.elim _ _

  45 theorem fixedPartition_joint_placement {I V : Type*} [Fintype I] [Fintype V]
  46     {k : ℕ} {d : Fin k → ℕ} [Nonempty (FixedPartition V k d)]
  47     (x : I ↪ V) (a : I ↪ Fin k) :
  48     (FiniteLaw.uniform (FixedPartition V k d)).probability
  49       (fun Q => ∀ i, Q.1 (x i) = a i) =
  50       ((∏ i, d (a i) : ℕ) : ℝ) / (Fintype.card V).descFactorial (Fintype.card I) := by
  51   letI : Nonempty (I ↪ V) := ⟨x⟩
  52   let P := Classical.choice (inferInstance : Nonempty (FixedPartition V k d))
  53   have h := uniformEmbedding_cell_probability P a
  54   rw [← uniformPermutation_inverse_images x, finiteLaw_map_probability] at h
  55   rw [← fixedPartition_uniform P, finiteLaw_map_probability]
  56   exact h
