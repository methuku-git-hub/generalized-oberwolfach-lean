  17 def SuccessorPartitionGood {V I : Type*} [Fintype V] {k : ℕ} {d : Fin k → ℕ}
  18     (G : SimpleGraph V) (π : Equiv.Perm V) (a : I → Fin k)
  19     (m : ℕ) (D η : ℝ) (P : FixedPartition V k d) : Prop :=
  20   ∀ s b v, |(graphCellDegree G (choosePermutation π b) P v (a s) : ℝ) -
  21     D * ((m : ℝ) / Fintype.card V)| < η * m

  23 theorem successorPartitionGood_probability {V I : Type*} [Fintype V] [Nonempty V] [Fintype I]
  24     {k : ℕ} {d : Fin k → ℕ} [Nonempty (FixedPartition V k d)]
  25     (G : SimpleGraph V) (π : Equiv.Perm V) (a : I ↪ Fin k)
  26     (m : ℕ) (hm : 0 < m) (hsize : ∀ s, d (a s) = m) (D ε η : ℝ)
  27     (hη : 0 < η) (hε : ε ≤ η / 2)
  28     (hdeg : ∀ v, |(G.degree v : ℝ) - D| ≤ ε * Fintype.card V) :
  29     1 - 4 * Fintype.card I * Fintype.card V * Real.exp (-η ^ 2 * m / 2) ≤
  30       (FiniteLaw.uniform (FixedPartition V k d)).probability
  31         (SuccessorPartitionGood G π a m D η) := by
  32   have h := graphPartition_common_good G (choosePermutation π) (Finset.univ.map a) m hm
  33     (by intro z hz; obtain ⟨s,_,rfl⟩ := Finset.mem_map.mp hz; exact hsize s)
  34     D ε η hη hε hdeg
  35   have he : (fun P : FixedPartition V k d => ∀ z ∈ Finset.univ.map a, ∀ b v,
  36       |(graphCellDegree G (choosePermutation π b) P v z : ℝ) -
  37         D * ((m : ℝ) / Fintype.card V)| < η * m) =
  38         SuccessorPartitionGood G π a m D η := by
  39     funext P
  40     apply propext
  41     constructor
  42     · intro hp s; exact hp (a s) (Finset.mem_map.mpr ⟨s,Finset.mem_univ _,rfl⟩)
  43     · intro hp z hz
  44       obtain ⟨s,_,rfl⟩ := Finset.mem_map.mp hz
  45       exact hp s
  46   rw [he] at h
  47   simp only [Finset.card_map, Finset.card_univ, Fintype.card_bool] at h
  48   convert h using 1 <;> ring

  50 theorem successorPartitionGood_interface_degrees {V : Type*} [Fintype V]
  51     {ℓ k : ℕ} {d : Fin k → ℕ} (G : SimpleGraph V) (π : Equiv.Perm V)
  52     (b : Fin (ℓ + 2) → Bool)
  53     (a : Fin (ℓ + 2) ⊕ Fin (ℓ + 2) → Fin k) (m : ℕ) (D η : ℝ)
  54     (P : FixedPartition V k d) (hgood : SuccessorPartitionGood G π a m D η P) :
  55     (∀ s v, |(graphCellDegree G (Equiv.refl V) P v (a s) : ℝ) -
  56       D * ((m : ℝ) / Fintype.card V)| ≤ η * m) ∧
  57     (∀ i v, |(graphCellDegree G (choosePermutation π (b i)) P v
  58       (a (Sum.inl (finRotate (ℓ + 2) i))) : ℝ) -
  59       D * ((m : ℝ) / Fintype.card V)| ≤ η * m) := by
  60   constructor
  61   · intro s v
  62     exact (hgood s false v).le
  63   · intro i v
