  13 theorem OrdinaryTransferConfiguration.labels_injective {V U : Type*} {r : ℕ} {hr : 3≤r}
  14     {Q F : SimpleGraph V} {H : V → U → Prop} {v : V} {u u' : U} :
  15     Function.Injective (fun T : OrdinaryTransferConfiguration hr Q F H v u u' =>
  16       (T.labels : Fin r × Fin r → V)) := by
  17   intro T T' h
  18   have he : T.labels=T'.labels := Function.Embedding.ext (congrFun h)
  19   cases T
  20   cases T'
  21   cases he
  22   rfl

  24 theorem ordinary_transfer_configuration_root_count {V U : Type*} [Fintype V] [DecidableEq V]
  25     {r : ℕ} {hr : 3≤r} {Q F : SimpleGraph V} {H : V → U → Prop}
  26     {v : V} {u u' : U} (S : Finset V) :
  27     (∑ T : OrdinaryTransferConfiguration hr Q F H v u u',
  28       if (↑S:Set V)⊆Set.range T.labels then (1:ℝ) else 0)≤
  29       ((r:ℝ)^2)^S.card*(Fintype.card V:ℝ)^(r^2-S.card) := by
  30   let g (f : Fin r × Fin r → V) := if (∀ z∈S,z∈Set.range f) then (1:ℝ) else 0
  31   have hg (f : Fin r × Fin r → V) : 0≤g f := by dsimp [g];split_ifs <;> norm_num
  32   calc
  33     _ = ∑ T : OrdinaryTransferConfiguration hr Q F H v u u',g T.labels := by
  34       apply Finset.sum_congr rfl
  35       intro T _
  36       dsimp only [g]
  37       simp only [Set.subset_def,Finset.mem_coe]
  38     _ ≤ ∑ f : Fin r × Fin r → V,g f :=
  39       finite_injective_sum_le _ OrdinaryTransferConfiguration.labels_injective g hg
  40     _ ≤ _ := by
  41       have hh := functions_cover_roots_count ((⟨0,by omega⟩,⟨0,by omega⟩):Fin r × Fin r) S
  42       simpa only [Fintype.card_prod,Fintype.card_fin,← pow_two,Nat.cast_pow] using hh

  44 theorem OrdinaryTransferConfiguration.erased_rim_subset {V U : Type*} [Fintype V] [DecidableEq V]
  45     {r : ℕ} {hr : 3≤r} {Q F : SimpleGraph V} {H : V → U → Prop} {v : V} {u u' : U}
  46     (T : OrdinaryTransferConfiguration hr Q F H v u u') (D : OrdinaryWheel Q F H r)
  47     (hD : T.weight D≠0) :
  48     (↑((ordinaryWheelRim D).support.toFinset.erase v):Set V)⊆Set.range T.labels := by

  54 theorem ordinary_transfer_coefficient_sum_bound {V U : Type*} [Fintype V] [DecidableEq V]
  55     {r : ℕ} {hr : 3≤r} {Q F : SimpleGraph V} {H : V → U → Prop}
  56     {v : V} {u u' : U} (D : OrdinaryWheel Q F H r) :
  57     (∑ T : OrdinaryTransferConfiguration hr Q F H v u u',|T.weight D|)≤
  58       6*((r:ℝ)^2)^((ordinaryWheelRim D).support.toFinset.erase v).card*
  59         (Fintype.card V:ℝ)^(r^2-((ordinaryWheelRim D).support.toFinset.erase v).card) := by
  60   let S := (ordinaryWheelRim D).support.toFinset.erase v
  61   have hb (T : OrdinaryTransferConfiguration hr Q F H v u u') :
  62       |T.weight D|≤6*(if (↑S:Set V)⊆Set.range T.labels then (1:ℝ) else 0) := by
  63     by_cases hD : T.weight D=0
  64     · rw [hD,abs_zero]
  65       split_ifs <;> norm_num
  66     · rw [if_pos (T.erased_rim_subset D hD),mul_one]
  67       exact T.coefficient_bound D
  68   calc

  70         6*(if (↑S:Set V)⊆Set.range T.labels then (1:ℝ) else 0) :=
  71       Finset.sum_le_sum (fun T _ => hb T)
  72     _ = 6*∑ T : OrdinaryTransferConfiguration hr Q F H v u u',
  73         if (↑S:Set V)⊆Set.range T.labels then (1:ℝ) else 0 := by rw [Finset.mul_sum]
  74     _ ≤ _ := by
  75       have hh := mul_le_mul_of_nonneg_left (ordinary_transfer_configuration_root_count
  76         (hr:=hr) (Q:=Q) (F:=F) (H:=H) (v:=v) (u:=u) (u':=u') S) (by norm_num : (0:ℝ)≤6)
  77       simpa only [mul_assoc] using hh
