  13 theorem cyclic_path_weighting_with_formula {k : ℕ} {X : Type*} [Fintype X]
  14     {P : Fin (k + 2) → Type*} [∀ i, Fintype (P i)]
  15     (m : Fin (k + 2) → X → ℝ) (K : Fin (k + 2) → X → X → ℝ)
  16     (hm : ∀ i x, 0 < m i x)
  17     (hK : ∀ i, K i ∈ transportMatrices (m i) (m (finRotate (k + 2) i)))
  18     (c : Fin (k + 2)) (Q : X → X → ℝ)
  19     (hQ : ∀ x z, Q x z =
  20       walkTransport m K (finRotate (k + 2)) (finRotate (k + 2) c) (k + 1) z x)
  21     (B : ℝ) (hB : 0 < B) (hms : ∀ i, ∑ x, m i x = B)
  22     (w : ∀ i, P i → ℝ) (hw : ∀ i p, 0 ≤ w i p)
  23     (hws : ∀ i, ∑ p, w i p = B)
  24     (f : ∀ i, P i → X × X)
  25     (hf : ∀ i x z, (∑ p, if f i p = (x, z) then w i p else 0) =
  26       if i = c then Q x z else K i x z) :
  27     ∃ ω : (∀ i, P i) → ℝ,
  28       (∀ p, 0 ≤ ω p) ∧ (∑ p, ω p) = B ∧
  29       (∀ i q, (∑ p, if p i = q then ω p else 0) = w i q) ∧
  30       (∀ p, 0 < ω p → ∀ i,
  31         (f i (p i)).2 = (f (finRotate (k + 2) i) (p (finRotate (k + 2) i))).1) ∧
  32       (∀ p, (∀ i, (f i (p i)).2 =
  33           (f (finRotate (k + 2) i) (p (finRotate (k + 2) i))).1) →
  34         ω p = chainJoint m K (finRotate (k + 2)) (finRotate (k + 2) c) (k + 1)
  35           (fun j => (f (cyclicChainOrder c j) (p (cyclicChainOrder c j))).1) *
  36         ∏ i, w i (p i) / (if i = c then Q (f i (p i)).1 (f i (p i)).2
  37           else K i (f i (p i)).1 (f i (p i)).2)) := by
  38   let J := chainJoint m K (finRotate (k + 2)) (finRotate (k + 2) c) (k + 1)
  39   have hJ0 (v : Fin (k + 2) → X) : 0 ≤ J v :=
  40     chainJoint_nonneg m K (finRotate (k + 2)) (fun i x => (hm i x).le)
  41       (fun i x z => (hK i).1 x z) _ _ v
  42   have hJs : ∑ v, J v = B := by
  43     rw [chainJoint_total m K (finRotate (k + 2)) hm hK, hms]
  44   have hJm (i : Fin (k + 2)) (e : X × X) :
  45       (∑ v, if cyclicBoundaryPair c i v = e then J v else 0) =
  46         ∑ p, if f i p = e then w i p else 0 := by
  47     rcases e with ⟨x, z⟩
  48     rw [hf i x z]
  49     exact complementaryBoundaryJoint_marginal m K hm hK c Q hQ i x z
  50   obtain ⟨W, hW, htotal, hboundary, hpaths, hsupport, hformula⟩ :=
  51     weighted_conditional_realization_with_formula J w B hB hJ0 hw hJs hws
  52       (cyclicBoundaryPair c) f (by
  53         intro i e
  54         convert hJm i e using 1 <;>
  55           (apply Finset.sum_congr rfl; intro t _; split_ifs <;> rfl))
  56   let ω (p : ∀ i, P i) := ∑ v, W ⟨v, p⟩
  57   refine ⟨ω, ?_, ?_, ?_, ?_, ?_⟩
  58   · intro p
  59     exact Finset.sum_nonneg fun v _ => hW ⟨v, p⟩
  60   · dsimp only [ω]
  61     rw [Finset.sum_comm, ← Fintype.sum_sigma]
  62     exact htotal
  63   · intro i q
  64     change (∑ p, if p i = q then (∑ v, W ⟨v, p⟩) else 0) = _
  65     have hid (p : ∀ i, P i) :
  66         (if p i = q then (∑ v, W ⟨v, p⟩) else 0) =
  67         ∑ v, if p i = q then W ⟨v, p⟩ else 0 := by
  68       split_ifs <;> simp
  69     simp_rw [hid]
  70     rw [Finset.sum_comm]
  71     have hh := hpaths i q
  72     rw [Fintype.sum_sigma] at hh
  73     exact hh
  74   · intro p hp i
  75     by_contra hbad
  76     have hz (v : Fin (k + 2) → X) : W ⟨v, p⟩ = 0 := by
  77       by_contra hne
  78       have hv : 0 < W ⟨v, p⟩ := lt_of_le_of_ne (hW ⟨v, p⟩) (Ne.symm hne)
  79       have hs := hsupport v p hv
  80       apply hbad
  81       rw [hs i, hs (finRotate (k + 2) i)]
  82       exact cyclicBoundaryPair_consistent c i v
  83     have hzero : ω p = 0 := Finset.sum_eq_zero fun v _ => hz v
  84     rw [hzero] at hp
  85     exact (lt_irrefl 0) hp
  86   · intro p hp
  87     let v₀ (j : Fin (k + 2)) := (f (cyclicChainOrder c j) (p (cyclicChainOrder c j))).1
  88     have hzero (v : Fin (k + 2) → X) (hv : v ≠ v₀) : W ⟨v, p⟩ = 0 := by
  89       by_contra hne
  90       have hpos : 0 < W ⟨v, p⟩ := lt_of_le_of_ne (hW ⟨v, p⟩) (Ne.symm hne)
  91       apply hv
  92       funext j
  93       have hh := congrArg Prod.fst (hsupport v p hpos (cyclicChainOrder c j))
  94       simpa only [cyclicBoundaryPair, Equiv.symm_apply_apply, v₀] using hh.symm
  95     have hsum : ω p = W ⟨v₀, p⟩ := by
  96       exact Finset.sum_eq_single v₀ (fun v _ hv => hzero v hv)
  97         (fun h => False.elim (h (Finset.mem_univ v₀)))
  98     have hpair (i : Fin (k + 2)) : cyclicBoundaryPair c i v₀ = f i (p i) := by
  99       apply Prod.ext
 100       · change (f (cyclicChainOrder c ((cyclicChainOrder c).symm i))
 101           (p (cyclicChainOrder c ((cyclicChainOrder c).symm i)))).1 = _
 102         rw [(cyclicChainOrder c).apply_symm_apply i]
 103       · change (f (cyclicChainOrder c ((cyclicChainOrder c).symm (finRotate (k + 2) i)))
 104           (p (cyclicChainOrder c ((cyclicChainOrder c).symm (finRotate (k + 2) i))))).1 = _
 105         rw [(cyclicChainOrder c).apply_symm_apply]
 106         exact (hp i).symm
 107     rw [hsum, hformula]
 108     change J v₀ * _ = J v₀ * _
 109     apply congrArg (fun z => J v₀ * z)
 110     apply Finset.prod_congr rfl
 111     intro i _
 112     rw [hpair i]
 113     simp only [ite_true]
 114     apply congrArg (fun z => w i (p i) / z)
 115     convert hf i (f i (p i)).1 (f i (p i)).2 using 1 <;>
 116       (apply Finset.sum_congr rfl; intro q _; split_ifs <;> rfl)
