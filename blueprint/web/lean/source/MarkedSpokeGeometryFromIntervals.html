   9 theorem marked_spoke_geometry_from_marked_sets
  10     {V U:Type} [Fintype V] [DecidableEq V] [DecidableEq U]
  11     (π:Equiv.Perm V) (Y:U→Finset V) (H:Bool→V→U→Prop)
  12     (hmarked:∀ v u,H true v u↔v∈Y u)
  13     (hordinary:∀ v u,H false v u→v∉Y u ∧ v∉(Y u).image π):
  14     (∀ u v,¬(H false v u∧H true v u)) ∧
  15     (∀ u v z,H false v u→H true z u→v≠π z) := by
  16   constructor
  17   · intro u v h
  18     exact (hordinary v u h.1).1 ((hmarked v u).mp h.2)
  19   · intro u v z h0 hK hv
  20     have hz:z∈Y u:=(hmarked z u).mp hK
  21     apply (hordinary v u h0).2
  22     rw [hv]
  23     exact Finset.mem_image.mpr ⟨z,hz,rfl⟩

  25 theorem marked_spoke_geometry_restrict
  26     {V U W Z:Type} (π:Equiv.Perm V) (ρ:Equiv.Perm W)
  27     (f:W↪V) (g:Z→U) (hcomm:∀ x,f (ρ x)=π (f x)) (H:Bool→V→U→Prop)
  28     (htypes:∀ u v,¬(H false v u∧H true v u))
  29     (hsep:∀ u v z,H false v u→H true z u→v≠π z):
  30     (∀ u v,¬(H false (f v) (g u)∧H true (f v) (g u))) ∧
  31     (∀ u v z,H false (f v) (g u)→H true (f z) (g u)→v≠ρ z) := by
  32   constructor
  33   · intro u v
  34     exact htypes (g u) (f v)
  35   · intro u v z h0 hK heq
  36     apply hsep (g u) (f v) (f z) h0 hK
  37     rw [heq,hcomm]
  38 end Oberwolfach
