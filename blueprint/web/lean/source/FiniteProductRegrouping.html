  12 theorem finiteLaw_prod_map_left_probability
  13     {Ω Γ Ξ : Type*} [Fintype Ω] [Fintype Γ] [Fintype Ξ]
  14     (μ : FiniteLaw Ω) (ν : FiniteLaw Γ) (f : Ω→Ξ) (E : Ξ→Γ→Prop) :
  15     (μ.prod ν).probability (fun z=>E (f z.1) z.2)=
  16       ((μ.map f).prod ν).probability (fun z=>E z.1 z.2) := by
  17   have hleft : (μ.prod ν).probability (fun z=>E (f z.1) z.2)=
  18       ∑ γ,ν.mass γ*μ.probability (fun ω=>E (f ω) γ) := by
  19     unfold FiniteLaw.probability
  20     rw [Fintype.sum_prod_type,Finset.sum_comm]
  21     apply Finset.sum_congr rfl
  22     intro γ _
  23     rw [Finset.mul_sum]
  24     apply Finset.sum_congr rfl
  25     intro ω _
  26     simp only [FiniteLaw.prod]
  27     split_ifs <;> ring
  28   have hright : ((μ.map f).prod ν).probability (fun z=>E z.1 z.2)=
  29       ∑ γ,ν.mass γ*(μ.map f).probability (fun ξ=>E ξ γ) := by
  30     unfold FiniteLaw.probability
  31     rw [Fintype.sum_prod_type,Finset.sum_comm]
  32     apply Finset.sum_congr rfl
  33     intro γ _
  34     rw [Finset.mul_sum]
  35     apply Finset.sum_congr rfl
  36     intro ξ _
  37     simp only [FiniteLaw.prod]
  38     split_ifs <;> ring
  39   rw [hleft,hright]
  40   simp_rw [finiteLaw_map_probability]

  54 theorem finiteLaw_pi_prod_probability
  55     {I : Type*} [Fintype I] [DecidableEq I]
  56     {A B : I→Type*} [∀ i,Fintype (A i)] [∀ i,Fintype (B i)]
  57     (μ : ∀ i,FiniteLaw (A i)) (ν : ∀ i,FiniteLaw (B i))
  58     (E : (∀ i,A i)→(∀ i,B i)→Prop) :
  59     ((FiniteLaw.pi μ).prod (FiniteLaw.pi ν)).probability (fun z=>E z.1 z.2)=
  60       (FiniteLaw.pi (fun i=>(μ i).prod (ν i))).probability
  61         (fun σ=>E (fun i=>(σ i).1) (fun i=>(σ i).2)) := by
  62   let e : ((∀ i,A i)×(∀ i,B i))≃(∀ i,A i×B i) :=
  63     { toFun := fun z i=>(z.1 i,z.2 i)
  64       invFun := fun σ=>(fun i=>(σ i).1,fun i=>(σ i).2)
  65       left_inv := by intro z;cases z;rfl
  66       right_inv := by intro σ;rfl }
  67   unfold FiniteLaw.probability
  68   rw [← e.symm.sum_comp]
  69   apply Finset.sum_congr rfl
  70   intro σ _
  71   change (if E (fun i=>(σ i).1) (fun i=>(σ i).2) then
  72     (∏ i,(μ i).mass (σ i).1)*(∏ i,(ν i).mass (σ i).2) else 0)=
  73     if E (fun i=>(σ i).1) (fun i=>(σ i).2) then
  74       ∏ i,(μ i).mass (σ i).1*(ν i).mass (σ i).2 else 0
  75   rw [Finset.prod_mul_distrib]
