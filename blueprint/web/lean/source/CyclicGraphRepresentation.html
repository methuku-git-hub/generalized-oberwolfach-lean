  13 def leftCyclicInterface {k : ℕ} {X Y : Type*}
  14     (G : SimpleGraph ((Fin (k + 2) × X) ⊕ (Fin (k + 2) × Y)))
  15     (i : Fin (k + 2)) (x : X) (y : Y) : Prop :=
  16   G.Adj (Sum.inl (i, x)) (Sum.inr (i, y))
  17 
  18 def rightCyclicInterface {k : ℕ} {X Y : Type*}
  19     (G : SimpleGraph ((Fin (k + 2) × X) ⊕ (Fin (k + 2) × Y)))
  20     (i : Fin (k + 2)) (y : Y) (x : X) : Prop :=
  21   G.Adj (Sum.inr (i, y)) (Sum.inl (finRotate (k + 2) i, x))

  23 theorem cyclicGraph_interface_reconstruction {k : ℕ} {X Y : Type*}
  24     (G : SimpleGraph ((Fin (k + 2) × X) ⊕ (Fin (k + 2) × Y)))
  25     (hcyc : G ≤ cyclicInterfaceGraph (fun _ _ _ => True) (fun _ _ _ => True)) :
  26     G = cyclicInterfaceGraph (leftCyclicInterface G) (rightCyclicInterface G) := by
  27   ext a b
  28   have hcross (i j : Fin (k + 2)) (x : X) (y : Y) :
  29       G.Adj (Sum.inl (i, x)) (Sum.inr (j, y)) ↔
  30       (i = j ∧ leftCyclicInterface G i x y) ∨
  31         (i = finRotate (k + 2) j ∧ rightCyclicInterface G j y x) := by
  32     constructor
  33     · intro h
  34       have hc := hcyc h
  35       change (i = j ∧ True) ∨ (i = finRotate (k + 2) j ∧ True) at hc
  36       rcases hc with ⟨rfl, _⟩ | ⟨rfl, _⟩
  37       · exact Or.inl ⟨rfl, h⟩
  38       · exact Or.inr ⟨rfl, h.symm⟩
  39     · rintro (⟨rfl, h⟩ | ⟨rfl, h⟩)
  40       · exact h
  41       · exact h.symm
  42   cases a with
  43   | inl a =>
  44     cases b with
  45     | inl b =>
  46       change G.Adj (Sum.inl a) (Sum.inl b) ↔ False
  47       exact ⟨fun h => hcyc h, False.elim⟩
  48     | inr b => exact hcross a.1 b.1 a.2 b.2
  49   | inr a =>
  50     cases b with
  51     | inr b =>
  52       change G.Adj (Sum.inr a) (Sum.inr b) ↔ False
  53       exact ⟨fun h => hcyc h, False.elim⟩
  54     | inl b =>
  55       change G.Adj (Sum.inr a) (Sum.inl b) ↔ _
  56       rw [G.adj_comm]
  57       exact hcross b.1 a.1 b.2 a.2

  59 theorem cyclic_graph_has_fractional_cycles {k : ℕ} {X Y V : Type*}
  60     [Fintype X] [Fintype Y] [DecidableEq X] [DecidableEq Y] [Nonempty X]
  61     [Fintype V] [DecidableEq V]
  62     (G : SimpleGraph V)
  63     (e : ((Fin (k + 2) × X) ⊕ (Fin (k + 2) × Y)) ≃ V)
  64     (hcyc : G.comap e ≤
  65       cyclicInterfaceGraph (fun _ _ _ => True) (fun _ _ _ => True))
  66     (δ : ℝ) (hn : 0 < Fintype.card Y)
  67     (hthreshold : (k + 2 : ℝ) * Fintype.card Y ≤ (2 * (k + 2) - 1) * δ)
  68     (hdL : ∀ i x, δ ≤ (Finset.univ.filter (leftCyclicInterface (G.comap e) i x)).card)
  69     (hY : ∀ i y,
  70       (Finset.univ.filter (fun x => leftCyclicInterface (G.comap e) i x y)).card =
  71       (Finset.univ.filter (rightCyclicInterface (G.comap e) i y)).card)
  72     (hX : ∀ i z,
  73       (Finset.univ.filter (fun y => rightCyclicInterface (G.comap e) i y z)).card =
  74       (Finset.univ.filter (leftCyclicInterface (G.comap e) (finRotate (k + 2) i) z)).card) :
  75     HasFractionalDecomposition G (fun H => IsEmbeddedCycle H ((k + 2) * 2) ∧
  76       ∃ p : Fin (k + 2) → X × Y × X, H = (alternatingTupleGraph p).map e.toEmbedding) := by
  77   have hh := cyclicInterfaceGraph_has_fractional_cycles
  78     (leftCyclicInterface (G.comap e)) (rightCyclicInterface (G.comap e))
  79     δ hn hthreshold hdL hY hX
  80   rw [← cyclicGraph_interface_reconstruction (G.comap e) hcyc] at hh
  81   have hm := hh.map e
  82   have hG : (G.comap e).map e.toEmbedding = G := by
  83     ext u v
  84     simp only [← SimpleGraph.comap_symm _ e, SimpleGraph.comap_adj,
  85       Equiv.toEmbedding_apply, Equiv.apply_symm_apply]
  86   rw [hG] at hm
  87   apply hm.mono_property
  88   rintro H ⟨K, ⟨hK, p, rfl⟩, rfl⟩
  89   exact ⟨hK.map e.toEmbedding, p, rfl⟩
