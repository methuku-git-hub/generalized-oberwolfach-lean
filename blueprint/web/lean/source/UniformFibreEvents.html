  14 theorem uniform_fibre_probability_bound {Ω Λ : Type*} [Fintype Ω] [Nonempty Ω] [Fintype Λ]
  15     (f : Ω → Λ) (A : Ω → Prop) (ε : ℝ) (hε : 0 ≤ ε)
  16     (h : ∀ y (hy : Nonempty {ω : Ω // f ω=y}),
  17       letI := hy
  18       (FiniteLaw.uniform {ω : Ω // f ω=y}).probability (fun ω => A ω.val) ≤ ε) :
  19     (FiniteLaw.uniform Ω).probability A ≤ ε := by
  20   let μ := FiniteLaw.uniform Ω
  21   have hb (y : Λ) : μ.probability (fun ω => A ω ∧ f ω=y) ≤ μ.probability (fun ω => f ω=y)*ε := by
  22     by_cases hn : Nonempty {ω : Ω // f ω=y}
  23     · letI := hn
  24       have hh := uniform_probability_intersection (fun ω : Ω => f ω=y) A
  25       have he : (fun ω : Ω => A ω ∧ f ω=y)=(fun ω => f ω=y ∧ A ω) := by
  26         funext ω;exact propext and_comm
  27       rw [he]
  28       change (FiniteLaw.uniform Ω).probability _ ≤ _
  29       rw [hh]
  30       exact mul_le_mul_of_nonneg_left (h y hn) (finiteLaw_probability_nonneg _ _)
  31     · have hz : μ.probability (fun ω => A ω ∧ f ω=y)=0 := by
  32         unfold FiniteLaw.probability
  33         apply Finset.sum_eq_zero
  34         intro ω _
  35         have hne : f ω≠y := fun he => hn ⟨⟨ω,he⟩⟩
  36         simp [hne]
  37       rw [hz]
  38       exact mul_nonneg (finiteLaw_probability_nonneg _ _) hε
  39   have hsum : (∑ y,μ.probability (fun ω => f ω=y))=1 := by
  40     have hh := finiteLaw_probability_value_partition μ (fun _ => True) f
  41     have ht : μ.probability (fun _ => True)=1 := by simpa only [FiniteLaw.probability,ite_true] using μ.total
  42     simpa only [true_and,ht] using hh.symm
  43   calc
  44     _ = ∑ y,μ.probability (fun ω => A ω ∧ f ω=y) := finiteLaw_probability_value_partition μ A f
  45     _ ≤ ∑ y,μ.probability (fun ω => f ω=y)*ε := Finset.sum_le_sum (fun y _ => hb y)
  46     _ = ε := by rw [← Finset.sum_mul,hsum,one_mul]

  51     (hε : ∀ j∈tests,0 ≤ ε j)
  52     (h : ∀ j∈tests,∀ y (hy : Nonempty {ω : Ω // labels j ω=y}),
  53       letI := hy
  54       (FiniteLaw.uniform {ω : Ω // labels j ω=y}).probability (fun ω => Bad j ω.val) ≤ ε j) :
  55     1-∑ j∈tests,ε j ≤ (FiniteLaw.uniform Ω).probability (fun ω => ∀ j∈tests,¬Bad j ω) := by
