  18 theorem canonicalMarkedCycle_restrict_iff (PE : E → Prop) (PK : K → Prop)
  19     (hE : Function.Injective (fun e=>(tail e,head e)))
  20     (hK : Function.Injective (fun e=>(ktail e,khead e)))
  21     (C : MarkedCycleTuple tail head ktail khead m π b) :
  22     IsMarkedCycleTuple (fun e : {e // PE e}=>tail e.val) (fun e=>head e.val)
  23       (fun e : {e // PK e}=>ktail e.val) (fun e=>khead e.val) m π b C.val ↔
  24         ∀ i,Sum.elim PE PK (markedCycleTypedEdge tail head ktail khead m π b C i) := by
  25   constructor
  26   · intro h i
  27     refine Fin.lastCases ?_ (fun j=>?_) i
  28     · obtain ⟨e,he⟩ := h.2.2
  29       have hh := Classical.choose_spec C.property.2.2
  30       have heq : Classical.choose C.property.2.2=e.val := hK (Prod.ext (hh.1.trans he.1.symm) (hh.2.trans he.2.symm))
  31       simpa only [markedCycleTypedEdge,Fin.snoc_last,Sum.elim_inr,heq] using e.property
  32     · obtain ⟨e,he⟩ := h.2.1 j
  33       have hh := Classical.choose_spec (C.property.2.1 j)
  34       have heq : Classical.choose (C.property.2.1 j)=e.val := hE (Prod.ext (hh.1.trans he.1.symm) (hh.2.trans he.2.symm))
  35       simpa only [markedCycleTypedEdge,Fin.snoc_castSucc,Sum.elim_inl,heq] using e.property
  36   · intro h
  37     refine ⟨C.property.1,?_,?_⟩
  38     · intro j
  39       have hj := h j.castSucc
  40       simp only [markedCycleTypedEdge,Fin.snoc_castSucc,Sum.elim_inl] at hj
  41       exact ⟨⟨Classical.choose (C.property.2.1 j),hj⟩,Classical.choose_spec (C.property.2.1 j)⟩
  42     · have hj := h (Fin.last m)
  43       simp only [markedCycleTypedEdge,Fin.snoc_last,Sum.elim_inr] at hj
  44       exact ⟨⟨Classical.choose C.property.2.2,hj⟩,Classical.choose_spec C.property.2.2⟩

  46 theorem markedWheelSurvives_iff_typed_occurrences (PE : E → Prop) (PK : K → Prop)
  47     (hE : Function.Injective (fun e=>(tail e,head e)))
  48     (hK : Function.Injective (fun e=>(ktail e,khead e)))
  49     (W : MarkedWheel tail head ktail khead m π b H₀ Hₖ) :
  50     MarkedWheelSurvives PE PK H₀' Hₖ' W ↔
  51       ∀ e∈typedMarkedOccurrenceResources W,
  52         Sum.elim (Sum.elim PE PK) (fun s=>if s.1 then Hₖ' s.2.1 s.2.2 else H₀' s.2.1 s.2.2) e := by
  53   rw [MarkedWheelSurvives,canonicalMarkedCycle_restrict_iff PE PK hE hK]
  54   constructor
  55   · rintro ⟨hc,hh⟩ e he
  56     cases e with
  57     | inl e =>
  58       obtain ⟨i,rfl⟩ := (typedMarkedOccurrenceResources_rim W e).mp he
  59       exact hc i
  60     | inr s =>
  61       obtain ⟨i,rfl⟩ := (typedMarkedOccurrenceResources_spoke W s).mp he
  62       dsimp only [Sum.elim_inr]
  63       refine Fin.cases ?_ (fun j=>?_) i
  64       · simpa only [decide_true,ite_true] using hh.1
  65       · simpa only [Fin.succ_ne_zero,decide_false,Bool.false_eq_true,ite_false] using hh.2 j
  66   · intro h
  67     constructor
  68     · intro i
  69       apply h (Sum.inl (markedCycleTypedEdge tail head ktail khead m π b W.1 i))
  70       rw [typedMarkedOccurrenceResources_rim]
  71       exact ⟨i,rfl⟩
  72     · have hh (i : Fin (m+1)) := h (Sum.inr (decide (i=0),W.1.val i,W.2.val))
  73         ((typedMarkedOccurrenceResources_spoke W _).mpr ⟨i,rfl⟩)
  74       constructor
  75       · simpa only [Sum.elim_inr,decide_true,ite_true] using hh 0
