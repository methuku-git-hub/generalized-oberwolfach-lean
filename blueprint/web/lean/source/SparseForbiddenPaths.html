  14 theorem sparse_path_prefix_within (Q F : SimpleGraph V) (A W Z Ω : Finset V) (hAΩ : A⊆Ω) (x : V) (hxΩ : x∈Ω) (hx : x∈Z)
  15     (hZ : ∀ v∈Z,∀ w∈Z,¬F.Adj v w) (k : ℕ) (L b : ℝ)
  16     (hL : 0≤L) (hF : ∀ v,(F.degree v:ℝ)≤L)
  17     (hcount : ∀ v∈Ω,b≤((A∩Q.neighborFinset v∩Q.neighborFinset v).card:ℝ))
  18     (hbudget : W.card+((Z.card:ℝ)+k+1)*(L+1)<b) :
  19     ∃ z,∃ p : Q.Walk x z,p.IsPath ∧ p.length=k ∧
  20       (∀ v∈p.support,v=x ∨ v∈A ∧ v∉W ∧ v∉Z) ∧
  21       (∀ v∈p.support,∀ w∈p.support,¬F.Adj v w) ∧
  22       (∀ v∈p.support,∀ w∈Z,¬F.Adj v w) := by
  23   induction k with
  24   | zero =>
  25     refine ⟨x,SimpleGraph.Walk.nil,by simp,by simp,?_,?_,?_⟩
  26     · intro v hv
  27       have hvx : v=x:=by simpa using hv
  28       exact Or.inl hvx
  29     · intro v hv w hw
  30       have hvx : v=x:=by simpa using hv
  31       have hwx : w=x:=by simpa using hw
  32       rw [hvx,hwx]
  33       exact F.loopless.irrefl x
  34     · intro v hv w hw
  35       have hvx : v=x:=by simpa using hv
  36       rw [hvx]
  37       exact hZ x hx w hw
  38   | succ k ih =>
  39     have hsmall : W.card+((Z.card:ℝ)+k+1)*(L+1)<b:=by
  40       push_cast at hbudget
  41       nlinarith only [hbudget,hL]
  42     obtain ⟨z,p,hp,hlen,hclass,hind,hroot⟩:=ih hsmall
  43     have hsupport : p.support.toFinset.card≤k+1:=by
  44       exact (List.toFinset_card_le _).trans_eq (by rw [p.length_support,hlen])
  45     have hused : ((Z∪p.support.toFinset).card:ℝ)≤Z.card+k+1:=by
  46       have hh:=(Finset.card_union_le Z p.support.toFinset).trans (Nat.add_le_add_left hsupport Z.card)
  47       exact_mod_cast hh
  48     have hb' : W.card+((Z∪p.support.toFinset).card:ℝ)*(L+1)<b:=by
  49       have hh:=mul_le_mul_of_nonneg_right hused (show 0≤L+1 by linarith)
  50       linarith only [hh,hsmall]
  51     obtain ⟨v,hvA,hvW,hvZ,hzv,_,hFv⟩:=sparse_common_neighbour_avoiding Q F A W
  52       (Z∪p.support.toFinset) z z L b hL hF (hcount z (by
  53         rcases hclass z p.end_mem_support with he|hz
  54         · exact he ▸ hxΩ
  55         · exact hAΩ hz.1)) hb'
  56     have hvp : v∉p.support:=fun h=>hvZ (Finset.mem_union_right _ (List.mem_toFinset.mpr h))
  57     have hvZ' : v∉Z:=fun h=>hvZ (Finset.mem_union_left _ h)
  58     refine ⟨v,p.concat hzv,hp.concat hvp hzv,by simp [hlen],?_,?_,?_⟩
  59     · intro w hw
  60       rw [SimpleGraph.Walk.support_concat] at hw
  61       rcases List.mem_append.mp hw with hw|hw
  62       · exact hclass w hw
  63       · have hwv:=List.mem_singleton.mp hw;subst w;exact Or.inr ⟨hvA,hvW,hvZ'⟩
  64     · intro w hw z' hz'
  65       simp only [SimpleGraph.Walk.support_concat,List.mem_append,List.mem_singleton] at hw hz'
  66       rcases hw with hw|he <;> rcases hz' with hz'|he'
  67       · exact hind w hw z' hz'
  68       · rw [he'];exact fun he=>hFv w (Finset.mem_union_right _ (List.mem_toFinset.mpr hw)) he.symm
  69       · rw [he];exact hFv z' (Finset.mem_union_right _ (List.mem_toFinset.mpr hz'))
  70       · rw [he,he'];exact F.loopless.irrefl v
  71     · intro w hw z' hz'
  72       simp only [SimpleGraph.Walk.support_concat,List.mem_append,List.mem_singleton] at hw
  73       rcases hw with hw|rfl
  74       · exact hroot w hw z' hz'
  75       · exact hFv z' (Finset.mem_union_left _ hz')

  77 theorem sparse_fixed_length_forbidden_path_within (Q F : SimpleGraph V) (A W Z Ω : Finset V) (hAΩ : A⊆Ω)
  78     (x y : V) (hxΩ : x∈Ω) (hyΩ : y∈Ω) (hxy : x≠y) (hx : x∈Z) (hy : y∈Z)
  79     (hZ : ∀ v∈Z,∀ w∈Z,¬F.Adj v w) (k : ℕ) (hk : 2≤k) (L b : ℝ)
  80     (hL : 0≤L) (hF : ∀ v,(F.degree v:ℝ)≤L)
  81     (hcount : ∀ v∈Ω,∀ w∈Ω,b≤((A∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  82     (hbudget : W.card+((Z.card:ℝ)+k)*(L+1)<b) :
  83     ∃ p : Q.Walk x y,p.IsPath ∧ p.length=k ∧
  84       (∀ v∈p.support,v=x ∨ v=y ∨ v∈A ∧ v∉W ∧ v∉Z) ∧
  85       (∀ v∈p.support,∀ w∈p.support,¬F.Adj v w) ∧
  86       (∀ v∈p.support,∀ w∈Z,¬F.Adj v w) := by
  87   have hkR : ((k-2:ℕ):ℝ)+2=k:=by exact_mod_cast (show k-2+2=k by omega)
  88   obtain ⟨z,p,hp,hlen,hclass,hind,hroot⟩:=sparse_path_prefix_within Q F A W Z Ω hAΩ x hxΩ hx hZ (k-2) L b hL hF
  89     (fun v hv=>hcount v hv v hv) (by nlinarith only [hbudget,hkR,hL])
  90   have hsupport : p.support.toFinset.card≤k-1:=by
  91     have hh:=List.toFinset_card_le p.support
  92     rw [p.length_support,hlen] at hh
  93     omega
  94   have hused : ((Z∪p.support.toFinset).card:ℝ)≤Z.card+k:=by
  95     have hh:=(Finset.card_union_le Z p.support.toFinset).trans (Nat.add_le_add_left (hsupport.trans (show k-1≤k by omega)) Z.card)
  96     exact_mod_cast hh
  97   have hb' : W.card+((Z∪p.support.toFinset).card:ℝ)*(L+1)<b:=by
  98     have hh:=mul_le_mul_of_nonneg_right hused (show 0≤L+1 by linarith)
  99     linarith only [hh,hbudget]
 100   obtain ⟨v,hvA,hvW,hvZ,hzv,hyv,hFv⟩:=sparse_common_neighbour_avoiding Q F A W
 101     (Z∪p.support.toFinset) z y L b hL hF (hcount z (by
 102       rcases hclass z p.end_mem_support with he|hz
 103       · exact he ▸ hxΩ
 104       · exact hAΩ hz.1) y hyΩ) hb'
 105   have hvp : v∉p.support:=fun h=>hvZ (Finset.mem_union_right _ (List.mem_toFinset.mpr h))
 106   have hvZ' : v∉Z:=fun h=>hvZ (Finset.mem_union_left _ h)
 107   have hyp : y∉p.support:=by
 108     intro h
 109     rcases hclass y h with he|hrest
 110     · exact hxy he.symm
 111     · exact hrest.2.2 hy
 112   have hyp' : y∉(p.concat hzv).support:=by
 113     simp only [SimpleGraph.Walk.support_concat,List.mem_append,List.mem_singleton,not_or]
 114     exact ⟨hyp,fun he=>hvZ' (he ▸ hy)⟩
 115   have hlist (w:V) : w∈((p.concat hzv).concat hyv.symm).support ↔ w∈p.support ∨ w=v ∨ w=y:=by
 116     simp only [SimpleGraph.Walk.support_concat,List.mem_append,List.mem_singleton,or_assoc]
 117   have hind' : ∀ w∈p.support,¬F.Adj w v:=fun w hw he=>hFv w (Finset.mem_union_right _ (List.mem_toFinset.mpr hw)) he.symm
 118   have hvin : ∀ w∈Z,¬F.Adj v w:=fun w hw=>hFv w (Finset.mem_union_left _ hw)
 119   refine ⟨(p.concat hzv).concat hyv.symm,(hp.concat hvp hzv).concat hyp' hyv.symm,?_,?_,?_,?_⟩
 120   · simp only [SimpleGraph.Walk.length_concat,hlen];omega
 121   · intro w hw
 122     rcases (hlist w).mp hw with hw|rfl|rfl
 123     · rcases hclass w hw with he|hc
 124       · exact Or.inl he
 125       · exact Or.inr (Or.inr hc)
 126     · exact Or.inr (Or.inr ⟨hvA,hvW,hvZ'⟩)
 127     · exact Or.inr (Or.inl rfl)
 128   · intro w hw t ht
 129     rcases (hlist w).mp hw with hw'|he|he <;> rcases (hlist t).mp ht with ht'|he'|he'
 130     · exact hind w hw' t ht'
 131     · rw [he'];exact hind' w hw'
 132     · rw [he'];exact hroot w hw' y hy
 133     · rw [he];exact fun h=>hind' t ht' h.symm
 134     · rw [he,he'];exact F.loopless.irrefl v
 135     · rw [he,he'];exact hvin y hy
 136     · rw [he];exact fun h=>hroot t ht' y hy h.symm
 137     · rw [he,he'];exact fun h=>hvin y hy h.symm
 138     · rw [he,he'];exact F.loopless.irrefl y
 139   · intro w hw t ht
 140     rcases (hlist w).mp hw with hw'|he|he
 141     · exact hroot w hw' t ht
 142     · rw [he];exact hvin t ht
 143     · rw [he];exact hZ y hy t ht
