  12 theorem ordinary_cycle_pair_weight_bound {V:Type} [Fintype V] [DecidableEq V]
  13     (G:SimpleGraph V) (r:ℕ) (u v:V) (huv:u≠v)
  14     (hn:0<Fintype.card V) (B:ℝ) (hB:0≤B)
  15     (w:OrdinaryCycle G r→ℝ)
  16     (hw:∀ C,w C≤B/(Fintype.card V:ℝ)^(r-2)):
  17     (∑ C:OrdinaryCycle G r,
  18       if u∈C.val.val.support ∧ v∈C.val.val.support then w C else 0)≤(r:ℝ)^2*B := by
  19   let b:=B/(Fintype.card V:ℝ)^(r-2)
  20   have hb:0≤b:=by dsimp [b];positivity
  21   have hn0:(Fintype.card V:ℝ)^(r-2)≠0:=by exact ne_of_gt (pow_pos (by exact_mod_cast hn) _)
  22   have hcount:
  23     (∑ C:OrdinaryCycle G r,
  24       if u∈C.val.val.support ∧ v∈C.val.val.support then (1:ℝ) else 0)≤
  25         (r:ℝ)^2*(Fintype.card V:ℝ)^(r-2) := by
  26     apply (finite_subtype_sum_le (fun C:EmbeddedCycleFamily V r=>C.val≤G)
  27       (fun C=>if u∈C.val.support ∧ v∈C.val.support then (1:ℝ) else 0)
  28       (fun C=>by split_ifs <;> norm_num)).trans
  29     simpa only [embedded_cycle_support_range] using ordinary_cycle_two_vertex_count r u v huv
  30   calc
  31     _ ≤ ∑ C:OrdinaryCycle G r,
  32       (if u∈C.val.val.support ∧ v∈C.val.val.support then (1:ℝ) else 0)*b := by
  33         apply Finset.sum_le_sum
  34         intro C _
  35         by_cases h:u∈C.val.val.support ∧ v∈C.val.val.support
  36         · simpa [h,b] using hw C
  37         · simp [h]
  38     _ = (∑ C:OrdinaryCycle G r,
  39       if u∈C.val.val.support ∧ v∈C.val.val.support then (1:ℝ) else 0)*b := by rw [Finset.sum_mul]
  40     _ ≤ ((r:ℝ)^2*(Fintype.card V:ℝ)^(r-2))*b:=mul_le_mul_of_nonneg_right hcount hb
  41     _ = (r:ℝ)^2*B:=by dsimp [b];field_simp
  42 end Oberwolfach
