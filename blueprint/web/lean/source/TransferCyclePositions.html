  12 theorem cycleGraph_adj_natural_cases (r : ℕ) (hr : 3 ≤ r) (i j : Fin r)
  13     (h : (SimpleGraph.cycleGraph r).Adj i j) :
  14     (i:ℕ)+1=j ∨ (j:ℕ)+1=i ∨ ((i:ℕ)=0 ∧ (j:ℕ)=r-1) ∨ ((j:ℕ)=0 ∧ (i:ℕ)=r-1) := by
  15   have hi := i.isLt
  16   have hj := j.isLt
  17   rcases (cycleGraph_modular_adj (by omega) i j).mp h with h|h
  18   · by_cases hlt : (i:ℕ)+1 < r
  19     · rw [Nat.mod_eq_of_lt hlt] at h
  20       exact Or.inl h
  21     · have he : (i:ℕ)+1=r := by omega
  22       rw [he,Nat.mod_self] at h
  23       exact Or.inr (Or.inr (Or.inr ⟨h.symm,by omega⟩))
  24   · by_cases hlt : (j:ℕ)+1 < r
