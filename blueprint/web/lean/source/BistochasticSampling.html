  13 theorem bistochastic_sampling {X Y : Type*} [Fintype X] [Fintype Y]
  14     [DecidableEq X] [DecidableEq Y]
  15     (e : X ≃ Y) (M : X → Y → ℝ) (hM : ∀ x y, 0 ≤ M x y)
  16     (hrow : ∀ x, ∑ y, M x y = 1) (hcol : ∀ y, ∑ x, M x y = 1) :
  17     ∃ μ : FiniteLaw (Equiv.Perm X),
  18       ∀ x y, (∑ σ, if e (σ x) = y then μ.mass σ else 0) = M x y := by
  19   let A : Matrix X X ℝ := fun x z => M x (e z)
  20   have hA : A ∈ doublyStochastic ℝ X := by
  21     rw [mem_doublyStochastic_iff_sum]
  22     refine ⟨fun x z => hM x (e z), ?_, fun z => hcol (e z)⟩
  23     intro x
  24     dsimp only [A]
  25     rw [Equiv.sum_comp e]
  26     exact hrow x
  27   obtain ⟨w, hw, ht, he⟩ := exists_eq_sum_perm_of_mem_doublyStochastic hA
  28   refine ⟨⟨w, hw, ht⟩, ?_⟩
  29   intro x y
  30   have hh := congrArg (fun B : Matrix X X ℝ => B x (e.symm y)) he
  31   simp only [Matrix.sum_apply, Matrix.smul_apply, smul_eq_mul,
  32     PEquiv.toMatrix_apply, Equiv.toPEquiv_apply, Option.mem_def,
  33     Option.some.injEq, A, Equiv.apply_symm_apply, mul_ite, mul_one, mul_zero] at hh
  34   convert hh using 1
  35   apply Finset.sum_congr rfl
  36   intro σ _
  37   simp only [← Equiv.eq_symm_apply]
  38 
  39 end Oberwolfach
