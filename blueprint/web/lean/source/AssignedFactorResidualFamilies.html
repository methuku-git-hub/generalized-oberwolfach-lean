  13 theorem assigned_factor_residual_families

  16     (F:SimpleGraph X) (hF:F.IsRegularOfDegree 2)
  17     (E:∀C:F.ConnectedComponent,SimpleGraph.cycleGraph (Fintype.card C) ≃g C.toSimpleGraph)
  18     (c d cutoff:ℕ) (pathComp:A⊕B→F.ConnectedComponent) (cycles:J↪F.ConnectedComponent)
  19     (hlong:∀i,cutoff≤Fintype.card (pathComp i)) (hshort:∀j,Fintype.card (cycles j)<cutoff)
  20     (ℓ:J→ℕ) (hsize:∀j,Fintype.card (cycles j)=ℓ j)
  21     (ext:∀i:A⊕B,Fin (c*d+5)↪X)
  22     (hext:∀i j,i≠j→Disjoint (Finset.univ.image (ext i)) (Finset.univ.image (ext j)))
  23     (hcomponent:∀i k,ext i k∈(pathComp i).supp)
  24     (u v:A⊕B→X) (P:∀i,F.Walk (u i) (v i)) (hp:∀i,(P i).IsPath)
  25     (hsupp:∀i,(P i).support=List.ofFn (ext i)):
  26     (∀i j,i≠j→Disjoint (P i).support.toFinset (P j).support.toFinset) ∧
  27     ∃(root:J→X) (C:∀j,F.Walk (root j) (root j)),
  28       (∀j,(C j).IsCycle) ∧ (∀j,(C j).length=ℓ j) ∧
  29       (∀j k,k<(C j).length→factorCycleOrientation F E ((C j).getVert k)=(C j).getVert (k+1)) ∧
  30       (∀j,(C j).toSubgraph.verts=(cycles j).supp) ∧
  31       (∀i j,i≠j→Disjoint (C i).support.toFinset (C j).support.toFinset) ∧
  32       (∀i j,Disjoint (P i).support.toFinset (C j).support.toFinset):=by
  33   have hP:∀i j,i≠j→Disjoint (P i).support.toFinset (P j).support.toFinset:=by
  34     intro i j hij
  35     apply Finset.disjoint_left.mpr
  36     intro x hx hy
  37     have hi:x∈Finset.univ.image (ext i):=by
  38       rw [hsupp i] at hx
  39       simpa only [List.mem_toFinset,List.mem_ofFn,Finset.mem_image,Finset.mem_univ,true_and] using hx
  40     have hj:x∈Finset.univ.image (ext j):=by
  41       rw [hsupp j] at hy
  42       simpa only [List.mem_toFinset,List.mem_ofFn,Finset.mem_image,Finset.mem_univ,true_and] using hy
  43     exact Finset.disjoint_left.mp (hext i j hij) hi hj
  44   refine ⟨hP,?_⟩
  45   obtain ⟨root,C,hC,hlenC,hdir,hcomp,hdis⟩:=factor_component_family_has_forward_cycles F hF E cycles
  46   have hcross:=factor_short_cycles_disjoint_from_long_reserved_paths F pathComp cycles cutoff hlong hshort
  47     (fun _=>c*d+5) (fun i=>ext i) u v P hsupp hcomponent root C hcomp
  48   exact ⟨root,C,hC,fun j=>(hlenC j).trans (hsize j),hdir,hcomp,hdis,hcross⟩
