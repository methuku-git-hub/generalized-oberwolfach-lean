  14 theorem ordinary_cycle_sample_vertex_mean {V:Type} [Fintype V] [DecidableEq V]
  15     (G:SimpleGraph V) (r:ℕ) [DecidableEq (OrdinaryCycle G r)]
  16     (w:OrdinaryCycle G r→ℝ)
  17     (hw:∀ e:G.edgeSet,∑ C,w C*ordinaryCycleIncidence G r C e=1)
  18     (scale:ℝ) (hp:∀ C,0≤scale*w C) (hp1:∀ C,scale*w C≤1) (v:V):
  19     (FiniteLaw.pi (fun C=>FiniteLaw.retentionCoin (scale*w C) (hp C) (hp1 C))).expectation
  20       (independentCycleCount (fun C=>v∈C.val.val.support))=scale*(G.degree v:ℝ)/2 := by
  21   rw [independent_cycle_count_expectation]
  22   have h:=ordinary_cycle_fractional_vertex_weight G r w hw v
  23   have he:(∑ C:OrdinaryCycle G r,@ite ℝ (v∈C.val.val.support) (Classical.propDecidable _) (scale*w C) 0)=
  24       scale*(∑ C:OrdinaryCycle G r,if v∈C.val.val.support then w C else 0):=by
  25     rw [Finset.mul_sum]
  26     apply Finset.sum_congr rfl
  27     intro C _
  28     by_cases hmem:v∈C.val.val.support <;> simp [hmem]
  29   rw [he,←h]
  30   ring

  32 theorem ordinary_cycle_sample_pair_mean_bound {V:Type} [Fintype V] [DecidableEq V]
  33     (G:SimpleGraph V) (r:ℕ) [DecidableEq (OrdinaryCycle G r)]
  34     (u v:V) (huv:u≠v) (hn:0<Fintype.card V)
  35     (B scale:ℝ) (hB:0≤B) (hscale:0≤scale) (w:OrdinaryCycle G r→ℝ)
  36     (hw:∀ C,w C≤B/(Fintype.card V:ℝ)^(r-2))
  37     (hp:∀ C,0≤scale*w C) (hp1:∀ C,scale*w C≤1):
  38     (FiniteLaw.pi (fun C=>FiniteLaw.retentionCoin (scale*w C) (hp C) (hp1 C))).expectation
  39       (independentCycleCount (fun C=>u∈C.val.val.support ∧ v∈C.val.val.support))≤scale*((r:ℝ)^2*B) := by
  40   rw [independent_cycle_count_expectation]
  41   have h:=ordinary_cycle_pair_weight_bound G r u v huv hn B hB w hw
  42   have he:(∑ C:OrdinaryCycle G r,
  43       @ite ℝ (u∈C.val.val.support ∧ v∈C.val.val.support) (Classical.propDecidable _) (scale*w C) 0)=
  44       scale*(∑ C:OrdinaryCycle G r,
  45         if u∈C.val.val.support ∧ v∈C.val.val.support then w C else 0):=by
  46     rw [Finset.mul_sum]
  47     apply Finset.sum_congr rfl
  48     intro C _
  49     by_cases hmem:u∈C.val.val.support ∧ v∈C.val.val.support <;> simp [hmem]
  50   rw [he]
  51   exact mul_le_mul_of_nonneg_left h hscale
