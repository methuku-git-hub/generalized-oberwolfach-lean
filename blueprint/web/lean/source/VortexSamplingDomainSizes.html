  11 theorem nested_contracting_difference_sizes {X : Type*} [DecidableEq X]
  12     (A : ℕ→Finset X) (ε : ℝ) (he : 0≤ε) (he1 : ε≤1/4)
  13     (hnest : ∀ i j,i≤ j→A j⊆A i)
  14     (hstep : ∀ i,((A (i+1)).card:ℝ)≤ε*(A i).card)
  15     (i d : ℕ) (hd : 1≤ d) :
  16     (3/4:ℝ)*(A i).card≤(A i\A (i+d)).card ∧
  17     ((A i\A (i+d)).card:ℝ)≤(A i).card ∧
  18     ((A (i+1)\A (i+1+d)).card:ℝ)≤(2*ε)*(A i\A (i+d)).card := by
  19   have hlate : ((A (i+d)).card:ℝ)≤(A (i+1)).card := by
  20     exact_mod_cast Finset.card_le_card (hnest (i+1) (i+d) (by omega))
  21   have hu := hstep i
  22   have hec := mul_le_mul_of_nonneg_right he1 (Nat.cast_nonneg (A i).card)
  23   have hc := real_card_sdiff (hnest i (i+d) (by omega))
  24   have hlo : (3/4:ℝ)*(A i).card≤(A i\A (i+d)).card := by
  25     linarith only [hc,hlate,hu,hec]
  26   have hu' : ((A (i+1)\A (i+1+d)).card:ℝ)≤(A (i+1)).card := by
  27     exact_mod_cast Finset.card_le_card (Finset.sdiff_subset : A (i+1)\A (i+1+d)⊆A (i+1))
  28   have hp := mul_le_mul_of_nonneg_left hlo he
  29   have hnonneg : (0:ℝ)≤ε*(A i).card := mul_nonneg he (Nat.cast_nonneg _)
  30   refine ⟨hlo,?_,?_⟩
  31   · exact_mod_cast Finset.card_le_card (Finset.sdiff_subset : A i\A (i+d)⊆A i)
  32   · nlinarith only [hp,hu',hu,hnonneg]

  34 theorem Vortex.number_of_levels_le_order
  35     {V U : Type*} [Fintype V] [Fintype U] {G : SimpleGraph V}
  36     {D m : ℕ} {β ε : ℝ} (Q : Vortex U G D β ε m)
  37     (he : 0<ε) (he1 : ε<1) (hm : 0< m) : Q.length+1≤ Fintype.card V := by
  38   have hp (i : ℕ) (hi : i≤ Q.length) : 0<(Q.vertices i).card := by
  39     have hh := Q.vertices_lower hi
  40     have hm' : (0:ℝ)< m := by exact_mod_cast hm
  41     exact_mod_cast (show (0:ℝ)<(Q.vertices i).card by nlinarith only [hh,mul_pos he hm'])
  42   have hdec (i : ℕ) (hi : i< Q.length) : (Q.vertices (i+1)).card<(Q.vertices i).card := by
  43     have hn : (0:ℝ)<(Q.vertices i).card := by exact_mod_cast hp i hi.le
  44     have hs := (Q.consecutive_size_bounds he hm hi).2
  45     have hm := mul_lt_mul_of_pos_right he1 hn
  46     exact_mod_cast (show ((Q.vertices (i+1)).card:ℝ)<(Q.vertices i).card by linarith only [hs,hm])
  47   have hbound : ∀ i,i≤ Q.length→i+(Q.vertices i).card≤ Fintype.card V := by
  48     intro i
  49     induction i with
  50     | zero=>intro hi;rw [Q.sizes 0 hi];simp [floorOrbit_zero]
  51     | succ i ih=>intro hi;have hh:=ih (by omega);have hd:=hdec i (by omega);omega
  52   have hh:=hbound Q.length le_rfl
  53   have hpos:=hp Q.length le_rfl
  54   omega

  56 theorem Vortex.sampling_difference_sizes
  57     {V U : Type*} [Fintype V] [Fintype U] {G : SimpleGraph V}
  58     {D m : ℕ} {β ε ρ₀ : ℝ} (Q : Vortex U G D β ε m)
  59     (hreg : G.IsRegularOfDegree D) (hn : 0< Fintype.card V)
  60     (hD : (Fintype.card V:ℝ)/2≤ D) (hr : 0<ρ₀)
  61     (hrlo : ρ₀≤ initialIndexDensity U D) (hb : 0≤β) (hb1 : β≤1/4)
  62     (he : 0<ε) (he1 : ε≤1/4) (i d : ℕ) (hd : 1≤ d) (hi : i≤ Q.length) :
  63     let A := Q.extendedVertices i\Q.extendedVertices (i+d)
  64     let B := Q.extendedIndices i\Q.extendedIndices (i+d);
  65     3*ε*m/8≤(A.card:ℝ) ∧
  66     3*ρ₀*ε*m/64≤(B.card:ℝ) ∧
  67     ((Q.extendedVertices (i+1)\Q.extendedVertices (i+1+d)).card:ℝ)≤(2*ε)*A.card ∧
  68     ((Q.extendedIndices (i+1)\Q.extendedIndices (i+1+d)).card:ℝ)≤(2*ε)*B.card ∧
  69     ((Q.vertices i).card:ℝ)≤(4/3)*A.card ∧
  70     ((Q.indices i).card:ℝ)≤(4/3)*B.card := by
  71   intro A B
  72   have hv := nested_contracting_difference_sizes Q.extendedVertices ε he.le he1
  73     (fun i j h=>Q.extendedVertices_subset h) (fun i=>(Q.extended_successor_size_bounds he.le i).1) i d hd
  74   have hu := nested_contracting_difference_sizes Q.extendedIndices ε he.le he1
  75     (fun i j h=>Q.extendedIndices_subset h) (fun i=>(Q.extended_successor_size_bounds he.le i).2) i d hd
  76   change (3/4:ℝ)*(Q.extendedVertices i).card≤ A.card ∧ _ ∧ _ at hv
  77   change (3/4:ℝ)*(Q.extendedIndices i).card≤ B.card ∧ _ ∧ _ at hu
  78   have hvl := hv.1
  79   have hul := hu.1
  80   rw [Q.extendedVertices_eq hi] at hvl
  81   rw [Q.extendedIndices_eq hi] at hul
  82   have hl := Q.vertices_lower hi
  83   have hulow := Q.index_size_lower hreg hn hD hr hrlo hb hb1 hi
  84   have hscale := mul_le_mul_of_nonneg_left hl (show 0≤ρ₀/8 by positivity)
  85   refine ⟨?_,?_,hv.2.2,hu.2.2,?_,?_⟩
  86   · nlinarith only [hvl,hl]
  87   · nlinarith only [hul,hulow,hscale]
  88   · linarith only [hvl]
  89   · linarith only [hul]
