   9 theorem ordinary_profile_budgets (m:ℕ) (β L:ℝ) (hβ:0<β):
  10     ∃ N:ℕ,∀ n:ℕ,N≤n →
  11       (m+3:ℝ)*(L+1)≤β/2*n ∧
  12       2*(m+1:ℝ)^2≤ordinaryLocalCountCoeff m β*n ∧
  13       2*(m:ℝ)*(2*(m+1)+(m+1)*m)≤ordinaryLocalCountCoeff m β*n := by
  14   let c:=ordinaryLocalCountCoeff m β
  15   have hc:0<c:=ordinaryLocalCountCoeff_pos m β hβ
  16   let x:ℝ:=(m+3)*(L+1)/(β/2)
  17   let y:ℝ:=2*(m+1:ℝ)^2/c
  18   let z:ℝ:=2*(m:ℝ)*(2*(m+1)+(m+1)*m)/c
  19   obtain ⟨N,hN⟩:=exists_nat_gt (max x (max y z))
  20   refine ⟨N,?_⟩
  21   intro n hn
  22   have hn':(N:ℝ)≤n:=by exact_mod_cast hn
  23   have hx:x≤n:=(le_max_left _ _).trans (le_of_lt hN) |>.trans hn'
  24   have hy:y≤n:=((le_max_left y z).trans (le_max_right x (max y z))).trans (le_of_lt hN) |>.trans hn'
  25   have hz:z≤n:=((le_max_right y z).trans (le_max_right x (max y z))).trans (le_of_lt hN) |>.trans hn'
  26   refine ⟨?_,?_,?_⟩
  27   · dsimp [x] at hx
  28     have hh: (m+3:ℝ)*(L+1)≤n*(β/2):=(div_le_iff₀ (by positivity)).mp hx
  29     nlinarith only [hh]
  30   · dsimp [y] at hy
  31     have hh: 2*(m+1:ℝ)^2≤n*c:=(div_le_iff₀ hc).mp hy
  32     nlinarith only [hh]
  33   · dsimp [z] at hz
  34     have hh:2*(m:ℝ)*(2*(m+1)+(m+1)*m)≤n*c:=(div_le_iff₀ hc).mp hz
  35     nlinarith only [hh]
