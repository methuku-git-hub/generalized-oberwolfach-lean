  14 theorem bounded_matrix_fractional_matching
  15     {V : Type*} [Fintype V] [DecidableEq V]
  16     (M : V → V → ℕ) (hM : ∀ x y, M x y ≤ 1)
  17     (β : ℝ) (hβ : 0<β) (hβhalf : β<1/2) (hn : 0<Fintype.card V)
  18     (hrow : ∀ x, (1/2+β)*Fintype.card V ≤ ((∑ y, M x y : ℕ) : ℝ))
  19     (hcol : ∀ y, (1/2+β)*Fintype.card V ≤ ((∑ x, M x y : ℕ) : ℝ)) :
  20     ∃ z : V → V → ℝ,
  21       (∀ x y, M x y=0 → z x y=0) ∧
  22       (∀ x y, M x y=1 → β/(5*Fintype.card V) ≤ z x y ∧
  23         z x y ≤ 5/(β*Fintype.card V)) ∧
  24       (∀ x, ∑ y, z x y=1) ∧ (∀ y, ∑ x, z x y=1) := by
  25   obtain ⟨w,hw,hl,hr⟩ := bounded_bipartite_fractional_matching M hM β hβ hβhalf hn hrow hcol
  26   let z (x y : V) : ℝ := ∑ i : Fin (M x y), w ⟨x,y,i⟩
  27   refine ⟨z,?_,?_,?_,?_⟩
  28   · intro x y hzero
  29     apply Finset.sum_eq_zero
  30     intro i _
  31     have hi := i.isLt
  32     omega
  33   · intro x y hone
  34     constructor
  35     · have hh := Finset.sum_le_sum (fun (i : Fin (M x y)) (_ : i∈Finset.univ) => (hw ⟨x,y,i⟩).1)
  36       simpa only [Finset.sum_const,Finset.card_univ,Fintype.card_fin,hone,Nat.cast_one,one_nsmul] using hh
  37     · have hh := Finset.sum_le_sum (fun (i : Fin (M x y)) (_ : i∈Finset.univ) => (hw ⟨x,y,i⟩).2)
  38       simpa only [Finset.sum_const,Finset.card_univ,Fintype.card_fin,hone,Nat.cast_one,one_nsmul] using hh
  39   · intro x
  40     have hh := hl x
  41     rw [Fintype.sum_sigma] at hh
  42     have he (u : V) : (∑ e : (Σ y, Fin (M u y)),
  43         if u=x then w ⟨u,e⟩ else 0)=if u=x then ∑ y, z u y else 0 := by
  44       by_cases h : u=x
  45       · subst u
  46         simp only [ite_true,Fintype.sum_sigma,z]
  47       · simp [h]
  48     simp_rw [he] at hh
  49     simpa using hh
  50   · intro y
  51     have hh := hr y
  52     rw [Fintype.sum_sigma] at hh
  53     have he (x : V) : (∑ e : (Σ v, Fin (M x v)),
  54         if e.1=y then w ⟨x,e⟩ else 0)=z x y := by
  55       rw [Fintype.sum_sigma]
  56       have he' (v : V) : (∑ i : Fin (M x v),
  57           if v=y then w ⟨x,v,i⟩ else 0)=if v=y then z x v else 0 := by
  58         by_cases h : v=y
  59         · subst v
  60           simp only [ite_true,z]
  61         · simp [h]
  62       simp_rw [he']
  63       simp
  64     simpa only [he] using hh

  66 /-- The coefficient of an unordered graph edge is represented by a symmetric
  67 function of its endpoints, zero on nonedges. Every vertex has total load one. -/
  68 theorem bounded_perfect_fractional_matching
  69     {V : Type*} [Fintype V] (Q : SimpleGraph V) (β : ℝ)
  70     (hβ : 0<β) (hβhalf : β<1/2)
  71     (hn : 4/β ≤ (Fintype.card V : ℝ))
  72     (hdegree : ∀ v, (1/2+β)*Fintype.card V ≤ (Q.degree v : ℝ)) :
  73     ∃ w : V → V → ℝ,
  74       (∀ u v, w u v=w v u) ∧
  75       (∀ u v, ¬ Q.Adj u v → w u v=0) ∧
  76       (∀ u v, Q.Adj u v → β/(5*Fintype.card V) ≤ w u v ∧
  77         w u v ≤ 5/(β*Fintype.card V)) ∧
  78       (∀ v, ∑ u, w v u=1) := by
  79   classical
  80   let M (x y : V) : ℕ := if Q.Adj x y then 1 else 0
  81   have hM (x y : V) : M x y≤1 := by dsimp [M];split_ifs <;> omega
  82   have hrow (x : V) : ∑ y, M x y=Q.degree x := by
  83     rw [SimpleGraph.degree,SimpleGraph.neighborFinset_eq_filter,Finset.card_filter]
  84   have hcol (y : V) : ∑ x, M x y=Q.degree y := by
  85     rw [← hrow y]
  86     apply Finset.sum_congr rfl
  87     intro x _
  88     simp only [M,Q.adj_comm]
  89   have hnpos : 0<Fintype.card V := by
  90     have hh : (0 : ℝ)<Fintype.card V := lt_of_lt_of_le (div_pos (by norm_num) hβ) hn
  91     exact_mod_cast hh
  92   obtain ⟨z,hzero,hbound,hl,hr⟩ := bounded_matrix_fractional_matching M hM β hβ hβhalf hnpos
  93     (fun x => by rw [hrow];exact hdegree x)
  94     (fun y => by rw [hcol];exact hdegree y)
  95   refine ⟨fun u v => (z u v+z v u)/2,?_,?_,?_,?_⟩
  96   · intro u v;ring
  97   · intro u v huv
  98     dsimp only
  99     rw [hzero u v (by simp [M,huv]),hzero v u (by simp [M,Q.adj_comm,huv])]
 100     norm_num
 101   · intro u v huv
 102     have h₁ := hbound u v (by simp [M,huv])
 103     have h₂ := hbound v u (by simp [M,huv.symm])
 104     constructor <;> linarith
 105   · intro v
 106     rw [← Finset.sum_div,Finset.sum_add_distrib,hl,hr]
 107     norm_num

 109 end Oberwolfach
