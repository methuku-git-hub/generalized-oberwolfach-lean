  14 theorem signed_cycle_edge_isolator_of_fractional {V : Type*} [Fintype V] [DecidableEq V] {r : ℕ}
  15     (G : SimpleGraph V) (u v : V) (huv : G.Adj u v)
  16     (hG : HasFractionalDecomposition G (fun C => IsEmbeddedCycle C r))
  17     (hD : HasFractionalDecomposition (G\SimpleGraph.edge u v) (fun C => IsEmbeddedCycle C r)) :
  18     ∃ I : SimpleGraph V → ℝ,
  19       (∀ C,I C≠0 → C≤G ∧ IsEmbeddedCycle C r) ∧
  20       (∀ C,|I C|≤1) ∧
  21       (∀ x y,(∑ C,if C.Adj x y then I C else 0)=
  22         if (SimpleGraph.edge u v).Adj x y then 1 else 0) ∧
  23       (∀ x,(∑ C,if C.degree x≠0 then I C else 0)=
  24         (1/2:ℝ)*(if x=u ∨ x=v then 1 else 0)) := by
  25   obtain ⟨w,hw,hs,he⟩ := hG
  26   obtain ⟨w',hw',hs',he'⟩ := hD
  27   let I (C : SimpleGraph V) := w C-w' C
  28   have hsub : SimpleGraph.edge u v≤G := (SimpleGraph.edge_le_iff G).mpr (Or.inr huv)
  29   have hI (C : SimpleGraph V) (hC : I C≠0) : C≤G ∧ IsEmbeddedCycle C r := by
  30     by_cases hp : 0<w C
  31     · exact hs C hp
  32     · have hp' : 0<w' C := by dsimp [I] at hC;rcases ne_iff_lt_or_gt.mp hC with hh|hh <;> linarith [hw C,hw' C]
  33       exact ⟨(hs' C hp').1.trans sdiff_le,(hs' C hp').2⟩
  34   have hload (x y : V) : (∑ C,if C.Adj x y then I C else 0)=
  35       if (SimpleGraph.edge u v).Adj x y then 1 else 0 := by
  36     calc
  37       _ = (∑ C,if C.Adj x y then w C else 0)-(∑ C,if C.Adj x y then w' C else 0) := by
  38         rw [← Finset.sum_sub_distrib]
  39         apply Finset.sum_congr rfl
  40         intro C _
  41         dsimp [I]
  42         split_ifs <;> ring
  43       _ = _ := by
  44         rw [he,he']
  45         by_cases hg : G.Adj x y
  46         · by_cases hj : (SimpleGraph.edge u v).Adj x y <;> simp [hg,hj,SimpleGraph.sdiff_adj]
  47         · have hj : ¬(SimpleGraph.edge u v).Adj x y := fun hj => hg (hsub hj)
  48           simp [hg,hj,SimpleGraph.sdiff_adj]
  49   refine ⟨I,hI,?_,hload,?_⟩
  50   · intro C
  51     have h1 := fractional_cycle_coefficient_le_one G r w hw hs he C
  52     have h2 := fractional_cycle_coefficient_le_one (G\SimpleGraph.edge u v) r w' hw' hs' he' C
  53     apply abs_le.mpr
  54     dsimp [I]
  55     constructor <;> linarith [hw C,hw' C]
  56   · intro x
  57     have hh := signed_cycle_vertex_load r I (fun C hC => (hI C hC).2) x
  58     have hload' : (∑ y,∑ C,if C.Adj x y then I C else 0)=
  59         ((SimpleGraph.edge u v).degree x:ℝ) := by
  60       calc
  61         _ = ∑ y,if (SimpleGraph.edge u v).Adj x y then (1:ℝ) else 0 := by
  62           apply Finset.sum_congr rfl
  63           intro y _
  64           exact hload x y
  65         _ = _ := by
  66           convert graph_adjacency_row_sum (SimpleGraph.edge u v) x using 1
  67           · apply Finset.sum_congr rfl
  68             intro y _
  69             split_ifs <;> rfl
  70           · simp only [← SimpleGraph.card_neighborSet_eq_degree,← Nat.card_eq_fintype_card]
  71     rw [hload'] at hh
  72     have hd := single_edge_degree u v huv.ne x
  73     simp only [← SimpleGraph.card_neighborSet_eq_degree,← Nat.card_eq_fintype_card] at hh hd ⊢
  74     rw [hd] at hh
  75     split_ifs at hh ⊢ <;> norm_num at hh ⊢ <;> linarith

  77 theorem signed_cycle_edge_isolator (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  78     ∃ N : ℕ,∀ n≥N,∀ G : SimpleGraph (Fin n),
  79       (∀ x,(cycleThreshold r+α)*(n:ℝ)≤G.degree x) →
  80       ∀ u v,G.Adj u v →
  81       ∃ I : SimpleGraph (Fin n) → ℝ,
  82         (∀ C,I C≠0 → C≤G ∧ IsEmbeddedCycle C r) ∧
  83         (∀ C,|I C|≤1) ∧
  84         (∀ x y,(∑ C,if C.Adj x y then I C else 0)=
  85           if (SimpleGraph.edge u v).Adj x y then 1 else 0) ∧
  86         (∀ x,(∑ C,if C.degree x≠0 then I C else 0)=
  87           (1/2:ℝ)*(if x=u ∨ x=v then 1 else 0)) := by
  88   obtain ⟨N₁,hN₁⟩ := cycle_fractional_positive_margin r hr (α/2) (by positivity)
  89   obtain ⟨N₂,hN₂⟩ := eventually_constant_le_linear 1 (α/2) (by positivity)
  90   refine ⟨max N₁ N₂,?_⟩
  91   intro n hn G hdeg u v huv
  92   apply signed_cycle_edge_isolator_of_fractional G u v huv
  93   · apply hN₁ n ((le_max_left _ _).trans hn) G
  94     intro x
  95     have ha : 0≤α*(n:ℝ) := mul_nonneg hα.le (Nat.cast_nonneg n)
  96     nlinarith [hdeg x]
  97   · apply hN₁ n ((le_max_left _ _).trans hn) (G\SimpleGraph.edge u v)
  98     intro x
  99     have hc := congrArg (fun a : ℕ => (a:ℝ)) (single_edge_deletion_degree G u v huv x)
 100     simp only [Nat.cast_add] at hc
 101     have hb : ((if x=u ∨ x=v then 1 else 0 : ℕ):ℝ)≤1 := by split_ifs <;> norm_num
 102     have hn' := hN₂ n ((le_max_right _ _).trans hn)
 103     have hg := hdeg x
 104     simp only [← SimpleGraph.card_neighborSet_eq_degree,← Nat.card_eq_fintype_card] at hc hg hn' ⊢
 105     nlinarith
