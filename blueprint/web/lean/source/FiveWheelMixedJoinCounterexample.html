   1 import Oberwolfach.MarkedBoundaryExtensionCounts
   2 /-! A two-step ordinary join and individual marked continuations do not
   3 alone imply a mixed marked-then-ordinary join. This concerns a proposed
   4 proof substitution, not the manuscript theorem. -/
   5 set_option autoImplicit false
   6 namespace Oberwolfach
   7 
   8 abbrev fiveWheelOrdinaryRelation (x y : Fin 6) : Prop :=
   9   x ≠ y ∧ ¬ ((x = 1 ∨ x = 2) ∧ y = 0)
  10 
  11 abbrev fiveWheelMarkedRelation (x y : Fin 6) : Prop :=
  12   x ≠ y ∧ (y = 1 ∨ y = 2)
  13 
  14 abbrev fiveWheelMarkedSpokes (x : Fin 6) : Prop := x = 1 ∨ x = 2
  15 
  16 theorem ordinary_joins_and_marked_steps_do_not_imply_mixed_join :
  17     (∀ x : Fin 6, ∃ y, fiveWheelMarkedRelation x y ∧ fiveWheelMarkedSpokes y) ∧
  18     (∀ x z : Fin 6, x ≠ z → ∃ y,
  19       fiveWheelOrdinaryRelation x y ∧ fiveWheelOrdinaryRelation y z) ∧
  20     (∃ x z : Fin 6, x ≠ z ∧ ¬∃ y,
  21       fiveWheelMarkedRelation x y ∧ fiveWheelMarkedSpokes y ∧
  22       fiveWheelOrdinaryRelation y z) := by
  23   constructor
  24   · intro x
  25     fin_cases x <;> first | exact ⟨1,by decide⟩ | exact ⟨2,by decide⟩
  26   constructor
  27   · intro x z hxz
  28     fin_cases x <;> fin_cases z <;>
  29       first | contradiction | exact ⟨0,by decide⟩ | exact ⟨1,by decide⟩ |
  30         exact ⟨2,by decide⟩ | exact ⟨3,by decide⟩ | exact ⟨4,by decide⟩ |
  31         exact ⟨5,by decide⟩
  32   · refine ⟨3,0,by decide,?_⟩
  33     rintro ⟨y,hy⟩
  34     fin_cases y <;> simp_all [fiveWheelOrdinaryRelation, fiveWheelMarkedRelation,
  35       fiveWheelMarkedSpokes]
  36 end Oberwolfach
