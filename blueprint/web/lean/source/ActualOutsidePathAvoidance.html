  11 theorem actual_outside_buffered_path_image_avoids_reserved

  13     [DecidableEq K] [DecidableEq J] (F:SimpleGraph X)
  14     (data:ActualResidualData (K:=K) (J:=J) F)
  15     (hCycle:data.t.biUnion (fun j=>(data.C j).support.toFinset)⊆data.Cembed)
  16     (n:I→ℕ) (inner:∀i,Fin (n i+1)↪X)
  17     (fC:{x // x∉bufferedPathStrictInterior n inner}↪V)
  18     (ψ:X→V) (Φ g:{x // x∈data.remaining}→V)
  19     (Anchors:Finset {x // x∈data.remaining}) (Lists Y:Finset V)
  20     (hold:∀(x:{x // x∉bufferedPathStrictInterior n inner}),
  21       x.val∈data.oldDomain→fC x=ψ x.val)
  22     (hnew:∀(x:{x // x∉bufferedPathStrictInterior n inner}) (hx:x.val∈data.remaining),
  23       fC x=Φ ⟨x.val,hx⟩)
  24     (hAnchor:∀x∈Anchors,Φ x=g x)
  25     (hFree:∀x,x∉Anchors→Φ x∈Lists)
  26     (hOldAvoid:∀x,x∈data.oldDomain→x∉bufferedPathFullDomain n inner→ψ x∉Y)
  27     (hAnchorAvoid:∀x∈Anchors,x.val∉bufferedPathFullDomain n inner → g x∉Y)
  28     (hListAvoid:Disjoint Lists Y):
  29     Disjoint (outsideBufferedPathImage n inner fC) Y:=by

  31   apply Finset.disjoint_left.mpr
  32   intro v hv hvY
  33   obtain ⟨x,hx,rfl⟩:=Finset.mem_image.mp hv
  34   have hxA:x.val∉bufferedPathFullDomain n inner:=(Finset.mem_filter.mp hx).2
  35   have hcover:x.val∈data.oldDomain ∨ x.val∈data.remaining:=by
  36     have hh:x.val∈data.oldDomain∪data.remaining:=by rw [data.old_domain_union_remaining hCycle];exact Finset.mem_univ _

  38   rcases hcover with hxold | hxnew

  41   · let z:{x // x∈data.remaining}:=⟨x.val,hxnew⟩
  42     have heq:fC x=Φ z:=hnew x hxnew

  44     · rw [heq,hAnchor z hz] at hvY
  45       exact hAnchorAvoid z hz hxA hvY

  47       rw [heq] at hvY
