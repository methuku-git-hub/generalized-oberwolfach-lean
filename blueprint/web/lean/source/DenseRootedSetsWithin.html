   4 /-! Count dense conflict-free rooted sets inside a specified vertex domain.
   5 The ambient graph is retained in the conclusion, so external-root callers
   6 can first enlarge their neighbourhood and then apply this result. -/

  15 theorem dense_rooted_sets_within (s : ℕ) (α : ℝ) (hα : 0<α) :
  16     ∃ K : ℕ,∀ k : ℕ,K≤k → ∀ L : ℝ,0≤L → ∃ N : ℕ,
  17     ∀ (V : Type) [Fintype V] [DecidableEq V],
  18     ∀ (P F : SimpleGraph V) (B T : Finset V),N≤B.card → T.card=s → T⊆B → ∀ δ : ℝ,
  19     (∀ v∈B,(δ+α)*(B.card:ℝ)≤((P.neighborFinset v∩B).card:ℝ)) →
  20     (∀ v,(F.degree v:ℝ)≤L) → (∀ x∈T,∀ y∈T,¬F.Adj x y) →
  21     (B.card:ℝ)^k/((2:ℝ)^(k+1)*k.factorial)≤
  22       Fintype.card {S : FixedSubset V (s+k) // T⊆S.val ∧ S.val⊆B ∧
  23         (∀ v∈S.val,(δ+α/2)*(s+k:ℝ)<((P.neighborFinset v∩S.val).card:ℝ)) ∧
  24         (∀ x∈S.val,∀ y∈S.val,¬F.Adj x y)} := by
  25   obtain ⟨K,hK⟩ := rootedSubset_dense_conflict_free_many s α hα
  26   refine ⟨K,?_⟩
  27   intro k hk L hL
  28   obtain ⟨N,hN⟩ := hK k hk L hL
  29   refine ⟨N,?_⟩
  30   intro V _ _ P F B T hNB hT hTB δ hP hF hind
  31   let Z := T.subtype (fun v=>v∈B)
  32   have hZ : Z.card=s := by
  33     dsimp [Z]
  34     rw [Finset.card_subtype]
  35     have he : T.filter (fun v=>v∈B)=T := Finset.filter_eq_self.mpr hTB
  36     rw [he,hT]
  37   clear hT
  38   subst s
  39   have hcardB : Fintype.card {v : V // v∈B}=B.card := Fintype.card_coe B
  40   have hp (v : {v : V // v∈B}) :
  41       (δ+α)*Fintype.card {v : V // v∈B}≤((P.induce (B:Set V)).degree v:ℝ) := by
  42     rw [hcardB,induced_degree_intersection]
  43     exact hP v.val v.property
  44   have hf (v : {v : V // v∈B}) : ((F.induce (B:Set V)).degree v:ℝ)≤L := by
  45     have hh : ((F.induce (B:Set V)).degree v:ℝ)≤F.degree v.val := by
  46       exact_mod_cast induced_degree_le_ambient F B v
  47     exact hh.trans (hF v.val)
  48   have hz : ∀ x∈Z,∀ y∈Z,¬(F.induce (B:Set V)).Adj x y := by
  49     intro x hx y hy hxy
  50     exact hind x.val (Finset.mem_subtype.mp hx) y.val (Finset.mem_subtype.mp hy) hxy
  51   have hh := hN {v : V // v∈B} (by simpa only [hcardB] using hNB)
  52     (P.induce (B:Set V)) (F.induce (B:Set V)) Z rfl δ hp hf hz
  53   let A := {S : RootedSubset {v : V // v∈B} Z k //
  54     (∀ v∈S.val.val,(δ+α/2)*(Z.card+k:ℝ)<(((P.induce (B:Set V)).neighborFinset v∩S.val.val).card:ℝ)) ∧
  55     (∀ x∈S.val.val,∀ y∈S.val.val,¬(F.induce (B:Set V)).Adj x y)}
  56   let D := {S : FixedSubset V (Z.card+k) // T⊆S.val ∧ S.val⊆B ∧
  57     (∀ v∈S.val,(δ+α/2)*(Z.card+k:ℝ)<((P.neighborFinset v∩S.val).card:ℝ)) ∧
  58     (∀ x∈S.val,∀ y∈S.val,¬F.Adj x y)}
  59   let f (S : A) : D := ⟨inducedSetEmbedding B S.val.val,by
  60     refine ⟨?_,inducedSetEmbedding_subset B S.val.val,?_,
  61       inducedSetEmbedding_conflict_free F B S.val.val S.property.2⟩
  62     · intro v hv
  63       have hvB := hTB hv
  64       have hvZ : (⟨v,hvB⟩:{v : V // v∈B})∈Z := Finset.mem_subtype.mpr hv
  65       exact Finset.mem_map.mpr ⟨⟨v,hvB⟩,S.val.property hvZ,rfl⟩
  66     · intro v hv
  67       obtain ⟨v',hv',rfl⟩ := Finset.mem_map.mp hv
  68       have hh := S.property.1 v' hv'
  69       rw [inducedSetEmbedding_degree] at hh
  70       exact hh⟩
  71   have hinj : Function.Injective f := by
  72     intro S T he
  73     apply Subtype.ext
  74     apply Subtype.ext
  75     exact inducedSetEmbedding_injective B (Z.card+k) (congrArg Subtype.val he)
  76   have hc : Fintype.card A≤Fintype.card D := Fintype.card_le_of_injective f hinj
  77   have hh' : (B.card:ℝ)^k/((2:ℝ)^(k+1)*k.factorial)≤Fintype.card A := by
  78     simpa only [A,hcardB] using hh
  79   have hout := hh'.trans (show (Fintype.card A:ℝ)≤Fintype.card D by exact_mod_cast hc)
  80   simpa only [D,Nat.cast_add] using hout
