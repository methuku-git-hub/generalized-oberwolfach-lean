  13 theorem finiteLaw_sum_positive_support {Ω : Type*} [Fintype Ω]
  14     (μ : FiniteLaw Ω) (g : Ω → ℝ) (hg : ∀ ω, μ.mass ω = 0 → g ω = 0) :
  15     (∑ ω : {ω : Ω // 0 < μ.mass ω}, g ω.1) = ∑ ω, g ω := by
  16   apply Fintype.sum_of_injective Subtype.val Subtype.val_injective
  17   · intro ω hω
  18     apply hg ω
  19     by_contra hn
  20     exact hω ⟨⟨ω, lt_of_le_of_ne (μ.nonneg ω) (Ne.symm hn)⟩, rfl⟩
  21   · intro ω; rfl

  23 def FiniteLaw.support {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) :
  24     FiniteLaw {ω : Ω // 0 < μ.mass ω} where
  25   mass ω := μ.mass ω.1
  26   nonneg ω := μ.nonneg ω.1
  27   total := (finiteLaw_sum_positive_support μ μ.mass (fun _ h => h)).trans μ.total

  29 theorem finiteLaw_support_probability {Ω : Type*} [Fintype Ω]
  30     (μ : FiniteLaw Ω) (A : Ω → Prop) :
  31     μ.support.probability (fun ω => A ω.1) = μ.probability A := by
  32   unfold FiniteLaw.probability
  33   exact finiteLaw_sum_positive_support μ (fun ω => if A ω then μ.mass ω else 0)
  34     (by intro ω h; simp [h])

  36 theorem finiteLaw_support_expectation {Ω : Type*} [Fintype Ω]
  37     (μ : FiniteLaw Ω) (f : Ω → ℝ) :
  38     μ.support.expectation (fun ω => f ω.1) = μ.expectation f := by

  43 theorem finiteLaw_support_nonempty {Ω : Type*} [Fintype Ω]
  44     (μ : FiniteLaw Ω) : Nonempty {ω : Ω // 0 < μ.mass ω} := by
  45   have h : 0 < ∑ ω, μ.mass ω := by rw [μ.total]; norm_num
