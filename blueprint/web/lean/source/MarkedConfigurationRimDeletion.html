  20 theorem physical_deletion_degree_sum (D:SimpleGraph V) (delta:ℝ)

  22     (∀ x,(∑ y:V,if D.Adj x y then (1:ℝ) else 0)≤delta) ∧
  23     (∀ y,(∑ x:V,if D.Adj x y then (1:ℝ) else 0)≤delta):=by
  24   have he (x:V):(∑ y:V,if D.Adj x y then (1:ℝ) else 0)=D.degree x:=by
  25     rw [←SimpleGraph.card_neighborFinset_eq_degree,SimpleGraph.neighborFinset_eq_filter]
  26     simp only [Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
