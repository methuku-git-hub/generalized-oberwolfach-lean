   5 /-! Minimum squared Euclidean length in the convex hull of a finite family.
   6 The ambient finite function space carries its usual topology; the energy
   7 is written explicitly as the sum of the squared coordinates. -/

  16 def quadraticEnergy (z : I → ℝ) : ℝ := ∑ i,(z i)^2
  17 
  18 theorem quadraticEnergy_nonneg (z : I → ℝ) : 0≤quadraticEnergy z :=
  19   Finset.sum_nonneg (fun i _=>sq_nonneg (z i))

  21 theorem quadraticEnergy_mix (u v : I → ℝ) {θ : ℝ} (hθ0 : 0≤θ) (hθ1 : θ≤1) :
  22     quadraticEnergy (θ•u+(1-θ)•v)≤
  23       θ*quadraticEnergy u+(1-θ)*quadraticEnergy v := by
  24   simp only [quadraticEnergy,Finset.mul_sum,← Finset.sum_add_distrib]
  25   apply Finset.sum_le_sum
  26   intro i _
  27   change (θ*u i+(1-θ)*v i)^2≤θ*(u i)^2+(1-θ)*(v i)^2
  28   have h := mul_nonneg (mul_nonneg hθ0 (sub_nonneg.mpr hθ1)) (sq_nonneg (u i-v i))
  29   nlinarith

  31 theorem finiteHull_energy_minimum (d : Ω → I → ℝ) (A : Finset Ω) (hA : A.Nonempty) :
  32     ∃ z∈convexHull ℝ (d '' (A : Set Ω)),
  33       ∀ y∈convexHull ℝ (d '' (A : Set Ω)),quadraticEnergy z≤quadraticEnergy y := by
  34   have hc := ((A.finite_toSet).image d).isCompact_convexHull ℝ
  35   have hn : (convexHull ℝ (d '' (A : Set Ω))).Nonempty := by
  36     obtain ⟨a,ha⟩ := hA
  37     exact ⟨d a,subset_convexHull ℝ _ (Set.mem_image_of_mem d ha)⟩
  38   exact hc.exists_isMinOn hn
  39     (show ContinuousOn quadraticEnergy _ by unfold quadraticEnergy;fun_prop)

  41 def finiteHullEnergy (d : Ω → I → ℝ) (A : Finset Ω) : ℝ :=
  42   if hA : A.Nonempty then quadraticEnergy (Classical.choose (finiteHull_energy_minimum d A hA))
  43   else 0

  45 theorem finiteHullEnergy_attained (d : Ω → I → ℝ) (A : Finset Ω) (hA : A.Nonempty) :
  46     ∃ z∈convexHull ℝ (d '' (A : Set Ω)),quadraticEnergy z=finiteHullEnergy d A ∧
  47       ∀ y∈convexHull ℝ (d '' (A : Set Ω)),finiteHullEnergy d A≤quadraticEnergy y := by
  48   rw [finiteHullEnergy,dif_pos hA]
  49   exact ⟨_,(Classical.choose_spec (finiteHull_energy_minimum d A hA)).1,rfl,
  50     (Classical.choose_spec (finiteHull_energy_minimum d A hA)).2⟩

  52 theorem finiteHullEnergy_nonneg (d : Ω → I → ℝ) (A : Finset Ω) :
  53     0≤finiteHullEnergy d A := by
  54   unfold finiteHullEnergy
  55   split_ifs
  56   · exact quadraticEnergy_nonneg _
  57   · exact le_refl _

  59 theorem finiteHullEnergy_le (d : Ω → I → ℝ) (A : Finset Ω) (hA : A.Nonempty)
  60     {z : I → ℝ} (hz : z∈convexHull ℝ (d '' (A : Set Ω))) :
  61     finiteHullEnergy d A≤quadraticEnergy z := by
  62   obtain ⟨_,_,_,hmin⟩ := finiteHullEnergy_attained d A hA
  63   exact hmin z hz

  65 theorem finiteHullEnergy_antitone (d : Ω → I → ℝ) {A B : Finset Ω}
  66     (hA : A.Nonempty) (hAB : A⊆B) : finiteHullEnergy d B≤finiteHullEnergy d A := by
  67   obtain ⟨z,hz,he,_⟩ := finiteHullEnergy_attained d A hA
  68   rw [← he]
  69   exact finiteHullEnergy_le d B (hA.mono hAB)
  70     ((convexHull_mono (Set.image_mono hAB)) hz)

  72 theorem finiteHull_coordinate_bounds (d : Ω → I → ℝ) (A : Finset Ω)
  73     (lo hi : I → ℝ) (hd : ∀ a∈A,∀ i,lo i≤d a i ∧ d a i≤hi i)
  74     {z : I → ℝ} (hz : z∈convexHull ℝ (d '' (A : Set Ω))) :
  75     ∀ i,lo i≤z i ∧ z i≤hi i := by
  76   have hsub : d '' (A : Set Ω)⊆Set.Icc lo hi := by
  77     rintro _ ⟨a,ha,rfl⟩
  78     exact ⟨fun i=>(hd a ha i).1,fun i=>(hd a ha i).2⟩
  79   have hb := (convexHull_min hsub (convex_Icc lo hi)) hz
  80   exact fun i=>⟨hb.1 i,hb.2 i⟩

  82 theorem finiteHull_lift_linear {J : Type*} (P : (I → ℝ) →ₗ[ℝ] (J → ℝ))
  83     (d : Ω → I → ℝ) (A : Finset Ω) {u : J → ℝ}
  84     (hu : u∈convexHull ℝ ((fun a=>P (d a)) '' (A : Set Ω))) :
  85     ∃ z∈convexHull ℝ (d '' (A : Set Ω)),P z=u := by
  86   have he : (fun a=>P (d a)) '' (A : Set Ω)=P '' (d '' (A : Set Ω)) :=

  88   rw [he,← P.image_convexHull] at hu
  89   exact hu
