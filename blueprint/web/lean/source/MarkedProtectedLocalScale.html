  12 private theorem protected_local_positive_linear_scale {I:Type} [Fintype I]
  13     (a b:I→ ℝ) (ha:∀ i,0≤a i) (hb:∀ i,0<b i):
  14     ∃ k:ℝ,0<k∧∀ x:ℝ,0≤x→ x≤k→ ∀ i,a i*x≤b i:=by
  15   classical
  16   let C:ℝ:=1+∑ i,a i/b i
  17   have hsum:0≤∑ i,a i/b i:=Finset.sum_nonneg (fun i _=>div_nonneg (ha i) (hb i).le)
  18   have hC:0<C:=by dsimp [C];linarith
  19   refine ⟨1/C,by positivity,?_⟩
  20   intro x hx hxk i
  21   have hi:a i/b i≤∑ j,a j/b j:=Finset.single_le_sum
  22     (fun j _=>div_nonneg (ha j) (hb j).le) (Finset.mem_univ i)
  23   have hiC:a i/b i≤C:=by dsimp [C];linarith
  24   have hi':a i≤C*b i:=(div_le_iff₀ (hb i)).mp hiC
  25   have hix:=mul_le_mul_of_nonneg_left hxk (ha i)
  26   apply hix.trans
  27   rw [mul_one_div]
  28   apply (div_le_iff₀ hC).mpr
  29   nlinarith only [hi']
