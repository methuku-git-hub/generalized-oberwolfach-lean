  13 theorem prescribed_wheel_family_exact_union (C:T→OrdinaryWheel Q F H r)

  15     (hdis:∀ i j,i≠j → Disjoint (ordinaryWheelResources (C i)) (ordinaryWheelResources (C j)))
  16     (hinner:∀ i,ordinaryWheelResources (C i)∩E={e i}) :

  19       ∀ i,(ordinaryWheelResources (C i)∩Finset.univ.image e)={e i} :=by
