  13     {V E K:Type*} [DecidableEq V] (G:SimpleGraph V) (tail head:E→V) (ktail khead:K→V)
  14     (m b:ℕ) (π:Equiv.Perm V) (hb:1≤b)
  15     (C:MarkedCycleTuple tail head ktail khead m π b)
  16     (hroot:π (C.val 0)≠C.val 0)
  17     (hE:∀e:E,G.Adj (tail e) (head e))
  18     (hK:∀e:K,G.Adj (ktail e) (π (khead e))):
  19     ∃f:Fin ((m+1)+1)↪V,
  20       (∀i:Fin (m+1),f i.castSucc=C.val i) ∧ f (Fin.last (m+1))=π (C.val 0) ∧
  21     ∃p:G.Walk (C.val 0) (π (C.val 0)), ∃hlen:p.length=m+1,
  22       p.IsPath ∧ p.support=List.ofFn f ∧
  23       ∀i:Fin (m+1),
  24         (p.darts[i.val]' (by simpa [SimpleGraph.Walk.length_darts,hlen] using i.isLt)).toProd=
  25           markedPhysicalArc tail head ktail khead π (markedCycleTypedEdge tail head ktail khead m π b C i) := by
