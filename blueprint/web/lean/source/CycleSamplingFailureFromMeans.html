  12 theorem cycle_sampling_failure_from_means
  13     (θ a q d:ℝ) (hθ:0<θ) (ha:0<a) (hq:0≤q) (hd:0<d)
  14     (hscale:2*Real.exp 1*q≤d):
  15     ∃ N:ℕ,∀ (V:Type) [Fintype V],N≤Fintype.card V→
  16     ∀ (S:Finset (V×V)) (mV:V→ℝ) (mP:V×V→ℝ),
  17     (∀ v,a*(Fintype.card V:ℝ)^2≤mV v)→
  18     (∀ z∈S,0<mP z ∧ mP z≤q*Fintype.card V)→
  19     (∑ v,2*Real.exp (-θ^2*mV v/3))+
  20       (∑ z∈S,(Real.exp 1*mP z/(d*Fintype.card V))^(d*Fintype.card V))<1:=by
  21   let c:=θ^2*a/3
  22   let b:=Real.log 2*d
  23   have hc:0<c:=by dsimp [c];positivity
  24   have hb:0<b:=by dsimp [b];positivity
  25   obtain ⟨N,hN⟩:=cycle_sampling_finite_union_margin c b hc hb
  26   refine ⟨max N 1,?_⟩
  27   intro V _ hn S mV mP hV hP
  28   have hnpos:0<(Fintype.card V:ℝ):=by
  29     exact_mod_cast (show 0<Fintype.card V by omega)
  30   let errV:=fun v=>2*Real.exp (-θ^2*mV v/3)
  31   let errP:=fun z=>(Real.exp 1*mP z/(d*Fintype.card V))^(d*Fintype.card V)
  32   apply hN V ((le_max_left N 1).trans hn) S errV errP
  33   · intro v
  34     apply cycle_sampling_relative_tail_bound θ (mV v) c (Fintype.card V)
  35     have hh:=mul_le_mul_of_nonneg_left (hV v) (sq_nonneg θ)
  36     dsimp [c];nlinarith only [hh]
  37   · intro z hz
  38     have hm:0<mP z:=(hP z hz).1
  39     have ht:0<d*(Fintype.card V:ℝ):=mul_pos hd hnpos
  40     have hhalf:Real.exp 1*mP z/(d*Fintype.card V)≤1/2:=by
  41       apply (div_le_iff₀ ht).2
  42       have h1:=mul_le_mul_of_nonneg_left (hP z hz).2 (Real.exp_pos 1).le
  43       have h2:=mul_le_mul_of_nonneg_right hscale hnpos.le
  44       nlinarith only [h1,h2]
  45     have hh:=cycle_sampling_large_tail_half (mP z) (d*Fintype.card V) hm ht hhalf
  46     dsimp [errP,b] at *
  47     convert hh using 1 <;> ring_nf
