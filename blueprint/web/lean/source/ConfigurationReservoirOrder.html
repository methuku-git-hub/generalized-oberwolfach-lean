  15 theorem configuration_reservoir_order (h c : ℕ) (K K' a L δ : ℝ)
  16     (hK : 0 < K) (hK' : 0 < K') (ha : 0 < a) (hL : 0 < L) (hδ : 0 < δ) :
  17     ∃ γ : ℝ,0 < γ ∧ ∃ n₀ : ℕ,∀ n : ℕ,n₀ ≤ n →
  18     ∀ (C I E : Type*) [Fintype C] [Fintype I] [DecidableEq C] [DecidableEq E],
  19     ∀ (A : C → Finset E) (T : I → Finset C) (P : Finset (C×C)),
  20       (∀ z∈P,z.1≠z.2) →
  21       (∀ a b,a≠b → ¬Disjoint (A a) (A b) → (a,b)∈P) →
  22       (∀ j,(A j).card ≤ h+1) →
  23       (Fintype.card C:ℝ) ≤ K*(n:ℝ)^(h+1) →
  24       (P.card:ℝ) ≤ K'*(n:ℝ)^(2*h+1) →
  25       (Fintype.card I:ℝ) ≤ L*(n:ℝ)^c →
  26       (∀ i,a*(n:ℝ)^(h+1) ≤ (T i).card) →
  27     ∃ S : Finset C,((S.biUnion A).card:ℝ) ≤ δ*n ∧
  28       (∀ j∈S,∀ k∈S,j≠k → Disjoint (A j) (A k)) ∧
  29       ∀ i,γ*n ≤ ((S∩T i).card:ℝ) := by
  30   let ρ := min 1 (min (δ/(4*(h+1)*K)) (a/(32*K')))
  31   have hρ : 0 < ρ := by dsimp [ρ];positivity
  32   have hρ1 : ρ ≤ 1 := min_le_left _ _
  33   have hρδ : ρ ≤ δ/(4*(h+1)*K) := (min_le_right _ _).trans (min_le_left _ _)
  34   have hρa : ρ ≤ a/(32*K') := (min_le_right _ _).trans (min_le_right _ _)
  35   have hsmall₁ : 4*(h+1)*K*ρ ≤ δ := by
  36     have hh := (le_div_iff₀ (by positivity : (0:ℝ)<4*(h+1)*K)).mp hρδ
  37     nlinarith only [hh]
  38   have hsmall₂ : 32*K'*ρ ≤ a := by
  39     have hh := (le_div_iff₀ (by positivity : 0<32*K')).mp hρa
  40     nlinarith only [hh]
  41   obtain ⟨n₁,hn₁⟩ := eventually_stretchedExp_le_inv_pow (2*L) c
  42     (c:=ρ*a/12) (p:=1) (by positivity) (by norm_num) 1
  43   refine ⟨a*ρ/4,by positivity,max n₁ 3,?_⟩
  44   intro n hn C I E _ _ _ _ A T P hP hconflict hA hC hPairs hI hT
  45   have hn3 : 3 ≤ n := (le_max_right _ _).trans hn
  46   have hnR : (0:ℝ)<n := by exact_mod_cast (show 0<n by omega)
  47   have hnR1 : (1:ℝ) ≤ n := by exact_mod_cast (show 1≤n by omega)
  48   let p := ρ/(n:ℝ)^h
  49   let u := K*ρ*n
  50   let v := K'*ρ^2*n
  51   have hp : 0 ≤ p := by dsimp [p];positivity
  52   have hp1 : p ≤ 1 := by
  53     apply (div_le_one (pow_pos hnR _)).mpr
  54     exact hρ1.trans (one_le_pow₀ hnR1)
  55   have hu : 0 < u := by dsimp [u];positivity
  56   have hv : 0 < v := by dsimp [v];positivity
  57   have hpow₁ : p*(n:ℝ)^(h+1)=ρ*n := by
  58     dsimp [p]
  59     rw [pow_succ]
  60     field_simp
  61   have hpow₂ : p^2*(n:ℝ)^(2*h+1)=ρ^2*n := by
  62     dsimp [p]
  63     rw [show 2*h+1=(h+h)+1 by omega,pow_succ (n:ℝ) (h+h),pow_add]
  64     field_simp
  65     <;> ring
  66   have hs : p*Fintype.card C ≤ u := by
  67     calc
  68       _ ≤ p*(K*(n:ℝ)^(h+1)) := mul_le_mul_of_nonneg_left hC hp
  69       _ = K*(p*(n:ℝ)^(h+1)) := by ring
  70       _ = _ := by rw [hpow₁];dsimp [u];ring
  71   have hps : p^2*P.card ≤ v := by
  72     calc
  73       _ ≤ p^2*(K'*(n:ℝ)^(2*h+1)) := mul_le_mul_of_nonneg_left hPairs (sq_nonneg p)
  74       _ = K'*(p^2*(n:ℝ)^(2*h+1)) := by ring
  75       _ = _ := by rw [hpow₂];dsimp [v];ring
  76   have hmean (i : I) : ρ*a*n ≤ p*(T i).card := by
  77     have hh := mul_le_mul_of_nonneg_left (hT i) hp
  78     have heq : p*(a*(n:ℝ)^(h+1))=ρ*a*n := by
  79       calc
  80         _ = a*(p*(n:ℝ)^(h+1)) := by ring
  81         _ = _ := by rw [hpow₁];ring
  82     rwa [heq] at hh
  83   have htail := hn₁ n ((le_max_left _ _).trans hn)
  84   simp only [Real.rpow_natCast,Real.rpow_one,pow_one] at htail
  85   have hfailure : ∑ i : I,2*Real.exp (-p*(T i).card/12)<1/2 := by
  86     calc
  87       _ ≤ ∑ _i : I,2*Real.exp (-(ρ*a/12)*(n:ℝ)) := by
  88         apply Finset.sum_le_sum
  89         intro i _
  90         apply mul_le_mul_of_nonneg_left (Real.exp_le_exp.mpr (by linarith [hmean i])) (by norm_num)
  91       _ = (Fintype.card I:ℝ)*(2*Real.exp (-(ρ*a/12)*(n:ℝ))) := by simp
  92       _ ≤ (L*(n:ℝ)^c)*(2*Real.exp (-(ρ*a/12)*(n:ℝ))) := mul_le_mul_of_nonneg_right hI (by positivity)
  93       _ = (2*L)*(n:ℝ)^c*Real.exp (-(ρ*a/12)*(n:ℝ)) := by ring
  94       _ ≤ 1/(n:ℝ) := htail
  95       _ < 1/2 := by
  96         apply (div_lt_iff₀ hnR).mpr
  97         have hh : (3:ℝ) ≤ n := by exact_mod_cast hn3
  98         linarith
  99   obtain ⟨S,hS,hdisj,hcompat⟩ := disjoint_configuration_selection_finite A T P hP hconflict
 100     p u v hp hp1 hu hv hs hps hfailure
 101   refine ⟨S,?_,hdisj,?_⟩
 102   · have hcard : (S.biUnion A).card ≤ S.card*(h+1) := by
 103       calc
 104         _ ≤ ∑ j∈S,(A j).card := Finset.card_biUnion_le
 105         _ ≤ ∑ _j∈S,(h+1) := Finset.sum_le_sum (fun j _ => hA j)
 106         _ = _ := by simp
 107     have hc : ((S.biUnion A).card:ℝ) ≤ (S.card:ℝ)*(h+1) := by exact_mod_cast hcard
 108     have hs' := mul_le_mul_of_nonneg_right hS (by positivity : (0:ℝ) ≤ h+1)
 109     have hm := mul_le_mul_of_nonneg_right hsmall₁ hnR.le
 110     dsimp [u] at hs'
 111     nlinarith only [hc,hs',hm]
 112   · intro i
 113     have hh := hcompat i
 114     have hm := mul_le_mul_of_nonneg_right hsmall₂ (mul_nonneg hρ.le hnR.le)
 115     have hi := hmean i
 116     dsimp [v] at hh
 117     nlinarith only [hh,hm,hi]
