  12 theorem prescribed_empty_segment_edges
  13     (Q F:SimpleGraph X) (A:Finset X) (D:X→X→Prop)
  14     (hDQ:∀ x y,D x y→Q.Adj x y) (seg:PrescribedPathSegment X)
  15     (hfree:seg.free=[])
  16     (hleft:∀ a,seg.left=some a→a∈A) (hright:∀ b,seg.right=some b→b∈A)
  17     (hchain:List.IsChain (fun x y=>D x y ∧ ¬F.Adj x y) (prescribedPathSegmentVertices seg)):
  18     prescribedOrderedPathEdges (prescribedPathSegmentVertices seg)⊆prescribedInducedSourceEdges Q A := by
  19   rcases seg with ⟨left,as,right⟩
  20   simp only at hfree
  21   subst as
  22   cases left with
  23   | none =>
  24     cases right <;> simp [prescribedPathSegmentVertices,prescribedOrderedPathEdges]
  25   | some a =>
  26     cases right with
  27     | none => simp [prescribedPathSegmentVertices,prescribedOrderedPathEdges]
  28     | some b =>
  29       have hc:List.IsChain (fun x y=>D x y ∧ ¬F.Adj x y) [a,b]:=by
  30         simpa [prescribedPathSegmentVertices] using hchain
  31       have hab:D a b:=((List.isChain_cons_cons.mp hc).1).1
  32       have ha:a∈A:=hleft a rfl
  33       have hb:b∈A:=hright b rfl
  34       intro e he
  35       have he':e=s(a,b):=by simpa [prescribedPathSegmentVertices,prescribedOrderedPathEdges] using he
  36       subst e
  37       apply Finset.mem_filter.mpr
  38       constructor
  39       · simpa only [SimpleGraph.mem_edgeFinset,SimpleGraph.mem_edgeSet] using hDQ a b hab
  40       · intro x hx
  41         simp only [Sym2.toFinset_mk_eq,Finset.mem_insert,Finset.mem_singleton] at hx
  42         rcases hx with rfl|rfl <;> assumption
