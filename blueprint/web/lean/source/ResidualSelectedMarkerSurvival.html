  11 def selectedRunInteriors {F:SimpleGraph V} {u v:V} (p:F.Walk u v)
  12     (c d:ℕ) (T:Finset ℕ) (hT:T⊆Finset.range d):Finset V :=
  13   (selectedSegmentRuns d T).attach.biUnion (fun r=>
  14     (selectedRunWalk p c r.1.1 r.1.2 ((selected_run_mem d T hT r.1.1 r.1.2).mp r.2).1).support.toFinset \
  15       {p.getVert (2+c*r.1.1),p.getVert (2+c*(r.1.2+1))})

  17 theorem selected_marker_survival_iff {F:SimpleGraph V} {u v:V}
  18     (p:F.Walk u v) (hp:p.IsPath) (c d:ℕ) (hc:0<c)
  19     (hlen:p.length=c*d+4) (T:Finset ℕ) (hT:T⊆Finset.range d)
  20     (j:ℕ) (hj:j≤d):
  21     p.getVert (2+c*j)∉selectedRunInteriors p c d T hT ↔
  22       j∈survivingSegmentMarkers d T := by
  23   rw [surviving_segment_marker_iff_avoids_runs d T hT j]
  24   constructor
  25   · intro h
  26     refine ⟨hj,?_⟩
  27     intro a b hab hmid
  28     let r : {r // r∈selectedSegmentRuns d T}:=⟨(a,b),hab⟩
  29     have hr: IsSelectedSegmentRun T a b := (selected_run_mem d T hT a b).mp hab
  30     have hbT:b∈T := hr.2.1 b hr.1 le_rfl
  31     have hbd:b<d := by have hh:=hT hbT;simpa using hh
  32     apply h
  33     apply Finset.mem_biUnion.mpr
  34     refine ⟨r,by simp,?_⟩
  35     exact (selected_run_marker_interior_iff p hp c d a b j hc hr.1 hbd hj hlen).2 hmid
  36   · rintro ⟨_,h⟩ hmem
  37     obtain ⟨r,_,hrmem⟩:=Finset.mem_biUnion.mp hmem
  38     rcases r with ⟨⟨a,b⟩,hab⟩
  39     have hr: IsSelectedSegmentRun T a b := (selected_run_mem d T hT a b).mp hab
  40     have hbT:b∈T := hr.2.1 b hr.1 le_rfl
  41     have hbd:b<d := by have hh:=hT hbT;simpa using hh
  42     have hmid: a<j ∧ j≤b := (selected_run_marker_interior_iff p hp c d a b j hc hr.1 hbd hj hlen).1 hrmem
  43     exact h a b hab hmid
