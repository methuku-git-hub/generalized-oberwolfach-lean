  12 theorem residual_actual_graph_and_prescriptions (F:SimpleGraph V) (hF:F.IsRegularOfDegree 2)
  13     (s₁ s₂:Finset I) (hs:Disjoint s₁ s₂) (u v:I→V) (P:∀i,F.Walk (u i) (v i))
  14     (hp:∀i∈s₁∪s₂,(P i).IsPath)
  15     (hP:(↑(s₁∪s₂):Set I).PairwiseDisjoint (fun i=>(P i).support.toFinset))
  16     (c d:ℕ) (hc:3≤c) (hd:0<d) (hlen:∀i∈s₁∪s₂,(P i).length=c*d+4)
  17     (T:I→Finset ℕ) (hT:∀i,T i⊆Finset.range d)
  18     (t:Finset J) (root:J→V) (C:∀j,F.Walk (root j) (root j))
  19     (hcycle:∀j∈t,(C j).IsCycle)
  20     (hC:(↑t:Set J).PairwiseDisjoint (fun j=>(C j).support.toFinset))
  21     (hcross:∀i∈s₁∪s₂,∀j∈t,Disjoint (P i).support.toFinset (C j).support.toFinset):
  22     let Z:=DeletedPathIndex s₁ s₂ d T
  23     let U:Z→V:=deletedPathStart F s₁ s₂ u v P c d T
  24     let W:Z→V:=deletedPathEnd F s₁ s₂ u v P c d T
  25     let paths:∀z:Z,F.Walk (U z) (W z):=deletedPathWalk F s₁ s₂ u v P c d T (fun i _=>hT i)
  26     let deletedV:=(t.biUnion (fun j=>(C j).support.toFinset)) ∪
  27       (Finset.univ.biUnion (fun z=>(paths z).support.toFinset \ {U z,W z}))
  28     let deletedE:=(t.biUnion (fun j=>(C j).edges.toFinset)) ∪
  29       (Finset.univ.biUnion (fun z=>(paths z).edges.toFinset))
  30     let Q:=F.deleteEdges (deletedE:Set (Sym2 V))
  31     let R:=Finset.univ\deletedV
  32     let A₀ := (s₁.biUnion (fun i=>{(P i).getVert 2,(P i).getVert (2+c*d)})) ∪
  33       (R ∩ s₂.biUnion (fun i=>(Finset.range (d+1)).image (fun j=>(P i).getVert (2+c*j))))
  34     let missing:=∑i∈s₂,(d-(T i).card)
  35     let runs:=∑i∈s₂,(selectedSegmentRuns d (T i)).card;
  36     A₀⊆R ∧ (∀ x∈A₀,∀y∈A₀,¬Q.Adj x y) ∧
  37       A₀.card=2*s₁.card+s₂.card+missing+runs ∧
  38       Q.edgeFinset.card+s₁.card+runs=R.card ∧
  39       R.card≤Q.edgeFinset.card+A₀.card ∧
  40       ∀ x∈deletedV,∀y,¬Q.Adj x y := by

  48   let Q:=F.deleteEdges (deletedE:Set (Sym2 V))

  52     apply Finset.disjoint_left.mpr
  53     intro x hx hy
  54     obtain ⟨j,hj,hxj⟩:=Finset.mem_biUnion.mp hy
  55     exact Finset.disjoint_left.mp (hcross i hi j hj) hx hxj
  56   have hid:=deleted_path_family_interiors_identity F s₁ s₂ hs u v P c d T hT
  57   have hr:=residual_graph_for_actual_path_family F hF s₁ s₂ hs u v P hp hP c d (by omega) hd hlen T
  58     (by intro i hi;exact hT i) t root C hcycle hC hcross
  59   have ha:=residual_actual_prescribed_set F Q hF (F.deleteEdges_le _) s₁ s₂ hs u v P hp hP c d hc hd hlen T hT Cset hCset
  60   have hsum:=(maximal_run_family_counts s₂ d hd T (by intro i hi;exact hT i)).2.1
  61   dsimp only at hid hr ha hsum ⊢
