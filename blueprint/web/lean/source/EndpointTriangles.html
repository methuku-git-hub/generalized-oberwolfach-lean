  15 abbrev EndpointTriangle (Q F : SimpleGraph V) (S T : Finset V) :=
  16   {X : Finset V // X⊆S∪T ∧ (X∩S).card=2 ∧ (X∩T).card=1 ∧
  17     (∀ x∈X,∀ y∈X,x≠y → Q.Adj x y) ∧ ∀ x∈X,∀ y∈X,¬F.Adj x y}

  19 def endpointTriangleVertices {Q F : SimpleGraph V} {S T : Finset V}
  20     (C : EndpointTriangle Q F S T) : Finset V := C.val

  22 theorem endpointTriangleVertices_card {Q F : SimpleGraph V} {S T : Finset V}
  23     (hST : Disjoint S T) (C : EndpointTriangle Q F S T) :
  24     (endpointTriangleVertices C).card=3 := by
  25   have hu : C.val=(C.val∩S)∪(C.val∩T) := by
  26     ext v
  27     simp only [Finset.mem_union,Finset.mem_inter]
  28     constructor
  29     · intro hv;exact (Finset.mem_union.mp (C.property.1 hv)).elim (fun h=>Or.inl ⟨hv,h⟩) (fun h=>Or.inr ⟨hv,h⟩)
  30     · tauto
  31   change C.val.card=3
  32   rw [hu,Finset.card_union_of_disjoint (hST.mono Finset.inter_subset_right Finset.inter_subset_right)]
  33   rw [C.property.2.1,C.property.2.2.1]

  42 theorem endpoint_triangle_from_tuple (Q F : SimpleGraph V) (S T : Finset V)
  43     (hST : Disjoint S T) (f : Fin 3→V) (hf : Function.Injective f)
  44     (hT : f 0∈T) (hS : ∀ j : Fin 2,f j.succ∈S)
  45     (hQ : ∀ i j,i≠j → Q.Adj (f i) (f j)) (hF : ∀ i j,¬F.Adj (f i) (f j)) :
  46     ∃ C : EndpointTriangle Q F S T,endpointTriangleVertices C=Finset.univ.image f := by
  47   let X := Finset.univ.image f
  48   have hzeroS : f 0∉S := fun h=>Finset.disjoint_left.mp hST h hT
  49   have hrowT (j : Fin 2) : f j.succ∉T := fun h=>Finset.disjoint_left.mp hST (hS j) h
  50   have hXS : X∩S=(Finset.univ.image (fun j : Fin 2=>f j.succ)) := by
  51     ext v
  52     simp only [X,Finset.mem_inter,Finset.mem_image,Finset.mem_univ,true_and]
  53     constructor
  54     · rintro ⟨⟨i,rfl⟩,hi⟩
  55       rcases Fin.eq_zero_or_eq_succ i with rfl|⟨j,rfl⟩
  56       · exact (hzeroS hi).elim
  57       · exact ⟨j,rfl⟩
  58     · rintro ⟨j,rfl⟩;exact ⟨⟨j.succ,rfl⟩,hS j⟩
  59   have hXT : X∩T={f 0} := by
  60     ext v
  61     simp only [X,Finset.mem_inter,Finset.mem_image,Finset.mem_univ,true_and,Finset.mem_singleton]
  62     constructor
  63     · rintro ⟨⟨i,rfl⟩,hi⟩
  64       rcases Fin.eq_zero_or_eq_succ i with rfl|⟨j,rfl⟩
  65       · rfl
  66       · exact (hrowT j hi).elim
  67     · rintro rfl;exact ⟨⟨0,rfl⟩,hT⟩
  68   refine ⟨⟨X,?_,?_,?_,?_,?_⟩,rfl⟩
  69   · intro v hv
  70     obtain ⟨i,_,rfl⟩ := Finset.mem_image.mp hv
  71     rcases Fin.eq_zero_or_eq_succ i with rfl|⟨j,rfl⟩
  72     · exact Finset.mem_union_right _ hT
  73     · exact Finset.mem_union_left _ (hS j)
  74   · rw [hXS,Finset.card_image_of_injective _ (show Function.Injective (fun j : Fin 2=>f j.succ) from fun i j h=>(Fin.succ_injective 2) (hf h))]
  75     simp
  76   · rw [hXT];simp
  77   · intro v hv w hw hne
  78     obtain ⟨i,_,rfl⟩ := Finset.mem_image.mp hv
  79     obtain ⟨j,_,rfl⟩ := Finset.mem_image.mp hw
  80     exact hQ i j (fun h=>hne (congrArg f h))
  81   · intro v hv w hw
  82     obtain ⟨i,_,rfl⟩ := Finset.mem_image.mp hv
  83     obtain ⟨j,_,rfl⟩ := Finset.mem_image.mp hw
  84     exact hF i j

  92 theorem endpoint_triangle_from_vertices (Q F : SimpleGraph V) (S T : Finset V)
  93     (hST : Disjoint S T) (u v w : V) (hu : u∈T) (hv : v∈S) (hw : w∈S) (hvw : v≠w)
  94     (huvQ : Q.Adj u v) (huwQ : Q.Adj u w) (hvwQ : Q.Adj v w)
  95     (huvF : ¬F.Adj u v) (huwF : ¬F.Adj u w) (hvwF : ¬F.Adj v w) :
  96     ∃ C : EndpointTriangle Q F S T,endpointTriangleVertices C={u,v,w} := by
  97   have huv : u≠v := by rintro rfl;exact Finset.disjoint_left.mp hST hv hu
  98   have huw : u≠w := by rintro rfl;exact Finset.disjoint_left.mp hST hw hu
  99   have hvuQ := huvQ.symm
 100   have hwuQ := huwQ.symm
 101   have hwvQ := hvwQ.symm
 102   have hvuF : ¬F.Adj v u := fun h=>huvF h.symm
 103   have hwuF : ¬F.Adj w u := fun h=>huwF h.symm
 104   have hwvF : ¬F.Adj w v := fun h=>hvwF h.symm
 105   have hf : Function.Injective (![u,v,w] : Fin 3→V) := by
 106     intro i j he;fin_cases i <;> fin_cases j <;> simp_all
 107   obtain ⟨C,hC⟩ := endpoint_triangle_from_tuple Q F S T hST ![u,v,w] hf hu
 108     (by intro j;fin_cases j <;> simp_all)
 109     (by intro i j hij;fin_cases i <;> fin_cases j <;> simp_all)
 110     (by intro i j;fin_cases i <;> fin_cases j <;> simp_all)
 111   refine ⟨C,hC.trans ?_⟩
 112   ext z
 113   simp only [Finset.mem_image,Finset.mem_univ,true_and,Finset.mem_insert,Finset.mem_singleton]
 114   constructor
 115   · rintro ⟨i,rfl⟩;fin_cases i <;> simp
 116   · rintro (rfl|rfl|rfl)
 117     · exact ⟨0,rfl⟩
 118     · exact ⟨1,rfl⟩
 119     · exact ⟨2,rfl⟩
