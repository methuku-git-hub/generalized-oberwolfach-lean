  10 theorem ordinary_cycle_two_vertex_count {V : Type*} [Fintype V] [DecidableEq V]
  11     (r : ℕ) (u v : V) (huv : u≠v) :
  12     (∑ C : EmbeddedCycleFamily V r,
  13       if u∈Set.range (embeddedCycleEmbedding C) ∧ v∈Set.range (embeddedCycleEmbedding C) then (1:ℝ) else 0)≤
  14       (r:ℝ)^2*(Fintype.card V:ℝ)^(r-2) := by
  15   let g (f : Fin r → V) := ∑ i,∑ j,if i≠j ∧ f i=u ∧ f j=v then (1:ℝ) else 0
  16   have hg (f : Fin r → V) : 0≤g f := by
  17     exact Finset.sum_nonneg (fun _ _ => Finset.sum_nonneg (fun _ _ => by split_ifs <;> norm_num))
  18   have hC (C : EmbeddedCycleFamily V r) : (if u∈Set.range (embeddedCycleEmbedding C) ∧ v∈Set.range (embeddedCycleEmbedding C) then (1:ℝ) else 0)≤g (embeddedCycleEmbedding C) := by
  19     by_cases hadj : u∈Set.range (embeddedCycleEmbedding C) ∧ v∈Set.range (embeddedCycleEmbedding C)
  20     · rw [if_pos hadj]
  21       obtain ⟨⟨i,hi⟩,⟨j,hj⟩⟩ := hadj
  22       have hne : i≠j := by intro he;apply huv;rw [← hi,← hj,he]
  23       have hjSum := Finset.single_le_sum (s:=Finset.univ)
  24         (f:=fun j => if i≠j ∧ embeddedCycleEmbedding C i=u ∧ embeddedCycleEmbedding C j=v then (1:ℝ) else 0)
  25         (fun _ _ => by split_ifs <;> norm_num) (Finset.mem_univ j)
  26       have hc : i≠j ∧ embeddedCycleEmbedding C i=u ∧ embeddedCycleEmbedding C j=v := ⟨hne,hi,hj⟩
  27       rw [if_pos hc] at hjSum
  28       apply hjSum.trans
  29       dsimp only [g]
  30       exact Finset.single_le_sum (f:=fun i : Fin r => ∑ j : Fin r,
  31         if i≠j ∧ embeddedCycleEmbedding C i=u ∧ embeddedCycleEmbedding C j=v then (1:ℝ) else 0)
  32         (fun _ _ => Finset.sum_nonneg (fun _ _ => by split_ifs <;> norm_num)) (Finset.mem_univ i)
  33     · simp only [if_neg hadj]
  34       exact hg _
  35   calc
  36     _ ≤ ∑ C : EmbeddedCycleFamily V r,g (embeddedCycleEmbedding C) := Finset.sum_le_sum (fun C _ => hC C)
  37     _ ≤ ∑ f : Fin r → V,g f := finite_injective_sum_le _ embeddedCycleEmbedding_injective g hg
  38     _ = ∑ i : Fin r,∑ j : Fin r,∑ f : Fin r → V,
  39         if i≠j ∧ f i=u ∧ f j=v then (1:ℝ) else 0 := by
  40       dsimp [g]
  41       rw [Finset.sum_comm]
  42       apply Finset.sum_congr rfl
  43       intro i _
  44       rw [Finset.sum_comm]
  45     _ ≤ ∑ _i : Fin r,∑ _j : Fin r,(Fintype.card V:ℝ)^(r-2) := by
  46       apply Finset.sum_le_sum
  47       intro i _
  48       apply Finset.sum_le_sum
  49       intro j _
  50       by_cases hij : i=j
  51       · simp only [hij,ne_eq,not_true_eq_false,false_and,ite_false,Finset.sum_const_zero]
  52         positivity
  53       · have hh := two_position_indicator_total i j hij u v
  54         simp only [Fintype.card_fin] at hh
  55         simp only [hij,ne_eq,not_false_eq_true,true_and]
  56         convert hh.le using 1
  57         apply Finset.sum_congr rfl
  58         intro f _
  59         split_ifs <;> rfl
  60     _ = _ := by simp only [Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul];ring
