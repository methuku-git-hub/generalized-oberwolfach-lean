   9 theorem walk_interval_mem_support_iff {F:SimpleGraph V} {u v:V} (p:F.Walk u v)
  10     (a b:ℕ) (hab:a≤b) (hb:b≤p.length) (x:V):
  11     x∈(walkInterval p a b hab).support ↔
  12       ∃ j,a≤j ∧ j≤b ∧ p.getVert j=x := by
  13   constructor
  14   · intro hx
  15     obtain ⟨k,hk,hkle⟩:=SimpleGraph.Walk.mem_support_iff_exists_getVert.mp hx
  16     have hlen:=walk_interval_length p a b hab hb
  17     have hkle':k≤b-a := by omega
  18     rw [walk_interval_getVert p a b hab k hkle'] at hk
  19     exact ⟨a+k,by omega,by omega,hk⟩
  20   · rintro ⟨j,haj,hjb,hjx⟩
  21     apply SimpleGraph.Walk.mem_support_iff_exists_getVert.mpr
  22     refine ⟨j-a,?_,?_⟩
  23     · rw [walk_interval_getVert p a b hab (j-a) (by omega)]
  24       simpa [Nat.add_sub_of_le haj] using hjx
  25     · rw [walk_interval_length p a b hab hb]
  26       omega

  36     (hp:p.IsPath) (a b a' b':ℕ) (hab:a≤b) (ha'b':a'≤b')
