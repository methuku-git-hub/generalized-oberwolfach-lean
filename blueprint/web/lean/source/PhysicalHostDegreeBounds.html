  13 theorem ordinary_physical_degree_row (Q:SimpleGraph V) (H:V→U→Prop) (x:V):
  14     (ordinaryPhysicalGraph Q H).degree (Sum.inl x)=Q.degree x+(Finset.univ.filter (H x)).card:=by
  15   have he:(ordinaryPhysicalGraph Q H).neighborFinset (Sum.inl x)=
  16       (Q.neighborFinset x).image Sum.inl∪(Finset.univ.filter (H x)).image Sum.inr:=by
  17     ext z
  18     rw [SimpleGraph.mem_neighborFinset]
  19     cases z <;> simp [ordinaryPhysicalGraph]
  20   have hd:Disjoint ((Q.neighborFinset x).image (Sum.inl:V→V⊕U))
  21       ((Finset.univ.filter (H x)).image Sum.inr):=by
  22     apply Finset.disjoint_left.mpr
  23     intro z hz hw
  24     obtain ⟨v,_,hv⟩:=Finset.mem_image.mp hz
  25     obtain ⟨u,_,hu⟩:=Finset.mem_image.mp hw
  26     exact Sum.inl_ne_inr (hv.trans hu.symm)
  27   rw [←SimpleGraph.card_neighborFinset_eq_degree,he,Finset.card_union_of_disjoint hd,
  28     Finset.card_image_of_injective _ Sum.inl_injective,Finset.card_image_of_injective _ Sum.inr_injective,
  29     SimpleGraph.card_neighborFinset_eq_degree]

  40 theorem ordinary_physical_degree_bound (Q:SimpleGraph V) (H:V→U→Prop) (r s c D:ℝ)
  41     (hr:∀ x,(Q.degree x:ℝ)≤r) (hs:∀ x,((Finset.univ.filter (H x)).card:ℝ)≤s)
  42     (hc:∀ u,((Finset.univ.filter (fun x=>H x u)).card:ℝ)≤c)
  43     (hrs:r+s≤D) (hcD:c≤D):
  44     ∀ z,((ordinaryPhysicalGraph Q H).degree z:ℝ)≤D:=by
  45   intro z
  46   cases z with
  47   | inl x=>rw [ordinary_physical_degree_row,Nat.cast_add];exact (add_le_add (hr x) (hs x)).trans hrs
  48   | inr u=>rw [ordinary_physical_degree_column];exact (hc u).trans hcD
