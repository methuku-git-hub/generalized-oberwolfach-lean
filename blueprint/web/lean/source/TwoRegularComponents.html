  17 theorem cycle_support_eq_tail_support {V : Type*} {G : SimpleGraph V}
  18     {v : V} {p : G.Walk v v} (hp : p.IsCycle) (x : V) :
  19     x ∈ p.support ↔ x ∈ p.tail.support := by
  20   rw [← SimpleGraph.Walk.cons_support_tail hp.not_nil, List.mem_cons]
  21   constructor
  22   · rintro (rfl | hx)
  23     · exact p.tail.end_mem_support
  24     · exact hx
  25   · exact Or.inr
  26 
  27 theorem cycle_length_eq_support_ncard {V : Type*} [DecidableEq V]
  28     {G : SimpleGraph V} {v : V} {p : G.Walk v v} (hp : p.IsCycle) :
  29     p.length = ({x | x ∈ p.support} : Set V).ncard := by
  30   have heq : ({x | x ∈ p.support} : Set V) = (p.tail.support.toFinset : Set V) := by
  31     ext x
  32     simp only [Set.mem_ofPred_eq, Finset.mem_coe, List.mem_toFinset]
  33     exact cycle_support_eq_tail_support hp x
  34   rw [heq, Set.ncard_coe_finset, List.toFinset_card_of_nodup hp.isPath_tail.support_nodup,
  35     SimpleGraph.Walk.length_support]
  36   exact (SimpleGraph.Walk.length_tail_add_one hp.not_nil).symm

  38 theorem regular_two_component_cycle {V : Type*} [Fintype V] [DecidableEq V]
  39     (G : SimpleGraph V) (hG : G.IsRegularOfDegree 2) (v : V) :
  40     ∃ p : G.Walk v v, p.IsCycle ∧
  41       (∀ x, x ∈ p.support ↔ G.Reachable v x) ∧
  42       p.length = Nat.card {x : V // G.Reachable v x} ∧
  43       (∀ x y, G.Reachable v x → (p.toSubgraph.Adj x y ↔ G.Adj x y)) := by
  44   have hcycles := regular_two_isCycles G hG
  45   have hn : (G.neighborSet v).Nonempty := by
  46     apply Set.nonempty_of_ncard_ne_zero
  47     rw [SimpleGraph.ncard_neighborSet, hG v]
  48     decide
  49   obtain ⟨p, hp, hverts⟩ := hcycles.exists_cycle_toSubgraph_verts_eq_connectedComponentSupp
  50     (c := G.connectedComponentMk v) (by rfl) hn
  51   have hs (x : V) : x ∈ p.support ↔ G.Reachable v x := by
  52     rw [← SimpleGraph.Walk.mem_verts_toSubgraph, hverts,
  53       SimpleGraph.ConnectedComponent.mem_supp_iff, SimpleGraph.ConnectedComponent.eq]
  54     exact G.reachable_comm
  55   refine ⟨p, hp, hs, ?_, ?_⟩
  56   · rw [cycle_length_eq_support_ncard hp]
  57     have heq : ({x | x ∈ p.support} : Set V) = {x | G.Reachable v x} :=
  58       Set.ext hs
  59     rw [heq]
  60     rfl
  61   · intro x y hx
  62     apply hp.adj_toSubgraph_iff_of_isCycles hcycles
  63     exact p.mem_verts_toSubgraph.mpr ((hs x).mpr hx)

  65 theorem AllowedCycleLengths_iff_component_cycles {n : ℕ}
  66     (F : SimpleGraph (Fin n)) (hF : F.IsRegularOfDegree 2) (S : Set ℕ) :
  67     AllowedCycleLengths F S ↔
  68       ∀ v, ∃ p : F.Walk v v, p.IsCycle ∧ p.length ∈ S ∧
  69         (∀ x, x ∈ p.support ↔ F.Reachable v x) := by
  70   constructor
  71   · intro h v
  72     obtain ⟨p, hp, hs, hlen, _⟩ := regular_two_component_cycle F hF v
  73     exact ⟨p, hp, hlen ▸ h v, hs⟩
  74   · intro h v
  75     obtain ⟨p, hp, hlen, hs⟩ := h v
  76     have hc : p.length = Nat.card {x : Fin n // F.Reachable v x} := by
  77       rw [cycle_length_eq_support_ncard hp]
  78       have heq : ({x | x ∈ p.support} : Set (Fin n)) = {x | F.Reachable v x} :=
  79         Set.ext hs
  80       rw [heq]
  81       rfl
  82     exact hc ▸ hlen
