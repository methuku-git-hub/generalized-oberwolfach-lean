   4 /-! Deleting one six-cycle in the third part adjusts edge divisibility while
   5 preserving even degrees and all edges in the first part. -/

  12 def fourBarrierSixEmbedding {t : ℕ} (ht : 6 ≤ t) :
  13     Fin 6 ↪ Fin ((t + t) + (t - 1)) :=
  14   (Fin.castLEEmb ht).trans (fourBarrierC t)
  15 
  16 def fourBarrierSixCycle {t : ℕ} (ht : 6 ≤ t) :
  17     SimpleGraph (Fin ((t + t) + (t - 1))) :=
  18   (SimpleGraph.cycleGraph 6).map (fourBarrierSixEmbedding ht)
  19 
  20 def fourBarrierGraph {t : ℕ} (ht : 6 ≤ t) :
  21     SimpleGraph (Fin ((t + t) + (t - 1))) :=
  22   fourBarrierBase t \ fourBarrierSixCycle ht
  23 
  24 theorem fourBarrierSixCycle_le {t : ℕ} (ht : 6 ≤ t) :
  25     fourBarrierSixCycle ht ≤ fourBarrierBase t := by
  26   apply SimpleGraph.map_le_iff_le_comap.mpr
  27   intro u v huv
  28   change (fourBarrierBase t).Adj
  29     (fourBarrierC t (Fin.castLEEmb ht u)) (fourBarrierC t (Fin.castLEEmb ht v))
  30   exact (fourBarrierBase_adj_C _ _ _).mpr ((Fin.castLEEmb ht).injective.ne huv.ne)
  31 
  32 theorem fourBarrierSixCycle_card {t : ℕ} (ht : 6 ≤ t) :
  33     (fourBarrierSixCycle ht).edgeFinset.card = 6 := by
  34   have hh := SimpleGraph.card_edgeFinset_map (fourBarrierSixEmbedding ht) (SimpleGraph.cycleGraph 6)
  35   have hc := cycleGraph_edge_card (by decide : 3 ≤ 6)
  36   simp only [← Set.ncard_coe_finset, SimpleGraph.coe_edgeFinset] at hh hc
  37   simpa only [fourBarrierSixCycle, ← Set.ncard_coe_finset, SimpleGraph.coe_edgeFinset]
  38     using hh.trans hc

  40 theorem fourBarrierSixCycle_degree {t : ℕ} (ht : 6 ≤ t)
  41     (x : Fin ((t + t) + (t - 1))) :
  42     (fourBarrierSixCycle ht).degree x = 0 ∨ (fourBarrierSixCycle ht).degree x = 2 := by
  43   have hh := mapped_regular_degree_or_zero (SimpleGraph.cycleGraph 6)
  44     (d := 2)
  45     (by intro v; simpa only [← SimpleGraph.ncard_neighborSet] using cycleGraph_regular (by decide : 3 ≤ 6) v)
  46     (fourBarrierSixEmbedding ht) x
  47   simpa only [fourBarrierSixCycle, ← SimpleGraph.ncard_neighborSet] using hh

  49 theorem fourBarrierGraph_degrees {t : ℕ} (ht : 6 ≤ t)
  50     (x : Fin ((t + t) + (t - 1))) :
  51     Even ((fourBarrierGraph ht).degree x) ∧
  52       2 * t - 4 ≤ (fourBarrierGraph ht).degree x := by
  53   have hbase := fourBarrierBase_degrees t x
  54   have hdel := fourBarrierSixCycle_degree ht x
  55   have hdeg := graph_degree_sdiff (fourBarrierBase t) (fourBarrierSixCycle ht)
  56     (fourBarrierSixCycle_le ht) x
  57   have hdeg' : (fourBarrierGraph ht).degree x =
  58       (fourBarrierBase t).degree x - (fourBarrierSixCycle ht).degree x := by
  59     simpa only [fourBarrierGraph, ← SimpleGraph.ncard_neighborSet] using hdeg
  60   rw [hdeg']
  61   rcases hdel with h | h
  62   · rw [h, Nat.sub_zero]
  63     exact ⟨hbase.1, by omega⟩
  64   · rw [h]
  65     exact ⟨hbase.1.tsub (by decide : Even 2), by omega⟩

  67 theorem fourBarrierGraph_edge_card {t : ℕ} (ht : 6 ≤ t) :
  68     (fourBarrierGraph ht).edgeFinset.card = 3 * t * (t - 1) - 6 := by
  69   have hh := graph_edge_count_sdiff (fourBarrierBase t) (fourBarrierSixCycle ht)
  70     (fourBarrierSixCycle_le ht)
  71   rw [fourBarrierBase_edge_card, fourBarrierSixCycle_card] at hh
  72   simpa only [fourBarrierGraph, ← Set.ncard_coe_finset, SimpleGraph.coe_edgeFinset] using hh

  74 theorem fourBarrierGraph_internal_edges {t : ℕ} (ht : 6 ≤ t) :
  75     (fourBarrierGraph ht).edgeFinset.filter (colourInternalEdge (fourBarrierColour t)) =
  76       (fourBarrierBase t).edgeFinset.filter (colourInternalEdge (fourBarrierColour t)) := by
  77   have hh := graph_filtered_edges_sdiff (fourBarrierBase t) (fourBarrierSixCycle ht)
  78     (colourInternalEdge (fourBarrierColour t)) (by
  79       intro e he hp hdel
  80       induction e using Sym2.ind with
  81       | _ u v =>
  82         rw [SimpleGraph.mem_edgeFinset, SimpleGraph.mem_edgeSet] at hdel
  83         rw [colourInternalEdge_pair] at hp
  84         change ((SimpleGraph.cycleGraph 6).map (fourBarrierSixEmbedding ht)).Adj u v at hdel
  85         rw [SimpleGraph.map_adj] at hdel
  86         obtain ⟨a, b, hab, rfl, rfl⟩ := hdel
  87         have hc : fourBarrierColour t (fourBarrierSixEmbedding ht a) = 2 :=
  88           fourBarrierColour_C t (Fin.castLEEmb ht a)
  89         rw [hc] at hp
  90         norm_num at hp)
  91   ext e
  92   have hh' := Finset.ext_iff.mp hh e
  93   simpa only [Finset.mem_filter, SimpleGraph.mem_edgeFinset, fourBarrierGraph] using hh'

  95 theorem fourBarrierGraph_no_decomposition {t : ℕ} (ht : 6 ≤ t)
  96     (hodd : Odd (t.choose 2)) :
  97     ¬ HasCycleDecomposition (fourBarrierGraph ht) 4 := by
  98   intro hdec
  99   have hallowed : ∀ u v, (fourBarrierGraph ht).Adj u v →
 100       fourBarrierAllowed (fourBarrierColour t u) (fourBarrierColour t v) := by
 101     intro u v huv
 102     exact fourBarrierBase_allowed t u v huv.1
 103   have hh := hdec.four_internal_even (fourBarrierColour t) hallowed
 104   rw [fourBarrierGraph_internal_edges, fourBarrierBase_internal_card] at hh
 105   exact (Nat.not_even_iff_odd.mpr hodd) hh
