  12     (K:ℕ) (θ:ℝ) (b:U→ℝ):

  14       ((Finset.univ.filter (fun v=>longPreparationCycleClass (F u) (hF u) K v=some i)).card:ℝ)/n

  16     let η:=fun i u=>(1-θ)*(if θ/(100*(K:ℝ))≤fraction u i then fraction u i else 0)/b u;
  17     ∀ u,(∑ i:{i:Fin K // i∈retained},η i.val u)=∑ i,η i u:=by
  18   intro fraction retained η u
