  16 theorem sequential_marked_row_covers (x:Fin n→V) (hx:Function.Injective x)
  17     (A:Finset V) (hxA:∀ i,x i∉A) (required:Fin n→Finset (Finset V⊕(V×U))) (allowedHubs:Fin n→Finset U)
  18     (hstep:∀ i:Fin n,∀ M:Finset (MarkedWheel tail head ktail khead m π b H₀ HK),
  19       (∀ W∈M,x i∉markedWheelRows W) →
  20       (∀ a,(spanningRestriction (markedPackingRimGraph M) A).degree a≤n*(4*(K+4))) →
  21       (∀ a∈A,(Finset.univ.filter (markedPackingSpokes M a)).card≤n*(K+4)) →
  22       (∀ u,(Finset.univ.filter (fun a=>markedPackingSpokes M a u)).card≤n*(m+1)) →
  23       ∃ N:Finset (MarkedWheel tail head ktail khead m π b H₀ HK),MarkedRowCover N (x i) A (required i) (allowedHubs i) K ∧
  24         Disjoint (N.biUnion markedWheelPhysicalResources) (M.biUnion markedWheelPhysicalResources)) :
  25     ∃ C:Fin n→Finset (MarkedWheel tail head ktail khead m π b H₀ HK),
  26       (∀ i,MarkedRowCover (C i) (x i) A (required i) (allowedHubs i) K) ∧
  27       (∀ i j,i≠j → Disjoint ((C i).biUnion markedWheelPhysicalResources) ((C j).biUnion markedWheelPhysicalResources)) := by
  28   let allowed:=fun (i:Fin n) (M:Finset (MarkedWheel tail head ktail khead m π b H₀ HK))=>MarkedRowCover M (x i) A (required i) (allowedHubs i) K
  29   let compatible:=fun M N:Finset (MarkedWheel tail head ktail khead m π b H₀ HK)=>
  30     Disjoint (M.biUnion markedWheelPhysicalResources) (N.biUnion markedWheelPhysicalResources)
  31   obtain ⟨C,hC,hcross⟩:=sequential_select_prefix n allowed compatible ∅ (by
  32     intro i C hC
  33     let M:=Finset.univ.biUnion C
  34     have hMr (a:V):(spanningRestriction (markedPackingRimGraph M) A).degree a≤n*(4*(K+4)):=by
  35       have hd (j:Fin n):(spanningRestriction (markedPackingRimGraph (C j)) A).degree a≤4*(K+4):=by
  36         by_cases hj:j.val < i.val
  37         · exact (hC.2.1 j hj).rimDegree a
  38         · rw [hC.1 j (by omega),markedPackingRimGraph_empty,spanningRestriction_degree]
  39           simp
  40       have he:spanningRestriction (markedPackingRimGraph M) A=
  41           ⨆ j,spanningRestriction (markedPackingRimGraph (C j)) A:=by
  42         dsimp only [M]
  43         rw [markedPackingRimGraph_biUnion]
  44         ext v w
  45         simp only [spanningRestriction_adj,SimpleGraph.iSup_adj]
  46         aesop
  47       have hh:=graph_iSup_local_degree_bound (fun j=>spanningRestriction (markedPackingRimGraph (C j)) A)
  48         Finset.univ a (4*(K+4)) hd
  49       simp only [Finset.inter_univ,SimpleGraph.card_neighborFinset_eq_degree] at hh
  50       have hcard:(Finset.univ.filter (fun j:Fin n=>∃ z∈(Finset.univ:Finset V),
  51           (spanningRestriction (markedPackingRimGraph (C j)) A).Adj a z)).card≤n:=by
  52         simpa using Finset.card_filter_le (s:=(Finset.univ:Finset (Fin n)))
  53           (p:=fun j=>∃ z∈(Finset.univ:Finset V),(spanningRestriction (markedPackingRimGraph (C j)) A).Adj a z)
  54       have hh':(⨆ j,spanningRestriction (markedPackingRimGraph (C j)) A).degree a≤n*(4*(K+4)):=
  55         hh.trans (by simpa only [Nat.mul_comm] using Nat.mul_le_mul_left (4*(K+4)) hcard)
  56       convert hh' using 1 <;> rw [he]
  57     have hMrow (a:V) (ha:a∈A):(Finset.univ.filter (markedPackingSpokes M a)).card≤n*(K+4):=by
  58       have hj (j:Fin n):(Finset.univ.filter (markedPackingSpokes (C j) a)).card≤K+4:=by
  59         by_cases hji:j.val < i.val
  60         · exact (hC.2.1 j hji).rowDegree a ha
  61         · rw [hC.1 j (by omega)];simp [markedPackingSpokes]
  62       have hh:=relation_iUnion_degree_bound (fun j=>markedPackingSpokes (C j) a) (K+4) hj
  63       simpa only [Fintype.card_fin,←markedPackingSpokes_biUnion,M] using hh
  64     have hMcol (u:U):(Finset.univ.filter (fun a=>markedPackingSpokes M a u)).card≤n*(m+1):=by
  65       have hj (j:Fin n):(Finset.univ.filter (fun a=>markedPackingSpokes (C j) a u)).card≤m+1:=by
  66         by_cases hji:j.val < i.val
  67         · exact (hC.2.1 j hji).columnDegree u
  68         · rw [hC.1 j (by omega)];simp [markedPackingSpokes]
  69       have hh:=relation_iUnion_degree_bound (fun j a=>markedPackingSpokes (C j) a u) (m+1) hj
  70       simpa only [Fintype.card_fin,←markedPackingSpokes_biUnion,M] using hh
  71     have hav:∀ W∈M,x i∉markedWheelRows W:=by
  72       intro W hW hix
  73       obtain ⟨j,_,hWj⟩:=Finset.mem_biUnion.mp hW
  74       have hji:j.val < i.val:=by
  75         by_contra hn
  76         rw [hC.1 j (by omega)] at hWj
  77         exact Finset.notMem_empty _ hWj
  78       rcases Finset.mem_insert.mp ((hC.2.1 j hji).rows W hWj hix) with he|he
  79       · have hh:=congrArg Fin.val (hx he);omega
  80       · exact hxA i he
  81     obtain ⟨N,hN,hNM⟩:=hstep i M hav hMr hMrow hMcol
  82     refine ⟨N,hN,?_⟩
  83     intro j hj
  84     apply hNM.symm.mono_left
  85     intro e he
  86     obtain ⟨W,hW,heW⟩:=Finset.mem_biUnion.mp he
  87     exact Finset.mem_biUnion.mpr ⟨W,Finset.mem_biUnion.mpr ⟨j,Finset.mem_univ _,hW⟩,heW⟩)
  88   refine ⟨C,hC,?_⟩
  89   intro i j hij
  90   rcases lt_or_gt_of_ne hij with h|h
  91   · exact hcross i j h
  92   · exact (hcross j i h).symm
