   6 theorem walk_ofFn_support_adjacent
   7     {V:Type*} (G:SimpleGraph V) (N:ℕ) (f:Fin (N+1)→V)
   8     {a b:V} (p:G.Walk a b) (hlen:p.length=N) (hsupport:p.support=List.ofFn f):
   9     ∀k:Fin N,G.Adj (f ⟨k.val,by omega⟩) (f ⟨k.val+1,by omega⟩) := by
  10   have hv (k:Fin (N+1)):p.getVert k.val=f k:=by
  11     rw [p.getVert_eq_support_getElem (by omega)]
  12     simp only [hsupport, List.getElem_ofFn]
  13   intro k
  14   have hadj:=p.adj_getVert_succ (i:=k.val) (by omega)
  15   rw [hv ⟨k.val,by omega⟩,hv ⟨k.val+1,by omega⟩] at hadj
  16   exact hadj
