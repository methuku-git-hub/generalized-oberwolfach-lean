  17 theorem ordinaryOldPair_embedding (G:SimpleGraph V₀) (H:V₀→U₀→Prop)
  18     (φ:V₀↪V) (ψ:U₀↪U) (f:V₀⊕(I×W)↪V) (g:U₀⊕I↪U)
  19     (hf:∀ a,f (.inl a)=φ a) (hg:∀ b,g (.inl b)=ψ b) :
  20     ((ordinaryOldRim G).map f=G.map φ) ∧
  21     ordinaryMappedSpokes (ordinaryOldSpokes H) f g=ordinaryMappedSpokes H φ ψ := by
  22   constructor
  23   · rw [ordinaryOldRim,SimpleGraph.map_map]
  24     congr 1
  25     funext a
  26     exact hf a
  27   · funext a b
  28     apply propext
  29     constructor
  30     · rintro ⟨x,y,h,hx,hy⟩
  31       cases x with
  32       | inr x=>exact h.elim
  33       | inl x=>
  34         cases y with
  35         | inr y=>exact h.elim
  36         | inl y=>exact ⟨x,y,h,(hf x).symm.trans hx,(hg y).symm.trans hy⟩
  37     · rintro ⟨x,y,h,hx,hy⟩
  38       exact ⟨.inl x,.inl y,h,(hf x).trans hx,(hg y).trans hy⟩

  40 theorem ordinary_embedded_attachment_resources (G:SimpleGraph V₀) (H:V₀→U₀→Prop)
  41     (C:SimpleGraph W) (z:W) (ρ:I→V₀)
  42     (φ:V₀↪V) (ψ:U₀↪U) (f:V₀⊕(I×W)↪V) (g:U₀⊕I↪U)
  43     (hf:∀ a,f (.inl a)=φ a) (hg:∀ b,g (.inl b)=ψ b) :
  44     let old:=ordinaryHostResources (G.map φ) (ordinaryMappedSpokes H φ ψ)
  45     let extra:=ordinaryHostResources ((ordinaryAttachmentRim ⊥ C z ρ).map f)
  46       (ordinaryMappedSpokes (ordinaryAttachmentSpokes (fun (_:V₀) (_:U₀)=>False) z ρ) f g)
  47     let full:=ordinaryHostResources ((ordinaryAttachmentRim G C z ρ).map f)
  48       (ordinaryMappedSpokes (ordinaryAttachmentSpokes H z ρ) f g)
  49     full=old∪extra ∧ Disjoint old extra := by
  50   dsimp only
  51   obtain ⟨hr,hs⟩:=ordinaryOldPair_embedding G H φ ψ f g hf hg
  52   have he:ordinaryHostResources ((ordinaryOldRim G).map f) (ordinaryMappedSpokes (ordinaryOldSpokes H) f g)=
  53       ordinaryHostResources (G.map φ) (ordinaryMappedSpokes H φ ψ):=by rw [hr,hs]
  54   constructor
  55   · rw [←ordinaryHostResources_embedding_image,ordinaryAttachment_resources_union,Finset.image_union,
  56       ordinaryHostResources_embedding_image,ordinaryHostResources_embedding_image,he]
  57   · have hh:Disjoint
  58         ((ordinaryHostResources (ordinaryOldRim G) (ordinaryOldSpokes H)).image (ordinaryPairResourceEmbedding f g))
  59         ((ordinaryHostResources (ordinaryAttachmentRim ⊥ C z ρ)
  60           (ordinaryAttachmentSpokes (fun (_:V₀) (_:U₀)=>False) z ρ)).image (ordinaryPairResourceEmbedding f g)):=
  61       by
  62         apply Finset.disjoint_left.mpr
  63         intro e he he'
  64         obtain ⟨a,ha,rfl⟩:=Finset.mem_image.mp he
  65         obtain ⟨b,hb,heq⟩:=Finset.mem_image.mp he'
  66         have hab: b=a := (ordinaryPairResourceEmbedding f g).injective heq
  67         subst b
  68         exact Finset.disjoint_left.mp (ordinaryAttachment_resources_disjoint G H C z ρ) ha hb
  69     rwa [ordinaryHostResources_embedding_image,ordinaryHostResources_embedding_image,he] at hh
