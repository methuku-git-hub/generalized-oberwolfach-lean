  18 abbrev ProfileRemainingClass (Z : Finset (Σ i,V i)) (i : I) :=
  19   {v : V i // (⟨i,v⟩ : Σ i,V i)∉Z}

  21 def profileComplementEmbedding (Z : Finset (Σ i,V i)) :
  22     (Σ i,ProfileRemainingClass Z i) ↪ (Σ i,V i) where
  23   toFun v := ⟨v.1,v.2.val⟩
  24   inj' := by
  25     rintro ⟨i,v⟩ ⟨j,w⟩ he
  26     have hij : i=j := congrArg Sigma.fst he
  27     subst j
  28     have hvw : v.val=w.val := eq_of_heq (Sigma.mk.inj_iff.mp he).2
  29     have hvw' : v=w := Subtype.ext hvw
  30     subst w
  31     rfl

  33 theorem profileComplementEmbedding_range (Z : Finset (Σ i,V i)) (v : Σ i,V i) :
  34     (∃ w,profileComplementEmbedding Z w=v) ↔ v∉Z := by
  35   constructor
  36   · rintro ⟨⟨i,w⟩,rfl⟩;exact w.property
  37   · rcases v with ⟨i,v⟩
  38     intro hv;exact ⟨⟨i,⟨v,hv⟩⟩,rfl⟩

  57 theorem profile_restricted_resource_class_card {C : Type*} [Fintype C] [DecidableEq C]
  58     (R : C → Finset (Σ i,V i)) (Z : Finset (Σ i,V i))
  59     (D : RestrictedResource R (profileComplementEmbedding Z)) (i : I) :
  60     (Finset.univ.filter (fun v : ProfileRemainingClass Z i =>
  61       (⟨i,v⟩ : Σ i,ProfileRemainingClass Z i)∈restrictedResourceSet R (profileComplementEmbedding Z) D)).card=
  62       (Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈R D.val)).card := by
  63   let A := {v : ProfileRemainingClass Z i //
  64     (⟨i,v⟩ : Σ i,ProfileRemainingClass Z i)∈restrictedResourceSet R (profileComplementEmbedding Z) D}
  65   let B := {v : V i // (⟨i,v⟩ : Σ i,V i)∈R D.val}
  66   let f (v : A) : B := ⟨v.val.val,(restrictedResourceSet_mem R _ D _).mp v.property⟩
  67   have hf : Function.Bijective f := by
  68     constructor
  69     · intro v w he
  70       apply Subtype.ext
  71       apply Subtype.ext
  72       exact congrArg (fun z : B => z.val) he
  73     · intro v
  74       have hv : (⟨i,v.val⟩ : Σ i,V i)∉Z :=
  75         (profileComplementEmbedding_range Z _).mp (D.property _ v.property)
  76       refine ⟨⟨⟨v.val,hv⟩,?_⟩,rfl⟩
  77       exact (restrictedResourceSet_mem R _ D _).mpr v.property
  78   have hh := Fintype.card_congr (Equiv.ofBijective f hf)
  79   simpa only [A,B,Fintype.card_subtype] using hh
