  15 theorem marked_finite_correction_physical_incidence
  16     (tail head:E→V) (ktail khead:K→V) (π:Equiv.Perm V)
  17     (A:Finset (E⊕K)) (S:Finset (V×U)) (δ:ℝ) (hδ:0≤δ)
  18     (hA:∀ v,((A.filter (fun a=>Sum.elim tail ktail a=v∨Sum.elim head khead a=v)).card:ℝ)≤δ)
  19     (hV:∀ v,((S.filter (fun z=>z.1=v)).card:ℝ)≤δ)
  20     (hU:∀ u,((S.filter (fun z=>z.2=u)).card:ℝ)≤δ)
  21     (x:V⊕U):
  22     (((markedFiniteResources tail head ktail khead π A S).filter (markedPhysicalIncident x)).card:ℝ)≤3*δ:=by
  23   have hArc (v:V):((A.filter (fun a=>v∈markedPhysicalEdge tail head ktail khead π a)).card:ℝ)≤2*δ:=by
  24     let P:=fun v a=>Sum.elim tail ktail a=v∨Sum.elim head khead a=v
  25     have hsub:A.filter (fun a=>v∈markedPhysicalEdge tail head ktail khead π a)⊆
  26         A.filter (P v)∪A.filter (P (π.symm v)):=by
  27       intro a ha
  28       obtain ⟨hmem,hi⟩:=Finset.mem_filter.mp ha
  29       cases a with
  30       | inl a=>
  31         apply Finset.mem_union.mpr;left
  32         exact Finset.mem_filter.mpr ⟨hmem,by simpa only [P,markedPhysicalEdge,Finset.mem_insert,Finset.mem_singleton,Sum.elim_inl,eq_comm] using hi⟩
  33       | inr a=>
  34         have hh:v=ktail a∨v=π (khead a):=by simpa only [markedPhysicalEdge,Sum.elim_inr,Finset.mem_insert,Finset.mem_singleton] using hi
  35         rcases hh with ht|hh
  36         · exact Finset.mem_union.mpr (Or.inl (Finset.mem_filter.mpr ⟨hmem,Or.inl ht.symm⟩))
  37         · apply Finset.mem_union.mpr;right
  38           apply Finset.mem_filter.mpr;refine ⟨hmem,Or.inr ?_⟩
  39           change khead a=π.symm v
  40           rw [hh,π.symm_apply_apply]
  41     have hc:=(Finset.card_le_card hsub).trans (Finset.card_union_le _ _)
  42     have hr:=Nat.cast_le (α:=ℝ).mpr hc
  43     simp only [Nat.cast_add] at hr
  44     linarith only [hr,hA v,hA (π.symm v)]
  45   let AR:=A.image (fun a=>(Sum.inl (markedPhysicalEdge tail head ktail khead π a):Finset V⊕(V×U)))
  46   let SR:=S.image (Sum.inr:V×U→Finset V⊕(V×U))
  47   have hsum:(((markedFiniteResources tail head ktail khead π A S).filter (markedPhysicalIncident x)).card:ℝ)≤
  48       (AR.filter (markedPhysicalIncident x)).card+(SR.filter (markedPhysicalIncident x)).card:=by
  49     have hh:=Finset.card_union_le (AR.filter (markedPhysicalIncident x)) (SR.filter (markedPhysicalIncident x))
  50     rw [←Finset.filter_union] at hh
  51     exact_mod_cast hh
  52   cases x with
  53   | inl v=>
  54     have hr:((AR.filter (markedPhysicalIncident (Sum.inl v))).card:ℝ)≤2*δ:=by
  55       have hh:((AR.filter (markedPhysicalIncident (Sum.inl v))).card)≤(A.filter (fun a=>v∈markedPhysicalEdge tail head ktail khead π a)).card:=by
  56         dsimp only [AR];rw [Finset.filter_image]
  57         simpa only [markedPhysicalIncident,←Set.ncard_coe_finset,Finset.coe_filter] using
  58           (Finset.card_image_le (s:=A.filter (fun a=>v∈markedPhysicalEdge tail head ktail khead π a))
  59             (f:=fun a=>(Sum.inl (markedPhysicalEdge tail head ktail khead π a):Finset V⊕(V×U))))
  60       exact (Nat.cast_le.mpr hh).trans (hArc v)
  61     have hs:((SR.filter (markedPhysicalIncident (Sum.inl v))).card:ℝ)≤δ:=by
  62       have hh:(SR.filter (markedPhysicalIncident (Sum.inl v))).card≤(S.filter (fun z=>z.1=v)).card:=by
  63         dsimp only [SR];rw [Finset.filter_image]
  64         simpa only [markedPhysicalIncident,←Set.ncard_coe_finset,Finset.coe_filter] using
  65           (Finset.card_image_le (s:=S.filter (fun z=>z.1=v)) (f:=(Sum.inr:V×U→Finset V⊕(V×U))))
  66       exact (Nat.cast_le.mpr hh).trans (hV v)
  67     linarith only [hsum,hr,hs]
  68   | inr u=>
  69     have hr:(AR.filter (markedPhysicalIncident (Sum.inr u))).card=0:=by
  70       simp only [AR,Finset.filter_image,markedPhysicalIncident,Finset.filter_false,Finset.image_empty,Finset.card_empty]
  71     have hs:((SR.filter (markedPhysicalIncident (Sum.inr u))).card:ℝ)≤δ:=by
  72       have hh:(SR.filter (markedPhysicalIncident (Sum.inr u))).card≤(S.filter (fun z=>z.2=u)).card:=by
  73         dsimp only [SR];rw [Finset.filter_image]
  74         simpa only [markedPhysicalIncident,←Set.ncard_coe_finset,Finset.coe_filter] using
  75           (Finset.card_image_le (s:=S.filter (fun z=>z.2=u)) (f:=(Sum.inr:V×U→Finset V⊕(V×U))))
  76       exact (Nat.cast_le.mpr hh).trans (hU u)
  77     rw [hr,Nat.cast_zero,zero_add] at hsum
  78     linarith only [hsum,hs,hδ]
