  15 theorem ordinary_resources_disjoint_of_rows (K L:Finset (Sym2 V⊕(V×U)))
  16     (h:Disjoint (ordinaryResourceRows K) (ordinaryResourceRows L)) : Disjoint K L := by
  17   apply Finset.disjoint_left.mpr
  18   intro e he he'
  19   cases e with
  20   | inl e=>
  21     induction e using Sym2.ind with
  22     | _ v w=>
  23       exact Finset.disjoint_left.mp h
  24         (show v∈ordinaryResourceRows K from Finset.mem_biUnion.mpr ⟨.inl s(v,w),he,by simp⟩)
  25         (show v∈ordinaryResourceRows L from Finset.mem_biUnion.mpr ⟨.inl s(v,w),he',by simp⟩)
  26   | inr p=>
  27     exact Finset.disjoint_left.mp h
  28       (show p.1∈ordinaryResourceRows K from Finset.mem_biUnion.mpr ⟨.inr p,he,by simp⟩)
  29       (show p.1∈ordinaryResourceRows L from Finset.mem_biUnion.mpr ⟨.inr p,he',by simp⟩)

  37 theorem OrdinaryPairHostEmbedding.resources_disjoint {G₀:SimpleGraph V₀} {Hs:V₀→U₀→Prop}
  38     {G₁:SimpleGraph V₁} {Ht:V₁→U₁→Prop} {Q:SimpleGraph V} {H:V→U→Prop}
  39     (e:OrdinaryPairHostEmbedding G₀ Hs Q H) (e':OrdinaryPairHostEmbedding G₁ Ht Q H)
  40     (h:∀ a b,e.row a≠e'.row b) : Disjoint e.resources e'.resources := by
  41   apply ordinary_resources_disjoint_of_rows
  42   apply (show Disjoint (Finset.univ.image e.row) (Finset.univ.image e'.row) from ?_).mono e.resource_rows e'.resource_rows
  43   apply Finset.disjoint_left.mpr
  44   intro v hv hv'
  45   obtain ⟨a,ha,rfl⟩:=Finset.mem_image.mp hv
  46   obtain ⟨b,hb,hba⟩:=Finset.mem_image.mp hv'
  47   exact h a b hba.symm
