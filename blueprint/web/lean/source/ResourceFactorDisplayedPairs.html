  14 theorem resource_factor_pairs_card (R : C → Finset V) (F : C → Finset (V×V)) (M : Finset C)
  15     (hM : ∀ C₀∈M,∀ D∈M,C₀≠D → Disjoint (R C₀) (R D))
  16     (hsize : ∀ C₀∈M,(F C₀).card ≤ (R C₀).card) :
  17     (M.biUnion F).card ≤ (M.biUnion R).card := by
  18   have hdis : (M : Set C).PairwiseDisjoint R := by
  19     intro x hx y hy hxy
  20     exact hM x hx y hy hxy
  21   calc
  22     _ ≤ ∑ C₀∈M,(F C₀).card := Finset.card_biUnion_le
  23     _ ≤ ∑ C₀∈M,(R C₀).card := Finset.sum_le_sum hsize
  24     _ = _ := (Finset.card_biUnion hdis).symm

  26 theorem resource_factor_pairs_incidence (R : C → Finset V) (F : C → Finset (V×V)) (M : Finset C)
  27     (hM : ∀ C₀∈M,∀ D∈M,C₀≠D → Disjoint (R C₀) (R D))
  28     (hF : ∀ C₀∈M,∀ e∈F C₀,e.1∈R C₀ ∧ e.2∈R C₀)
  29     (d : ℕ) (hd : ∀ C₀∈M,∀ v,((F C₀).filter (fun e => v=e.1 ∨ v=e.2)).card ≤ d) (v : V) :
  30     ((M.biUnion F).filter (fun e => v=e.1 ∨ v=e.2)).card ≤ d := by
  31   by_cases hx : ∃ C₀∈M,v∈R C₀
  32   · obtain ⟨C₀,hC,hv⟩ := hx
  33     have hs : (M.biUnion F).filter (fun e => v=e.1 ∨ v=e.2) ⊆
  34         (F C₀).filter (fun e => v=e.1 ∨ v=e.2) := by
  35       intro e he
  36       obtain ⟨he,hve⟩ := Finset.mem_filter.mp he
  37       obtain ⟨D,hD,heD⟩ := Finset.mem_biUnion.mp he
  38       have hvD : v∈R D := by
  39         rcases hve with h | h
  40         · exact h ▸ (hF D hD e heD).1
  41         · exact h ▸ (hF D hD e heD).2
  42       have hDC : D=C₀ := by
  43         by_contra hh
  44         exact Finset.disjoint_left.mp (hM D hD C₀ hC hh) hvD hv
  45       subst D
  46       exact Finset.mem_filter.mpr ⟨heD,hve⟩
  47     exact (Finset.card_le_card hs).trans (hd C₀ hC v)
  48   · have hz : (M.biUnion F).filter (fun e => v=e.1 ∨ v=e.2)=∅ := by
  49       apply Finset.eq_empty_iff_forall_notMem.mpr
  50       intro e he
  51       obtain ⟨he,hve⟩ := Finset.mem_filter.mp he
  52       obtain ⟨D,hD,heD⟩ := Finset.mem_biUnion.mp he
  53       have hvD : v∈R D := by
  54         rcases hve with h | h
  55         · exact h ▸ (hF D hD e heD).1
  56         · exact h ▸ (hF D hD e heD).2
  57       exact hx ⟨D,hD,hvD⟩
  58     rw [hz,Finset.card_empty]
  59     exact Nat.zero_le _

  61 theorem resource_pair_avoidance_disjoint (R : C → Finset V) (F : C → Finset (V×V))
  62     (M : Finset C) (E : Finset (V×V))
  63     (hF : ∀ C₀∈M,∀ e∈F C₀,e.1∈R C₀ ∧ e.2∈R C₀)
  64     (havoid : ∀ C₀∈M,∀ e∈E,¬(e.1∈R C₀ ∧ e.2∈R C₀)) :
  65     Disjoint E (M.biUnion F) := by
  66   apply Finset.disjoint_left.mpr
  67   intro e he hM
  68   obtain ⟨C₀,hC,heC⟩ := Finset.mem_biUnion.mp hM
  69   exact havoid C₀ hC e he (hF C₀ hC e heC)
