  13 theorem rawIntervalCovered_successor_blocked
  14     {V Γ Color : Type} [Fintype V] [DecidableEq V]
  15     (π : Equiv.Perm V) (d : ℕ) (hd : 1≤d) (σ : RawIntervalOutcome V Γ Color)
  16     (x : V) (g : Γ) (hx : rawIntervalCovered π d σ x g) :
  17     rawIntervalBlocked π d σ (π x) := by
  18   obtain ⟨a,ha,hxa⟩ := hx
  19   refine ⟨g,a,ha,?_⟩
  20   obtain ⟨i,hi⟩ := (permutationWindow_mem_iff π 0 (d-1) a x).mp hxa
  21   apply (permutationWindow_mem_iff π 0 d a (π x)).mpr
  22   refine ⟨⟨i.val+1,by have hh:=i.isLt;omega⟩,?_⟩
  23   simp only [pow_zero,Equiv.Perm.one_apply] at hi ⊢
  24   rw [pow_succ',Equiv.Perm.mul_apply,hi]

  34   have h := h₀.symm.trans hK

  38     {V U Γ Color : Type} [Fintype V] [DecidableEq V]
  39     (π : Equiv.Perm V) (d : ℕ) (hd : 1≤d) (σ : RawIntervalFamilyOutcome V U Γ Color)
  40     (g : Γ) (j : Color) (v : V) (u : U) :
  41     ¬(rawMarkedSpokes π d σ g j false (π v) u ∧ rawMarkedSpokes π d σ g j true v u) := by
  42   rintro ⟨h₀,hK⟩
  43   have hn := (rawIntervalState_label_iff π d (fun e=>σ (u,e)) (π v) j).mp h₀

  45   exact hn.1 (rawIntervalCovered_successor_blocked π d hd _ v g hc)
