  14 theorem IsEmbeddedCycle.triangle_adj {G : SimpleGraph V} (hG : IsEmbeddedCycle G 3) (x y : V) :
  15     G.Adj x y ↔ x∈G.support ∧ y∈G.support ∧ x≠y := by
  16   constructor
  17   · intro h;exact ⟨h.mem_support_left,h.mem_support_right,h.ne⟩
  18   · obtain ⟨_,f,rfl⟩:=hG
  19     rw [cycle_map_support (by norm_num : 3≤3) f]
  20     rintro ⟨⟨i,rfl⟩,⟨j,rfl⟩,hne⟩
  21     have hij : i≠j:=fun h=>hne (congrArg f h)
  22     apply (SimpleGraph.map_adj f _ _ _).mpr
  23     refine ⟨i,j,?_,rfl,rfl⟩
  24     fin_cases i <;> fin_cases j <;> first | exact (hij rfl).elim | decide

  26 theorem triangle_cycle_from_vertices_support (G : SimpleGraph V) (x v w : V)
  27     (hxv : x≠v) (hxw : x≠w) (hvw : v≠w)
  28     (hxvG : G.Adj x v) (hxwG : G.Adj x w) (hvwG : G.Adj v w) :
  29     ∃ C : OrdinaryCycle G 3,C.val.val.support.toFinset={x,v,w} := by
  30   obtain ⟨C,hC⟩:=triangle_cycle_from_vertices G x v w hxv hxw hvw hxvG hxwG hvwG
  31   refine ⟨C,hC.trans ?_⟩
  32   ext z
