  11 theorem prescribedVertices_card_le {I X V : Type*}
  12     [Fintype I] [Fintype X] [DecidableEq I] [DecidableEq V]
  13     (S : Finset V) (cl : V → I) (val : V → X)
  14     (hinj : Function.Injective (fun v => (cl v, val v))) :
  15     Fintype.card {f : I → X // ∀ v ∈ S, f (cl v) = val v} ≤
  16       Fintype.card X ^ (Fintype.card I - S.card) := by
  17   classical
  18   by_cases hex : ∃ f : I → X, ∀ v ∈ S, f (cl v) = val v
  19   · obtain ⟨f₀, hf₀⟩ := hex
  20     have hcl : Set.InjOn cl S := by
  21       intro u hu v hv huv
  22       apply hinj
  23       apply Prod.ext huv
  24       change val u = val v
  25       rw [← hf₀ u hu, ← hf₀ v hv, huv]
  26     have hcard : (S.image cl).card = S.card := Finset.card_image_of_injOn hcl
  27     have hp (f : I → X) : (∀ v ∈ S, f (cl v) = val v) ↔
  28         (∀ i ∈ S.image cl, f i = f₀ i) := by
  29       constructor
  30       · intro h i hi
  31         obtain ⟨v, hv, rfl⟩ := Finset.mem_image.mp hi
  32         exact (h v hv).trans (hf₀ v hv).symm
  33       · intro h v hv
  34         exact (h (cl v) (Finset.mem_image.mpr ⟨v, hv, rfl⟩)).trans (hf₀ v hv)
  35     have he := Fintype.card_congr (Equiv.subtypeEquivRight hp)
  36     rw [he, fixedCoordinates_card, hcard]
  37   · have hz : IsEmpty {f : I → X // ∀ v ∈ S, f (cl v) = val v} :=
  38       ⟨fun f => hex ⟨f.val, f.property⟩⟩
  39     letI := hz
  40     simp only [Fintype.card_eq_zero, zero_le]

  42 theorem prescribedVertices_weight_le {I X V : Type*}
  43     [Fintype I] [Fintype X] [DecidableEq I] [DecidableEq V]
  44     (S : Finset V) (cl : V → I) (val : V → X)
  45     (hinj : Function.Injective (fun v => (cl v, val v)))
  46     (w : (I → X) → ℝ) (M : ℝ) (hM : 0 ≤ M) (hw : ∀ f, w f ≤ M) :
  47     (∑ f : I → X, if (∀ v ∈ S, f (cl v) = val v) then w f else 0) ≤
  48       (Fintype.card X : ℝ) ^ (Fintype.card I - S.card) * M := by
  49   classical
  50   calc
  51     _ ≤ ∑ f : I → X, if (∀ v ∈ S, f (cl v) = val v) then M else 0 := by
  52       apply Finset.sum_le_sum
  53       intro f _
  54       split_ifs
  55       · exact hw f
  56       · exact le_refl _
  57     _ = (Fintype.card {f : I → X // ∀ v ∈ S, f (cl v) = val v} : ℝ) * M := by
  58       simp only [← Finset.sum_filter, Finset.sum_const, nsmul_eq_mul,
  59         Fintype.card_subtype]
  60     _ ≤ _ := mul_le_mul_of_nonneg_right (by
  61       exact_mod_cast prescribedVertices_card_le S cl val hinj) hM
