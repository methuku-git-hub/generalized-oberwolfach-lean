  14 theorem sparse_first_corner_error (H : A → B → Prop) (p q : B → ℝ)
  15     (L δ D : ℝ) (h : SparseIntersectionData H p q L δ D)
  16     (ha : 0<Fintype.card A) (hb : 0<Fintype.card B) (x : A) (u : B) :
  17     |q u/((Fintype.card A:ℝ)*Fintype.card B)*
  18       (∑ y,bipAdj H y u*bipWeightedCodegree H q x y)-1|≤
  19       L*(2*δ+(L^2+1)*(D/(Fintype.card A:ℝ))) := by
  20   let a : ℝ := Fintype.card A
  21   let b : ℝ := Fintype.card B
  22   have ha' : 0<a := Nat.cast_pos.mpr ha
  23   have hb' : 0<b := Nat.cast_pos.mpr hb
  24   have hL := h.L_nonneg
  25   have hδ := h.delta_nonneg
  26   have hD := h.D_nonneg
  27   have hS := bipWeightedCodegree_error_sum H q L δ D hL hδ h.inverse_bounds
  28     h.exceptionalSets h.exceptionalSets_card h.exceptionalSets_error x
  29   have he : (∑ y,bipAdj H y u*bipWeightedCodegree H q x y)-a*b*p u=
  30       (∑ y,bipAdj H y u*(bipWeightedCodegree H q x y-b))+b*(bipColumnDegree H u-a*p u) := by
  31     unfold bipColumnDegree
  32     simp only [mul_sub,Finset.sum_sub_distrib,Finset.mul_sum,Finset.sum_mul]
  33     ring
  34   have hsum : |(∑ y,bipAdj H y u*bipWeightedCodegree H q x y)-a*b*p u|≤
  35       a*δ*b+D*(L^2+1)*b+b*(δ*a) := by
  36     rw [he]
  37     calc
  38       _ ≤ |∑ y,bipAdj H y u*(bipWeightedCodegree H q x y-b)|+|b*(bipColumnDegree H u-a*p u)| := abs_add_le _ _
  39       _ ≤ (∑ y,|bipWeightedCodegree H q x y-b|)+b*(δ*a) := by
  40         apply add_le_add
  41         · have hh := weighted_sum_abs_le (fun y=>bipAdj H y u)
  42             (fun y=>bipWeightedCodegree H q x y-b) 1 (by norm_num)
  43             (fun y=>by rw [abs_of_nonneg (bipAdj_nonneg H y u)];exact bipAdj_le_one H y u)
  44           simpa only [one_mul] using hh
  45         · rw [abs_mul,abs_of_nonneg hb'.le]
  46           exact mul_le_mul_of_nonneg_left (h.degree_error u) hb'.le
  47       _ ≤ _ := add_le_add hS le_rfl
  48   have he' : q u/(a*b)*(∑ y,bipAdj H y u*bipWeightedCodegree H q x y)-1=
  49       q u/(a*b)*((∑ y,bipAdj H y u*bipWeightedCodegree H q x y)-a*b*p u) := by
  50     symm
  51     calc
  52       _ = q u/(a*b)*(∑ y,bipAdj H y u*bipWeightedCodegree H q x y)-q u*p u := by field_simp <;> ring
  53       _ = _ := by rw [h.inverse_identity u]
  54   change |q u/(a*b)*(∑ y,bipAdj H y u*bipWeightedCodegree H q x y)-1|≤_
  55   rw [he',abs_mul,abs_of_nonneg (div_nonneg (h.inverse_bounds u).1 (by positivity))]
  56   calc
  57     _ ≤ (L/(a*b))*(a*δ*b+D*(L^2+1)*b+b*(δ*a)) := by
  58       exact mul_le_mul (div_le_div_of_nonneg_right (h.inverse_bounds u).2 (by positivity)) hsum
  59         (abs_nonneg _) (by positivity)
  60     _ = _ := by field_simp <;> ring
