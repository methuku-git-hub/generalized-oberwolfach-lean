  14 def IsMarkedCycleTuple {V E K : Type*} (tail head : E → V) (ktail khead : K → V)
  15     (m : ℕ) (π : Equiv.Perm V) (b : ℕ) (C : Fin (m+1) → V) : Prop :=
  16   PermutationSeparated π b C ∧
  17   (∀ j : Fin m,∃ e,tail e=C j.castSucc ∧ head e=C j.succ) ∧
  18   ∃ e,ktail e=C (Fin.last m) ∧ khead e=C 0

  21     (m : ℕ) (π : Equiv.Perm V) (b : ℕ) :=

  24 def markedCycleIncidence {V E K : Type*} [DecidableEq V]
  25     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (C : Fin (m+1) → V) : E⊕K → ℝ :=
  26   Sum.elim (fun e => ∑ j : Fin m,directedEdgeIndicator tail head (C j.castSucc) (C j.succ) e)
  27     (directedEdgeIndicator ktail khead (C (Fin.last m)) (C 0))

  29 def markedBalanceSubspace {V E K : Type*} [Fintype V] [Fintype E] [Fintype K] [DecidableEq V]
  30     (tail head : E → V) (ktail khead : K → V) (m : ℕ) : Submodule ℝ (E⊕K → ℝ) where
  31   carrier := {w | occurrenceSumLinear (Sum.elim tail ktail) w=occurrenceSumLinear (Sum.elim head khead) w ∧
  32     (∑ e : E,w (Sum.inl e))=(m:ℝ)*(∑ e : K,w (Sum.inr e))}
  33   zero_mem' := by simp
  34   add_mem' := by
  35     intro w u hw hu
  36     constructor
  37     · simp only [map_add,hw.1,hu.1]
  38     · simp only [Pi.add_apply,Finset.sum_add_distrib,hw.2,hu.2,mul_add]
  39   smul_mem' := by
  40     intro a w hw
  41     constructor
  42     · simp only [map_smul,hw.1]
  43     · simp only [Pi.smul_apply,smul_eq_mul,← Finset.mul_sum,hw.2]
  44       ring

  46 theorem markedCycleIncidence_nonneg {V E K : Type*} [DecidableEq V]

  48     0≤markedCycleIncidence tail head ktail khead m C e := by

  56   | inr e =>
  57     simp only [markedCycleIncidence,Sum.elim_inr,directedEdgeIndicator]
  58     split_ifs <;> norm_num

  60 theorem markedCycleIncidence_balanced {V E K : Type*} [Fintype V] [Fintype E] [Fintype K] [DecidableEq V]
  61     (tail head : E → V) (ktail khead : K → V)
  62     (hinj : Function.Injective (fun e => (tail e,head e)))
  63     (hkinj : Function.Injective (fun e => (ktail e,khead e)))
  64     (m : ℕ) (C : Fin (m+1) → V)
  65     (hO : ∀ j : Fin m,∃ e,tail e=C j.castSucc ∧ head e=C j.succ)
  66     (hK : ∃ e,ktail e=C (Fin.last m) ∧ khead e=C 0) :
  67     markedCycleIncidence tail head ktail khead m C∈markedBalanceSubspace tail head ktail khead m := by
  68   classical
  69   constructor
  70   · funext v
  71     change (∑ e : E⊕K,if Sum.elim tail ktail e=v then markedCycleIncidence tail head ktail khead m C e else 0)=
  72       ∑ e : E⊕K,if Sum.elim head khead e=v then markedCycleIncidence tail head ktail khead m C e else 0
  73     simp only [Fintype.sum_sum_type,markedCycleIncidence,Sum.elim_inl,Sum.elim_inr]
  74     have ho (f : E → V) : (∑ e,if f e=v then (∑ j : Fin m,directedEdgeIndicator tail head (C j.castSucc) (C j.succ) e) else 0)=
  75         ∑ j : Fin m,occurrenceSumLinear f (directedEdgeIndicator tail head (C j.castSucc) (C j.succ)) v := by
  76       calc
  77         _ = ∑ e,∑ j : Fin m,if f e=v then directedEdgeIndicator tail head (C j.castSucc) (C j.succ) e else 0 := by
  78           apply Finset.sum_congr rfl
  79           intro e _
  80           by_cases h : f e=v <;> simp only [h,ite_true,ite_false,Finset.sum_const_zero]
  81         _ = _ := Finset.sum_comm
  82     rw [ho tail,ho head]
  83     change (∑ j : Fin m,occurrenceSumLinear tail (directedEdgeIndicator tail head (C j.castSucc) (C j.succ)) v)+
  84       occurrenceSumLinear ktail (directedEdgeIndicator ktail khead (C (Fin.last m)) (C 0)) v=
  85       (∑ j : Fin m,occurrenceSumLinear head (directedEdgeIndicator tail head (C j.castSucc) (C j.succ)) v)+
  86       occurrenceSumLinear khead (directedEdgeIndicator ktail khead (C (Fin.last m)) (C 0)) v
  87     simp only [directedEdgeIndicator_tail tail head hinj _ _ (hO _),
  88       directedEdgeIndicator_head tail head hinj _ _ (hO _),
  89       directedEdgeIndicator_tail ktail khead hkinj _ _ hK,
  90       directedEdgeIndicator_head ktail khead hkinj _ _ hK]
  91     have hfirst := Fin.sum_univ_castSucc (fun j : Fin (m+1) => if C j=v then (1:ℝ) else 0)
  92     have hlast := Fin.sum_univ_succ (fun j : Fin (m+1) => if C j=v then (1:ℝ) else 0)
  93     linarith
