  11 theorem ordinary_profile_weight_rescale (r : ℕ) (hr : 1≤r)

  13     (hb : 0≤b) (hB : 0≤B) (hlow : β*n/2≤r*k) (hhigh : (r:ℝ)*k≤n)
  14     (hw : b/n^(r-1)≤w ∧ w≤B/n^(r-1)) :
  15     (b*(β/2)^(r-1)/(r:ℝ)^(r-1))/k^(r-1)≤w ∧
  16       w≤(B/(r:ℝ)^(r-1))/k^(r-1) := by
  17   have hrR : (0:ℝ)<r := by exact_mod_cast (show 0<r by omega)
  18   have hden : (β/2*n)^(r-1)≤((r:ℝ)*k)^(r-1) :=
  19     pow_le_pow_left₀ (by positivity) (by nlinarith only [hlow]) _
  20   have hden' : (β/2)^(r-1)*n^(r-1)≤(r:ℝ)^(r-1)*k^(r-1) := by
  21     simpa only [mul_pow] using hden
  22   constructor
  23   · apply le_trans _ hw.1
  24     rw [div_div]
  25     apply (div_le_div_iff₀ (by positivity : (0:ℝ)<(r:ℝ)^(r-1)*k^(r-1)) (by positivity)).mpr
  26     nlinarith only [mul_le_mul_of_nonneg_left hden' hb]
  27   · apply hw.2.trans
  28     rw [div_div]
  29     apply div_le_div_of_nonneg_left hB (by positivity)
