  13 theorem triangle_transformer_sample (z : ℕ) (β ρ : ℝ)
  14     (hβ : 0<β) (hβ1 : β≤1) (hρ : 0<ρ) :
  15     ∃ s₀ : ℕ,∀ s : ℕ,s₀≤s → ∃ N : ℕ,∀ (V : Type) [Fintype V] [DecidableEq V],
  16       N≤Fintype.card V → ∀ (G : SimpleGraph V) (A Z : Finset V),Z.card=z →
  17       ρ*Fintype.card V≤(A.card:ℝ) →
  18       (∀ v∈A∪Z,(3/4+β/2)*(A.card:ℝ)≤((G.neighborFinset v∩A).card:ℝ)) →
  19       ∃ S : Finset V,S.card=s ∧ S⊆A ∧ Disjoint S Z ∧
  20         (∀ v∈S∪Z,(3/4+β/8)*(s:ℝ)≤((G.neighborFinset v∩S).card:ℝ)) := by
  21   obtain ⟨K,hK⟩:=external_rooted_dense_sets z β ρ hβ hβ1 hρ
  22   obtain ⟨M,hM⟩:=exists_nat_gt ((8/β)*z)
  23   refine ⟨max K M,?_⟩
  24   intro s hs
  25   obtain ⟨c,hc,hcN⟩:=hK s ((le_max_left K M).trans hs)
  26   obtain ⟨N,hN⟩:=hcN 0 (le_refl _)
  27   refine ⟨max N 1,?_⟩
  28   intro V _ _ hn G A Z hZ hA hdeg
  29   have hcount:=hN V ((le_max_left N 1).trans hn) G ⊥ A Z hZ hA (3/4)
  30     (by norm_num) (by norm_num) hdeg (by simp) (by simp)
  31   have hnpos : 0<(Fintype.card V:ℝ) := by
  32     exact_mod_cast (le_max_right N 1).trans hn
  33   have hpos : 0<c*(Fintype.card V:ℝ)^s := by positivity
  34   have hcardpos := hpos.trans_le hcount
  35   have hcardNat : 0<Fintype.card {T : FixedSubset V (z+s) // Z⊆T.val ∧ T.val⊆A∪Z ∧
  36       (∀ v∈T.val,(3/4+β/4)*(z+s:ℝ)≤((G.neighborFinset v∩T.val).card:ℝ)) ∧
  37       (∀ x∈T.val,∀ y∈T.val,¬(⊥:SimpleGraph V).Adj x y)} := by exact_mod_cast hcardpos
  38   obtain ⟨T⟩:=Fintype.card_pos_iff.mp hcardNat
  39   let S:=T.val.val\Z
  40   have hScard : S.card=s := by
  41     dsimp [S]
  42     rw [Finset.card_sdiff_of_subset T.property.1,T.val.property,hZ]
  43     omega
  44   have hSZ : S∪Z=T.val.val := Finset.sdiff_union_of_subset T.property.1
  45   have hSA : S⊆A := by
  46     intro v hv
  47     have hh:=Finset.mem_sdiff.mp hv
  48     rcases Finset.mem_union.mp (T.property.2.1 hh.1) with h|h
  49     · exact h
  50     · exact (hh.2 h).elim
  51   refine ⟨S,hScard,hSA,Finset.sdiff_disjoint,?_⟩
  52   intro v hv
  53   have hT:=T.property.2.2.1 v (hSZ ▸ hv)
  54   have hsplit : G.neighborFinset v∩T.val.val⊆(G.neighborFinset v∩S)∪Z := by
  55     intro w hw
  56     have hwT:=(Finset.mem_inter.mp hw).2
  57     by_cases hwZ:w∈Z
  58     · exact Finset.mem_union_right _ hwZ
  59     · exact Finset.mem_union_left _ (Finset.mem_inter.mpr
  60         ⟨(Finset.mem_inter.mp hw).1,Finset.mem_sdiff.mpr ⟨hwT,hwZ⟩⟩)
  61   have hcard:=(Finset.card_le_card hsplit).trans (Finset.card_union_le _ _)
  62   rw [hZ] at hcard
  63   have hcardR : ((G.neighborFinset v∩T.val.val).card:ℝ)≤(G.neighborFinset v∩S).card+(z:ℝ) := by
  64     exact_mod_cast hcard
  65   have hMs : (M:ℝ)≤s := by exact_mod_cast (le_max_right K M).trans hs
  66   have hmargin:=mul_le_mul_of_nonneg_right (hM.le.trans hMs) (by positivity : (0:ℝ)≤β/8)
  67   have he : ((8/β)*(z:ℝ))*(β/8)=z := by field_simp
  68   rw [he] at hmargin
  69   have hnonneg : 0≤(3/4+β/4)*(z:ℝ) := by positivity
  70   nlinarith only [hT,hcardR,hmargin,hnonneg]
