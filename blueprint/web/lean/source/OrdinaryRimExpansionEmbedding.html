  13 theorem ordinary_rim_expansion_embedding (β : ℝ) (hβ : 0<β) (r b : ℕ) (hr : 3≤r) :
  14     ∃ n₀ : ℕ,∀ n : ℕ,n₀≤n →
  15     ∀ (V U : Type*) [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U],Fintype.card V≤n →
  16     ∀ (G : SimpleGraph V) (H : V → U → Prop),
  17     (∀ a,(1/2+β)*(n:ℝ)≤G.degree a) →
  18     (∀ a,(1-1/(2*(r:ℝ)))*(Fintype.card U:ℝ)≤(Finset.univ.filter (H a)).card) →
  19     ∀ (R : Finset V) (B : Finset U) (x y : V),R.card≤b → x∈R → y∈R → 2*B.card<Fintype.card U →
  20     ∃ (f : Fin r ↪ V) (u : U),
  21       (∀ i j,(SimpleGraph.cycleGraph r).Adj i j → G.Adj (f i) (f j)) ∧
  22       (∀ i,H (f i) u) ∧ (∀ i,f i∉R) ∧ u∉B ∧
  23       G.Adj y (f ⟨0,by omega⟩) ∧ G.Adj x (f ⟨r-1,by omega⟩) ∧
  24       (∀ i,f i≠x ∧ f i≠y) := by
  25   obtain ⟨n₁,hpath⟩ := fixed_length_path β hβ (r-1) b (by omega)
  26   let n₂ := Nat.ceil (((b:ℝ)+1)/β)+1
  27   refine ⟨max n₁ n₂,?_⟩
  28   intro n hn V U _ _ _ _ hV G H hdeg hH R B x y hR hx hy hB
  29   have hceil : (Nat.ceil (((b:ℝ)+1)/β):ℝ)<n := by
  30     exact_mod_cast (show Nat.ceil (((b:ℝ)+1)/β)<n by have h: n₂≤n:=(le_max_right _ _).trans hn;dsimp [n₂] at h;omega)
  31   have hmargin : (b:ℝ)+1<(n:ℝ)*β :=
  32     (div_lt_iff₀ hβ).mp ((Nat.le_ceil (((b:ℝ)+1)/β)).trans_lt hceil)
  33   have hRR : (R.card:ℝ)≤b := by exact_mod_cast hR
  34   have hnR : (0:ℝ)≤n := Nat.cast_nonneg _
  35   have hy_size : R.card<(G.neighborFinset x).card := by
  36     rw [SimpleGraph.card_neighborFinset_eq_degree]
  37     have hh : (R.card:ℝ)<G.degree x := by nlinarith [hdeg x]
  38     exact_mod_cast hh
  39   obtain ⟨y',hy',hy'R⟩ := Finset.exists_mem_notMem_of_card_lt_card hy_size
  40   have hxy' : G.Adj x y' := by simpa only [SimpleGraph.mem_neighborFinset] using hy'
  41   have hx_size : R.card<(G.neighborFinset y∩G.neighborFinset y').card := by
  42     have hh := dense_common_neighbours G n β hV hdeg y y'
  43     have hlt : (R.card:ℝ)<((G.neighborFinset y∩G.neighborFinset y').card:ℝ) := by nlinarith
  44     exact_mod_cast hlt
  45   obtain ⟨x',hx',hx'R⟩ := Finset.exists_mem_notMem_of_card_lt_card hx_size
  46   have hyx' : G.Adj y x' := by simpa only [SimpleGraph.mem_neighborFinset] using (Finset.mem_inter.mp hx').1
  47   have hy'x' : G.Adj y' x' := by simpa only [SimpleGraph.mem_neighborFinset] using (Finset.mem_inter.mp hx').2
  48   obtain ⟨p,hp,hlen,havoid⟩ := hpath n ((le_max_left _ _).trans hn) V hV G hdeg
  49     x' y' hy'x'.ne.symm R hR hx'R hy'R
  50   let f := pathCycleEmbedding p hp hlen
  51   have hfavoid : ∀ i,f i∉R := pathCycleEmbedding_avoids (R:Set V) p hp hlen havoid
  52   let A := Finset.univ.image f
  53   have hA : A.card≤r := by
  54     exact (Finset.card_image_le).trans (by simp)
  55   obtain ⟨u,hu,hAu⟩ := common_spoke_hub_avoiding H A B r (by omega) hA (fun a _ => hH a) hB
  56   refine ⟨f,u,pathCycleEmbedding_hom hr p hp hlen hy'x'.symm,?_,hfavoid,hu,?_,?_,?_⟩
  57   · intro i
  58     exact hAu _ (Finset.mem_image.mpr ⟨i,Finset.mem_univ _,rfl⟩)
