  14 theorem marked_free_rim_and_spoke_extension_disjoint (hm:1≤m) (hb:1≤b)
  15     (C:I→MarkedWheel tail head ktail khead m π b H₀ HK)
  16     (W:MarkedWheel tail head ktail khead m π b H₀ HK)
  17     (X:Finset V) (T:I→Finset (Finset V)) (T':Finset (Finset V))
  18     (hold:∀ i e,e∈markedCyclePhysicalEdges tail head ktail khead m π b (C i).1→e⊆X∨e∈T i)
  19     (hnew:∀ e,e∈markedCyclePhysicalEdges tail head ktail khead m π b W.1→e⊆X∨e∈T')
  20     (houtside:∀ i e,e∈T i→¬e⊆X) (hroots:∀ i,Disjoint (T i) T')
  21     (hspokes:∀ i,(C i).2.val=W.2.val→Disjoint (markedWheelRows (C i)) (markedWheelRows W))
  22     (hD:Disjoint (markedWheelRimGraph W) (spanningRestriction (markedFamilyRim C) X)):
  23     ∀ i,Disjoint (markedWheelPhysicalResources (C i)) (markedWheelPhysicalResources W) := by
  24   intro i
  25   apply Finset.disjoint_left.mpr
  26   intro e he hw
  27   cases e with
  28   | inr e=>
  29     have hc:=(markedWheelPhysicalResources_spoke (C i) e.1 e.2).mp he
  30     have hn:=(markedWheelPhysicalResources_spoke W e.1 e.2).mp hw
  31     exact Finset.disjoint_left.mp (hspokes i (hc.1.trans hn.1.symm)) hc.2 hn.2
  32   | inl e=>
  33     rw [markedWheelPhysicalResources_rim] at he hw
  34     rcases hold i e he with hx|ht
  35     · have hcard:e.card=2:=by
  36         obtain ⟨a,ha,h⟩:=Finset.mem_image.mp he
  37         rw [←h]
  38         exact markedCycle_physical_edge_card tail head ktail khead m π b hm hb (C i).1 a
  39       obtain ⟨x,y,hxy,rfl⟩:=Finset.card_eq_two.mp hcard
  40       have hd:(spanningRestriction (markedFamilyRim C) X).Adj x y:=
  41         (spanningRestriction_adj _ _ _ _).mpr
  42           ⟨SimpleGraph.iSup_adj.mpr ⟨i,hxy,he⟩,hx (by simp),hx (by simp)⟩
  43       exact disjoint_graphs_no_common_edge _ _ hD x y ⟨hxy,hw⟩ hd
  44     · rcases hnew e hw with hx|ht'
  45       · exact houtside i e ht hx
  46       · exact Finset.disjoint_left.mp (hroots i) ht ht'
