  10 theorem residual_actual_prescribed_set_bound (F Q:SimpleGraph V)
  11     (hF:F.IsRegularOfDegree 2) (hQ:Q≤F)
  12     (s₁ s₂:Finset I) (hs:Disjoint s₁ s₂)
  13     (u v:I→V) (P:∀i,F.Walk (u i) (v i))
  14     (hp:∀i∈s₁∪s₂,(P i).IsPath)
  15     (hP:(↑(s₁∪s₂):Set I).PairwiseDisjoint (fun i=>(P i).support.toFinset))
  16     (c d:ℕ) (hc:3≤c) (hd:0<d) (hlen:∀i∈s₁∪s₂,(P i).length=c*d+4)
  17     (T:I→Finset ℕ) (hT:∀i,T i⊆Finset.range d) (C:Finset V)
  18     (hC:∀i∈s₁∪s₂,Disjoint (P i).support.toFinset C):
  19     let R:=Finset.univ \ (C ∪ (s₁∪s₂).biUnion (pathFamilyLocalDeletion F s₁ u v P c d T hT))
  20     let A₀ := (s₁.biUnion (fun i=>{(P i).getVert 2,(P i).getVert (2+c*d)})) ∪
  21       (R ∩ s₂.biUnion (fun i=>(Finset.range (d+1)).image (fun j=>(P i).getVert (2+c*j))))
  22     let missing:=∑i∈s₂,(d-(T i).card);
  23     A₀.card≤2*(s₁.card+s₂.card)+2*missing := by
  24   have hA:=residual_actual_prescribed_set F Q hF hQ s₁ s₂ hs u v P hp hP c d hc hd hlen T hT C hC
  25   have hr:=(maximal_run_family_counts s₂ d hd T (by intro i hi;exact hT i)).2.1
  26   dsimp only at hA hr ⊢
  27   omega
