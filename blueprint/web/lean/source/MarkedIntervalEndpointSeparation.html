  10 theorem marked_interval_roots_disjoint_endpoints
  11     {V I:Type*} [Fintype V] [Fintype I] [DecidableEq V]
  12     (π:Equiv.Perm V) (k:I→ℕ) (a:I→V)
  13     (hinj:∀j,Function.Injective (fun i:Fin (k j+1)=>(π^i.val) (a j)))
  14     (hdis:∀j l:I,j≠l→Disjoint (permutationWindow π 0 (k j) (a j))
  15       (permutationWindow π 0 (k l) (a l))):
  16     Disjoint (markedIntervalRootSet π k a)
  17       (Finset.univ.image (fun j=>(π^(k j)) (a j))):=by
  18   classical
  19   apply Finset.disjoint_left.mpr
  20   intro x hx hy
  21   obtain ⟨j,_,hj⟩:=Finset.mem_biUnion.mp hx
  22   obtain ⟨i,_,hi⟩:=Finset.mem_image.mp hj
  23   obtain ⟨l,_,hl⟩:=Finset.mem_image.mp hy
  24   have hroot:(π^i.val) (a j)∈permutationWindow π 0 (k j) (a j):=by
  25     unfold permutationWindow
  26     exact Finset.mem_image.mpr ⟨⟨i.val,by omega⟩,Finset.mem_univ _,by simp; rfl⟩
  27   have hend:(π^(k l)) (a l)∈permutationWindow π 0 (k l) (a l):=by
  28     unfold permutationWindow
  29     exact Finset.mem_image.mpr ⟨⟨k l,by omega⟩,Finset.mem_univ _,by simp; rfl⟩
  30   by_cases hjl:j=l
  31   · subst l
  32     have heq:(π^i.castSucc.val) (a j)=(π^(Fin.last (k j)).val) (a j):=hi.trans hl.symm
  33     have he:=hinj j heq
  34     have hv:=congrArg Fin.val he
  35     simp only [Fin.val_castSucc,Fin.val_last] at hv
  36     omega
  37   · apply Finset.disjoint_left.mp (hdis j l hjl) hroot
  38     have hv:(π^i.val) (a j)=(π^(k l)) (a l):=hi.trans hl.symm
