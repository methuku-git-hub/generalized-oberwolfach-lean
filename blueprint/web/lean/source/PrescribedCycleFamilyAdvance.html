  13 theorem prescribed_cycle_family_advance
  14     (QV:I→Finset X) (Q:I→SimpleGraph X) (A₀:I→Finset X) (ψ:I→X→V)
  15     (E₀:I→Finset (Sym2 V)) (J:SimpleGraph V)

  20     (hold:∀ v,((Finset.univ.filter (fun i=>∃ x,x∈A₀ i ∧ ψ i x=v)).card:ℝ)≤δ*Fintype.card V)
  21     (hJ:∀ v,(J.degree v:ℝ)≤δ*Fintype.card V)
  22     (k:I) (Dsrc:X→X→Prop) (Fsrc:SimpleGraph X) (hDQ:∀ x y,Dsrc x y→(Q k).Adj x y)
  23     (D:V→V→Prop) (L:Finset V) (hext:IsExtendingList D L β)
  24     (a:X) (xs:List X) (hnodup:(a::xs).Nodup) (hlen:3≤(a::xs).length)
  25     (hsourcecycle:DirectedCycleThrough Dsrc Fsrc (a::xs))
  26     (st:PrescribedFamilyState I X V)
  27     (hvalid:st.Valid QV Q A₀ ψ E₀ J (Real.sqrt δ*Fintype.card V))
  28     (hinit:prescribedInducedSourceEdges (Q k) (A₀ k)⊆st.sourceEdges k)
  29     (hfree:∀ x∈a::xs,x∉A₀ k→x∉st.assigned k) (hsubset:∀ x∈a::xs,x∈QV k):
  30     ∃ st':PrescribedFamilyState I X V,
  31       st'.Valid QV Q A₀ ψ E₀ J (Real.sqrt δ*Fintype.card V) ∧
  32       st'.assigned=Function.update st.assigned k (st.assigned k∪(a::xs).toFinset) ∧
  33       (∀ i,st.assigned i⊆st'.assigned i) ∧ (∀ i,st.sourceEdges i⊆st'.sourceEdges i) ∧
  34       (∀ i x,x∈st.assigned i→st'.vertexMap i x=st.vertexMap i x) ∧
  35       (∀ x∈a::xs,x∉A₀ k→st'.vertexMap k x∈L) ∧
  36       (∀ e∈prescribedDirectedPathEdges ((a::xs)++[a]),
  37         (e.1∈A₀ k ∧ e.2∈A₀ k) ∨ D (st'.vertexMap k e.1) (st'.vertexMap k e.2)) ∧
  38       (a::xs).toFinset⊆st'.assigned k ∧
  39       prescribedOrderedPathEdges ((a::xs)++[a])⊆st'.sourceEdges k := by

  41   · obtain ⟨z,hzin,hza⟩:=hex

  44         hQsize hQdegree hold hJ k Dsrc Fsrc hDQ D L hext a xs hnodup hlen ha hsourcecycle st hvalid hinit hfree hsubset

  46         rcases List.mem_cons.mp hzin with hz|hz
  47         · subst z;exact (ha hza).elim
  48         · exact hz
  49       obtain ⟨ys,hrotNodup,hrotLen,hrotSet,hrotChain,hrotPhys,hrotDir⟩:=

  66       refine ⟨last,hl,?_,hA,hE,hpres,?_,?_,?_,?_⟩
  67       · simpa only [←hrotSet] using hExact
  68       · intro x hx hnot;exact hL x ((hmem x).mpr hx) hnot
  69       · intro e he;apply harcs e;rw [←hrotDir];exact he
  70       · rw [hrotSet];exact hvertices
  71       · rw [hrotPhys];exact hedges
  72   · have hnoA:∀ x∈a::xs,x∉A₀ k:=by intro x hx hxa;exact hex ⟨x,hx,hxa⟩

  74       prescribed_cycle_free_advance QV Q A₀ ψ E₀ J β δ hβ hβ1 hδ hδbound hnβ hn hI

  76     refine ⟨last,hl,hExact,hA,hE,hpres,hL,?_,hvertices,hedges⟩
  77     intro e he;exact Or.inr (harcs e he)
