  14 theorem finite_real_common_upper_bound {I : Type*} [Finite I] (f : I → ℝ) :
  15     ∃ K : ℝ,1<K ∧ ∀ i,f i≤K := by
  16   obtain ⟨M,hM⟩ := (Set.finite_range f).bddAbove
  17   refine ⟨max 2 M,lt_of_lt_of_le (by norm_num) (le_max_left _ _),?_⟩
  18   intro i
  19   exact (hM (Set.mem_range_self i)).trans (le_max_right _ _)

  21 theorem profile_sampling_numerical_margins (c : ℕ) (hc : 2≤c)
  22     (A Dₐ α ξ : ℝ) (hA : 0≤A) (hDₐ : 0≤Dₐ) (hα : 0<α) (hξ : 0<ξ) (hξ1 : ξ≤1) :
  23     ∃ N : ℕ,c+c*c≤N ∧ ∀ t : ℕ,N≤t → ∀ F₀ D : ℝ,0≤F₀ → 0≤D →
  24       ∃ k₀ : ℝ,1<k₀ ∧ ∀ k : ℝ,k₀≤k →
  25         0<k ∧ 2*(t:ℝ)≤k ∧ 2*F₀≤k ∧
  26         2*(c-1:ℕ)*(F₀+1+(c-1:ℕ))≤k ∧
  27         2*(c*c:ℕ)*(F₀+c+(c*c:ℕ))≤k ∧
  28         ((ξ/4)+F₀*(A/k))+(1+(ξ/4)+F₀*(A/k))*
  29           (2*(c-1:ℕ)*(((1:ℝ)+(c-1:ℕ))/t+(F₀+1+(c-1:ℕ))/k))≤ξ ∧
  30         2*(c*c:ℕ)*(((c:ℝ)+(c*c:ℕ))/t+(F₀+c+(c*c:ℕ))/k)≤1/4 ∧
  31         4*(F₀+c)*Dₐ≤α*k ∧ 4*D*(t:ℝ)^2/k≤1/4 := by
  32   let H : ℝ := (c-1:ℕ)
  33   let h : ℝ := (c*c:ℕ)
  34   obtain ⟨N,hN⟩ := exists_nat_gt (max ((c+c*c:ℕ):ℝ) (max (16*H*(1+H)/ξ) (16*h*((c:ℝ)+h))))
  35   have hNc : c+c*c≤N := by
  36     have hh : ((c+c*c:ℕ):ℝ)<N := (le_max_left _ _).trans_lt hN
  37     exact_mod_cast hh.le
  38   refine ⟨N,hNc,?_⟩
  39   intro t ht F₀ D hF₀ hD
  40   have htR : (N:ℝ)≤t := by exact_mod_cast ht
  41   have htpos : (0:ℝ)<t := by exact_mod_cast (show 0<t by omega)
  42   have htv : 16*H*(1+H)≤ξ*t := by
  43     have hbound : 16*H*(1+H)/ξ≤(t:ℝ) :=
  44       (le_max_left _ _).trans ((le_max_right _ _).trans (hN.le.trans htR))
  45     nlinarith [(div_le_iff₀ hξ).mp hbound]
  46   have htr : 16*h*((c:ℝ)+h)≤t :=
  47     (le_max_right _ _).trans ((le_max_right _ _).trans (hN.le.trans htR))
  48   let f : Fin 9 → ℝ := ![2*t,2*F₀,2*H*(F₀+1+H),2*h*(F₀+c+h),
  49     16*H*(F₀+1+H)/ξ,16*h*(F₀+c+h),4*F₀*A/ξ,4*(F₀+c)*Dₐ/α,16*D*t^2]
  50   obtain ⟨k₀,hk₀,hf⟩ := finite_real_common_upper_bound f
  51   refine ⟨k₀,hk₀,?_⟩
  52   intro k hk
  53   have hb (i : Fin 9) : f i≤k := (hf i).trans hk
  54   have hkpos : 0<k := lt_of_lt_of_le (by linarith) hk
  55   have h0 : 2*(t:ℝ)≤k := by simpa only [f,Matrix.cons_val_zero] using hb 0
  56   have h1 : 2*F₀≤k := by simpa [f] using hb 1
  57   have h2 : 2*H*(F₀+1+H)≤k := by simpa [f] using hb 2
  58   have h3 : 2*h*(F₀+c+h)≤k := by simpa [f] using hb 3
  59   have h4 : 16*H*(F₀+1+H)≤ξ*k := by

  61     nlinarith [(div_le_iff₀ hξ).mp hh]
  62   have h5 : 16*h*(F₀+c+h)≤k := by simpa [f] using hb 5
  63   have h6 : 4*F₀*A≤ξ*k := by

  65     nlinarith [(div_le_iff₀ hξ).mp hh]
  66   have h7 : 4*(F₀+c)*Dₐ≤α*k := by
