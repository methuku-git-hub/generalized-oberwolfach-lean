  13 def alternatingIndex {k : ℕ} (i : Fin (k + 2)) (b : Fin 2) : Fin ((k + 2) * 2) :=
  14   finProdFinEquiv (i, b)

  16 theorem alternatingIndex_injective {k : ℕ} :
  17     Function.Injective (fun p : Fin (k + 2) × Fin 2 => alternatingIndex p.1 p.2) :=
  18   finProdFinEquiv.injective

  20 theorem alternatingIndex_next_zero {k : ℕ} (i : Fin (k + 2)) :
  21     cycleNext (alternatingIndex i 0) = alternatingIndex i 1 := by
  22   apply Fin.ext
  23   rw [cycleNext_val]
  24   have hi := i.isLt
  25   change (if 0 + 2 * i.val + 1 < (k + 2) * 2 then 0 + 2 * i.val + 1 else 0) = 1 + 2 * i.val
  26   split_ifs <;> omega
  27 
  28 theorem alternatingIndex_next_one {k : ℕ} (i : Fin (k + 2)) :
  29     cycleNext (alternatingIndex i 1) = alternatingIndex (finRotate (k + 2) i) 0 := by
  30   apply Fin.ext
  31   rw [cycleNext_val, finRotate_apply]
  32   have hi := i.isLt
  33   change (if 1 + 2 * i.val + 1 < (k + 2) * 2 then 1 + 2 * i.val + 1 else 0) =
  34     0 + 2 * ((i.val + 1) % (k + 2))
  35   by_cases h : i.val + 1 < k + 2
  36   · rw [Nat.mod_eq_of_lt h]
  37     split_ifs <;> omega
  38   · have he : i.val + 1 = k + 2 := by omega
  39     rw [he, Nat.mod_self]
  40     split_ifs <;> omega

  42 theorem cycleGraph_adj_next {r : ℕ} (hr : 2 ≤ r) (a b : Fin r) :
  43     (SimpleGraph.cycleGraph r).Adj a b ↔ cycleNext a = b ∨ cycleNext b = a := by
  44   rw [cycleGraph_modular_adj hr]
  45   simp only [Fin.ext_iff, cycleNext]

  47 theorem alternatingIndex_eq_iff {k : ℕ} (i j : Fin (k + 2)) (a b : Fin 2) :
  48     alternatingIndex i a = alternatingIndex j b ↔ i = j ∧ a = b := by
  49   change finProdFinEquiv (i, a) = finProdFinEquiv (j, b) ↔ _
  50   rw [Equiv.apply_eq_iff_eq, Prod.mk.injEq]

  52 theorem alternatingIndex_adj {k : ℕ} (i j : Fin (k + 2)) (a b : Fin 2) :
  53     (SimpleGraph.cycleGraph ((k + 2) * 2)).Adj
  54       (alternatingIndex i a) (alternatingIndex j b) ↔
  55     (a = 0 ∧ b = 1 ∧ (i = j ∨ i = finRotate (k + 2) j)) ∨
  56     (a = 1 ∧ b = 0 ∧ (j = i ∨ j = finRotate (k + 2) i)) := by
  57   rw [cycleGraph_adj_next (by omega)]
  58   fin_cases a <;> fin_cases b <;>
  59     simp [alternatingIndex_next_zero, alternatingIndex_next_one, alternatingIndex_eq_iff, eq_comm, or_comm]

  61 def alternatingVertex {k : ℕ} {X Y : Type*}
  62     (p : Fin (k + 2) → X × Y × X) (i : Fin (k + 2)) (b : Fin 2) :
  63     (Fin (k + 2) × X) ⊕ (Fin (k + 2) × Y) :=
  64   if b = 0 then Sum.inl (i, (p i).1) else Sum.inr (i, (p i).2.1)
  65 
  66 theorem alternatingVertex_injective {k : ℕ} {X Y : Type*}
  67     (p : Fin (k + 2) → X × Y × X) :
  68     Function.Injective (fun t : Fin (k + 2) × Fin 2 => alternatingVertex p t.1 t.2) := by
  69   rintro ⟨i, a⟩ ⟨j, b⟩ h
  70   fin_cases a <;> fin_cases b <;> simp [alternatingVertex] at h ⊢
  71   all_goals exact h.1

  73 def alternatingCycleEmbedding {k : ℕ} {X Y : Type*}
  74     (p : Fin (k + 2) → X × Y × X) :
  75     Fin ((k + 2) * 2) ↪ ((Fin (k + 2) × X) ⊕ (Fin (k + 2) × Y)) :=
  76   (finProdFinEquiv.symm.toEmbedding).trans
  77     ⟨fun t => alternatingVertex p t.1 t.2, alternatingVertex_injective p⟩

  79 theorem alternatingCycleEmbedding_apply {k : ℕ} {X Y : Type*}

  81     alternatingCycleEmbedding p (alternatingIndex i b) = alternatingVertex p i b := by
