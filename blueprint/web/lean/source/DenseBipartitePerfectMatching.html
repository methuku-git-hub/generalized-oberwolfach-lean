  12 theorem dense_bipartite_perfect_matching {X Y : Type*} [Fintype X] [Fintype Y]
  13     (E : X→Y→Prop) (hcard : Fintype.card X=Fintype.card Y)
  14     (hrow : ∀ x,Fintype.card Y≤2*(Finset.univ.filter (E x)).card)
  15     (hcol : ∀ y,Fintype.card X≤2*(Finset.univ.filter (fun x=>E x y)).card) :
  16     ∃ σ : X≃Y,∀ x,E x (σ x) := by
  17   classical
  18   have hhall (S : Finset X) : S.card≤(S.biUnion (fun x=>Finset.univ.filter (E x))).card := by
  19     let N := S.biUnion (fun x=>Finset.univ.filter (E x))
  20     rcases S.eq_empty_or_nonempty with rfl|hS
  21     · simp
  22     by_cases hsmall : 2*S.card≤Fintype.card Y
  23     · obtain ⟨x,hx⟩ := hS
  24       have hsub : Finset.univ.filter (E x)⊆N :=
  25         fun y hy=>Finset.mem_biUnion.mpr ⟨x,hx,hy⟩
  26       have hn := Finset.card_le_card hsub
  27       have hd := hrow x
  28       change S.card≤N.card
  29       omega
  30     · have he : N=Finset.univ := by
  31         apply Finset.eq_univ_of_forall
  32         intro y
  33         by_contra hy
  34         have hsub : Finset.univ.filter (fun x=>E x y)⊆Finset.univ\S := by
  35           intro x hx
  36           refine Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,?_⟩
  37           intro hxS
  38           exact hy (Finset.mem_biUnion.mpr ⟨x,hxS,Finset.mem_filter.mpr
  39             ⟨Finset.mem_univ _,(Finset.mem_filter.mp hx).2⟩⟩)
  40         have hh := Finset.card_le_card hsub
  41         rw [Finset.card_sdiff_of_subset (Finset.subset_univ _),Finset.card_univ] at hh
  42         have hd := hcol y
  43         have hSle := Finset.card_le_univ S
  44         omega
  45       change S.card≤N.card
  46       rw [he,Finset.card_univ,← hcard]
  47       exact Finset.card_le_univ S
  48   obtain ⟨f,hf,hE⟩ := (Finset.all_card_le_biUnion_card_iff_existsInjective'
  49     (fun x=>Finset.univ.filter (E x))).mp hhall
  50   exact ⟨Equiv.ofBijective f ((Fintype.bijective_iff_injective_and_card f).mpr ⟨hf,hcard⟩),
  51     fun x=>(Finset.mem_filter.mp (hE x)).2⟩
