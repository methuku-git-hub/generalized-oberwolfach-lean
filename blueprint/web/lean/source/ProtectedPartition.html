  16 /-- Correct all colour-class degrees while retaining every protected edge.
  17 The last four inequalities control the numbers of removed and added arcs
  18 separately, at both endpoints. The input cut inequalities are exactly those
  19 needed for the integral prescribed-degree theorem. -/
  20 theorem protectedBipartiteRegularPartition
  21     {V : Type*} [Fintype V] [DecidableEq V] (k q : ℕ) (m : Fin k → ℕ)
  22     (M L : Fin k → V → V → ℕ)
  23     (hq : ∀ a, q ≤ m a)
  24     (hL : ∀ a x y, L a x y ≤ M a x y)
  25     (hout : ∀ x, ∑ a, ∑ y, M a x y = ∑ a, m a)
  26     (hin : ∀ y, ∑ a, ∑ x, M a x y = ∑ a, m a)
  27     (hlowout : ∀ a x, m a - q ≤ ∑ y, M a x y)
  28     (hlowin : ∀ a y, m a - q ≤ ∑ x, M a x y)
  29     (hcut : ∀ a (S T : Finset V),
  30       (∑ x ∈ S, ((∑ y, M a x y) - (m a - q))) ≤
  31         (∑ y ∈ T, ((∑ x, M a x y) - (m a - q))) +
  32         ∑ x, ∑ y, if x ∈ S ∧ y ∉ T then L a x y else 0) :
  33     ∃ P : Fin k → V → V → ℕ,
  34       (∀ x y, ∑ a, P a x y = ∑ a, M a x y) ∧
  35       (∀ a x, ∑ y, P a x y = m a) ∧
  36       (∀ a y, ∑ x, P a x y = m a) ∧
  37       (∀ a x y, M a x y - L a x y ≤ P a x y) ∧
  38       (∀ a x, ∑ y, (M a x y - P a x y) ≤
  39         (∑ y, M a x y) - (m a - q)) ∧
  40       (∀ a y, ∑ x, (M a x y - P a x y) ≤
  41         (∑ x, M a x y) - (m a - q)) ∧
  42       (∀ a x, ∑ y, (P a x y - M a x y) ≤ q) ∧
  43       (∀ a y, ∑ x, (P a x y - M a x y) ≤ q) ∧
  44       (∀ x y, (∀ a, L a x y = 0) → ∀ a, P a x y = M a x y) := by
  45   classical
  46   let f a x := (∑ y, M a x y) - (m a - q)
  47   let g a y := (∑ x, M a x y) - (m a - q)
  48   have hfg (a : Fin k) : ∑ x, f a x = ∑ y, g a y := by
  49     dsimp [f, g]
  50     rw [Finset.sum_tsub_distrib _ (fun x _ => hlowout a x),
  51       Finset.sum_tsub_distrib _ (fun y _ => hlowin a y)]
  52     rw [Finset.sum_comm]
  53   have hex (a : Fin k) : ∃ T : V → V → ℕ,
  54       (∀ x y, T x y ≤ L a x y) ∧
  55       (∀ x, ∑ y, T x y = f a x) ∧
  56       (∀ y, ∑ x, T x y = g a y) :=
  57     bipartite_prescribedMultiplicity (L a) (f a) (g a) (hfg a) (hcut a)
  58   choose T hTcap hTout hTin using hex
  59   have hTM (a : Fin k) (x y : V) : T a x y ≤ M a x y :=
  60     (hTcap a x y).trans (hL a x y)
  61   let C a x y := M a x y - T a x y
  62   have hCout (a : Fin k) (x : V) : ∑ y, C a x y = m a - q := by
  63     dsimp [C]
  64     rw [Finset.sum_tsub_distrib _ (fun y _ => hTM a x y), hTout]
  65     dsimp [f]
  66     exact Nat.sub_sub_self (hlowout a x)
  67   have hCin (a : Fin k) (y : V) : ∑ x, C a x y = m a - q := by
  68     dsimp [C]
  69     rw [Finset.sum_tsub_distrib _ (fun x _ => hTM a x y), hTin]
  70     dsimp [g]
  71     exact Nat.sub_sub_self (hlowin a y)
  72   have hm : ∑ a, (m a - q) + k * q = ∑ a, m a := by
  73     calc
  74       _ = (∑ a, (m a - q)) + ∑ _ : Fin k, q := by simp
  75       _ = ∑ a, (m a - q + q) := (Finset.sum_add_distrib).symm
  76       _ = _ := by simp [Nat.sub_add_cancel (hq _)]
  77   have hfout (x : V) : ∑ a, f a x = k * q := by
  78     dsimp [f]
  79     rw [Finset.sum_tsub_distrib _ (fun a _ => hlowout a x), hout]
  80     omega
  81   have hgin (y : V) : ∑ a, g a y = k * q := by
  82     dsimp [g]
  83     rw [Finset.sum_tsub_distrib _ (fun a _ => hlowin a y), hin]
  84     omega
  85   let N x y := ∑ a, T a x y
  86   have hNout (x : V) : ∑ y, N x y = k * q := by
  87     dsimp [N]
  88     rw [Finset.sum_comm]
  89     simp only [hTout, hfout]
  90   have hNin (y : V) : ∑ x, N x y = k * q := by
  91     dsimp [N]
  92     rw [Finset.sum_comm]
  93     simp only [hTin, hgin]
  94   obtain ⟨Q, hQsum, hQout, hQin⟩ :=
  95     regularBipartite_groupedDecomposition k q N hNout hNin
  96   let P a x y := C a x y + Q a x y
  97   have hremove (a : Fin k) (x y : V) : M a x y - P a x y ≤ T a x y := by
  98     dsimp [P, C]
  99     omega
 100   have hadd (a : Fin k) (x y : V) : P a x y - M a x y ≤ Q a x y := by
 101     dsimp [P, C]
 102     omega
 103   refine ⟨P, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_⟩
 104   · intro x y
 105     simp only [P, Finset.sum_add_distrib, hQsum, N]
 106     rw [← Finset.sum_add_distrib]
 107     apply Finset.sum_congr rfl
 108     intro a _
 109     exact Nat.sub_add_cancel (hTM a x y)
 110   · intro a x
 111     simp only [P, Finset.sum_add_distrib, hCout, hQout]
 112     exact Nat.sub_add_cancel (hq a)
 113   · intro a y
 114     simp only [P, Finset.sum_add_distrib, hCin, hQin]
 115     exact Nat.sub_add_cancel (hq a)
 116   · intro a x y
 117     have ht := hTcap a x y
 118     dsimp [P, C]
 119     omega
 120   · intro a x
 121     calc
 122       _ ≤ ∑ y, T a x y := Finset.sum_le_sum (fun y _ => hremove a x y)
 123       _ = _ := hTout a x
 124   · intro a y
 125     calc
 126       _ ≤ ∑ x, T a x y := Finset.sum_le_sum (fun x _ => hremove a x y)
 127       _ = _ := hTin a y
 128   · intro a x
 129     calc
 130       _ ≤ ∑ y, Q a x y := Finset.sum_le_sum (fun y _ => hadd a x y)
 131       _ = _ := hQout a x
 132   · intro a y
 133     calc
 134       _ ≤ ∑ x, Q a x y := Finset.sum_le_sum (fun x _ => hadd a x y)
 135       _ = _ := hQin a y
 136   · intro x y hzero a
 137     have ht (b : Fin k) : T b x y = 0 := by
 138       have hb := hTcap b x y
 139       rw [hzero b] at hb
 140       omega
 141     have hqsum : ∑ b, Q b x y = 0 := by
 142       rw [hQsum]
 143       simp [N, ht]
 144     have hqa : Q a x y = 0 := by
 145       have hb := Finset.single_le_sum (fun b _ => Nat.zero_le (Q b x y))
 146         (Finset.mem_univ a)
 147       rw [hqsum] at hb
 148       omega
 149     simp [P, C, ht, hqa]

 151 /-- The deterministic part of the protected regular partition, expressed
 152 on the original simple graph. `Z` specifies pairs whose colour must remain
 153 unchanged. All capacity cuts are imposed after excluding protected edges.
 154 The discrepancy is the degree of the symmetric difference with the
 155 original colour class. -/
 156 theorem protectedRegularPartition
 157     {n k q : ℕ} {e : ℝ} (G : SimpleGraph (Fin n)) (m : Fin k → ℕ)
 158     (M L : Fin k → Fin n → Fin n → ℕ) (Z : Fin n → Fin n → Prop)
 159     (hq : ∀ a, q ≤ m a)
 160     (hL : ∀ a x y, L a x y ≤ M a x y)
 161     (hphysical : ∀ x y, (∑ a, M a x y) + (∑ a, M a y x) =
 162       if G.Adj x y then 1 else 0)
 163     (hout : ∀ x, ∑ a, ∑ y, M a x y = ∑ a, m a)
 164     (hin : ∀ y, ∑ a, ∑ x, M a x y = ∑ a, m a)
 165     (hlowout : ∀ a x, m a - q ≤ ∑ y, M a x y)
 166     (hlowin : ∀ a y, m a - q ≤ ∑ x, M a x y)
 167     (hupout : ∀ a x, ((∑ y, M a x y : ℕ) : ℝ) ≤ m a + e)
 168     (hupin : ∀ a y, ((∑ x, M a x y : ℕ) : ℝ) ≤ m a + e)
 169     (hcut : ∀ a (S T : Finset (Fin n)),
 170       (∑ x ∈ S, ((∑ y, M a x y) - (m a - q))) ≤
 171         (∑ y ∈ T, ((∑ x, M a x y) - (m a - q))) +
 172         ∑ x, ∑ y, if x ∈ S ∧ y ∉ T then L a x y else 0)
 173     (hZ : ∀ a x y, Z x y → L a x y = 0 ∧ L a y x = 0) :
 174     ∃ H : Fin k → SimpleGraph (Fin n),
 175       EdgePartition G H ∧ (∀ a, (H a).IsRegularOfDegree (2 * m a)) ∧
 176       (∀ a x y, Z x y → ((H a).Adj x y ↔ 0 < M a x y + M a y x)) ∧
 177       (∀ a x, ((Finset.univ.filter (fun y =>
 178         Xor ((H a).Adj x y) (0 < M a x y + M a y x))).card : ℝ) ≤ 4 * q + 2 * e) := by
 179   classical
 180   obtain ⟨P, hPsum, hPout, hPin, _, hremout, hremin, haddout, haddin, hfixed⟩ :=
 181     protectedBipartiteRegularPartition k q m M L hq hL hout hin hlowout hlowin hcut
 182   obtain ⟨H, hH, hHdeg, hHadj⟩ := regularPartition_of_orientedMultiplicities G P m
 183     (by simpa only [hPsum] using hphysical) hPout hPin
 184   refine ⟨H, hH, hHdeg, ?_, ?_⟩
 185   · intro a x y hz
 186     rw [hHadj, hfixed x y (fun b => (hZ b x y hz).1),
 187       hfixed y x (fun b => (hZ b x y hz).2)]
 188   · intro a x
 189     have hpoint (y : Fin n) :
 190         (if Xor ((H a).Adj x y) (0 < M a x y + M a y x) then 1 else 0 : ℕ) ≤
 191           (M a x y - P a x y) + (M a y x - P a y x) +
 192           (P a x y - M a x y) + (P a y x - M a y x) := by
 193       rw [hHadj]
 194       split_ifs with hh
 195       · rcases hh with ⟨hp, hm⟩ | ⟨hp, hm⟩ <;> omega
 196       · omega
 197     have hremovedout : ((∑ y, (M a x y - P a x y) : ℕ) : ℝ) ≤
 198         ((∑ y, M a x y : ℕ) : ℝ) - ((m a : ℝ) - q) := by
 199       have hh : ((∑ y, (M a x y - P a x y) : ℕ) : ℝ) ≤
 200           (((∑ y, M a x y) - (m a - q) : ℕ) : ℝ) := by
 201         exact_mod_cast hremout a x
 202       simpa only [Nat.cast_sub (hlowout a x), Nat.cast_sub (hq a)] using hh
 203     have hremovedin : ((∑ y, (M a y x - P a y x) : ℕ) : ℝ) ≤
 204         ((∑ y, M a y x : ℕ) : ℝ) - ((m a : ℝ) - q) := by
 205       have hh : ((∑ y, (M a y x - P a y x) : ℕ) : ℝ) ≤
 206           (((∑ y, M a y x) - (m a - q) : ℕ) : ℝ) := by
 207         exact_mod_cast hremin a x
 208       simpa only [Nat.cast_sub (hlowin a x), Nat.cast_sub (hq a)] using hh
 209     have haddedout : ((∑ y, (P a x y - M a x y) : ℕ) : ℝ) ≤ q := by
 210       exact_mod_cast haddout a x
 211     have haddedin : ((∑ y, (P a y x - M a y x) : ℕ) : ℝ) ≤ q := by
 212       exact_mod_cast haddin a x
 213     have hnat : (Finset.univ.filter (fun y =>
 214         Xor ((H a).Adj x y) (0 < M a x y + M a y x))).card ≤
 215         (∑ y, (M a x y - P a x y)) + (∑ y, (M a y x - P a y x)) +
 216         (∑ y, (P a x y - M a x y)) + (∑ y, (P a y x - M a y x)) := by
 217       rw [Finset.card_filter, ← Finset.sum_add_distrib, ← Finset.sum_add_distrib,
 218         ← Finset.sum_add_distrib]
 219       exact Finset.sum_le_sum (fun y _ => hpoint y)
 220     have hreal : ((Finset.univ.filter (fun y =>
 221         Xor ((H a).Adj x y) (0 < M a x y + M a y x))).card : ℝ) ≤
 222         ((∑ y, (M a x y - P a x y) : ℕ) : ℝ) + ((∑ y, (M a y x - P a y x) : ℕ) : ℝ) +
 223         ((∑ y, (P a x y - M a x y) : ℕ) : ℝ) + ((∑ y, (P a y x - M a y x) : ℕ) : ℝ) := by
 224       exact_mod_cast hnat
 225     have huo := hupout a x
 226     have hui := hupin a x
 227     linarith

 229 /-- Deterministic completion from the actual large-cut and near-degree
 230 estimates in the protected-partition proof. This discharges the auxiliary
 231 prescribed-degree cut inequalities, rather than assuming them. It does not
 232 assert the probabilistic event producing the initial orientation and cuts. -/
 233 theorem protectedRegularPartition_of_largeCuts
 234     {n k q : ℕ} {e : ℝ} (G : SimpleGraph (Fin n)) (m : Fin k → ℕ)
 235     (M L : Fin k → Fin n → Fin n → ℕ) (Z : Fin n → Fin n → Prop)
 236     (δ h : Fin k → ℝ)
 237     (hn : 0 < n) (he : 0 ≤ e) (heq : e < q)
 238     (hq : ∀ a, q ≤ m a)
 239     (hL : ∀ a x y, L a x y ≤ M a x y)
 240     (hphysical : ∀ x y, (∑ a, M a x y) + (∑ a, M a y x) =
 241       if G.Adj x y then 1 else 0)
 242     (hout : ∀ x, ∑ a, ∑ y, M a x y = ∑ a, m a)
 243     (hin : ∀ y, ∑ a, ∑ x, M a x y = ∑ a, m a)
 244     (hnearout : ∀ a x, (m a : ℝ) - e ≤ ((∑ y, M a x y : ℕ) : ℝ) ∧
 245       ((∑ y, M a x y : ℕ) : ℝ) ≤ m a + e)
 246     (hnearin : ∀ a y, (m a : ℝ) - e ≤ ((∑ x, M a x y : ℕ) : ℝ) ∧
 247       ((∑ x, M a x y : ℕ) : ℝ) ≤ m a + e)
 248     (hδ : ∀ a, 0 ≤ δ a ∧ δ a < 1 / 4)
 249     (hmargin : ∀ a, 2 * e / ((q : ℝ) - e) ≤ δ a)
 250     (hcapacity : ∀ a, (q : ℝ) + e ≤ h a * n / 2)
 251     (hlarge : ∀ a (S T : Finset (Fin n)),
 252       1 - 2 * δ a < (S.card : ℝ) / n + (T.card : ℝ) / n →
 253       h a * n * n * min ((S.card : ℝ) / n) ((T.card : ℝ) / n) ≤
 254         ((∑ x, ∑ y, if x ∈ S ∧ y ∈ T then L a x y else 0 : ℕ) : ℝ))
 255     (hZ : ∀ a x y, Z x y → L a x y = 0 ∧ L a y x = 0) :
 256     ∃ H : Fin k → SimpleGraph (Fin n),
 257       EdgePartition G H ∧ (∀ a, (H a).IsRegularOfDegree (2 * m a)) ∧
 258       (∀ a x y, Z x y → ((H a).Adj x y ↔ 0 < M a x y + M a y x)) ∧
 259       (∀ a x, ((Finset.univ.filter (fun y =>
 260         Xor ((H a).Adj x y) (0 < M a x y + M a y x))).card : ℝ) ≤ 4 * q + 2 * e) := by
 261   classical
 262   have hlowout (a : Fin k) (x : Fin n) : m a - q ≤ ∑ y, M a x y := by
 263     have hb := (hnearout a x).1
 264     have hc : ((m a - q : ℕ) : ℝ) ≤ ((∑ y, M a x y : ℕ) : ℝ) := by
 265       rw [Nat.cast_sub (hq a)]
 266       linarith
 267     exact_mod_cast hc
 268   have hlowin (a : Fin k) (y : Fin n) : m a - q ≤ ∑ x, M a x y := by
 269     have hb := (hnearin a y).1
 270     have hc : ((m a - q : ℕ) : ℝ) ≤ ((∑ x, M a x y : ℕ) : ℝ) := by
 271       rw [Nat.cast_sub (hq a)]
 272       linarith
 273     exact_mod_cast hc
 274   let f a x := (∑ y, M a x y) - (m a - q)
 275   let g a y := (∑ x, M a x y) - (m a - q)
 276   have hfg (a : Fin k) : ∑ x, f a x = ∑ y, g a y := by
 277     dsimp [f, g]
 278     rw [Finset.sum_tsub_distrib _ (fun x _ => hlowout a x),
 279       Finset.sum_tsub_distrib _ (fun y _ => hlowin a y), Finset.sum_comm]
 280   have hf (a : Fin k) (x : Fin n) :
 281       (q : ℝ) - e ≤ (f a x : ℝ) ∧ (f a x : ℝ) ≤ q + e := by
 282     dsimp [f]
 283     rw [Nat.cast_sub (hlowout a x), Nat.cast_sub (hq a)]
 284     have hb := hnearout a x
 285     constructor <;> linarith [hb.1, hb.2]
 286   have hg (a : Fin k) (y : Fin n) :
 287       (q : ℝ) - e ≤ (g a y : ℝ) ∧ (g a y : ℝ) ≤ q + e := by
 288     dsimp [g]
 289     rw [Nat.cast_sub (hlowin a y), Nat.cast_sub (hq a)]
 290     have hb := hnearin a y
 291     constructor <;> linarith [hb.1, hb.2]
 292   have hcut (a : Fin k) := intervalDemands_cut (L a) (f a) (g a)
 293     (by simpa using hn) he heq (hδ a).1 (hδ a).2 (hmargin a)
 294     (by simpa using hcapacity a) (hfg a) (hf a) (hg a)
 295     (by simpa using hlarge a)
 296   exact protectedRegularPartition G m M L Z hq hL hphysical hout hin hlowout hlowin
 297     (fun a x => (hnearout a x).2) (fun a y => (hnearin a y).2) hcut hZ
