   7 theorem long_reserved_manuscript_capacity
   8     {I:Type*} [Fintype I] (L n K requests:ℕ) (length:I→ℕ) (η:ℝ)
   9     (hL:0<L) (hη:0<η) (hη1:η≤1)
  10     (hK:2*(L:ℝ)/η≤K) (hlength:∀i,(K:ℝ)≤length i)
  11     (hsum:η*n≤∑i:I,(length i:ℝ))
  12     (hrequests:(requests:ℝ)≤η*n/(2*(L:ℝ))):
  13     requests≤∑i:I,length i/L := by
  14   have hK' : 2*(L:ℝ)≤η*(K:ℝ):=by simpa [mul_comm] using (div_le_iff₀ hη).mp hK
  15   have hKnonneg:0≤(K:ℝ):=by positivity
  16   have hlarge:∀i,L≤length i:=by
  17     intro i
  18     have hh:0≤(1-η)*(K:ℝ):=mul_nonneg (by linarith) hKnonneg
  19     have hreal:(L:ℝ)≤length i:=by have hi:=hlength i;nlinarith
  20     exact_mod_cast hreal
  21   have hden:0<2*(L:ℝ):=by exact_mod_cast (show 0<2*L by omega)
  22   have hreq:2*L*requests≤∑i:I,length i:=by
  23     have hh:((2:ℝ)*(L:ℝ))*(requests:ℝ)≤η*n:=by simpa [mul_comm] using (le_div_iff₀ hden).mp hrequests
  24     have hreal:((2*L*requests:ℕ):ℝ)≤∑i:I,(length i:ℝ):=by push_cast;nlinarith
  25     exact_mod_cast hreal
  26   exact long_reserved_total_block_capacity L hL length hlarge requests hreq
