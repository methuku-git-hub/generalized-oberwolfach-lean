  21 theorem exists_twoClass_root_order (σ : ∀ i,FixedSubset (V i) (twoClassProfile m i)) :
  22     ∃ x : Fin (m+1) → Σ i,V i,Function.Injective x ∧
  23       Finset.univ.image x=profileRootSet σ ∧
  24       (x 0).1=true ∧ ∀ j : Fin m,(x j.succ).1=false := by
  25   let eK : Fin 1 ≃ (σ true).val := Fintype.equivOfCardEq (by
  26     rw [Fintype.card_fin,Fintype.card_coe,(σ true).property];rfl)
  27   let eO : Fin m ≃ (σ false).val := Fintype.equivOfCardEq (by
  28     rw [Fintype.card_fin,Fintype.card_coe,(σ false).property];rfl)
  29   let x : Fin (m+1) → Σ i,V i := Fin.cons (α:=fun _ => Σ i,V i)
  30     ⟨true,(eK 0).val⟩ (fun j => ⟨false,(eO j).val⟩)
  31   have hxi : Function.Injective x := by
  32     intro i j
  33     refine Fin.cases ?_ (fun i => ?_) i
  34     · refine Fin.cases ?_ (fun j => ?_) j
  35       · intro _;rfl
  36       · intro he
  37         have hh := congrArg Sigma.fst he
  38         exact Bool.noConfusion hh
  39     · refine Fin.cases ?_ (fun j => ?_) j
  40       · intro he
  41         have hh := congrArg Sigma.fst he
  42         exact Bool.noConfusion hh
  43       · intro he
  44         have hv : (eO i).val=(eO j).val := (Sigma.mk.inj_iff.mp he).2 |> eq_of_heq
  45         exact congrArg Fin.succ (eO.injective (Subtype.ext hv))
  46   refine ⟨x,hxi,?_,rfl,fun _ => rfl⟩
  47   ext v
  48   constructor
  49   · intro hv

  51     refine Fin.cases ?_ (fun j => ?_) i
  52     · exact (profileRootSet_mem σ true _).mpr (eK 0).property
  53     · exact (profileRootSet_mem σ false _).mpr (eO j).property
  54   · intro hv

  56     cases i
  57     · let v' : (σ false).val := ⟨v,(profileRootSet_mem σ false v).mp hv⟩
