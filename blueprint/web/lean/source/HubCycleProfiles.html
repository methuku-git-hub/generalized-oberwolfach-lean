  16     (hK : (x 0).1=true) (hO : ∀ j : Fin m,(x j.succ).1=false) (i : Bool) :
  17     (Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈Finset.univ.image x)).card=
  18       twoClassProfile m i := by

  21   · have he : Finset.univ.filter (fun j => (x j).1=false)=Finset.univ.erase 0 := by
