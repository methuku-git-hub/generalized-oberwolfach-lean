  11 theorem residual_slice_card_of_single_incidence
  12     {E U:Type*} [DecidableEq E] [Fintype U]
  13     (H:Finset E) (D:U→Finset E) (P:U→Prop)
  14     (hsub:∀w,D w⊆H)
  15     (hdis:∀w z,w≠z→Disjoint (D w) (D z))
  16     (hH:H.card=Fintype.card U)
  17     (hD:∀w,(D w).card=if P w then 1 else 0):
  18     (H\Finset.univ.biUnion D).card=(Finset.univ.filter (fun w=>¬P w)).card:=by
  19   classical
  20   have hU:Finset.univ.biUnion D⊆H:=by
  21     intro e he
  22     obtain ⟨w,_,hw⟩:=Finset.mem_biUnion.mp he
  23     exact hsub w hw
  24   have hcard:(Finset.univ.biUnion D).card=(Finset.univ.filter P).card:=by
  25     rw [Finset.card_biUnion (by intro w _ z _ h;exact hdis w z h)]
  26     simp only [hD]
  27     simpa only [Nat.cast_id] using (Finset.sum_boole (R:=ℕ) P Finset.univ)
  28   have hp:=Finset.card_filter_add_card_filter_not (s:=(Finset.univ:Finset U)) P
  29   rw [Finset.card_sdiff_of_subset hU,hcard,hH]
  30   simp only [Finset.card_univ] at hp
