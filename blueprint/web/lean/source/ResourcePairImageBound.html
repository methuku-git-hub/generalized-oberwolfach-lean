  11 theorem resourceOmittedPairWeight_image_le {A B I : Type*}
  12     [Fintype A] [Fintype B] [Fintype I] [DecidableEq A] [DecidableEq B]
  13     (R : I → Finset A) (S : I → Finset B) (f : A → B)
  14     (hmem : ∀ i a,a∈R i → f a∈S i)
  15     (hinj : ∀ i a,a∈R i → ∀ b,b∈R i → f a=f b → a=b)
  16     (w : I → ℝ) (hw : ∀ i,0≤w i) (e g : A) :
  17     resourceOmittedPairWeight R w e g≤resourceOmittedPairWeight S w (f e) (f g) := by
  18   unfold resourceOmittedPairWeight
  19   apply Finset.sum_le_sum
  20   intro i _
  21   by_cases h : e∈R i ∧ g∈R i ∧ g≠e
  22   · rw [if_pos h,if_pos ⟨hmem i e h.1,hmem i g h.2.1,fun he=>h.2.2 (hinj i g h.2.1 e h.1 he)⟩]
  23   · rw [if_neg h]
  24     split_ifs <;> first | exact hw i | rfl

  26 end Oberwolfach
