  14 theorem finiteLaw_coordinate_localLemma_degree
  15     {E I : Type*} [Fintype E] [DecidableEq E] [Fintype I] [DecidableEq I]
  16     {State : E → Type*} [∀ e,Fintype (State e)]
  17     (μ : ∀ e,FiniteLaw (State e)) (A : I → (∀ e,State e) → Prop)
  18     (S : I → Finset E)
  19     (hs : ∀ i σ τ,(∀ e∈S i,σ e=τ e) → (A i σ ↔ A i τ))
  20     {r s : ℕ} (hr : 0<r) (hspos : 0<s)
  21     (hsize : ∀ i,(S i).card≤r)
  22     (hinc : ∀ e,(Finset.univ.filter (fun i=>e∈S i)).card≤s)
  23     (hp : ∀ i,(FiniteLaw.pi μ).probability (A i)≤1/(4*((r*s : ℕ) : ℝ))) :
  24     ∃ σ,0<(FiniteLaw.pi μ).mass σ ∧ ∀ i,¬A i σ := by
  25   classical
  26   let N : I → Finset I := fun i=>
  27     Finset.univ.filter (fun j=>j≠i ∧ ¬Disjoint (S i) (S j))
  28   have hdeg (i : I) : (N i).card≤r*s := by
  29     have hsub : N i⊆(S i).biUnion (fun e=>Finset.univ.filter (fun j=>e∈S j)) := by
  30       intro j hj
  31       have hdis := (Finset.mem_filter.mp hj).2.2
  32       obtain ⟨e,he,hj⟩ := Finset.not_disjoint_iff.mp hdis
  33       exact Finset.mem_biUnion.mpr ⟨e,he,Finset.mem_filter.mpr ⟨Finset.mem_univ _,hj⟩⟩
  34     calc
  35       (N i).card ≤ ((S i).biUnion (fun e=>Finset.univ.filter (fun j=>e∈S j))).card :=
  36         Finset.card_le_card hsub
  37       _ ≤ ∑ e∈S i,(Finset.univ.filter (fun j=>e∈S j)).card := Finset.card_biUnion_le
  38       _ ≤ ∑ _e∈S i,s := Finset.sum_le_sum (fun e _=>hinc e)
  39       _ = (S i).card*s := by simp
  40       _ ≤ r*s := Nat.mul_le_mul_right _ (hsize i)
  41   have hind : ∀ i T,i∉T → Disjoint T (N i) →
  42       (FiniteLaw.pi μ).probability (fun σ=>A i σ ∧ avoidsEvents A T σ)=
  43         (FiniteLaw.pi μ).probability (A i)*
  44           (FiniteLaw.pi μ).probability (avoidsEvents A T) := by
  45     intro i T hi hdis
  46     apply finiteLaw_supported_avoidance_independent μ A S hs
  47     intro j hj
  48     by_contra hn
  49     have hji : j≠i := fun he=>hi (he ▸ hj)
  50     have hjN : j∈N i := Finset.mem_filter.mpr ⟨Finset.mem_univ _,hji,hn⟩
  51     exact Finset.disjoint_left.mp hdis hj hjN
  52   have hpos := finiteLaw_symmetric_localLemma (FiniteLaw.pi μ) A N
  53     (Nat.mul_pos hr hspos) hdeg hind hp Finset.univ
  54   obtain ⟨σ,hσ,hA⟩ := finiteLaw_positive_event_exists (FiniteLaw.pi μ)
  55     (avoidsEvents A Finset.univ) hpos
  56   exact ⟨σ,hσ,by simpa [avoidsEvents] using hA⟩
