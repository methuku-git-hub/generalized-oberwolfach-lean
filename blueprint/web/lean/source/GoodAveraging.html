  13 theorem finiteLaw_expectation_indicator_value {Ω : Type*} [Fintype Ω]
  14     (μ : FiniteLaw Ω) (E : Ω → Prop) (k : ℝ) [DecidablePred E] :
  15     μ.expectation (fun ω => if E ω then k else 0) = k * μ.probability E := by
  16   unfold FiniteLaw.expectation FiniteLaw.probability
  17   rw [Finset.mul_sum]
  18   apply Finset.sum_congr rfl
  19   intro ω _
  20   by_cases h : E ω <;> simp [h, mul_comm]

  22 theorem averagedWeight_good_incidence_bounds {Ω T : Type*} [Fintype Ω] [Fintype T]
  23     (μ : FiniteLaw Ω) (w : Ω → T → ℝ) (I : T → Prop) (E Good : Ω → Prop)
  24     (ρ δ L U : ℝ) (hρ : 0 < ρ) (hL : 0 ≤ L)
  25     (hgood : 1 - δ ≤ μ.probability Good)
  26     (hload : ∀ ω, (if E ω ∧ Good ω then L else 0) ≤ (∑ t, if I t then w ω t else 0) ∧
  27       (∑ t, if I t then w ω t else 0) ≤ (if E ω then U else 0)) :
  28     L * (μ.probability E - δ) / ρ ≤ (∑ t, if I t then averagedWeight μ w ρ t else 0) ∧
  29       (∑ t, if I t then averagedWeight μ w ρ t else 0) ≤ U * μ.probability E / ρ := by
  30   rw [averagedWeight_incidence]
  31   have hlo := finiteLaw_expectation_mono μ _ _ (fun ω => (hload ω).1)
  32   have hhi := finiteLaw_expectation_mono μ _ _ (fun ω => (hload ω).2)
  33   rw [finiteLaw_expectation_indicator_value μ (fun ω => E ω ∧ Good ω) L] at hlo
  34   rw [finiteLaw_expectation_indicator_value μ E U] at hhi
  35   constructor
  36   · exact div_le_div_of_nonneg_right
  37       ((mul_le_mul_of_nonneg_left (finiteLaw_good_intersection μ E Good δ hgood) hL).trans hlo) hρ.le
  38   · exact div_le_div_of_nonneg_right hhi hρ.le

  40 theorem averagedWeight_good_edge_bounds {Ω T : Type*} [Fintype Ω] [Fintype T]
  41     (μ : FiniteLaw Ω) (w : Ω → T → ℝ) (I : T → Prop) (E Good : Ω → Prop)
  42     (ρ δ p ε : ℝ) (hρ : 0 < ρ) (hp : 0 ≤ p) (hp1 : p ≤ 1) (hε : 0 ≤ ε)
  43     (hprob : μ.probability E = ρ) (hgood : 1 - δ ≤ μ.probability Good)
  44     (hscale : δ ≤ ε * ρ)
  45     (hload : ∀ ω, (if E ω ∧ Good ω then 1 - p else 0) ≤ (∑ t, if I t then w ω t else 0) ∧
  46       (∑ t, if I t then w ω t else 0) ≤ (if E ω then 1 else 0)) :
  47     1 - p - ε ≤ (∑ t, if I t then averagedWeight μ w ρ t else 0) ∧
  48       (∑ t, if I t then averagedWeight μ w ρ t else 0) ≤ 1 := by
  49   obtain ⟨hl,hu⟩ := averagedWeight_good_incidence_bounds μ w I E Good ρ δ (1-p) 1 hρ
  50     (sub_nonneg.mpr hp1) hgood hload
  51   rw [hprob] at hl hu
  52   constructor
  53   · apply le_trans _ hl
  54     apply (le_div_iff₀ hρ).mpr
  55     have h := mul_le_mul_of_nonneg_left hscale (sub_nonneg.mpr hp1)
  56     nlinarith [mul_nonneg hp hε]
  57   · simpa only [one_mul, div_self hρ.ne'] using hu

  60     (μ : FiniteLaw Ω) (w : Ω → T → ℝ) (ρ : ℝ) (S : T → Prop)
