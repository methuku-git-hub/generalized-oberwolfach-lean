  15 theorem approximate_marked_cycle_coefficients {V E K : Type*}
  16     [Fintype V] [Nonempty V] [Fintype E] [Fintype K] [DecidableEq V]
  17     (tail head : E → V) (ktail khead : K → V)
  18     (hinj : Function.Injective (fun e => (tail e,head e)))
  19     (hkinj : Function.Injective (fun e => (ktail e,khead e)))
  20     (m : ℕ) (hm : 2≤m) (π : Equiv.Perm V) (b : ℕ)
  21     (hsep : ∀ e,¬PermutationClose π b (tail e) (head e))
  22     (hksep : ∀ e,¬PermutationClose π b (ktail e) (khead e))
  23     (f : Fin m → E → ℝ) (w : E → ℝ) (wK : K → ℝ) (hsum : ∀ e,(∑ j,f j e)=w e)
  24     (hf : ∀ j e,1/(2*(m:ℝ))≤f j e ∧ f j e≤2/(m:ℝ))
  25     (hwK : ∀ e,1/2≤wK e ∧ wK e≤3/2)
  26     (u : (Fin (m+1) → V) → ℝ) (hu : ∀ C,0≤u C)
  27     (he : ∀ i x y,(∑ C,if C i=x ∧ C (finRotate (m+1) i)=y then u C else 0)=
  28       copiedMarkedCapacity tail head ktail khead m f wK i x y)
  29     (B : ℝ) (hB : 0≤B) (hupper : ∀ C,u C≤B/(Fintype.card V:ℝ)^(m-1)) :
  30     ∃ q : MarkedCycleTuple tail head ktail khead m π b → ℝ,
  31       (∀ C,0≤q C ∧ q C≤B/(Fintype.card V:ℝ)^(m-1)) ∧
  32       ‖(∑ C,q C • markedCycleIncidence tail head ktail khead m C.val)-Sum.elim w wK‖≤
  33         (m:ℝ)*((m+1:ℝ)^2*(2*b+1)*B)/(Fintype.card V:ℝ) := by
  34   classical
  35   let a := copiedMarkedCapacity tail head ktail khead m f wK
  36   let q := separatedWeight π b u
  37   have hq (C : Fin (m+1) → V) : 0≤q C ∧ q C≤u C := separatedWeight_bounds π b u hu C
  38   have ha := (copiedMarkedCapacity_bounds tail head ktail khead hinj hkinj m (by omega) f wK hf hwK).1
  39   have hmr : (0:ℝ)<m := Nat.cast_pos.mpr (by omega)
  40   have hs (C : Fin (m+1) → V) (hC : 0<q C) : IsMarkedCycleTuple tail head ktail khead m π b C := by
  41     obtain ⟨hsepC,_,hpos⟩ := separated_capacity_support π b a ha u hu
  42       (by
  43         intro i x y
  44         convert he i x y using 1
  45         apply Finset.sum_congr rfl
  46         intro C _
  47         split_ifs <;> rfl) C hC
  48     obtain ⟨hO,hK⟩ := (copied_marked_support_iff tail head ktail khead hinj hkinj m f wK
  49       (fun j e => (by positivity : (0:ℝ)<1/(2*m)).trans_le (hf j e).1)
  50       (fun e => by linarith [(hwK e).1]) C).mp hpos
  51     exact ⟨hsepC,hO,hK⟩
  52   refine ⟨fun C => q C.val,fun C => ⟨(hq C.val).1,(hq C.val).2.trans (hupper C.val)⟩,?_⟩
  53   rw [marked_cycle_weight_restriction tail head ktail khead m π b q (fun C => (hq C).1) hs]
  54   have hl := separated_marked_edge_loads tail head ktail khead hinj hkinj m hm π b hsep hksep
  55     f w wK hsum u hu he B hB hupper
  56   have hnonneg : 0≤((m+1:ℝ)^2*(2*b+1)*B)/(Fintype.card V:ℝ) := by positivity
  57   apply (pi_norm_le_iff_of_nonneg (show 0≤(m:ℝ)*((m+1:ℝ)^2*(2*b+1)*B)/(Fintype.card V:ℝ) by positivity)).mpr
  58   intro e
  59   simp only [Pi.sub_apply,Finset.sum_apply,Pi.smul_apply,smul_eq_mul,Real.norm_eq_abs]
  60   cases e with
  61   | inl e => simpa only [q,mul_div_assoc,Sum.elim_inl] using hl.1 e
  62   | inr e =>
  63     have hh := hl.2 e
  64     have hmr1 : (1:ℝ)≤m := by exact_mod_cast (by omega : 1≤m)
  65     have he : ((m+1:ℝ)^2*(2*b+1)*B)/(Fintype.card V:ℝ)≤
  66         (m:ℝ)*((m+1:ℝ)^2*(2*b+1)*B)/(Fintype.card V:ℝ) := by
  67       simpa only [one_mul,mul_div_assoc] using mul_le_mul_of_nonneg_right hmr1 hnonneg
  68     exact hh.trans he
