   8 theorem ordinary_frequency_marginal_bounds
   9     (s θ c long short omitted:ℝ)
  10     (hs0:0≤s) (hs1:s≤1/100) (hθ0:0≤θ) (hθ1:θ≤1/100) (hc:8≤c)
  11     (hlong:0≤long) (hshort:0≤short) (homit:0≤omitted) (homitmax:omitted≤θ/50)
  12     (hfreq:long+short+omitted=1-s):
  13     let a:=(1-θ)*long/c
  14     let b:=(c-1)*a+(1-θ)*short
  15     let u:=1-s-a-b;
  16     u=θ*(1-s)+(1-θ)*omitted ∧ θ/2≤u ∧ u≤2*θ ∧
  17     0≤a ∧ a≤1/c ∧ 1/2≤b ∧ s+a+b+u=1 := by
  18   intro a b u
  19   have hc0:0<c:=by linarith
  20   have hθc:0≤1-θ:=by linarith
  21   have hlong1:long≤1:=by linarith
  22   have hlongprod:0≤θ*long:=mul_nonneg hθ0 hlong
  23   have homitprod:0≤θ*omitted:=mul_nonneg hθ0 homit
  24   have hsprod:0≤θ*s:=mul_nonneg hθ0 hs0
  25   have hθsbound:0≤θ*(1/100-s):=mul_nonneg hθ0 (by linarith)
  26   have ha:0≤a:=by dsimp [a];positivity
  27   have ha1:a≤1/c:=by
  28     dsimp [a]
  29     apply (div_le_div_iff_of_pos_right hc0).2
  30     nlinarith
  31   have hca:c*a=(1-θ)*long:=by dsimp [a];field_simp
  32   have hu:u=θ*(1-s)+(1-θ)*omitted:=by dsimp [u,b];nlinarith
  33   have huLo:θ/2≤u:=by rw [hu];nlinarith
  34   have huHi:u≤2*θ:=by rw [hu];nlinarith
  35   have ha8:a≤1/8:=by
  36     have hh:1/c≤(1/8:ℝ):=by apply (div_le_div_iff₀ hc0 (by norm_num)).2;nlinarith
  37     exact ha1.trans hh
  38   have hb:1/2≤b:=by dsimp [u] at huHi;nlinarith
  39   refine ⟨hu,huLo,huHi,ha,ha1,hb,?_⟩
  40   dsimp [u];ring
  41 end Oberwolfach
