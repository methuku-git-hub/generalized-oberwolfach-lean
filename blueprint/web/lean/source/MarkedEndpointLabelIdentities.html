  49 theorem marked_endpoint_labels_parent_error
  50     (P : SimpleGraph V) (π : Equiv.Perm V) (d : ℕ) (A : Finset V)
  51     (hInv : ∀ z,π z∈A ↔ z∈A) (mark incoming : Bool) (x : V)
  52     (σ : RawMarkedOutcome V U Γ Color) (p err D delta : ℝ) (hp : 0≤p) (hp1 : p≤1)
  53     (hcount : |((markedEndpointLabels P π d A (mark,incoming,x) σ).card:ℝ)-
  54       p*(markedRowDomain P π (3*d) A {(mark,incoming,x)} ∅).card|≤err)
  55     (hparent : |((P.neighborFinset (if mark && incoming then π x else x)∩A).card:ℝ)-D|≤delta) :
  56     |((markedEndpointLabels P π d A (mark,incoming,x) σ).card:ℝ)-p*D|≤
  57       err+(6*(d:ℝ)+1)+p*delta := by
  58   rw [marked_endpoint_labels_card,marked_single_arc_domain] at hcount
  59   have he := markedRestrictedDegree_parent_error P π (3*d) (fun e=>σ (Sum.inl e))
  60     mark incoming x A hInv p err hp hp1 hcount
  61   rw [marked_endpoint_labels_card]
  62   have hdiff : |p*((P.neighborFinset (if mark && incoming then π x else x)∩A).card:ℝ)-p*D|≤p*delta := by
  63     rw [←mul_sub,abs_mul,abs_of_nonneg hp]
  64     exact mul_le_mul_of_nonneg_left hparent hp
  65   have hc := (abs_sub_le (markedRestrictedDegree P π (3*d) (fun e=>σ (Sum.inl e)) mark incoming x A:ℝ)
  66     (p*((P.neighborFinset (if mark && incoming then π x else x)∩A).card:ℝ)) (p*D)).trans (add_le_add he hdiff)
  67   convert hc using 1 <;> push_cast <;> ring
