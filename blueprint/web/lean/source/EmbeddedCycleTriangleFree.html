  13  theorem cycleGraph_no_triangle (r:ℕ) (hr:4≤r) (i j k:Fin r)
  14     (hij:(SimpleGraph.cycleGraph r).Adj i j)
  15     (hjk:(SimpleGraph.cycleGraph r).Adj j k)
  16     (hki:(SimpleGraph.cycleGraph r).Adj k i):False:=by

  24  theorem IsEmbeddedCycle.no_triangle {V:Type} {C:SimpleGraph V} {r:ℕ}
  25     (hC:IsEmbeddedCycle C r) (hr:4≤r) (x y z:V)
  26     (hxy:C.Adj x y) (hyz:C.Adj y z):¬C.Adj z x:=by
  27   intro hzx
  28   obtain ⟨_,f,rfl⟩:=hC
  29   obtain ⟨i,j,hij,rfl,rfl⟩:=(SimpleGraph.map_adj f _ _ _).mp hxy
  30   obtain ⟨k,l,hkl,hkj,rfl⟩:=(SimpleGraph.map_adj f _ _ _).mp hyz
  31   have hkj':k=j:=f.injective hkj
  32   subst k
  33   have hli:(SimpleGraph.cycleGraph r).Adj l i:=SimpleGraph.map_adj_apply.mp hzx
  34   exact cycleGraph_no_triangle r hr i j l hij hkl hli
