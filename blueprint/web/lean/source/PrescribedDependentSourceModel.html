  17 theorem prescribed_lifted_source_vertex_membership (pieces:∀ i,List (PrescribedSourcePiece (X i)))
  18     (i:I) (x:X i):
  19     Sigma.mk i x∈prescribedSourceVertices (prescribedLiftDependentPieces Finset.univ.toList pieces) i ↔
  20       x∈prescribedDependentVertices pieces i := by
  21   rw [prescribed_source_model_vertices]
  22   simp only [prescribedDependentVertices,Finset.mem_filter,Finset.mem_univ,true_and]
  23   constructor
  24   · rintro ⟨p,hp,hpi,hx⟩
  25     obtain ⟨j,hj,hp⟩:=List.mem_flatMap.mp hp
  26     obtain ⟨q,hq,rfl⟩:=List.mem_map.mp hp
  27     simp only at hpi
  28     subst j
  29     rw [prescribed_piece_map_vertices] at hx
  30     obtain ⟨z,hz,hzx⟩:=List.mem_map.mp hx
  31     have hzx':z=x:=sigma_mk_injective hzx
  32     subst z
  33     exact ⟨q,hq,hz⟩
  34   · rintro ⟨q,hq,hx⟩
  35     refine ⟨(i,q.map (Sigma.mk i)),?_,rfl,?_⟩
  36     · apply List.mem_flatMap.mpr
  37       exact ⟨i,by simp,List.mem_map.mpr ⟨q,hq,rfl⟩⟩
  38     · rw [prescribed_piece_map_vertices]
  39       exact List.mem_map.mpr ⟨x,hx,rfl⟩

  41 theorem prescribed_lifted_source_arc_membership (pieces:∀ i,List (PrescribedSourcePiece (X i)))
  42     (i:I) (x y:X i):
  43     prescribedSourceDigraph (prescribedLiftDependentPieces Finset.univ.toList pieces) i (Sigma.mk i x) (Sigma.mk i y) ↔
  44       prescribedDependentDigraph pieces i x y := by
  45   constructor
  46   · rintro ⟨p,hp,hpi,he⟩
  47     obtain ⟨j,hj,hp⟩:=List.mem_flatMap.mp hp
  48     obtain ⟨q,hq,rfl⟩:=List.mem_map.mp hp
  49     simp only at hpi
  50     subst j
  51     rw [prescribed_piece_map_arcs] at he
  52     obtain ⟨e,he,heq⟩:=Finset.mem_image.mp he
  53     rcases e with ⟨u,v⟩
  54     have hu:u=x:=sigma_mk_injective (Prod.mk.inj heq).1
  55     have hv:v=y:=sigma_mk_injective (Prod.mk.inj heq).2
  56     subst u;subst v
  57     exact ⟨q,hq,he⟩
  58   · rintro ⟨q,hq,he⟩
  59     refine ⟨(i,q.map (Sigma.mk i)),?_,rfl,?_⟩
  60     · apply List.mem_flatMap.mpr
  61       exact ⟨i,by simp,List.mem_map.mpr ⟨q,hq,rfl⟩⟩
  62     · rw [prescribed_piece_map_arcs]
  63       exact Finset.mem_image.mpr ⟨(x,y),he,rfl⟩

  65     (i:I) (z:Σ i,X i)

  77 theorem prescribed_lifted_source_vertices (pieces:∀ i,List (PrescribedSourcePiece (X i)))
  78     (i:I):
  79     prescribedSourceVertices (prescribedLiftDependentPieces Finset.univ.toList pieces) i =
  80       (prescribedDependentVertices pieces i).image (Sigma.mk i) := by
  81   ext z
  82   rcases z with ⟨j,x⟩
  83   by_cases hji:j=i
  84   · subst j
  85     rw [prescribed_lifted_source_vertex_membership]
  86     constructor
  87     · intro hx;exact Finset.mem_image.mpr ⟨x,hx,rfl⟩
  88     · intro hx
  89       obtain ⟨y,hy,he⟩:=Finset.mem_image.mp hx
  90       have h:y=x:=sigma_mk_injective he
  91       simpa [h] using hy
  92   · constructor
  93     · intro hz;exact (hji (prescribed_lifted_source_vertex_fiber pieces i ⟨j,x⟩ hz)).elim
  94     · intro hz
  95       obtain ⟨y,hy,he⟩:=Finset.mem_image.mp hz
  96       have hh:i=j:=congrArg Sigma.fst he
  97       exact (hji hh.symm).elim
