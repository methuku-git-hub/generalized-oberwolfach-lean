  16 theorem ordinaryFiveUnion_card {α:Type} [DecidableEq α] (L R:Finset α) (X:Fin 3→Finset α)
  17     (M:ℕ) (hL:L.card≤M) (hR:R.card≤M) (hX:∀ i,(X i).card≤M) :
  18     (ordinaryFiveUnion L R X).card≤5*M := by
  19   have hU:(Finset.univ.biUnion X).card≤3*M:=by
  20     calc
  21       _≤∑ i:Fin 3,(X i).card:=Finset.card_biUnion_le
  22       _≤∑ _:Fin 3,M:=Finset.sum_le_sum (fun i _=>hX i)
  23       _=3*M:=by simp
  24   exact (Finset.card_union_le _ _).trans (by
  25     have hLR:=Finset.card_union_le L R
  26     omega)
