  19 theorem canonical_marked_wheel_distinct_rim_pair (hm : 3≤m)
  20     (hn : 0<Fintype.card V) (hu : Fintype.card U≤Fintype.card V)
  21     (w : MarkedWheel tail head ktail khead m π b H₀ Hₖ → ℝ) (B : ℝ) (hB : 0≤B)
  22     (hw : ∀ W,w W≤B/(Fintype.card V:ℝ)^m) (e f : E⊕K) (hef : e≠f) :
  23     (∑ W,if CanonicalMarkedArcMem W.1 e ∧
  24       CanonicalMarkedArcMem W.1 f then w W else 0)≤
  25       ((m:ℝ)+1)^4*B/Fintype.card V := by
  26   let P (W : MarkedWheel tail head ktail khead m π b H₀ Hₖ) :=
  27     CanonicalMarkedArcMem W.1 e ∧
  28     CanonicalMarkedArcMem W.1 f
  29   change (∑ W,if P W then w W else 0)≤_
  30   by_cases hex : ∃ W,P W
  31   · obtain ⟨W,hW⟩ := hex
  32     let S : Finset V := {Sum.elim tail ktail e,Sum.elim head khead e,Sum.elim tail ktail f,Sum.elim head khead f}
  33     have hS : 3≤S.card := canonicalMarkedArc_pair_root_card (by omega) W.1 e f hef hW.1 hW.2
  34     have hS4 : S.card≤4 := by
  35       dsimp [S]
  36       have h1 := Finset.card_insert_le (Sum.elim tail ktail e)
  37         ({Sum.elim head khead e,Sum.elim tail ktail f,Sum.elim head khead f}:Finset V)
  38       have h2 := Finset.card_insert_le (Sum.elim head khead e)
  39         ({Sum.elim tail ktail f,Sum.elim head khead f}:Finset V)
  40       have h3 := Finset.card_insert_le (Sum.elim tail ktail f) ({Sum.elim head khead f}:Finset V)
  41       simp only [Finset.card_singleton] at h3
  42       omega
  43     have hbound := marked_wheel_free_hub_event hn hu w B hB hw S (by omega) (by omega) P
  44       (by
  45         intro Z hZ v hv
  46         have he := canonicalMarkedArc_endpoints_present Z.1 e hZ.1
  47         have hf := canonicalMarkedArc_endpoints_present Z.1 f hZ.2
  48         simp only [S,Finset.mem_insert,Finset.mem_singleton] at hv
  49         rcases hv with rfl|rfl|rfl|rfl
  50         · exact he.1
  51         · exact he.2
  52         · exact hf.1
  53         · exact hf.2)
  54     have ht := hbound.trans (uniform_wheel_pair_scale _ _ B
  55       (by exact_mod_cast (show 1≤Fintype.card V by omega))
  56       (by linarith [show (0:ℝ)≤m from Nat.cast_nonneg m]) hB S.card (S.card-2) hS4 (by omega))
  57     convert ht using 1
  58     apply Finset.sum_congr rfl
  59     intro Z _
  60     by_cases h : P Z <;> simp only [h,ite_true,ite_false]
  61   · have hfalse : ∀ W,¬P W := by simpa only [not_exists] using hex
  62     simp only [hfalse,ite_false,Finset.sum_const_zero]
  63     positivity
