  13 theorem marked_deep_host_enumerate_roots
  14     (P:SimpleGraph V) (π:Equiv.Perm V) (d m:ℕ)
  15     (σ:RawMarkedOutcome V U Γ Color) (H:Bool→V→U→Prop)
  16     (I T:Finset V) (J W:Finset U) (hTI:T⊆I) (hWJ:W⊆J)
  17     (hdis:∀ x u,¬(H false x u∧H true x u))
  18     (hshift:∀ u x z,H false x u→H true z u→x≠π z)
  19     (hb:MarkedDeepHostBalanced P π d m σ H I T J W):
  20     let E:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) false
  21     let K:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) true
  22     let src:=Sum.elim (fun e:E=>e.val.1) (fun e:K=>e.val.1)
  23     let dst:=Sum.elim (fun e:E=>e.val.2) (fun e:K=>e.val.2)
  24     let F:=Iᶜ;
  25     ∃ (u:Fin W.card→U) (v:Fin T.card→V),
  26       Function.Injective u∧Function.Injective v∧Finset.univ.image u=W∧Finset.univ.image v=T∧
  27       (∀ i x,¬(H false x (u i)∧H true x (u i)))∧
  28       (∀ i x z,H false x (u i)→H true z (u i)→x≠π z)∧
  29       (∀ i,Fintype.card (hubNeighbourClass (H false) (H true) (u i) false)=
  30         m*Fintype.card (hubNeighbourClass (H false) (H true) (u i) true))∧
  31       (∀ i x,H false x (u i)∨H true x (u i)→x∈F)∧
  32       (∀ i,v i∉F)∧(∀ i z,¬(H false (v i) z∧H true (v i) z))∧
  33       (∀ i,∀ a:E⊕K,dst a=v i→src a∈F)∧
  34       (∀ i,∀ a:E⊕K,src a=v i→dst a∈F)∧
  35       (∀ i,(Finset.univ.filter (fun a:E⊕K=>dst a=v i)).card=
  36         (Finset.univ.filter (fun a:E⊕K=>src a=v i)).card)∧
  37       (∀ i,(Finset.univ.filter (fun a:E⊕K=>dst a=v i)).card=
  38         (Finset.univ.filter (fun z:V×U=>(H false z.1 z.2∨H true z.1 z.2)∧z.1=v i)).card)∧
  39       (∀ i,(Finset.univ.filter (fun a:E⊕K=>dst a=v i∧markedArcType a=true)).card=
  40         (Finset.univ.filter (fun z:V×U=>H true z.1 z.2∧z.1=v i)).card)∧
  41       (∀ i,Disjoint ((Finset.univ.filter (H false (v i)))∪(Finset.univ.filter (H true (v i))))
  42         (Finset.univ.image u)):=by
  43   intro E K src dst F
  44   obtain ⟨hbal,hratio,hinc,hout,hrowSupport,hhubSupport⟩:=hb
  45   let eu:Fin W.card≃W:=(Fintype.equivFinOfCardEq (Fintype.card_coe W)).symm
  46   let ev:Fin T.card≃T:=(Fintype.equivFinOfCardEq (Fintype.card_coe T)).symm
  47   let u:Fin W.card→U:=fun i=>(eu i).val
  48   let v:Fin T.card→V:=fun i=>(ev i).val
  49   have hu:Function.Injective u:=fun i j h=>eu.injective (Subtype.ext h)
  50   have hv:Function.Injective v:=fun i j h=>ev.injective (Subtype.ext h)
  51   have huW (i):u i∈W:=(eu i).property
  52   have hvT (i):v i∈T:=(ev i).property
  53   have huimage:Finset.univ.image u=W:=by
  54     ext z
  55     simp only [Finset.mem_image,Finset.mem_univ,true_and]
  56     constructor
  57     · rintro ⟨i,rfl⟩;exact huW i
  58     · intro hz
  59       obtain ⟨i,hi⟩:=eu.surjective ⟨z,hz⟩
  60       exact ⟨i,by simp only [u,hi]⟩
  61   have hvimage:Finset.univ.image v=T:=by
  62     ext x
  63     simp only [Finset.mem_image,Finset.mem_univ,true_and]
  64     constructor
  65     · rintro ⟨i,rfl⟩;exact hvT i
  66     · intro hx
  67       obtain ⟨i,hi⟩:=ev.surjective ⟨x,hx⟩
  68       exact ⟨i,by simp only [v,hi]⟩
  69   refine ⟨u,v,hu,hv,huimage,hvimage,(fun i x=>hdis x (u i)),
  70     (fun i x z=>hshift (u i) x z),(fun i=>hratio (u i) (huW i)),
  71     (fun i x hx=>Finset.mem_compl.mpr (hhubSupport (u i) (huW i) x hx)),
  72     (fun i=>by change v i∉Iᶜ; simpa only [Finset.mem_compl,not_not] using hTI (hvT i)),
  73     (fun i z=>hdis (v i) z),
  74     (fun i a ha=>by
  75       apply Finset.mem_compl.mpr
  76       apply hinc a
  77       change dst a∈T
  78       rw [ha]
  79       exact hvT i),
  80     (fun i a ha=>by
  81       apply Finset.mem_compl.mpr
  82       apply hout a
  83       change src a∈T
  84       rw [ha]
  85       exact hvT i),
  86     (fun i=>(hbal (v i) (hvT i)).1), (fun i=>(hbal (v i) (hvT i)).2.1),
  87     (fun i=>(hbal (v i) (hvT i)).2.2),?_⟩
  88   intro i
  89   rw [huimage]
  90   apply Finset.disjoint_left.mpr
  91   intro z hz hzW
  92   have hs:H false (v i) z∨H true (v i) z:=by
  93     simpa only [Finset.mem_union,Finset.mem_filter,Finset.mem_univ,true_and] using hz
  94   exact hrowSupport (v i) (hvT i) z hs (hWJ hzW)
