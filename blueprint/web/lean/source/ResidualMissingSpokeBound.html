  11 theorem residual_missing_positions_le_uncovered_spokes
  12     {I V:Type} [DecidableEq I] [Fintype V] [DecidableEq V]
  13     (s₂:Finset I) (d:ℕ) (T:I→Finset ℕ) (hT:∀i∈s₂,T i⊆Finset.range d)
  14     (f:I→ℕ→V) (Available Covered:Finset V)
  15     (hvalid:∀i∈s₂,∀j<d,f i j∈Available)
  16     (hinj:∀i∈s₂,∀j<d,∀i'∈s₂,∀j'<d,f i j=f i' j'→i=i' ∧ j=j')
  17     (hcovered:∀i∈s₂,∀j<d,f i j∈Covered→j∈T i):
  18     (∑i∈s₂,(d-(T i).card))≤(Available\Covered).card:=by
  19   let Missing:=s₂.sigma (fun i=>(Finset.range d)\T i)
  20   let g:(Σ _ : I, ℕ)→V:=fun p=>f p.1 p.2
  21   have hcard:Missing.card=∑i∈s₂,(d-(T i).card):=by
  22     rw [show Missing=s₂.sigma (fun i=>(Finset.range d)\T i) from rfl,Finset.card_sigma]
  23     apply Finset.sum_congr rfl
  24     intro i hi
  25     rw [Finset.card_sdiff_of_subset (hT i hi),Finset.card_range]
  26   have hmaps:Missing.image g⊆Available\Covered:=by
  27     intro x hx
  28     obtain ⟨⟨i,j⟩,hij,rfl⟩:=Finset.mem_image.mp hx
  29     have hi:i∈s₂:=(Finset.mem_sigma.mp hij).1
  30     have hj:=Finset.mem_sdiff.mp (Finset.mem_sigma.mp hij).2
  31     have hjd:j<d:=Finset.mem_range.mp hj.1
  32     refine Finset.mem_sdiff.mpr ⟨hvalid i hi j hjd,?_⟩
  33     intro hc
  34     exact hj.2 (hcovered i hi j hjd hc)
  35   have hinj':Set.InjOn g ↑Missing:=by
  36     intro a ha b hb hab
  37     rcases a with ⟨i,j⟩;rcases b with ⟨i',j'⟩
  38     have hi: i∈s₂:=(Finset.mem_sigma.mp ha).1
  39     have hi': i'∈s₂:=(Finset.mem_sigma.mp hb).1
  40     have hj:j<d:=Finset.mem_range.mp (Finset.mem_sdiff.mp (Finset.mem_sigma.mp ha).2).1
  41     have hj':j'<d:=Finset.mem_range.mp (Finset.mem_sdiff.mp (Finset.mem_sigma.mp hb).2).1
  42     obtain ⟨rfl,rfl⟩:=hinj i hi j hj i' hi' j' hj' hab
  43     rfl
  44   have hh:=Finset.card_le_card hmaps
  45   rw [Finset.card_image_iff.mpr hinj',hcard] at hh
  46   exact hh
