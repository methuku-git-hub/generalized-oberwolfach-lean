  13 theorem related_fixed_selections_card {I X : Type*}
  14     [Fintype I] [Fintype X] [DecidableEq I] [Nonempty X]
  15     (R : X → X → Prop) (q : ℕ) (hR : ∀ x, Fintype.card {y // R x y} ≤ q)
  16     (A : Finset I) (a : I → X) (i j : I) (hi : i ∉ A) (hij : i ≠ j) :
  17     Fintype.card {f : I → X // (∀ k ∈ A, f k = a k) ∧ R (f j) (f i)} ≤
  18       q * Fintype.card X ^ (Fintype.card I - A.card - 1) := by
  19   let x₀ : X := Classical.choice (inferInstance : Nonempty X)
  20   let G := {g : I → X // ∀ k ∈ insert i A, g k = Function.update a i x₀ k}
  21   let B := {f : I → X // (∀ k ∈ A, f k = a k) ∧ R (f j) (f i)}
  22   let g (f : B) : G := ⟨Function.update f.1 i x₀, by
  23     intro k hk
  24     rcases Finset.mem_insert.mp hk with h | h
  25     · subst k
  26       simp
  27     · have hki : k ≠ i := by intro he; exact hi (he ▸ h)
  28       simp only [Function.update_of_ne hki]
  29       exact f.2.1 k h⟩
  30   let F (f : B) : Σ g : G, {x : X // R (g.1 j) x} :=
  31     ⟨g f, f.1 i, by simpa only [g, Function.update_of_ne hij.symm] using f.2.2⟩
  32   have hF : Function.Injective F := by
  33     intro f f' he
  34     have hg : g f = g f' := congrArg Sigma.fst he
  35     have hv : f.1 i = f'.1 i := congrArg (fun z : (Σ g : G, {x : X // R (g.1 j) x}) => z.2.1) he
  36     apply Subtype.ext
  37     funext k
  38     by_cases hk : k = i
  39     · simpa only [hk] using hv
  40     · have hh := congrArg (fun z : G => z.1 k) hg
  41       simpa only [g, Function.update_of_ne hk] using hh
  42   calc
  43     Fintype.card B ≤ Fintype.card (Σ g : G, {x : X // R (g.1 j) x}) :=
  44       Fintype.card_le_of_injective F hF
  45     _ = ∑ g : G, Fintype.card {x : X // R (g.1 j) x} := Fintype.card_sigma
  46     _ ≤ ∑ _g : G, q := Finset.sum_le_sum (fun g _ => hR (g.1 j))
  47     _ = Fintype.card G * q := by simp
  48     _ = _ := by
  49       rw [show Fintype.card G = Fintype.card X ^ (Fintype.card I - (insert i A).card) from
  50         fixedCoordinates_card (insert i A) (Function.update a i x₀)]
  51       rw [Finset.card_insert_of_notMem hi]
  52       have hsub : Fintype.card I - (A.card + 1) = Fintype.card I - A.card - 1 := by omega
  53       rw [hsub, Nat.mul_comm]
