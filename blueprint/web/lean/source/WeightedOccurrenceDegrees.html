  11 theorem weighted_occurrence_degree_error {V E : Type*} [Fintype E] [DecidableEq V]
  12     (f : E → V) (w : E → ℝ) (κ : ℝ) (hκ : 0≤κ) (hw : ∀ e,|w e-1|≤κ) (v : V) :
  13     |occurrenceSumLinear f w v-occurrenceDegree f v|≤κ*occurrenceDegree f v := by
  14   have hd : (∑ e,if f e=v then (1:ℝ) else 0)=occurrenceDegree f v := by
  15     unfold occurrenceDegree
  16     apply Finset.sum_congr rfl
  17     intro e _
  18     split_ifs <;> rfl
  19   rw [← hd]
  20   change |(∑ e,if f e=v then w e else 0)-(∑ e,if f e=v then (1:ℝ) else 0)|≤_
  21   rw [← Finset.sum_sub_distrib,Finset.mul_sum]
  22   calc
  23     _ ≤ ∑ e,|(if f e=v then w e else 0)-(if f e=v then (1:ℝ) else 0)| := Finset.abs_sum_le_sum_abs _ _
  24     _ ≤ _ := by
  25       apply Finset.sum_le_sum
  26       intro e _
  27       by_cases h : f e=v
  28       · simpa only [h,ite_true,mul_one] using hw e
  29       · simp only [h,ite_false,sub_self,abs_zero,mul_zero,le_refl]

  31 theorem weighted_occurrence_degree_near {V E : Type*} [Fintype E] [DecidableEq V]
  32     (f : E → V) (w : E → ℝ) (κ ε n D : ℝ) (hκ : 0≤κ)
  33     (hw : ∀ e,|w e-1|≤κ) (v : V)
  34     (hdeg : occurrenceDegree f v≤n) (hnear : |occurrenceDegree f v-D|≤ε*n) :
  35     |occurrenceSumLinear f w v-D|≤(ε+κ)*n := by
  36   have hweight := (weighted_occurrence_degree_error f w κ hκ hw v).trans
  37     (mul_le_mul_of_nonneg_left hdeg hκ)
  38   rw [show occurrenceSumLinear f w v-D=
  39     (occurrenceSumLinear f w v-occurrenceDegree f v)+(occurrenceDegree f v-D) by ring]
  40   have hh := (abs_add_le _ _).trans (add_le_add hweight hnear)
  41   nlinarith

  43 theorem occurrence_degree_mean_near {V E : Type*} [Fintype V] [Fintype E]
  44     (f : E → V) (D ε : ℝ) (hn : 0<Fintype.card V)
  45     (hnear : ∀ v,|occurrenceDegree f v-D|≤ε) :
  46     |(Fintype.card E : ℝ)/Fintype.card V-D|≤ε := by
  47   have hnr : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr hn
  48   have he : (Fintype.card E : ℝ)/Fintype.card V-D=
  49       (∑ v,(occurrenceDegree f v-D))/Fintype.card V := by
  50     rw [Finset.sum_sub_distrib,occurrenceDegree_total]
  51     simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul]
  52     field_simp
  53   rw [he,abs_div,abs_of_pos hnr]
  54   apply (div_le_iff₀ hnr).mpr
  55   have hh := (Finset.abs_sum_le_sum_abs (s := Finset.univ) (f := fun v => occurrenceDegree f v-D)).trans
  56     (Finset.sum_le_sum (fun v _ => hnear v))
  57   simpa only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul,mul_comm] using hh

  59 theorem occurrence_degree_near_mean {V E : Type*} [Fintype V] [Fintype E]
  60     (f : E → V) (D ε : ℝ) (hn : 0<Fintype.card V)
  61     (hnear : ∀ v,|occurrenceDegree f v-D|≤ε) (v : V) :
  62     |occurrenceDegree f v-(Fintype.card E : ℝ)/Fintype.card V|≤2*ε := by
  63   have hm := occurrence_degree_mean_near f D ε hn hnear
  64   rw [show occurrenceDegree f v-(Fintype.card E : ℝ)/Fintype.card V=
  65     (occurrenceDegree f v-D)-((Fintype.card E : ℝ)/Fintype.card V-D) by ring]
  66   have hh := (abs_sub _ _).trans (add_le_add (hnear v) hm)
  67   linarith
