  11     (x ell theta : ℝ) (hc : 0 < c) (hd : 0 < d) (hK : 0 < K)
  12     (hx : 0 < x) (hθ : 0 < theta) (hell : (1:ℝ)/2 ≤ ell)
  13     (hmin : ∀ i, K ≤ a i)
  14     (htotal : ((∑ i, a i : ℕ) : ℝ) = ell*x^4)
  15     (hcut : 5/((c:ℝ)*(d:ℝ)+5) + (c:ℝ)*(d:ℝ)/K ≤ theta/2)
  16     (hlarge : 8*(c:ℝ)/theta < x)
  17     (hdemand : (demand:ℝ) ≤ (1-theta)*ell*x^4/((c:ℝ)*(d:ℝ)) + 2*x^3/(d:ℝ)) :
  18     demand ≤ ∑ i, a i/(c*d+5) := by
  19   have hcR : (0:ℝ) < c := by exact_mod_cast hc
  20   have hdR : (0:ℝ) < d := by exact_mod_cast hd
  21   have herr := reserved_root_error_bound x c d theta hx hcR hdR hθ hlarge
  22   have hcut' : 5/((↑(c*d):ℝ)+5) + (↑(c*d):ℝ)/K ≤ theta/2 := by
  23     simpa only [Nat.cast_mul] using hcut
  24   have herr' : 2*x^3/(d:ℝ) < theta*x^4/(4*(↑(c*d):ℝ)) := by
  25     simpa only [Nat.cast_mul] using herr
  26   have hdemand' : (demand:ℝ) ≤ (1-theta)*ell*x^4/(↑(c*d):ℝ) + 2*x^3/(d:ℝ) := by
  27     simpa only [Nat.cast_mul] using hdemand
  28   exact long_reserved_capacity_from_cutoff a (c*d) K demand (x^4) ell theta (2*x^3/(d:ℝ))
