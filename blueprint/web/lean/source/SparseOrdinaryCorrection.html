  17 theorem sparse_ordinary_degree_correction
  18     (r : ℕ) (hr : 3≤r) (ω h δ ξ : ℝ)
  19     (hω : 0<ω) (hh : 0<h) (hδ : 0<δ) (hδsmall : δ<1/4)
  20     (hξ : 0<ξ) (hξh : ξ≤h/100) :
  21     ∃ η₀ : ℝ,0<η₀ ∧ ∃ n₀ : ℕ,∀ η : ℝ,0≤η → η≤η₀ →
  22       ∀ (V : Type u) (U : Type v) [Fintype V] [Fintype U] [DecidableEq V] [DecidableEq U],
  23       n₀≤Fintype.card V → ω*Fintype.card V≤Fintype.card U →
  24       Fintype.card U≤Fintype.card V →
  25       ∀ (Q FQ : SimpleGraph V) (H FH : V → U → Prop) (A : V → V → ℕ),
  26       (∀ x y,A x y+A y x=if (Q \ FQ).Adj x y then 1 else 0) →
  27       BipartiteLargeCut A δ h →
  28       BipartiteLargeCut (fun v u=>if H v u ∧ ¬FH v u then 1 else 0) δ h →
  29       (∀ v,|(Q.degree v : ℝ)-2*(Finset.univ.filter (H v)).card|≤η*Fintype.card V) →
  30       ∃ (JQ : SimpleGraph V) (JH : V → U → Prop),
  31         JQ≤Q \ FQ ∧ (∀ v u,JH v u → H v u ∧ ¬FH v u) ∧
  32         (∀ v,(Q \ JQ).degree v=2*(Finset.univ.filter (fun u=>H v u ∧ ¬JH v u)).card) ∧
  33         (∀ u,r∣(Finset.univ.filter (fun v=>H v u ∧ ¬JH v u)).card) ∧
  34         (∀ v,(JQ.degree v : ℝ)+(Finset.univ.filter (JH v)).card≤10*ξ*Fintype.card V) ∧
  35         (∀ u,((Finset.univ.filter (fun v=>JH v u)).card : ℝ)≤10*ξ*Fintype.card V) := by

  37   obtain ⟨n₀,hn₀,hscale⟩ := sparse_correction_order_threshold r hδ hω hξ
  38   refine ⟨δ*ω*ξ/1000,by positivity,n₀,?_⟩
  39   intro η hη hηsmall V U _ _ _ _ hn hωn hsn Q FQ H FH A hA hcutA hcutH hb
  40   let n := Fintype.card V
  41   have hn1 : 1≤n := hn₀.trans hn
  42   have hnpos : 0<n := by omega
  43   have hnR : (0 : ℝ)<n := by exact_mod_cast hnpos
  44   have hspos : 0<Fintype.card U := by
  45     have hpos : (0 : ℝ)<Fintype.card U := (mul_pos hω hnR).trans_le hωn
  46     exact_mod_cast hpos
  47   have hω1 : ω≤1 := by
  48     have hsR : (Fintype.card U : ℝ)≤n := by exact_mod_cast hsn
  49     nlinarith
  50   let base := Nat.floor (ξ*n)
  51   have hnscale : 100*((r : ℝ)+2)≤δ*ω*ξ*n := hscale n hn
  52   obtain ⟨hrδ,hunit,hpositive,hrelative,hcapacity,hQbound,hHbound⟩ :=
  53     sparse_correction_numeric (by omega : 0<r) hδ (by linarith : δ≤1) hω hω1
  54       hξ hξh hη hηsmall hnscale
  55   change (r : ℝ)≤δ*base at hrδ
  56   change 1≤δ*ω*base at hunit
  57   change η*n+2≤2*ω*base at hpositive
  58   change 3/2+η*n≤δ*(ω*base-η*n/2) at hrelative
  59   change (base : ℝ)+r+η*n/2≤h*n/2 at hcapacity
  60   change 2*((base : ℝ)+r+1)+η*n≤4*ξ*n at hQbound
  61   change (base : ℝ)+r+1≤2*ξ*n at hHbound
  62   let b (v : V) : ℤ := (Q.degree v : ℤ)-2*(Finset.univ.filter (H v)).card
  63   let d (u : U) := (Finset.univ.filter (fun v=>H v u)).card
  64   have hb' (v : V) : |(b v : ℝ)|≤η*n := by
  65     simpa only [b,Int.cast_sub,Int.cast_mul,Int.cast_ofNat,Int.cast_natCast] using hb v
  66   have heven : Even (∑ v,b v) := ordinary_degree_imbalance_even Q H
  67   obtain ⟨J,K,hJ,hK,hbalance,hresidue,hJbound,hKrowbound,hKcolbound⟩ :=
  68     sparse_integer_correction_matrices A (fun v u=>if H v u ∧ ¬FH v u then 1 else 0)
  69       d b base r (by omega) hnpos hspos hsn hω hωn hδ.le hδsmall
  70       (by positivity : 0≤η*n) hrδ hunit hpositive hrelative hcapacity hb' heven hcutA hcutH
  71   obtain ⟨JQ,hJQ,hJQdeg⟩ := graph_of_suborientation (Q \ FQ) A J (fun x y=>by
  72     by_cases hx : (Q \ FQ).Adj x y
  73     · simpa only [hx,ite_true] using hA x y
  74     · simpa only [hx,ite_false] using hA x y) hJ
  75   obtain ⟨JH,hJH,hJHrow,hJHcol⟩ := relation_from_submatrix
  76     (fun v u=>H v u ∧ ¬FH v u) K (fun v u=>by
  77       by_cases hvu : H v u ∧ ¬FH v u
  78       · simpa only [hvu,ite_true] using hK v u
  79       · simpa only [hvu,ite_false] using hK v u)
  80   have hJQ' : JQ≤Q := hJQ.trans (by intro x y hx;exact hx.1)
  81   have hJH' : ∀ v u,JH v u → H v u := fun v u hv=>(hJH v u hv).1
  82   have hHrowle (v : V) : (Finset.univ.filter (JH v)).card≤(Finset.univ.filter (H v)).card := by
  83     apply Finset.card_le_card
  84     intro u hu
  85     simp only [Finset.mem_filter,Finset.mem_univ,true_and] at hu ⊢
  86     exact hJH' v u hu
  87   refine ⟨JQ,JH,hJQ,hJH,?_,?_,?_,?_⟩
  88   · intro v
  89     have hhbal := hbalance v
  90     rw [← hJHrow v] at hhbal
  91     have hphys := hJQdeg v
  92     have hdq := SimpleGraph.degree_le_of_le hJQ' (v:=v)
  93     have hdh := hHrowle v
  94     rw [graph_degree_sdiff Q JQ hJQ',relation_deleted_row_degree H JH hJH']
  95     dsimp [b] at hhbal
  96     omega
  97   · intro u
  98     rw [relation_deleted_column_degree H JH hJH']
  99     apply Nat.ModEq.dvd'
 100     change (Finset.univ.filter (fun v=>JH v u)).card%r=(Finset.univ.filter (fun v=>H v u)).card%r
 101     rw [hJHcol]
 102     exact hresidue u
 103   · intro v
 104     have hj : (JQ.degree v : ℝ)≤4*ξ*n := by
 105       rw [hJQdeg]

 108     have hk : ((Finset.univ.filter (JH v)).card : ℝ)≤2*ξ*n := by
 109       rw [hJHrow]
 110       exact (hKrowbound v).trans hHbound
 111     change _≤10*ξ*n
 112     nlinarith [mul_nonneg hξ.le hnR.le]
 113   · intro u
