  13 theorem enlarged_neighbour_density {V : Type*} [Fintype V] [DecidableEq V]
  14     (P : SimpleGraph V) (T Z : Finset V) (δ α : ℝ)
  15     (hnonneg : 0≤δ+α/2)
  16     (hmargin : (δ+α/2)*(Z.card:ℝ)≤α*(T.card:ℝ)/2)
  17     (hdeg : ∀ v∈T∪Z,(δ+α)*(T.card:ℝ)≤((P.neighborFinset v∩T).card:ℝ)) :
  18     ∀ v∈T∪Z,(δ+α/2)*((T∪Z).card:ℝ)≤((P.neighborFinset v∩(T∪Z)).card:ℝ) := by
  19   intro v hv
  20   have hc : ((T∪Z).card:ℝ)≤T.card+Z.card := by exact_mod_cast Finset.card_union_le T Z
  21   have hd : ((P.neighborFinset v∩T).card:ℝ)≤(P.neighborFinset v∩(T∪Z)).card := by
  22     exact_mod_cast Finset.card_le_card (Finset.inter_subset_inter_left Finset.subset_union_left)
  23   have hm := mul_le_mul_of_nonneg_left hc hnonneg
  24   nlinarith only [hmargin,hdeg v hv,hd,hm]

  26 theorem enlarged_induced_degree {V : Type*} [Fintype V] [DecidableEq V]
  27     (P : SimpleGraph V) (T Z : Finset V) (δ α : ℝ)
  28     (hnonneg : 0≤δ+α/2)
  29     (hmargin : (δ+α/2)*(Z.card:ℝ)≤α*(T.card:ℝ)/2)
  30     (hdeg : ∀ v∈T∪Z,(δ+α)*(T.card:ℝ)≤((P.neighborFinset v∩T).card:ℝ)) :
  31     ∀ v : {v // v∈((T∪Z:Finset V):Set V)},
  32       (δ+α/2)*Fintype.card {v // v∈((T∪Z:Finset V):Set V)}≤((P.induce ((T∪Z:Finset V):Set V)).degree v:ℝ) := by
  33   intro v
  34   rw [induced_degree_intersection]
  35   have hc : Fintype.card {v // v∈((T∪Z:Finset V):Set V)}=(T∪Z).card := by
  36     rw [Fintype.card_subtype]
  37     congr 1
  38     ext x
  39     simp
  40   rw [hc]
  41   exact enlarged_neighbour_density P T Z δ α hnonneg hmargin hdeg v.val v.property
