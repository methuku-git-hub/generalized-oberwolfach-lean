  10 theorem regularization_constants (β : ℝ) (hβ : 0 < β) :
  11     ∃ δ h C η₀ : ℝ,
  12       0 < δ ∧ δ ≤ 1 / 8 ∧ 0 < h ∧ h ≤ β - δ ∧ 0 < C ∧ C = 4 / h ∧ 0 < η₀ ∧
  13       ∀ η : ℝ, 0 < η → η < η₀ →
  14         0 < Real.sqrt η ∧ Real.sqrt η ≤ 1 / 4 ∧ Real.sqrt η ≤ δ / 8 ∧
  15           0 < C * Real.sqrt η ∧ C * Real.sqrt η ≤ 1 := by
  16   let δ := min (β / 4) (1 / 8)
  17   let h := min (β / 2) (1 / 4)
  18   have hδ : 0 < δ := lt_min (by positivity) (by norm_num)
  19   have hδ8 : δ ≤ 1 / 8 := min_le_right _ _
  20   have hh : 0 < h := lt_min (by positivity) (by norm_num)
  21   have hhβ : h ≤ β - δ := by
  22     have hδβ : δ ≤ β / 4 := min_le_left _ _
  23     have hhβ : h ≤ β / 2 := min_le_left _ _
  24     linarith
  25   let C := 4 / h
  26   have hC : 0 < C := div_pos (by norm_num) hh
  27   let η₀ := min ((δ / 8) ^ 2) ((1 / C) ^ 2)
  28   have hη₀ : 0 < η₀ := lt_min (pow_pos (by positivity) _) (pow_pos (by positivity) _)
  29   refine ⟨δ, h, C, η₀, hδ, hδ8, hh, hhβ, hC, rfl, hη₀, ?_⟩
  30   intro η hη hsmall
  31   have hs := (Real.sqrt_pos.mpr hη)
  32   have hsd : Real.sqrt η < δ / 8 := (Real.sqrt_lt' (by positivity)).mpr
  33     (hsmall.trans_le (min_le_left _ _))
  34   have hsC : Real.sqrt η < 1 / C := (Real.sqrt_lt' (by positivity)).mpr
  35     (hsmall.trans_le (min_le_right _ _))
  36   refine ⟨hs, by linarith, hsd.le, mul_pos hC hs, ?_⟩
  37   have hh := (lt_div_iff₀ hC).mp hsC
  38   nlinarith

  40 theorem regularization_order_threshold (η : ℝ) (hη : 0 < η) :
  41     ∃ m₀ : ℕ, 1 ≤ m₀ ∧ ∀ m : ℕ, m₀ ≤ m → 0 < m ∧ 1 ≤ η * m := by
  42   refine ⟨⌈1 / η⌉₊ + 1, by omega, ?_⟩
  43   intro m hm
  44   have hmpos : 0 < m := by omega
  45   have hceil : (1 / η : ℝ) ≤ m :=
  46     (Nat.le_ceil _).trans (by exact_mod_cast (show ⌈1 / η⌉₊ ≤ m by omega))
  47   exact ⟨hmpos, by nlinarith [(div_le_iff₀ hη).mp hceil]⟩
