   4 /-! Simultaneous construction of the rim paths in a bounded ordinary
   5 transformer. The forbidden set is enlarged by the actual earlier path
   6 supports, whose total size is bounded before each application. -/

  15  theorem separated_transformer_paths (Q F:SimpleGraph V)
  16     (A:I→Finset V) (W Ω:Finset V) (Z:I→Finset V) (hAΩ:∀ i,A i⊆Ω)
  17     (x y x' y':I→V) (hroots:∀ i,x i∈Z i∧y i∈Z i∧x' i∈Z i∧y' i∈Z i)
  18     (hΩ:∀ i,x i∈Ω∧y i∈Ω∧x' i∈Ω∧y' i∈Ω)
  19     (hZ:∀ i,∀ v∈Z i,∀ w∈Z i,¬F.Adj v w)
  20     (r:I→ℕ) (d:ℕ) (hr:∀ i,4≤r i) (hd:∀ i,r i-2≤d) (L b:ℝ) (hL:0≤L) (hF:∀ v,(F.degree v:ℝ)≤L)
  21     (hcount:∀ i,∀ v∈Ω,∀ w∈Ω,b≤((A i∩Q.neighborFinset v∩Q.neighborFinset w).card:ℝ))
  22     (hthree:∀ i,r i=4→∀ a∈A i,b≤(((A i∩Q.neighborFinset a)∩Q.neighborFinset (y i)∩Q.neighborFinset (y' i)).card:ℝ))
  23     (hbudget:∀ i,W.card+Fintype.card I*d*(L+1)+(((Z i).card:ℝ)+r i+2)*(L+1)<b):
  24     ∃ p:I→(Σ a:V,Σ c:V,Q.Walk a c),
  25       (∀ i,(p i).2.2.IsPath ∧ (p i).2.2.length=r i-3 ∧
  26         Q.Adj (x i) (p i).1 ∧ Q.Adj (x' i) (p i).1 ∧
  27         Q.Adj (y i) (p i).2.1 ∧ Q.Adj (y' i) (p i).2.1 ∧
  28         (∀ v∈(p i).2.2.support,v∈A i∧v∉W∧v∉Z i) ∧
  29         (∀ v∈(p i).2.2.support,∀ w∈(p i).2.2.support,¬F.Adj v w) ∧
  30         (∀ v∈(p i).2.2.support,∀ w∈Z i,¬F.Adj v w)) ∧
  31       ∀ i j,i≠j→Disjoint (p i).2.2.support.toFinset (p j).2.2.support.toFinset ∧
  32         (∀ v∈(p i).2.2.support,∀ w∈(p j).2.2.support,¬F.Adj v w):=by
  33   let S:(Σ a:V,Σ c:V,Q.Walk a c)→Finset V:=fun p=>p.2.2.support.toFinset
  34   let Good:=fun (i:I) (p:Σ a:V,Σ c:V,Q.Walk a c)=>
  35     p.2.2.IsPath ∧ p.2.2.length=r i-3 ∧ Q.Adj (x i) p.1 ∧ Q.Adj (x' i) p.1 ∧
  36       Q.Adj (y i) p.2.1 ∧ Q.Adj (y' i) p.2.1 ∧
  37       (∀ v∈p.2.2.support,v∈A i∧v∉Z i) ∧
  38       (∀ v∈p.2.2.support,∀ w∈p.2.2.support,¬F.Adj v w) ∧
  39       (∀ v∈p.2.2.support,∀ w∈Z i,¬F.Adj v w)
  40   have hchoose:∀ i,∀ B:Finset V,(B.card:ℝ)≤W.card+Fintype.card I*d*(L+1)→
  41       ∃ p,Good i p ∧ (S p).card≤d ∧ Disjoint (S p) B:=by
  42     intro i B hB
  43     obtain ⟨a,c,p,hp,hlen,hxa,hx'a,hyc,hy'c,hclass,hind,hroot⟩:=
  44       transformer_rim_common_path Q F (A i) B Ω (Z i) (hAΩ i) (x i) (y i) (x' i) (y' i)
  45         (hroots i) (hΩ i) (hZ i) (r i) (hr i) L b hL hF (hcount i) (hthree i)
  46         (by linarith only [hB,hbudget i])
  47     refine ⟨⟨a,c,p⟩,⟨hp,hlen,hxa,hx'a,hyc,hy'c,?_,hind,hroot⟩,?_,?_⟩
  48     · intro v hv;exact ⟨(hclass v hv).1,(hclass v hv).2.2⟩
  49     · have hh:=List.toFinset_card_le p.support
  50       rw [p.length_support,hlen] at hh
  51       dsimp only [S]
  52       have hh' := hd i
  53       have hh'' := hr i
  54       omega
  55     · apply Finset.disjoint_left.mpr
  56       intro v hv hBv
  57       exact (hclass v (List.mem_toFinset.mp hv)).2.1 hBv
  58   obtain ⟨p,hp,hdis⟩:=finite_conflict_separated_choices S Good W d F L hL hF hchoose
  59   refine ⟨p,?_,?_⟩
  60   rotate_left
  61   · intro i j hij
  62     exact ⟨(hdis i j hij).1,fun v hv w hw=>(hdis i j hij).2 v (List.mem_toFinset.mpr hv) w (List.mem_toFinset.mpr hw)⟩
  63   intro i
  64   obtain ⟨hi,hcard,hW⟩:=hp i
  65   rcases hi with ⟨hpath,hlen,hxa,hx'a,hyc,hy'c,hclass,hind,hroot⟩
  66   refine ⟨hpath,hlen,hxa,hx'a,hyc,hy'c,?_,hind,hroot⟩
  67   intro v hv
  68   exact ⟨(hclass v hv).1,fun hw=>Finset.disjoint_left.mp hW (List.mem_toFinset.mpr hv) hw,(hclass v hv).2⟩
