  23     have hxory:x∈e ∨ y∈e:=by
  24       by_contra h

  29       rcases hv with rfl|rfl
  30       · exact (not_or.mp h).1 (by simpa using hv')
  31       · exact (not_or.mp h).2 (by simpa using hv')
  32     rcases hxory with hx|hy
  33     · apply Finset.mem_union_left
  34       rw [SimpleGraph.incidenceFinset_eq_filter]
