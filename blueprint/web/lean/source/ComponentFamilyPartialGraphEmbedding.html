   4 set_option autoImplicit false

   7 attribute [local instance] Classical.propDecidable

  10 theorem component_family_partial_graph_embedding
  11     (F:SimpleGraph (Fin n)) (C:A→F.ConnectedComponent)
  12     (hC:Function.Injective C) (f:∀a,C a↪Fin n)
  13     (hdis:∀a b,a≠b→Disjoint (Finset.univ.image (f a)) (Finset.univ.image (f b)))
  14     (R:A→Fin n→Fin n→Prop)
  15     (hR:∀a (x y:C a),(C a).toSimpleGraph.Adj x y→R a (f a x) (f a y)):
  16     ∃ψ:{v:Fin n // ∃a,v∈(C a).supp}↪Fin n,
  17       (∀a (x:C a),ψ ⟨x.val,⟨a,x.property⟩⟩=f a x) ∧
  18       (∀x y,F.Adj x.val y.val→∃a,R a (ψ x) (ψ y)) :=by
  19   obtain ⟨ψ,hψ⟩:=component_family_partial_embedding F C hC f hdis
  20   refine ⟨ψ,hψ,?_⟩
  21   intro x y hxy
  22   obtain ⟨a,hx⟩:=x.property
  23   have hy:y.val∈(C a).supp:=(C a).mem_supp_of_adj_mem_supp hx hxy
  24   let x':C a:=⟨x.val,hx⟩
  25   let y':C a:=⟨y.val,hy⟩
  26   have hxψ:ψ x=f a x':=by
  27     have h:=hψ a x'
  28     convert h using 1 <;> congr 1
  29   have hyψ:ψ y=f a y':=by
  30     have h:=hψ a y'
  31     convert h using 1 <;> congr 1
  32   refine ⟨a,?_⟩
  33   rw [hxψ,hyψ]
  34   exact hR a x' y' hxy
  35 end Oberwolfach
