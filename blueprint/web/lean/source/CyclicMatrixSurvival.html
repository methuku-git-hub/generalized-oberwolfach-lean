  13 def cyclicMatrixEndpoints {k : ℕ} {X : Type*} (v : Fin (k + 2) → X × X) :
  14     Fin (k + 2) ⊕ Fin (k + 2) → X × X
  15   | Sum.inl i => v i
  16   | Sum.inr i => ((v i).2, (v (finRotate (k + 2) i)).1)

  18 theorem cyclicMatrixTupleValid_iff {k : ℕ} {X : Type*}
  19     (M : (Fin (k + 2) ⊕ Fin (k + 2)) → X → X → ℕ)
  20     (v : Fin (k + 2) → X × X) : CyclicMatrixTupleValid M v ↔
  21       ∀ i, 0 < M i (cyclicMatrixEndpoints v i).1 (cyclicMatrixEndpoints v i).2 := by
  22   constructor
  23   · intro h i; cases i with
  24     | inl i => exact (h i).1
  25     | inr i => exact (h i).2
  26   · intro h i; exact ⟨h (Sum.inl i), h (Sum.inr i)⟩

  28 theorem cyclicMatrix_tuple_survival {Ω X : Type*} [Fintype Ω] {k : ℕ}
  29     (μ : FiniteLaw Ω)
  30     (M : (Fin (k + 2) ⊕ Fin (k + 2)) → X → X → ℕ)
  31     (R : Ω → (Fin (k + 2) ⊕ Fin (k + 2)) → X → X → ℕ)
  32     (p : ℝ)
  33     (hp : ∀ i x y, 0 < M i x y → μ.probability (fun ω => R ω i x y = 0) ≤ p)
  34     (v : Fin (k + 2) → X × X) (hv : CyclicMatrixTupleValid M v) :
  35     1 - (2 * (k + 2 : ℝ)) * p ≤ μ.probability (fun ω => CyclicMatrixTupleValid (R ω) v) := by
  36   have h := finiteLaw_common_event_lower μ Finset.univ
  37     (fun i ω => R ω i (cyclicMatrixEndpoints v i).1 (cyclicMatrixEndpoints v i).2 = 0)
  38     (fun _ => p)
  39     (fun i _ => hp i _ _ ((cyclicMatrixTupleValid_iff M v).mp hv i))
  40   simpa only [Finset.sum_const, Finset.card_univ, Fintype.card_sum, Fintype.card_fin,
  41     nsmul_eq_mul, Nat.cast_add, Nat.cast_ofNat, Finset.mem_univ, forall_const,
  42     ← Nat.pos_iff_ne_zero, ← cyclicMatrixTupleValid_iff, two_mul] using h

  44 theorem finiteLaw_binary_mean {Ω : Type*} [Fintype Ω]
  45     (μ : FiniteLaw Ω) (R : Ω → ℕ) (hR : ∀ ω, R ω ≤ 1) :
  46     μ.expectation (fun ω => (R ω : ℝ)) = μ.probability (fun ω => 0 < R ω) := by
  47   unfold FiniteLaw.expectation FiniteLaw.probability
  48   apply Finset.sum_congr rfl
  49   intro ω _
  50   have h := hR ω
  51   by_cases hz : R ω = 0
  52   · simp [hz]
  53   · have h1 : R ω = 1 := by omega
  54     simp [h1]

  56 theorem finiteLaw_retained_binary_mean {Ω : Type*} [Fintype Ω]
  57     (μ : FiniteLaw Ω) (M : ℕ) (hM : M ≤ 1) (R : Ω → ℕ)
  58     (hR : ∀ ω, R ω ≤ M) (p : ℝ)
  59     (hp : 0 < M → μ.probability (fun ω => R ω = 0) ≤ p) :
  60     (1 - p) * M ≤ μ.expectation (fun ω => (R ω : ℝ)) ∧
  61       μ.expectation (fun ω => (R ω : ℝ)) ≤ M := by
  62   constructor
  63   · by_cases hz : M = 0
  64     · have hz' : ∀ ω, R ω = 0 := fun ω => Nat.le_zero.mp (hz ▸ hR ω)
  65       simp only [hz, Nat.cast_zero, mul_zero, hz', finiteLaw_expectation_const, le_refl]
  66     · have h1 : M = 1 := by omega
  67       rw [finiteLaw_binary_mean μ R (fun ω => (hR ω).trans hM), h1, Nat.cast_one, mul_one]
  68       have he := finiteLaw_probability_compl μ (fun ω => R ω = 0)
  69       simp only [← Nat.pos_iff_ne_zero] at he
  70       linarith [hp (by omega)]
  71   · have h := finiteLaw_expectation_mono μ (fun ω => (R ω : ℝ)) (fun _ => (M : ℝ))
  72       (fun ω => by exact_mod_cast hR ω)
  73     simpa only [finiteLaw_expectation_const] using h
