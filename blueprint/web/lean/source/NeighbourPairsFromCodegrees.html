  14 theorem ordinary_neighbour_pairs_from_codegrees (p:ℝ) (hp:0<p) (hp1:p≤1) :
  15     ∃ ε:ℝ,0<ε ∧ ∀ b:ℕ,∃ N:ℕ,
  16       ∀ (X Y:Type) [Fintype X] [Fintype Y],∀ m:ℕ,N≤m →
  17       Fintype.card X=2*m → Fintype.card Y=m →
  18       ∀ (H:X→Y→Prop) (F:SimpleGraph X),
  19       (∀ x,F.degree x≤b) →
  20       (∀ x,(p-ε)*m≤∑ u,bipAdj H x u) →
  21       (∀ u,p*m≤∑ x,bipAdj H x u) →
  22       (∀ x y,x≠y → ¬F.Adj x y → ∑ u,bipAdj H x u*bipAdj H y u≤(p^2+ε)*m) →
  23       ∃ v w:Y↪X,(∀ u,H (v u) u∧H (w u) u∧¬F.Adj (v u) (w u)) ∧
  24         Disjoint (Set.range v) (Set.range w) ∧ (∀ x,(∃ u,v u=x)∨∃ u,w u=x) := by
  25   let η:=p/64
  26   have hη:0<η:=by dsimp [η];positivity
  27   let ε:=min (p/2) (p^2*η/24)
  28   have hε:0<ε:=by dsimp [ε];positivity
  29   refine ⟨ε,hε,?_⟩
  30   intro b
  31   obtain ⟨N₀,hN⟩:=ordinary_neighbour_pair_assignment (p/4) (by positivity) b
  32   obtain ⟨N₁,hN₁⟩:=exists_nat_gt (8*(b+1)/(p^2*η^2):ℝ)
  33   refine ⟨max N₀ (max N₁ 1),?_⟩
  34   intro X Y _ _ m hm hX hY H F hF hrow hcol hcodeg
  35   classical
  36   have hmpos:0<m:=by omega
  37   have hmass:(b+1:ℝ)≤p^2*η^2*m/8:=by
  38     have hh:=(div_lt_iff₀ (mul_pos (sq_pos_of_pos hp) (sq_pos_of_pos hη))).mp
  39       (hN₁.trans_le (Nat.cast_le.mpr (show N₁≤m by omega)))
  40     nlinarith only [hh]
  41   let E:X→Finset X:=fun x=>insert x (F.neighborFinset x)
  42   have hE (x:X):((E x).card:ℝ)≤b+1:=by
  43     have hh:(E x).card≤b+1:=by
  44       have hh:=Finset.card_insert_le x (F.neighborFinset x)
  45       rw [SimpleGraph.card_neighborFinset_eq_degree] at hh
  46       exact hh.trans (Nat.add_le_add_right (hF x) 1)
  47     exact_mod_cast hh
  48   have hcodeg' (x y:X) (hy:y∉E x):∑ u,bipAdj H x u*bipAdj H y u≤(p^2+ε)*Fintype.card Y:=by
  49     have hne:x≠y:=by intro he;apply hy;simp [E,he]
  50     have hFxy:¬F.Adj x y:=by intro he;apply hy;exact Finset.mem_insert_of_mem ((F.mem_neighborFinset x y).mpr he)
  51     simpa only [hY] using hcodeg x y hne hFxy
  52   apply hN X Y m (by omega) hX hY H F
  53   · intro x
  54     have he:((Finset.univ.filter (H x)).card:ℝ)=∑ u,bipAdj H x u:=by simp only [bipAdj,Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  55     rw [he]
  56     have hsmall:ε≤p/2:=min_le_left _ _
  57     have hh:=mul_le_mul_of_nonneg_right hsmall (Nat.cast_nonneg m : (0:ℝ)≤m)
  58     have hpos:0≤p*m:=by positivity
  59     nlinarith only [hrow x,hh,hpos]
  60   · intro u
  61     have he:((Finset.univ.filter (fun x=>H x u)).card:ℝ)=∑ x,bipAdj H x u:=by simp only [bipAdj,Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  62     rw [he]
  63     have hpos:0≤p*m:=by positivity
  64     nlinarith only [hcol u,hpos]
  65   · exact hF
  66   · intro S T hS hT
  67     have hSl:η*Fintype.card Y≤(S.card:ℝ):=by simpa only [hY,η,div_div,show (4*16:ℝ)=64 by norm_num] using hS
  68     have hTl:η*Fintype.card Y≤(T.card:ℝ):=by simpa only [hY,η,div_div,show (4*16:ℝ)=64 by norm_num] using hT
  69     have hh:=bipartite_cuts_from_degrees_and_codegrees H p ε (b+1) η hp hp1 hε.le hη
  70       (by simpa only [hY] using hmpos) E hE (by simpa only [hY] using hrow) hcodeg'
  71       (min_le_right _ _) (by simpa only [hY] using hmass) S T hSl hTl
  72     have hnonneg:0≤p*S.card*T.card:=by positivity
  73     nlinarith only [hh,hnonneg]
