  18     (he:markedPhysicalEdge tail head ktail khead π (markedCycleTypedEdge tail head ktail khead m π b C i)=

  35  theorem marked_rim_cycle_family_packing
  36     (hp:Function.Injective (markedPhysicalEdge tail head ktail khead π))
  37     (C:I→MarkedCycleTuple tail head ktail khead m π b) (R:Finset V) (P:I→Finset V)
  38     (old:I→Finset V) (hold:Function.Injective old)
  39     (hvertices:∀ i a,(C i).val a∈R∪P i)
  40     (hPR:∀ i,Disjoint (P i) R) (hPP:∀ i j,i≠j→Disjoint (P i) (P j))
  41     (hroot:∀ i a,(C i).val a∈R→(C i).val (finRotate (m+1) a)∈R→
  42       markedPhysicalEdge tail head ktail khead π (markedCycleTypedEdge tail head ktail khead m π b (C i) a)=old i):
  43     ∀ i j,i≠j→Disjoint (markedCyclePhysicalEdges tail head ktail khead m π b (C i))
  44       (markedCyclePhysicalEdges tail head ktail khead m π b (C j)):=by
  45   intro i j hij
  46   apply Finset.disjoint_left.mpr
  47   intro e he hf
  48   obtain ⟨a,_,ha⟩:=Finset.mem_image.mp he
  49   obtain ⟨z,_,hz⟩:=Finset.mem_image.mp hf
  50   have hc:=marked_cycle_common_physical_endpoints hp (C i) (C j) a z (ha.trans hz.symm)
  51   have htail:(C i).val a∈R:=private_path_common_vertex R (P i) (P j) (hPR i) (hPR j) (hPP i j hij) _
  52     (hvertices i a) (hc.1 ▸ hvertices j z)
  53   have hhead:(C i).val (finRotate (m+1) a)∈R:=private_path_common_vertex R (P i) (P j)
  54     (hPR i) (hPR j) (hPP i j hij) _ (hvertices i _) (hc.2 ▸ hvertices j _)
  55   have htail':(C j).val z∈R:=hc.1 ▸ htail
  56   have hhead':(C j).val (finRotate (m+1) z)∈R:=hc.2 ▸ hhead
  57   exact hij (hold ((hroot i a htail hhead).symm.trans ((ha.trans hz.symm).trans (hroot j z htail' hhead'))))
