   9 theorem long_reserved_capacity_from_rpow
  10     {I : Type*} [Fintype I] (a : I → ℕ) (c d K demand : ℕ)
  11     (N ell theta : ℝ) (hc : 0 < c) (hd : 0 < d) (hK : 0 < K)
  12     (hN : 0 < N) (hθ : 0 < theta) (hell : (1:ℝ)/2 ≤ ell)
  13     (hmin : ∀ i, K ≤ a i)
  14     (htotal : ((∑ i, a i : ℕ) : ℝ) = ell*N)
  15     (hcut : 5/((c:ℝ)*(d:ℝ)+5) + (c:ℝ)*(d:ℝ)/K ≤ theta/2)
  16     (hlarge : 8*(c:ℝ)/theta < N^((1:ℝ)/4))
  17     (hdemand : (demand:ℝ) ≤ (1-theta)*ell*N/((c:ℝ)*(d:ℝ)) + 2*N^((3:ℝ)/4)/(d:ℝ)) :
  18     demand ≤ ∑ i, a i/(c*d+5) := by
  19   have h4 : (N^((1:ℝ)/4))^4 = N := by
  20     rw [← Real.rpow_natCast, ← Real.rpow_mul (le_of_lt hN)]
  21     norm_num
  22   have h3 : (N^((1:ℝ)/4))^3 = N^((3:ℝ)/4) := by
  23     rw [← Real.rpow_natCast, ← Real.rpow_mul (le_of_lt hN)]
  24     norm_num
  25   have htotal' : ((∑ i, a i : ℕ) : ℝ) = ell*(N^((1:ℝ)/4))^4 := by
  26     simpa only [h4] using htotal
  27   have hdemand' : (demand:ℝ) ≤ (1-theta)*ell*(N^((1:ℝ)/4))^4/((c:ℝ)*(d:ℝ)) +
  28       2*(N^((1:ℝ)/4))^3/(d:ℝ) := by
  29     simpa only [h4,h3] using hdemand
  30   exact long_reserved_capacity_from_root a c d K demand (N^((1:ℝ)/4)) ell theta hc hd hK
  31     (Real.rpow_pos_of_pos hN _) hθ hell hmin htotal' hcut hlarge hdemand'
  32 end Oberwolfach
