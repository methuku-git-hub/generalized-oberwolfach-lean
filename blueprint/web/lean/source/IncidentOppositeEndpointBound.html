  12 theorem incident_opposite_endpoint_bound {A D V:Type} [Fintype D] [DecidableEq D] [DecidableEq V]

  14     (S:Finset A) (r:V) (X:Finset V) (hS:∀ a∈S,tail a=r∨head a=r):
  15     (S.filter (fun a=>(if tail a=r then head a else tail a)∈X)).card≤
  16       2*Fintype.card D*X.card:=by
  17   classical
  18   let T:=S.filter (fun a=>(if tail a=r then head a else tail a)∈X)
  19   let f:=fun a:A=>(tag a,decide (tail a=r),if tail a=r then head a else tail a)
  20   have hf:Set.InjOn f T:=by
  21     intro a ha b hb hab
  22     have ha':=hS a (Finset.mem_filter.mp ha).1
  23     have hb':=hS b (Finset.mem_filter.mp hb).1
  24     have he0:tag a=tag b:=congrArg Prod.fst hab
  25     have he1:decide (tail a=r)=decide (tail b=r):=congrArg (fun z=>z.2.1) hab
  26     have he2:(if tail a=r then head a else tail a)=(if tail b=r then head b else tail b):=
  27       congrArg (fun z=>z.2.2) hab
  28     apply hinj
  29     by_cases hat:tail a=r
  30     · have hbt:tail b=r:=by simpa only [hat,decide_true,true_eq_decide_iff] using he1
  31       simpa only [hat,hbt,ite_true,Prod.mk.injEq,true_and,and_true] using And.intro he0 he2
  32     · have hbt:tail b≠r:=by simpa only [hat,decide_false,false_eq_decide_iff] using he1
  33       have hah:head a=r:=ha'.resolve_left hat
  34       have hbh:head b=r:=hb'.resolve_left hbt
  35       simpa only [hat,hbt,ite_false,hah,hbh,Prod.mk.injEq,true_and,and_true] using And.intro he0 he2
  36   have hsub:T.image f⊆Finset.univ.product (Finset.univ.product X):=by
  37     intro z hz
  38     obtain ⟨a,ha,rfl⟩:=Finset.mem_image.mp hz
  39     exact Finset.mem_product.mpr ⟨Finset.mem_univ _,Finset.mem_product.mpr
  40       ⟨Finset.mem_univ _,(Finset.mem_filter.mp ha).2⟩⟩
  41   have hcard:=(Finset.card_le_card hsub)
  42   rw [Finset.card_image_iff.mpr hf] at hcard
  43   apply hcard.trans_eq
