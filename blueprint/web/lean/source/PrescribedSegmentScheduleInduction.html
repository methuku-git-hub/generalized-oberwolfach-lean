  11 def PrescribedSegmentsCompatible (A:Finset X) (D:X→X→Prop) (F:SimpleGraph X)
  12     (segs:List (PrescribedPathSegment X)):Prop:=
  13   ∀ seg∈segs,
  14     (∀ a,seg.left=some a→a∈A) ∧ (∀ b,seg.right=some b→b∈A) ∧
  15     (∀ a b,seg.left=some a→seg.right=some b→a≠b) ∧
  16     List.IsChain (fun x y=>D x y ∧ ¬F.Adj x y) (prescribedPathSegmentVertices seg)
