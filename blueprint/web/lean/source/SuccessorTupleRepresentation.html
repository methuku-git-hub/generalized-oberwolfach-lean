  15 def pairPositionEquiv (k : ℕ) : (Fin (k+2) ⊕ Fin (k+2)) ≃ Fin ((k+2)*2) where
  16   toFun s := match s with
  17     | Sum.inl i => alternatingIndex i 0
  18     | Sum.inr i => alternatingIndex i 1
  19   invFun j := let p := finProdFinEquiv.symm j
  20     if p.2 = 0 then Sum.inl p.1 else Sum.inr p.1
  21   left_inv s := by cases s <;> simp [alternatingIndex]
  22   right_inv j := by
  23     obtain ⟨⟨i,b⟩,rfl⟩ := finProdFinEquiv.surjective j
  24     fin_cases b <;> simp [alternatingIndex]

  34 def pairSequenceEquiv (k : ℕ) (V : Type*) :
  35     (Fin (k+2) → V × V) ≃ (Fin ((k+2)*2) → V) where
  36   toFun t j := pairTupleLabels t ((pairPositionEquiv k).symm j)
  37   invFun f i := (f (alternatingIndex i 0),f (alternatingIndex i 1))
  38   left_inv t := by
  39     funext i
  40     simp only [pairPositionEquiv_symm_zero,pairPositionEquiv_symm_one,pairTupleLabels,Prod.mk.eta]
  41   right_inv f := by
  42     funext j
  43     obtain ⟨s,rfl⟩ := (pairPositionEquiv k).surjective j
  44     cases s with
  45     | inl i =>
  46       change pairTupleLabels (fun a => (f (alternatingIndex a 0),f (alternatingIndex a 1)))
  47         ((pairPositionEquiv k).symm ((pairPositionEquiv k) (Sum.inl i))) = f ((pairPositionEquiv k) (Sum.inl i))
  48       rw [Equiv.symm_apply_apply]
  49       rfl
  50     | inr i =>
  51       change pairTupleLabels (fun a => (f (alternatingIndex a 0),f (alternatingIndex a 1)))
  52         ((pairPositionEquiv k).symm ((pairPositionEquiv k) (Sum.inr i))) = f ((pairPositionEquiv k) (Sum.inr i))
  53       rw [Equiv.symm_apply_apply]
  54       rfl

  68 def SuccessorSequenceValid {V : Type*} {c : ℕ} (G : SimpleGraph V) (π : Equiv.Perm V)

  72 theorem successor_pair_sequence_valid {V : Type*} {k : ℕ}
  73     (G : SimpleGraph V) (π : Equiv.Perm V) (t : Fin (k+2) → V × V) :
  74     TwistedCyclicTupleValid G (fun i => choosePermutation π (closingInterface k i)) t ↔
  75       SuccessorSequenceValid G π (pairSequenceEquiv k V t) := by
  76   have hzero (i : Fin (k+2)) : (alternatingIndex i 0).val+1 ≠ (k+2)*2 := by
  77     change 0+2*i.val+1 ≠ (k+2)*2
  78     have hi := i.isLt
  79     omega
  80   have hone (i : Fin (k+2)) : (alternatingIndex i 1).val+1 = (k+2)*2 ↔ i.val+1=k+2 := by
  81     change 1+2*i.val+1 = (k+2)*2 ↔ i.val+1=k+2
  82     omega
  83   have hz (i : Fin (k+2)) :
  84       G.Adj (pairSequenceEquiv k V t (alternatingIndex i 0))
  85         (if (alternatingIndex i 0).val+1=(k+2)*2 then
  86           π (pairSequenceEquiv k V t (cycleNext (alternatingIndex i 0)))
  87         else pairSequenceEquiv k V t (cycleNext (alternatingIndex i 0))) ↔ G.Adj (t i).1 (t i).2 := by
  88     simp only [hzero,ite_false,alternatingIndex_next_zero,pairSequenceEquiv_zero,pairSequenceEquiv_one]
  89   have ho (i : Fin (k+2)) :
  90       G.Adj (pairSequenceEquiv k V t (alternatingIndex i 1))
  91         (if (alternatingIndex i 1).val+1=(k+2)*2 then
  92           π (pairSequenceEquiv k V t (cycleNext (alternatingIndex i 1)))
  93         else pairSequenceEquiv k V t (cycleNext (alternatingIndex i 1))) ↔
  94       G.Adj (t i).2 (choosePermutation π (closingInterface k i) (t (finRotate (k+2) i)).1) := by
  95     simp only [hone,alternatingIndex_next_one,pairSequenceEquiv_zero,pairSequenceEquiv_one]
  96     by_cases hh : i.val=k+1 <;> simp [choosePermutation,closingInterface,hh]
  97   constructor
  98   · intro ht j
  99     obtain ⟨⟨i,b⟩,rfl⟩ := finProdFinEquiv.surjective j
 100     fin_cases b
 101     · exact (hz i).mpr (ht i).1
 102     · exact (ho i).mpr (ht i).2
 103   · intro ht i
 104     exact ⟨(hz i).mp (ht _),(ho i).mp (ht _)⟩

 117 theorem permutationSeparated_comp_equiv {I J V : Type*} (π : Equiv.Perm V) (b : ℕ)
 118     (e : I ≃ J) (f : J → V) :
 119     PermutationSeparated π b (fun i => f (e i)) ↔ PermutationSeparated π b f := by
 120   constructor
 121   · intro h i j hij
 122     have hne : e.symm i ≠ e.symm j := fun he => hij (e.symm.injective he)
 123     simpa only [e.apply_symm_apply] using h (e.symm i) (e.symm j) hne
 124   · intro h i j hij
 125     exact h (e i) (e j) (fun he => hij (e.injective he))

 133 theorem pairSequence_weight_incidence {V : Type*} [Fintype V] {k : ℕ}
 134     (w : (Fin (k+2) → V × V) → ℝ) (E : (Fin ((k+2)*2) → V) → Prop) [DecidablePred E] :
 135     (∑ f, if E f then w ((pairSequenceEquiv k V).symm f) else 0) =
 136       ∑ t, if E (pairSequenceEquiv k V t) then w t else 0 := by
 137   have h := Fintype.sum_equiv (pairSequenceEquiv k V)
 138     (fun t => if E (pairSequenceEquiv k V t) then w t else 0)
 139     (fun f => if E f then w ((pairSequenceEquiv k V).symm f) else 0)
 140     (fun t => by rw [Equiv.symm_apply_apply])
 141   exact h.symm
