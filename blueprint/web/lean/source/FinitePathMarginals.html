  11 def pathVertexLoad {X : Type*} [Fintype X] {k : ℕ}
  12     (w : (Fin (k+1) → X) → ℝ) (j : Fin (k+1)) (x : X) : ℝ :=
  13   ∑ p, if p j=x then w p else 0
  14 
  15 def pathEdgeLoad {X : Type*} [Fintype X] {k : ℕ}
  16     (w : (Fin (k+1) → X) → ℝ) (j : Fin k) (x y : X) : ℝ :=
  17   ∑ p, if p j.castSucc=x ∧ p j.succ=y then w p else 0

  19 theorem pathEdgeLoad_row {X : Type*} [Fintype X] {k : ℕ}
  20     (w : (Fin (k+1) → X) → ℝ) (j : Fin k) (x : X) :
  21     (∑ y, pathEdgeLoad w j x y) = pathVertexLoad w j.castSucc x := by
  22   unfold pathEdgeLoad pathVertexLoad
  23   rw [Finset.sum_comm]
  24   apply Finset.sum_congr rfl
  25   intro p _
  26   simp [ite_and]

  37 theorem pathEdgeLoad_nonneg {X : Type*} [Fintype X] {k : ℕ}
  38     (w : (Fin (k+1) → X) → ℝ) (hw : ∀ p, 0 ≤ w p)
  39     (j : Fin k) (x y : X) : 0 ≤ pathEdgeLoad w j x y := by
  40   apply Finset.sum_nonneg
  41   intro p _
  42   split_ifs <;> first | exact hw p | exact le_refl _

  70 theorem residual_path_capacities_balanced {X : Type*} [Fintype X] {k : ℕ}
  71     (a : Fin k → X → X → ℝ) (w : (Fin (k+1) → X) → ℝ)
  72     (j l : Fin k) (hjl : j.succ=l.castSucc)
  73     (hbal : ∀ y, (∑ x, a j x y)=∑ z, a l y z) :
  74     ∀ y, (∑ x, (a j x y-pathEdgeLoad w j x y)) =
  75       ∑ z, (a l y z-pathEdgeLoad w l y z) := by
  76   intro y
  77   rw [Finset.sum_sub_distrib,Finset.sum_sub_distrib,
  78     pathEdgeLoad_column,pathEdgeLoad_row,hjl,hbal]

  80 theorem residual_path_capacity_bounds {X : Type*} [Fintype X] {k : ℕ}
  81     (a : Fin k → X → X → ℝ) (w : (Fin (k+1) → X) → ℝ)
  82     (ha : ∀ j x y, 0 ≤ a j x y) (hw : ∀ p, 0 ≤ w p)
  83     (hreserve : ∀ j x y, pathEdgeLoad w j x y ≤ (11/27)*a j x y) :
  84     ∀ j x y, a j x y/2 ≤ a j x y-pathEdgeLoad w j x y ∧
  85       a j x y-pathEdgeLoad w j x y ≤ a j x y := by
  86   intro j x y
  87   have h0 := pathEdgeLoad_nonneg w hw j x y
  88   have h1 := hreserve j x y
  89   have h2 := ha j x y
