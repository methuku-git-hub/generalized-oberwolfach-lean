  15 theorem orientationMatrix_degree
  16     {V : Type*} [Fintype V] (G : SimpleGraph V) (M : V → V → ℕ)
  17     (hphysical : ∀ x y, M x y + M y x = if G.Adj x y then 1 else 0) (x : V) :
  18     (∑ y, M x y) + (∑ y, M y x) = G.degree x := by
  19   rw [← Finset.sum_add_distrib]
  20   simp only [hphysical]
  21   rw [← Finset.card_filter]
  22   congr 1
  23   ext y

  26 /-- An exact statement of the balanced-orientation claim used by the
  27 protected partition. The only hypotheses on the graph are even degrees
  28 and its minimum-degree bound. All concentration, routing, rounding and
  29 asymptotic choices are discharged. -/
  30 theorem balancedOrientation_highProbability_pow {α : ℝ} (hα : 0 < α) (q : ℕ) :
  31     ∃ N₀ : ℕ, ∀ (V : Type u) [Fintype V] [LinearOrder V],
  32       let n := Fintype.card V
  33       N₀ ≤ n → ∀ G : SimpleGraph V,
  34       (∀ v, Even (G.degree v)) →
  35       (∀ v, (1 / 2 + α) * n ≤ (G.degree v : ℝ)) →
  36       1 - 1 / (n : ℝ) ^ q ≤
  37         (FiniteLaw.pi (fun _ : Sym2 (V) => FiniteLaw.fairCoin)).probability (fun σ =>
  38           ∀ F : SimpleGraph (V), (∀ v, (F.degree v : ℝ) ≤ α * n / 100) →
  39           ∃ M : V → V → ℕ,
  40             (∀ x y, M x y + M y x = if G.Adj x y then 1 else 0) ∧
  41             (∀ v, (∑ w, M v w) = G.degree v / 2 ∧ (∑ w, M w v) = G.degree v / 2) ∧
  42             (∀ x y, M x y ≠ sampledOrientation G σ x y ∨
  43               M y x ≠ sampledOrientation G σ y x → ¬ F.Adj x y) ∧
  44             ∀ x, ((Finset.univ.filter (fun y => M x y ≠ sampledOrientation G σ x y ∨
  45               M y x ≠ sampledOrientation G σ y x)).card : ℝ) ≤ 4 * (n : ℝ) ^ (4 / 5 : ℝ)) := by
  46   obtain ⟨N₁, h₁⟩ := eventually_orientation_numeric hα
  47   obtain ⟨N₂, h₂⟩ := eventually_orientation_failure_small_pow hα q
  48   refine ⟨max N₁ N₂, fun V _ _ => ?_⟩
  49   intro n hn G heven hmin
  50   obtain ⟨hnpos, hL, hmargin, hchange⟩ := h₁ n (le_trans (le_max_left _ _) hn)
  51   have hnreal : (0 : ℝ) < n := by exact_mod_cast hnpos
  52   have hfailure := h₂ n (le_trans (le_max_right _ _) hn)
  53   let d (v : V) := G.degree v / 2
  54   have hd (v : V) : 2 * d v = G.degree v := by
  55     obtain ⟨k, hk⟩ := heven v
  56     dsimp [d]
  57     omega
  58   have hc := commonNeighbours_of_minDegree G hmin
  59   have hcommon (x y : V) (_hxy : x ≠ y) :
  60       8 * (α / 4) * Fintype.card (V) ≤
  61         ((G.neighborFinset x ∩ G.neighborFinset y).card : ℝ) := by
  62     have hh := hc x y
  63     linarith
  64   have hdegree (v : V) : 0 < G.degree v := by
  65     have hpos : (0 : ℝ) < (1 / 2 + α) * n := by positivity
  66     have hh := hpos.trans_le (hmin v)
  67     exact_mod_cast hh
  68   have hprob := sampledOrientation_balancing_event G G le_rfl d hd
  69     (Nat.ceil ((n : ℝ) ^ (4 / 5 : ℝ))) hL hnpos
  70     (a := (n : ℝ) ^ (3 / 5 : ℝ)) (f := α * n / 100) (β := α / 4)
  71     (by positivity) (by positivity) (by positivity) hcommon
  72     hmargin
  73   have hsum := orientation_failure_sum_bound G G hdegree
  74     (a := (n : ℝ) ^ (3 / 5 : ℝ)) (α := α) (fun x y _ => hc x y)
  75   change _ ≤ 2 * n * Real.exp (-2 * ((n : ℝ) ^ (3 / 5 : ℝ)) ^ 2 / n) +
  76     2 * (n : ℝ) ^ 2 * Real.exp (-α * n / 32) at hsum
  77   rw [threeFifths_tail_exponent hnreal] at hsum
  78   have hbase : 1 - 1 / (n : ℝ) ^ q ≤
  79       (FiniteLaw.pi (fun _ : Sym2 (V) => FiniteLaw.fairCoin)).probability (fun σ =>
  80         ∀ F : SimpleGraph (V), (∀ v, (F.degree v : ℝ) ≤ α * n / 100) →
  81         ∃ M : V → V → ℕ,
  82           (∀ x y, M x y + M y x = if G.Adj x y then 1 else 0) ∧
  83           (∀ v, ∑ w, M v w = d v) ∧
  84           (∀ x y, M x y ≠ sampledOrientation G σ x y ∨ M y x ≠ sampledOrientation G σ y x →
  85             G.Adj x y ∧ ¬ F.Adj x y) ∧
  86           ∀ x, ((Finset.univ.filter (fun y => M x y ≠ sampledOrientation G σ x y ∨
  87             M y x ≠ sampledOrientation G σ y x)).card : ℝ) ≤
  88               (n : ℝ) ^ (3 / 5 : ℝ) + 2 * Nat.ceil ((n : ℝ) ^ (4 / 5 : ℝ))) := by
  89     apply le_trans ?_ hprob
  90     linarith [hsum, hfailure]
  91   apply hbase.trans
  92   apply finiteLaw_probability_mono

  94   obtain ⟨M, hphys, hout, havoid, hsize⟩ := hσ F hF
  95   refine ⟨M, hphys, ?_, fun x y h => (havoid x y h).2,
  96     fun x => (hsize x).trans hchange⟩
  97   intro v
  98   have hh := orientationMatrix_degree G M hphys v
  99   rw [hout v] at hh
