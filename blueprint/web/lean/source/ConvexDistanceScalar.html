  13 theorem convexDistance_log_bound {r : ℝ} (hr0 : 1/2≤r) (hr1 : r≤1) :
  14     -(2*r-1)*Real.log r+(1-r)^2≤Real.log (2-r) := by
  15   let f : ℝ → ℝ := fun x=>Real.log (2-x)+(2*x-1)*Real.log x-(1-x)^2
  16   have hd (x : ℝ) (hx : x∈Set.Icc (1/2 : ℝ) 1) :
  17       HasDerivAt f (-(2-x)⁻¹+2*Real.log x+(2*x-1)*x⁻¹+2*(1-x)) x := by
  18     have hx0 : x≠0 := by linarith [hx.1]
  19     have hx2 : 2-x≠0 := by linarith [hx.2]
  20     convert (((((hasDerivAt_const x 2).sub (hasDerivAt_id x)).log hx2).add
  21       ((((hasDerivAt_const x 2).mul (hasDerivAt_id x)).sub (hasDerivAt_const x 1)).mul
  22         ((hasDerivAt_id x).log hx0))).sub
  23       (((hasDerivAt_const x 1).sub (hasDerivAt_id x)).pow 2)) using 1 <;>
  24       dsimp [f] <;> try ring
  25     all_goals
  26       funext y
  27       simp only [Pi.add_apply,Pi.mul_apply,Pi.sub_apply,Pi.pow_apply,Pi.ofNat_apply,id_eq]
  28       ring
  29   have hm : AntitoneOn f (Set.Icc (1/2 : ℝ) 1) := by
  30     apply antitoneOn_of_deriv_nonpos (convex_Icc _ _)
  31     · intro x hx
  32       exact (hd x hx).continuousAt.continuousWithinAt
  33     · intro x hx
  34       exact (hd x (interior_subset hx)).differentiableAt.differentiableWithinAt
  35     · intro x hx
  36       have hx' := interior_subset hx
  37       rw [(hd x hx').deriv]
  38       have hx0 : 0<x := by linarith [hx'.1]
  39       have hx2 : 0<2-x := by linarith [hx'.2]
  40       have hl := Real.log_le_sub_one_of_pos hx0
  41       have he : -(2-x)⁻¹+(2*x-1)*x⁻¹ =
  42           -2*(x-1)^2/(x*(2-x)) := by
  43         field_simp
  44         <;> ring
  45       have hn : -2*(x-1)^2/(x*(2-x))≤0 :=
  46         div_nonpos_of_nonpos_of_nonneg (by nlinarith [sq_nonneg (x-1)]) (by positivity)
  47       rw [← he] at hn
  48       linarith
  49   have hh := hm ⟨hr0,hr1⟩ (show (1 : ℝ)∈Set.Icc (1/2) 1 by constructor <;> norm_num) hr1
  50   dsimp [f] at hh
  51   norm_num at hh
  52   linarith
