  13 theorem boundary_graph_degree (G:SimpleGraph W) (Ω:Finset W)
  14     (x:I↪W) (A:I→Finset W)
  15     (hout:∀ z∉Ω,∀ v,G.Adj z v↔∃ i,x i=z ∧ v∈A i)
  16     (d k:ℕ) (hinner:∀ v,(spanningRestriction G Ω).degree v≤d)
  17     (hmem:∀ v,(Finset.univ.filter (fun i=>v∈A i)).card≤k)
  18     (v:W) (hv:v∈Ω):G.degree v≤d+k:=by
  19   have hsub:G.neighborFinset v⊆(spanningRestriction G Ω).neighborFinset v∪
  20       (Finset.univ.filter (fun i=>v∈A i)).image x:=by
  21     intro z hz
  22     have hz':G.Adj v z:=(G.mem_neighborFinset v z).mp hz
  23     by_cases hΩ:z∈Ω
  24     · exact Finset.mem_union_left _ (((spanningRestriction G Ω).mem_neighborFinset v z).mpr
  25         ((spanningRestriction_adj G Ω v z).mpr ⟨hz',hv,hΩ⟩))
  26     · obtain ⟨i,hi,ha⟩:=(hout z hΩ v).mp hz'.symm
  27       exact Finset.mem_union_right _ (Finset.mem_image.mpr ⟨i,by simp [ha],hi⟩)
  28   calc
  29     G.degree v= (G.neighborFinset v).card:=rfl
  30     _≤((spanningRestriction G Ω).neighborFinset v∪
  31         (Finset.univ.filter (fun i=>v∈A i)).image x).card:=Finset.card_le_card hsub
  32     _≤((spanningRestriction G Ω).neighborFinset v).card+
  33         ((Finset.univ.filter (fun i=>v∈A i)).image x).card:=Finset.card_union_le _ _
  34     _≤d+k:=Nat.add_le_add (hinner v) (Finset.card_image_le.trans (hmem v))

  36 theorem boundary_graph_root_degree (G:SimpleGraph W) (Ω:Finset W)
  37     (x:I↪W) (A:I→Finset W) (hx:∀ i,x i∉Ω)
  38     (hout:∀ z∉Ω,∀ v,G.Adj z v↔∃ i,x i=z ∧ v∈A i)
  39     (i:I):G.degree (x i)=(A i).card:=by
  40   change (G.neighborFinset (x i)).card=(A i).card
  41   congr 1
  42   ext v
  43   rw [SimpleGraph.mem_neighborFinset,hout (x i) (hx i)]
  44   constructor
  45   · rintro ⟨j,hj,hv⟩
  46     exact x.injective hj ▸ hv
  47   · intro hv;exact ⟨i,rfl,hv⟩
