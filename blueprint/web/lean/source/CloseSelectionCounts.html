  14 theorem bad_fixed_selections_card {I X : Type*}
  15     [Fintype I] [Fintype X] [DecidableEq I] [Nonempty X]
  16     (R : X → X → Prop) (hR : ∀ x y, R x y → R y x)
  17     (q : ℕ) (hdegree : ∀ x, Fintype.card {y // R x y} ≤ q)
  18     (A : Finset I) (a : I → X)
  19     (hsep : ∀ i ∈ A, ∀ j ∈ A, i ≠ j → ¬ R (a i) (a j)) :
  20     Fintype.card {f : I → X // (∀ i ∈ A, f i = a i) ∧
  21       ∃ i j, i ≠ j ∧ R (f i) (f j)} ≤
  22       Fintype.card I ^ 2 * q * Fintype.card X ^ (Fintype.card I - A.card - 1) := by
  23   let C (p : I × I) := Finset.univ.filter (fun f : I → X =>
  24     (∀ i ∈ A, f i = a i) ∧ p.1 ∉ A ∧ p.1 ≠ p.2 ∧ R (f p.2) (f p.1))
  25   have hsub : Finset.univ.filter (fun f : I → X =>
  26       (∀ i ∈ A, f i = a i) ∧ ∃ i j, i ≠ j ∧ R (f i) (f j)) ⊆
  27       Finset.univ.biUnion C := by
  28     intro f hf
  29     obtain ⟨hf, i, j, hij, hrel⟩ := (Finset.mem_filter.mp hf).2
  30     by_cases hi : i ∈ A
  31     · have hj : j ∉ A := by
  32         intro hj
  33         apply hsep i hi j hj hij
  34         simpa only [hf i hi, hf j hj] using hrel
  35       exact Finset.mem_biUnion.mpr ⟨(j, i), Finset.mem_univ _,
  36         Finset.mem_filter.mpr ⟨Finset.mem_univ _, hf, hj, hij.symm, hrel⟩⟩
  37     · exact Finset.mem_biUnion.mpr ⟨(i, j), Finset.mem_univ _,
  38         Finset.mem_filter.mpr ⟨Finset.mem_univ _, hf, hi, hij, hR _ _ hrel⟩⟩
  39   have hC (p : I × I) : (C p).card ≤
  40       q * Fintype.card X ^ (Fintype.card I - A.card - 1) := by
  41     by_cases hi : p.1 ∈ A
  42     · simp [C, hi]
  43     by_cases hij : p.1 = p.2
  44     · simp [C, hij]
  45     have heq : (C p).card = Fintype.card {f : I → X //
  46         (∀ i ∈ A, f i = a i) ∧ R (f p.2) (f p.1)} := by
  47       simp only [C, ne_eq, hi, hij, not_false_eq_true, true_and, and_true, Fintype.card_subtype]
  48     rw [heq]
  49     exact related_fixed_selections_card R q hdegree A a p.1 p.2 hi hij
  50   calc
  51     _ ≤ (Finset.univ.biUnion C).card := by
  52       rw [Fintype.card_subtype]
  53       exact Finset.card_le_card hsub
  54     _ ≤ ∑ p : I × I, (C p).card := Finset.card_biUnion_le
  55     _ ≤ ∑ _p : I × I, q * Fintype.card X ^ (Fintype.card I - A.card - 1) :=
  56       Finset.sum_le_sum (fun p _ => hC p)
  57     _ = _ := by simp only [Finset.sum_const, Finset.card_univ, nsmul_eq_mul,
  58       Fintype.card_prod, Nat.cast_id]; ring
