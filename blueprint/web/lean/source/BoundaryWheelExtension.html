  15 theorem boundary_wheel_after_deletion (Q D F:SimpleGraph V) (H E:V→U→Prop)
  16     (A W Ω:Finset V) (hAΩ:A⊆Ω) (x v w:V) (u:U) (hvΩ:v∈Ω) (hwΩ:w∈Ω)
  17     (hvw:v≠w) (hxv:Q.Adj x v) (hxw:Q.Adj x w)
  18     (hDxv:¬D.Adj x v) (hDxw:¬D.Adj x w)
  19     (hFroot:∀ a∈({x,v,w}:Finset V),∀ b∈({x,v,w}:Finset V),¬F.Adj a b)
  20     (hHroot:H x u∧H v u∧H w u) (hEroot:¬E x u∧¬E v u∧¬E w u)
  21     (hHA:∀ a∈A,H a u) (r:ℕ) (hr:4≤r) (L d e b:ℝ) (hL:0≤L)
  22     (hF:∀ a,(F.degree a:ℝ)≤L) (hD:∀ a∈Ω,(D.degree a:ℝ)≤d)
  23     (hE:((Finset.univ.filter (fun a=>E a u)).card:ℝ)≤e)
  24     (hcount:∀ a∈Ω,∀ c∈Ω,b≤((A∩Q.neighborFinset a∩Q.neighborFinset c).card:ℝ))
  25     (hbudget:W.card+((r:ℝ)+3)*(L+1)+2*d+e<b):
  26     ∃ C:OrdinaryWheel Q F H r,ordinaryWheelHub C=u ∧
  27       (ordinaryWheelRim C).Adj x v ∧ (ordinaryWheelRim C).Adj x w ∧
  28       Disjoint (ordinaryWheelResources C) (ordinaryHostResources D E) ∧
  29       (∀ a∈(ordinaryWheelRim C).support,a∈({x,v,w}:Finset V) ∨ a∈A∧a∉W):=by
  30   let Z:=Finset.univ.filter (fun a=>E a u)
  31   let A₀:=A\Z
  32   have hcounts (a:V) (ha:a∈Ω) (c:V) (hc:c∈Ω):
  33       b-2*d-e≤((A₀∩(Q\D).neighborFinset a∩(Q\D).neighborFinset c).card:ℝ):=by
  34     have hdel:=two_common_neighbours_after_deletion Q D A a c
  35     have hrem:=Finset.card_le_card_sdiff_add_card
  36       (s:=A∩(Q\D).neighborFinset a∩(Q\D).neighborFinset c) (t:=Z)
  37     have he:(A∩(Q\D).neighborFinset a∩(Q\D).neighborFinset c)\Z=
  38         A₀∩(Q\D).neighborFinset a∩(Q\D).neighborFinset c:=by
  39       ext z;simp only [A₀,Finset.mem_sdiff,Finset.mem_inter];tauto
  40     rw [he] at hrem
  41     have hd':((A∩Q.neighborFinset a∩Q.neighborFinset c).card:ℝ)≤
  42       (A∩(Q\D).neighborFinset a∩(Q\D).neighborFinset c).card+D.degree a+D.degree c:=by exact_mod_cast hdel
  43     have hr':((A∩(Q\D).neighborFinset a∩(Q\D).neighborFinset c).card:ℝ)≤
  44       (A₀∩(Q\D).neighborFinset a∩(Q\D).neighborFinset c).card+Z.card:=by exact_mod_cast hrem
  45     linarith only [hd',hr',hD a ha,hD c hc,hE,hcount a ha c hc]
  46   obtain ⟨C,hCu,hCv,hCw,hCs⟩:=ordinary_root_wheel_from_common_counts_within (Q\D) F H
  47     A₀ W Ω (Finset.sdiff_subset.trans hAΩ) x v w hvΩ hwΩ u hvw
  48     ⟨hxv,hDxv⟩ ⟨hxw,hDxw⟩ hFroot hHroot (fun a ha=>hHA a (Finset.mem_sdiff.mp ha).1)
  49     r hr L (b-2*d-e) hL hF
  50     (by intro a ha c hc;convert hcounts a ha c hc using 1;congr 2;ext z;simp only [Finset.mem_inter,SimpleGraph.mem_neighborFinset])
  51     (by linarith only [hbudget])
  52   let C':OrdinaryWheel Q F H r:=ordinaryWheelInclude sdiff_le C
  53   have hspoke (a:V) (ha:a∈(ordinaryWheelRim C').support):¬E a u:=by
  54     rcases hCs a ha with hroot|hfree
  55     · simp only [Finset.mem_insert,Finset.mem_singleton] at hroot
  56       rcases hroot with rfl|rfl|rfl
  57       · exact hEroot.1
  58       · exact hEroot.2.1
  59       · exact hEroot.2.2
  60     · have hn: a∉Z:=(Finset.mem_sdiff.mp hfree.1).2
  61       simpa only [Z,Finset.mem_filter,Finset.mem_univ,true_and] using hn
  62   refine ⟨C',hCu,hCv,hCw,?_,?_⟩
  63   · apply Finset.disjoint_left.mpr
  64     intro z hz hzD
  65     cases z with
  66     | inl edge=>
  67       rw [ordinaryWheelResources_rim] at hz
  68       rw [ordinaryHostResources_rim] at hzD
  69       induction edge using Sym2.ind with
  70       | _ a c=>exact ((C.val.1.property hz).2 hzD)
  71     | inr au=>
  72       rcases au with ⟨a,u'⟩
  73       rw [ordinaryWheelResources_spoke] at hz
  74       rw [ordinaryHostResources_spoke] at hzD
  75       have he:u'=u:=hz.1.symm.trans hCu
  76       exact hspoke a hz.2 (he ▸ hzD)
  77   · intro a ha
  78     rcases hCs a ha with hroot|hfree
  79     · exact Or.inl hroot
  80     · exact Or.inr ⟨(Finset.mem_sdiff.mp hfree.1).1,hfree.2⟩
