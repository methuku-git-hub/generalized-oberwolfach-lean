  16 theorem independentSet_disjoint_neighbours {V : Type*} [Fintype V]
  17     (G : SimpleGraph V) (S : Finset V)
  18     (hS : ∀ u∈S,∀ v∈S,¬G.Adj u v) : Disjoint S (setNeighbours G S) := by
  19   apply Finset.disjoint_left.mpr
  20   intro u hu hn
  21   obtain ⟨v,hv,hvu⟩ := (Finset.mem_filter.mp hn).2

  24 theorem joint_coinRetention_probability
  25     {V C : Type*} [Fintype V] [DecidableEq V] [Fintype C]
  26     (G : SimpleGraph V) (μ : V → FiniteLaw C)
  27     (a : V → C → ℝ) (ha : ∀ v c,0≤a v c ∧ a v c≤1)
  28     (S : Finset V) (hS : ∀ u∈S,∀ v∈S,¬G.Adj u v) (c : C) :
  29     (FiniteLaw.pi (fun u=>(μ u).withCoin (a u) (ha u))).probability
  30       (fun σ=>∀ v∈S,coinRetainedColour G σ v=some c)=
  31       (∏ v∈S,(μ v).mass c*a v c)*
  32         ∏ u∈setNeighbours G S,(1-(μ u).mass c) := by
  33   classical
  34   let ν := fun u=>(μ u).withCoin (a u) (ha u)
  35   let N := setNeighbours G S
  36   have hdis : Disjoint S N := independentSet_disjoint_neighbours G S hS
  37   let A : V → C×Bool → Prop := fun u x=>
  38     if u∈S then x.1=c ∧ x.2=true else if u∈N then x.1≠c else True
  39   have hevent (σ : V → C×Bool) :
  40       (∀ v∈S,coinRetainedColour G σ v=some c) ↔ ∀ u,A u (σ u) := by
  41     constructor
  42     · intro h u
  43       by_cases hu : u∈S
  44       · obtain ⟨hc,hcoin,_⟩ := (coinRetainedColour_eq_some G σ u c).mp (h u hu)
  45         simp [A,hu,hc,hcoin]
  46       · by_cases hn : u∈N
  47         · obtain ⟨v,hv,hvu⟩ := (Finset.mem_filter.mp hn).2
  48           obtain ⟨hc,_,hkeep⟩ := (coinRetainedColour_eq_some G σ v c).mp (h v hv)
  49           simpa [A,hu,hn,hc] using hkeep u hvu
  50         · simp [A,hu,hn]
  51     · intro h v hv
  52       obtain ⟨hc,hcoin⟩ : (σ v).1=c ∧ (σ v).2=true := by simpa [A,hv] using h v
  53       apply (coinRetainedColour_eq_some G σ v c).mpr
  54       refine ⟨hc,hcoin,?_⟩
  55       intro u hadj
  56       have hn : u∈N := Finset.mem_filter.mpr ⟨Finset.mem_univ _,v,hv,hadj⟩
  57       have hu : u∉S := fun hu=>Finset.disjoint_left.mp hdis hu hn
  58       simpa [A,hu,hn,hc] using h u
  59   rw [finiteLaw_probability_congr (FiniteLaw.pi ν) _ _ hevent,
  60     finiteLaw_pi_probability_forall ν A]
  61   have hcoord (u : V) : (ν u).probability (A u)=
  62       (if u∈S then (μ u).mass c*a u c else 1)*
  63         (if u∈N then 1-(μ u).mass c else 1) := by
  64     by_cases hs : u∈S
  65     · have hn : u∉N := fun hn=>Finset.disjoint_left.mp hdis hs hn
  66       simp only [A,hs,ite_true,hn,ite_false,mul_one]
  67       exact finiteLaw_withCoin_success _ _ _ _
  68     · by_cases hn : u∈N
  69       · simp only [A,hs,ite_false,hn,ite_true,one_mul]
  70         exact finiteLaw_withCoin_not_colour _ _ _ _
  71       · simp only [A,hs,hn,ite_false,one_mul]
  72         exact finiteLaw_probability_true _
  73   simp_rw [hcoord]
  74   rw [Finset.prod_mul_distrib]
  75   simp only [Finset.prod_ite_mem,Finset.univ_inter]
  76   rfl
