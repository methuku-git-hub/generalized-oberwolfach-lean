  17  theorem marked_same_direction_physical_injective (π:Equiv.Perm V) (j o:Bool)
  18     (v w z:V) (hne:v≠w):
  19     Set.InjOn (fun a=>markedPairPhysical π (markedRowRim a z)) (markedSameDirectionRequests j o v w):=by
  20   have hp:markedPairPhysical π (markedRowRim (j,o,v) z)≠
  21       markedPairPhysical π (markedRowRim (j,o,w) z):=by
  22     intro he
  23     cases j <;> cases o
  24     · exact hne (Sym2.congr_left.mp he)
  25     · exact hne (Sym2.congr_right.mp he)
  26     · exact hne (Sym2.congr_left.mp he)
  27     · exact hne (π.injective (Sym2.congr_right.mp he))
  28   intro a ha b hb he
  29   simp only [markedSameDirectionRequests,Finset.mem_coe,Finset.mem_insert,Finset.mem_singleton] at ha hb
  30   rcases ha with rfl|rfl <;> rcases hb with rfl|rfl
  31   · rfl
  32   · exact (hp he).elim
  33   · exact (hp he.symm).elim
  34   · rfl

  36  theorem marked_same_direction_domain_lower (P:SimpleGraph V) (π:Equiv.Perm V) (b:ℕ)
  37     (A:Finset V) (hA:A.image π=A) (j o:Bool) (v w:V) (hne:v≠w)
  38     (γ:ℝ) (hγ1:γ≤1)
  39     (hdeg:∀ x,(1/2+γ/2)*A.card≤((A∩P.neighborFinset x).card:ℝ)):
  40     γ*A.card-2*(2*(b:ℝ)+1)≤
  41       ((markedRowDomain P π b A (markedSameDirectionRequests j o v w) ∅).card:ℝ):=by
  42   let τ:Equiv.Perm V:=if j && !o then π else Equiv.refl V
  43   let f:V→V:=fun x=>if j && o then π x else x
  44   have hτ:A.image τ=A:=by cases j <;> cases o <;> simp [τ,hA]
  45   have hR:({f v,f w}:Finset V).card≤2:=(Finset.card_insert_le _ _).trans (by simp)
  46   have hc:=permutedNeighbourChoices_card_lower P π τ b A {f v,f w} {v,w} γ hγ1 hR
  47     (by intro x hx;rw [hτ,Finset.inter_comm];exact hdeg x)
  48   have hsub:permutedNeighbourChoices P π τ b A {f v,f w} {v,w}⊆
  49       markedRowDomain P π b A (markedSameDirectionRequests j o v w) ∅:=by
  50     intro z hz
  51     obtain ⟨hzA,hzP,hzsep⟩:=Finset.mem_filter.mp hz
  52     have hvP:=hzP (f v) (by simp)
  53     have hwP:=hzP (f w) (by simp)
  54     have hvsep:=hzsep v (by simp)
  55     have hwsep:=hzsep w (by simp)
  56     apply Finset.mem_filter.mpr
  57     refine ⟨hzA,?_,marked_same_direction_physical_injective π j o v w z hne,by simp⟩
  58     intro a ha
  59     simp only [markedSameDirectionRequests,Finset.mem_insert,Finset.mem_singleton] at ha
  60     rcases ha with rfl|rfl
  61     · cases j <;> cases o <;>
  62         simpa [markedRowEligible,markedRowRim,τ,f] using
  63           And.intro hvsep (by first | exact hvP | exact hvP.symm)
  64     · cases j <;> cases o <;>
  65         simpa [markedRowEligible,markedRowRim,τ,f] using
  66           And.intro hwsep (by first | exact hwP | exact hwP.symm)
  67   have hpair:(({v,w}:Finset V).card:ℝ)≤2:=by
  68     exact_mod_cast (Finset.card_insert_le v ({w}:Finset V)).trans (by simp)
  69   have hloss:=mul_le_mul_of_nonneg_right hpair (by positivity : (0:ℝ)≤2*b+1)
  70   have hcard:((permutedNeighbourChoices P π τ b A {f v,f w} {v,w}).card:ℝ)≤
  71       (markedRowDomain P π b A (markedSameDirectionRequests j o v w) ∅).card:=by
  72     exact_mod_cast Finset.card_le_card hsub
  73   linarith only [hc,hloss,hcard]
