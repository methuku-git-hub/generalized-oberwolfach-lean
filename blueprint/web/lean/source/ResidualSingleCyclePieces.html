   2 /-! Represent a retained part of an ordered source cycle by directed source
   3 pieces: the cycle itself if intact, and directed paths if cut. -/
   4 set_option autoImplicit false
   5 noncomputable section
   6 namespace Oberwolfach
   7 attribute [local instance] Classical.propDecidable
   8 variable {V:Type} [Fintype V] [DecidableEq V]

  10 theorem retained_cycle_source_pieces (H:SimpleGraph V) (R:Finset V)
  11     (hisol:∀x,x∉R→∀y,¬H.Adj x y) (D:V→V→Prop)
  12     (a:V) (xs:List V) (hn:(a::xs).Nodup) (hlen:3≤(a::xs).length)
  13     (hchain:List.IsChain D ((a::xs)++[a])):
  14     ∃ ps:List (PrescribedSourcePiece V),
  15       ps.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset) ∧
  16       (∀p∈ps,p.OriginalValid D) ∧
  17       (∀x,x∈sourcePieceVerticesUnion ps ↔ x∈a::xs ∧ x∈R) ∧
  18       sourcePieceEdgesUnion ps=(prescribedOrderedPathEdges ((a::xs)++[a])).filter (fun e=>e∈H.edgeFinset) := by
  19   by_cases hkeep:∀e∈prescribedOrderedPathEdges ((a::xs)++[a]),e∈H.edgeFinset
  20   · have hsurv: (a::xs).toFinset⊆R := intact_cycle_vertices_survive H R hisol a xs (by omega) hkeep
  21     refine ⟨[.cycle a xs],by simp,?_,?_,?_⟩
  22     · intro p hp
  23       have hp':p=.cycle a xs := by simpa using hp
  24       subst p
  25       exact ⟨hn,hlen,hchain⟩
  26     · intro x
  27       have heq:x∈sourcePieceVerticesUnion [.cycle a xs] ↔ x∈a::xs := by
  28         simp [sourcePieceVerticesUnion,PrescribedSourcePiece.vertices]
  29       rw [heq]
  30       constructor
  31       · intro hx
  32         exact ⟨hx,hsurv (List.mem_toFinset.mpr hx)⟩
  33       · exact And.left
  34     · have heq:sourcePieceEdgesUnion [.cycle a xs]=prescribedOrderedPathEdges ((a::xs)++[a]) := by
  35         simp [sourcePieceEdgesUnion,PrescribedSourcePiece.edges]
  36       rw [heq]
  37       exact (Finset.filter_true_of_mem hkeep).symm
  38   · push_neg at hkeep
  39     obtain ⟨e,he,hmissing⟩:=hkeep
  40     obtain ⟨blocks,hpair,hblocks,hvertices,hedges⟩:=cut_cycle_into_retained_paths H R hisol D a xs hn hchain e he hmissing
  41     obtain ⟨ps,hp,hvalid,hpsverts,hpsedges⟩:=path_blocks_as_source_pieces D blocks hpair
  42       (by intro block hb;exact ⟨(hblocks block hb).1,(hblocks block hb).2.2⟩)
  43     refine ⟨ps,hp,hvalid,?_,?_⟩
  44     · intro x
  45       exact (hpsverts x).trans (hvertices x)
  46     · exact hpsedges.trans hedges
  47 end Oberwolfach
