  17 def markedDegreeEndpoints (π : Equiv.Perm V) (mark incoming : Bool) (x y : V) : V×V :=
  18   if incoming then (y,if mark then π x else x) else (x,if mark then π y else y)

  20 def markedDegreeDomain (P : SimpleGraph V) (π : Equiv.Perm V) (b : ℕ)
  21     (mark incoming : Bool) (x : V) (A : Finset V) : Finset V :=
  22   A.filter (fun y=>¬PermutationClose π b x y ∧
  23     P.Adj (markedDegreeEndpoints π mark incoming x y).1 (markedDegreeEndpoints π mark incoming x y).2)

  25 def markedRestrictedDegree (P : SimpleGraph V) (π : Equiv.Perm V) (b : ℕ)
  26     (σ : Sym2 V→Option (Bool×Bool)) (mark incoming : Bool) (x : V) (A : Finset V) : ℕ :=
  27   ∑ y∈A,markedPhysicalRim P π b σ mark (if incoming then y else x) (if incoming then x else y)

  29 theorem markedDegreeEdge_injective (π : Equiv.Perm V) (mark incoming : Bool) (x : V) :
  30     Function.Injective (fun y=>s((markedDegreeEndpoints π mark incoming x y).1,
  31       (markedDegreeEndpoints π mark incoming x y).2)) := by
  32   cases mark <;> cases incoming
  33   · exact incidentPair_injective x
  34   · intro y z h
  35     change s(y,x)=s(z,x) at h
  36     exact incidentPair_injective x (Sym2.eq_swap.trans (h.trans Sym2.eq_swap))
  37   · intro y z h
  38     exact π.injective (incidentPair_injective x h)
  39   · intro y z h
  40     change s(y,π x)=s(z,π x) at h
  41     exact incidentPair_injective (π x) (Sym2.eq_swap.trans (h.trans Sym2.eq_swap))

  58 theorem markedRestrictedDegree_tail (P : SimpleGraph V) (π : Equiv.Perm V) (b : ℕ)
  59     (p k : ℝ) (hp : 0≤p) (hk : 0≤k) (hsum : 2*p+2*k≤1)
  60     (mark incoming : Bool) (x : V) (A : Finset V) (t : ℝ) (ht : 0≤t) :
  61     (FiniteLaw.pi (fun _ : Sym2 V=>markedPhysicalLaw p k hp hk hsum)).probability
  62       (fun σ=>t≤|(markedRestrictedDegree P π b σ mark incoming x A:ℝ)-
  63         (if mark then k else p)*(markedDegreeDomain P π b mark incoming x A).card|)≤
  64       2*Real.exp (-2*t^2/A.card) := by
  65   let ends := markedDegreeEndpoints π mark incoming x
  66   have hh := finiteLaw_singleton_patterns_tail_at_scale
  67     (fun _ : Sym2 V=>markedPhysicalLaw p k hp hk hsum)
  68     (markedDegreeDomain P π b mark incoming x A) (fun y=>s((ends y).1,(ends y).2))
  69     (fun y _=>some (mark,orientationBit (ends y).1 (ends y).2))
  70     (fun y hy z hz h=>markedDegreeEdge_injective π mark incoming x h)
  71     (n:=(A.card:ℝ)) (by exact_mod_cast Finset.card_filter_le A _) ht
  72   have hmass (y : V) : (markedPhysicalLaw p k hp hk hsum).mass
  73       (some (mark,orientationBit (ends y).1 (ends y).2))=if mark then k else p := by cases mark <;> rfl
  74   simp only [hmass,Finset.sum_const,nsmul_eq_mul] at hh
  75   dsimp [ends] at hh
