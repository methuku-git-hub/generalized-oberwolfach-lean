  37 theorem rawIntervalHubNeighbourhood_common_event
  38     (hnoshort : ∀ x k,0<k → k<2*d+1 → (π^k) x≠x)
  39     (hn : 0<Fintype.card V) (P : SimpleGraph V) (j : Color) (pH : ℝ) (hj : ∀ u,s u j=pH)
  40     {I : Type} [Fintype I] (J : I → Finset U)
  41     (hJ : ∀ i,(J i).Nonempty) (k : ℕ) (hk : 0<k) (hsize : ∀ i,(J i).card≤k)
  42     (t : ℝ) (ht : 0<t) :
  43     1-((Fintype.card I:ℝ)*(Fintype.card V+1))*
  44       (2*Real.exp (-2*t^2/((4*(d:ℝ)+3)^2*((k:ℝ)*Fintype.card V))))≤
  45       (FiniteLaw.pi μfamily).probability (fun σ=>∀ i,
  46         let T := rawIntervalHubNeighbourhood π d σ j (J i)
  47         (|(T.card:ℝ)-pH^(J i).card*Fintype.card V|<t) ∧
  48         (∀ v,|((P.neighborFinset v∩T).card:ℝ)-pH^(J i).card*P.degree v|<t)) := by
  49   let A : I×Option V → Finset V := fun a=>match a.2 with
  50     | none=>Finset.univ
  51     | some v=>P.neighborFinset v
  52   let JJ : I×Option V → Finset U := fun a=>J a.1
  53   have hh := rawIntervalNeighbourCount_common_event π d hd q hq hQ s hs hS hnoshort hn j A JJ
  54     (fun a=>hJ a.1) k hk (fun a=>hsize a.1) t ht
  55   have hcard : (Fintype.card (I×Option V):ℝ)=(Fintype.card I:ℝ)*(Fintype.card V+1) := by
  56     simp only [Fintype.card_prod,Fintype.card_option,Nat.cast_mul,Nat.cast_add,Nat.cast_one]
  57   rw [hcard] at hh
  58   apply hh.trans
  59   apply finiteLaw_probability_mono
  60   intro σ hσ i
  61   have hprod : (∏ u∈J i,s u j)=pH^(J i).card := by simp only [hj,Finset.prod_const]
  62   constructor
  63   · have hi := hσ (i,none)
