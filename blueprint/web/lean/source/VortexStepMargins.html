  11 theorem eventually_vortex_step_margins {ε : ℝ} (he : 0 < ε) :
  12     ∃ N : ℕ, ∀ n : ℕ, N ≤ n →
  13       2 ≤ n ∧ 2 ≤ ε * n ∧
  14       (4 * (n : ℝ) / ε) * Real.exp (-ε * (n : ℝ) ^ (1 / 3 : ℝ)) < 1 := by

  16     (c := ε) (p := 1 / 3) he (by norm_num) 1
  17   refine ⟨max 2 (max N (Nat.ceil (2 / ε))), fun n hn => ?_⟩
  18   have hn2 : 2 ≤ n := by omega
  19   have hnreal : (1 : ℝ) < n := by exact_mod_cast (show 1 < n by omega)
  20   have hceil : (Nat.ceil (2 / ε) : ℝ) ≤ n := by exact_mod_cast (show Nat.ceil (2 / ε) ≤ n by omega)
  21   have he2 : 2 ≤ ε * n := by
  22     have hh := (Nat.le_ceil (2 / ε)).trans hceil
  23     have hh' := (div_le_iff₀ he).mp hh
  24     nlinarith
  25   refine ⟨hn2, he2, ?_⟩
  26   have hh := hN n (by omega)
  27   simp only [Real.rpow_one, pow_one] at hh
  28   have hsame : (4 * (n : ℝ) / ε) * Real.exp (-ε * (n : ℝ) ^ (1 / 3 : ℝ)) =
  29       (4 / ε) * n * Real.exp (-ε * (n : ℝ) ^ (1 / 3 : ℝ)) := by ring
