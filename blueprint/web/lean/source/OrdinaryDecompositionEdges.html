  15 def ordinaryHostResources (Q : SimpleGraph V) (H : V → U → Prop) : Finset (Sym2 V ⊕ (V×U)) :=
  16   Finset.univ.filter (fun e => ordinaryResourceState Q H e=true)

  18 theorem ordinaryHostResources_rim (Q : SimpleGraph V) (H : V → U → Prop) (e : Sym2 V) :
  19     Sum.inl e∈ordinaryHostResources Q H ↔ e∈Q.edgeSet := by
  20   simp [ordinaryHostResources,ordinaryResourceState]

  22 theorem ordinaryHostResources_spoke (Q : SimpleGraph V) (H : V → U → Prop) (v : V) (u : U) :
  23     Sum.inr (v,u)∈ordinaryHostResources Q H ↔ H v u := by
  24   simp [ordinaryHostResources,ordinaryResourceState]

  26 theorem ordinaryHostResources_union (Q P : SimpleGraph V) (H K : V → U → Prop) :
  27     ordinaryHostResources (Q⊔P) (fun v u => H v u ∨ K v u)=ordinaryHostResources Q H ∪ ordinaryHostResources P K := by
  28   ext e
  29   cases e with
  30   | inl e => simp only [Finset.mem_union,ordinaryHostResources_rim,SimpleGraph.edgeSet_sup,Set.mem_union]
  31   | inr vu => rcases vu with ⟨v,u⟩; simp only [Finset.mem_union,ordinaryHostResources_spoke]

  33 theorem ordinaryHostResources_sdiff (Q P : SimpleGraph V) (H K : V → U → Prop) :
  34     ordinaryHostResources (Q\P) (fun v u => H v u ∧ ¬K v u)=ordinaryHostResources Q H \ ordinaryHostResources P K := by
  35   ext e
  36   cases e with
  37   | inl e => simp only [Finset.mem_sdiff,ordinaryHostResources_rim,SimpleGraph.edgeSet_sdiff,Set.mem_diff]
  38   | inr vu => rcases vu with ⟨v,u⟩; simp only [Finset.mem_sdiff,ordinaryHostResources_spoke]

  40 theorem ordinaryHostResources_injective (Q P : SimpleGraph V) (H K : V → U → Prop)
  41     (h : ordinaryHostResources Q H=ordinaryHostResources P K) : Q=P ∧ H=K := by
  42   constructor
  43   · apply SimpleGraph.ext
  44     funext v w
  45     have hh := congrArg (fun S => Sum.inl s(v,w)∈S) h
  46     simpa only [ordinaryHostResources_rim,SimpleGraph.mem_edgeSet] using hh
  47   · funext v u
  48     have hh := congrArg (fun S => Sum.inr (v,u)∈S) h
  49     simpa only [ordinaryHostResources_spoke] using hh

  51 theorem ordinaryWheelResources_subset_host {P F : SimpleGraph V} {Hbase : V → U → Prop} {r : ℕ}
  52     (W : OrdinaryWheel P F Hbase r) (Q : SimpleGraph V) (H : V → U → Prop) :
  53     ordinaryWheelResources W ⊆ ordinaryHostResources Q H ↔ OrdinaryWheelSurvives W Q H := by
