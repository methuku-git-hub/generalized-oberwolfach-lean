  16     (σ τ:RawMarkedOutcome V U Γ Color) (H H':Bool→V→U→Prop)

  19     let K:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) true

  43     let Q:=DS⊔SC
  44     let edgeInst:Fintype D.edgeSet:=@SimpleGraph.fintypeEdgeSet V D (Sym2.instFintype)
  45       (fun x y=>Classical.propDecidable (D.Adj x y))
  46     let σU:=rawMarkedDeleteRims σ (@SimpleGraph.edgeFinset V D edgeInst)
  47     let HU:=markedDeletePhysicalSpokes H Q
  48     let σ₁:=rawMarkedDeleteRims σ DR.edgeFinset
  49     let H₁:=markedDeletePhysicalSpokes H DS
  50     let σS:=rawMarkedDeleteRims σ₁ RC.edgeFinset

  52     let EU:=MarkedSampleArc P π (3*d) (fun e=>σU (Sum.inl e)) false
  53     let KU:=MarkedSampleArc P π (3*d) (fun e=>σU (Sum.inl e)) true;
  54     ∀ M:Finset (MarkedWheel (fun e:EU=>e.val.1) (fun e=>e.val.2)
  55       (fun e:KU=>e.val.1) (fun e=>e.val.2) m π (3*d) (HU false) (HU true)),
  56     (∀ W∈M,∀ Z∈M,W≠Z→Disjoint (markedWheelPhysicalResources W) (markedWheelPhysicalResources Z))→
