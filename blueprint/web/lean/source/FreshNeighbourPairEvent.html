  17       N≤Fintype.card V→Fintype.card J≤(Fintype.card V)^2→
  18       ∀ (G F:J→SimpleGraph V) (A B:J→Finset V) (p:J→ℝ),
  19       (∀ j,p₀≤p j ∧ p j≤1)→
  20       (∀ j,a*Fintype.card V≤((A j).card:ℝ) ∧ a*Fintype.card V≤((B j).card:ℝ))→
  21       (∀ j,Disjoint (A j) (B j))→
  22       (∀ j v,(F j).degree v≤b)→
  23       (∀ j v,v∈A j→|(((B j)∩(G j).neighborFinset v).card:ℝ)-p j*(B j).card|≤η*(B j).card)→
  24       (∀ j v,v∈B j→|(((A j)∩(G j).neighborFinset v).card:ℝ)-p j*(A j).card|≤η*(A j).card)→
  25       (∀ j v,v∈A j→∀ w,w∈A j→v≠w→¬(F j).Adj v w→
  26         ((((B j)∩(G j).neighborFinset v)∩(G j).neighborFinset w).card:ℝ)≤(p j^2+η)*(B j).card)→
  27       1-1/(Fintype.card V:ℝ)^q≤
  28         (FiniteLaw.pi (fun _:V=>FiniteLaw.retentionCoin ρ hρ.le hρ1)).probability
  29         (fun σ=>∀ j (A'' B'':Finset V) (D:SimpleGraph V),A''⊆A j→B''⊆B j→
  30           ((changed A'' ((A j)∩bernoulliVertexSet σ)).card:ℝ)≤ζ*(ρ*Fintype.card V)→
  31           ((changed B'' ((B j)∩bernoulliVertexSet σ)).card:ℝ)≤ζ*(ρ*Fintype.card V)→
  32           (∀ v,(D.degree v:ℝ)≤ζ*(ρ*Fintype.card V))→
  33           A''.card=2*B''.card→Nonempty (EndpointPairing ((G j)\D) (F j) A'' B''))
