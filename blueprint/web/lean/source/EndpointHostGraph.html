   4 /-! The graph of the endpoint-triangle problem: row edges come from R and
   5 row--index edges come from H. The external root x is used to define the two
   6 neighbour sets, and need not belong to the row domain A. -/

  15 def endpointHostGraph (R H : SimpleGraph V) (A B : Finset V) : SimpleGraph V where
  16   Adj v w := (v∈A ∧ w∈A ∧ R.Adj v w) ∨
  17     ((v∈A ∧ w∈B ∨ v∈B ∧ w∈A) ∧ H.Adj v w)
  18   symm := ⟨by
  19     intro v w h
  20     rcases h with ⟨hv,hw,he⟩|⟨h,he⟩
  21     · exact Or.inl ⟨hw,hv,he.symm⟩
  22     · exact Or.inr ⟨h.elim (fun h=>Or.inr ⟨h.2,h.1⟩) (fun h=>Or.inl ⟨h.2,h.1⟩),he.symm⟩⟩
  23   loopless := ⟨by intro v h;rcases h with h|h;exact R.loopless.irrefl v h.2.2;exact H.loopless.irrefl v h.2⟩

  25 theorem endpointHostGraph_row (R H : SimpleGraph V) (A B : Finset V) (hAB : Disjoint A B)
  26     {v w : V} (hv : v∈A) (hw : w∈A) :
  27     (endpointHostGraph R H A B).Adj v w ↔ R.Adj v w := by
  28   have hvB : v∉B := fun h=>Finset.disjoint_left.mp hAB hv h
  29   have hwB : w∉B := fun h=>Finset.disjoint_left.mp hAB hw h
  30   simp [endpointHostGraph,hv,hw,hvB,hwB]

  32 theorem endpointHostGraph_spoke (R H : SimpleGraph V) (A B : Finset V) (hAB : Disjoint A B)
  33     {v w : V} (hv : v∈A) (hw : w∈B) :
  34     (endpointHostGraph R H A B).Adj v w ↔ H.Adj v w := by
  35   have hwA : w∉A := fun h=>Finset.disjoint_left.mp hAB h hw
  36   simp [endpointHostGraph,hv,hw,hwA]

  38 theorem endpointHostGraph_row_common (R H : SimpleGraph V) (A B : Finset V) (hAB : Disjoint A B)
  39     (x : V) (Y J : Finset V)
  40     (hY : Y⊆common A {x} (fun v=>R.neighborFinset v)) (hJ : J⊆B) :
  41     common (common A {x} (fun v=>R.neighborFinset v)) (Y∪J)
  42       (fun v=>(endpointHostGraph R H A B).neighborFinset v)=
  43     common (common A (insert x Y) (fun v=>R.neighborFinset v)) J (fun v=>H.neighborFinset v) := by
  44   ext z
  45   simp only [common,Finset.mem_filter,Finset.mem_singleton,forall_eq,Finset.mem_insert,
  46     SimpleGraph.mem_neighborFinset]
  47   constructor
  48   · rintro ⟨⟨hzA,hxz⟩,hh⟩
  49     refine ⟨⟨hzA,?_⟩,?_⟩
  50     · intro v hv
  51       rcases hv with rfl|hv
  52       · exact hxz
  53       · have hvA := ((mem_common_iff _ _ _ _).mp (hY hv)).1
  54         exact (endpointHostGraph_row R H A B hAB hvA hzA).mp (hh v (Finset.mem_union_left _ hv))
  55     · intro v hv
  56       exact ((endpointHostGraph_spoke R H A B hAB hzA (hJ hv)).mp
  57         (hh v (Finset.mem_union_right _ hv)).symm).symm
  58   · rintro ⟨⟨hzA,hh⟩,hJz⟩
  59     refine ⟨⟨hzA,hh x (Or.inl rfl)⟩,?_⟩
  60     intro v hv
  61     rcases Finset.mem_union.mp hv with hv|hv
  62     · have hvA := ((mem_common_iff _ _ _ _).mp (hY hv)).1
  63       exact (endpointHostGraph_row R H A B hAB hvA hzA).mpr (hh v (Or.inr hv))
  64     · exact ((endpointHostGraph_spoke R H A B hAB hzA (hJ hv)).mpr (hJz v hv).symm).symm

  66 theorem endpointHostGraph_index_common (R H : SimpleGraph V) (A B : Finset V) (hAB : Disjoint A B)
  67     (x : V) (Y : Finset V) (hY : Y⊆A) :

  71   ext z
