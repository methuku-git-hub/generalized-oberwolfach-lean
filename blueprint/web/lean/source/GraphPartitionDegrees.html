  15 def graphCellDegree {V : Type*} [Fintype V] {k : ℕ} {d : Fin k → ℕ}
  16     (G : SimpleGraph V) (σ : Equiv.Perm V) (P : FixedPartition V k d)
  17     (v : V) (a : Fin k) : ℕ :=
  18   (Finset.univ.filter (fun x => G.Adj v (σ x) ∧ P.1 x = a)).card

  20 theorem permutation_neighbour_count {V : Type*} [Fintype V]
  21     (G : SimpleGraph V) (σ : Equiv.Perm V) (v : V) :
  22     (Finset.univ.filter (fun x => G.Adj v (σ x))).card = G.degree v := by
  23   rw [Finset.card_filter, ← G.card_neighborFinset_eq_degree]
  24   have he : (Finset.univ.filter (G.Adj v)) = G.neighborFinset v := by
  25     ext x
  26     simp
  27   rw [← he, Finset.card_filter]
  28   exact Fintype.sum_equiv σ _ _ (fun _ => rfl)

  30 theorem split_neighbour_count {X Y : Type*} [Fintype X] [Fintype Y]
  31     (G : SimpleGraph (X ⊕ Y)) (σ : Equiv.Perm (X ⊕ Y)) (v : X ⊕ Y) :
  32     (Finset.univ.filter (fun x => G.Adj v (σ (Sum.inl x)))).card +
  33       (Finset.univ.filter (fun y => G.Adj v (σ (Sum.inr y)))).card = G.degree v := by
  34   rw [← permutation_neighbour_count G σ v]
  35   simp only [Finset.card_filter, Fintype.sum_sum_type]

  37 theorem rooted_graphCellDegree {X Y : Type*} [Fintype X] [Fintype Y]
  38     {k : ℕ} {r : X → Fin k} {d : Fin k → ℕ}
  39     (G : SimpleGraph (X ⊕ Y)) (σ : Equiv.Perm (X ⊕ Y))
  40     (P : RootedPartition X Y k r d) (v : X ⊕ Y) (a : Fin k) :
  41     graphCellDegree G σ P.1 v a = rootedCellTest P
  42       (Finset.univ.filter (fun x => G.Adj v (σ (Sum.inl x))))
  43       (Finset.univ.filter (fun y => G.Adj v (σ (Sum.inr y)))) a := by
  44   unfold graphCellDegree rootedCellTest
  45   congr 1
  46   ext x
  47   cases x <;> simp

  49 theorem rooted_graph_degree_tail {X Y : Type*} [Fintype X] [Fintype Y] [Nonempty Y]
  50     {k : ℕ} (r : X → Fin k) (d : Fin k → ℕ)
  51     [Nonempty (RootedPartition X Y k r d)] [Nonempty (FixedPartition Y k d)]
  52     (G : SimpleGraph (X ⊕ Y)) (σ : Equiv.Perm (X ⊕ Y))
  53     (v : X ⊕ Y) (a : Fin k) (D ε η t : ℝ) (ht : 0 ≤ t)
  54     (hdeg : |(G.degree v : ℝ) - D| ≤ ε * ((Fintype.card X : ℝ) + Fintype.card Y))
  55     (hmargin : t + 2 * Fintype.card X + ε * (rootCellSize r a + d a) ≤
  56       η * (rootCellSize r a + d a)) :
  57     (FiniteLaw.uniform (RootedPartition X Y k r d)).probability
  58       (fun P => η * (rootCellSize r a + d a) ≤
  59         |(graphCellDegree G σ P.1 v a : ℝ) -
  60           D * (((rootCellSize r a : ℝ) + d a) /
  61             ((Fintype.card X : ℝ) + Fintype.card Y))|) ≤
  62       2 * Real.exp (-2 * t ^ 2 / (d a : ℝ)) := by
  63   simp_rw [rooted_graphCellDegree]
  64   apply rootedPartition_degree_tail r d _ _ a D ε η t ht _ hmargin
  65   simpa only [← Nat.cast_add, split_neighbour_count] using hdeg

  67 /-- A single event controls every vertex and every prescribed interface
  68 class for each of the fixed permutations. It also controls external
  69 vertices, since the test variable ranges over the whole vertex set. -/
  70 theorem rooted_graph_degrees_common_event {X Y J : Type*}
  71     [Fintype X] [Fintype Y] [Nonempty Y] [Fintype J]
  72     {k : ℕ} (r : X → Fin k) (d : Fin k → ℕ)
  73     [Nonempty (RootedPartition X Y k r d)] [Nonempty (FixedPartition Y k d)]
  74     (G : SimpleGraph (X ⊕ Y)) (σ : J → Equiv.Perm (X ⊕ Y))
  75     (classes : Finset (Fin k)) (D ε η t : ℝ) (ht : 0 ≤ t)
  76     (hdeg : ∀ v, |(G.degree v : ℝ) - D| ≤
  77       ε * ((Fintype.card X : ℝ) + Fintype.card Y))
  78     (hmargin : ∀ a ∈ classes, t + 2 * Fintype.card X + ε * (rootCellSize r a + d a) ≤
  79       η * (rootCellSize r a + d a)) :
  80     1 - ∑ a ∈ classes, (Fintype.card J : ℝ) *
  81         (Fintype.card X + Fintype.card Y) *
  82           (2 * Real.exp (-2 * t ^ 2 / (d a : ℝ))) ≤
  83     (FiniteLaw.uniform (RootedPartition X Y k r d)).probability
  84       (fun P => ∀ a ∈ classes, ∀ j v,
  85         |(graphCellDegree G (σ j) P.1 v a : ℝ) -
  86           D * (((rootCellSize r a : ℝ) + d a) /
  87             ((Fintype.card X : ℝ) + Fintype.card Y))| <
  88           η * (rootCellSize r a + d a)) := by
  89   let tests := classes ×ˢ (Finset.univ : Finset (J × (X ⊕ Y)))
  90   have h := finiteLaw_common_event_lower
  91     (FiniteLaw.uniform (RootedPartition X Y k r d)) tests
  92     (fun z P => η * (rootCellSize r z.1 + d z.1) ≤
  93       |(graphCellDegree G (σ z.2.1) P.1 z.2.2 z.1 : ℝ) -
  94         D * (((rootCellSize r z.1 : ℝ) + d z.1) /
  95           ((Fintype.card X : ℝ) + Fintype.card Y))|)
  96     (fun z => 2 * Real.exp (-2 * t ^ 2 / (d z.1 : ℝ)))
  97     (by
  98       intro z hz
  99       exact rooted_graph_degree_tail r d G (σ z.2.1) z.2.2 z.1 D ε η t ht
 100         (hdeg _) (hmargin _ (Finset.mem_product.mp hz).1))
 101   simp only [tests, Finset.sum_product, Finset.sum_const, Finset.card_univ,
 102     Fintype.card_prod, Fintype.card_sum, Nat.cast_mul, Nat.cast_add, nsmul_eq_mul,
 103     Finset.mem_product, Finset.mem_univ, and_true, Prod.forall,
 104     forall_const, not_le, mul_assoc] at h
 105   simp only [mul_assoc]
 106   apply h.trans_eq
 107   congr 1
 108   ext P
 109   constructor

 111   · intro hp a j v ha; exact hp a ha j v
