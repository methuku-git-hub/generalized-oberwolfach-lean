  14 def optionalSuccessor {V : Type*} (π : Equiv.Perm V) (a : Bool) (x : V) : V := if a then π x else x

  16 theorem separated_optionalSuccessor_index {I V : Type*}
  17     (π : Equiv.Perm V) (b : ℕ) (hb : 1≤b) (f : I→V) (hf : PermutationSeparated π b f)
  18     (a c : Bool) (i j : I) (he : optionalSuccessor π a (f i)=optionalSuccessor π c (f j)) : i=j := by
  19   by_contra hij
  20   cases a <;> cases c
  21   · exact hf i j hij (Or.inl he)
  22   · exact (permutationSeparated_successor_avoid π b hb f hf j i (Ne.symm hij)) he.symm
  23   · exact (permutationSeparated_successor_avoid π b hb f hf i j hij) he
  24   · exact hf i j hij (Or.inl (π.injective he))

  26 theorem separated_root_optionalSuccessor_ne {V : Type*} {m : ℕ}
  27     (π : Equiv.Perm V) (b : ℕ) (hb : 1≤b) (v : V) (f : Fin m→V)
  28     (hf : PermutationSeparated π b (Fin.cons v f)) (a c : Bool) (i : Fin m) :
  29     optionalSuccessor π a v≠optionalSuccessor π c (f i) := by
  30   intro he
  31   have hh := separated_optionalSuccessor_index π b hb (Fin.cons v f) hf a c 0 i.succ he
  32   exact Fin.succ_ne_zero i hh.symm

  34 def pathInternalType (j : Bool) {m : ℕ} (i : Fin m) : Bool := decide (j=false ∧ i.val=1)

  36 def pathPhysicalTail {V : Type*} {m : ℕ} (v v' : V) (z : Fin (m+1)→V) : Fin m⊕(Bool×Bool)→V
  37   | Sum.inl i=>z i.castSucc
  38   | Sum.inr (closing,side)=>if closing then z (Fin.last m) else if side then v' else v

  40 def pathPhysicalHead {V : Type*} {m : ℕ} (π : Equiv.Perm V) (j : Bool)
  41     (v v' : V) (z : Fin (m+1)→V) : Fin m⊕(Bool×Bool)→V
  42   | Sum.inl i=>optionalSuccessor π (pathInternalType j i) (z i.succ)
  43   | Sum.inr (closing,side)=>if closing then optionalSuccessor π j (if side then v' else v) else z 0

  45 def pathPhysicalEdge {V : Type*} {m : ℕ} (π : Equiv.Perm V) (j : Bool)
  46     (v v' : V) (z : Fin (m+1)→V) (i : Fin m⊕(Bool×Bool)) : Sym2 V :=
  47   s(pathPhysicalTail v v' z i,pathPhysicalHead π j v v' z i)

  49 theorem pathPhysicalEdge_internal_injective {V : Type*} {m : ℕ}
  50     (π : Equiv.Perm V) (b : ℕ) (hb : 1≤b) (j : Bool) (v v' : V) (z : Fin (m+1)→V)
  51     (hz : PermutationSeparated π b z) :
  52     Function.Injective (fun i : Fin m=>pathPhysicalEdge π j v v' z (Sum.inl i)) := by
  53   intro i k he
  54   change s(z i.castSucc,optionalSuccessor π (pathInternalType j i) (z i.succ))=
  55     s(z k.castSucc,optionalSuccessor π (pathInternalType j k) (z k.succ)) at he
  56   rcases Sym2.eq_iff.mp he with ⟨ht,_⟩|⟨h₁,h₂⟩
  57   · exact Fin.castSucc_injective m (permutationSeparated_injective π b z hz ht)
  58   · have h₁' := separated_optionalSuccessor_index π b hb z hz false (pathInternalType j k) i.castSucc k.succ h₁
  59     have h₂' := separated_optionalSuccessor_index π b hb z hz (pathInternalType j i) false i.succ k.castSucc h₂
  60     have hv₁:=congrArg Fin.val h₁'
  61     have hv₂:=congrArg Fin.val h₂'
  62     simp only [Fin.val_castSucc,Fin.val_succ] at hv₁ hv₂
  63     omega

  65 theorem pathPhysicalEdge_internal_ne_root {V : Type*} {m : ℕ}
  66     (π : Equiv.Perm V) (b : ℕ) (hb : 1≤b) (j : Bool) (v v' : V) (z : Fin (m+1)→V)
  67     (hv : PermutationSeparated π b (Fin.cons v z)) (hv' : PermutationSeparated π b (Fin.cons v' z))
  68     (i : Fin m) (closing side : Bool) :
  69     pathPhysicalEdge π j v v' z (Sum.inl i)≠pathPhysicalEdge π j v v' z (Sum.inr (closing,side)) := by
  70   let w := if side then v' else v
  71   have hw : PermutationSeparated π b (Fin.cons w z) := by dsimp [w];cases side <;> assumption
  72   have hroot (a c : Bool) (l : Fin (m+1)) := separated_root_optionalSuccessor_ne π b hb w z hw a c l
  73   intro he
  74   cases closing
  75   · change s(z i.castSucc,optionalSuccessor π (pathInternalType j i) (z i.succ))=s(w,z 0) at he
  76     rcases Sym2.eq_iff.mp he with ⟨ht,_⟩|⟨_,hh⟩
  77     · exact hroot false false i.castSucc ht.symm
  78     · exact hroot false (pathInternalType j i) i.succ hh.symm
  79   · change s(z i.castSucc,optionalSuccessor π (pathInternalType j i) (z i.succ))=
  80       s(z (Fin.last m),optionalSuccessor π j w) at he
  81     rcases Sym2.eq_iff.mp he with ⟨_,hh⟩|⟨ht,_⟩
  82     · exact hroot j (pathInternalType j i) i.succ hh.symm
  83     · exact hroot j false i.castSucc ht.symm

  85 theorem pathPhysicalEdge_root_injective {V : Type*} {m : ℕ}
  86     (π : Equiv.Perm V) (b : ℕ) (hb : 1≤b) (hm : 1≤m) (j : Bool) (v v' : V) (hvv' : v≠v')
  87     (z : Fin (m+1)→V) (hz : PermutationSeparated π b z)
  88     (hv : PermutationSeparated π b (Fin.cons v z)) (hv' : PermutationSeparated π b (Fin.cons v' z)) :
  89     Function.Injective (fun i : Bool×Bool=>pathPhysicalEdge π j v v' z (Sum.inr i)) := by
  90   have hside : Function.Injective (fun a : Bool=>if a then v' else v) := by
  91     intro a c he;cases a <;> cases c <;> simp_all
  92   have hsucc : Function.Injective (optionalSuccessor π j) := by
  93     cases j
  94     · exact Function.injective_id
  95     · exact π.injective
  96   have hne : z 0≠z (Fin.last m) := by
  97     intro he
  98     have hh:=congrArg Fin.val (permutationSeparated_injective π b z hz he)
  99     simp only [Fin.val_zero,Fin.val_last] at hh
 100     omega
 101   have hroot (side a c : Bool) (i : Fin (m+1)) :
 102       optionalSuccessor π a (if side then v' else v)≠optionalSuccessor π c (z i) := by
 103     apply separated_root_optionalSuccessor_ne π b hb _ z _ a c i
 104     cases side <;> assumption
 105   rintro ⟨a,c⟩ ⟨a',c'⟩ he
 106   cases a <;> cases a'
 107   · change s((if c then v' else v),z 0)=s((if c' then v' else v),z 0) at he
 108     rcases Sym2.eq_iff.mp he with ⟨hr,_⟩|⟨hr,_⟩
 109     · exact congrArg (fun a=>(false,a)) (hside hr)
 110     · exact (hroot c false false 0 hr).elim
 111   · change s((if c then v' else v),z 0)=s(z (Fin.last m),optionalSuccessor π j (if c' then v' else v)) at he
 112     rcases Sym2.eq_iff.mp he with ⟨hr,_⟩|⟨_,hh⟩
 113     · exact (hroot c false false (Fin.last m) hr).elim
 114     · exact (hne hh).elim
 115   · change s(z (Fin.last m),optionalSuccessor π j (if c then v' else v))=s((if c' then v' else v),z 0) at he
 116     rcases Sym2.eq_iff.mp he with ⟨hr,_⟩|⟨hh,_⟩
 117     · exact (hroot c' false false (Fin.last m) hr.symm).elim
 118     · exact (hne hh.symm).elim
 119   · change s(z (Fin.last m),optionalSuccessor π j (if c then v' else v))=
 120       s(z (Fin.last m),optionalSuccessor π j (if c' then v' else v)) at he
 121     rcases Sym2.eq_iff.mp he with ⟨_,hh⟩|⟨hh,_⟩
 122     · exact congrArg (fun a=>(true,a)) (hside (hsucc hh))
 123     · exact (hroot c' j false (Fin.last m) hh.symm).elim

 125 theorem pathPhysicalEdge_injective {V : Type*} {m : ℕ}
 126     (π : Equiv.Perm V) (b : ℕ) (hb : 1≤b) (hm : 1≤m) (j : Bool) (v v' : V) (hvv' : v≠v')
 127     (z : Fin (m+1)→V) (hz : PermutationSeparated π b z)
 128     (hv : PermutationSeparated π b (Fin.cons v z)) (hv' : PermutationSeparated π b (Fin.cons v' z)) :
 129     Function.Injective (pathPhysicalEdge π j v v' z) := by
 130   intro i k he
 131   cases i with
 132   | inl i=>cases k with
 133     | inl k=>exact congrArg Sum.inl (pathPhysicalEdge_internal_injective π b hb j v v' z hz he)
 134     | inr k=>exact (pathPhysicalEdge_internal_ne_root π b hb j v v' z hv hv' i k.1 k.2 he).elim
 135   | inr i=>cases k with
 136     | inl k=>exact (pathPhysicalEdge_internal_ne_root π b hb j v v' z hv hv' k i.1 i.2 he.symm).elim
 137     | inr k=>exact congrArg Sum.inr (pathPhysicalEdge_root_injective π b hb hm j v v' hvv' z hz hv hv' he)
