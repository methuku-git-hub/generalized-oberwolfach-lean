  22 def retainedIntervalState (π : Equiv.Perm V) (d : ℕ) (σ : RawIntervalOutcome V Γ Color)
  23     (R : V→Γ→Prop) (x : V) : Option (Γ⊕Color) :=
  24   if h : ∃ g,retainedIntervalCovered π d R x g then some (Sum.inl (Classical.choose h))
  25   else if retainedIntervalBlocked π d R x then none
  26   else (σ (Sum.inr (Sum.inr x))).map Sum.inr

  28 theorem retainedIntervalState_eq_of_no_deletion
  29     (π : Equiv.Perm V) (d : ℕ) (σ : RawIntervalOutcome V Γ Color) (R : V→Γ→Prop)
  30     (hR : ∀ a g,R a g → rawIntervalColouredStart π d σ a g) (x : V)
  31     (hkeep : ∀ a g,rawIntervalColouredStart π d σ a g → x∈permutationWindow π 0 d a → R a g) :
  32     retainedIntervalState π d σ R x=rawIntervalState π d σ x := by
  33   have hc : retainedIntervalCovered π d R x=rawIntervalCovered π d σ x := by
  34     funext g
  35     apply propext
  36     constructor
  37     · rintro ⟨a,ha,hx⟩;exact ⟨a,hR a g ha,hx⟩
  38     · rintro ⟨a,ha,hx⟩
  39       exact ⟨a,hkeep a g ha (permutationWindow_mono π (le_refl 0) (Nat.sub_le d 1) a hx),hx⟩
  40   have hb : retainedIntervalBlocked π d R x=rawIntervalBlocked π d σ x := by
  41     apply propext
  42     constructor
  43     · rintro ⟨g,a,ha,hx⟩;exact ⟨g,a,hR a g ha,hx⟩
  44     · rintro ⟨g,a,ha,hx⟩;exact ⟨g,a,hkeep a g ha hx,hx⟩
  45   unfold retainedIntervalState rawIntervalState
  46   rw [hc,hb]

  48 theorem retainedIntervalState_change_witness
  49     (π : Equiv.Perm V) (d : ℕ) (σ : RawIntervalOutcome V Γ Color) (R : V→Γ→Prop)

  61     (π : Equiv.Perm V) (d : ℕ) (σ : RawIntervalOutcome V Γ Color) (R : V→Γ→Prop)
  62     (hR : ∀ a g,R a g → rawIntervalColouredStart π d σ a g) (g₀ : Γ)
  63     (hother : ∀ a g,g≠g₀ → rawIntervalColouredStart π d σ a g → R a g) (x : V)
  64     (hchange : retainedIntervalState π d σ R x≠rawIntervalState π d σ x) :
  65     ∃ a,rawIntervalColouredStart π d σ a g₀ ∧ ¬R a g₀ ∧ x∈permutationWindow π 0 d a := by

  71 theorem permutation_power_mem_invariant (π : Equiv.Perm V) (S : Finset V)
  72     (hS : ∀ x,π x∈S ↔ x∈S) (k : ℕ) (a : V) : (π^k) a∈S ↔ a∈S := by
  73   induction k with
  74   | zero=>rfl
  75   | succ k ih=>simpa only [pow_succ',Equiv.Perm.mul_apply,hS] using ih

  77 theorem permutationWindow_mem_invariant (π : Equiv.Perm V) (S : Finset V)
  78     (hS : ∀ x,π x∈S ↔ x∈S) (d : ℕ) (a x : V)
  79     (hx : x∈permutationWindow π 0 d a) : x∈S ↔ a∈S := by
  80   obtain ⟨i,hi⟩ := (permutationWindow_mem_iff π 0 d a x).mp hx

  85 theorem retainedIntervalState_changes_card
  86     (π : Equiv.Perm V) (d : ℕ) (σ : RawIntervalOutcome V Γ Color) (R : V→Γ→Prop)
  87     (hR : ∀ a g,R a g → rawIntervalColouredStart π d σ a g) (g₀ : Γ)
  88     (hother : ∀ a g,g≠g₀ → rawIntervalColouredStart π d σ a g → R a g)
  89     (S : Finset V) (hS : ∀ x,π x∈S ↔ x∈S) :
  90     (S.filter (fun x=>retainedIntervalState π d σ R x≠rawIntervalState π d σ x)).card≤
  91       (d+1)*(S.filter (fun a=>rawIntervalColouredStart π d σ a g₀ ∧ ¬R a g₀)).card := by
  92   let A := S.filter (fun a=>rawIntervalColouredStart π d σ a g₀ ∧ ¬R a g₀)
  93   have hsub : S.filter (fun x=>retainedIntervalState π d σ R x≠rawIntervalState π d σ x)⊆
  94       A.biUnion (fun a=>permutationWindow π 0 d a) := by
  95     intro x hx
  96     obtain ⟨hxS,hx⟩ := Finset.mem_filter.mp hx
  97     obtain ⟨a,ha,hn,hxa⟩ := retainedIntervalState_change_selected π d σ R hR g₀ hother x hx
  98     apply Finset.mem_biUnion.mpr
  99     exact ⟨a,Finset.mem_filter.mpr ⟨(permutationWindow_mem_invariant π S hS d a x hxa).mp hxS,ha,hn⟩,hxa⟩
 100   calc
 101     _ ≤ (A.biUnion (fun a=>permutationWindow π 0 d a)).card := Finset.card_le_card hsub
 102     _ ≤ ∑ a∈A,(permutationWindow π 0 d a).card := Finset.card_biUnion_le
 103     _ ≤ ∑ _a∈A,(d+1) := Finset.sum_le_sum (fun a _=>by simpa using permutationWindow_card_le π 0 d a)
