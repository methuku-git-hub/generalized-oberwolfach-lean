  15 theorem ordinary_cycle_event_weight_bound {V : Type*} [Fintype V] [DecidableEq V]
  16     (G : SimpleGraph V) (r : ℕ) (hr : 0<r) (hn : 0<Fintype.card V)
  17     (a : OrdinaryCycle G r → ℝ) (ha : ∀ C,0≤a C)
  18     (B : ℝ) (hB : 0≤B) (hb : ∀ C,a C≤B/(Fintype.card V:ℝ)^(r-2))
  19     (S : Finset V) (hS : 2≤S.card) (E : OrdinaryCycle G r → Prop)
  20     (hE : ∀ C,E C → (↑S:Set V)⊆C.val.val.support) :
  21     (∑ C,if E C then a C else 0)≤(r:ℝ)^S.card*B/(Fintype.card V:ℝ)^(S.card-2) := by
  22   have hn' : (0:ℝ)<Fintype.card V := Nat.cast_pos.mpr hn
  23   have hM : 0≤B/(Fintype.card V:ℝ)^(r-2) := by positivity
  24   by_cases hSr : S.card≤r
  25   · calc
  26       _ ≤ ∑ C : OrdinaryCycle G r,if (↑S:Set V)⊆C.val.val.support then B/(Fintype.card V:ℝ)^(r-2) else 0 := by
  27         apply Finset.sum_le_sum
  28         intro C _
  29         by_cases h : E C
  30         · rw [if_pos h,if_pos (hE C h)]
  31           exact hb C
  32         · rw [if_neg h]
  33           split_ifs
  34           · exact hM
  35           · exact le_refl _
  36       _ = (∑ C : OrdinaryCycle G r,if (↑S:Set V)⊆C.val.val.support then (1:ℝ) else 0)*
  37           (B/(Fintype.card V:ℝ)^(r-2)) := by
  38         rw [Finset.sum_mul]
  39         apply Finset.sum_congr rfl
  40         intro C _
  41         split_ifs <;> simp
  42       _ ≤ (∑ C : EmbeddedCycleFamily V r,if (↑S:Set V)⊆C.val.support then (1:ℝ) else 0)*
  43           (B/(Fintype.card V:ℝ)^(r-2)) := by
  44         apply mul_le_mul_of_nonneg_right _ hM
  45         exact finite_subtype_sum_le (fun C : EmbeddedCycleFamily V r => C.val≤G)
  46           (fun C => if (↑S:Set V)⊆C.val.support then (1:ℝ) else 0) (fun C => by split_ifs <;> norm_num)
  47       _ ≤ ((r:ℝ)^S.card*(Fintype.card V:ℝ)^(r-S.card))*
  48           (B/(Fintype.card V:ℝ)^(r-2)) :=
  49         mul_le_mul_of_nonneg_right (ordinary_cycle_root_count r hr S) hM
  50       _ = _ := by
  51         rw [mul_assoc,normalized_transversal_power _ _ _ _ hn'.ne' hS hSr]
  52         ring
  53   · have hfalse (C : OrdinaryCycle G r) : ¬E C := by
  54       intro hc
  55       have hsub : S⊆C.val.val.support.toFinset := by
  56         intro v hv
  57         exact Set.mem_toFinset.mpr (hE C hc hv)
  58       have hh := Finset.card_le_card hsub
  59       rw [C.val.property.support_card] at hh
  60       exact hSr hh
  61     simp only [hfalse,ite_false,Finset.sum_const_zero]
  62     positivity

  64 theorem ordinary_cycle_adjacent_pair_load {V : Type*} [Fintype V] [DecidableEq V]
  65     (G : SimpleGraph V) (r : ℕ) (hr : 0<r) (hn : 0<Fintype.card V)
  66     (a : OrdinaryCycle G r → ℝ) (ha : ∀ C,0≤a C)
  67     (B : ℝ) (hB : 0≤B) (hb : ∀ C,a C≤B/(Fintype.card V:ℝ)^(r-2))
  68     (x y z : V) (hxy : x≠y) (hxz : x≠z) (hyz : y≠z) :
  69     (∑ C,if C.val.val.Adj x y ∧ C.val.val.Adj x z then a C else 0)≤
  70       (r:ℝ)^3*B/Fintype.card V := by
  71   have hc : ({x,y,z}:Finset V).card=3 := by simp [hxy,hxz,hyz]
  72   have hh := ordinary_cycle_event_weight_bound G r hr hn a ha B hB hb {x,y,z}
  73     (by rw [hc];norm_num) (fun C => C.val.val.Adj x y ∧ C.val.val.Adj x z) (by
  74       intro C h v hv
  75       simp only [Finset.coe_insert,Finset.coe_singleton,Set.mem_insert_iff,Set.mem_singleton_iff] at hv
  76       rcases hv with rfl|rfl|rfl
  77       · exact h.1.mem_support_left
  78       · exact h.1.mem_support_right
  79       · exact h.2.mem_support_right)
  80   simp only [hc,show 3-2=1 from rfl,pow_one] at hh
  81   convert hh using 1
  82   apply Finset.sum_congr rfl
  83   intro C _
  84   split_ifs <;> rfl
