  12 theorem finRotate_ne_self {k : ℕ} (i : Fin (k + 2)) :
  13     finRotate (k + 2) i ≠ i := by
  14   intro h
  15   rw [finRotate_apply] at h
  16   have h1 : (1 : Fin (k + 2)) = 0 := add_left_cancel (show i + 1 = i + 0 by simpa using h)
  17   have hv := congrArg Fin.val h1
  18   norm_num at hv

  20 theorem positive_weight_has_positive_marginal {Ω E : Type*} [Fintype Ω]
  21     (w : Ω → ℝ) (hw : ∀ a, 0 ≤ w a) (f : Ω → E) (a : Ω) (ha : 0 < w a) :
  22     0 < ∑ b, if f b = f a then w b else 0 := by
  23   have hle : w a ≤ ∑ b, if f b = f a then w b else 0 := by
  24     have hh := Finset.single_le_sum
  25       (f := fun b => if f b = f a then w b else 0)
  26       (s := Finset.univ) (a := a)
  27       (fun b _ => by split_ifs; exact hw b; exact le_refl _) (Finset.mem_univ a)
  28     simpa only [ite_true] using hh
  29   exact lt_of_lt_of_le ha hle

  32 theorem cyclic_weighting_graph_equations {k : ℕ} {X Y : Type*}
  33     [Fintype X] [Fintype Y]
  34     (L : Fin (k + 2) → X → Y → Prop) (R : Fin (k + 2) → Y → X → Prop)
  35     (ω : (Fin (k + 2) → X × Y × X) → ℝ)
  36     (hω : ∀ p, 0 ≤ ω p)
  37     (hleft : ∀ i x y, (∑ p, if ((p i).1, (p i).2.1) = (x, y) then ω p else 0) =
  38       interfaceCapacity (L i) x y)
  39     (hright : ∀ i y z, (∑ p, if ((p i).2.1, (p i).2.2) = (y, z) then ω p else 0) =
  40       interfaceCapacity (R i) y z)
  41     (hclosed : ∀ p, 0 < ω p → ∀ i, (p i).2.2 = (p (finRotate (k + 2) i)).1) :
  42     (∀ p, 0 < ω p → alternatingTupleGraph p ≤ cyclicInterfaceGraph L R) ∧
  43     (∀ u v, (∑ p, if (alternatingTupleGraph p).Adj u v then ω p else 0) =
  44       if (cyclicInterfaceGraph L R).Adj u v then (1 : ℝ) else 0) := by
  45   have hsupp (p : Fin (k + 2) → X × Y × X) (hp : 0 < ω p) :
  46       (∀ i, L i (p i).1 (p i).2.1) ∧ (∀ i, R i (p i).2.1 (p i).2.2) := by
  47     constructor <;> intro i
  48     · have hh := positive_weight_has_positive_marginal ω hω
  49         (fun p => ((p i).1, (p i).2.1)) p hp
  50       have he := hleft i (p i).1 (p i).2.1
  51       simp only [Prod.mk.injEq] at hh he
  52       rw [he] at hh
  53       by_contra h
  54       simp [interfaceCapacity, h] at hh
  55     · have hh := positive_weight_has_positive_marginal ω hω
  56         (fun p => ((p i).2.1, (p i).2.2)) p hp
  57       have he := hright i (p i).2.1 (p i).2.2
  58       simp only [Prod.mk.injEq] at hh he
  59       rw [he] at hh
  60       by_contra h
  61       simp [interfaceCapacity, h] at hh
  62   refine ⟨fun p hp => alternatingTupleGraph_le L R p (hsupp p hp).1
  63     (hsupp p hp).2 (hclosed p hp), ?_⟩
  64   have hcross (i j : Fin (k + 2)) (x : X) (y : Y) :
  65       (∑ p, if (alternatingTupleGraph p).Adj (Sum.inl (i, x)) (Sum.inr (j, y)) then ω p else 0) =
  66       if (cyclicInterfaceGraph L R).Adj (Sum.inl (i, x)) (Sum.inr (j, y)) then 1 else 0 := by
  67     by_cases hij : i = j
  68     · subst j
  69       have hne : i ≠ finRotate (k + 2) i := (finRotate_ne_self i).symm
  70       simpa only [alternatingTupleGraph_adj_cross, cyclicInterfaceGraph,
  71         eq_self_iff_true, true_or, true_and, hne, false_and, or_false,
  72         ← Prod.mk.injEq, interfaceCapacity] using hleft i x y
  73     · by_cases hnext : i = finRotate (k + 2) j
  74       · have heq (p : Fin (k + 2) → X × Y × X) :
  75           (if (alternatingTupleGraph p).Adj (Sum.inl (i, x)) (Sum.inr (j, y))
  76             then ω p else 0) =
  77           if ((p j).2.1, (p j).2.2) = (y, x) then ω p else 0 := by
  78           by_cases hz : ω p = 0
  79           · simp [hz]
  80           · have hp := lt_of_le_of_ne (hω p) (Ne.symm hz)
  81             have hc := hclosed p hp j
  82             simp only [alternatingTupleGraph_adj_cross, hnext, or_true,
  83               true_and, Prod.mk.injEq]
  84             simp only [← hc, and_comm]
  85         simp_rw [heq]
  86         rw [hright]
  87         simp [cyclicInterfaceGraph, hnext, finRotate_ne_self j, interfaceCapacity]
  88       · simp only [alternatingTupleGraph_adj_cross, cyclicInterfaceGraph,
  89           hij, hnext, false_or, false_and, ite_false, Finset.sum_const_zero]
  90   intro u v
  91   cases u with
  92   | inl u =>
  93     cases v with
  94     | inl v =>
  95       simp only [(alternatingTupleGraph_no_same _).1 u v, ite_false,
  96         Finset.sum_const_zero, cyclicInterfaceGraph]
  97     | inr v => exact hcross u.1 v.1 u.2 v.2
  98   | inr u =>
  99     cases v with
 100     | inl v =>
 101       simpa only [SimpleGraph.adj_comm] using hcross v.1 u.1 v.2 u.2
 102     | inr v =>
 103       simp only [(alternatingTupleGraph_no_same _).2 u v, ite_false,
 104         Finset.sum_const_zero, cyclicInterfaceGraph]
