  12 theorem prescribed_cycle_rooted_advance
  13     (QV:I→Finset X) (Q:I→SimpleGraph X) (A₀:I→Finset X) (ψ:I→X→V)
  14     (E₀:I→Finset (Sym2 V)) (J:SimpleGraph V)
  15     (β δ:ℝ) (hβ:0<β) (hβ1:β≤1) (hδ:0<δ) (hδbound:δ≤(β/100)^2)
  16     (hnβ:100/β≤(Fintype.card V:ℝ)) (hn:2/Real.sqrt δ≤(Fintype.card V:ℝ))
  17     (hI:(Fintype.card I:ℝ)≤Fintype.card V)
  18     (hQsize:∀ i,((QV i).card:ℝ)≤δ*Fintype.card V) (hQdegree:∀ i x,(Q i).degree x≤2)
  19     (hold:∀ v,((Finset.univ.filter (fun i=>∃ x,x∈A₀ i ∧ ψ i x=v)).card:ℝ)≤δ*Fintype.card V)
  20     (hJ:∀ v,(J.degree v:ℝ)≤δ*Fintype.card V)
  21     (k:I) (Dsrc:X→X→Prop) (Fsrc:SimpleGraph X) (hDQ:∀ x y,Dsrc x y→(Q k).Adj x y)
  22     (D:V→V→Prop) (L:Finset V) (hext:IsExtendingList D L β)
  23     (a:X) (xs:List X) (hnodup:(a::xs).Nodup) (hlen:3≤(a::xs).length) (ha:a∈A₀ k)
  24     (hsourcecycle:DirectedCycleThrough Dsrc Fsrc (a::xs))
  25     (st:PrescribedFamilyState I X V)
  26     (hvalid:st.Valid QV Q A₀ ψ E₀ J (Real.sqrt δ*Fintype.card V))
  27     (hinit:prescribedInducedSourceEdges (Q k) (A₀ k)⊆st.sourceEdges k)
  28     (hfree:∀ x∈a::xs,x∉A₀ k→x∉st.assigned k) (hsubset:∀ x∈a::xs,x∈QV k):
  29     ∃ st':PrescribedFamilyState I X V,
  30       st'.Valid QV Q A₀ ψ E₀ J (Real.sqrt δ*Fintype.card V) ∧
  31       st'.assigned=Function.update st.assigned k (st.assigned k∪(a::xs).toFinset) ∧
  32       (∀ i,st.assigned i⊆st'.assigned i) ∧ (∀ i,st.sourceEdges i⊆st'.sourceEdges i) ∧
  33       (∀ i x,x∈st.assigned i→st'.vertexMap i x=st.vertexMap i x) ∧
  34       (∀ x∈a::xs,x∉A₀ k→st'.vertexMap k x∈L) ∧
  35       (∀ e∈prescribedDirectedPathEdges ((a::xs)++[a]),
  36         (e.1∈A₀ k ∧ e.2∈A₀ k) ∨ D (st'.vertexMap k e.1) (st'.vertexMap k e.2)) ∧
  37       (a::xs).toFinset⊆st'.assigned k ∧
  38       prescribedOrderedPathEdges ((a::xs)++[a])⊆st'.sourceEdges k := by
  39   by_cases hex:∃ b∈xs,b∈A₀ k
  40   · obtain ⟨b,hbin,hb⟩:=hex

  42       hQsize hQdegree hold hJ k Dsrc Fsrc hDQ D L hext a b xs hnodup ha hb hbin hsourcecycle st hvalid hinit hfree hsubset
  43   · have hnoA:∀ x∈xs,x∉A₀ k:=by
  44       intro x hx hxa;exact hex ⟨x,hx,hxa⟩
  45     have hlen':2≤xs.length:=by simp only [List.length_cons] at hlen;omega
  46     obtain ⟨st',hv,haexact,hamono,hemono,hpres,hL,harcs,hvertices,hedges⟩:=
  47       prescribed_cycle_one_boundary_advance QV Q A₀ ψ E₀ J β δ hβ hβ1 hδ hδbound hnβ hn hI
  48         hQsize hQdegree hold hJ k Dsrc Fsrc hDQ D L hext a xs hnodup hlen' ha hnoA hsourcecycle st hvalid hinit hfree hsubset
  49     refine ⟨st',hv,haexact,hamono,hemono,hpres,hL,?_,hvertices,hedges⟩
  50     intro e he;exact Or.inr (harcs e he)
