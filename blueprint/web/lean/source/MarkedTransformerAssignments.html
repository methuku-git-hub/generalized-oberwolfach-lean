  14  theorem marked_transformer_attachment_assignment {X Y:Type*}
  15     [Fintype X] [Fintype Y]
  16     (own:X≃Y) (K:Finset X) (A:Finset Y)
  17     (hKA:K.card≤A.card) (hA:2≤A.card ∨ K.card=0)
  18     (hrem:2≤Fintype.card X-K.card):
  19     ∃ σ:X≃Y,(∀ x,σ x≠own x)∧∀ x∈K,σ x∈A:=by
  20   classical
  21   let Bad:K→A→Prop:=fun x y=>own x.val=y.val
  22   have hrow (x:K):(Finset.univ.filter (Bad x)).card≤1:=by
  23     apply Finset.card_le_one.mpr
  24     intro y hy z hz
  25     apply Subtype.ext
  26     exact (Finset.mem_filter.mp hy).2.symm.trans (Finset.mem_filter.mp hz).2
  27   have hcol (y:A):(Finset.univ.filter (fun x=>Bad x y)).card≤1:=by
  28     apply Finset.card_le_one.mpr
  29     intro x hx z hz
  30     apply Subtype.ext
  31     exact own.injective ((Finset.mem_filter.mp hx).2.trans (Finset.mem_filter.mp hz).2.symm)
  32   obtain ⟨fK,hfK⟩:=injection_avoiding_partial_matching Bad hrow hcol
  33     (by simpa using hKA) (by simpa using hA)
  34   let g:K→Y:=fun x=>(fK x).val
  35   have hg:Function.Injective g:=fun x y h=>fK.injective (Subtype.ext h)
  36   let T:Finset Y:=Finset.univ.image g
  37   have hT:T.card=K.card:=by
  38     dsimp only [T]
  39     rw [Finset.card_image_of_injective _ hg,Finset.card_univ,Fintype.card_coe]
  40   let Xr:=((Finset.univ:Finset X)\K)
  41   let Yr:=((Finset.univ:Finset Y)\T)
  42   have hXr:Fintype.card Xr=Fintype.card X-K.card:=by
  43     simp only [Xr,Fintype.card_coe,Finset.card_sdiff_of_subset (Finset.subset_univ _),Finset.card_univ]
  44   have hYr:Fintype.card Yr=Fintype.card Y-K.card:=by
  45     simp only [Yr,Fintype.card_coe,Finset.card_sdiff_of_subset (Finset.subset_univ _),Finset.card_univ,hT]
  46   let BadR:Xr→Yr→Prop:=fun x y=>own x.val=y.val
  47   have hrowR (x:Xr):(Finset.univ.filter (BadR x)).card≤1:=by
  48     apply Finset.card_le_one.mpr
  49     intro y hy z hz
  50     apply Subtype.ext
  51     exact (Finset.mem_filter.mp hy).2.symm.trans (Finset.mem_filter.mp hz).2
  52   have hcolR (y:Yr):(Finset.univ.filter (fun x=>BadR x y)).card≤1:=by
  53     apply Finset.card_le_one.mpr
  54     intro x hx z hz
  55     apply Subtype.ext
  56     exact own.injective ((Finset.mem_filter.mp hx).2.trans (Finset.mem_filter.mp hz).2.symm)
  57   have hcard:Fintype.card X=Fintype.card Y:=Fintype.card_congr own
  58   obtain ⟨fR,hfR⟩:=bijection_avoiding_partial_matching BadR hrowR hcolR
  59     (by rw [hXr,hYr,hcard]) (by rw [hYr,← hcard];omega)
  60   let f:X→Y:=fun x=>if hx:x∈K then g ⟨x,hx⟩ else
  61     (fR ⟨x,Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,hx⟩⟩).val
  62   have hgin (x:K):g x∈T:=Finset.mem_image.mpr ⟨x,Finset.mem_univ _,rfl⟩
  63   have hRout (x:Xr):(fR x).val∉T:=(Finset.mem_sdiff.mp (fR x).property).2
  64   have hf:Function.Injective f:=by
  65     intro x y h
  66     by_cases hx:x∈K <;> by_cases hy:y∈K
  67     · simp only [f,dite_eq_left hx,dite_eq_left hy] at h
  68       exact congrArg Subtype.val (hg h)
  69     · simp only [f,dite_eq_left hx,dite_eq_right hy] at h
  70       exact (hRout ⟨y,Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,hy⟩⟩ (h ▸ hgin ⟨x,hx⟩)).elim
  71     · simp only [f,dite_eq_right hx,dite_eq_left hy] at h
  72       exact (hRout ⟨x,Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,hx⟩⟩ (h.symm ▸ hgin ⟨y,hy⟩)).elim
  73     · simp only [f,dite_eq_right hx,dite_eq_right hy] at h
  74       exact congrArg Subtype.val (fR.injective (Subtype.ext h))
  75   let σ:X≃Y:=Equiv.ofBijective f ((Fintype.bijective_iff_injective_and_card f).mpr ⟨hf,hcard⟩)
  76   refine ⟨σ,?_,?_⟩
  77   · intro x
  78     change f x≠own x
  79     by_cases hx:x∈K
  80     · simp only [f,dite_eq_left hx]
  81       exact Ne.symm (hfK ⟨x,hx⟩)
  82     · simp only [f,dite_eq_right hx]
  83       exact Ne.symm (hfR ⟨x,Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,hx⟩⟩)
  84   · intro x hx
  85     change f x∈A
  86     simpa only [f,dite_eq_left hx,g] using (fK ⟨x,hx⟩).property
