  17 theorem embedded_profile_pair_probability {W : Type*} [DecidableEq W]
  18     (e : (Σ i,V i) ↪ W) (m : I → ℕ)
  19     [∀ i,Nonempty (FixedSubset (V i) (m i))]
  20     (hm : ∀ i,m i ≤ Fintype.card (V i)) (ρ : ℝ) (hρ : 0 ≤ ρ)
  21     (hρm : ∀ i,(m i:ℝ)/Fintype.card (V i) ≤ ρ) (x y : W) (hxy : x≠y) :
  22     (FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (m i)))).probability
  23       (fun σ => x∈(profileRootSet σ).map e ∧ y∈(profileRootSet σ).map e) ≤ ρ^2 := by
  24   by_cases hx : ∃ u,e u=x
  25   · obtain ⟨u,rfl⟩ := hx
  26     by_cases hy : ∃ v,e v=y
  27     · obtain ⟨v,rfl⟩ := hy
  28       simpa only [Finset.mem_map,e.injective.eq_iff,exists_eq_right] using
  29         profile_pair_inclusion_upper m hm ρ hρ hρm u v (fun he => hxy (congrArg e he))
  30     · have hp : (fun σ : ∀ i,FixedSubset (V i) (m i) =>
  31           e u∈(profileRootSet σ).map e ∧ y∈(profileRootSet σ).map e)=fun _ => False := by
  32         funext σ
  33         apply propext
  34         simp only [iff_false]
  35         intro hh
  36         obtain ⟨v,_,he⟩ := Finset.mem_map.mp hh.2
  37         exact hy ⟨v,he⟩
  38       rw [hp,finiteLaw_probability_false]
  39       positivity
  40   · have hp : (fun σ : ∀ i,FixedSubset (V i) (m i) =>
  41         x∈(profileRootSet σ).map e ∧ y∈(profileRootSet σ).map e)=fun _ => False := by
  42       funext σ
  43       apply propext
  44       simp only [iff_false]
  45       intro hh
  46       obtain ⟨v,_,he⟩ := Finset.mem_map.mp hh.1
  47       exact hx ⟨v,he⟩
  48     rw [hp,finiteLaw_probability_false]
  49     positivity

  51 theorem embedded_profile_forbidden_pairs_probability {W : Type*} [DecidableEq W]
  52     (e : (Σ i,V i) ↪ W) (m : I → ℕ)
  53     [∀ i,Nonempty (FixedSubset (V i) (m i))]
  54     (hm : ∀ i,m i ≤ Fintype.card (V i)) (ρ : ℝ) (hρ : 0 ≤ ρ)
  55     (hρm : ∀ i,(m i:ℝ)/Fintype.card (V i) ≤ ρ)
  56     (E : Finset (W×W)) (hE : ∀ e∈E,e.1≠e.2) :
  57     (FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (m i)))).probability
  58       (fun σ => ∃ f∈E,f.1∈(profileRootSet σ).map e ∧ f.2∈(profileRootSet σ).map e) ≤
  59       (E.card:ℝ)*ρ^2 := by
  60   apply (finiteLaw_union_bound _ E (fun f σ => f.1∈(profileRootSet σ).map e ∧ f.2∈(profileRootSet σ).map e)).trans
  61   calc
  62     _ ≤ ∑ _f∈E,ρ^2 := Finset.sum_le_sum (fun f hf => embedded_profile_pair_probability e m hm ρ hρ hρm f.1 f.2 (hE f hf))
  63     _ = _ := by simp

  65 theorem profile_remaining_forbidden_pairs_probability
  66     (a : I → ℕ) (ha : ∀ i,1 ≤ a i) (k F t D : ℝ)
  67     (hk : 0 < k) (ht : 0 ≤ t) (hF : 2*F ≤ k)
  68     (hn : ∀ i,(Fintype.card (V i):ℝ)=a i*k) (Z : Finset (Σ i,V i))
  69     (hZ : (Z.card:ℝ) ≤ F) (m : I → ℕ)
  70     [∀ i,Nonempty (FixedSubset (ProfileRemainingClass Z i) (m i))]
  71     (hm : ∀ i,m i ≤ Fintype.card (ProfileRemainingClass Z i))
  72     (hmt : ∀ i,(m i:ℝ) ≤ a i*t)
  73     (E : Finset ((Σ i,V i)×(Σ i,V i))) (hE : ∀ e∈E,e.1≠e.2)
  74     (hsize : (E.card:ℝ) ≤ D*k) :
  75     (FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (ProfileRemainingClass Z i) (m i)))).probability
  76       (fun σ => ∃ f∈E,f.1∈(profileRootSet σ).map (profileComplementEmbedding Z) ∧
  77         f.2∈(profileRootSet σ).map (profileComplementEmbedding Z)) ≤ 4*D*t^2/k := by
  78   have hh := embedded_profile_forbidden_pairs_probability (profileComplementEmbedding Z) m hm (2*t/k)
  79     (by positivity) (profile_remaining_sampling_ratio a ha k F t hk ht hF hn Z hZ m hmt) E hE
  80   apply hh.trans
  81   calc
  82     _ ≤ (D*k)*(2*t/k)^2 := mul_le_mul_of_nonneg_right hsize (sq_nonneg _)
  83     _ = _ := by field_simp;ring
