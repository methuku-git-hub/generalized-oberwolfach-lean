   8 theorem marked_active_hub_density_arithmetic
   9     (n nV nU I J T R A B ε ρ ω:ℝ)
  10     (hn:0≤n) (hnV:0≤nV) (hnU:0≤nU) (hI:0≤I) (hJ:0≤J)
  11     (he:0<ε) (hehalf:ε≤1/2) (hρ:0<ρ) (hω:0≤ω)
  12     (hV:nV≤n) (hU:nU≤n) (hIlo:(ε/2)*n≤I) (hJlo:(ρ/8)*I≤J)
  13     (hT:T≤ε*I) (hR:R≤ε*J) (hA:A=I-T) (hB:B=J-R)
  14     (hωv:ω≤ε/8) (hωu:ω≤ρ*ε/64):
  15     ω*(nV+nU)≤A ∧ ω*(nV+nU)≤B:=by
  16   have hsum:nV+nU≤2*n:=by linarith
  17   have hAlo:I/2≤A:=by
  18     rw [hA]
  19     nlinarith [mul_nonneg (sub_nonneg.mpr hehalf) hI]
  20   have hBlo:J/2≤B:=by
  21     rw [hB]
  22     nlinarith [mul_nonneg (sub_nonneg.mpr hehalf) hJ]
  23   have hAparent:(ε/4)*n≤A:=by linarith [hIlo,hAlo]
  24   have hJparent:(ρ*ε/16)*n≤J:=by
  25     have hh:=mul_le_mul_of_nonneg_left hIlo (show 0≤ρ/8 by positivity)
  26     nlinarith [hh,hJlo]
  27   have hBparent:(ρ*ε/32)*n≤B:=by linarith [hJparent,hBlo]
  28   have hleft:ω*(nV+nU)≤2*ω*n:=by nlinarith [mul_le_mul_of_nonneg_left hsum hω]
  29   have hrightA:2*ω*n≤(ε/4)*n:=by
  30     have hh:=mul_le_mul_of_nonneg_right hωv hn
  31     nlinarith [hh]
  32   have hrightB:2*ω*n≤(ρ*ε/32)*n:=by
  33     have hh:=mul_le_mul_of_nonneg_right hωu hn
  34     nlinarith [hh]
  35   exact ⟨hleft.trans (hrightA.trans hAparent),hleft.trans (hrightB.trans hBparent)⟩
  36 end Oberwolfach
