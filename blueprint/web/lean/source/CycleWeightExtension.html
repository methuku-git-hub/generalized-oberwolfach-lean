  11 def cycleWeightExtend {c : ℕ} {V : Type*} (a : Fin c → V → V → ℝ)
  12     (w : DirectedTransversalCycle a → ℝ) (v : Fin c → V) : ℝ :=
  13   if hv : ∀ i,0<a i (v i) (v (finRotate c i)) then w ⟨v,hv⟩ else 0

  37     (a : Fin c → V → V → ℝ) (w : DirectedTransversalCycle a → ℝ)
  38     (i : Fin c) (x y : V) :
  39     (∑ v,if v i=x ∧ v (finRotate c i)=y then cycleWeightExtend a w v else 0)=
  40       ∑ v : DirectedTransversalCycle a,if v.val i=x ∧ v.val (finRotate c i)=y then w v else 0 := by

  44     (fun v => if v i=x ∧ v (finRotate c i)=y then cycleWeightExtend a w v else 0)
  45     (fun v : S => if v.val i=x ∧ v.val (finRotate c i)=y then w v else 0)
