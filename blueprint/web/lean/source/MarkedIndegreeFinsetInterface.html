   2 /-! Finite-set form of the marked-indegree correction. The arc universe is a

   4 the symmetric difference of the old and new marked sets. -/

  14 def directedInCount (P:V×V→Prop) (M:Finset {e:V×V // P e}) (v:V) : ℕ :=
  15   (M.filter (fun e=>e.val.2=v)).card

  17 def changedDirectedPhysicalEdges (P:V×V→Prop)
  18     (M M₀:Finset {e:V×V // P e}) : Finset (Finset V) :=
  19   (symmDiff M M₀).image (fun e=>({e.val.1,e.val.2}:Finset V))

  21 theorem marked_indegree_correction_finsets
  22     (P:V×V→Prop) (M₀ A:Finset {e:V×V // P e}) (b:ℕ) (rho eta:ℝ)
  23     (hrho:0<rho) (heta:0<eta) (hetaR:eta≤rho)
  24     (hdev:∀ v, |(directedInCount P M₀ v:ℝ)-(b:ℝ)|≤eta*Fintype.card V)
  25     (hmarked:∀ v, rho*Fintype.card V≤(directedInCount P (A∩M₀) v:ℝ))
  26     (hunmarked:∀ v, rho*Fintype.card V≤(directedInCount P (A\M₀) v:ℝ)) :
  27     ∃ M:Finset {e:V×V // P e},
  28       (∀ v, directedInCount P M v=b) ∧ symmDiff M M₀⊆A ∧
  29       (∀ v, ((physicalEdgeGraph (changedDirectedPhysicalEdges P M M₀)).degree v:ℝ) ≤
  30         (1+2/rho)*eta*Fintype.card V+(Fintype.card V:ℝ)^(3/5:ℝ)) := by
  31   let B₀ : {e:V×V // P e}→Bool := fun e=>decide (e∈M₀)
  32   have hcount : ∀ v, markedHeadCount (fun e:{e:V×V // P e}=>e.val.2) B₀ v = directedInCount P M₀ v := by
  33     intro v
  34     have heq : Finset.univ.filter (fun e:{e:V×V // P e}=>e.val.2=v ∧ B₀ e=true) =
  35         M₀.filter (fun e=>e.val.2=v) := by
  36       ext e
  37       simp [B₀, and_comm]
  38     unfold markedHeadCount directedInCount
  39     rw [← Finset.card_filter,heq]
  40   have hflex : ∀ v flag,
  41       markedFlexibleIncoming (fun e:{e:V×V // P e}=>e.val.2) B₀ (fun e=>e∈A) v flag =
  42       directedInCount P (if flag then A∩M₀ else A\M₀) v := by
  43     intro v flag
  44     unfold markedFlexibleIncoming directedInCount
  45     congr 1
  46     ext e
  47     cases flag <;> simp [B₀,Finset.mem_filter,Finset.mem_sdiff,Finset.mem_inter, and_assoc, and_comm, and_left_comm]
  48   have hdev' : ∀v, |(markedHeadCount (fun e:{e:V×V // P e}=>e.val.2) B₀ v:ℝ)-(b:ℝ)|≤eta*Fintype.card V := by
  49     intro v; rw [hcount]; exact hdev v
  50   have hm : ∀v, rho*Fintype.card V≤(markedFlexibleIncoming (fun e:{e:V×V // P e}=>e.val.2) B₀ (fun e=>e∈A) v true:ℝ) := by
  51     intro v; rw [hflex]; exact hmarked v
  52   have hu : ∀v, rho*Fintype.card V≤(markedFlexibleIncoming (fun e:{e:V×V // P e}=>e.val.2) B₀ (fun e=>e∈A) v false:ℝ) := by
  53     intro v; rw [hflex]; exact hunmarked v
  54   obtain ⟨B,hB,hA,hdeg⟩ := marked_indegree_correction_oriented_graph P B₀ (fun e=>e∈A) b rho eta
  55     hrho heta hetaR hdev' hm hu
  56   let M : Finset {e:V×V // P e} := Finset.univ.filter (fun e=>B e=true)
  57   have hmem : ∀ e, e∈M ↔ B e=true := by intro e; simp [M]
  58   have hchange : ∀ e, e∈symmDiff M M₀ ↔ B e≠B₀ e := by
  59     intro e
  60     rw [Finset.mem_symmDiff]
  61     simp only [hmem,B₀,decide_eq_true_eq]
  62     by_cases he:e∈M₀ <;> cases hb:B e <;> simp [he,hb]
  63   have hedges : changedDirectedPhysicalEdges P M M₀ =
  64       retypedPhysicalEdges (fun e:{e:V×V // P e}=>e.val.1) (fun e=>e.val.2) B B₀ := by
  65     unfold changedDirectedPhysicalEdges retypedPhysicalEdges
  66     congr 1
  67     ext e
  68     simp only [Finset.mem_symmDiff,Finset.mem_filter,Finset.mem_univ,true_and]
  69     simpa only [Finset.mem_symmDiff] using hchange e
  70   refine ⟨M,?_,?_,?_⟩
  71   · intro v
  72     have hc : directedInCount P M v = markedHeadCount (fun e:{e:V×V // P e}=>e.val.2) B v := by
  73       have heq : Finset.univ.filter (fun e:{e:V×V // P e}=>e.val.2=v ∧ B e=true) =
  74           M.filter (fun e=>e.val.2=v) := by
  75         ext e
  76         simp [hmem e, and_comm]
  77       unfold directedInCount markedHeadCount
  78       rw [← Finset.card_filter,heq]
  79     rw [hc]; exact hB v
  80   · intro e he
  81     exact hA e ((hchange e).mp he)
  82   · intro v
  83     rw [hedges]
  84     exact hdeg v
