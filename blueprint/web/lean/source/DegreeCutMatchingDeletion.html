  14 theorem filter_card_le_kept_add_deleted {Z : Type*} [Fintype Z]
  15     (P Q : Z→Prop) :
  16     (Finset.univ.filter P).card≤(Finset.univ.filter (fun z=>P z∧¬Q z)).card+
  17       (Finset.univ.filter Q).card := by
  18   classical
  19   have hs : Finset.univ.filter P⊆(Finset.univ.filter (fun z=>P z∧¬Q z))∪(Finset.univ.filter Q):=by
  20     intro z hz
  21     by_cases hq : Q z
  22     · exact Finset.mem_union_right _ (Finset.mem_filter.mpr ⟨Finset.mem_univ _,hq⟩)
  23     · exact Finset.mem_union_left _ (Finset.mem_filter.mpr ⟨Finset.mem_univ _,(Finset.mem_filter.mp hz).2,hq⟩)
  24   exact (Finset.card_le_card hs).trans (Finset.card_union_le _ _)

  26 theorem degree_cut_matching_after_deletion {X Y : Type*} [Fintype X] [Fintype Y]
  27     (E J : X→Y→Prop) (hcard : Fintype.card X=Fintype.card Y)
  28     (a b lam : ℝ) (ha : 0<a) (hb : 0≤b) (hlam : 0<lam) (hmargin : b<lam*a)
  29     (hrow : ∀ x,a+b≤((Finset.univ.filter (E x)).card:ℝ))
  30     (hcol : ∀ y,a+b≤((Finset.univ.filter (fun x=>E x y)).card:ℝ))
  31     (hjrow : ∀ x,((Finset.univ.filter (J x)).card:ℝ)≤b)
  32     (hjcol : ∀ y,((Finset.univ.filter (fun x=>J x y)).card:ℝ)≤b)
  33     (hcut : ∀ S : Finset X,∀ T : Finset Y,a≤(S.card:ℝ) → a≤(T.card:ℝ) →
  34       lam*S.card*T.card≤(bipartiteMatrixCut (fun x y=>if E x y then 1 else 0) S T:ℝ)) :
  35     ∃ σ : X≃Y,∀ x,E x (σ x)∧¬J x (σ x) := by
  36   classical
  37   apply degree_cut_perfect_matching (fun x y=>E x y∧¬J x y) hcard a ha.le
  38   · intro x
  39     have hh:=filter_card_le_kept_add_deleted (E x) (J x)
  40     have hhR : ((Finset.univ.filter (E x)).card:ℝ)≤
  41       (Finset.univ.filter (fun y=>E x y∧¬J x y)).card+(Finset.univ.filter (J x)).card:=by exact_mod_cast hh
  42     have hk : a≤((Finset.univ.filter (fun y=>E x y∧¬J x y)).card:ℝ):=by linarith only [hhR,hrow x,hjrow x]
  43     convert hk using 1 <;> congr!
  44   · intro y
  45     have hh:=filter_card_le_kept_add_deleted (fun x=>E x y) (fun x=>J x y)
  46     have hhR : ((Finset.univ.filter (fun x=>E x y)).card:ℝ)≤
  47       (Finset.univ.filter (fun x=>E x y∧¬J x y)).card+(Finset.univ.filter (fun x=>J x y)).card:=by exact_mod_cast hh
  48     have hk : a≤((Finset.univ.filter (fun x=>E x y∧¬J x y)).card:ℝ):=by linarith only [hhR,hcol y,hjcol y]
  49     convert hk using 1 <;> congr!
  50   · intro S T hS hT hSn hTn
  51     by_contra hn
  52     have hinc : ∀ x∈S,∀ y∈T,E x y → J x y:=by
  53       intro x hx y hy he
  54       by_contra hj
  55       exact hn ⟨x,hx,y,hy,he,hj⟩
  56     have hupper : (bipartiteMatrixCut (fun x y=>if E x y then 1 else 0) S T:ℝ)≤(S.card:ℝ)*b:=by
  57       simp only [bipartiteMatrixCut,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  58       calc
  59         _ ≤ ∑ _x∈S,b:=by
  60           apply Finset.sum_le_sum
  61           intro x hx
  62           calc
  63             _ ≤ ∑ y∈T,if J x y then (1:ℝ) else 0:=by
  64               apply Finset.sum_le_sum
  65               intro y hy
  66               by_cases he : E x y
  67               · simp [he,hinc x hx y hy he]
  68               · simp only [he,ite_false];positivity
  69             _ ≤ ∑ y,if J x y then (1:ℝ) else 0:=Finset.sum_le_sum_of_subset_of_nonneg (Finset.subset_univ T) (by intro y hy hyt;positivity)
  70             _ ≤ b:=by simpa using hjrow x
  71         _ = _:=by simp
  72     have hpositive : (0:ℝ)<S.card:=Nat.cast_pos.mpr hSn.card_pos
  73     have hlarge : b<lam*T.card:=hmargin.trans_le (mul_le_mul_of_nonneg_left hT hlam.le)
  74     have hstrict:=mul_lt_mul_of_pos_left hlarge hpositive
  75     have hlower:=hcut S T hS hT
  76     nlinarith only [hstrict,hupper,hlower]
