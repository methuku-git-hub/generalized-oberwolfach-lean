  24       (finRotate (k + 2) i) (k + 1) z x)

  32           ∏ j ∈ (Finset.univ.erase c).erase (finRotate (k + 2) c),
  33             cyclicInterfaceMarginal L j (p j).1) := by
  34   have hC : 0 < (intervalCommon (L c) (R c) (p c).1 (p c).2.2 : ℝ) := by
  35     exact_mod_cast Finset.card_pos.mpr ⟨(p c).2.1, Finset.mem_filter.mpr
  36       ⟨Finset.mem_univ _, hp c⟩⟩
  37   have hh := cyclic_sector_formula (cyclicInterfaceMarginal L)
  38     (cyclicResidualSystem L R Q) hm hH c (Q.val c) (hfixed c)
  39     (fun i => (p i).1)
  40     (fun i => if i = c then cyclicReservePath L R Q i (p i)
  41       else cyclicResidualPath L R Q i (p i))
  42     (fun i => cyclicResidualPath L R Q i (p i))

  51 def cyclicSectorLower (k : ℕ) (α : ℝ) : ℝ :=
  52   ((((2 * (k + 2) - 1) * α / (k + 2)) ^ 2 / (k + 1)) ^ (k + 1)) /
  53     ((1 / (k + 2)) ^ k)
  54 
  55 def cyclicSectorUpper (k : ℕ) (α : ℝ) : ℝ :=
  56   ((1 / ((k + 1) * ((2 * (k + 2) - 1) * α / (k + 2)) *
  57     ((k + 2) / (2 * (k + 2) - 1)))) ^ (k + 1)) /
  58     ((1 / (2 * (k + 2) - 1)) * (1 / (2 * (k + 2) - 1)) ^ k)
