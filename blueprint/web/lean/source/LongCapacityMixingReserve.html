  12     (ha : ∀ i x y, 0 ≤ a i x y) (hd : ∀ i x, 0 < d i x)
  13     (hrow : ∀ i x, ∑ y, a i x y=d i x)
  14     (hcol : ∀ i y, ∑ x, a i x y=d (i+1) y)
  15     (M : ℝ) (hM : 0 < M) (hms : ∀ i, ∑ x, d i x=M)
  16     (s L : Fin 3 → ℕ) (t : ℕ)
  17     (hbd : ∀ i x, d (s i+L i) x=d (s (finRotate 3 i)) x)
  18     (J : Finset ℕ) (hJ : J.card ≤ 1) (κ B : ℝ)
  19     (hκ : 0 < κ) (hκ1 : κ < 1) (hB : 1 ≤ B)
  20     (ht : 1+3*⌈Real.log (10*(B+1)) / (-Real.log (1-κ))⌉₊ ≤ t)
  21     (hL : ∀ i, 4*t+1 ≤ L i)
  22     (hupper : ∀ i j, s i ≤ j → j < s i+L i → j ∉ J → ∀ x y,
  23       capacityTransition a d j x y ≤ B*capacityProfile d M (j+1) y)
  24     (hminor : ∀ i j, s i ≤ j → j+3 ≤ s i+L i → j ∉ J → j+1 ∉ J → j+2 ∉ J →
  25       ∀ x y, κ*capacityProfile d M (j+3) y ≤ transitionWalk (capacityTransition a d) j 3 x y) :
  26     (∀ i x z, |transitionWalk (capacityTransition a d) (s i) (L i) x z-
  27       capacityProfile d M (s i+L i) z| ≤ (1/10)*capacityProfile d M (s i+L i) z) ∧
  28     (∀ (Q : transportProduct (threeIntervalMarginal (intervalDegrees d s L)) (finRotate 3))
  29       i j x y, pathEdgeLoad (longCapacityReserve a d s L Q i) j x y ≤ (11/27)*a (s i+j.val) x y) := by
  30   let R := capacityTransition a d
  31   let π := capacityProfile d M
  32   have hR := capacityTransition_isTransition a d ha hd hrow
  33   obtain ⟨hπpos,hπsum⟩ := capacityProfile_distribution d M hM hd hms
  34   have hπ := capacityProfile_stationary a d M hd hcol
  35   have hMp (j : ℕ) (x : X) : M*π j x=d j x := by
  36     change M*(d j x/M)=d j x
  37     field_simp [hM.ne']
  38   have hmix (i : Fin 3) := long_interval_mixing R π hR (fun j x => (hπpos j x).le)
  39     hπsum hπ J hJ κ B hκ hκ1 hB (s i) (L i) t (hupper i) (hminor i) ht (by have := hL i; omega)
  40   refine ⟨hmix,?_⟩
  41   intro Q i j x y
  42   have hQ : Q.val i ∈ transportMatrices (fun x => M*π (s i) x/3) (fun z => M*π (s i+L i) z/3) := by
  43     simp only [hMp,hbd]
  44     exact Q.property i (Set.mem_univ i)
  45   have hh := long_reserved_path_edge_bound R π hR hπpos hπsum hπ J hJ κ B M hκ hκ1 hB
  46     (s i) (L i) t ht (hL i) (hupper i) (hminor i) (Q.val i) hQ j x y
  47   rw [capacityTransition_reconstruct a d M hM.ne' (fun i x => (hd i x).ne')] at hh
  48   exact hh
