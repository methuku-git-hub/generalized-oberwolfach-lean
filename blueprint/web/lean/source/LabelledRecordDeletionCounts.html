  18 theorem labelled_cycle_records_constrained_count (R : C → Finset V)
  19     (S : O → Finset (Fin h ⊕ Fin c)) (X Y : Finset V)
  20     (P : Finset (Fin h ⊕ Fin c)) (J : ℕ)
  21     (hJ : ∀ U : Finset V,Fintype.card {C₀ : C // R C₀=U} ≤ J) :
  22     Fintype.card {D : LabelledCycleRecord R S X // ∀ j∈P,D.1.val j∈Y} ≤
  23       X.card^c*Y.card^(freeLabelPositions P).card*
  24         Fintype.card V^(h-(freeLabelPositions P).card)*J^Fintype.card O := by
  25   let A := {D : LabelledCycleRecord R S X // ∀ j∈P,D.1.val j∈Y}
  26   let F := {f : RootedLabelMap V h c X // ∀ j∈P,f.val j∈Y}
  27   let B := Σ f : F,CycleIdentifierAssignments R S f.val.val
  28   let enc (D : A) : B := ⟨⟨D.val.1,D.property⟩,D.val.2⟩
  29   have hinj : Function.Injective enc := by
  30     intro D E he
  31     apply Subtype.ext
  32     exact congrArg (fun b : B => (⟨b.1.val,b.2⟩ : LabelledCycleRecord R S X)) he
  33   have hB : Fintype.card B ≤ Fintype.card F*J^Fintype.card O := by
  34     rw [Fintype.card_sigma]
  35     calc
  36       _ ≤ ∑ _f : F,J^Fintype.card O :=
  37         Finset.sum_le_sum (fun f _ => cycle_identifier_assignments_card R S J hJ f.val.val)
  38       _ = _ := by simp
  39   exact (Fintype.card_le_of_injective enc hinj).trans (hB.trans
  40     (Nat.mul_le_mul_right _ (rooted_label_maps_constrained_count X Y P)))

  42 theorem labelled_cycle_record_free_coordinate_count (R : C → Finset V)
  43     (S : O → Finset (Fin h ⊕ Fin c)) (X : Finset V) (J : ℕ)
  44     (hJ : ∀ U : Finset V,Fintype.card {C₀ : C // R C₀=U} ≤ J) (j : Fin h) (v : V) :
  45     Fintype.card {D : LabelledCycleRecord R S X // D.1.val (Sum.inl j)=v} ≤
  46       X.card^c*Fintype.card V^(h-1)*J^Fintype.card O := by
  47   have hh := labelled_cycle_records_constrained_count R S X {v} {Sum.inl j} J hJ
  48   have he : freeLabelPositions ({Sum.inl j} : Finset (Fin h ⊕ Fin c))={j} := by
  49     ext k;simp [freeLabelPositions]
  50   simp only [he,Finset.card_singleton,pow_one,one_pow,mul_one] at hh
  51   have hp (D : LabelledCycleRecord R S X) :
  52       (∀ k∈({Sum.inl j} : Finset (Fin h ⊕ Fin c)),D.1.val k∈({v} : Finset V)) ↔
  53         D.1.val (Sum.inl j)=v := by simp
  54   have hcard := Fintype.card_congr (Equiv.subtypeEquivRight hp)
  55   calc
  56     _ = Fintype.card {D : LabelledCycleRecord R S X //
  57         ∀ k∈({Sum.inl j} : Finset (Fin h ⊕ Fin c)),D.1.val k∈({v} : Finset V)} := hcard.symm
  58     _ ≤ _ := hh

  60 theorem labelled_absorber_free_vertex_count (R : C → Finset V)
  61     (S : T → VertexAbsorberTemplate c) (X : Finset V) (J : ℕ)
  62     (hJ : ∀ U : Finset V,Fintype.card {C₀ : C // R C₀=U} ≤ J) (v : V) :
  63     Fintype.card {D : LabelledAbsorberFamily R S X //
  64       v∈labelledAbsorberFree R (S D.1) X D.2} ≤
  65       (c*c)*Fintype.card T*X.card^c*Fintype.card V^(c*c-1)*J^(2*c+1) := by
  66   have hc (j : Fin (c*c)) : Fintype.card {D : LabelledAbsorberFamily R S X //
  67       D.2.1.val (Sum.inl j)=v} ≤
  68       Fintype.card T*(X.card^c*Fintype.card V^(c*c-1)*J^(2*c+1)) := by
  69     have he : Fintype.card {D : LabelledAbsorberFamily R S X // D.2.1.val (Sum.inl j)=v}=
  70         ∑ t,Fintype.card {D : LabelledCycleRecord R (S t).val X // D.1.val (Sum.inl j)=v} := by
  71       convert finite_sigma_predicate_card (fun t (D : LabelledCycleRecord R (S t).val X) =>
  72         D.1.val (Sum.inl j)=v) using 1 <;> congr!
  73     rw [he]
  74     calc
  75       _ ≤ ∑ _t : T,X.card^c*Fintype.card V^(c*c-1)*J^(2*c+1) := by
  76         apply Finset.sum_le_sum
  77         intro t _
  78         have hh := labelled_cycle_record_free_coordinate_count R (S t).val X J hJ j v
  79         simpa only [Fintype.card_sum,Fintype.card_fin,show c+(c+1)=2*c+1 by omega] using hh
  80       _ = _ := by simp
  81   have hp (D : LabelledAbsorberFamily R S X) :
  82       v∈labelledAbsorberFree R (S D.1) X D.2 ↔ ∃ j : Fin (c*c),D.2.1.val (Sum.inl j)=v := by
  83     simp [labelledAbsorberFree]
  84   rw [Fintype.card_congr (Equiv.subtypeEquivRight hp)]
  85   calc
  86     _ ≤ ∑ j : Fin (c*c),Fintype.card {D : LabelledAbsorberFamily R S X // D.2.1.val (Sum.inl j)=v} := by
  87       convert finite_exists_card_le_sum (fun j (D : LabelledAbsorberFamily R S X) => D.2.1.val (Sum.inl j)=v) using 1 <;> congr!
  88     _ ≤ ∑ _j : Fin (c*c),Fintype.card T*(X.card^c*Fintype.card V^(c*c-1)*J^(2*c+1)) :=
  89       Finset.sum_le_sum (fun j _ => hc j)
  90     _ = _ := by simp only [Finset.sum_const,Finset.card_univ,Fintype.card_fin,smul_eq_mul];ring

  92 theorem labelled_absorber_free_deletion_count (R : C → Finset V)
  93     (S : T → VertexAbsorberTemplate c) (X Z : Finset V) (J : ℕ)
  94     (hJ : ∀ U : Finset V,Fintype.card {C₀ : C // R C₀=U} ≤ J) :
  95     Fintype.card {D : LabelledAbsorberFamily R S X //
  96       ¬Disjoint (labelledAbsorberFree R (S D.1) X D.2) Z} ≤
  97       Z.card*((c*c)*Fintype.card T*X.card^c*Fintype.card V^(c*c-1)*J^(2*c+1)) := by
  98   have hp (D : LabelledAbsorberFamily R S X) :
  99       ¬Disjoint (labelledAbsorberFree R (S D.1) X D.2) Z ↔
 100         ∃ v : Z,v.val∈labelledAbsorberFree R (S D.1) X D.2 := by
 101     constructor
 102     · intro hd
 103       by_contra hn
 104       exact hd (Finset.disjoint_right.mpr (fun v hv hw => hn ⟨⟨v,hv⟩,hw⟩))
 105     · rintro ⟨v,hv⟩ hd;exact Finset.disjoint_left.mp hd hv v.property
 106   rw [Fintype.card_congr (Equiv.subtypeEquivRight hp)]
 107   calc
 108     _ ≤ ∑ v : Z,Fintype.card {D : LabelledAbsorberFamily R S X //
 109         v.val∈labelledAbsorberFree R (S D.1) X D.2} := by
 110       convert finite_exists_card_le_sum (fun (v : Z) (D : LabelledAbsorberFamily R S X) =>
 111         v.val∈labelledAbsorberFree R (S D.1) X D.2) using 1 <;> congr!
 112     _ ≤ ∑ _v : Z,(c*c)*Fintype.card T*X.card^c*Fintype.card V^(c*c-1)*J^(2*c+1) :=
 113       Finset.sum_le_sum (fun v _ => labelled_absorber_free_vertex_count R S X J hJ v.val)
