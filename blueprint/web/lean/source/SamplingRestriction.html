  12 theorem finiteLaw_pi_restrict
  13     {E : Type*} [Fintype E] [DecidableEq E]
  14     {State : E → Type*} [∀ e, Fintype (State e)]
  15     (μ : ∀ e, FiniteLaw (State e)) (A : Finset E) :
  16     (FiniteLaw.pi μ).map (fun σ (e : A) => σ e) =
  17       FiniteLaw.pi (fun e : A => μ e) := by
  18   apply finiteLaw_ext
  19   intro τ
  20   let s (e : E) : State e := if he : e ∈ A then τ ⟨e, he⟩ else Classical.choice (finiteLaw_nonempty (μ e))
  21   have heq (σ : ∀ e, State e) : (fun e : A => σ e) = τ ↔ ∀ e ∈ A, σ e = s e := by
  22     constructor
  23     · intro h e he
  24       simpa only [s, dite_eq_left he] using congrFun h ⟨e, he⟩
  25     · intro h
  26       funext e
  27       simpa only [s, dite_eq_left e.2] using h e e.2
  28   have hp (σ : ∀ e, State e) : patternValue A s σ =
  29       if (fun e : A => σ e) = τ then 1 else 0 := by
  30     by_cases h : (fun e : A => σ e) = τ
  31     · have hh := heq σ |>.mp h
  32       simp only [h, ite_true]
  33       unfold patternValue
  34       exact Finset.prod_eq_one (fun e he => by simp [hh e he])
  35     · have hh : ∃ e ∈ A, σ e ≠ s e := by
  36         by_contra hn
  37         apply h
  38         exact (heq σ).mpr (by simpa using hn)
  39       obtain ⟨e, he, hne⟩ := hh
  40       simp only [h, ite_false]
  41       unfold patternValue
  42       exact Finset.prod_eq_zero he (by simp [hne])
  43   have hm : ((FiniteLaw.pi μ).map (fun σ (e : A) => σ e)).mass τ =
  44       (FiniteLaw.pi μ).expectation (patternValue A s) := by
  45     dsimp only [FiniteLaw.map, FiniteLaw.expectation]
  46     apply Finset.sum_congr rfl
  47     intro σ _
  48     rw [hp]
  49     split_ifs <;> simp
  50   rw [hm, finiteLaw_expectation_pattern]
  51   dsimp only [FiniteLaw.pi]
  52   rw [Finset.prod_subtype A (fun _ => Iff.rfl)]
  53   apply Finset.prod_congr rfl
  54   intro e _
  55   simp [s, e.2]
