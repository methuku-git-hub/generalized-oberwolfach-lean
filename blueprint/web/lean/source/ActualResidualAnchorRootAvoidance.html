  17     (fC:{x // x∉bufferedPathStrictInterior n inner}↪V)

  20     (hold:∀(x:{x // x∉bufferedPathStrictInterior n inner}),x.val∈data.oldDomain→fC x=ψ x.val)
  21     (hnew:∀(x:{x // x∉bufferedPathStrictInterior n inner}) (hx:x.val∈data.remaining),fC x=Φ ⟨x.val,hx⟩)
  22     (hAnchor:∀x∈B,Φ x=g x) (hFree:∀x,x∉B→Φ x∈Lists)
  23     (hImage:B.image g=(residualSubtypeSet data.remaining data.prescribed).image (fun x=>ψ x.1)∪E∪Z)
  24     (hEAvoid:Disjoint E Y) (hZAvoid:Disjoint Z Y) (hListAvoid:Disjoint Lists Y)
  25     (hOldAvoid:∀x,x∈data.oldDomain→x∉bufferedPathFullDomain n inner→ψ x∉Y):
  26     Disjoint (outsideBufferedPathImage n inner fC) Y:=by

  30   rw [hImage] at hg
  31   rcases Finset.mem_union.mp hg with hleft|hZ
  32   · rcases Finset.mem_union.mp hleft with hpres|hE
  33     · obtain ⟨y,hy,heq⟩:=Finset.mem_image.mp hpres
  34       have hyPres:y.1∈data.prescribed:=(Finset.mem_filter.mp hy).2
  35       have hyOld:y.1∈data.oldDomain:=by

  42         intro h;exact Finset.disjoint_left.mp hDR h y.property
  43       have hf:fC ⟨x.1,hxNot⟩=fC ⟨y.1,hyNot⟩:=by

  45           fC ⟨x.1,hxNot⟩ = Φ x := hnew ⟨x.1,hxNot⟩ x.property
  46           _ = g x := hAnchor x hx
