  14 theorem subtype_pair_filter_row {A B : Type*} [Fintype A] [Fintype B]
  15     (P Q : A → B → Prop) (a : A) :
  16     ((Finset.univ.filter (fun z : {z : A×B // P z.1 z.2}=>z.val.1=a ∧ Q z.val.1 z.val.2)).card:ℝ)=
  17       ∑ b,if P a b ∧ Q a b then (1:ℝ) else 0 := by
  18   have hh := finite_sum_restrict_of_zero (fun z : A×B=>P z.1 z.2)
  19     (fun z=>if P z.1 z.2 ∧ z.1=a ∧ Q z.1 z.2 then (1:ℝ) else 0)
  20     (by intro z hz;simp only [hz,false_and,ite_false])
  21   have he : (∑ z : {z : A×B // P z.1 z.2},if P z.val.1 z.val.2 ∧ z.val.1=a ∧ Q z.val.1 z.val.2 then (1:ℝ) else 0)=
  22       ∑ z : {z : A×B // P z.1 z.2},if z.val.1=a ∧ Q z.val.1 z.val.2 then (1:ℝ) else 0 := by
  23     apply Finset.sum_congr rfl
  24     intro z _
  25     simp only [z.property,true_and]

  27   rw [← Finset.sum_boole,← hh,Fintype.sum_prod_type]
  28   rw [Finset.sum_eq_single a]
  29   · simp only [true_and]
  30   · intro x _ hxa
  31     apply Finset.sum_eq_zero

  51 theorem deleted_occurrence_incidence_le {A V : Type*} [Fintype A]
  52     (f g : A → V) (P : A → Prop) (v : V) (d : ℝ)
  53     (hf : (∑ e,if ¬P e ∧ f e=v then (1:ℝ) else 0)≤d)
  54     (hg : (∑ e,if ¬P e ∧ g e=v then (1:ℝ) else 0)≤d) :
  55     ((Finset.univ.filter (fun e=>(f e=v ∨ g e=v) ∧ ¬P e)).card:ℝ)≤2*d := by
  56   rw [← Finset.sum_boole]
  57   calc
  58     _ ≤ ∑ e,((if ¬P e ∧ f e=v then (1:ℝ) else 0)+(if ¬P e ∧ g e=v then 1 else 0)) := by
  59       apply Finset.sum_le_sum
  60       intro e _
  61       by_cases hp : P e <;> by_cases hfe : f e=v <;> by_cases hge : g e=v <;> simp [hp,hfe,hge]
  62     _ ≤ 2*d := by rw [Finset.sum_add_distrib];linarith only [hf,hg]
