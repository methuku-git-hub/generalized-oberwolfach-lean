  13 theorem finiteLaw_bounded_certificate_tail
  14     {I Ω : Type*} [Fintype I] [Fintype Ω]
  15     (μ : I → FiniteLaw Ω) (R : (I → Ω) → ℝ)
  16     {c D M t : ℝ} (hc : 0<c) (hD : 0<D) (ht : 0≤t)
  17     (hcert : HasCoordinateCertificateBound R c D)
  18     (hrange : ∀ x,0≤R x ∧ R x≤M)
  19     (hsmall : M*(4*Real.exp (-t^2/(4*c*D)))≤t) :
  20     (FiniteLaw.pi μ).probability
  21       (fun x=>3*t≤|R x-(FiniteLaw.pi μ).expectation R|)≤
  22         4*Real.exp (-t^2/(4*c*D)) := by
  23   classical
  24   let ν := FiniteLaw.pi μ
  25   obtain ⟨z,hlo,hhi⟩ := finiteLaw_exists_median ν R
  26   let m := R z
  27   have hM0 : 0≤M := (hrange z).1.trans (hrange z).2
  28   have hbound (x : I → Ω) : |R x-m|≤M := by
  29     apply abs_le.mpr
  30     constructor <;> dsimp [m] <;> linarith [(hrange x).1,(hrange x).2,(hrange z).1,(hrange z).2]
  31   have hmed := finiteLaw_certificate_median_tail μ R hc hD ht hcert hlo hhi
  32   have hm : |ν.expectation R-m|≤2*t := by
  33     have he : ν.expectation (fun x=>R x-m)=ν.expectation R-m := by
  34       unfold FiniteLaw.expectation
  35       simp only [mul_sub,Finset.sum_sub_distrib,← Finset.sum_mul,ν.total,one_mul]
  36     have ha := finiteLaw_abs_expectation_le ν (fun x=>R x-m)
  37     rw [he] at ha
  38     have hb := finiteLaw_expectation_abs_truncate ν (fun x=>R x-m) ht hbound
  39     have hp := mul_le_mul_of_nonneg_left hmed hM0
  40     linarith
  41   have hsubset (x : I → Ω) (hx : 3*t≤|R x-ν.expectation R|) :
  42       t≤|R x-m| := by
  43     have htri := abs_sub_le (R x) m (ν.expectation R)
  44     rw [abs_sub_comm m (ν.expectation R)] at htri
  45     linarith
  46   exact (finiteLaw_probability_mono ν _ _ hsubset).trans hmed
