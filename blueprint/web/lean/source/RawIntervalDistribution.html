  28 theorem finiteLaw_exclusive_union {Ω J : Type*} [Fintype Ω]
  29     (μ : FiniteLaw Ω) (T : Finset J) (E : J → Ω → Prop)
  30     (hunique : ∀ ω,∀ i∈T,∀ j∈T,E i ω → E j ω → i=j) :
  31     μ.probability (fun ω=>∃ i∈T,E i ω)=∑ i∈T,μ.probability (E i) := by
  32   classical
  33   unfold FiniteLaw.probability
  34   rw [Finset.sum_comm]
  35   apply Finset.sum_congr rfl
  36   intro ω _
  37   by_cases he : ∃ i∈T,E i ω
  38   · obtain ⟨i,hi,hei⟩ := he
  39     rw [if_pos ⟨i,hi,hei⟩]
  40     symm
  41     rw [Finset.sum_eq_single i]
  42     · exact if_pos hei
  43     · intro j hj hji
  44       exact if_neg (fun hej=>hji (hunique ω j hj i hi hej hei))
  45     · intro h;exact (h hi).elim
  46   · have hn (i : J) (hi : i∈T) : ¬E i ω := fun h=>he ⟨i,hi,h⟩
  47     rw [if_neg he]
  48     symm
  49     exact Finset.sum_eq_zero (fun i hi=>if_neg (hn i hi))

  60 theorem rawIntervalKept_probability
  61     (hnoshort : ∀ x k,0<k → k<2*d+1 → (π^k) x≠x) (a : V) :
  62     (FiniteLaw.pi μraw).probability (fun σ=>rawIntervalKept π d (fun x=>σ (Sum.inl x)) a)=
  63       isolatedStartProbability d := by
  64   let T := permutationWindow π d d a
  65   let R : Finset (RawIntervalCoordinate V) := T.image Sum.inl
  66   let value : RawIntervalOutcome V Γ Color := fun e=>match e with
  67     | Sum.inl x=>decide (x=a)
  68     | Sum.inr (Sum.inl _)=>none
  69     | Sum.inr (Sum.inr _)=>none
  70   have hself : a∈T := permutationWindow_self π d d a
  71   have hpred (σ : RawIntervalOutcome V Γ Color) :
  72       rawIntervalKept π d (fun x=>σ (Sum.inl x)) a ↔ ∀ e∈R,σ e=value e := by
  73     constructor
  74     · intro h e he
  75       obtain ⟨x,hx,rfl⟩ := Finset.mem_image.mp he
  76       by_cases hx0 : x=a
  77       · subst x;simpa [value] using h.1
  78       · simpa [value,hx0] using h.2 x hx hx0
  79     · intro h
  80       refine ⟨?_,?_⟩
  81       · have hh:=h (Sum.inl a) (Finset.mem_image.mpr ⟨a,hself,rfl⟩)
  82         simpa [value] using hh
  83       · intro x hx hxa
  84         have hh:=h (Sum.inl x) (Finset.mem_image.mpr ⟨x,hx,rfl⟩)
  85         simpa [value,hxa] using hh
  86   have hp : (FiniteLaw.pi μraw).probability (fun σ=>rawIntervalKept π d (fun x=>σ (Sum.inl x)) a)=
  87       (FiniteLaw.pi μraw).expectation (patternValue R value) := by
  88     unfold FiniteLaw.probability FiniteLaw.expectation
  89     apply Finset.sum_congr rfl
  90     intro σ _
  91     rw [patternValue_indicator]
  92     simp only [hpred,mul_ite,mul_one,mul_zero]
  93   rw [hp,finiteLaw_expectation_pattern]
  94   dsimp only [R]
  95   rw [Finset.prod_image (by intro x _ y _ h;exact Sum.inl.inj h)]
  96   have hmass (x : V) : (μraw (Sum.inl x)).mass (value (Sum.inl x))=
  97       if x=a then intervalActivation d else 1-intervalActivation d := by
  98     by_cases hxa : x=a <;> simp [rawIntervalCoordinateLaw,FiniteLaw.retentionCoin,value,hxa]
  99   simp_rw [hmass]
 100   rw [← Finset.mul_prod_erase _ _ hself]
 101   simp only [if_pos rfl]
 102   have hprod : (∏ x∈T.erase a,if x=a then intervalActivation d else 1-intervalActivation d)=
 103       (1-intervalActivation d)^(2*d) := by
 104     have hc : T.card=2*d+1 := by
 105       simpa only [T,show d+d=2*d by omega] using permutationWindow_card π d d
 106         (by intro x k hk hkb;exact hnoshort x k hk (by omega)) a
 107     have herase : (T.erase a).card=2*d := by rw [Finset.card_erase_of_mem hself,hc];omega
 108     simp only [Finset.prod_congr rfl (fun x hx=>if_neg (Finset.mem_erase.mp hx).1),
 109       Finset.prod_const,herase]
 110   rw [hprod]
 111   rfl
