  16 theorem graph_iSup_local_degree_bound (F : J→SimpleGraph V) (T : Finset V) (v : V) (D : ℕ)
  17     (hD : ∀ j,(F j).degree v ≤ D) :
  18     (((⨆ j,F j).neighborFinset v)∩T).card ≤
  19       D*(Finset.univ.filter (fun j=>∃ w∈T,(F j).Adj v w)).card := by
  20   let A := Finset.univ.filter (fun j=>∃ w∈T,(F j).Adj v w)
  21   have he : ((⨆ j,F j).neighborFinset v)∩T=A.biUnion (fun j=>(F j).neighborFinset v∩T) := by
  22     ext w
  23     simp only [Finset.mem_inter,SimpleGraph.mem_neighborFinset,SimpleGraph.iSup_adj,Finset.mem_biUnion,A,
  24       Finset.mem_filter,Finset.mem_univ,true_and]
  25     constructor
  26     · rintro ⟨⟨j,hj⟩,hw⟩;exact ⟨j,⟨w,hw,hj⟩,hj,hw⟩
  27     · rintro ⟨j,hj,hjw,hw⟩;exact ⟨⟨j,hjw⟩,hw⟩
  28   rw [he]
  29   calc
  30     _ ≤ ∑ j∈A,((F j).neighborFinset v∩T).card := Finset.card_biUnion_le
  31     _ ≤ ∑ _j∈A,D := Finset.sum_le_sum (fun j _=>
  32       ((Finset.card_le_card (Finset.inter_subset_left)).trans (by simpa using hD j)))
  33     _ = _ := by simp [A,Nat.mul_comm]

  35 theorem graph_iSup_global_degree_bound (F : J→SimpleGraph V) (S : J→Finset V) (D : ℕ)
  36     (hsup : ∀ j,GraphSupportedOn (S j) (F j)) (hD : ∀ j v,(F j).degree v ≤ D) (v : V) :
  37     (⨆ j,F j).degree v ≤ D*(Finset.univ.filter (fun j=>v∈S j)).card := by
  38   have hh := graph_iSup_local_degree_bound F Finset.univ v D (fun j=>hD j v)
  39   simp only [Finset.inter_univ,SimpleGraph.card_neighborFinset_eq_degree] at hh
  40   apply hh.trans
  41   apply Nat.mul_le_mul_left
  42   apply Finset.card_le_card
  43   intro j hj
  44   obtain ⟨w,hw,hvw⟩ := (Finset.mem_filter.mp hj).2
  45   exact Finset.mem_filter.mpr ⟨Finset.mem_univ _,(hsup j v w hvw).1⟩

  47 theorem graphSelection_counter_eq (S : Fin N→Finset V) (F : Fin N→SimpleGraph V)
  48     (x : SelectionVertexTest S)
  49     (hfuture : ∀ j,x.val.1.val ≤ j.val → F j=⊥)
  50     (hsup : ∀ j,GraphSupportedOn (S j) (F j)) :
  51     (Finset.univ.filter (fun j=>∃ w∈S x.val.1,(F j).Adj x.val.2 w)).card=
  52       selectionCounter (graphSelectionActive S) (graphSelectionHit S) F x := by
  53   apply congrArg Finset.card
  54   ext j
  55   simp only [selectionCounter,Finset.mem_filter,Finset.mem_univ,true_and,graphSelectionActive,graphSelectionHit]
  56   constructor
  57   · rintro ⟨w,hw,hvw⟩
  58     have hji : j.val < x.val.1.val := by
  59       by_contra hn
  60       have hj := hfuture j (by omega)
  61       rw [hj] at hvw
  62       exact hvw.elim
  63     exact ⟨⟨hji,(hsup j x.val.2 w hvw).1⟩,w,hw,hvw⟩
  64   · rintro ⟨ha,hh⟩;exact hh

  66 theorem spanningRestriction_degree (F : SimpleGraph V) (S : Finset V) (v : V) :
  67     (spanningRestriction F S).degree v=
  68       if v∈S then (F.neighborFinset v∩S).card else 0 := by
  69   have he : (spanningRestriction F S).neighborFinset v=
  70       if v∈S then F.neighborFinset v∩S else ∅ := by

  72     by_cases hv : v∈S <;> simp [SimpleGraph.mem_neighborFinset,spanningRestriction_adj,hv]
