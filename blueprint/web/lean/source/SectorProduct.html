  10 theorem sector_product_cancel {I : Type*} [Fintype I] [DecidableEq I]
  11     (c : I) (H R : I → ℝ) (Q C M : ℝ)
  12     (hH : ∀ i, i ≠ c → H i ≠ 0) (hQ : Q ≠ 0) (hC : C ≠ 0) (hM : M ≠ 0) :
  13     ((∏ i ∈ Finset.univ.erase c, H i) / M) *
  14       (∏ i, if i = c then (Q / C) / Q else R i / H i) =
  15       (∏ i ∈ Finset.univ.erase c, R i) / (C * M) := by
  16   classical
  17   rw [← Finset.mul_prod_erase Finset.univ
  18     (fun i => if i = c then (Q / C) / Q else R i / H i) (Finset.mem_univ c)]
  19   simp only [ite_true]
  20   have he :
  21       (∏ i ∈ Finset.univ.erase c, if i = c then (Q / C) / Q else R i / H i) =
  22         ∏ i ∈ Finset.univ.erase c, R i / H i := by
  23     apply Finset.prod_congr rfl
  24     intro i hi
  25     simp only [(Finset.mem_erase.mp hi).1, ite_false]
  26   rw [he, Finset.prod_div_distrib]
  27   have hp : (∏ i ∈ Finset.univ.erase c, H i) ≠ 0 :=
  28     Finset.prod_ne_zero_iff.mpr (fun i hi => hH i (Finset.mem_erase.mp hi).1)
  29   field_simp

  38 theorem walkTransport_pos {I X : Type*} [Fintype X] [Nonempty X]
  39     (m : I → X → ℝ) (K : I → X → X → ℝ) (s : I → I)
  40     (hm : ∀ i x, 0 < m i x) (hK : ∀ i x z, 0 < K i x z)
  41     (i : I) (j : ℕ) (x z : X) :
  42     0 < walkTransport m K s i (j + 1) x z := by
  43   induction j generalizing i x z with
  44   | zero => rw [walkTransport_one m K s hm]; exact hK i x z
  45   | succ j ih =>
  46     change 0 < composeTransport (m (s i)) (K i)
  47       (walkTransport m K s (s i) (j + 1)) x z
  48     apply Finset.sum_pos _ Finset.univ_nonempty
  49     intro y _
  50     exact div_pos (mul_pos (hK i x y) (ih (s i) y z)) (hm (s i) y)
