  13 theorem matching_integer_root_scale (a H n : ℕ) (ha : 0<a) (hH : 2≤H)
  14     (hn : H^a≤n) :
  15     let h := Nat.nthRoot a n
  16     H≤h ∧ h^a≤n ∧ n≤h^(2*a) := by
  17   dsimp only
  18   have hHh : H≤Nat.nthRoot a n := (Nat.le_nthRoot_iff (by omega)).mpr hn
  19   have hh : 2≤Nat.nthRoot a n := hH.trans hHh
  20   refine ⟨hHh,(Nat.pow_nthRoot_le_iff).mpr (Or.inl (by omega)),?_⟩
  21   have hnext := (Nat.lt_pow_nthRoot_add_one (by omega : a≠0) n).le
  22   have hs : Nat.nthRoot a n+1≤(Nat.nthRoot a n)^2 := by nlinarith
  23   calc
  24     _ ≤ (Nat.nthRoot a n+1)^a := hnext
  25     _ ≤ ((Nat.nthRoot a n)^2)^a := Nat.pow_le_pow_left hs a
  26     _ = _ := (pow_mul _ _ _).symm
