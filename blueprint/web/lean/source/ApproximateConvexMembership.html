  12 theorem functional_unit_ball_maximum
  13     {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E] [FiniteDimensional ℝ E]
  14     (f : E →L[ℝ] ℝ) :
  15     ∃ z : E, ‖z‖ ≤ 1 ∧ 0 ≤ f z ∧ ∀ e : E, f e ≤ f z * ‖e‖ := by
  16   obtain ⟨z, hz, hmax⟩ := (isCompact_closedBall (0 : E) 1).exists_isMaxOn
  17     (show (Metric.closedBall (0 : E) 1).Nonempty from ⟨0, by simp⟩)
  18     f.continuous.continuousOn
  19   have hz' : ‖z‖ ≤ 1 := by simpa using hz
  20   have hd : 0 ≤ f z := by simpa using hmax (show (0 : E) ∈ Metric.closedBall 0 1 by simp)
  21   refine ⟨z, hz', hd, ?_⟩
  22   intro e
  23   by_cases he : e = 0
  24   · simp [he]
  25   have hepos : 0 < ‖e‖ := norm_pos_iff.mpr he
  26   have heball : ‖e‖⁻¹ • e ∈ Metric.closedBall (0 : E) 1 := by
  27     simp only [Metric.mem_closedBall, dist_zero_right, norm_smul, Real.norm_eq_abs,
  28       abs_of_pos (inv_pos.mpr hepos), inv_mul_cancel₀ hepos.ne', le_refl]
  29   have h := hmax heball
  30   change f (‖e‖⁻¹ • e) ≤ f z at h
  31   rw [map_smul, smul_eq_mul] at h
  32   have hm := mul_le_mul_of_nonneg_left h hepos.le
  33   simpa only [← mul_assoc, mul_inv_cancel₀ hepos.ne', one_mul, mul_comm] using hm

  35 theorem exact_membership_of_nearby_approximations
  36     {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E] [FiniteDimensional ℝ E]
  37     (Z : Set E) (hZconv : Convex ℝ Z) (hZclosed : IsClosed Z)
  38     (w : E) (κ ε : ℝ) (hκ : 0 < κ) (hεκ : ε < κ)
  39     (happrox : ∀ w' : E, ‖w' - w‖ ≤ κ → ∃ y ∈ Z, ‖y - w'‖ ≤ ε) : w ∈ Z := by
  40   by_contra hnot
  41   obtain ⟨f, u, hfu, huf⟩ := geometric_hahn_banach_closed_point hZconv hZclosed hnot
  42   obtain ⟨z, hz, hd, hbound⟩ := functional_unit_ball_maximum f
  43   have hnear : ‖(w + κ • z) - w‖ ≤ κ := by
  44     simp only [add_sub_cancel_left, norm_smul, Real.norm_eq_abs, abs_of_pos hκ]
  45     nlinarith
  46   obtain ⟨y, hy, herror⟩ := happrox (w + κ • z) hnear
  47   have herror' : ‖(w + κ • z) - y‖ ≤ ε := by rwa [norm_sub_rev]
  48   have hdual := hbound ((w + κ • z) - y)
  49   rw [map_sub, map_add, map_smul, smul_eq_mul] at hdual
  50   have hscaled := mul_le_mul_of_nonneg_left herror' hd
  51   have hpositive := mul_nonneg (le_of_lt (sub_pos.mpr hεκ)) hd
  52   have hsep := hfu y hy
  53   nlinarith
