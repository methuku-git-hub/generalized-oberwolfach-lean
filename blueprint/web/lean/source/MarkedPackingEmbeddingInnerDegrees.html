  13 theorem spanningRestriction_map_preimage {V W:Type} [DecidableEq V] [DecidableEq W]
  14     (G:SimpleGraph V) (f:V↪W) (X:Finset V) (I:Finset W)
  15     (hX:∀ v,f v∈I↔v∈X):
  16     spanningRestriction (G.map f) I=(spanningRestriction G X).map f:=by
  17   ext a b
  18   rw [spanningRestriction_adj,SimpleGraph.map_adj,SimpleGraph.map_adj]
  19   constructor
  20   · rintro ⟨⟨v,w,hvw,rfl,rfl⟩,hv,hw⟩
  21     exact ⟨v,w,(spanningRestriction_adj G X v w).mpr ⟨hvw,(hX v).mp hv,(hX w).mp hw⟩,rfl,rfl⟩
  22   · rintro ⟨v,w,hvw,rfl,rfl⟩
  23     obtain ⟨hvw,hv,hw⟩:=(spanningRestriction_adj G X v w).mp hvw
  24     exact ⟨⟨v,w,hvw,rfl,rfl⟩,(hX v).mpr hv,(hX w).mpr hw⟩

  33 theorem markedPackingResources_embedding_inner_degrees
  34     (f:V↪V') (g:U↪U')
  35     (M:Finset (MarkedWheel tail head ktail khead m π b H₀ HK))
  36     (N:Finset (MarkedWheel tail' head' ktail' khead' m' π' b' H₀' HK'))
  37     (he:N.biUnion markedWheelPhysicalResources=(M.biUnion markedWheelPhysicalResources).image (markedResourceEmbedding f g))
  38     (X:Finset V) (Z:Finset U) (I:Finset V') (J:Finset U')
  39     (hX:∀ x,f x∈I↔x∈X) (hZ:∀ u,g u∈J↔u∈Z):
  40     (∀ v,(spanningRestriction (markedPackingRimGraph N) I).degree (f v)=
  41       (spanningRestriction (markedPackingRimGraph M) X).degree v)∧
  42     (∀ v,(J.filter (markedPackingSpokes N (f v))).card=(Z.filter (markedPackingSpokes M v)).card)∧
  43     (∀ u,(I.filter (fun x=>markedPackingSpokes N x (g u))).card=(X.filter (fun x=>markedPackingSpokes M x u)).card):=by
  44   obtain ⟨hr,hs⟩:=markedPackingResources_embedding_graphs f g M N he
  45   refine ⟨?_,?_,?_⟩
  46   · intro v
  47     simp only [←SimpleGraph.ncard_neighborSet]
  48     rw [hr,spanningRestriction_map_preimage _ f X I hX,SimpleGraph.neighborSet_map,
  49       Set.ncard_image_of_injective _ f.injective]
  50   · intro v
  51     have hh:J.filter (markedPackingSpokes N (f v))=(Z.filter (markedPackingSpokes M v)).image g:=by
  52       ext u
  53       simp only [Finset.mem_filter,Finset.mem_image]
  54       constructor
  55       · rintro ⟨hu,hs'⟩
  56         obtain ⟨v',w,hv,hw,hm⟩:=(hs _ _).mp hs'
  57         have hv':v'=v:=f.injective hv
  58         subst v'
  59         exact ⟨w,⟨(hZ w).mp (hw.symm ▸ hu),hm⟩,hw⟩
  60       · rintro ⟨w,⟨hw,hm⟩,rfl⟩
  61         exact ⟨(hZ w).mpr hw,(hs _ _).mpr ⟨v,w,rfl,rfl,hm⟩⟩
  62     rw [hh,Finset.card_image_of_injective _ g.injective]
  63   · intro u
  64     have hh:I.filter (fun x=>markedPackingSpokes N x (g u))=(X.filter (fun x=>markedPackingSpokes M x u)).image f:=by
  65       ext v
  66       simp only [Finset.mem_filter,Finset.mem_image]
  67       constructor
  68       · rintro ⟨hv,hs'⟩
  69         obtain ⟨w,u',hw,hu,hm⟩:=(hs _ _).mp hs'
  70         have hu':u'=u:=g.injective hu
  71         subst u'
  72         exact ⟨w,⟨(hX w).mp (hw.symm ▸ hv),hm⟩,hw⟩
  73       · rintro ⟨w,⟨hw,hm⟩,rfl⟩
  74         exact ⟨(hX w).mpr hw,(hs _ _).mpr ⟨w,u,rfl,rfl,hm⟩⟩
  75     rw [hh,Finset.card_image_of_injective _ f.injective]
