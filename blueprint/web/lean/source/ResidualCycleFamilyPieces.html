   9 theorem retained_cycle_family_source_pieces (H:SimpleGraph V) (R:Finset V)
  10     (hisol:∀x,x∉R→∀y,¬H.Adj x y) (D:V→V→Prop):
  11     ∀ orig:List (PrescribedSourcePiece V),
  12     orig.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset)→
  13     (∀p∈orig,∃a xs,p=.cycle a xs ∧ p.OriginalValid D)→
  14     ∃ ps:List (PrescribedSourcePiece V),
  15       ps.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset) ∧
  16       (∀p∈ps,p.OriginalValid D) ∧
  17       (∀x,x∈sourcePieceVerticesUnion ps ↔ x∈sourcePieceVerticesUnion orig ∧ x∈R) ∧
  18       sourcePieceEdgesUnion ps=(sourcePieceEdgesUnion orig).filter (fun e=>e∈H.edgeFinset) := by
  19   intro orig
  20   induction orig with
  21   | nil =>
  22     intro _ _
  23     refine ⟨[],by simp,by simp,?_,?_⟩
  24     · intro x;simp [sourcePieceVerticesUnion]
  25     · simp [sourcePieceEdgesUnion]
  26   | cons p orig ih =>
  27     intro hpair hvalid
  28     have hpairs:=List.pairwise_cons.mp hpair
  29     obtain ⟨a,xs,hp,hcycle⟩:=hvalid p (by simp)
  30     subst p
  31     obtain ⟨front,hfront,hfrontvalid,hfrontV,hfrontE⟩:=
  32       retained_cycle_source_pieces H R hisol D a xs hcycle.1 hcycle.2.1 hcycle.2.2
  33     have htailvalid:∀p∈orig,∃a xs,p=.cycle a xs ∧ p.OriginalValid D := by
  34       intro q hq
  35       exact hvalid q (by simp [hq])
  36     obtain ⟨tail,htail,htailvalid',htailV,htailE⟩:=ih hpairs.2 htailvalid
  37     refine ⟨front++tail,?_,?_,?_,?_⟩
  38     · rw [List.pairwise_append]
  39       refine ⟨hfront,htail,?_⟩
  40       intro p hp q hq
  41       apply Finset.disjoint_left.mpr
  42       intro x hxp hxq
  43       have hxfront:x∈sourcePieceVerticesUnion front := source_piece_vertex_mem_union front p hp x (List.mem_toFinset.mp hxp)
  44       have hxtail:x∈sourcePieceVerticesUnion tail := source_piece_vertex_mem_union tail q hq x (List.mem_toFinset.mp hxq)
  45       have hxa:x∈a::xs := (hfrontV x).1 hxfront |>.1
  46       have hxorig:x∈sourcePieceVerticesUnion orig := (htailV x).1 hxtail |>.1
  47       obtain ⟨q',hq',hxq'⟩:=Finset.mem_biUnion.mp hxorig
  48       have hdis:=hpairs.1 q' (List.mem_toFinset.mp hq')
  49       exact Finset.disjoint_left.mp hdis (List.mem_toFinset.mpr hxa) hxq'
  50     · intro q hq
  51       rcases List.mem_append.mp hq with hq|hq
  52       · exact hfrontvalid q hq
  53       · exact htailvalid' q hq
  54     · intro x
  55       rw [source_piece_vertices_union_append]
  56       have horig:sourcePieceVerticesUnion (.cycle a xs::orig) =
  57           (a::xs).toFinset∪sourcePieceVerticesUnion orig := by
  58         simp [sourcePieceVerticesUnion,PrescribedSourcePiece.vertices]
  59       rw [horig]
  60       simp only [Finset.mem_union,List.mem_toFinset,hfrontV x,htailV x]
  61       tauto
  62     · rw [source_piece_edges_union_append,hfrontE,htailE]
  63       have horig:sourcePieceEdgesUnion (.cycle a xs::orig) =
  64           prescribedOrderedPathEdges ((a::xs)++[a])∪sourcePieceEdgesUnion orig := by
  65         simp [sourcePieceEdgesUnion,PrescribedSourcePiece.edges]
  66       rw [horig,Finset.filter_union]
  67 end Oberwolfach
