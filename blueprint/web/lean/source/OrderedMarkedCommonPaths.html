  14 def rootedCycleOrder {V : Type*} {m : ℕ} (σ : Equiv.Perm (Fin (m+1))) (v : V)
  15     (z : Fin m → V) : Fin (m+1) → V := fun k=>Fin.cases v z (σ k)

  17 abbrev OrderedCommonMarkedPath {V U E K : Type*}
  18     (tail head : E → V) (ktail khead : K → V) (m : ℕ) (π : Equiv.Perm V) (b : ℕ)
  19     (H₀ Hₖ : V → U → Prop) (σ : Equiv.Perm (Fin (m+1))) (v v' : V) (u u' : U) :=
  20   {z : Fin m → V //
  21     IsMarkedCycleTuple tail head ktail khead m π b (rootedCycleOrder σ v z) ∧
  22     IsMarkedCycleTuple tail head ktail khead m π b (rootedCycleOrder σ v' z) ∧
  23     ∀ l : Fin m,
  24       (if markedPositionType (σ.symm l.succ) then Hₖ (z l) u else H₀ (z l) u) ∧
  25       (if markedPositionType (σ.symm l.succ) then Hₖ (z l) u' else H₀ (z l) u')}

  27 def orderedCommonMarkedPath_to_pair {V U E K : Type*}
  28     {tail head : E → V} {ktail khead : K → V} {m : ℕ} {π : Equiv.Perm V} {b : ℕ}
  29     {H₀ Hₖ : V → U → Prop} {σ : Equiv.Perm (Fin (m+1))} {v v' : V} {u u' : U}
  30     (hroot : (if markedPositionType (σ.symm 0) then Hₖ v u else H₀ v u) ∧
  31       (if markedPositionType (σ.symm 0) then Hₖ v u' else H₀ v u') ∧
  32       (if markedPositionType (σ.symm 0) then Hₖ v' u else H₀ v' u) ∧
  33       (if markedPositionType (σ.symm 0) then Hₖ v' u' else H₀ v' u')) :
  34     OrderedCommonMarkedPath tail head ktail khead m π b H₀ Hₖ σ v v' u u' ↪
  35       MarkedReplacementPair tail head ktail khead m π b H₀ Hₖ (σ.symm 0) v v' u u' where
  36   toFun z := by
  37     have hcompat (w : V) (h : U)
  38         (hr : if markedPositionType (σ.symm 0) then Hₖ w h else H₀ w h)
  39         (hf : ∀ l : Fin m,if markedPositionType (σ.symm l.succ) then Hₖ (z.val l) h else H₀ (z.val l) h) :
  40         MarkedHubCompatible H₀ Hₖ (rootedCycleOrder σ w z.val) h := by
  41       apply (markedHubCompatible_iff_positions _ _ _ _).mpr
  42       intro k
  43       have hk := Fin.eq_zero_or_eq_succ (σ k)
  44       rcases hk with hk|⟨l,hl⟩
  45       · have hk' : k=σ.symm 0 := σ.injective (by simpa only [Equiv.apply_symm_apply] using hk)
  46         subst k
  47         simpa only [rootedCycleOrder,Equiv.apply_symm_apply,Fin.cases_zero] using hr
  48       · have hk' : k=σ.symm l.succ := σ.injective (by simpa only [Equiv.apply_symm_apply] using hl)
  49         subst k
  50         simpa only [rootedCycleOrder,Equiv.apply_symm_apply,Fin.cases_succ] using hf l
  51     refine ⟨⟨⟨rootedCycleOrder σ v z.val,z.property.1⟩,⟨rootedCycleOrder σ v' z.val,z.property.2.1⟩⟩,
  52       ?_,?_,?_,hcompat v u hroot.1 (fun l=>(z.property.2.2 l).1),
  53       hcompat v u' hroot.2.1 (fun l=>(z.property.2.2 l).2),
  54       hcompat v' u hroot.2.2.1 (fun l=>(z.property.2.2 l).1),
  55       hcompat v' u' hroot.2.2.2 (fun l=>(z.property.2.2 l).2)⟩
  56     · simp only [rootedCycleOrder,Equiv.apply_symm_apply,Fin.cases_zero]
  57     · simp only [rootedCycleOrder,Equiv.apply_symm_apply,Fin.cases_zero]
  58     · intro k hk
  59       have hσ : σ k≠0 := by intro he;exact hk (σ.injective (by simpa only [Equiv.apply_symm_apply] using he))
  60       obtain ⟨l,hl⟩ := Fin.eq_succ_of_ne_zero hσ
  61       simp only [rootedCycleOrder,hl,Fin.cases_succ]
  62   inj' z w he := by
  63     apply Subtype.ext
  64     funext l
  65     have hc := congrArg (fun P=>P.val.1.val (σ.symm l.succ)) he
  66     simpa only [rootedCycleOrder,Equiv.apply_symm_apply,Fin.cases_succ] using hc

  68 theorem orderedCommonMarkedPath_card_le {V U E K : Type*} [Fintype V]
  69     {tail head : E → V} {ktail khead : K → V} {m : ℕ} {π : Equiv.Perm V} {b : ℕ}
  70     {H₀ Hₖ : V → U → Prop} {σ : Equiv.Perm (Fin (m+1))} {v v' : V} {u u' : U}
  71     (hroot : (if markedPositionType (σ.symm 0) then Hₖ v u else H₀ v u) ∧
  72       (if markedPositionType (σ.symm 0) then Hₖ v u' else H₀ v u') ∧
  73       (if markedPositionType (σ.symm 0) then Hₖ v' u else H₀ v' u) ∧
  74       (if markedPositionType (σ.symm 0) then Hₖ v' u' else H₀ v' u')) :
  75     Fintype.card (OrderedCommonMarkedPath tail head ktail khead m π b H₀ Hₖ σ v v' u u')≤
  76       Fintype.card (MarkedReplacementPair tail head ktail khead m π b H₀ Hₖ (σ.symm 0) v v' u u') :=
  77   Fintype.card_le_of_injective _ (orderedCommonMarkedPath_to_pair hroot).injective
