  13 theorem finite_measurement_sparsification
  14     {E J : Type} [Fintype E] [Fintype J] (z : E → J → ℝ) (w : E → ℝ)
  15     (hw : ∀ e,0≤w e) (hW : 0<∑ e,w e) :
  16     ∃ w' : E → ℝ,(∀ e,0≤w' e) ∧ (∀ e,w e=0 → w' e=0) ∧
  17       (∑ e,w' e=∑ e,w e) ∧ (∀ j,∑ e,w' e*z e j=∑ e,w e*z e j) ∧
  18       (Finset.univ.filter (fun e=>0<w' e)).card≤Fintype.card J+1 := by
  19   classical
  20   let W := ∑ e,w e
  21   have hW' : 0<W := hW
  22   let T := Finset.univ.filter (fun e=>0<w e)
  23   let x := ∑ e,(w e/W) • z e
  24   have hfilter : (∑ e∈T,w e/W)=1 := by
  25     dsimp only [T]
  26     rw [Finset.sum_filter]
  27     calc
  28       _ = ∑ e,w e/W := by
  29         apply Finset.sum_congr rfl
  30         intro e _
  31         by_cases he : 0<w e
  32         · simp [he]
  33         · have hzero : w e=0 := le_antisymm (le_of_not_gt he) (hw e)
  34           simp [hzero]
  35       _ = 1 := by rw [← Finset.sum_div];exact div_self hW'.ne'
  36   have hxT : (∑ e∈T,(w e/W) • z e)=x := by
  37     dsimp only [T,x]
  38     rw [Finset.sum_filter]
  39     apply Finset.sum_congr rfl
  40     intro e _
  41     by_cases he : 0<w e
  42     · simp [he]
  43     · have hzero : w e=0 := le_antisymm (le_of_not_gt he) (hw e)
  44       simp [hzero]
  45   have hx : x∈convexHull ℝ (z '' {e | 0<w e}) := by
  46     rw [← hxT]
  47     apply (convex_convexHull ℝ (z '' {e | 0<w e})).sum_mem
  48       (fun e _=>div_nonneg (hw e) hW'.le) hfilter
  49     intro e he
  50     exact subset_convexHull ℝ _ ⟨e,(Finset.mem_filter.mp he).2,rfl⟩
  51   obtain ⟨I,hI,z',a,hrange,haff,ha,hatotal,hxsum⟩ := eq_pos_convex_span_of_mem_convexHull hx
  52   letI := hI
  53   have hex : ∀ i : I,∃ e : E,0<w e ∧ z e=z' i := fun i=>hrange ⟨i,rfl⟩
  54   choose e he hze using hex
  55   let w' := fun f : E=>W*(∑ i : I,if e i=f then a i else 0)
  56   have hw' : ∀ f,0≤w' f := by
  57     intro f
  58     apply mul_nonneg hW'.le
  59     exact Finset.sum_nonneg (fun i _=>by split_ifs;exact (ha i).le;exact le_refl _)
  60   have hmass : (∑ f,w' f)=W := by
  61     dsimp only [w']
  62     rw [← Finset.mul_sum,Finset.sum_comm]
  63     simp only [Finset.sum_ite_eq,Finset.mem_univ,ite_true,hatotal,mul_one]
  64   have hcoordinate (j : J) : ∑ f,w' f*z f j=∑ f,w f*z f j := by
  65     have hxj := congrFun hxsum j
  66     simp only [Finset.sum_apply,Pi.smul_apply,smul_eq_mul,x] at hxj
  67     have htarget : W*(∑ i,a i*z' i j)=∑ f,w f*z f j := by
  68       rw [hxj,Finset.mul_sum]
  69       apply Finset.sum_congr rfl
  70       intro f _
  71       field_simp
  72     calc
  73       _ = W*(∑ i : I,a i*z (e i) j) := by
  74         dsimp only [w']
  75         simp only [mul_assoc,Finset.sum_mul,← Finset.mul_sum]
  76         rw [Finset.sum_comm]
  77         congr 1
  78         apply Finset.sum_congr rfl
  79         intro i _
  80         simp only [ite_mul,zero_mul,Finset.sum_ite_eq,Finset.mem_univ,ite_true]
  81       _ = W*(∑ i : I,a i*z' i j) := by simp only [hze]
  82       _ = _ := htarget
  83   have hsupport : Finset.univ.filter (fun f=>0<w' f)⊆Finset.univ.image e := by
  84     intro f hf
  85     by_contra hno
  86     have hall : ∀ i,e i≠f := by simpa only [Finset.mem_image,Finset.mem_univ,true_and,not_exists] using hno
  87     have hz : w' f=0 := by simp only [w',hall,ite_false,Finset.sum_const_zero,mul_zero]
  88     have hh := (Finset.mem_filter.mp hf).2
  89     rw [hz] at hh
  90     exact (lt_irrefl 0) hh
  91   have hcard : Fintype.card I≤Fintype.card J+1 := by
  92     have hh := haff.card_le_finrank_succ
  93     have hd := (vectorSpan ℝ (Set.range z')).finrank_le
  94     rw [Module.finrank_fintype_fun_eq_card] at hd
  95     omega
  96   refine ⟨w',hw',?_,hmass,hcoordinate,?_⟩
  97   · intro f hf
  98     have hall : ∀ i,e i≠f := by
  99       intro i hi
 100       have hh := he i
 101       rw [hi,hf] at hh
 102       exact (lt_irrefl 0) hh
 103     simp only [w',hall,ite_false,Finset.sum_const_zero,mul_zero]
 104   · exact (Finset.card_le_card hsupport).trans ((Finset.card_image_le).trans (by simpa only [Finset.card_univ] using hcard))
