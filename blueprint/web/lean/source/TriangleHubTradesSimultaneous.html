  11 def TriangleHubTradeAt (G:SimpleGraph V) (H:V→U→Prop) (A Z:Finset V) (u u':U) (R:SimpleGraph V):Prop:=
  12     let Splus:V→U→Prop:=fun v z=>(v∈A∪Z∧z=u)∨(v∈A∧z=u')
  13     let Sminus:V→U→Prop:=fun v z=>(v∈A∪Z∧z=u')∨(v∈A∧z=u)
  14     HasOrdinaryWheelDecomposition R Splus 3 ∧ HasOrdinaryWheelDecomposition R Sminus 3 ∧
  15     HasResourceDecomposition (ordinaryWheelResources (P:=G) (F:=⊥) (Hbase:=H) (r:=3)) (ordinaryHostResources R Splus) ∧
  16     HasResourceDecomposition (ordinaryWheelResources (P:=G) (F:=⊥) (Hbase:=H) (r:=3)) (ordinaryHostResources R Sminus) ∧
  17     R≤G ∧ (∀ v w,R.Adj v w→v∈A∪Z ∧ w∈A∪Z) ∧ (∀ v w,v∈Z→w∈Z→¬R.Adj v w) ∧
  18     (∀ v z,Splus v z→H v z) ∧ (∀ v z,Sminus v z→H v z) ∧
  19     (∀ e:Sym2 V⊕(V×U),
  20       (if e∈ordinaryHostResources R Splus then (1:ℤ) else 0)-
  21       (if e∈ordinaryHostResources R Sminus then 1 else 0)=
