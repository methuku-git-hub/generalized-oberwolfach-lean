  12 theorem finiteLaw_exponential_interpolation
  13     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X Y : Ω → ℝ)
  14     {a b θ : ℝ} (hθ0 : 0≤θ) (hθ1 : θ≤1)
  15     (hX : μ.expectation (fun ω=>Real.exp (X ω))≤Real.exp a)
  16     (hY : μ.expectation (fun ω=>Real.exp (Y ω))≤Real.exp b) :
  17     μ.expectation (fun ω=>Real.exp (θ*X ω+(1-θ)*Y ω))≤
  18       Real.exp (θ*a+(1-θ)*b) := by
  19   have hpoint (ω : Ω) :
  20       Real.exp (θ*X ω+(1-θ)*Y ω)≤
  21         Real.exp (θ*a+(1-θ)*b)*
  22           (θ*Real.exp (-a)*Real.exp (X ω)+(1-θ)*Real.exp (-b)*Real.exp (Y ω)) := by
  23     have hc := convexOn_exp.2 (Set.mem_univ (X ω-a)) (Set.mem_univ (Y ω-b))
  24       hθ0 (sub_nonneg.mpr hθ1) (by ring : θ+(1-θ)=1)
  25     simp only [smul_eq_mul] at hc
  26     have hh := mul_le_mul_of_nonneg_left hc (Real.exp_pos (θ*a+(1-θ)*b)).le
  27     rw [← Real.exp_add] at hh
  28     have he : θ*a+(1-θ)*b+(θ*(X ω-a)+(1-θ)*(Y ω-b))=
  29         θ*X ω+(1-θ)*Y ω := by ring
  30     rw [he] at hh
  31     simpa only [sub_eq_add_neg,Real.exp_add,Real.exp_neg,mul_assoc,mul_comm,mul_left_comm] using hh
  32   have he : μ.expectation (fun ω=>Real.exp (θ*a+(1-θ)*b)*
  33       (θ*Real.exp (-a)*Real.exp (X ω)+(1-θ)*Real.exp (-b)*Real.exp (Y ω)))=
  34       Real.exp (θ*a+(1-θ)*b)*
  35         (θ*Real.exp (-a)*μ.expectation (fun ω=>Real.exp (X ω))+
  36           (1-θ)*Real.exp (-b)*μ.expectation (fun ω=>Real.exp (Y ω))) := by
  37     simp only [FiniteLaw.expectation,Finset.mul_sum,← Finset.sum_add_distrib]
  38     apply Finset.sum_congr rfl
  39     intro ω _
  40     ring
  41   have hx := mul_le_mul_of_nonneg_left hX (mul_nonneg hθ0 (Real.exp_pos (-a)).le)
  42   have hy := mul_le_mul_of_nonneg_left hY
  43     (mul_nonneg (sub_nonneg.mpr hθ1) (Real.exp_pos (-b)).le)
  44   have hxa : θ*Real.exp (-a)*Real.exp a=θ := by
  45     rw [mul_assoc,← Real.exp_add];simp
  46   have hyb : (1-θ)*Real.exp (-b)*Real.exp b=1-θ := by
  47     rw [mul_assoc,← Real.exp_add];simp
  48   rw [hxa] at hx
  49   rw [hyb] at hy
  50   calc
  51     _ ≤ _ := finiteLaw_expectation_mono μ _ _ hpoint
  52     _ = _ := he
  53     _ ≤ Real.exp (θ*a+(1-θ)*b)*1 :=
  54       mul_le_mul_of_nonneg_left (by linarith) (Real.exp_pos _).le
  55     _ = _ := mul_one _
