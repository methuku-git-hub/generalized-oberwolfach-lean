  10 theorem factor_pruned_family_removal {W:Type} [Fintype W] {n:ℕ} (hn:0<n)
  11     (K:ℕ) (hK:0<K) (classify:W→Option (Fin K))
  12     (D:ℕ) (δ α κL κS:ℝ) (hδ:0≤δ) (hα:0<α)
  13     (hL0:0≤κL) (hS0:0≤κS) (hL:κL≤α/1000) (hS:κS≤α/(1000*(K:ℝ)))
  14     (G:SimpleGraph (Fin n)) (hG:G.IsRegularOfDegree D) (hEven:Even D)
  15     (hdegree:(δ+α)*(n:ℝ)≤D) (F:W→SimpleGraph (Fin n))
  16     (hF:∀ w,(F w).IsRegularOfDegree 2):
  17     let fibres:=fun j:Option (Fin K)=>Finset.univ.filter (fun w=>classify w=j)
  18     let small:=fun j:Option (Fin K)=>
  19       ((fibres j).card:ℝ)<(if j=none then κL*n else κS*n)
  20     let removed:=Finset.univ.filter (fun w=>small (classify w))
  21     let list:=removed.toList.map F;
  22     (∀ P:SimpleGraph (Fin n),∀ d:ℕ,P.IsRegularOfDegree d→Even d→
  23       (δ+α/2)*(n:ℝ)≤d→∀ w∈removed,
  24       HasFactorDecomposition P (fun _:Fin (d/2)=>F w))→
  25     ∃ R:SimpleGraph (Fin n),∃ pieces:List (SimpleGraph (Fin n)),
  26       R.IsRegularOfDegree (D-2*removed.card) ∧ R≤G ∧
  27       (δ+α/2)*(n:ℝ)≤((D-2*removed.card:ℕ):ℝ) ∧
  28       List.Forall₂ (fun H A=>Nonempty (H ≃g A) ∧ H.IsRegularOfDegree 2) pieces list ∧
  29       (∀ H∈pieces,H≤G ∧ ∀ u v,H.Adj u v→¬R.Adj u v) ∧
  30       pieces.Pairwise (fun H J=>∀ u v,H.Adj u v→¬J.Adj u v) ∧
  31       (∀ u v,G.Adj u v ↔ R.Adj u v ∨ ∃ H∈pieces,H.Adj u v):=by
  32   intro fibres small removed list hdecomp
  33   have hb:=factor_small_class_budget K hK classify (n:ℝ) α κL κS (by positivity) hα hL0 hS0 hL hS
  34   have hb':(removed.card:ℝ)≤α*n/500:=by simpa [fibres,small,removed] using hb
  35   have hlen:list.length=removed.card:=by simp [list]
  36   have hsmall:(list.length:ℝ)≤α*n/500:=by rw [hlen];exact hb'
  37   have hlist:∀ A∈list,A.IsRegularOfDegree 2:=by
  38     intro A hA
  39     obtain ⟨w,_,rfl⟩:=List.mem_map.mp hA
  40     exact hF w
  41   have hdec:∀ P:SimpleGraph (Fin n),∀ d:ℕ,P.IsRegularOfDegree d→Even d→
  42       (δ+α/2)*(n:ℝ)≤d→∀ A∈list,HasFactorDecomposition P (fun _:Fin (d/2)=>A):=by
  43     intro P d hp he hd A hA
  44     obtain ⟨w,hw,rfl⟩:=List.mem_map.mp hA
  45     exact hdecomp P d hp he hd w (by simpa using hw)
  46   have hh:=factor_small_class_from_decompositions hn D δ α hδ hα G hG hEven hdegree list hlist hsmall hdec
  47   simpa [hlen] using hh
