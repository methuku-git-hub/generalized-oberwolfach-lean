  13     (F:∀i,SimpleGraph (X i)) (hF:∀i,(F i).IsRegularOfDegree 2):
  14     ∃ (D:∀i,X i→X i→Prop) (orig:∀i,List (PrescribedSourcePiece (X i))),
  15       (∀i,(orig i).Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset)) ∧
  16       (∀i p,p∈orig i→∃a xs,p=.cycle a xs ∧ p.OriginalValid (D i)) ∧
  17       (∀i,sourcePieceVerticesUnion (orig i)=Finset.univ ∧ sourcePieceEdgesUnion (orig i)=(F i).edgeFinset) ∧
  18       (∀i x y,D i x y ↔ (x,y)∈sourcePieceArcsUnion (orig i)) ∧
  19       ∀ (H:∀i,SimpleGraph (X i)) (R:∀i,Finset (X i)),
  20         (∀i,H i≤F i)→(∀i x,x∉R i→∀y,¬(H i).Adj x y)→
  21         ∃ pieces:∀i,List (PrescribedSourcePiece {x // x∈R i}),
  22           (∀i,(pieces i).Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset)) ∧
  23           (∀i p,p∈pieces i→p.Shape ∧ p.OriginalValid (fun x y=>D i x.1 y.1)) ∧
  24           (∀i,prescribedDependentVertices pieces i=Finset.univ) ∧
  25           (∀i,prescribedDependentGraph pieces i=(H i).induce (↑(R i):Set (X i))) ∧
  26           ∀i x y,prescribedDependentDigraph pieces i x y ↔ D i x.1 y.1 ∧ (H i).Adj x.1 y.1 := by
