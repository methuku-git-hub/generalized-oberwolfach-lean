  14 theorem resource_restriction_absorption_completion (R : C → Finset V) (e : W ↪ V)
  15     (Z : Finset V) (he : ∀ v,(∃ w,e w=v) ↔ v∉Z)
  16     (M : Finset (RestrictedResource R e))
  17     (hM : ∀ D∈M,∀ E∈M,D≠E → Disjoint (restrictedResourceSet R e D) (restrictedResourceSet R e E))
  18     (habs : HasResourceDecomposition R (Z∪(Finset.univ\M.biUnion (restrictedResourceSet R e)).image e)) :
  19     HasResourceDecomposition R Finset.univ := by
  20   let A := M.biUnion (restrictedResourceSet R e)
  21   have hA : HasResourceDecomposition R (A.image e) :=
  22     restricted_resource_decomposition_lift R e A ⟨M,hM,rfl⟩
  23   have hdisj : Disjoint (A.image e) (Z∪(Finset.univ\A).image e) := by
  24     apply Finset.disjoint_left.mpr
  25     intro v hv hw
  26     obtain ⟨x,hx,rfl⟩ := Finset.mem_image.mp hv
  27     rcases Finset.mem_union.mp hw with hz | hw
  28     · exact (he _).mp ⟨x,rfl⟩ hz
  29     · obtain ⟨y,hy,hxy⟩ := Finset.mem_image.mp hw
  30       have hxy' := e.injective hxy
  31       subst y
  32       exact (Finset.mem_sdiff.mp hy).2 hx
  33   have hU : A.image e∪(Z∪(Finset.univ\A).image e)=Finset.univ := by
  34     apply Finset.eq_univ_of_forall
  35     intro v
  36     by_cases hz : v∈Z
  37     · exact Finset.mem_union_right _ (Finset.mem_union_left _ hz)
  38     · obtain ⟨x,rfl⟩ := (he v).mpr hz
  39       by_cases hx : x∈A
  40       · exact Finset.mem_union_left _ (Finset.mem_image.mpr ⟨x,hx,rfl⟩)
  41       · exact Finset.mem_union_right _ (Finset.mem_union_right _
  42           (Finset.mem_image.mpr ⟨x,Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,hx⟩,rfl⟩))
  43   rw [← hU]
  44   exact resource_decomposition_union R _ _ hA habs hdisj
