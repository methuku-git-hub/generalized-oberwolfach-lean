  13 theorem signed_cycle_edge_isolator_on_subset (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  14     ∃ N : ℕ,∀ (V : Type) [Fintype V] [DecidableEq V],∀ (G : SimpleGraph V) (S : Finset V),
  15       N≤S.card →
  16       (∀ z∈S,(cycleThreshold r+α)*S.card≤((G.neighborFinset z∩S).card:ℝ)) →
  17       ∀ u v,u∈S → v∈S → G.Adj u v →
  18       ∃ I : SimpleGraph V → ℝ,
  19         (∀ C,I C≠0 → C≤spanningRestriction G S ∧ IsEmbeddedCycle C r) ∧
  20         (∀ C,|I C|≤1) ∧
  21         (∀ x y,(∑ C,if C.Adj x y then I C else 0)=
  22           if (SimpleGraph.edge u v).Adj x y then 1 else 0) ∧
  23         (∀ x,(∑ C,if x∈C.support then I C else 0)=
  24           (1/2:ℝ)*(if x=u ∨ x=v then 1 else 0)) := by
  25   obtain ⟨N,hN⟩ := fractional_cycle_edge_deletion r hr α hα
  26   refine ⟨N,?_⟩
  27   intro V _ _ G S hS hdeg u v hu hv huv
  28   let uS : (S:Set V) := ⟨u,hu⟩
  29   let vS : (S:Set V) := ⟨v,hv⟩
  30   obtain ⟨hB,hD⟩ := hN (S:Set V) (by simpa using hS) (G.induce (S:Set V)) (by
  31     intro z
  32     have he := induced_degree_intersection G S z
  33     simp only [← SimpleGraph.ncard_neighborSet] at he ⊢
  34     rw [he]
  35     simpa only [Fintype.card_coe,Finset.coe_sort_coe] using hdeg z.val z.property)
  36   have hB' := hB.map_embedded_cycles (Function.Embedding.subtype (S:Set V))
  37   have hD' := (hD uS vS huv).map_embedded_cycles (Function.Embedding.subtype (S:Set V))
  38   rw [graph_map_delete_edge] at hD'
  39   change HasFractionalDecomposition (spanningRestriction G S\SimpleGraph.edge u v)
  40     (fun C => IsEmbeddedCycle C r) at hD'
  41   have huvs : (spanningRestriction G S).Adj u v :=
  42     (spanningRestriction_adj G S u v).mpr ⟨huv,hu,hv⟩
  43   obtain ⟨I,hs,hb,he,hvtx⟩ := signed_cycle_edge_isolator_of_fractional
  44     (spanningRestriction G S) u v huvs hB' hD'
  45   refine ⟨I,hs,hb,he,?_⟩
  46   intro x
  47   calc
  48     _ = ∑ C : SimpleGraph V,if C.degree x≠0 then I C else 0 := by
  49       apply Finset.sum_congr rfl
  50       intro C _
  51       have hc : (x∈C.support) ↔ C.degree x≠0 :=
  52         (C.degree_pos_iff_mem_support x).symm.trans Nat.pos_iff_ne_zero
  53       simp only [hc]
  54     _ = _ := hvtx x
