  12 theorem prescribed_induced_physical_image (Q:SimpleGraph X) (D:X→X→Prop)
  13     (hrep:∀x y,Q.Adj x y→D x y ∨ D y x) (hDQ:∀x y,D x y→Q.Adj x y)
  14     (A:Finset X) (f:X→V):
  15     (prescribedInducedSourceEdges Q A).image (Sym2.map f)=
  16       ((prescribedInducedArcs D A).image (Prod.map f f)).image prescribedArcPhysical := by
  17   ext e
  18   constructor
  19   · intro he
  20     obtain ⟨a,ha,hae⟩:=Finset.mem_image.mp he
  21     induction a using Sym2.ind with
  22     | _ x y =>
  23       have hxy:Q.Adj x y ∧ x∈A ∧ y∈A:=(prescribed_induced_edge_mk Q A x y).mp ha
  24       rcases hrep x y hxy.1 with h|h
  25       · apply Finset.mem_image.mpr
  26         refine ⟨(f x,f y),?_,?_⟩
  27         · apply Finset.mem_image.mpr
  28           refine ⟨(x,y),?_,rfl⟩
  29           simp [prescribedInducedArcs,hxy.2.1,hxy.2.2,h]
  30         · simpa [prescribedArcPhysical] using hae
  31       · apply Finset.mem_image.mpr
  32         refine ⟨(f y,f x),?_,?_⟩
  33         · apply Finset.mem_image.mpr
  34           refine ⟨(y,x),?_,rfl⟩
  35           simp [prescribedInducedArcs,hxy.2.1,hxy.2.2,h]
  36         · simpa [prescribedArcPhysical,Sym2.eq_swap] using hae
  37   · intro he
  38     obtain ⟨e',he',heq⟩:=Finset.mem_image.mp he
  39     obtain ⟨a,ha,hae⟩:=Finset.mem_image.mp he'
  40     rcases a with ⟨x,y⟩
  41     have hh:x∈A ∧ y∈A ∧ D x y:=by simpa [prescribedInducedArcs] using ha
  42     apply Finset.mem_image.mpr
  43     refine ⟨s(x,y),(prescribed_induced_edge_mk Q A x y).mpr ⟨hDQ x y hh.2.2,hh.1,hh.2.1⟩,?_⟩
  44     subst e'
  45     simpa [prescribedArcPhysical] using heq
  46 end Oberwolfach
