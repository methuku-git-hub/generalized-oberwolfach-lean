  33 theorem normalized_signed_demand_bounds
  34     (degree Δ E n s B δ ε r : ℝ) (hE : 0≤E) (hn : 0<n) (hs : 0<s) (hB : 1≤B)
  35     (hδ : 0≤δ) (hδsmall : δ<1/4)
  36     (hdeg : |degree-E/n|≤ε) (hΔ : |Δ|≤r)
  37     (hsmall : s*ε+r ≤ δ*s*(E/n)) :
  38     0≤(s*degree+Δ)/(B*s) ∧
  39       (s*degree+Δ)/(B*s) ≤ (1+δ)*(E/B)/n := by
  40   have hBp : 0<B := lt_of_lt_of_le zero_lt_one hB
  41   have hcenter : 0≤s*(E/n) := mul_nonneg hs.le (div_nonneg hE hn.le)
  42   have herror : |s*degree+Δ-s*(E/n)| ≤ δ*s*(E/n) := by
  43     rw [show s*degree+Δ-s*(E/n)=s*(degree-E/n)+Δ by ring]
  44     calc
  45       _ ≤ |s*(degree-E/n)|+|Δ| := abs_add_le _ _
  46       _ = s*|degree-E/n|+|Δ| := by rw [abs_mul,abs_of_pos hs]
  47       _ ≤ s*ε+r := add_le_add (mul_le_mul_of_nonneg_left hdeg hs.le) hΔ
  48       _ ≤ _ := hsmall
  49   obtain ⟨hlo,hup⟩ := abs_le.mp herror
  50   have hmul : δ*(s*(E/n)) ≤ s*(E/n) := mul_le_of_le_one_left hcenter (by linarith)
  51   have hg : 0≤s*degree+Δ := by nlinarith
  52   refine ⟨div_nonneg hg (mul_pos hBp hs).le,?_⟩
  53   have hgu : s*degree+Δ ≤ (1+δ)*s*(E/n) := by nlinarith
  54   have hh := div_le_div_of_nonneg_right hgu (mul_pos hBp hs).le
  55   convert hh using 1 <;> field_simp [hs.ne',hBp.ne',hn.ne']
