  11 theorem finiteLaw_poisson_mgf_upper_tail
  12     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X : Ω → ℝ)
  13     {a : ℝ} (ha : 0 ≤ a)
  14     (hmgf : ∀ s, μ.expectation (fun ω => Real.exp (s * X ω)) ≤
  15       Real.exp (a * (Real.exp s - 1)))
  16     {ε : ℝ} (he : 0 ≤ ε) (he1 : ε ≤ 1) :
  17     μ.probability (fun ω => (1 + ε) * a ≤ X ω) ≤ Real.exp (-ε ^ 2 * a / 3) := by
  18   have hpos : 0 < 1 + ε := by linarith
  19   have hlog : 0 ≤ Real.log (1 + ε) := Real.log_nonneg (by linarith)
  20   calc
  21     _ ≤ Real.exp (-Real.log (1 + ε) * ((1 + ε) * a)) *
  22         μ.expectation (fun ω => Real.exp (Real.log (1 + ε) * X ω)) :=
  23       finiteLaw_exponential_markov μ X _ hlog
  24     _ ≤ Real.exp (-Real.log (1 + ε) * ((1 + ε) * a)) *
  25         Real.exp (a * (Real.exp (Real.log (1 + ε)) - 1)) :=
  26       mul_le_mul_of_nonneg_left (hmgf _) (Real.exp_pos _).le
  27     _ = Real.exp (-a * ((1 + ε) * Real.log (1 + ε) - ε)) := by
  28       rw [← Real.exp_add, Real.exp_log hpos]
  29       congr 1
  30       ring
  31     _ ≤ _ := by
  32       apply Real.exp_le_exp.mpr
  33       have h := mul_le_mul_of_nonneg_left (chernoff_upper_entropy he he1) ha
  34       nlinarith

  36 theorem finiteLaw_poisson_mgf_lower_tail
  37     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X : Ω → ℝ)
  38     {a : ℝ} (ha : 0 ≤ a)
  39     (hmgf : ∀ s, μ.expectation (fun ω => Real.exp (s * X ω)) ≤
  40       Real.exp (a * (Real.exp s - 1)))
  41     {ε : ℝ} (he : 0 ≤ ε) :
  42     μ.probability (fun ω => X ω ≤ (1 - ε) * a) ≤ Real.exp (-ε ^ 2 * a / 2) := by
  43   have hm : μ.expectation (fun ω => Real.exp (ε * (-X ω))) ≤
  44       Real.exp (a * (Real.exp (-ε) - 1)) := by
  45     simpa only [mul_neg, neg_mul] using hmgf (-ε)
  46   have ht := finiteLaw_exponential_markov μ (fun ω => -X ω) (-((1 - ε) * a)) he
  47   have hev : (fun ω => -((1 - ε) * a) ≤ -X ω) = (fun ω => X ω ≤ (1 - ε) * a) := by
  48     funext ω
  49     exact propext neg_le_neg_iff
  50   rw [hev] at ht
  51   calc
  52     _ ≤ Real.exp (-ε * (-((1 - ε) * a))) * μ.expectation (fun ω => Real.exp (ε * (-X ω))) := ht
  53     _ ≤ Real.exp (-ε * (-((1 - ε) * a))) * Real.exp (a * (Real.exp (-ε) - 1)) :=
  54       mul_le_mul_of_nonneg_left hm (Real.exp_pos _).le
  55     _ = Real.exp (a * (Real.exp (-ε) - 1 + ε - ε ^ 2)) := by
  56       rw [← Real.exp_add]
  57       congr 1
  58       ring
  59     _ ≤ _ := by
  60       apply Real.exp_le_exp.mpr
  61       have h := mul_le_mul_of_nonneg_left (exp_neg_upper_quadratic he) ha
  62       nlinarith

  64 theorem finiteLaw_poisson_mgf_absolute_tail
  65     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X : Ω → ℝ)
  66     {a : ℝ} (ha : 0 ≤ a)
  67     (hmgf : ∀ s, μ.expectation (fun ω => Real.exp (s * X ω)) ≤
  68       Real.exp (a * (Real.exp s - 1)))
  69     {ε : ℝ} (he : 0 ≤ ε) (he1 : ε ≤ 1) :
  70     μ.probability (fun ω => ε * a ≤ |X ω - a|) ≤ 2 * Real.exp (-ε ^ 2 * a / 3) := by
  71   classical
  72   let A (b : Bool) (ω : Ω) := if b then (1 + ε) * a ≤ X ω else X ω ≤ (1 - ε) * a
  73   have hev : (fun ω => ε * a ≤ |X ω - a|) =
  74       (fun ω => ∃ b ∈ (Finset.univ : Finset Bool), A b ω) := by
  75     funext ω
  76     apply propext
  77     simp only [Finset.mem_univ, true_and, Bool.exists_bool, A, Bool.false_eq_true, ite_false, ite_true, le_abs]
  78     constructor
  79     · rintro (h | h) <;> [right; left] <;> linarith
  80     · rintro (h | h) <;> [right; left] <;> linarith
  81   rw [hev]
  82   have hu := finiteLaw_poisson_mgf_upper_tail μ X ha hmgf he he1
  83   have hl := finiteLaw_poisson_mgf_lower_tail μ X ha hmgf he
  84   have hcomp : Real.exp (-ε ^ 2 * a / 2) ≤ Real.exp (-ε ^ 2 * a / 3) := by
  85     apply Real.exp_le_exp.mpr
  86     nlinarith [mul_nonneg (sq_nonneg ε) ha]
  87   have hsum := finiteLaw_union_bound μ Finset.univ A
  88   simp only [Fintype.sum_bool, A, Bool.false_eq_true, ite_false, ite_true] at hsum
  89   linarith

  91 theorem finiteLaw_poisson_mgf_large_tail
  92     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X : Ω → ℝ)
  93     {a : ℝ} (ha : 0 < a)
  94     (hmgf : ∀ s, μ.expectation (fun ω => Real.exp (s * X ω)) ≤
  95       Real.exp (a * (Real.exp s - 1)))
  96     {t : ℝ} (ht : a ≤ t) :
  97     μ.probability (fun ω => t ≤ X ω) ≤ (Real.exp 1 * a / t) ^ t := by
  98   have htpos : 0 < t := lt_of_lt_of_le ha ht
  99   have hquot : 0 < t / a := div_pos htpos ha
 100   have hlog : 0 ≤ Real.log (t / a) :=
 101     Real.log_nonneg ((one_le_div ha).mpr ht)
 102   calc
 103     _ ≤ Real.exp (-Real.log (t / a) * t) *
 104         μ.expectation (fun ω => Real.exp (Real.log (t / a) * X ω)) :=
 105       finiteLaw_exponential_markov μ X t hlog
 106     _ ≤ Real.exp (-Real.log (t / a) * t) *
 107         Real.exp (a * (Real.exp (Real.log (t / a)) - 1)) :=
 108       mul_le_mul_of_nonneg_left (hmgf _) (Real.exp_pos _).le
 109     _ = Real.exp (t * (1 + Real.log a - Real.log t) - a) := by
 110       rw [← Real.exp_add, Real.exp_log hquot, Real.log_div htpos.ne' ha.ne']
 111       congr 1
 112       field_simp
 113       ring
