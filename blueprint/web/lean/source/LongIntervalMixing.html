  11 theorem transitionWalk_mixing_from_subinterval {X : Type*} [Fintype X]
  12     (R : ℕ → X → X → ℝ) (π : ℕ → X → ℝ)
  13     (hR : ∀ i, IsTransition (R i))
  14     (hπ : ∀ i, transitionApply (π i) (R i) = π (i+1))
  15     (a L s t : ℕ) (has : a ≤ s) (hst : s+t ≤ a+L) (ε : ℝ)
  16     (hblock : ∀ x, RelativeError (transitionWalk R s t x) (π (s+t)) ε) :
  17     ∀ x, RelativeError (transitionWalk R a L x) (π (a+L)) ε := by
  18   have ha : a+(s-a)=s := Nat.add_sub_of_le has
  19   have hfirst : transitionCompose (transitionWalk R a (s-a)) (transitionWalk R s t) =
  20       transitionWalk R a ((s-a)+t) := by
  21     simpa only [ha] using (transitionWalk_add R a (s-a) t).symm
  22   have hstart : a+((s-a)+t)=s+t := by omega
  23   have hfull : transitionCompose
  24       (transitionCompose (transitionWalk R a (s-a)) (transitionWalk R s t))
  25       (transitionWalk R (s+t) (a+L-(s+t))) = transitionWalk R a L := by
  26     rw [hfirst]
  27     have hh := (transitionWalk_add R a ((s-a)+t) (a+L-(s+t))).symm
  28     rw [hstart] at hh
  29     convert hh using 1 <;> congr 1 <;> omega
  30   intro x
  31   have hh := mixed_block_with_prefix_suffix (transitionWalk R a (s-a)) (transitionWalk R s t)
  32     (transitionWalk R (s+t) (a+L-(s+t))) (π (s+t)) ε
  33     (transitionWalk_isTransition R hR _ _) (transitionWalk_isTransition R hR _ _).1 hblock x
  34   have hstat := transitionWalk_stationary R π hπ (s+t) (a+L-(s+t))
  35   have hend : s+t+(a+L-(s+t))=a+L := Nat.add_sub_of_le hst
  36   rw [hend] at hstat
  37   rw [hfull,hstat] at hh
  38   exact hh

  47     (hupper : ∀ i, a ≤ i → i < a+t → i ∉ J → ∀ x y, R i x y ≤ B*π (i+1) y)

  49       ∀ x y, κ*π (i+3) y ≤ transitionWalk R i 3 x y)
  50     (ht : 1+3*⌈Real.log (10*(B+1)) / (-Real.log (1-κ))⌉₊ ≤ t)
  51     (hclean : ∀ j, a ≤ j → j < a+t → j ∉ J) :
  52     ∀ x, RelativeError (transitionWalk R a t x) (π (a+t)) (1/10) := by
  53   let q := ⌈Real.log (10*(B+1)) / (-Real.log (1-κ))⌉₊

  55     (by dsimp [q]; omega) (hupper a le_rfl (by omega) (hclean a le_rfl (by omega)))
  56   · intro j hj x y
  57     have hq : 1+3*q ≤ t := ht
  58     have h0 := hclean (a+1+j*3) (by omega) (by omega)
  59     have h1 := hclean (a+1+j*3+1) (by omega) (by omega)
  60     have h2 := hclean (a+1+j*3+2) (by omega) (by omega)

  62       hminor (a+1+j*3) (by omega) (by omega) h0 h1 h2 x y
