  14 theorem common_neighbour_card_lower {V I : Type*} [DecidableEq V]
  15     (A : Finset V) (J : Finset I) (N : I → Finset V) (δ : ℝ)
  16     (hdeg : ∀ j∈J,δ*(A.card:ℝ)≤((A∩N j).card:ℝ)) :
  17     (1-(J.card:ℝ)*(1-δ))*(A.card:ℝ)≤((common A J N).card:ℝ) := by
  18   let D := J.biUnion (fun j=>A\N j)
  19   have he : common A J N=A\D := by
  20     ext x
  21     simp only [common,D,Finset.mem_filter,Finset.mem_sdiff,Finset.mem_biUnion]
  22     constructor
  23     · rintro ⟨ha,hall⟩
  24       exact ⟨ha,by rintro ⟨j,hj,_,hn⟩;exact hn (hall j hj)⟩
  25     · rintro ⟨ha,hnot⟩
  26       refine ⟨ha,?_⟩
  27       intro j hj
  28       by_contra hn
  29       exact hnot ⟨j,hj,ha,hn⟩
  30   have hdeficit (j : I) (hj : j∈J) : ((A\N j).card:ℝ)≤(1-δ)*(A.card:ℝ) := by
  31     have hh : ((A\N j).card:ℝ)+(A∩N j).card=A.card := by
  32       exact_mod_cast Finset.card_sdiff_add_card_inter A (N j)
  33     nlinarith only [hh,hdeg j hj]
  34   have hD : (D.card:ℝ)≤(J.card:ℝ)*((1-δ)*(A.card:ℝ)) := by
  35     calc
  36       _ ≤ ∑ j∈J,((A\N j).card:ℝ) := by exact_mod_cast (Finset.card_biUnion_le : D.card≤_)
  37       _ ≤ ∑ _j∈J,(1-δ)*(A.card:ℝ) := Finset.sum_le_sum hdeficit
  38       _ = _ := by simp only [Finset.sum_const,nsmul_eq_mul]
  39   have hc : (A.card:ℝ)≤(A\D).card+D.card := by exact_mod_cast (Finset.card_le_card_sdiff_add_card (s:=A) (t:=D))
  40   rw [he]
  41   nlinarith only [hc,hD]

  44 theorem forbidden_used_neighbours_card {V : Type*} [Fintype V] [DecidableEq V]
  45     (F : SimpleGraph V) (S : Finset V) (L : ℝ) (hF : ∀ v,(F.degree v:ℝ)≤L) :
  46     ((S∪S.biUnion (fun v=>F.neighborFinset v)).card:ℝ)≤(S.card:ℝ)*(L+1) := by
  47   have hu : ((S∪S.biUnion (fun v=>F.neighborFinset v)).card:ℝ)≤S.card+(S.biUnion (fun v=>F.neighborFinset v)).card := by
  48     exact_mod_cast Finset.card_union_le S (S.biUnion (fun v=>F.neighborFinset v))
  49   have hb : ((S.biUnion (fun v=>F.neighborFinset v)).card:ℝ)≤(S.card:ℝ)*L := by
  50     calc
  51       _ ≤ ∑ v∈S,((F.neighborFinset v).card:ℝ) := by exact_mod_cast (Finset.card_biUnion_le : (S.biUnion (fun v=>F.neighborFinset v)).card≤_)
  52       _ ≤ ∑ _v∈S,L := Finset.sum_le_sum (fun v _=>by simpa only [SimpleGraph.card_neighborFinset_eq_degree] using hF v)
  53       _ = _ := by simp only [Finset.sum_const,nsmul_eq_mul]
  54   nlinarith only [hu,hb]

  57 theorem dense_common_neighbour_avoiding_used {V : Type*} [Fintype V] [DecidableEq V]
  58     (P F : SimpleGraph V) (A R S : Finset V) (δ L : ℝ)
  59     (hdeg : ∀ v∈R,δ*(A.card:ℝ)≤((A∩P.neighborFinset v).card:ℝ))
  60     (hF : ∀ v,(F.degree v:ℝ)≤L) :
  61     (1-(R.card:ℝ)*(1-δ))*(A.card:ℝ)-(S.card:ℝ)*(L+1)≤
  62       ((common A R (fun v=>P.neighborFinset v)\(S∪S.biUnion (fun v=>F.neighborFinset v))).card:ℝ) := by
  63   have hh := common_neighbour_card_lower A R (fun v=>P.neighborFinset v) δ hdeg
  64   have hb := forbidden_used_neighbours_card F S L hF

  66       (common A R (fun v=>P.neighborFinset v)\(S∪S.biUnion (fun v=>F.neighborFinset v))).card+
  67       (S∪S.biUnion (fun v=>F.neighborFinset v)).card := by
