  11 def nestedLevel {V : Type*} (B : ℕ → Finset V) (t : ℕ) (x : V) : ℕ :=
  12   Nat.findGreatest (fun i => x ∈ B i) t

  18 theorem mem_nestedLevel {V : Type*} [Fintype V] (B : ℕ → Finset V) (t : ℕ)
  19     (hzero : B 0 = Finset.univ) (x : V) : x ∈ B (nestedLevel B t x) :=
  20   Nat.findGreatest_spec (P := fun i => x ∈ B i) (Nat.zero_le t)

  23 theorem mem_nested_iff_le_level {V : Type*} [Fintype V]
  24     (B : ℕ → Finset V) (t : ℕ) (hzero : B 0 = Finset.univ)
  25     (hsub : ∀ i < t, B (i + 1) ⊆ B i) (x : V) {i : ℕ} (hi : i ≤ t) :
  26     x ∈ B i ↔ i ≤ nestedLevel B t x := by
  27   constructor
  28   · exact fun hx => Nat.le_findGreatest hi hx
  29   · intro hix
  30     exact finset_chain_subset B t hsub hix (nestedLevel_le B t x) (mem_nestedLevel B t hzero x)
