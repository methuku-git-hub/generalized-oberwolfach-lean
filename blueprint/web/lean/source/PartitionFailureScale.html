  24 theorem partition_failure_eventually_small (c : ℕ) (hc : 0 < c)
  25     (η : ℝ) (hη : 0 < η) :
  26     ∃ N : ℕ, ∀ n : ℕ, N ≤ n →
  27       0 < n / c ∧ n ≤ 2 * c * (n / c) ∧ partitionFailure c η n ≤ 1 / (n : ℝ) := by
  28   have hc' : (0 : ℝ) < c := by exact_mod_cast hc
  29   have hκ : 0 < η ^ 2 / (4 * c) := by positivity
  30   obtain ⟨N,hN⟩ := eventually_stretchedExp_le_inv_pow (4 * c) 1
  31     (c := η ^ 2 / (4 * c)) (p := 1) hκ (by norm_num) 1
  32   refine ⟨max N c, ?_⟩
  33   intro n hn
  34   have hnc : c ≤ n := (le_max_right _ _).trans hn
  35   obtain ⟨hm, hscale, _⟩ := equal_cell_size_bounds n c hc hnc
  36   refine ⟨hm, hscale, ?_⟩
  37   have hscaler : (n : ℝ) ≤ 2 * c * (n / c : ℕ) := by exact_mod_cast hscale
  38   have hexp : -η ^ 2 * (n / c : ℕ) / 2 ≤ -(η ^ 2 / (4 * c)) * n := by
  39     apply (mul_le_mul_iff_right₀ (show (0 : ℝ) < 4 * c by positivity)).mp
  40     field_simp
  41     nlinarith [mul_le_mul_of_nonneg_left hscaler (sq_nonneg η)]
  42   calc
  43     partitionFailure c η n ≤ 4 * c * n * Real.exp (-(η ^ 2 / (4 * c)) * n) :=
  44       mul_le_mul_of_nonneg_left (Real.exp_le_exp.mpr hexp) (by positivity)
  45     _ ≤ _ := by
  46       simpa only [Real.rpow_one, pow_one] using hN n ((le_max_left _ _).trans hn)
