  11 theorem factor_pruned_class_indices {W:Type} [Fintype W] {n:ℕ} (hn:0<n)
  12     (K:ℕ) (hK:0<K) (classify:W→Option (Fin K))
  13     (D:ℕ) (hEven:Even D) (hsize:Fintype.card W=D/2) (hDn:D≤n)
  14     (δ α κL κS:ℝ) (hδ:0≤δ) (hα:0<α)
  15     (hL0:0<κL) (hS0:0<κS) (hL:κL≤α/1000) (hS:κS≤α/(1000*(K:ℝ)))
  16     (hdegree:(δ+α)*(n:ℝ)≤D):
  17     let fibres:=fun j:Option (Fin K)=>Finset.univ.filter (fun w=>classify w=j)
  18     let cutoff:=fun j:Option (Fin K)=>if j=none then κL else κS
  19     let small:=fun j:Option (Fin K)=>((fibres j).card:ℝ)<cutoff j*n
  20     let removed:=Finset.univ.filter (fun w=>small (classify w))
  21     let remaining:={w:W // w∉removed}
  22     let cf:=fun w:remaining=>classify w.val
  23     let D0:=D-2*removed.card
  24     let fibres0:=fun j:Option (Fin K)=>Finset.univ.filter (fun w:remaining=>cf w=j)
  25     let C:=Finset.univ.filter (fun j=>(fibres0 j).Nonempty)
  26     let k:=C.card
  27     let e:Fin k≃↥C:=(Fintype.equivFinOfCardEq (Fintype.card_coe C)).symm
  28     let m:=fun a:Fin k=>(fibres0 (e a).val).card
  29     let ρ:=fun a:Fin k=>2*(m a:ℝ)/(D0:ℝ);
  30     0<D0 ∧ Even D0 ∧ 0<k ∧ k≤K+1 ∧ (∀ a,0<m a) ∧
  31     (∑ a,m a)=D0/2 ∧ (∀ a,0<ρ a) ∧ (∑ a,ρ a)=1 ∧
  32     (∀ a,ρ a/2*D0=m a) ∧ (∀ a,2*min κL κS≤ρ a):=by
  33   intro fibres cutoff small removed remaining cf D0 fibres0 C k e m ρ
  34   have hp:=factor_pruned_family_parameters hn K hK classify D hEven hsize hDn δ α κL κS
  35     hδ hα hL0.le hS0.le hL hS hdegree
  36   have hp':0<D0 ∧ Fintype.card remaining=D0/2 ∧ Even D0 ∧
  37       (∀ j,(∃ w:remaining,cf w=j)→2*cutoff j≤
  38         2*((Finset.univ.filter (fun w:remaining=>cf w=j)).card:ℝ)/D0):=by
  39     exact ⟨hp.1,hp.2.1,hp.2.2.1,hp.2.2.2.2⟩
  40   have hsz:2*Fintype.card remaining=D0:=by
  41     rw [hp'.2.1]
  42     obtain ⟨t,ht⟩:=hp'.2.2.1
  43     omega
  44   have hc:=factor_nonempty_class_indices cf D0 hp'.1 hsz
  45   have hbound:k≤K+1:=factor_nonempty_class_count_bound K cf
  46   have hlow:∀ a,2*min κL κS≤ρ a:=by
  47     intro a
  48     have ha:(fibres0 (e a).val).Nonempty:=(Finset.mem_filter.mp (e a).property).2
  49     obtain ⟨w,hw⟩:=ha
  50     have hw':cf w=(e a).val:=(Finset.mem_filter.mp hw).2
  51     have hρ:=hp'.2.2.2 (e a).val ⟨w,hw'⟩
  52     have hmin:min κL κS≤cutoff (e a).val:=by
  53       dsimp [cutoff]
  54       split
  55       · exact min_le_left _ _
  56       · exact min_le_right _ _
  57     have hh:=mul_le_mul_of_nonneg_left hmin (by norm_num: (0:ℝ)≤2)
  58     exact hh.trans hρ
  59   exact ⟨hp'.1,hp'.2.2.1,hc.1,hbound,hc.2.1,hc.2.2.1,hc.2.2.2.1,
  60     hc.2.2.2.2.1,hc.2.2.2.2.2,hlow⟩
