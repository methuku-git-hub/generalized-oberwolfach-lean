  28     (hopen : markedArcPresent tail head ktail khead jout v a)
  29     (hclose : markedArcPresent tail head ktail khead jin z v)
  30     (hopenD : ¬D.Adj v (if jout then π a else a))
  31     (hcloseD : ¬D.Adj z (if jin then π v else v))
  32     (hbudget : W.card+((Z.card:ℝ)+l+4)*(2*b+1)+2*δ<c) :
  33     ∃ p : Fin (l+3) ↪ V, ∃ C : MarkedWheel tail head ktail khead (l+3) π b H₀ HK,

  37       PermutationSeparated π b p ∧ Disjoint (markedWheelRimGraph C) D := by
