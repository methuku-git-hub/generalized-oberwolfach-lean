  20     (s:Finset I) (u v:I→V) (P:∀ i,F.Walk (u i) (v i))
  21     (hp:∀i∈s,(P i).IsPath)
  22     (t:Finset J) (root:J→V) (C:∀ j,F.Walk (root j) (root j))
  23     (hc:∀j∈t,(C j).IsCycle)
  24     (x y:V) (hxy:F.Adj x y)
  25     (hx:x∈(t.biUnion (fun j=>(C j).support.toFinset)) ∪
  26        (s.biUnion (fun i=>(P i).support.toFinset \ {u i,v i}))):
  27     s(x,y)∈(t.biUnion (fun j=>(C j).edges.toFinset)) ∪
  28        (s.biUnion (fun i=>(P i).edges.toFinset)) := by
