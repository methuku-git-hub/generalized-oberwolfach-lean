  11 theorem prescribed_position_pair_vertex_zip:
  12     ∀ (es:List (X×X)) (ms:List (V×V)),es.length=ms.length→∀ (vs:List X) (zs:List V),
  13       (es.flatMap prescribedPairVertices++vs).zip (ms.flatMap prescribedPairVertices++zs)=
  14         (es.zip ms).flatMap (fun p=>[(p.1.1,p.2.1),(p.1.2,p.2.2)])++vs.zip zs := by
  15   intro es
  16   induction es with
  17   | nil =>
  18     intro ms h vs zs
  19     cases ms with
  20     | nil => simp [prescribedPairVertices]
  21     | cons m ms => simp at h
  22   | cons e es ih =>
  23     intro ms h vs zs
  24     cases ms with
  25     | nil => simp at h
  26     | cons m ms =>
  27       have hl:es.length=ms.length:=by simpa using h
  28       simp only [List.flatMap_cons,prescribedPairVertices,List.cons_append,List.nil_append,
  29         List.zip_cons_cons]
  30       rw [ih ms hl vs zs]
