  11 theorem codegree_cut_numeric (p ε D b s t v η:ℝ)
  12     (hp:0<p) (hb:0<b) (hη:0<η) (hs:η*b≤s) (ht:η*b≤t)
  13     (hε:ε≤p^2*η/24) (hD:D≤p^2*η^2*b/8)
  14     (hq:(v-p*s*t)^2≤t*(3*ε*s^2*b+D*s*b)) : p/2*s*t≤v := by
  15   have hspos:0<s:=(mul_pos hη hb).trans_le hs
  16   have htpos:0<t:=(mul_pos hη hb).trans_le ht
  17   have hεb:3*ε*b≤p^2*t/8:=by
  18     have hh:=mul_le_mul_of_nonneg_right hε hb.le
  19     have hh':=mul_le_mul_of_nonneg_left ht (sq_nonneg p)
  20     nlinarith only [hh,hh']
  21   have hproduct:η^2*b^2≤s*t:=by
  22     have hh:=mul_le_mul hs ht (mul_pos hη hb).le hspos.le
  23     nlinarith only [hh]
  24   have hDb:D*b≤p^2*s*t/8:=by
  25     have hh:=mul_le_mul_of_nonneg_right hD hb.le
  26     have hh':=mul_le_mul_of_nonneg_left hproduct (sq_nonneg p)
  27     nlinarith only [hh,hh']
  28   have hterms:3*ε*s^2*b+D*s*b≤p^2*s^2*t/4:=by
  29     have hh:=mul_le_mul_of_nonneg_right hεb (sq_nonneg s)
  30     have hh':=mul_le_mul_of_nonneg_right hDb hspos.le
  31     nlinarith only [hh,hh']
  32   have hu:=mul_le_mul_of_nonneg_left hterms htpos.le
  33   have hu':(v-p*s*t)^2≤(p*s*t/2)^2:=by nlinarith only [hq,hu]
  34   by_contra hv
  35   have hv':v<p*s*t/2:=by linarith only [lt_of_not_ge hv]
  36   have hpos:0<p*s*t/2:=by positivity
  37   have hsq:(p*s*t/2)^2<(p*s*t-v)^2:=
  38     (sq_lt_sq₀ hpos.le (by linarith only [hv',hpos])).mpr (by linarith only [hv'])
  39   nlinarith only [hsq,hu']

  45     (hrow:∀ x,(p-ε)*Fintype.card Y≤∑ u,bipAdj H x u)
  46     (hcodeg:∀ x y,y∉E x → ∑ u,bipAdj H x u*bipAdj H y u≤(p^2+ε)*Fintype.card Y)
  47     (hsmall:ε≤p^2*η/24) (hlarge:D≤p^2*η^2*Fintype.card Y/8)
  48     (S:Finset X) (T:Finset Y)
  49     (hS:η*Fintype.card Y≤(S.card:ℝ)) (hT:η*Fintype.card Y≤(T.card:ℝ)) :
