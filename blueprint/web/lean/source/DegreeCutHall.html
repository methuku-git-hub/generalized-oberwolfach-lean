  13 theorem degree_cut_perfect_matching {X Y : Type*} [Fintype X] [Fintype Y]
  14     (E : X→Y→Prop) (hcard : Fintype.card X=Fintype.card Y)
  15     (a : ℝ) (ha : 0≤a)
  16     (hrow : ∀ x,a≤((Finset.univ.filter (E x)).card:ℝ))
  17     (hcol : ∀ y,a≤((Finset.univ.filter (fun x=>E x y)).card:ℝ))
  18     (hcut : ∀ S : Finset X,∀ T : Finset Y,a≤(S.card:ℝ) → a≤(T.card:ℝ) →
  19       S.Nonempty → T.Nonempty → ∃ x∈S,∃ y∈T,E x y) :
  20     ∃ σ : X≃Y,∀ x,E x (σ x) := by
  21   classical
  22   have hhall (S : Finset X) : S.card≤(S.biUnion (fun x=>Finset.univ.filter (E x))).card := by
  23     let N:=S.biUnion (fun x=>Finset.univ.filter (E x))
  24     by_contra hfail
  25     have hlt : N.card<S.card:=lt_of_not_ge hfail
  26     have hS : S.Nonempty:=Finset.card_pos.mp (by omega)
  27     obtain ⟨x,hx⟩:=hS
  28     have hsub : Finset.univ.filter (E x)⊆N:=fun y hy=>Finset.mem_biUnion.mpr ⟨x,hx,hy⟩
  29     have hSlarge : a≤(S.card:ℝ):=(hrow x).trans ((Nat.cast_le.mpr (Finset.card_le_card hsub)).trans (Nat.cast_le.mpr hlt.le))
  30     have hNlt : N.card<Fintype.card Y:=hlt.trans_le (by simpa [hcard] using Finset.card_le_univ S)
  31     have hT : (Finset.univ\N).Nonempty:=by
  32       apply Finset.card_pos.mp
  33       rw [Finset.card_sdiff_of_subset (Finset.subset_univ _),Finset.card_univ]
  34       omega
  35     obtain ⟨y,hy⟩:=hT
  36     have hnot : y∉N:=(Finset.mem_sdiff.mp hy).2
  37     have hsub' : Finset.univ.filter (fun x=>E x y)⊆Finset.univ\S:=by
  38       intro x hx'
  39       refine Finset.mem_sdiff.mpr ⟨Finset.mem_univ _,?_⟩
  40       intro hxS
  41       exact hnot (Finset.mem_biUnion.mpr ⟨x,hxS,Finset.mem_filter.mpr ⟨Finset.mem_univ _,(Finset.mem_filter.mp hx').2⟩⟩)
  42     have hcomp : (Finset.univ\S).card≤(Finset.univ\N).card:=by
  43       rw [Finset.card_sdiff_of_subset (Finset.subset_univ _),Finset.card_sdiff_of_subset (Finset.subset_univ _),Finset.card_univ,Finset.card_univ,hcard]
  44       omega
  45     have hTlarge : a≤((Finset.univ\N).card:ℝ):=(hcol y).trans ((Nat.cast_le.mpr (Finset.card_le_card hsub')).trans (Nat.cast_le.mpr hcomp))
  46     obtain ⟨x',hx',y',hy',he⟩:=hcut S (Finset.univ\N) hSlarge hTlarge ⟨x,hx⟩ ⟨y,hy⟩
  47     exact (Finset.mem_sdiff.mp hy').2 (Finset.mem_biUnion.mpr ⟨x',hx',Finset.mem_filter.mpr ⟨Finset.mem_univ _,he⟩⟩)
  48   obtain ⟨f,hf,hE⟩:=(Finset.all_card_le_biUnion_card_iff_existsInjective' (fun x=>Finset.univ.filter (E x))).mp hhall
  49   exact ⟨Equiv.ofBijective f ((Fintype.bijective_iff_injective_and_card f).mpr ⟨hf,hcard⟩),fun x=>(Finset.mem_filter.mp (hE x)).2⟩
