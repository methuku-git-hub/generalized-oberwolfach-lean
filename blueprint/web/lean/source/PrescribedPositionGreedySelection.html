  11 theorem prescribed_position_greedy_selection (Q:SimpleGraph X) (D:X→X→Prop)
  12     (hdeg:∀ x,Q.degree x≤2) (hrep:∀x y,Q.Adj x y→D x y ∨ D y x)
  13     (S:Finset X) (m z:ℕ) (hE:2*S.card+8*m≤Q.edgeFinset.card)
  14     (hV:S.card+4*m+3*z≤Fintype.card X):
  15     ∃ (es:List (X×X)) (vs:List X),es.length=m ∧ vs.length=z ∧
  16       (∀ e∈es,D e.1 e.2 ∧ Q.Adj e.1 e.2 ∧ e.1∉S ∧ e.2∉S) ∧
  17       es.Pairwise (prescribedPairsApart Q) ∧ (∀ v∈vs,v∉S) ∧
  18       (∀ e∈es,∀ v∈vs,v∉prescribedPairClosed Q e) ∧
  19       vs.Pairwise (prescribedSinglesApart Q) := by
  20   obtain ⟨es,heslen,hes,hpair⟩:=prescribed_position_greedy_pairs Q D hdeg hrep m S hE
  21   let C:Finset X:=es.toFinset.biUnion (prescribedPairClosed Q)
  22   have hC:C.card≤4*m:=by
  23     calc
  24       C.card≤∑ e∈es.toFinset,(prescribedPairClosed Q e).card:=Finset.card_biUnion_le
  25       _≤∑ _e∈es.toFinset,4:=by
  26         apply Finset.sum_le_sum
  27         intro e he
  28         exact prescribed_edge_closed_neighbourhood_card Q hdeg e.1 e.2 (hes e (List.mem_toFinset.mp he)).2.1
  29       _=4*es.toFinset.card:=by simp [Nat.mul_comm]
  30       _≤4*es.length:=Nat.mul_le_mul_left 4 (List.toFinset_card_le es)
  31       _=4*m:=by rw [heslen]
  32   let T:=S∪C
  33   have hT:T.card≤S.card+4*m:=(Finset.card_union_le _ _).trans (Nat.add_le_add_left hC _)
  34   have hbudget:T.card+3*z≤Fintype.card X:=by omega
  35   obtain ⟨vs,hvslen,hvs,hvsPair⟩:=prescribed_position_greedy_singles Q hdeg z T hbudget
  36   refine ⟨es,vs,heslen,hvslen,hes,hpair,?_,?_,hvsPair⟩
  37   · intro v hv hs;exact hvs v hv (Finset.mem_union_left _ hs)
  38   · intro e he v hv hclosed
  39     have hvC:v∈C:=Finset.mem_biUnion.mpr ⟨e,List.mem_toFinset.mpr he,hclosed⟩
  40     exact hvs v hv (Finset.mem_union_right _ hvC)
