  14 def directedBoundaryNeighbours (tail head:A→W) (coin:A→E) (X:Finset W)
  15     (σ:E→Bool) (x:W):Finset W :=
  16   X.filter (fun y=>∃ a,((tail a=x∧head a=y)∨(head a=x∧tail a=y))∧σ (coin a)=true)

  18 theorem boundary_inside_image_eq_neighbours (tail head:A→W) (coin:A→E) (X:Finset W)
  19     (σ:E→Bool) (x:W) (hx:x∉X):
  20     (Finset.univ.filter (fun a:DirectedBoundaryEdge tail head X=>
  21       boundaryOutside tail head X a=x∧σ (coin a.val)=true)).image (boundaryInside tail head X)=
  22       directedBoundaryNeighbours tail head coin X σ x := by
  23   ext y
  24   constructor
  25   · intro hy
  26     obtain ⟨a,ha,hay⟩:=Finset.mem_image.mp hy
  27     have hao:boundaryOutside tail head X a=x:=(Finset.mem_filter.mp ha).2.1
  28     have hs:σ (coin a.val)=true:=(Finset.mem_filter.mp ha).2.2
  29     apply Finset.mem_filter.mpr
  30     refine ⟨hay ▸ boundaryInside_mem tail head X a,⟨a.val,?_,hs⟩⟩
  31     have he:=boundary_ordered_endpoints tail head X a
  32     rw [hao,hay] at he
  33     by_cases ht:tail a.val∈X
  34     · simp only [ht,ite_true] at he
  35       exact Or.inr ⟨he.2,he.1⟩
  36     · simp only [ht,ite_false] at he
  37       exact Or.inl he
  38   · intro hy
  39     obtain ⟨hy,⟨a,ha,hs⟩⟩:=Finset.mem_filter.mp hy
  40     rcases ha with ⟨ht,hh⟩|⟨hh,ht⟩
  41     · have hp:(tail a∉X∧head a∈X)∨(tail a∈X∧head a∉X):=Or.inl ⟨ht.symm ▸ hx,hh.symm ▸ hy⟩
  42       refine Finset.mem_image.mpr ⟨⟨a,hp⟩,?_,?_⟩
  43       · apply Finset.mem_filter.mpr
  44         exact ⟨Finset.mem_univ _,by simp [boundaryOutside,ht,hx,hs]⟩
  45       · simp only [boundaryInside,ht,hx,ite_false,hh]
  46     · have hp:(tail a∉X∧head a∈X)∨(tail a∈X∧head a∉X):=Or.inr ⟨ht.symm ▸ hy,hh.symm ▸ hx⟩
  47       refine Finset.mem_image.mpr ⟨⟨a,hp⟩,?_,?_⟩
  48       · apply Finset.mem_filter.mpr
  49         exact ⟨Finset.mem_univ _,by simp [boundaryOutside,ht,hy,hh,hs]⟩
  50       · simp only [boundaryInside,ht,hy,ite_true]

  53     (π:Equiv.Perm W) (σ:E→Bool) (x:W) (hx:x∉X):
  54     directedBoundarySupport tail head coin X π σ x=
