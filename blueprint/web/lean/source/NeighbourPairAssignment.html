  16 theorem neighbour_pairs_from_balanced_split {X Y : Type} [Fintype X] [Fintype Y]
  17     (m : ℕ) (hm : 0<m) (hY : Fintype.card Y=m)
  18     (E : X→Y→Prop) (F : SimpleGraph X) (lam b : ℝ)
  19     (hlam : 0<lam) (hb : 0≤b) (hsmall : b≤lam*m/4)
  20     (hmargin : b<lam^2*m/16)
  21     (P : FixedPartition X 2 (fun _=>m))
  22     (hrow : ∀ x,lam*m≤((Finset.univ.filter (E x)).card:ℝ))
  23     (hcol : ∀ y i,lam*m/2≤(((Finset.univ.filter (fun x=>E x y))∩(fixedPartitionCell P i).val).card:ℝ))
  24     (hF : ∀ x,(F.degree x:ℝ)≤b)
  25     (hcut : ∀ S : Finset X,∀ T : Finset Y,lam/16*m≤(S.card:ℝ) → lam/16*m≤(T.card:ℝ) →
  26       lam*S.card*T.card≤(bipartiteMatrixCut (fun x y=>if E x y then 1 else 0) S T:ℝ)) :
  27     ∃ v w : Y↪X,(∀ y,E (v y) y∧E (w y) y∧¬F.Adj (v y) (w y)) ∧
  28       Disjoint (Set.range v) (Set.range w) ∧ (∀ x,(∃ y,v y=x)∨∃ y,w y=x) := by
  29   classical
  30   have hmR : (0:ℝ)<m:=Nat.cast_pos.mpr hm
  31   let f (i:Fin 2):Fin m↪X:=partitionCellEmbedding P i m rfl
  32   have hcell (i:Fin 2) (y:Y) : lam*m/2≤((Finset.univ.filter (fun j=>E (f i j) y)).card:ℝ):=by
  33     rw [show (Finset.univ.filter (fun j=>E (f i j) y)).card=
  34       ((Finset.univ.filter (fun x=>E x y))∩(fixedPartitionCell P i).val).card from partition_cell_predicate_card P i (fun x=>E x y)]
  35     exact hcol y i
  36   have hcut' (i:Fin 2) (S:Finset (Fin m)) (T:Finset Y)
  37       (hS : lam*m/4≤(S.card:ℝ)) (hT : lam*m/4≤(T.card:ℝ)) :
  38       lam*S.card*T.card≤(bipartiteMatrixCut (fun j y=>if E (f i j) y then 1 else 0) S T:ℝ):=by
  39     rw [bipartite_cut_embedding (f i) (fun x y=>if E x y then 1 else 0) S T]
  40     have hh:=hcut (S.map (f i)) T (by simp only [Finset.card_map];nlinarith only [hS,mul_pos hlam hmR]) (by nlinarith only [hT,mul_pos hlam hmR])
  41     simpa only [Finset.card_map] using hh
  42   have hcard : Fintype.card (Fin m)=Fintype.card Y:=by simp [hY]
  43   obtain ⟨σ,hσ⟩:=degree_cut_matching_after_deletion (fun j y=>E (f 0 j) y) (fun _ _=>False)
  44     hcard (lam*m/4) 0 lam (by positivity) (by norm_num) hlam (by positivity)
  45     (fun j=>by have hh:=hrow (f 0 j);linarith only [hh,mul_pos hlam hmR])
  46     (fun y=>by have hh:=hcell 0 y;linarith only [hh,mul_pos hlam hmR])
  47     (by intro j;simp) (by intro y;simp) (hcut' 0)
  48   let v : Y↪X:=σ.symm.toEmbedding.trans (f 0)
  49   have hv (y:Y):E (v y) y:=by simpa [v] using (hσ (σ.symm y)).1
  50   let J (j:Fin m) (y:Y):=F.Adj (v y) (f 1 j)
  51   have hjrow (j:Fin m) : ((Finset.univ.filter (J j)).card:ℝ)≤b:=by
  52     have hh:=embedding_predicate_card v (fun x=>F.Adj (f 1 j) x)
  53     have he : (Finset.univ.filter (J j)).card=(Finset.univ.filter (fun y=>F.Adj (f 1 j) (v y))).card:=by
  54       congr 1;ext y;simp only [Finset.mem_filter,Finset.mem_univ,true_and,J];exact F.adj_comm _ _
  55     rw [he]
  56     have hn : (Finset.univ.filter (fun x=>F.Adj (f 1 j) x)).card=F.degree (f 1 j):=by
  57       rw [SimpleGraph.degree];congr 1;ext x;simp
  58     rw [hn] at hh
  59     exact (Nat.cast_le.mpr hh).trans (hF _)
  60   have hjcol (y:Y) : ((Finset.univ.filter (fun j=>J j y)).card:ℝ)≤b:=by
  61     have hh:=embedding_predicate_card (f 1) (fun x=>F.Adj (v y) x)
  62     have hn : (Finset.univ.filter (fun x=>F.Adj (v y) x)).card=F.degree (v y):=by
  63       rw [SimpleGraph.degree];congr 1;ext x;simp
  64     rw [hn] at hh
  65     exact (Nat.cast_le.mpr hh).trans (hF _)
  66   have hmarg : b<lam*(lam*m/4):=by nlinarith only [hmargin,mul_pos (sq_pos_of_pos hlam) hmR]
  67   obtain ⟨τ,hτ⟩:=degree_cut_matching_after_deletion (fun j y=>E (f 1 j) y) J
  68     hcard (lam*m/4) b lam (by positivity) hb hlam hmarg
  69     (fun j=>by have hh:=hrow (f 1 j);linarith only [hh,hsmall,mul_pos hlam hmR])
  70     (fun y=>by have hh:=hcell 1 y;linarith only [hh,hsmall]) hjrow hjcol (hcut' 1)
  71   let w : Y↪X:=τ.symm.toEmbedding.trans (f 1)
  72   have hw (y:Y):E (w y) y∧¬F.Adj (v y) (w y):=by simpa [w,J] using hτ (τ.symm y)
  73   refine ⟨v,w,fun y=>⟨hv y,(hw y).1,(hw y).2⟩,?_,?_⟩
  74   · apply Set.disjoint_left.mpr
  75     rintro x ⟨y,rfl⟩ ⟨z,hz⟩
  76     have h0 : P.val (v y)=0:=partitionCellEmbedding_colour P 0 m rfl (σ.symm y)
  77     have h1 : P.val (w z)=1:=partitionCellEmbedding_colour P 1 m rfl (τ.symm z)
  78     rw [hz,h0] at h1
  79     exact (by decide : (0:Fin 2)≠1) h1
  80   · intro x
  81     have hcases : P.val x=0∨P.val x=1:=by omega
  82     rcases hcases with h0|h1
  83     · obtain ⟨j,hj⟩:=(partitionCellEmbedding_range P 0 m rfl x).mpr h0
  84       exact Or.inl ⟨σ j,by simpa [v,f] using hj⟩
  85     · obtain ⟨j,hj⟩:=(partitionCellEmbedding_range P 1 m rfl x).mpr h1
  86       exact Or.inr ⟨τ j,by simpa [w,f] using hj⟩

  88 theorem ordinary_neighbour_pair_assignment (lam : ℝ) (hlam : 0<lam) :
  89     ∀ b : ℕ,∃ N : ℕ,∀ (X Y : Type) [Fintype X] [Fintype Y],
  90       ∀ m : ℕ,N≤m → Fintype.card X=2*m → Fintype.card Y=m →
  91       ∀ (E : X→Y→Prop) (F : SimpleGraph X),
  92       (∀ x,lam*m≤((Finset.univ.filter (E x)).card:ℝ)) →
  93       (∀ y,2*lam*m≤((Finset.univ.filter (fun x=>E x y)).card:ℝ)) →
  94       (∀ x,F.degree x≤b) →
  95       (∀ S : Finset X,∀ T : Finset Y,lam/16*m≤(S.card:ℝ) → lam/16*m≤(T.card:ℝ) →
  96         lam*S.card*T.card≤(bipartiteMatrixCut (fun x y=>if E x y then 1 else 0) S T:ℝ)) →
  97       ∃ v w : Y↪X,(∀ y,E (v y) y∧E (w y) y∧¬F.Adj (v y) (w y)) ∧
  98         Disjoint (Set.range v) (Set.range w) ∧ (∀ x,(∃ y,v y=x)∨∃ y,w y=x) := by
  99   intro b
 100   obtain ⟨N,hN⟩:=bipartite_balanced_split lam hlam
 101   obtain ⟨L,hL⟩:=exists_nat_gt (max (4*b/lam) (16*b/lam^2):ℝ)
 102   refine ⟨max N (max L 1),?_⟩
 103   intro X Y _ _ m hm hX hY E F hrow hcol hF hcut
 104   have hmpos : 0<m:=by omega
 105   have hLm : (L:ℝ)≤m:=Nat.cast_le.mpr (by omega)
 106   have hsmall : (b:ℝ)≤lam*m/4:=by
 107     have hh:4*b/lam<(m:ℝ):=(le_max_left _ _).trans_lt (hL.trans_le hLm)
 108     have hh':4*b<m*lam:=(div_lt_iff₀ hlam).mp hh
 109     nlinarith only [hh']
 110   have hmargin : (b:ℝ)<lam^2*m/16:=by
 111     have hh:16*b/lam^2<(m:ℝ):=(le_max_right _ _).trans_lt (hL.trans_le hLm)
 112     have hh':16*b<m*lam^2:=(div_lt_iff₀ (sq_pos_of_pos hlam)).mp hh
 113     nlinarith only [hh']
 114   obtain ⟨P,hP⟩:=hN X Y m ((le_max_left _ _).trans hm) hX hY E hcol
 115   exact neighbour_pairs_from_balanced_split m hmpos hY E F lam b hlam (by positivity) hsmall hmargin P
 116     hrow hP (fun x=>Nat.cast_le.mpr (hF x)) hcut
