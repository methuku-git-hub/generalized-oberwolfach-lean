  11 theorem finite_pattern_incidence_square_sum {A I : Type*} [Fintype A] [DecidableEq A] [Fintype I]
  12     (R : I → Finset A) (w : I → ℝ) (hw : ∀ i,0≤w i)
  13     (a q L : ℝ) (ha : 0≤a) (hq : 0≤q)
  14     (hsize : ∀ i,((R i).card:ℝ)≤q) (htotal : (∑ i,w i)≤L)
  15     (hload : ∀ e,(∑ i,if e∈R i then w i else 0)≤a) :
  16     (∑ e,(∑ i,if e∈R i then w i else 0)^2)≤a*q*L := by
  17   have hnonneg (e : A) : (0:ℝ)≤∑ i,if e∈R i then w i else 0 := by
  18     apply Finset.sum_nonneg
  19     intro i _
  20     split_ifs;exact hw i;exact le_refl _
  21   have hsum : (∑ e,∑ i,if e∈R i then w i else 0)≤q*L := by
  22     rw [Finset.sum_comm]
  23     have he (i : I) : (∑ e,if e∈R i then w i else 0)=((R i).card:ℝ)*w i := by
  24       rw [← Finset.sum_filter]
  25       have he : Finset.univ.filter (fun e=>e∈R i)=R i := by ext e;simp
  26       rw [he,Finset.sum_const,nsmul_eq_mul]
  27     simp_rw [he]
  28     calc
  29       _ ≤ ∑ i,q*w i := Finset.sum_le_sum (fun i _=>mul_le_mul_of_nonneg_right (hsize i) (hw i))
  30       _ = q*∑ i,w i := (Finset.mul_sum _ _ _).symm
  31       _ ≤ q*L := mul_le_mul_of_nonneg_left htotal hq
  32   calc
  33     _ ≤ ∑ e,a*(∑ i,if e∈R i then w i else 0) := by
  34       apply Finset.sum_le_sum
  35       intro e _
  36       simpa only [pow_two] using mul_le_mul_of_nonneg_right (hload e) (hnonneg e)
  37     _ = a*(∑ e,∑ i,if e∈R i then w i else 0) := (Finset.mul_sum _ _ _).symm
  38     _ ≤ a*(q*L) := mul_le_mul_of_nonneg_left hsum ha
  39     _ = _ := by ring
