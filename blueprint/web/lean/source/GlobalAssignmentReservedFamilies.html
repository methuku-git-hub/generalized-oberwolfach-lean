  14     [DecidableEq U] [DecidableEq A] [DecidableEq B] [DecidableEq C]
  15     (F:U→SimpleGraph X) (hF:∀w,(F w).IsRegularOfDegree 2)
  16     (E:∀w (D:(F w).ConnectedComponent),SimpleGraph.cycleGraph (Fintype.card D) ≃g D.toSimpleGraph)
  17     (hub:C→U) (component:∀a,(F (hub a)).ConnectedComponent)
  18     (hinj:Function.Injective (fun a=>(⟨hub a,component a⟩:Σw,(F w).ConnectedComponent)))
  19     (ℓ:C→ℕ) (hsize:∀a,Fintype.card (component a)=ℓ a)
  20     (w:U) (components:I↪(F w).ConnectedComponent) (c d cutoff:ℕ)
  21     (hlong:∀i,cutoff≤Fintype.card (components i)) (hshort:∀a,ℓ a<cutoff)
  22     (hcapacity:Fintype.card (A⊕B)≤∑i:I,Fintype.card (components i)/(c*d+5)):
  23     let J:={a // hub a=w}
  24     let cycles:=assignedComponentsAtHub F hub component hinj w
  25     ∃(index:A⊕B↪LongReservedBlockIndex (F w) components c d)
  26       (ext:∀i:A⊕B,Fin (c*d+5)↪X) (inner:∀i:A⊕B,Fin (c*d+1)↪X),
  27       (∀i,ext i=longReservedSigmaBlock (F w) components c d (fun j=>E w (components j)) (index i) d (le_refl _)) ∧
  28       (∀i k,inner i k=ext i ⟨k.val+2,by omega⟩) ∧
  29       (∀i j,i≠j→Disjoint (Finset.univ.image (ext i)) (Finset.univ.image (ext j))) ∧
  30       ∃(P:∀i:A⊕B,(F w).Walk (ext i 0) (ext i (Fin.last (c*d+4)))) (root:J→X)
  31         (Cyc:∀j:J,(F w).Walk (root j) (root j)),
  32       (∀i,(P i).IsPath) ∧ (∀i,(P i).length=c*d+4) ∧ (∀i,(P i).support=List.ofFn (ext i)) ∧
  33       (∀i j,i≠j→Disjoint (P i).support.toFinset (P j).support.toFinset) ∧
  34       (∀j,(Cyc j).IsCycle) ∧ (∀j,(Cyc j).length=ℓ j.val) ∧
  35       (∀j k,k<(Cyc j).length→factorCycleOrientation (F w) (E w) ((Cyc j).getVert k)=(Cyc j).getVert (k+1)) ∧
  36       (∀j,(Cyc j).toSubgraph.verts=(cycles j).supp) ∧
  37       (∀i j,i≠j→Disjoint (Cyc i).support.toFinset (Cyc j).support.toFinset) ∧
  38       (∀i j,Disjoint (P i).support.toFinset (Cyc j).support.toFinset):=by

  44   exact assigned_reserved_source_families (F w) (hF w) (E w) components cycles c d cutoff hlong hs
