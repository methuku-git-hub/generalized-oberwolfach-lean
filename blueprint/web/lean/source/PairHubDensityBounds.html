   6 theorem pair_hub_density_bounds (r:ℕ) (p pmin s₁ s₂ smin:ℝ)
   7     (hpmin:0<pmin) (hp:pmin≤p) (hp1:p≤1)
   8     (hsmin:0<smin) (hs₁:smin≤s₁) (hs₂:smin≤s₂) (hs₁1:s₁≤1) (hs₂1:s₂≤1):
   9     0<pmin^r*(smin*smin)^(r-1) ∧
  10       pmin^r*(smin*smin)^(r-1)≤p^r*(s₁*s₂)^(r-1) ∧
  11       p^r*(s₁*s₂)^(r-1)≤1 := by
  12   have hp0:0≤p:=le_trans hpmin.le hp
  13   have hs10:0≤s₁:=le_trans hsmin.le hs₁
  14   have hs20:0≤s₂:=le_trans hsmin.le hs₂
  15   have hsm:0≤smin*smin:=mul_nonneg hsmin.le hsmin.le
  16   have hsprod:smin*smin≤s₁*s₂:=mul_le_mul hs₁ hs₂ hsmin.le hs10
  17   have hsprod1:s₁*s₂≤1:=by nlinarith [mul_nonneg hs10 (sub_nonneg.mpr hs₂1)]
  18   constructor
  19   · positivity
  20   constructor
  21   · exact mul_le_mul (pow_le_pow_left₀ hpmin.le hp _) (pow_le_pow_left₀ hsm hsprod _) (pow_nonneg hsm _) (pow_nonneg hp0 _)
  22   · have hpbound:p^r≤1:=by simpa using pow_le_one₀ hp0 hp1
