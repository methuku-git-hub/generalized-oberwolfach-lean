  16 theorem large_even_multiple_reciprocal (r : ℕ) (hr : 0<r) (α : ℝ) (hα : 0<α) :
  17     ∃ q : ℕ,0<q ∧ Even q ∧ r∣q ∧ 1/(q+1:ℝ)<α/4 := by
  18   let k : ℕ := ⌈4/α⌉₊+1
  19   let q := 2*r*k
  20   have hk : 0<k := by dsimp [k];omega
  21   have hq : 0<q := by dsimp [q];positivity
  22   refine ⟨q,hq,?_,?_,?_⟩
  23   · refine ⟨r*k,?_⟩
  24     dsimp [q]
  25     ring
  26   · refine ⟨2*k,?_⟩
  27     dsimp [q]
  28     ring
  29   · have hkr : 4/α<(k:ℝ) := by
  30       have hh := Nat.le_ceil (4/α)
  31       dsimp [k]
  32       push_cast
  33       linarith
  34     have hkq : k≤q := by
  35       dsimp [q]
  36       nlinarith
  37     have hqr : 4/α<(q:ℝ)+1 := by
  38       have hh : (k:ℝ)≤q := by exact_mod_cast hkq
  39       linarith
  40     have hm := (div_lt_iff₀ hα).mp hqr
  41     apply (div_lt_iff₀ (show (0:ℝ)<q+1 by positivity)).mpr
  42     nlinarith

  56 theorem cycle_fractional_positive_margin (r : ℕ) (hr : 3≤r) (α : ℝ) (hα : 0<α) :
  57     ∃ N : ℕ,∀ n≥N,∀ G : SimpleGraph (Fin n),
  58       (∀ v,(cycleThreshold r+α)*(n:ℝ)≤(G.degree v:ℝ)) →
  59       HasFractionalDecomposition G (fun H => IsEmbeddedCycle H r) := by
  60   obtain ⟨q,hq,hqeven,hrq,hqsmall⟩ := large_even_multiple_reciprocal r (by omega) α hα
  61   let ξ := min (α/8) (1/(2*(q+1:ℝ)))
  62   have hξ : 0<ξ := lt_min (by positivity) (by positivity)
  63   have hξsmall : ξ≤1/(q+1:ℝ) := by
  64     have hq1 : (0:ℝ)<q+1 := by positivity
  65     have hh := min_le_right (α/8) (1/(2*(q+1:ℝ)))
  66     have hc : 1/(2*(q+1:ℝ))≤1/(q+1:ℝ) := by
  67       apply one_div_le_one_div_of_le hq1
  68       linarith
  69     exact hh.trans hc
  70   have hloss : 1/(q+1:ℝ)+ξ<α/2 := by
  71     have hh := min_le_left (α/8) (1/(2*(q+1:ℝ)))
  72     dsimp only [ξ]
  73     linarith only [hh,hqsmall,hα]
  74   obtain ⟨N₁,hN₁⟩ := dense_divisible_complement_colouring q r hqeven (by omega) hrq α ξ hα hξ hξsmall
  75   obtain ⟨N₂,hN₂⟩ := cycleThreshold_positive_margin r (α:=α/2) (by positivity)
  76   refine ⟨max N₁ N₂,?_⟩
  77   intro n hn G hdeg
  78   have hdense (v : Fin n) : (1/2+α)*Fintype.card (Fin n)≤(G.degree v:ℝ) := by
  79     simp only [Fintype.card_fin]
  80     have hh := mul_le_mul_of_nonneg_right (cycleThreshold_half_le hr) (Nat.cast_nonneg (α:=ℝ) n)
  81     nlinarith [hdeg v]
  82   obtain ⟨τ,hdiv,hτdeg⟩ := hN₁ (Fin n) (by simpa only [Fintype.card_fin] using (le_max_left _ _).trans hn) G hdense
  83   have hdec (a : Fin (q+1)) : HasCycleDecomposition (G\edgeColourGraph G τ a) r := by
  84     apply hN₂ n ((le_max_right _ _).trans hn) (G\edgeColourGraph G τ a)
  85     · simpa only [CycleDivisible,← SimpleGraph.card_neighborSet_eq_degree,
  86         SimpleGraph.edgeFinset_card, ← Nat.card_eq_fintype_card] using hdiv a
  87     · intro v
  88       have hc := congrArg (fun t : ℕ => (t:ℝ))
  89         (degree_complement_add G (edgeColourGraph G τ a) (edgeColourGraph_le G τ a) v)
  90       simp only [Nat.cast_add] at hc
  91       have hJ := hτdeg a v
  92       simp only [Fintype.card_fin] at hJ
  93       have hDn : (G.degree v:ℝ)≤n := by
  94         exact_mod_cast (by simpa only [Fintype.card_fin] using (G.degree_lt_card_verts v).le)
  95       have hDdiv := div_le_div_of_nonneg_right hDn (show (0:ℝ)≤q+1 by positivity)
  96       have hsmall := mul_le_mul_of_nonneg_right hloss.le (Nat.cast_nonneg (α:=ℝ) n)
  97       have he : (1/(q+1:ℝ)+ξ)*n=(n:ℝ)/(q+1:ℝ)+ξ*n := by ring
  98       rw [he] at hsmall
  99       have hG := hdeg v
 100       simp only [← SimpleGraph.card_neighborSet_eq_degree,← Nat.card_eq_fintype_card] at hc hJ hDdiv hG ⊢
 101       nlinarith only [hc,hJ,hDdiv,hsmall,hG]
 102   exact fractionalDecomposition_of_divisible_complements G (edgeColourGraph G τ) (by omega)
 103     (edgeColourGraph_partition G τ) hdec
