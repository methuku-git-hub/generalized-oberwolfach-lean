  11 theorem long_reserved_interior_list_infix {V:Type*} (ℓ:ℕ) (f:Fin (ℓ+5)↪V):
  12     List.ofFn (longReservedInteriorEmbedding ℓ f) <:+: List.ofFn f := by
  13   have heq:List.ofFn (longReservedInteriorEmbedding ℓ f)=
  14       ((List.ofFn f).drop 2).take (ℓ+1):=by
  15     apply List.ext_getElem
  16     · simp; omega
  17     · intro i hi hj
  18       have hi':i<ℓ+1:=by simpa using hi
  19       simp only [List.getElem_take,List.getElem_drop,List.getElem_ofFn]
  20       change f ⟨i+2,by omega⟩=f ⟨2+i,by omega⟩
  21       congr 1
  22       apply Fin.ext
  23       simp only [Fin.val_mk]
  24       omega
  25   rw [heq]
  26   exact (List.take_prefix (ℓ+1) ((List.ofFn f).drop 2)).isInfix.trans
  27     (List.drop_suffix 2 (List.ofFn f)).isInfix
