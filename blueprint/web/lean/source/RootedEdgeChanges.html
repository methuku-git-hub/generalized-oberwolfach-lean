   4 /-! Counts of tuples affected by a changed edge. At least one endpoint is
   5 free. The degree of the realised change relation, not a sampling density,

  17 def fillFreeCoordinate (i : I) (x : X) (g : {j : I // j≠i} → X) : I → X :=
  18   fun j=>if h : j=i then x else g ⟨j,h⟩

  20 theorem sum_bound_from_one_coordinate (i : I) (F : (I → X) → ℝ) (d : ℝ)
  21     (h : ∀ g : {j : I // j≠i} → X,∑ x,F (fillFreeCoordinate i x g)≤d) :
  22     (∑ f : I → X,F f)≤(Fintype.card X:ℝ)^(Fintype.card I-1)*d := by
  23   classical
  24   have he : (∑ f : I → X,F f)=
  25       ∑ g : {j : I // j≠i} → X,∑ x,F (fillFreeCoordinate i x g) := by
  26     rw [← (Equiv.funSplitAt i X).symm.sum_comp F,Fintype.sum_prod_type,Finset.sum_comm]
  27     apply Finset.sum_congr rfl
  28     intro g _
  29     apply Finset.sum_congr rfl
  30     intro x _
  31     congr 1
  32     funext j
  33     by_cases hj : j=i
  34     · subst j;simp [Equiv.funSplitAt,Equiv.piSplitAt,fillFreeCoordinate]
  35     · simp [Equiv.funSplitAt,Equiv.piSplitAt,fillFreeCoordinate,hj]
  36   rw [he]
  37   calc
  38     _ ≤ ∑ _g : {j : I // j≠i} → X,d := Finset.sum_le_sum (fun g _=>h g)
  39     _ = _ := by
  40       simp only [Finset.sum_const,Finset.card_univ,nsmul_eq_mul,Fintype.card_fun,
  41         Fintype.card_subtype_compl,Fintype.card_unique,Nat.cast_pow]

  43 def rootedPositionValue (f : I → X) : X⊕I → X := Sum.elim id f

  45 def RootedEdgeHasFreeEndpoint (a b : X⊕I) : Prop :=
  46   match a,b with
  47   | Sum.inl _,Sum.inl _ => False
  48   | Sum.inr i,Sum.inr j => i≠j
  49   | _,_ => True

  51 theorem rooted_edge_change_count
  52     (D : X → X → Prop) (d : ℝ)
  53     (hout : ∀ x,(∑ y,if D x y then (1:ℝ) else 0)≤d)
  54     (hin : ∀ y,(∑ x,if D x y then (1:ℝ) else 0)≤d)
  55     (a b : X⊕I) (hab : RootedEdgeHasFreeEndpoint a b)
  56     (σ τ : Equiv.Perm X) :
  57     (∑ f : I → X,if D (σ (rootedPositionValue f a)) (τ (rootedPositionValue f b)) then (1:ℝ) else 0)≤
  58       (Fintype.card X:ℝ)^(Fintype.card I-1)*d := by
  59   classical
  60   cases a with
  61   | inl x =>
  62     cases b with
  63     | inl y => exact hab.elim
  64     | inr j =>
  65       apply sum_bound_from_one_coordinate j
  66       intro g
  67       simp only [rootedPositionValue,Sum.elim_inl,Sum.elim_inr,id_eq,fillFreeCoordinate,dif_pos rfl,dite_true]
  68       rw [Equiv.sum_comp τ (fun y=>if D (σ x) y then (1:ℝ) else 0)]
  69       exact hout (σ x)
  70   | inr i =>
  71     cases b with
  72     | inl y =>
  73       apply sum_bound_from_one_coordinate i
  74       intro g
  75       simp only [rootedPositionValue,Sum.elim_inl,Sum.elim_inr,id_eq,fillFreeCoordinate,dif_pos rfl,dite_true]
  76       rw [Equiv.sum_comp σ (fun x=>if D x (τ y) then (1:ℝ) else 0)]
  77       exact hin (τ y)
  78     | inr j =>
  79       have hne : i≠j := hab
  80       apply sum_bound_from_one_coordinate j
  81       intro g
  82       simp only [rootedPositionValue,Sum.elim_inr,fillFreeCoordinate,dif_pos rfl,dif_neg hne,dite_true]
  83       rw [Equiv.sum_comp τ (fun y=>if D (σ (g ⟨i,hne⟩)) y then (1:ℝ) else 0)]
  84       exact hout (σ (g ⟨i,hne⟩))
