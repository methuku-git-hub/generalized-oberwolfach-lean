  10 theorem triangle_transformer_available_neighbours
  11     {V:Type} [Fintype V] [DecidableEq V] (G:SimpleGraph V) (A:Finset V) (β:ℝ)
  12     (hβ:0<β) (hβ1:β≤1)
  13     (hdeg:∀ v,(3/4+β)*(Fintype.card V:ℝ)≤(G.degree v:ℝ))
  14     (hmiss:(((Finset.univ:Finset V)\A).card:ℝ)≤(β/2)*(Fintype.card V:ℝ)):
  15     (1/2)*(Fintype.card V:ℝ)≤(A.card:ℝ) ∧
  16     ∀ v,(3/4+β/2)*(A.card:ℝ)≤((G.neighborFinset v∩A).card:ℝ) := by
  17   have htotal:(A.card:ℝ)+(((Finset.univ:Finset V)\A).card:ℝ)=Fintype.card V:=by
  18     have hc := Finset.card_sdiff_add_card_eq_card (Finset.subset_univ A)
  19     have he : A.card + ((Finset.univ:Finset V)\A).card = Fintype.card V := by simpa [add_comm] using hc
  20     exact_mod_cast he
  21   have hn:0≤(Fintype.card V:ℝ):=by positivity
  22   have ha:(A.card:ℝ)≤Fintype.card V:=Nat.cast_le.mpr (Finset.card_le_univ A)
  23   constructor
  24   · nlinarith
  25   · intro v
  26     have hcard:(G.neighborFinset v∩A).card + (G.neighborFinset v\A).card=G.degree v:=by
  27       rw [Finset.card_inter_add_card_sdiff,SimpleGraph.card_neighborFinset_eq_degree]
  28     have hsubset:G.neighborFinset v\A⊆(Finset.univ:Finset V)\A:=by
  29       intro x hx
  30       exact Finset.mem_sdiff.mpr ⟨Finset.mem_univ x,(Finset.mem_sdiff.mp hx).2⟩
  31     have hsmall:((G.neighborFinset v\A).card:ℝ)≤(((Finset.univ:Finset V)\A).card:ℝ):=
  32       Nat.cast_le.mpr (Finset.card_le_card hsubset)
  33     have hc:((G.neighborFinset v∩A).card:ℝ)+((G.neighborFinset v\A).card:ℝ)=(G.degree v:ℝ):=by exact_mod_cast hcard
  34     have hv:=hdeg v
  35     nlinarith

  37 theorem triangle_transformer_available_sample_factors
  38     (L:ℕ) (β:ℝ) (hβ:0<β) (hβ1:β≤1) :
  39     ∃ s₀:ℕ,∀ s:ℕ,s₀≤s → 3∣s → ∃ N:ℕ,∀ (V:Type) [Fintype V] [DecidableEq V],
  40       N≤Fintype.card V → ∀ (G:SimpleGraph V) (A Z:Finset V),
  41       Z.card≤L → 3∣Z.card →
  42       (∀ v,(3/4+β)*(Fintype.card V:ℝ)≤(G.degree v:ℝ)) →
  43       (((Finset.univ:Finset V)\A).card:ℝ)≤(β/2)*(Fintype.card V:ℝ) →
  44       ∃ S:Finset V,S.card=s ∧ S⊆A ∧ Disjoint S Z ∧
  45       ∃ M₁ M₂:Finset (OrdinaryCycle G 3),
  46         (∀ C∈M₁,∀ D∈M₁,C≠D → Disjoint (ordinaryCycleVertices G 3 C) (ordinaryCycleVertices G 3 D)) ∧
  47         M₁.biUnion (ordinaryCycleVertices G 3)=S∪Z ∧
  48         (∀ C∈M₂,∀ D∈M₂,C≠D → Disjoint (ordinaryCycleVertices G 3 C) (ordinaryCycleVertices G 3 D)) ∧
  49         M₂.biUnion (ordinaryCycleVertices G 3)=S ∧
  50         (∀ C∈M₁,∀ D∈M₂,Disjoint C.val.val.edgeFinset D.val.val.edgeFinset) ∧
  51         (∀ C∈M₁,∀ v w,C.val.val.Adj v w → v∈S ∨ w∈S) := by
  52   obtain ⟨s₀,hs⟩:=triangle_transformer_sample_factors_uniform_old_size L β (1/2) hβ hβ1 (by norm_num)
  53   refine ⟨s₀,?_⟩
  54   intro s hs₀ hdiv
  55   obtain ⟨N,hN⟩:=hs s hs₀ hdiv
  56   refine ⟨N,?_⟩
  57   intro V _ _ hn G A Z hz hZ hdeg hmiss
  58   obtain ⟨hA,hAdeg⟩:=triangle_transformer_available_neighbours G A β hβ hβ1 hdeg hmiss
  59   exact hN V hn G A Z hz hZ hA (fun v _=>hAdeg v)
  60 end Oberwolfach
