  10 def BalancedSampleOrientationEvent {V:Type} [Fintype V] [LinearOrder V]
  11     (G:SimpleGraph V) (α:ℝ) (σ:Sym2 V→Bool):Prop :=
  12   ∀ F:SimpleGraph V,(∀v,(F.degree v:ℝ)≤α*Fintype.card V/100)→
  13     ∃ M:V→V→ℕ,
  14       (∀x y,M x y+M y x=if G.Adj x y then 1 else 0) ∧
  15       (∀v,(∑w,M v w)=G.degree v/2 ∧ (∑w,M w v)=G.degree v/2) ∧
  16       (∀x y,M x y≠sampledOrientation G σ x y ∨ M y x≠sampledOrientation G σ y x→¬F.Adj x y) ∧
  17       ∀x,((Finset.univ.filter (fun y=>M x y≠sampledOrientation G σ x y ∨
  18         M y x≠sampledOrientation G σ y x)).card:ℝ)≤4*(Fintype.card V:ℝ)^(4/5:ℝ)
