  12 theorem prescribed_source_model_oriented (ps:List (I×PrescribedSourcePiece X))
  13     (hshape:∀ p∈ps,p.2.Shape) (hapart:ps.Pairwise PrescribedPiecesApart):
  14     ∀ i x y,prescribedSourceDigraph ps i x y→¬prescribedSourceDigraph ps i y x := by
  15   intro i x y hxy hyx
  16   obtain ⟨p,hp,hpi,hpa⟩:=hxy
  17   obtain ⟨q,hq,hqi,hqa⟩:=hyx
  18   have heq:p=q:=by
  19     rcases prescribed_pairwise_rel_or_eq PrescribedPiecesApart prescribed_pieces_apart_symmetric ps hapart p hp q hq with heq|hdis
  20     · exact heq
  21     · rcases hdis with hne|hdis
  22       · exact (hne (hpi.trans hqi.symm)).elim
  23       · have hpx:x∈p.2.vertices:=(prescribed_source_piece_arc_endpoints p.2 (x,y) hpa).1
  24         have hqx:x∈q.2.vertices:=(prescribed_source_piece_arc_endpoints q.2 (y,x) hqa).2
  25         exact ((Finset.disjoint_left.mp hdis (List.mem_toFinset.mpr hpx)) (List.mem_toFinset.mpr hqx)).elim
  26   subst q
  27   exact prescribed_piece_no_reverse p.2 (hshape p hp) x y hpa hqa

  29 theorem prescribed_source_model_representation (ps:List (I×PrescribedSourcePiece X)):
  30     ∀ i x y,(prescribedSourceGraph ps i).Adj x y→
  31       prescribedSourceDigraph ps i x y ∨ prescribedSourceDigraph ps i y x := by
  32   intro i x y hxy
  33   have he:s(x,y)∈(prescribedSourceGraph ps i).edgeFinset:=SimpleGraph.mem_edgeFinset.mpr hxy
  34   obtain ⟨p,hp,hpi,hpE⟩:=prescribed_source_model_edges ps i _ he
  35   rw [prescribed_piece_edges_are_arc_images] at hpE
  36   obtain ⟨e,he,hphys⟩:=Finset.mem_image.mp hpE
  37   rcases e with ⟨u,v⟩
  38   have ha:prescribedSourceDigraph ps i u v:=⟨p,hp,hpi,he⟩
  39   change s(u,v)=s(x,y) at hphys
  40   rcases Sym2.eq_iff.mp hphys with ⟨hu,hv⟩|⟨hu,hv⟩
  41   · subst u;subst v;exact Or.inl ha
  42   · subst u;subst v;exact Or.inr ha
