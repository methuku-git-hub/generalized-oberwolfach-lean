  14 theorem prescribed_root_positions_count {I V : Type*} [Fintype I] [Fintype V]
  15     [DecidableEq I] [DecidableEq V] (i₀ : I) (S : Finset V) (q : S → I) :
  16     (∑ f : I → V,if (∀ s : S,f (q s)=s.val) then (1:ℝ) else 0)≤
  17       (Fintype.card V:ℝ)^(Fintype.card I-S.card) := by
  18   let cl (v : V) := if hv : v∈S then q ⟨v,hv⟩ else i₀
  19   have hp (f : I → V) : (∀ s : S,f (q s)=s.val) ↔ ∀ v∈S,f (cl v)=v := by
  20     constructor
  21     · intro h v hv
  22       simpa only [cl,dif_pos hv] using h ⟨v,hv⟩
  23     · intro h s
  24       simpa only [cl,dif_pos s.property] using h s.val s.property
  25   have hh := prescribedVertices_weight_le S cl id
  26     (by intro u v huv;exact congrArg Prod.snd huv) (fun _ => (1:ℝ)) 1 (by norm_num) (by intro _;rfl)
  27   simp only [mul_one] at hh
  28   convert hh using 1
  29   apply Finset.sum_congr rfl
  30   intro f _
  31   simp only [hp,id_eq]
  32   split_ifs <;> rfl

  34 theorem functions_cover_roots_count {I V : Type*} [Fintype I] [Fintype V]
  35     [DecidableEq I] [DecidableEq V] (i₀ : I) (S : Finset V) :
  36     (∑ f : I → V,if (∀ v∈S,v∈Set.range f) then (1:ℝ) else 0)≤
  37       (Fintype.card I:ℝ)^S.card*(Fintype.card V:ℝ)^(Fintype.card I-S.card) := by
  38   have hp (f : I → V) : (if (∀ v∈S,v∈Set.range f) then (1:ℝ) else 0)≤
  39       ∑ q : S → I,if (∀ s : S,f (q s)=s.val) then (1:ℝ) else 0 := by
  40     by_cases hf : ∀ v∈S,v∈Set.range f
  41     · let q (s : S) := Classical.choose (hf s.val s.property)
  42       have hq : ∀ s : S,f (q s)=s.val := fun s => Classical.choose_spec (hf s.val s.property)
  43       rw [if_pos hf]
  44       have hh := Finset.single_le_sum (s:=Finset.univ)
  45         (f:=fun q : S → I => if (∀ s : S,f (q s)=s.val) then (1:ℝ) else 0)
  46         (fun _ _ => by split_ifs <;> norm_num) (Finset.mem_univ q)
  47       simpa only [if_pos hq] using hh
  48     · rw [if_neg hf]
  49       exact Finset.sum_nonneg (fun _ _ => by split_ifs <;> norm_num)
  50   calc
  51     _ ≤ ∑ f : I → V,∑ q : S → I,if (∀ s : S,f (q s)=s.val) then (1:ℝ) else 0 :=
  52       Finset.sum_le_sum (fun f _ => hp f)
  53     _ = ∑ q : S → I,∑ f : I → V,if (∀ s : S,f (q s)=s.val) then (1:ℝ) else 0 := Finset.sum_comm
  54     _ ≤ ∑ _q : S → I,(Fintype.card V:ℝ)^(Fintype.card I-S.card) :=
  55       Finset.sum_le_sum (fun q _ => prescribed_root_positions_count i₀ S q)
  56     _ = _ := by simp only [Finset.sum_const,Finset.card_univ,Fintype.card_fun,
  57       Fintype.card_coe,nsmul_eq_mul,Nat.cast_pow]

  59 theorem ordinary_cycle_root_count {V : Type*} [Fintype V] [DecidableEq V]
  60     (r : ℕ) (hr : 0<r) (S : Finset V) :
  61     (∑ C : EmbeddedCycleFamily V r,if (↑S:Set V)⊆C.val.support then (1:ℝ) else 0)≤
  62       (r:ℝ)^S.card*(Fintype.card V:ℝ)^(r-S.card) := by
  63   let g (f : Fin r → V) := if (∀ v∈S,v∈Set.range f) then (1:ℝ) else 0
  64   have hg (f : Fin r → V) : 0≤g f := by dsimp [g];split_ifs <;> norm_num
  65   have hC (C : EmbeddedCycleFamily V r) :
  66       (if (↑S:Set V)⊆C.val.support then (1:ℝ) else 0)≤g (embeddedCycleEmbedding C) := by
  67     by_cases hS : (↑S:Set V)⊆C.val.support
  68     · have hf : ∀ v∈S,v∈Set.range (embeddedCycleEmbedding C : Fin r → V) := by
  69         intro v hv
  70         have hh := hS hv
  71         rw [embeddedCycleEmbedding_spec C,SimpleGraph.support_map] at hh
  72         obtain ⟨i,hi,hiv⟩ := hh
  73         exact ⟨i,hiv⟩
  74       rw [if_pos hS]
  75       dsimp only [g]
  76       split_ifs <;> norm_num
  77     · simp only [hS,ite_false]
  78       exact hg _
  79   calc
  80     _ ≤ ∑ C : EmbeddedCycleFamily V r,g (embeddedCycleEmbedding C) := Finset.sum_le_sum (fun C _ => hC C)
  81     _ ≤ ∑ f : Fin r → V,g f := finite_injective_sum_le _ embeddedCycleEmbedding_injective g hg
  82     _ ≤ _ := by
  83       simpa only [Fintype.card_fin] using functions_cover_roots_count (⟨0,hr⟩:Fin r) S
