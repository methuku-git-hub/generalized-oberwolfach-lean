  10 theorem deleted_path_family_properties (F:SimpleGraph V)
  11     (s₁ s₂:Finset I) (u v:I→V) (P:∀i,F.Walk (u i) (v i))
  12     (hp:∀i∈s₁∪s₂,(P i).IsPath) (c d:ℕ) (hc:0<c) (hd:0<d)
  13     (hlen:∀i∈s₁∪s₂,(P i).length=c*d+4)
  14     (T:I→Finset ℕ) (hT:∀i∈s₂,T i⊆Finset.range d):
  15     ∀ z:DeletedPathIndex s₁ s₂ d T,
  16       (deletedPathWalk F s₁ s₂ u v P c d T hT z).IsPath ∧
  17       deletedPathStart F s₁ s₂ u v P c d T z ≠ deletedPathEnd F s₁ s₂ u v P c d T z := by
  18   intro z
  19   cases z with
  20   | inl i =>
  21     have hi:i.1∈s₁∪s₂:=Finset.mem_union_left _ i.2
  22     have hright:2+c*d≤(P i.1).length := by rw [hlen i.1 hi];omega
  23     have hleft:2<2+c*d := by have hh:=Nat.mul_pos hc hd;omega
  24     exact ⟨walk_interval_isPath (P i.1) (hp i.1 hi) _ _ (by omega),
  25       path_interval_endpoints_ne (P i.1) (hp i.1 hi) _ _ hleft hright⟩
  26   | inr z =>
  27     rcases z with ⟨i,r⟩
  28     rcases r with ⟨⟨a,b⟩,hr⟩
  29     have hi:i.1∈s₁∪s₂:=Finset.mem_union_right _ i.2
  30     have hr':IsSelectedSegmentRun (T i.1) a b := (selected_run_mem d (T i.1) (hT i.1 i.2) a b).mp hr
  31     have hbT:b∈T i.1:=hr'.2.1 b hr'.1 le_rfl
  32     have hbd:b<d:=by have hh:=hT i.1 i.2 hbT;simpa using hh
  33     have hh:=selected_run_walk_properties (P i.1) (hp i.1 hi) c d a b hc hr'.1 hbd (hlen i.1 hi)
  34     exact ⟨hh.1,hh.2.2.1⟩
