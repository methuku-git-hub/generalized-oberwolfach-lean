  10 variable {X V:Type} [Fintype X] [Fintype V] [DecidableEq X] [DecidableEq V]

  12 theorem prescribed_positions (Q:SimpleGraph X) (D:X→X→Prop)
  13     (hdeg:∀x,Q.degree x≤2) (hrep:∀x y,Q.Adj x y→D x y ∨ D y x)
  14     (hDQ:∀x y,D x y→Q.Adj x y) (horiented:∀x y,D x y→¬D y x)
  15     (A:Finset X) (hA:∀x∈A,∀y∈A,¬Q.Adj x y) (f:X→V) (hinj:Set.InjOn f (↑A:Set X))
  16     (M:Finset (V×V)) (hM:IsPrescribedOrientedMatching M) (Z:Finset V)
  17     (hOldM:Disjoint (A.image f) (prescribedMatchingEndpoints M))
  18     (hOldZ:Disjoint (A.image f) Z) (hMZ:Disjoint (prescribedMatchingEndpoints M) Z)
  19     (hedges:Fintype.card X≤Q.edgeFinset.card+A.card)
  20     (hsize:8*A.card+8*M.card+4*Z.card+10≤Fintype.card X):
  21     ∃ (B:Finset X) (g:X→V),A⊆B ∧ Set.InjOn g (↑B:Set X) ∧ (∀x∈A,g x=f x) ∧
  22       B.image g=A.image f∪prescribedMatchingEndpoints M∪Z ∧
  23       (prescribedInducedArcs D B).image (Prod.map g g)=M := by
  24   let ms:=M.toList
  25   let zs:=Z.toList
  26   have hmn:(ms.flatMap prescribedPairVertices).Nodup:=prescribed_target_matching_list_nodup M hM
  27   have hzn:zs.Nodup:=Finset.nodup_toList Z
  28   have hend:(ms.flatMap prescribedPairVertices).toFinset=prescribedMatchingEndpoints M:=
  29     prescribed_target_matching_list_endpoints M
  30   have hz:zs.toFinset=Z:=by simp [zs]
  31   have hmz:List.Disjoint (ms.flatMap prescribedPairVertices) zs:=by
  32     apply List.disjoint_left.mpr
  33     intro x hx hy
  34     apply (Finset.disjoint_left.mp hMZ)
  35     · rw [←hend];exact List.mem_toFinset.mpr hx
  36     · rw [←hz];exact List.mem_toFinset.mpr hy
  37   have hold:Disjoint (A.image f) (ms.flatMap prescribedPairVertices++zs).toFinset:=by
  38     rw [List.toFinset_append,hend,hz]
  39     exact Finset.disjoint_union_right.mpr ⟨hOldM,hOldZ⟩
  40   have hsize':8*A.card+8*ms.length+4*zs.length+10≤Fintype.card X:=by simpa [ms,zs] using hsize
  41   obtain ⟨B,g,hAB,hginj,hgpres,himage,harcs⟩:=prescribed_positions_from_lists Q D hdeg hrep hDQ horiented
  42     A hA f hinj ms zs hmn hzn hmz hold hedges hsize'
  43   refine ⟨B,g,hAB,hginj,hgpres,?_,?_⟩
  44   · simpa [hend,hz] using himage
  45   · simpa [ms] using harcs
  46 end Oberwolfach
