  10 theorem saturated_vertex_count_bound {V:Type} [Fintype V] (f:V→ℕ) (t T:ℝ)
  11     (ht:0<t) (hused:(∑ v,(f v:ℝ))≤T):
  12     let S:=Finset.univ.filter (fun v=>(t:ℝ)≤f v);
  13     t*(S.card:ℝ)≤T := by
  14   intro S
  15   have hpoint:∑ v∈S,t≤∑ v∈S,(f v:ℝ):=by
  16     apply Finset.sum_le_sum
  17     intro v hv
  18     exact (Finset.mem_filter.mp hv).2
  19   have hsub:(∑ v∈S,(f v:ℝ))≤∑ v,(f v:ℝ):=
  20     Finset.sum_le_sum_of_subset_of_nonneg (Finset.subset_univ S) (fun v _ _=>by positivity)
  21   simp only [Finset.sum_const,nsmul_eq_mul] at hpoint
  22   nlinarith
