  14 def ordinaryRowCorrection {V U : Type*} [Fintype U]
  15     {Q F : SimpleGraph V} {H : V → U → Prop} {r : ℕ} (v : V)
  16     (z : U → ℝ)
  17     (T : {u // H v u} → {u // H v u} → OrdinaryWheel Q F H r → ℝ)
  18     (W : OrdinaryWheel Q F H r) : ℝ :=
  19   ∑ p : {u // H v u} × {u // H v u},
  20     (z p.1.val/(Fintype.card {u // H v u}:ℝ))*T p.1 p.2 W

  22 theorem ordinaryRowCorrection_loads {V U : Type*} [Fintype V] [Fintype U]
  23     [DecidableEq V] [DecidableEq U]
  24     {Q F : SimpleGraph V} {H : V → U → Prop} {r : ℕ} (v : V)
  25     (z : U → ℝ) (hz : ∑ u : {u // H v u},z u.val=0)
  26     (hout : ∀ u,¬H v u → z u=0)
  27     (T : {u // H v u} → {u // H v u} → OrdinaryWheel Q F H r → ℝ)
  28     (hR : ∀ i j x y,ordinaryRimLoad (T i j) x y=0)
  29     (hS : ∀ i j x h,ordinarySpokeLoad (T i j) x h=
  30       (if x=v then (1:ℝ) else 0)*((if h=i.val then 1 else 0)-(if h=j.val then 1 else 0))) :
  31     (∀ x y,ordinaryRimLoad (ordinaryRowCorrection v z T) x y=0) ∧
  32     (∀ x h,ordinarySpokeLoad (ordinaryRowCorrection v z T) x h=
  33       (if x=v then (1:ℝ) else 0)*z h) := by
  34   constructor
  35   · intro x y
  36     unfold ordinaryRowCorrection
  37     rw [ordinaryRimLoad_combination]
  38     simp only [hR,mul_zero,Finset.sum_const_zero]
  39   · intro x h
  40     unfold ordinaryRowCorrection
  41     rw [ordinarySpokeLoad_combination]
  42     simp only [hS,Fintype.sum_prod_type]
  43     have he := selected_pair_transfer_identity (H v) z hz hout h
  44     calc
  45       _ = (if x=v then (1:ℝ) else 0)*
  46           ((∑ i : {u // H v u},∑ j : {u // H v u},
  47             z i.val*((if h=i.val then (1:ℝ) else 0)-(if h=j.val then 1 else 0)))/
  48             (Fintype.card {u // H v u}:ℝ)) := by
  49         simp only [Finset.sum_div,Finset.mul_sum]
  50         apply Finset.sum_congr rfl
  51         intro i _
  52         apply Finset.sum_congr rfl
  53         intro j _
  54         ring
  55       _ = _ := by rw [he]
