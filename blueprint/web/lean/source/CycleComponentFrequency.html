  11 theorem component_size_vertex_count {V:Type} [Fintype V] [DecidableEq V]
  12     (G:SimpleGraph V) (r:ℕ):
  13     let T:=Finset.univ.filter (fun C:G.ConnectedComponent=>Fintype.card C=r)
  14     (Finset.univ.filter (fun v:V=>Fintype.card (G.connectedComponentMk v)=r)).card=r*T.card :=by
  15   intro T
  16   have hs:∀C∈T,(Finset.univ.filter (fun v:V=>G.connectedComponentMk v=C)).card=r:=by
  17     intro C hC
  18     have hCsize:Fintype.card C=r:=(Finset.mem_filter.mp hC).2
  19     have hset:Finset.univ.filter (fun v:V=>G.connectedComponentMk v=C)=C.supp.toFinset:=by
  20       ext v
  21       simp only [Finset.mem_filter,Finset.mem_univ,true_and,Set.mem_toFinset,
  22         SimpleGraph.ConnectedComponent.mem_supp_iff]
  23     rw [hset]
  24     rw [Set.toFinset_card]
  25     have heq:Fintype.card C.supp=Fintype.card C:=Fintype.card_congr (Equiv.refl _)
  26     exact heq.trans hCsize
  27   have h:=card_of_uniform_fibers G.connectedComponentMk T r hs
  28   apply Eq.trans ?_ h
  29   congr 1
  30   ext v
  31   simp only [Finset.mem_filter,Finset.mem_univ,true_and,T]
  32 end Oberwolfach

  37 theorem component_fraction_times_order_div_size {V:Type} [Fintype V] [DecidableEq V]
  38     (G:SimpleGraph V) (r:ℕ) (hr:0<r) (hV:0<Fintype.card V):
  39     let T:=Finset.univ.filter (fun C:G.ConnectedComponent=>Fintype.card C=r)
  40     (((Finset.univ.filter (fun v:V=>Fintype.card (G.connectedComponentMk v)=r)).card:ℝ)/
  41       (Fintype.card V:ℝ))*(Fintype.card V:ℝ)/(r:ℝ)=(T.card:ℝ) :=by
  42   intro T
  43   have hn:(Fintype.card V:ℝ)≠0:=by exact_mod_cast (Nat.ne_of_gt hV)
  44   have hrr:(r:ℝ)≠0:=by exact_mod_cast (Nat.ne_of_gt hr)
  45   have hc:=component_size_vertex_count G r
  46   dsimp only at hc
  47   rw [hc]
  48   push_cast
  49   field_simp
  50   rfl
  51 end Oberwolfach

  56 theorem long_preparation_fraction_component_count {n K:ℕ} (hn:0<n)
  57     (F:SimpleGraph (Fin n)) (hF:F.IsRegularOfDegree 2) (i:Fin K):
  58     let fraction:=((Finset.univ.filter (fun v=>longPreparationCycleClass F hF K v=some i)).card:ℝ)/n
  59     let T:=Finset.univ.filter (fun C:F.ConnectedComponent=>Fintype.card C=i.val+1)
  60     fraction*n/((i.val+1:ℕ):ℝ)=(T.card:ℝ) :=by
  61   intro fraction T
  62   have hset:Finset.univ.filter (fun v=>longPreparationCycleClass F hF K v=some i)=
  63       Finset.univ.filter (fun v=>Fintype.card (F.connectedComponentMk v)=i.val+1):=by
  64     ext v
  65     simp only [Finset.mem_filter,Finset.mem_univ,true_and]
  66     exact (long_preparation_cycle_class F hF K v).2 i
  67   dsimp only [fraction]
  68   rw [hset]
  69   simpa only [Fintype.card_fin] using
  70     component_fraction_times_order_div_size F (i.val+1) (by omega) (by simpa only [Fintype.card_fin] using hn)
  71 end Oberwolfach
