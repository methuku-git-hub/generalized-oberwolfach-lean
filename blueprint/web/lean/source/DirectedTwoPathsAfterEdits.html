  10 theorem directed_two_paths_after_edits
  11     {V:Type} [Fintype V] [DecidableEq V] (S:SimpleGraph V)
  12     (Raw:V→V→Prop) (M:V→V→ℕ) (b:ℝ) (hb:0≤b)
  13     (hhost:∀x y,0<M x y→S.Adj x y)
  14     (hchange:∀v,((Finset.univ.filter (fun w=>
  15       (Raw v w ∧ ¬0<M v w) ∨ (Raw w v ∧ ¬0<M w v))).card:ℝ)≤b):
  16     ∀x y,(((Finset.univ.filter (fun z=>Raw x z ∧ Raw z y)).card:ℝ)-2*b)≤
  17       ((Finset.univ.filter (fun z=>0<M x z ∧ 0<M z y ∧ S.Adj x z ∧ S.Adj z y)).card:ℝ):=by
  18   intro x y
  19   let R:=Finset.univ.filter (fun z=>Raw x z ∧ Raw z y)
  20   let T:=Finset.univ.filter (fun z=>0<M x z ∧ 0<M z y ∧ S.Adj x z ∧ S.Adj z y)
  21   let X:=Finset.univ.filter (fun z=>(Raw x z ∧ ¬0<M x z) ∨ (Raw z x ∧ ¬0<M z x))
  22   let Y:=Finset.univ.filter (fun z=>(Raw y z ∧ ¬0<M y z) ∨ (Raw z y ∧ ¬0<M z y))
  23   have hsub:R⊆(T∪X)∪Y:=by
  24     intro z hz
  25     have hh:Raw x z ∧ Raw z y:=(Finset.mem_filter.mp hz).2
  26     by_cases hx:0<M x z
  27     · by_cases hy:0<M z y
  28       · exact Finset.mem_union_left _ (Finset.mem_union_left _ (Finset.mem_filter.mpr ⟨Finset.mem_univ _,hx,hy,hhost _ _ hx,hhost _ _ hy⟩))
  29       · exact Finset.mem_union_right _ (Finset.mem_filter.mpr ⟨Finset.mem_univ _,Or.inr ⟨hh.2,hy⟩⟩)
  30     · exact Finset.mem_union_left _ (Finset.mem_union_right _ (Finset.mem_filter.mpr ⟨Finset.mem_univ _,Or.inl ⟨hh.1,hx⟩⟩))
  31   have hc:R.card≤T.card+X.card+Y.card:=by
  32     calc
  33       R.card≤((T∪X)∪Y).card:=Finset.card_le_card hsub
  34       _≤(T∪X).card+Y.card:=Finset.card_union_le _ _
  35       _≤T.card+X.card+Y.card:=Nat.add_le_add_right (Finset.card_union_le _ _) _
  36   have hcR:((R.card:ℕ):ℝ)≤T.card+X.card+Y.card:=by exact_mod_cast hc
  37   have hx:=hchange x
  38   have hy:=hchange y
  39   change (X.card:ℝ)≤b at hx
  40   change (Y.card:ℝ)≤b at hy
  41   change (R.card:ℝ)-2*b≤T.card
  42   linarith
  43 end Oberwolfach
