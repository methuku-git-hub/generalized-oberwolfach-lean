  18 theorem selected_hub_factor_wheels (Q F:SimpleGraph V) (H:V→U→Prop) (r:ℕ)
  19     (T Y:J→Finset V) (u:J↪U)
  20     (D:∀ j,Finset (LocalOrdinaryCycle Q F (T j) r))
  21     (hD:∀ j,∀ C∈D j,∀ C'∈D j,C≠C'→Disjoint (localOrdinaryVertices C) (localOrdinaryVertices C'))
  22     (hcover:∀ j,(D j).biUnion localOrdinaryVertices=Y j)
  23     (hH:∀ j,∀ v∈Y j,H v (u j))
  24     (hcross:∀ i j,i≠j→Disjoint (localOrdinaryFactorGraph (D i)) (localOrdinaryFactorGraph (D j))):
  25     ∃ W:(Σ j,D j)→OrdinaryWheel Q F H r,
  26       (∀ i,ordinaryWheelRim (W i)=i.2.val.val.val.val ∧ ordinaryWheelHub (W i)=u i.1) ∧
  27       wheelFamilyRim W=(⨆ j,localOrdinaryFactorGraph (D j)) ∧
  28       (∀ a z,wheelFamilySpokes W a z↔∃ j,u j=z ∧ a∈Y j) ∧
  29       (∀ i j,i≠j→Disjoint (ordinaryWheelResources (W i)) (ordinaryWheelResources (W j))) ∧
  30       HasOrdinaryWheelDecomposition (wheelFamilyRim W) (wheelFamilySpokes W) r:=by
  31   have hmem (j:J) (C:D j) (a:V) (ha:a∈C.val.val.val.val.support):a∈Y j:=by
  32     rw [←hcover j]
  33     exact Finset.mem_biUnion.mpr ⟨C.val,C.property,Set.mem_toFinset.mpr ha⟩
  34   let W:(Σ j,D j)→OrdinaryWheel Q F H r:=fun i=>
  35     ⟨(i.2.val.val,u i.1),⟨fun a ha=>hH i.1 a (hmem i.1 i.2 a ha),i.2.val.property.2⟩⟩
  36   have hrim (i:Σ j,D j):ordinaryWheelRim (W i)=i.2.val.val.val.val:=rfl
  37   have hhub (i:Σ j,D j):ordinaryWheelHub (W i)=u i.1:=rfl
  38   have hdis (i j:Σ k,D k) (hne:i≠j):
  39       Disjoint (ordinaryWheelResources (W i)) (ordinaryWheelResources (W j)):=by
  40     by_cases hij:i.1=j.1
  41     · rcases i with ⟨k,C⟩;rcases j with ⟨l,C'⟩
  42       dsimp only at hij;subst l
  43       have hc:C.val≠C'.val:=by
  44         intro he;exact hne (congrArg (Sigma.mk k) (Subtype.ext he))
  45       have hcc:=hD k C.val C.property C'.val C'.property hc
  46       apply Finset.disjoint_left.mpr
  47       intro e he hf
  48       cases e with
  49       | inl e=>
  50         rw [ordinaryWheelResources_rim] at he hf
  51         induction e using Sym2.ind with
  52         | _ a b=>
  53           exact Finset.disjoint_left.mp hcc (Set.mem_toFinset.mpr he.mem_support_left)
  54             (Set.mem_toFinset.mpr hf.mem_support_left)
  55       | inr az=>
  56         rw [ordinaryWheelResources_spoke] at he hf
  57         exact Finset.disjoint_left.mp hcc (Set.mem_toFinset.mpr he.2) (Set.mem_toFinset.mpr hf.2)
  58     · apply Finset.disjoint_left.mpr
  59       intro e he hf
  60       cases e with
  61       | inl e=>
  62         rw [ordinaryWheelResources_rim] at he hf
  63         induction e using Sym2.ind with
  64         | _ a b=>
  65           have hi:(localOrdinaryFactorGraph (D i.1)).Adj a b:=
  66             (localOrdinaryFactorGraph_adj _ a b).mpr ⟨i.2.val,i.2.property,he⟩
  67           have hj:(localOrdinaryFactorGraph (D j.1)).Adj a b:=
  68             (localOrdinaryFactorGraph_adj _ a b).mpr ⟨j.2.val,j.2.property,hf⟩
  69           exact disjoint_graphs_no_common_edge _ _ (hcross i.1 j.1 hij) a b hi hj
  70       | inr az=>
  71         rw [ordinaryWheelResources_spoke] at he hf
  72         exact hij (u.injective ((hhub i).symm.trans (he.1.trans (hf.1.symm.trans (hhub j)))))
  73   refine ⟨W,fun i=>⟨hrim i,hhub i⟩,?_,?_,hdis,wheelFamily_decomposition W hdis⟩
  74   · ext a b
  75     simp only [wheelFamilyRim,SimpleGraph.iSup_adj,localOrdinaryFactorGraph_adj]
  76     constructor
  77     · rintro ⟨i,hi⟩;exact ⟨i.1,i.2.val,i.2.property,hi⟩
  78     · rintro ⟨j,C,hC,h⟩;exact ⟨⟨j,⟨C,hC⟩⟩,h⟩
  79   · intro a z
  80     constructor
  81     · rintro ⟨i,hi,ha⟩
  82       exact ⟨i.1,(hhub i).symm.trans hi,hmem i.1 i.2 a ha⟩
  83     · rintro ⟨j,hj,ha⟩
  84       rw [←hcover j] at ha
  85       obtain ⟨C,hC,ha⟩:=Finset.mem_biUnion.mp ha
  86       exact ⟨⟨j,⟨C,hC⟩⟩,hj,Set.mem_toFinset.mp ha⟩
