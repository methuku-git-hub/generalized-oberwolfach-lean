  11 theorem finite_refined_exposure_common_event
  12     {Ω Λ:Type} [Fintype Ω] [Fintype Λ] {F:Λ→Type} [∀ a,Fintype (F a)]
  13     (μ:FiniteLaw Ω) (f:Ω→Λ) (ν:∀ a,FiniteLaw (F a))
  14     (E:(Σ a,F a)→Prop) (Extra:(Σ ω,F (f ω))→Prop) (a b:ℝ)
  15     (hE:1-a≤((μ.map f).sigma ν).probability E)
  16     (hExtra:1-b≤(μ.sigma (fun ω=>ν (f ω))).probability Extra):
  17     let law:=μ.sigma (fun ω=>ν (f ω));
  18     1-a-b≤law.probability (fun z=>E ⟨f z.1,z.2⟩ ∧ Extra z) ∧
  19     (a+b<1→∃ z,0<law.mass z ∧ E ⟨f z.1,z.2⟩ ∧ Extra z) := by
  20   intro law
  21   have hp:1-a≤law.probability (fun z=>E ⟨f z.1,z.2⟩):=by
  22     rw [finiteLaw_sigma_projection_probability]
  23     exact hE
  24   have hj:=finiteLaw_good_intersection law (fun z=>E ⟨f z.1,z.2⟩) Extra b hExtra
  25   have hbound:1-a-b≤law.probability (fun z=>E ⟨f z.1,z.2⟩ ∧ Extra z):=by linarith
  26   refine ⟨hbound,?_⟩
  27   intro hab
  28   let Good:=(fun z:Σ ω,F (f ω)=>E ⟨f z.1,z.2⟩ ∧ Extra z)
  29   letI : DecidablePred Good := fun z=>Classical.propDecidable (Good z)
  30   have hpos:0<law.probability Good:=by dsimp [Good];linarith
  31   unfold FiniteLaw.probability at hpos
  32   have hnonneg:∀ z∈Finset.univ,0≤(if Good z then law.mass z else 0):=by
  33     intro z _;split_ifs;exact law.nonneg z;norm_num
  34   obtain ⟨z,_,hz⟩:=(Finset.sum_pos_iff_of_nonneg hnonneg).mp hpos
  35   by_cases hzgood:Good z
  36   · exact ⟨z,by simpa [hzgood] using hz,hzgood⟩
  37   · simp [hzgood] at hz
  38 end Oberwolfach
