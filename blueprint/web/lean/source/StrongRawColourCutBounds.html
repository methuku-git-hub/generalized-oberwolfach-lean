  14 theorem matrixCut_large_of_degree_and_quadratic_strong
  15     {V : Type*} [Fintype V] (G : SimpleGraph V) (M : V → V → ℕ)
  16     {α ρ δ t q : ℝ} (hα : 0 ≤ α) (hαmax : α ≤ 1 / 2) (hρ : 0 ≤ ρ)
  17     (hδa : δ ≤ α / 32) (hδr : δ ≤ ρ / 64)
  18     (ht : t ≤ ρ * Fintype.card V / 32)
  19     (hq : q ≤ α * ρ ^ 2 / 1024)
  20     (hM : ∀ x y, M x y ≤ 1)
  21     (hdegree : ∀ x, (1 / 2 + α) * Fintype.card V ≤ (G.degree x : ℝ))
  22     (hrow : ∀ x, ρ * Fintype.card V / 4 - t ≤ ((∑ y, M x y : ℕ) : ℝ))
  23     (hcol : ∀ y, ρ * Fintype.card V / 4 - t ≤ ((∑ x, M x y : ℕ) : ℝ))
  24     (hcut : ∀ S T, |(matrixCut M S T : ℝ) - ρ / 2 * graphCut G S T| <
  25       q * (Fintype.card V : ℝ) ^ 2)
  26     (S T : Finset V)
  27     (hlarge : (1 - 2 * δ) * Fintype.card V < (S.card : ℝ) + T.card) :
  28     α * ρ / 4 * Fintype.card V * min (S.card : ℝ) T.card ≤ (matrixCut M S T : ℝ) := by
  29   let n : ℝ := Fintype.card V
  30   let s : ℝ := min (S.card : ℝ) T.card
  31   have hn : 0 ≤ n := Nat.cast_nonneg _
  32   have hs : 0 ≤ s := le_min (by positivity) (by positivity)
  33   by_cases hsmall : s ≤ ρ * n / 32
  34   · have hδn := mul_le_mul_of_nonneg_right hδr hn
  35     have hmargin : ρ * n / 8 ≤ ρ * n / 4 - t - ρ * n / 32 - 2 * δ * n := by
  36       change t ≤ ρ * n / 32 at ht
  37       nlinarith [mul_nonneg hρ hn]
  38     have hh := matrixCut_small_side M hM hrow hcol hmargin S T hlarge hsmall
  39     have hcoeff : α * ρ / 4 ≤ ρ / 8 := by nlinarith
  40     have hmul := mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_right hcoeff hn) hs
  41     dsimp [n, s] at hmul
  42     nlinarith
  43   · have hlin : ρ * n / 32 ≤ s := (lt_of_not_ge hsmall).le
  44     have hg := graphCut_large_of_minDegree G hdegree S T hlarge
  45     have hm := mul_le_mul_of_nonneg_left hg (div_nonneg hρ (by norm_num : (0 : ℝ) ≤ 2))
  46     have hc := (abs_lt.mp (hcut S T)).1
  47     have hqn := mul_le_mul_of_nonneg_right hq (sq_nonneg n)
  48     have hsn := mul_le_mul_of_nonneg_left hlin (show 0 ≤ α * ρ * n / 32 by positivity)
  49     have hδn := mul_le_mul_of_nonneg_right hδa (show 0 ≤ ρ / 2 * n * s by positivity)
  50     change (α - δ) * n * s ≤ (graphCut G S T : ℝ) at hg
  51     change ρ / 2 * ((α - δ) * n * s) ≤ ρ / 2 * (graphCut G S T : ℝ) at hm
  52     change -(q * n ^ 2) < (matrixCut M S T : ℝ) - ρ / 2 * graphCut G S T at hc
  53     change α * ρ / 4 * n * s ≤ (matrixCut M S T : ℝ)
  54     nlinarith [mul_nonneg hα (mul_nonneg hρ (mul_nonneg hn hs))]
