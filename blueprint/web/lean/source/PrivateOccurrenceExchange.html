   5 /-! Exchange each old occurrence with one specified occurrence in its
   6 private copy. All private copies have distinct indices, so the exchanges
   7 form an involution even when the old graph has repeated endpoints. -/

  14 def exchangePrivateOccurrence (k:A→W):A⊕((A⊕S)×W)→A⊕((A⊕S)×W)
  15   | .inl a=>.inr (.inl a,k a)
  16   | .inr (.inl a,w)=>if w=k a then .inl a else .inr (.inl a,w)
  17   | .inr (.inr s,w)=>.inr (.inr s,w)

  19 theorem exchangePrivateOccurrence_involutive (k:A→W):Function.Involutive (exchangePrivateOccurrence (S:=S) k):=by
  20   rintro (a|⟨a|s,w⟩)
  21   · simp [exchangePrivateOccurrence]
  22   · by_cases hw:w=k a <;> simp [exchangePrivateOccurrence,hw]
  23   · rfl
