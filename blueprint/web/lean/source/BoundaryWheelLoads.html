  41 theorem boundaryWheel_load_bound (C:I→OrdinaryWheel Q F H r) (x v w:I→V) (K D:ℕ)
  42     (hcap:∀ a,boundaryWheelFreeLoad C x v w a≤K)
  43     (hroot:∀ a,(Finset.univ.filter (fun i=>a∈({x i,v i,w i}:Finset V))).card≤D):
  44     ∀ a,wheelRimLoad C a≤K+D:=by
  45   intro a
  46   have hs:Finset.univ.filter (fun i=>a∈(ordinaryWheelRim (C i)).support)⊆
  47       (Finset.univ.filter (fun i=>a∈rootWheelInternal (C i) (x i) (v i) (w i)))∪
  48       (Finset.univ.filter (fun i=>a∈({x i,v i,w i}:Finset V))):=by
  49     intro i hi
  50     have ha:=(Finset.mem_filter.mp hi).2
  51     by_cases hr:a∈({x i,v i,w i}:Finset V)
  52     · exact Finset.mem_union_right _ (Finset.mem_filter.mpr ⟨Finset.mem_univ _,hr⟩)
  53     · exact Finset.mem_union_left _ (Finset.mem_filter.mpr
  54         ⟨Finset.mem_univ _,Finset.mem_sdiff.mpr ⟨Set.mem_toFinset.mpr ha,hr⟩⟩)
  55   exact ((Finset.card_le_card hs).trans (Finset.card_union_le _ _)).trans
  56     (Nat.add_le_add (hcap a) (hroot a))
