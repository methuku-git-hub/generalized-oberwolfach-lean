  14 theorem rootFirstSpokeType_free (m : ℕ) (hm : 2≤m) (j : Bool) (i : Fin (m+1)) :
  15     rootFirstSpokeType (m+1) j i.succ=decide (j=false ∧ i.val=2) := by
  16   unfold rootFirstSpokeType
  17   cases j
  18   · congr 1
  19     apply propext
  20     simp only [Bool.false_eq_true,ite_false,true_and,Fin.ext_iff,Fin.val_succ,Fin.val_natCast]
  21     change i.val+1=3%(m+2) ↔ i.val=2
  22     rw [Nat.mod_eq_of_lt (show 3<m+2 by omega)]
  23     omega
  24   · simp [Fin.succ_ne_zero]

  37 def survivingPath_to_common_path (hm : 2≤m) :
  38     {C : DenseRootMarkedPathFamily P π (3*d) m j
  39       (rawMarkedPairClass π d σ g j₀ u false) (rawMarkedPairClass π d σ g j₀ u true) v v' //
  40         MarkedPathRims P π (3*d) m j v v' τ C.val} ↪
  41       SampledCommonMarkedPath P π d m j v v' σ τ g j₀ u where
  42   toFun C := by
  43     refine ⟨C.val.val,?_,?_,?_⟩
  44     · exact markedPathRims_rootCycle P π (3*d) m j v v' τ hm C.val.val false C.val.property.1 C.property
  45     · exact markedPathRims_rootCycle P π (3*d) m j v v' τ hm C.val.val true C.val.property.2.1 C.property
  46     · intro i
  47       rw [rootFirstSpokeType_free m hm j i]
  48       have hmem := C.val.property.2.2.1 i
  49       by_cases hc : j=false ∧ i.val=2
  50       · simp only [hc,ite_true,decide_true] at hmem ⊢
  51         have hh := (Finset.mem_filter.mp hmem).2
  52         exact ⟨hh 0,hh 1⟩
  53       · simp only [hc,ite_false,decide_false,Bool.false_eq_true] at hmem ⊢
  54         have hh := (Finset.mem_filter.mp hmem).2
  55         exact ⟨hh 0,hh 1⟩
  56   inj' C D he := by

  58     apply Subtype.ext
