  17 def wheelFamilyRim (W : I → OrdinaryWheel G F H r) : SimpleGraph V := ⨆ i,ordinaryWheelRim (W i)
  18 def wheelFamilySpokes (W : I → OrdinaryWheel G F H r) (v : V) (u : U) : Prop :=
  19   ∃ i,ordinaryWheelHub (W i)=u ∧ v∈(ordinaryWheelRim (W i)).support

  21 theorem wheelFamily_resources (W : I → OrdinaryWheel G F H r) :
  22     ordinaryHostResources (wheelFamilyRim W) (wheelFamilySpokes W)=
  23       Finset.univ.biUnion (fun i => ordinaryWheelResources (W i)) := by
  24   ext e
  25   cases e with
  26   | inl e =>
  27     induction e using Sym2.ind with
  28     | _ v w =>
  29       simp only [ordinaryHostResources_rim,SimpleGraph.mem_edgeSet,wheelFamilyRim,SimpleGraph.iSup_adj,
  30         Finset.mem_biUnion,Finset.mem_univ,true_and,ordinaryWheelResources_rim]
  31   | inr vu =>
  32     rcases vu with ⟨v,u⟩
  33     simp only [ordinaryHostResources_spoke,wheelFamilySpokes,Finset.mem_biUnion,Finset.mem_univ,
  34       true_and,ordinaryWheelResources_spoke]

  36 theorem wheelFamily_subhost (W : I → OrdinaryWheel G F H r) :
  37     wheelFamilyRim W ≤ G ∧ ∀ v u,wheelFamilySpokes W v u → H v u := by
  38   constructor
  39   · exact iSup_le (fun i => (W i).val.1.property)
  40   · intro v u h
  41     obtain ⟨i,hi,hv⟩ := h
  42     rw [← hi]
  43     exact (W i).property.1 v hv

  45 theorem wheelFamily_decomposition (W : I → OrdinaryWheel G F H r)
  46     (hW : ∀ i j,i≠j → Disjoint (ordinaryWheelResources (W i)) (ordinaryWheelResources (W j))) :
  47     HasOrdinaryWheelDecomposition (wheelFamilyRim W) (wheelFamilySpokes W) r := by
  48   let M := Finset.univ.image W
  49   apply ordinary_packing_exact_decomposition M
  50   · intro A hA B hB hAB
  51     obtain ⟨i,_,rfl⟩ := Finset.mem_image.mp hA
  52     obtain ⟨j,_,rfl⟩ := Finset.mem_image.mp hB
  53     exact hW i j (fun h => hAB (congrArg W h))
  54   · rw [wheelFamily_resources]
  55     ext e
  56     simp only [M,Finset.mem_biUnion,Finset.mem_image,Finset.mem_univ,true_and]
  57     constructor
  58     · rintro ⟨A,⟨i,rfl⟩,he⟩;exact ⟨i,he⟩
  59     · rintro ⟨i,hi⟩;exact ⟨W i,⟨i,rfl⟩,hi⟩

  61 theorem wheel_avoids_old_edges (W : OrdinaryWheel G F H r) (R : Finset V) (B : Finset U) (x : V)
  62     (hroot : ∀ v∈(ordinaryWheelRim W).support,v∈R → v=x)
  63     (hhub : ordinaryWheelHub W∉B) (L : SimpleGraph V) (K : V → U → Prop)
  64     (hL : L.support ⊆ (R:Set V)) (hK : ∀ v u,K v u → u∈B) :
  65     Disjoint (ordinaryWheelResources W) (ordinaryHostResources L K) := by
  66   apply Finset.disjoint_left.mpr
  67   intro e he hf
  68   cases e with
  69   | inl e =>
  70     rw [ordinaryWheelResources_rim] at he
  71     rw [ordinaryHostResources_rim] at hf
  72     induction e using Sym2.ind with
  73     | _ v w =>
  74       have hv := hroot v he.mem_support_left (hL hf.mem_support_left)
  75       have hw := hroot w he.mem_support_right (hL hf.mem_support_right)
  76       exact he.ne (hv.trans hw.symm)
  77   | inr vu =>
  78     rcases vu with ⟨v,u⟩
  79     rw [ordinaryWheelResources_spoke] at he
  80     rw [ordinaryHostResources_spoke] at hf
  81     exact hhub (he.1.symm ▸ hK v u hf)

  83 theorem wheelFamily_avoids_old_edges (W : I → OrdinaryWheel G F H r)
  84     (R : Finset V) (B : Finset U) (x : I → V)
  85     (hroot : ∀ i,∀ v∈(ordinaryWheelRim (W i)).support,v∈R → v=x i)
  86     (hhub : ∀ i,ordinaryWheelHub (W i)∉B) (L : SimpleGraph V) (K : V → U → Prop)
  87     (hL : L.support ⊆ (R:Set V)) (hK : ∀ v u,K v u → u∈B) :
  88     Disjoint (ordinaryHostResources (wheelFamilyRim W) (wheelFamilySpokes W)) (ordinaryHostResources L K) := by
  89   rw [wheelFamily_resources]
  90   apply Finset.disjoint_left.mpr
  91   intro e he hf
  92   obtain ⟨i,_,hi⟩ := Finset.mem_biUnion.mp he
  93   exact (Finset.disjoint_left.mp (wheel_avoids_old_edges (W i) R B (x i) (hroot i) (hhub i) L K hL hK)) hi hf

  95 theorem wheelFamily_vertex_bound (W : I → OrdinaryWheel G F H r) :
  96     (wheelFamilyRim W).support.toFinset.card ≤ r*Fintype.card I ∧
  97     (Finset.univ.image (fun i => ordinaryWheelHub (W i))).card ≤ Fintype.card I := by
  98   constructor
  99   · have he : (wheelFamilyRim W).support.toFinset=
 100         Finset.univ.biUnion (fun i => (ordinaryWheelRim (W i)).support.toFinset) := by
 101       ext v
 102       simp only [Set.mem_toFinset,Finset.mem_biUnion,Finset.mem_univ,true_and,SimpleGraph.mem_support,
 103         wheelFamilyRim,SimpleGraph.iSup_adj]
 104       constructor
 105       · rintro ⟨w,i,h⟩;exact ⟨i,w,h⟩
 106       · rintro ⟨i,w,h⟩;exact ⟨w,i,h⟩
 107     rw [he]
 108     calc
 109       _ ≤ ∑ i,(ordinaryWheelRim (W i)).support.toFinset.card := Finset.card_biUnion_le
 110       _ = _ := by
 111         have hc (i : I) : (ordinaryWheelRim (W i)).support.toFinset.card=r := (W i).val.1.val.property.support_card
 112         simp only [hc,Finset.sum_const,Finset.card_univ,smul_eq_mul,Nat.mul_comm]
 113   · exact (Finset.card_image_le).trans_eq (Finset.card_univ)
