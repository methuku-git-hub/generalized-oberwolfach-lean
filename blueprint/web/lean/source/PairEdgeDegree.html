  13 theorem pair_edge_incident_card_le_two (tail head : I → V)
  14     (ht : Function.Injective tail) (hh : Function.Injective head) (v : V) :
  15     ((Finset.univ.image (fun i => ({tail i,head i} : Finset V))).filter (fun e => v∈e)).card≤2 := by
  16   rw [Finset.filter_image]
  17   apply Finset.card_image_le.trans
  18   have he : Finset.univ.filter (fun i => v∈({tail i,head i}:Finset V))=
  19       (Finset.univ.filter (fun i => tail i=v))∪(Finset.univ.filter (fun i => head i=v)) := by
  20     ext i;simp [eq_comm]
  21   rw [he]
  22   apply (Finset.card_union_le _ _).trans
  23   have h₁ : (Finset.univ.filter (fun i => tail i=v)).card≤1 := by
  24     apply Finset.card_le_one.mpr
  25     intro i hi j hj
  26     exact ht ((Finset.mem_filter.mp hi).2.trans (Finset.mem_filter.mp hj).2.symm)
  27   have h₂ : (Finset.univ.filter (fun i => head i=v)).card≤1 := by
  28     apply Finset.card_le_one.mpr
  29     intro i hi j hj
  30     exact hh ((Finset.mem_filter.mp hi).2.trans (Finset.mem_filter.mp hj).2.symm)
  31   omega
