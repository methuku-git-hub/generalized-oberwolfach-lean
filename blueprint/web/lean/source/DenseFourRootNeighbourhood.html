   8 theorem dense_four_root_neighbourhood (P:SimpleGraph V) (α:ℝ) (hα:0<α)
   9     (hP:∀ v,(3/4+α)*Fintype.card V≤(P.degree v:ℝ)):
  10     ∀ S:Finset V,S.card≤4→
  11       4*(min α (1/4))*Fintype.card V≤((common Finset.univ S (fun v=>P.neighborFinset v)).card:ℝ) := by
  12   intro S hS
  13   let ξ:=min α (1/4)
  14   let δ:=3/4+ξ
  15   have hξ1:ξ≤1/4:=min_le_right _ _
  16   have hξα:ξ≤α:=min_le_left _ _
  17   have hδ:δ≤1:=by dsimp [δ];linarith
  18   have hdeg:∀ v,δ*Fintype.card V≤(P.degree v:ℝ):=by
  19     intro v
  20     have hh:=mul_le_mul_of_nonneg_right hξα (Nat.cast_nonneg (Fintype.card V))
  21     dsimp [δ]
  22     nlinarith only [hh,hP v]
  23   have hc:=common_neighbour_card_lower (Finset.univ:Finset V) S
  24     (fun v=>P.neighborFinset v) δ (fun v _=>by
  25       simpa only [Finset.inter_univ,Finset.univ_inter,Finset.card_univ,
  26         SimpleGraph.card_neighborFinset_eq_degree] using hdeg v)
  27   simp only [Finset.card_univ] at hc
  28   have hSreal:(S.card:ℝ)≤4:=by exact_mod_cast hS
  29   have hcoef:1-4*(1-δ)≤1-(S.card:ℝ)*(1-δ):=by
  30     nlinarith only [mul_le_mul_of_nonneg_right hSreal (sub_nonneg.mpr hδ)]
  31   have hm:=mul_le_mul_of_nonneg_right hcoef (Nat.cast_nonneg (Fintype.card V))
  32   have heq:1-4*(1-δ)=4*ξ:=by dsimp [δ];ring
  33   rw [heq] at hm
  34   exact hm.trans hc
  35 end Oberwolfach
