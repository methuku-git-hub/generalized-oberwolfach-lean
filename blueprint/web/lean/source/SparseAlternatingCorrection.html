  15 def FourCycleIncident {H : A → B → Prop} (Q : BipartiteFourCycle H) (e : A×B) : Prop :=
  16   (e.1=Q.val.1.1 ∨ e.1=Q.val.1.2) ∧ (e.2=Q.val.2.1 ∨ e.2=Q.val.2.2)

  26 theorem sparse_alternating_correction_inverse_profile (H : A → B → Prop) (p q : B → ℝ)
  27     (L δ D : ℝ) (h : SparseIntersectionData H p q L δ D)
  28     (ha : 0<Fintype.card A) (hb : 0<Fintype.card B)
  29     (hsmall : δ+D/(Fintype.card A:ℝ)+1/(Fintype.card B:ℝ)≤1/(200*L^6))
  30     (z : A×B → ℝ) (hz : z∈zeroMarginalSpace H) :
  31     ∃ t : BipartiteFourCycle H → ℝ,
  32       (∀ e,z e=∑ Q,t Q*bipartiteFourCycleVector Q e) ∧
  33       (∀ e,(∑ Q,if FourCycleIncident Q e then |t Q| else 0)≤8*L^3*(1+9*L^3)*‖z‖) := by
  34   let S := zeroMarginalSpace H
  35   let T : S →ₗ[ℝ] S :=
  36     { toFun := fun w=>⟨sparseAlternatingMap H q w.val,sparseAlternatingMap_zero_marginals H q w.val⟩
  37       map_add' := by intro w w';apply Subtype.ext;exact (sparseAlternatingMap H q).map_add w.val w'.val
  38       map_smul' := by intro c w;apply Subtype.ext;exact (sparseAlternatingMap H q).map_smul c w.val }
  39   have herror (w : S) : ‖w-T w‖≤9*L^3*‖w‖ ∧ ‖(w-T w)-T (w-T w)‖≤(1/2:ℝ)*‖w‖ :=
  40     sparse_error_contraction H p q L δ D h ha hb hsmall w.val w.property
  41   obtain ⟨w,hw,hnorm⟩ := second_order_inverse T (9*L^3) (1/2) (by norm_num)
  42     (fun w=>(herror w).1) (fun w=>(herror w).2) ⟨z,hz⟩
  43   let t (Q : BipartiteFourCycle H) := sparseRectangleCoefficient H q w.val Q.val.1.1 Q.val.1.2 Q.val.2.1 Q.val.2.2
  44   refine ⟨t,?_,?_⟩
  45   · intro e
  46     have he : sparseAlternatingMap H q w.val e=z e := congrFun (congrArg Subtype.val hw) e
  47     rw [← he]
  48     exact rectangleCombination_fourCycles H (sparseRectangleCoefficient H q w.val)
  49       (sparseRectangleCoefficient_supported H q w.val) e
  50   · intro e
  51     have hload := sparseAlternating_coefficient_load H q w.val L h.L_nonneg h.inverse_bounds ha hb e
  52     have he : (∑ Q,if FourCycleIncident Q e then |t Q| else 0)=
  53         (∑ Q,|t Q| * |bipartiteFourCycleVector Q e|) := by
  54       simp only [bipartiteFourCycleVector_abs,mul_ite,mul_one,mul_zero]
  55     rw [he]
  56     calc
  57       _ ≤ 4*L^3*‖w.val‖ := hload
  58       _ ≤ 4*L^3*(((1+9*L^3)/(1-(1/2:ℝ)))*‖z‖) :=
  59         mul_le_mul_of_nonneg_left hnorm (by have := h.L_nonneg;positivity)
  60       _ = _ := by ring
