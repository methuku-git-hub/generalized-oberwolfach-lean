  11 theorem prescribed_forbidden_graph_from_partial_images
  12     (X:I→Type) [∀ i,Fintype (X i)] (Q:∀ i,SimpleGraph (X i))
  13     (φ:∀ i,X i↪V) (H:I→SimpleGraph V) (old fresh:I→V→Prop)
  14     (hsource:∀ i,H i≤(Q i).map (φ i)) (hsourceDegree:∀ i x,(Q i).degree x≤2)
  15     (hocc:∀ i v,(∃ x,φ i x=v)→ old i v ∨ fresh i v)
  16     (J:SimpleGraph V) (δ n t:ℝ)
  17     (hold:∀ v,((Finset.univ.filter (fun i=>old i v)).card:ℝ)≤δ*n)
  18     (hfresh:∀ v,((Finset.univ.filter (fun i=>fresh i v)).card:ℝ)≤t+1)
  19     (hJ:∀ v,(J.degree v:ℝ)≤δ*n):
  20     let G:SimpleGraph V:=⨆ i,H i;
  21     let F:=G⊔J;
  22     letI : DecidableRel F.Adj := fun _ _=>Classical.propDecidable _;
  23     (∀ v,(G.degree v:ℝ)≤2*δ*n+2*t+2 ∧ (F.degree v:ℝ)≤3*δ*n+2*t+2) ∧
  24     (∀ i,(H i).edgeFinset⊆F.edgeFinset) ∧ J.edgeFinset⊆F.edgeFinset := by
  25   intro G F
  26   letI : DecidableRel F.Adj := fun _ _=>Classical.propDecidable _
  27   have hcover:∀ v,G.neighborFinset v⊆Finset.univ.biUnion (fun i=>(H i).neighborFinset v):=by
  28     intro v w hw
  29     simp only [SimpleGraph.mem_neighborFinset] at hw
  30     change (⨆ i,H i).Adj v w at hw
  31     obtain ⟨i,hi⟩:=SimpleGraph.iSup_adj.mp hw
  32     simp only [Finset.mem_biUnion,Finset.mem_univ,true_and,SimpleGraph.mem_neighborFinset]
  33     exact ⟨i,hi⟩
  34   have hforbid:∀ v,F.neighborFinset v⊆G.neighborFinset v∪J.neighborFinset v:=by
  35     intro v w hw
  36     simp only [SimpleGraph.mem_neighborFinset] at hw
  37     change (G⊔J).Adj v w at hw
  38     simp only [Finset.mem_union,SimpleGraph.mem_neighborFinset,SimpleGraph.sup_adj]
  39     exact (SimpleGraph.sup_adj G J v w).mp hw
  40   have hbound:=prescribed_embedding_used_degree_from_partial_images X Q φ H old fresh hsource hsourceDegree hocc G J F hcover hforbid δ n t hold hfresh hJ
  41   refine ⟨hbound,?_,?_⟩
  42   · intro i
  43     apply SimpleGraph.edgeFinset_mono
  44     intro v w hw
  45     change (G⊔J).Adj v w
  46     apply (SimpleGraph.sup_adj G J v w).mpr
  47     left
  48     change (⨆ i,H i).Adj v w
  49     exact SimpleGraph.iSup_adj.mpr ⟨i,hw⟩
  50   · apply SimpleGraph.edgeFinset_mono
  51     intro v w hw
  52     change (G⊔J).Adj v w
  53     exact (SimpleGraph.sup_adj G J v w).mpr (Or.inr hw)
