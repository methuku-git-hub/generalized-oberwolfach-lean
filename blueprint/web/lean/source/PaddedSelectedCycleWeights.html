  11 theorem padded_selected_cycle_weights {V:Type} [Fintype V] [DecidableEq V]
  12     (G:SimpleGraph V) (K:ℕ) (n:Fin K→ℕ) (B:Fin K→ℝ) (hB:∀ i,0≤B i)
  13     (S:Finset (Fin K))
  14     (w:∀ i:↥S,OrdinaryCycle G (n i.val+2)→ℝ)
  15     (hw:∀ i C,0≤w i C ∧ w i C≤B i.val/(Fintype.card V:ℝ)^(n i.val))
  16     (he:∀ i x y,(∑ C,if C.val.val.Adj x y then w i C else 0)=if G.Adj x y then 1 else 0):
  17     ∃ wfull:∀ i:Fin K,OrdinaryCycle G (n i+2)→ℝ,
  18       (∀ i C,0≤wfull i C ∧ wfull i C≤B i/(Fintype.card V:ℝ)^(n i)) ∧
  19       (∀ i:↥S,∀ C,wfull i.val C=w i C) ∧
  20       (∀ i:↥S,∀ x y,(∑ C,if C.val.val.Adj x y then wfull i.val C else 0)=if G.Adj x y then 1 else 0) ∧
  21       (∀ i∉S,∀ C,wfull i C=0):=by
  22   let X:=fun i:Fin K=>OrdinaryCycle G (n i+2)
  23   let wfull:=padIndexedFunctions S X w
  24   refine ⟨wfull,?_,?_,?_,?_⟩
  25   · apply pad_indexed_functions_bounds S X w (fun i=>B i/(Fintype.card V:ℝ)^(n i))
  26     · intro i;have hi:=hB i;positivity
  27     · exact hw
  28   · intro i C
  29     exact pad_indexed_functions_on S X w i C
  30   · intro i x y
  31     have hwEq:∀ C,wfull i.val C=w i C:=pad_indexed_functions_on S X w i
  32     simp_rw [hwEq]
  33     exact he i x y
  34   · intro i hi C
  35     exact pad_indexed_functions_off S X w i hi C
  36 end Oberwolfach
