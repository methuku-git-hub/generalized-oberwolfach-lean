  11 theorem permutation_neighbour_filter_image
  12     {V : Type*} [Fintype V] [DecidableEq V] (P : SimpleGraph V) (π : Equiv.Perm V)
  13     (T : Finset V) (x : V) :
  14     (T.filter (fun v=>P.Adj x (π v))).image π=P.neighborFinset x∩T.image π := by
  15   ext z
  16   simp only [Finset.mem_image,Finset.mem_filter,Finset.mem_inter,SimpleGraph.mem_neighborFinset]
  17   constructor
  18   · rintro ⟨v,⟨hv,hadj⟩,heq⟩
  19     exact ⟨heq ▸ hadj,⟨v,hv,heq⟩⟩
  20   · rintro ⟨hadj,v,hv,heq⟩
  21     exact ⟨v,⟨hv,heq.symm ▸ hadj⟩,heq⟩

  29 theorem permutation_neighbour_domain_card
  30     {V : Type*} [Fintype V] [DecidableEq V] (P : SimpleGraph V) (π : Equiv.Perm V) (x : V) :
  31     (Finset.univ.filter (fun v=>P.Adj x (π v))).card=P.degree x := by
  32   rw [permutation_neighbour_filter_card]
