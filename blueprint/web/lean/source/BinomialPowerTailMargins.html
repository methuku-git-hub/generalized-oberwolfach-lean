  13 theorem binomial_tail_le_geometric (M k : ℕ) (p a : ℝ)
  14     (hp : 0≤p) (hp1 : p≤1) (ha : 0≤a) (hk : a*M≤(k:ℝ)) :
  15     (M.choose k:ℝ)*p^k≤(2*p^a)^M := by
  16   have hc : (M.choose k:ℝ)≤(2:ℝ)^M := by exact_mod_cast Nat.choose_le_two_pow M k
  17   have he : p^k≤(p^a)^M := by
  18     rw [← Real.rpow_natCast,← Real.rpow_natCast,← Real.rpow_mul hp]
  19     exact Real.rpow_le_rpow_of_exponent_ge' hp hp1 (by positivity) hk
  20   calc
  21     _ ≤ (2:ℝ)^M*((p^a)^M) := mul_le_mul hc he (by positivity) (by positivity)
  22     _ = _ := (mul_pow _ _ _).symm

  24 theorem eventually_binomial_deletion_tail (K a b : ℝ) (hK : 0≤K) (ha : 0<a) (hb : 0<b) :
  25     ∃ N : ℕ,1≤N ∧ ∀ n : ℕ,N≤n → ∀ M k : ℕ,∀ p : ℝ,
  26       0≤p → p≤K*(n:ℝ)^(-b) → a*M≤(k:ℝ) →
  27       (M.choose k:ℝ)*p^k≤(n:ℝ)^(-(a*b/2)*M) := by
  28   let c := a*b/2
  29   have hc : 0<c := by dsimp [c];positivity
  30   have hlim₁ := ((tendsto_rpow_neg_atTop hb).const_mul K).comp tendsto_natCast_atTop_atTop
  31   have hlim₂ := ((tendsto_rpow_neg_atTop hc).const_mul (2*K^a)).comp tendsto_natCast_atTop_atTop
  32   simp only [mul_zero] at hlim₁ hlim₂
  33   obtain ⟨N₁,hN₁⟩ := eventually_atTop.mp (hlim₁.eventually (eventually_lt_nhds (by norm_num : (0:ℝ)<1)))
  34   obtain ⟨N₂,hN₂⟩ := eventually_atTop.mp (hlim₂.eventually (eventually_lt_nhds (by norm_num : (0:ℝ)<1)))
  35   refine ⟨max 1 (max N₁ N₂),le_max_left _ _,?_⟩
  36   intro n hn M k p hp hrate hk
  37   have hnpos : (0:ℝ)<n := by exact_mod_cast (show 0<n by omega)
  38   have hsmall₁ : K*(n:ℝ)^(-b)≤1 := (hN₁ n (by omega)).le
  39   have hsmall₂ : (2*K^a)*(n:ℝ)^(-c)≤1 := (hN₂ n (by omega)).le
  40   have hbase : 2*p^a≤(n:ℝ)^(-c) := by
  41     have heq : 2*(K*(n:ℝ)^(-b))^a=((2*K^a)*(n:ℝ)^(-c))*(n:ℝ)^(-c) := by
  42       rw [Real.mul_rpow hK (by positivity),← Real.rpow_mul hnpos.le]
  43       have he : -b*a = -c + -c := by dsimp [c];ring
  44       rw [he,Real.rpow_add hnpos]
  45       ring
  46     calc
  47       _ ≤ 2*(K*(n:ℝ)^(-b))^a := mul_le_mul_of_nonneg_left (Real.rpow_le_rpow hp hrate ha.le) (by norm_num)
  48       _ = _ := heq
  49       _ ≤ 1*(n:ℝ)^(-c) := mul_le_mul_of_nonneg_right hsmall₂ (by positivity)
  50       _ = _ := one_mul _
  51   calc
  52     _ ≤ (2*p^a)^M := binomial_tail_le_geometric M k p a hp (hrate.trans hsmall₁) ha.le hk
  53     _ ≤ ((n:ℝ)^(-c))^M := pow_le_pow_left₀ (by positivity) hbase M
  54     _ = (n:ℝ)^(-(a*b/2)*M) := by rw [← Real.rpow_natCast,← Real.rpow_mul hnpos.le]
