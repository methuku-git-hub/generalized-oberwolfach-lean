  13 theorem marked_inner_free_packing_graphs
  14     (M:Finset (MarkedWheel tail head ktail khead m π b H₀ HK)) (I:Finset V) (J:Finset U)
  15     (hinner:(M.biUnion markedWheelPhysicalResources).filter (markedInnerResource I J)=∅):
  16     (∀ x∈I,∀ y∈I,¬(markedPackingRimGraph M).Adj x y) ∧
  17     (∀ x∈I,∀ u∈J,¬(markedFiniteSpokeGraph (M.biUnion markedWheelSpokePairs)).Adj (Sum.inl x) (Sum.inr u)) :=by
  18   have hno (e:Finset V⊕(V×U)) (he:e∈M.biUnion markedWheelPhysicalResources)
  19       (hi:markedInnerResource I J e):False:=by
  20     have hh:e∈(M.biUnion markedWheelPhysicalResources).filter (markedInnerResource I J):=
  21       Finset.mem_filter.mpr ⟨he,hi⟩
  22     rw [hinner] at hh
  23     exact Finset.notMem_empty e hh
  24   constructor
  25   · intro x hx y hy hadj
  26     have hed:{x,y}∈markedPackingRims M:=by
  27       have hh:=hadj
  28       simp only [markedPackingRimGraph,physicalEdgeGraph] at hh
  29       exact hh.2
  30     obtain ⟨C,hC,he⟩:=Finset.mem_biUnion.mp hed
  31     have hres:Sum.inl ({x,y}:Finset V)∈M.biUnion markedWheelPhysicalResources:=
  32       Finset.mem_biUnion.mpr ⟨C,hC,(markedWheelPhysicalResources_rim C _).mpr he⟩
  33     have hi:markedInnerResource I J (Sum.inl ({x,y}:Finset V)):=by
  34       change ({x,y}:Finset V)⊆I
  35       intro z hz
  36       simp only [Finset.mem_insert,Finset.mem_singleton] at hz
  37       rcases hz with rfl|rfl <;> assumption
  38     exact hno _ hres hi
  39   · intro x hx u hu hadj
  40     obtain ⟨C,hC,hCu,hCx⟩:=(markedFiniteSpokeGraph_packing_adj M x u).mp hadj
  41     have hres:Sum.inr (x,u)∈M.biUnion markedWheelPhysicalResources:=
  42       Finset.mem_biUnion.mpr ⟨C,hC,(markedWheelPhysicalResources_spoke C x u).mpr
  43         ⟨hCu,by simpa only [markedWheelRows] using hCx⟩⟩
  44     have hi:markedInnerResource I J (Sum.inr (x,u)):=⟨hx,hu⟩
  45     exact hno _ hres hi
