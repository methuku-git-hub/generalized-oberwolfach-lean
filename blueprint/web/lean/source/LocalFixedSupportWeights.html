  26     (S Y : Finset V) (hS : S.card ≤ c) :
  27     (∑ C₀,if S⊆R C₀ then inducedResourceWeight R w Y C₀ else 0) ≤
  28       (J:ℝ)*(Y.card:ℝ)^(c-S.card)*B := by
  29   have hcount : (Fintype.card {C₀ : C // S⊆R C₀ ∧ R C₀⊆Y}:ℝ) ≤
  30       J*(Y.card:ℝ)^(c-S.card) := by exact_mod_cast uniform_resources_local_fixed_set_count R c J hR hJ S Y hS

  32     _ ≤ ∑ C₀,if S⊆R C₀ ∧ R C₀⊆Y then B else 0 := by
