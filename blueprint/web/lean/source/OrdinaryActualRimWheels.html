  15  structure OrdinaryActualRimWheels (G Q F:SimpleGraph V) (H₀ H:V→U→Prop)
  16     (φ:V→V) (r:ℕ) (B:Finset U)
  17     (b:∀ v,G.neighborSet v≃({u:U // H₀ v u}×Fin 2))
  18     (σ:∀ v,Equiv.Perm (G.neighborSet v))
  19     (p:(G.edgeFinset⊕{e:V×U // H₀ e.1 e.2})→(Σ a:V,Σ c:V,Q.Walk a c)) where
  20   hub : G.edgeFinset↪U
  21   first : G.edgeFinset→OrdinaryWheel Q F H r
  22   second : G.edgeFinset→OrdinaryWheel Q F H r
  23   hub_avoid : ∀ e,hub e∉B
  24   attachments : ∀ e,∀ v∈transformerHubAttachments G σ
  25     (transformerRimEndpoint G Q H₀ p) (transformerSpokeEndpoint G Q H₀ b p) e,H v (hub e)
  26   first_hub : ∀ e,ordinaryWheelHub (first e)=hub e
  27   second_hub : ∀ e,ordinaryWheelHub (second e)=hub e
  28   first_support : ∀ e v,v∈(ordinaryWheelRim (first e)).support↔
  29     v=e.val.out.1∨v=e.val.out.2∨v∈(p (.inl e)).2.2.support
  30   second_support : ∀ e v,v∈(ordinaryWheelRim (second e)).support↔
  31     v=φ e.val.out.1∨v=φ e.val.out.2∨v∈(p (.inl e)).2.2.support
  32   first_disjoint : ∀ e f,e≠f→Disjoint (ordinaryWheelResources (first e)) (ordinaryWheelResources (first f))
  33   second_disjoint : ∀ e f,e≠f→Disjoint (ordinaryWheelResources (second e)) (ordinaryWheelResources (second f))
  34   rim_difference : ∀ i e,
  35     (if Sum.inl e∈ordinaryWheelResources (first i) then (1:ℤ) else 0)-
  36       (if Sum.inl e∈ordinaryWheelResources (second i) then 1 else 0)=
  37     (if e=s(i.val.out.1,i.val.out.2) then 1 else 0)+
  38       (if e=s(i.val.out.1,(p (.inl i)).1) then 1 else 0)+
  39       (if e=s(i.val.out.2,(p (.inl i)).2.1) then 1 else 0)-
  40       (if e=s(φ i.val.out.1,φ i.val.out.2) then 1 else 0)-
  41       (if e=s(φ i.val.out.1,(p (.inl i)).1) then 1 else 0)-
  42       (if e=s(φ i.val.out.2,(p (.inl i)).2.1) then 1 else 0)
  43   spoke_difference : ∀ i v z,
  44     (if Sum.inr (v,z)∈ordinaryWheelResources (first i) then (1:ℤ) else 0)-
  45       (if Sum.inr (v,z)∈ordinaryWheelResources (second i) then 1 else 0)=
  46     (if v=i.val.out.1∧z=hub i then 1 else 0)+(if v=i.val.out.2∧z=hub i then 1 else 0)-
  47       (if v=φ i.val.out.1∧z=hub i then 1 else 0)-(if v=φ i.val.out.2∧z=hub i then 1 else 0)

  49  theorem ordinary_actual_rim_wheels (G Q F:SimpleGraph V) (H₀ H:V→U→Prop) (φ:V→V)
  50     (b:∀ v,G.neighborSet v≃({u:U // H₀ v u}×Fin 2))
  51     (σ:∀ v,Equiv.Perm (G.neighborSet v))
  52     (p:(G.edgeFinset⊕{e:V×U // H₀ e.1 e.2})→(Σ a:V,Σ c:V,Q.Walk a c))
  53     (r:ℕ) (hr:3≤r) (R:Finset V) (B:Finset U)
  54     (hroot:∀ v∈G.support,v∈R∧φ v∈R)
  55     (hQ:G≤Q) (hφQ:∀ v w,G.Adj v w→Q.Adj (φ v) (φ w))
  56     (hcross:∀ v∈G.support,∀ w∈G.support,φ v≠w)
  57     (hφedge:Function.Injective (fun e:G.edgeFinset=>s(φ e.val.out.1,φ e.val.out.2)))
  58     (hpath:∀ e:G.edgeFinset,(p (.inl e)).2.2.IsPath∧(p (.inl e)).2.2.length=r-3)
  59     (hends:∀ e:G.edgeFinset,Q.Adj e.val.out.1 (p (.inl e)).1 ∧
  60       Q.Adj (φ e.val.out.1) (p (.inl e)).1 ∧ Q.Adj e.val.out.2 (p (.inl e)).2.1 ∧
  61       Q.Adj (φ e.val.out.2) (p (.inl e)).2.1)
  62     (havoid:∀ i,∀ v∈(p i).2.2.support,v∉R)
  63     (hdis:∀ i j,i≠j→Disjoint (p i).2.2.support.toFinset (p j).2.2.support.toFinset)
  64     (hFF:∀ i j,∀ v∈(p i).2.2.support,∀ w∈(p j).2.2.support,¬F.Adj v w)
  65     (hFR:∀ i,∀ v∈(p i).2.2.support,∀ w∈R,¬F.Adj v w)
  66     (hFroots:∀ e:G.edgeFinset,∀ v∈({e.val.out.1,e.val.out.2,φ e.val.out.1,φ e.val.out.2}:Finset V),
  67       ∀ w∈({e.val.out.1,e.val.out.2,φ e.val.out.1,φ e.val.out.2}:Finset V),¬F.Adj v w)
  68     (t:ℝ) (hhubs:∀ S:Finset V,S.card≤r+6→(∀ v∈S,∀ w∈S,¬F.Adj v w)→
  69       t≤((common (Finset.univ:Finset U) S (fun v=>Finset.univ.filter (H v))).card:ℝ))
  70     (hbudget:B.card+G.edgeFinset.card<t):
  71     Nonempty (OrdinaryActualRimWheels G Q F H₀ H φ r B b σ p):=by
  72   let a:=transformerRimEndpoint G Q H₀ p
  73   let z:=transformerSpokeEndpoint G Q H₀ b p
  74   let T:=transformerHubAttachments G σ a z
  75   let P:Finset V:=Finset.univ.biUnion (fun i=>(p i).2.2.support.toFinset)
  76   have hm (i) (v) (hv:v∈(p i).2.2.support):v∈P:=
  77     Finset.mem_biUnion.mpr ⟨i,Finset.mem_univ _,List.mem_toFinset.mpr hv⟩
  78   have ha (d:Σ v,G.neighborSet v):a d∈P:=hm _ _ (graphPathEndpoint_mem _ _)
  79   have hz (d:Σ v,G.neighborSet v):z d∈P:=hm _ _ (graphPathEndpoint_mem _ _)
  80   have hTP (e:G.edgeFinset):T e⊆P:=by
  81     intro v hv
  82     obtain ⟨i,_,hi⟩:=Finset.mem_biUnion.mp hv
  83     rcases Finset.mem_insert.mp hi with he|he
  84     · exact he ▸ ha _
  85     · exact (Finset.mem_singleton.mp he) ▸ hz _
  86   have hPF:∀ v∈P,∀ w∈P,¬F.Adj v w:=by
  87     intro v hv w hw
  88     obtain ⟨i,_,hi⟩:=Finset.mem_biUnion.mp hv
  89     obtain ⟨j,_,hj⟩:=Finset.mem_biUnion.mp hw
  90     exact hFF i j v (List.mem_toFinset.mp hi) w (List.mem_toFinset.mp hj)
  91   have hPR:∀ v∈P,∀ w∈R,¬F.Adj v w:=by
  92     intro v hv w hw
  93     obtain ⟨i,_,hi⟩:=Finset.mem_biUnion.mp hv
  94     exact hFR i v (List.mem_toFinset.mp hi) w hw
  95   have hRoots (e:G.edgeFinset):e.val.out.1∈R∧e.val.out.2∈R∧φ e.val.out.1∈R∧φ e.val.out.2∈R:=by
  96     have he:=edge_out_adj G e
  97     exact ⟨(hroot _ he.mem_support_left).1,(hroot _ he.mem_support_right).1,
  98       (hroot _ he.mem_support_left).2,(hroot _ he.mem_support_right).2⟩
  99   have hS (e:G.edgeFinset):∀ v∈({e.val.out.1,e.val.out.2,φ e.val.out.1,φ e.val.out.2}∪T e∪
 100       (p (.inl e)).2.2.support.toFinset:Finset V),
 101       ∀ w∈({e.val.out.1,e.val.out.2,φ e.val.out.1,φ e.val.out.2}∪T e∪
 102       (p (.inl e)).2.2.support.toFinset:Finset V),¬F.Adj v w:=by
 103     let Z:Finset V:={e.val.out.1,e.val.out.2,φ e.val.out.1,φ e.val.out.2}
 104     have hZR:Z⊆R:=by
 105       intro v hv
 106       simp only [Z,Finset.mem_insert,Finset.mem_singleton] at hv
 107       rcases hv with rfl|rfl|rfl|rfl
 108       · exact (hRoots e).1
 109       · exact (hRoots e).2.1
 110       · exact (hRoots e).2.2.1
 111       · exact (hRoots e).2.2.2
 112     have hsplit:∀ v∈(Z∪T e∪(p (.inl e)).2.2.support.toFinset),v∈Z∨v∈P:=by
 113       intro v hv
 114       rcases Finset.mem_union.mp hv with hv|hv
 115       · rcases Finset.mem_union.mp hv with hv|hv
 116         · exact Or.inl hv
 117         · exact Or.inr (hTP e hv)
 118       · exact Or.inr (hm _ _ (List.mem_toFinset.mp hv))
 119     intro v hv w hw
 120     rcases hsplit v hv with hv|hv <;> rcases hsplit w hw with hw|hw
 121     · exact hFroots e v hv w hw
 122     · exact fun hh=>hPR w hw v (hZR hv) hh.symm
 123     · exact hPR v hv w (hZR hw)
 124     · exact hPF v hv w hw
 125   have hinj:Function.Injective (fun e:G.edgeFinset=>s(e.val.out.1,e.val.out.2)):=by
 126     intro e f he
 127     apply Subtype.ext
 128     have hee:s(e.val.out.1,e.val.out.2)=e.val:=by rw [Sym2.mk];exact e.val.out_eq
 129     have hff:s(f.val.out.1,f.val.out.2)=f.val:=by rw [Sym2.mk];exact f.val.out_eq
 130     exact hee.symm.trans (he.trans hff)
 131   have hcross' (e:G.edgeFinset):φ e.val.out.1≠e.val.out.1∧φ e.val.out.1≠e.val.out.2∧
 132       φ e.val.out.2≠e.val.out.1∧φ e.val.out.2≠e.val.out.2:=by
 133     have he:=edge_out_adj G e
 134     exact ⟨hcross _ he.mem_support_left _ he.mem_support_left,
 135       hcross _ he.mem_support_left _ he.mem_support_right,
 136       hcross _ he.mem_support_right _ he.mem_support_left,
 137       hcross _ he.mem_support_right _ he.mem_support_right⟩
 138   obtain ⟨h,W,W',hT,hh,hs,hs',hd,hd',hrim,hspoke⟩:=ordinary_rim_wheel_families Q F H r hr R B T
 139     (fun e:G.edgeFinset=>e.val.out.1) (fun e=>e.val.out.2) (fun e=>φ e.val.out.1) (fun e=>φ e.val.out.2)
 140     (fun e=>p (.inl e)) hRoots (fun e=>hQ (edge_out_adj G e)) (fun e=>hφQ _ _ (edge_out_adj G e))
 141     hcross' hpath hends (fun e=>havoid (.inl e))
 142     (fun e f he=>hdis (.inl e) (.inl f) (fun hh=>he (Sum.inl.inj hh)))
 143     (fun e=>transformerHubAttachments_card G σ a z e) hS hinj hφedge t hhubs
 144     (by simpa only [Fintype.card_coe] using hbudget)
 145   exact ⟨⟨h,W,W',fun e=>(hT e).1,fun e=>(hT e).2,fun e=>(hh e).1,fun e=>(hh e).2,
 146     hs,hs',hd,hd',hrim,hspoke⟩⟩

 148  theorem OrdinaryActualRimWheels.root_hub_adjacency
 149     {G Q F:SimpleGraph V} {H₀ H:V→U→Prop} {φ:V→V} {r:ℕ} {B:Finset U}
 150     {b:∀ v,G.neighborSet v≃({u:U // H₀ v u}×Fin 2)} {σ:∀ v,Equiv.Perm (G.neighborSet v)}
 151     {p:(G.edgeFinset⊕{e:V×U // H₀ e.1 e.2})→(Σ a:V,Σ c:V,Q.Walk a c)}
 152     (E:OrdinaryActualRimWheels G Q F H₀ H φ r B b σ p):
 153     ∀ e,H e.val.out.1 (E.hub e)∧H (φ e.val.out.1) (E.hub e)∧
 154       H e.val.out.2 (E.hub e)∧H (φ e.val.out.2) (E.hub e):=by
 155   intro e
 156   have h1:∀ v∈(ordinaryWheelRim (E.first e)).support,H v (E.hub e):=by
 157     intro v hv
 158     have hh:H v (ordinaryWheelHub (E.first e)):=(E.first e).property.1 v hv
 159     rw [E.first_hub] at hh;exact hh
 160   have h2:∀ v∈(ordinaryWheelRim (E.second e)).support,H v (E.hub e):=by
 161     intro v hv
 162     have hh:H v (ordinaryWheelHub (E.second e)):=(E.second e).property.1 v hv
 163     rw [E.second_hub] at hh;exact hh
 164   exact ⟨h1 _ ((E.first_support e _).mpr (Or.inl rfl)),h2 _ ((E.second_support e _).mpr (Or.inl rfl)),
 165     h1 _ ((E.first_support e _).mpr (Or.inr (Or.inl rfl))),h2 _ ((E.second_support e _).mpr (Or.inr (Or.inl rfl)))⟩
