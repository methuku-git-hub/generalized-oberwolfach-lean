  16 theorem colorClassLottery_expectation (part : E → I) (color : E → C) (w : E → ℝ) :
  17     (FiniteLaw.pi (fun _ : I=>FiniteLaw.uniform C)).expectation
  18       (fun σ=>∑ i,colorClassWeight part color w i (σ i))=(∑ e,w e)/Fintype.card C := by
  19   let : MeasurableSpace C := ⊤
  20   let : MeasurableSingletonClass C := ⟨fun _=>trivial⟩
  21   calc
  22     _ = ∑ i,(FiniteLaw.pi (fun _ : I=>FiniteLaw.uniform C)).expectation
  23         (fun σ=>colorClassWeight part color w i (σ i)) := by
  24       unfold FiniteLaw.expectation
  25       simp_rw [Finset.mul_sum]
  26       rw [Finset.sum_comm]
  27     _ = ∑ i,(FiniteLaw.uniform C).expectation (colorClassWeight part color w i) := by
  28       apply Finset.sum_congr rfl
  29       intro i _
  30       exact finiteLaw_coordinate_expectation _ i _
  31     _ = ∑ i,(∑ c,colorClassWeight part color w i c)/Fintype.card C := by
  32       simp_rw [finiteLaw_uniform_expectation]
  33     _ = _ := by rw [← Finset.sum_div,colorClassWeight_total]

  35 theorem colorClassLottery_matching
  36     (R : E → Finset V) (part : E → I) (color : E → C)
  37     (hparts : ∀ e f,part e≠part f → Disjoint (R e) (R f))
  38     (hcolor : ∀ e f,e≠f → part e=part f → color e=color f → Disjoint (R e) (R f))
  39     (T : Finset J) (w : J → E → ℝ) (B : J → I → ℝ) (t : J → ℝ)
  40     (hw : ∀ j∈T,∀ e,0≤w j e)
  41     (hB : ∀ j∈T,∀ i c,colorClassWeight part color (w j) i c≤B j i)
  42     (ht : ∀ j∈T,0≤t j)
  43     (hsmall : (∑ j∈T,2*Real.exp (-2*(t j)^2/(∑ i,(B j i)^2)))<1) :
  44     ∃ M : Finset E,
  45       (∀ e∈M,∀ f∈M,e≠f → Disjoint (R e) (R f)) ∧
  46       (∀ j∈T,|(∑ e∈M,w j e)-(∑ e,w j e)/Fintype.card C|<t j) := by
  47   let μ := FiniteLaw.pi (fun _ : I=>FiniteLaw.uniform C)
  48   let X (j : J) (σ : I → C) := ∑ i,colorClassWeight part color (w j) i (σ i)
  49   have hnonneg (j : J) (hj : j∈T) (i : I) : 0≤B j i :=
  50     (colorClassWeight_nonneg part color (w j) (hw j hj) i (Classical.choice (inferInstance : Nonempty C))).trans
  51       (hB j hj i _)
  52   have htail (j : J) (hj : j∈T) : μ.probability (fun σ=>t j≤|X j σ-μ.expectation (X j)|)≤
  53       2*Real.exp (-2*(t j)^2/(∑ i,(B j i)^2)) := by
  54     exact finiteLaw_boundedDifferences (fun _ : I=>FiniteLaw.uniform C) (X j) (B j)
  55       (hnonneg j hj)
  56       (colorClassWeight_coordinate_change part color (w j) (hw j hj) (B j) (hB j hj)) (ht j hj)
  57   obtain ⟨σ,_hσ,hgood⟩ := finiteLaw_common_event_exists μ T
  58     (fun j σ=>t j≤|X j σ-μ.expectation (X j)|)
  59     (fun j=>2*Real.exp (-2*(t j)^2/(∑ i,(B j i)^2))) htail hsmall
  60   refine ⟨colorClassSelection part color σ,colorClassSelection_matching R part color hparts hcolor σ,?_⟩
  61   intro j hj
  62   have hh := lt_of_not_ge (hgood j hj)
  63   change |(∑ i,colorClassWeight part color (w j) i (σ i))-
  64     (FiniteLaw.pi (fun _ : I=>FiniteLaw.uniform C)).expectation
  65       (fun σ=>∑ i,colorClassWeight part color (w j) i (σ i))|<t j at hh
  66   rw [colorClassLottery_expectation] at hh
  67   rwa [colorClassSelection_weight]
