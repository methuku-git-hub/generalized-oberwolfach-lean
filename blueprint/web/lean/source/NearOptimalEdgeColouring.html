  13 theorem near_optimal_hypergraph_edge_colouring
  14     (k : ℕ) (hk : 0<k) (ε δ : ℝ) (hε : 0<ε) (hε1 : ε≤1/2)
  15     (hδ : 0<δ) (hδ1 : δ≤1) :
  16     ∃ D₀ : ℕ, 1≤D₀ ∧ ∀ D B : ℕ, D₀≤D → (B : ℝ)≤(D : ℝ)^(1-δ) →
  17       ∀ (V E : Type) [Fintype V] [Fintype E],
  18       ∀ (R : E → Finset V),
  19       (∀ e,(R e).card=k) →
  20       (∀ v,(hypergraphVertexClique R v).card≤D) →
  21       (∀ v w,v≠w → (Finset.univ.filter (fun e=>v∈R e ∧ w∈R e)).card≤B) →
  22       ∃ f : E → Fin ⌈(1+ε)*(D : ℝ)⌉₊,
  23         ∀ e g,(hypergraphLineGraph R).Adj e g → f e≠f g := by
  24   classical
  25   obtain ⟨N,hN⟩ := near_optimal_hypergraph_list_colouring k hk ε δ hε (by linarith) hδ hδ1
  26   refine ⟨max 2 N,by omega,?_⟩
  27   intro D B hD hB V E _ _ R hsize hdegree hcodegree
  28   have hD2 : 2≤D := (le_max_left _ _).trans hD
  29   have hDpos : (0 : ℝ)<D := by exact_mod_cast (show 0<D by omega)
  30   have hprod : 0≤(1+ε)*(D : ℝ) := by positivity
  31   have hcols : ⌈(1+ε)*(D : ℝ)⌉₊≤2*D := by
  32     apply Nat.ceil_le.mpr
  33     push_cast
  34     nlinarith
  35   obtain ⟨f,_hf,hproper⟩ := hN D B ((le_max_right _ _).trans hD) hB V E
  36     (Fin ⌈(1+ε)*(D : ℝ)⌉₊) (by simpa using hcols) R (fun _=>Finset.univ)
  37     ⌈(1+ε)*(D : ℝ)⌉₊ hsize (by intro e;simp) hdegree hcodegree
  38     (Nat.le_ceil _) hcols
  39   exact ⟨f,hproper⟩
