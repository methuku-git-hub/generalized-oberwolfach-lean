  15 theorem prescribed_one_ended_component_step
  16     (as:List X) (ha:as.Nodup) (A:Finset X) (f:X→V)
  17     (hinj:Set.InjOn f A) (hsource:∀ u∈as,u∉A) (a:X) (haA:a∈A)
  18     (S L:Finset V) (β:ℝ) (D:V→V→Prop) (hext:IsExtendingList D L β)
  19     (F J:SimpleGraph V) (M:ℝ) (hM:0≤M) (hdeg:∀ v,(F.degree v:ℝ)≤M) (forward:Bool)
  20     (hbudget:((A.image f∪S).card:ℝ)+as.length+1+M<β*Fintype.card V)
  21     (E:I→Finset (Sym2 V)) (k:I) (hpair:PairwisePhysicalDisjoint E)
  22     (hcover:∀ j,E j⊆F.edgeFinset) (hJ:J.edgeFinset⊆F.edgeFinset):
  23     ∃ g:X→V,Set.InjOn g (↑(A∪as.toFinset):Set X) ∧ (∀ u∈A,g u=f u) ∧
  24       (∀ u∈as,g u∈L ∧ g u∉S) ∧ DirectedPathFrom D F forward (g a) (as.map g) ∧
  25       (let added:=(prescribedOneSidedPhysicalEdges a as).image (Sym2.map g);
  26        PairwisePhysicalDisjoint (Function.update E k (E k∪added)) ∧ Disjoint added J.edgeFinset) := by

  28   exact ⟨g,hg,hpres,hlist,hpath,prescribed_one_ended_mapped_physical_step g D F J forward a as hpath E k hpair hcover hJ⟩
