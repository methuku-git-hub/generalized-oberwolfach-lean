  15 def expansionReindexResource (e : E≃E') (j : J≃J') :
  16     (Sym2 (ExpansionRow V E J W)⊕(ExpansionRow V E J W×ExpansionHub U E J)) →
  17     (Sym2 (ExpansionRow V E' J' W)⊕(ExpansionRow V E' J' W×ExpansionHub U E' J')) :=
  18   Sum.map (Sym2.map (expansionReindexRow e j))
  19     (Prod.map (expansionReindexRow e j) (expansionReindexHub e j))

  22     Function.Injective (expansionReindexResource (V:=V) (U:=U) (W:=W) e j) := by
  23   intro a b h
  24   cases a with
  25   | inl a =>
  26     cases b with
  27     | inl b => exact congrArg Sum.inl (Sym2.map.injective (expansionReindexRow e j).injective (Sum.inl.inj h))
  28     | inr b => cases h
  29   | inr a =>
  30     cases b with
  31     | inl b => cases h
  32     | inr b =>
  33       have hh := Sum.inr.inj h
  34       exact congrArg Sum.inr (Prod.ext ((expansionReindexRow e j).injective (congrArg Prod.fst hh))

  55       rw [ordinaryHostResources_spoke] at hmem

  57   · intro ha
  58     cases a with
  59     | inl a =>

  61       | _ b c =>
  62         obtain ⟨b,rfl⟩ := (expansionReindexRow (V:=V) (W:=W) e j).surjective b

  67     | inr p =>

  69       obtain ⟨b,rfl⟩ := (expansionReindexRow (V:=V) (W:=W) e j).surjective b
