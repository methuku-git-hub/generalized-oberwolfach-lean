   7 theorem normalized_weight_bounds (n S d c cmin a b w:ℝ)
   8     (hn:0<n) (hS:0<S) (hdlo:n/4≤d) (hdhi:d≤n)
   9     (hcmin:0<cmin) (hclo:cmin≤c) (hchi:c≤1)
  10     (ha:0≤a) (hb:0≤b) (hwlo:a/S≤w) (hwhi:w≤b/S):
  11     a/(S*n)≤w/d/c ∧ w/d/c≤(4*b/cmin)/(S*n) := by
  12   have hd:0<d:=by linarith
  13   have hc:0<c:=lt_of_lt_of_le hcmin hclo
  14   have hA:0≤a/S:=div_nonneg ha hS.le
  15   have hB:0≤b/S:=div_nonneg hb hS.le
  16   have hLow1:a/S/n≤a/S/d:=div_le_div_of_nonneg_left hA hd hdhi
  17   have hLow2:a/S/d≤(a/S/d)/c:=by
  18     have hh:=div_le_div_of_nonneg_left (div_nonneg hA hd.le) hc hchi
  19     simpa only [div_one] using hh
  20   have hLow3:(a/S/d)/c≤w/d/c:=
  21     div_le_div_of_nonneg_right (div_le_div_of_nonneg_right hwlo hd.le) hc.le
  22   have hlow:a/(S*n)≤w/d/c:=by
  23     rw [←div_div]
  24     exact hLow1.trans (hLow2.trans hLow3)
  25   have hUp1:w/d/c≤(b/S/d)/c:=
  26     div_le_div_of_nonneg_right (div_le_div_of_nonneg_right hwhi hd.le) hc.le
  27   have hn4:0<n/4:=by positivity
  28   have hUp2:(b/S/d)/c≤(b/S/(n/4))/c:=
  29     div_le_div_of_nonneg_right (div_le_div_of_nonneg_left hB hn4 hdlo) hc.le
  30   have hUp3:(b/S/(n/4))/c≤(b/S/(n/4))/cmin:=
  31     div_le_div_of_nonneg_left (div_nonneg hB hn4.le) hcmin hclo
  32   have hEq:(b/S/(n/4))/cmin=(4*b/cmin)/(S*n):=by
  33     field_simp
  34     <;> ring
  35   exact ⟨hlow,by rw [←hEq];exact hUp1.trans (hUp2.trans hUp3)⟩
