  16 theorem marked_wheel_correction {V U E K : Type*} [Fintype V] [Fintype U] [Fintype E] [Fintype K]
  17     [DecidableEq V] [DecidableEq U]
  18     {tail head : E → V} {ktail khead : K → V} {m : ℕ} {π : Equiv.Perm V} {b : ℕ}
  19     {H₀ Hₖ : V → U → Prop}
  20     (β η L δ D C : ℝ) (hβ : 0<β) (hη : 0≤η) (hC : 0≤C)
  21     (hn : 0<Fintype.card V) (hu : 0<Fintype.card U)
  22     (p q : Bool → U → ℝ)
  23     (hdata : ∀ j,SparseIntersectionData (if j then Hₖ else H₀) (p j) (q j) L δ D)
  24     (hsmall : δ+D/(Fintype.card V:ℝ)+1/(Fintype.card U:ℝ)≤1/(200*L^6))
  25     (i : Bool → Fin (m+1)) (hipos : ∀ j,markedPositionType (i j)=j)
  26     (hcommon : ∀ u u' : U,u≠u' → β*(Fintype.card V:ℝ)^(m+1)≤
  27       (Fintype.card (CommonMarkedCycle tail head ktail khead m π b H₀ Hₖ u u'):ℝ))
  28     (hpaths : ∀ j,∀ Q : BipartiteFourCycle (if j then Hₖ else H₀),
  29       β*(Fintype.card V:ℝ)^m≤
  30       (Fintype.card (MarkedReplacementPair tail head ktail khead m π b H₀ Hₖ (i j)
  31         Q.val.1.1 Q.val.1.2 Q.val.2.1 Q.val.2.2):ℝ))
  32     (hinj : Function.Injective (fun e => (tail e,head e)))
  33     (hkinj : Function.Injective (fun e => (ktail e,khead e)))
  34     (hK : ∀ v,occurrenceDegree khead v=∑ u,if Hₖ v u then (1:ℝ) else 0)
  35     (hout : ∀ v,occurrenceDegree (Sum.elim tail ktail) v=
  36       (∑ u,if H₀ v u then (1:ℝ) else 0)+(∑ u,if Hₖ v u then (1:ℝ) else 0))
  37     (hhub : ∀ u,(∑ v,if H₀ v u then (1:ℝ) else 0)=(m:ℝ)*(∑ v,if Hₖ v u then (1:ℝ) else 0))
  38     (ψ R : MarkedWheel tail head ktail khead m π b H₀ Hₖ → ℝ)
  39     (hR : ∀ e,markedWheelRimLoad R e=1-markedWheelRimLoad ψ e)
  40     (hRb : ∀ W,|R W|≤C*η/(Fintype.card V:ℝ)^m)
  41     (hRsp : ∀ j v u,|markedWheelSpokeLoad R j v u|≤(m+1)*C*η)
  42     (herr : ∀ j v u,|markedSpokeResidual ψ j v u|≤η) :
  43     ∃ w : MarkedWheel tail head ktail khead m π b H₀ Hₖ → ℝ,
  44       (∀ e,markedWheelRimLoad w e=1) ∧
  45       (∀ j v u,markedWheelSpokeLoad w j v u=if (if j then Hₖ v u else H₀ v u) then 1 else 0) ∧
  46       (∀ W,|w W-ψ W|≤markedWheelCorrectionConstant m β L C*η/(Fintype.card V:ℝ)^m) := by
  47   let ψ₁ := ψ+R
  48   have hR1 (e : E⊕K) : markedWheelRimLoad ψ₁ e=1 := by
  49     change markedWheelRimLoad (ψ+R) e=1
  50     rw [markedWheelRimLoad_add,hR]
  51     ring
  52   have hrow : ∀ j v,(∑ u,markedSpokeResidual ψ₁ j v u)=0 :=
  53     markedSpokeResidual_zero_rows hinj hkinj hK hout ψ₁ hR1
  54   have hcol : ∀ u,(∑ v,markedSpokeResidual ψ₁ false v u)=(m:ℝ)*(∑ v,markedSpokeResidual ψ₁ true v u) :=
  55     markedSpokeResidual_column_ratio hhub ψ₁
  56   let η₁ := (1+(m+1)*C)*η
  57   have hη₁ : 0≤η₁ := by dsimp [η₁];positivity
  58   have herr₁ (j : Bool) (v : V) (u : U) : |markedSpokeResidual ψ₁ j v u|≤η₁ := by
  59     have he : markedSpokeResidual ψ₁ j v u=markedSpokeResidual ψ j v u-markedWheelSpokeLoad R j v u := by
  60       unfold markedSpokeResidual
  61       change (if (if j then Hₖ v u else H₀ v u) then 1 else 0)-markedWheelSpokeLoad (ψ+R) j v u=_
  62       rw [markedWheelSpokeLoad_add]
  63       ring
  64     rw [he]
  65     have hh := abs_sub_le (markedSpokeResidual ψ j v u) 0 (markedWheelSpokeLoad R j v u)
  66     simp only [sub_zero,zero_sub,abs_neg] at hh
  67     calc
  68       _ ≤ η+(m+1)*C*η := hh.trans (add_le_add (herr j v u) (hRsp j v u))
  69       _ = _ := by dsimp only [η₁];ring
  70   obtain ⟨S,hSrim,hSspoke,hSb⟩ := marked_spoke_correction β η₁ L δ D hβ hη₁ hn hu p q hdata hsmall
  71     i hipos hcommon hpaths ψ₁ hrow hcol herr₁
  72   refine ⟨ψ₁+S,?_,?_,?_⟩
  73   · intro e
  74     rw [markedWheelRimLoad_add,hR1,hSrim,add_zero]
  75   · intro j v u
  76     rw [markedWheelSpokeLoad_add,hSspoke,markedSpokeResidual]
  77     ring
  78   · intro W
  79     have he : (ψ₁+S) W-ψ W=R W+S W := by dsimp only [ψ₁,Pi.add_apply];ring
  80     rw [he]
  81     calc
  82       _ ≤ |R W|+|S W| := abs_add_le _ _
  83       _ ≤ C*η/(Fintype.card V:ℝ)^m+markedSpokeCorrectionConstant m β L*η₁/(Fintype.card V:ℝ)^m :=
  84         add_le_add (hRb W) (hSb W)
  85       _ = _ := by dsimp only [markedWheelCorrectionConstant,η₁];ring
