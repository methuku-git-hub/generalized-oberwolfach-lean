  15 theorem prescribed_rim_wheel_inner_intersection (C:OrdinaryWheel Q F H r)
  16     (I X:Finset V) (J:Finset U) (hIX:Disjoint I X) (x y:V) (hx:x∈I) (hy:y∈I)
  17     (hxy:(ordinaryWheelRim C).Adj x y) (hu:ordinaryWheelHub C∉J)
  18     (hs:∀ a∈(ordinaryWheelRim C).support,a=x∨a=y∨a∈X) :
  19     ordinaryWheelResources C∩ordinaryHostResources (spanningRestriction Q I)
  20       (fun a u=>H a u∧a∈I∧u∈J)={Sum.inl s(x,y)}:=by
  21   have hroot (a:V) (ha:a∈(ordinaryWheelRim C).support) (hai:a∈I):a=x∨a=y:=by
  22     rcases hs a ha with h|h|h
  23     · exact Or.inl h
  24     · exact Or.inr h
  25     · exact (Finset.disjoint_left.mp hIX hai h).elim
  26   ext e
  27   constructor
  28   · intro he
  29     obtain ⟨heC,heI⟩:=Finset.mem_inter.mp he
  30     cases e with
  31     | inl e =>
  32       rw [ordinaryWheelResources_rim] at heC
  33       rw [ordinaryHostResources_rim] at heI
  34       induction e using Sym2.inductionOn with
  35       | _ a b =>
  36         have hi:=(spanningRestriction_adj Q I a b).mp heI
  37         rcases hroot a heC.mem_support_left hi.2.1 with ha|ha <;>
  38           rcases hroot b heC.mem_support_right hi.2.2 with hb|hb
  39         · exact (heC.ne (ha.trans hb.symm)).elim
  40         · simp [ha,hb]
  41         · simp [ha,hb,Sym2.eq_swap]
  42         · exact (heC.ne (ha.trans hb.symm)).elim
  43     | inr au =>
  44       rw [ordinaryWheelResources_spoke] at heC
  45       rw [ordinaryHostResources_spoke] at heI
  46       exact (hu (heC.1.symm ▸ heI.2.2)).elim
  47   · intro he
  48     have he':e=Sum.inl s(x,y):=Finset.mem_singleton.mp he
  49     subst e
  50     refine Finset.mem_inter.mpr ⟨(ordinaryWheelResources_rim C _).mpr hxy,?_⟩
  51     rw [ordinaryHostResources_rim]
  52     exact (spanningRestriction_adj Q I x y).mpr ⟨C.val.1.property hxy,hx,hy⟩

  54 theorem prescribed_spoke_wheel_inner_intersection (C:OrdinaryWheel Q F H r)
  55     (I X:Finset V) (J:Finset U) (hIX:Disjoint I X) (x:V) (u:U)
  56     (hx:x∈I) (hu:u∈J) (hCu:ordinaryWheelHub C=u) (hCx:x∈(ordinaryWheelRim C).support)
  57     (hs:∀ a∈(ordinaryWheelRim C).support,a=x∨a∈X) :
  58     ordinaryWheelResources C∩ordinaryHostResources (spanningRestriction Q I)
  59       (fun a u=>H a u∧a∈I∧u∈J)={Sum.inr (x,u)}:=by
  60   have hroot (a:V) (ha:a∈(ordinaryWheelRim C).support) (hai:a∈I):a=x:=by
  61     rcases hs a ha with h|h
  62     · exact h
  63     · exact (Finset.disjoint_left.mp hIX hai h).elim
  64   ext e
  65   constructor
  66   · intro he
  67     obtain ⟨heC,heI⟩:=Finset.mem_inter.mp he
  68     cases e with
  69     | inl e =>
  70       rw [ordinaryWheelResources_rim] at heC
  71       rw [ordinaryHostResources_rim] at heI
  72       induction e using Sym2.inductionOn with
  73       | _ a b =>
  74         have hi:=(spanningRestriction_adj Q I a b).mp heI
  75         exact (heC.ne ((hroot a heC.mem_support_left hi.2.1).trans (hroot b heC.mem_support_right hi.2.2).symm)).elim
  76     | inr au =>
  77       rw [ordinaryWheelResources_spoke] at heC
  78       rw [ordinaryHostResources_spoke] at heI
  79       have ha:=hroot au.1 heC.2 heI.2.1
  80       have hb:au.2=u:=heC.1.symm.trans hCu
  81       exact Finset.mem_singleton.mpr (congrArg Sum.inr (Prod.ext ha hb))
  82   · intro he
  83     have he':e=Sum.inr (x,u):=Finset.mem_singleton.mp he
  84     subst e
  85     refine Finset.mem_inter.mpr ⟨(ordinaryWheelResources_spoke C x u).mpr ⟨hCu,hCx⟩,?_⟩
  86     rw [ordinaryHostResources_spoke]
  87     exact ⟨hCu ▸ C.property.1 x hCx,hx,hu⟩
