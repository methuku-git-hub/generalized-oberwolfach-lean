  17 theorem profile_embedding_preimage_class_card (e : (Σ i,W i) ↪ (Σ i,V i))
  18     (he : ∀ v,(e v).1=v.1) (F : Finset (Σ i,V i))
  19     (hF : ∀ v∈F,∃ w,e w=v) (i : I) :
  20     (Finset.univ.filter (fun w : W i => e ⟨i,w⟩∈F)).card=
  21       (Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈F)).card := by
  22   let A : Finset (Σ i,W i) := Finset.univ.filter (fun w => e w∈F)
  23   have hmap : A.map e=F := by
  24     ext v
  25     constructor
  26     · intro hv
  27       obtain ⟨w,hw,rfl⟩ := Finset.mem_map.mp hv
  28       exact (Finset.mem_filter.mp hw).2
  29     · intro hv
  30       obtain ⟨w,rfl⟩ := hF v hv
  31       exact Finset.mem_map.mpr ⟨w,Finset.mem_filter.mpr ⟨Finset.mem_univ _,hv⟩,rfl⟩
  32   have hfilter : (A.filter (fun w => w.1=i)).map e=F.filter (fun v => v.1=i) := by
  33     ext v
  34     constructor
  35     · intro hv
  36       obtain ⟨w,hw,rfl⟩ := Finset.mem_map.mp hv
  37       obtain ⟨hw,hi⟩ := Finset.mem_filter.mp hw
  38       exact Finset.mem_filter.mpr ⟨(Finset.mem_filter.mp hw).2,(he w).trans hi⟩
  39     · intro hv
  40       obtain ⟨hv,hi⟩ := Finset.mem_filter.mp hv
  41       rw [← hmap] at hv
  42       obtain ⟨w,hw,rfl⟩ := Finset.mem_map.mp hv
  43       exact Finset.mem_map.mpr ⟨w,Finset.mem_filter.mpr ⟨hw,(he w).symm.trans hi⟩,rfl⟩
  44   have hcard := congrArg Finset.card hfilter
  45   rw [Finset.card_map,sigma_finset_class_card] at hcard
  46   rw [sigma_finset_class_card F i] at hcard
  47   simpa [A] using hcard
