  15 theorem startRemovedHubs_subset_iff (A : V→Finset U) (t : ℕ) (J : StartRetentionOutcome A t)
  16     (a : V) (T : Finset U) (hT : T⊆A a) :
  17     T⊆startRemovedHubs A t J a ↔ T.subtype (fun u=>u∈A a)⊆(J a).valᶜ := by
  18   constructor
  19   · intro h v hv
  20     have hvT : v.val∈T := by simpa using hv
  21     have hd := h hvT
  22     obtain ⟨_,hn⟩ := Finset.mem_sdiff.mp hd
  23     apply Finset.mem_compl.mpr
  24     intro hj
  25     exact hn (Finset.mem_map.mpr ⟨v,hj,rfl⟩)
  26   · intro h u hu
  27     have huA := hT hu
  28     have hn := h (show (⟨u,huA⟩ : {u // u∈A a})∈T.subtype (fun u=>u∈A a) by simpa using hu)
  29     apply Finset.mem_sdiff.mpr
  30     refine ⟨huA,?_⟩
  31     intro hj
  32     obtain ⟨v,hv,he⟩ := Finset.mem_map.mp hj
  33     apply Finset.mem_compl.mp hn
  34     have he' : v=(⟨u,huA⟩ : {u // u∈A a}) := Subtype.ext he
  35     exact he' ▸ hv

  37 theorem startRemovedHubs_joint_probability_le (A : V→Finset U) (t : ℕ) (ht : ∀ a,t≤(A a).card)
  38     (a : V) (T : Finset U) :
  39     (startRetentionLaw A t ht).probability (fun J=>T⊆startRemovedHubs A t J a)≤
  40       (((A a).card-t:ℕ):ℝ)^T.card/((A a).card:ℝ)^T.card := by
  41   by_cases hT : T⊆A a
  42   · letI : Nonempty (FixedSubset {u // u∈A a} t) := fixedSubset_nonempty (by simpa using ht a)
  43     simp_rw [startRemovedHubs_subset_iff A t _ a T hT]
  44     rw [startRetentionLaw_coordinate_event A t ht a
  45       (fun X=>T.subtype (fun u=>u∈A a)⊆X.valᶜ)]
  46     have hh := fixedSubset_joint_exclusion_le (V:={u // u∈A a}) (by simpa using ht a)
  47       (T.subtype (fun u=>u∈A a))
  48     have hc : (T.subtype (fun u=>u∈A a)).card=T.card := by
  49       rw [Finset.card_subtype]
  50       rw [Finset.filter_eq_self.mpr (fun u hu=>hT hu)]
  51     simp only [Finset.subset_iff,Finset.mem_compl] at hh ⊢
  52     simpa only [hc,Fintype.card_coe] using hh
  53   · have he : (fun J : StartRetentionOutcome A t=>T⊆startRemovedHubs A t J a)=(fun _=>False) := by
  54       funext J
  55       apply propext
  56       constructor
  57       · intro h;exact hT (h.trans Finset.sdiff_subset)
  58       · exact False.elim
  59     rw [he]
  60     simp only [FiniteLaw.probability,ite_false,Finset.sum_const_zero]
  61     positivity
