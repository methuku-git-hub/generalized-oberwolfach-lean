   8 theorem marked_sample_reference_density_hierarchy
   9     (m:ℕ) (hm:1≤m) (ρ s ρ₀ sstar:ℝ)
  10     (hρ₀:0<ρ₀) (hsstar:0<sstar) (hρlo:ρ₀≤ρ) (hρhi:ρ≤1)
  11     (hslo:sstar≤s) (hshi:s≤1/100):
  12     let t₀:=sstar/m
  13     let t:=s/m
  14     let pA0:=ρ₀*sstar/2
  15     let p0:=ρ₀*sstar/(2*m)
  16     let p:=ρ*s/2
  17     let k:=p/m;
  18     0<ρ₀ ∧ ρ₀≤1 ∧ 0<t₀ ∧ t₀≤1 ∧ t₀≤t ∧ 0<pA0 ∧ pA0≤1/2 ∧
  19       0<p0 ∧ p0≤1/2 ∧ p0≤p ∧ p0≤k ∧ p0≤t ∧
  20       pA0≤ρ*m*t/2 ∧ p0≤ρ*m*t/2 ∧ p0≤ρ*t/2 ∧
  21       p0/2≤ρ₀*t₀/4 ∧ (m:ℝ)*t≤1:=by
  22   intro t₀ t pA0 p0 p k
  23   have hmR:0<(m:ℝ):=by exact_mod_cast (show 0<m by omega)
  24   have hm1:(1:ℝ)≤m:=by exact_mod_cast hm
  25   have hρ:0<ρ:=hρ₀.trans_le hρlo
  26   have hs:0<s:=hsstar.trans_le hslo
  27   have hρ₀1:ρ₀≤1:=hρlo.trans hρhi
  28   have hsstar1:sstar≤1:=by linarith
  29   have hs1:s≤1:=by linarith
  30   have hprod:ρ₀*sstar≤ρ*s:=mul_le_mul hρlo hslo hsstar.le hρ.le
  31   have ht₀:0<t₀:=by dsimp [t₀];positivity
  32   have ht₀1:t₀≤1:=by dsimp [t₀];apply (div_le_iff₀ hmR).mpr;linarith
  33   have htle:t₀≤t:=by dsimp [t₀,t];exact div_le_div_of_nonneg_right hslo hmR.le
  34   have hPA:0<pA0:=by dsimp [pA0];positivity
  35   have hPA1:pA0≤1/2:=by
  36     have hh:=mul_le_mul hρ₀1 hsstar1 hsstar.le (by norm_num:(0:ℝ)≤1)
  37     dsimp [pA0];linarith
  38   obtain ⟨hp0,hp0p,hp0k,_,_,_,_,_,_⟩:=marked_seed_density_bounds m hm ρ s ρ₀ sstar hρ₀ hsstar hρlo hρhi hslo hshi
  39   have hkform:k=ρ*t/2:=by dsimp [k,p,t];field_simp <;> ring
  40   have hpform:p=ρ*m*t/2:=by dsimp [p,t];field_simp <;> ring
  41   have hp0t:p0≤t:=by
  42     have hk_t:k≤t:=by rw [hkform];nlinarith [mul_nonneg (sub_nonneg.mpr hρhi) (show 0≤t by dsimp [t];positivity)]
  43     exact hp0k.trans hk_t
  44   have hPAp:pA0≤p:=by dsimp [pA0,p];linarith
  45   have hp01:p0≤1/2:=by
  46     have hh:=mul_le_mul_of_nonneg_right hp0t (show (0:ℝ)≤1 by norm_num)
  47     dsimp [t] at hh
  48     have htbound:s/(m:ℝ)≤1/100:=by apply (div_le_iff₀ hmR).mpr;nlinarith
  49     linarith
  50   have hp0ref:p0/2≤ρ₀*t₀/4:=by dsimp [p0,t₀];field_simp <;> ring_nf <;> norm_num
  51   have htUpper:(m:ℝ)*t≤1:=by dsimp [t];field_simp;linarith
  52   refine ⟨hρ₀,hρ₀1,ht₀,ht₀1,htle,hPA,hPA1,hp0,hp01,hp0p,hp0k,hp0t,?_,?_,?_,hp0ref,htUpper⟩
  53   · rw [←hpform];exact hPAp
  54   · rw [←hpform];exact hp0p
  55   · rw [←hkform];exact hp0k
