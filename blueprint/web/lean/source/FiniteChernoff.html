  11     {I : Type*} [Fintype I] [DecidableEq I] {Ω : I → Type*} [∀ i, Fintype (Ω i)]
  12     (μ : ∀ i, FiniteLaw (Ω i)) (X : ∀ i, Ω i → ℝ) :
  13     (FiniteLaw.pi μ).expectation (fun σ => ∑ i, X i (σ i)) =
  14       ∑ i, (μ i).expectation (X i) := by
  15   let : ∀ i, MeasurableSpace (Ω i) := fun _ => ⊤
  16   have he : ∀ i, (FiniteLaw.pi μ).expectation (fun σ => X i (σ i)) =
  17       (μ i).expectation (X i) := fun i => finiteLaw_coordinate_expectation μ i (X i)
  18   simp only [FiniteLaw.expectation, Finset.mul_sum]
  19   rw [Finset.sum_comm]
  20   exact Finset.sum_congr rfl (fun i _ => he i)

  22 theorem finiteLaw_nonnegative_zero_mean
  23     {Ω : Type*} [Fintype Ω] (μ : FiniteLaw Ω) (X : Ω → ℝ)
  24     (hX : ∀ ω, 0 ≤ X ω) (hzero : μ.expectation X = 0)
  25     (ω : Ω) (hω : 0 < μ.mass ω) : X ω = 0 := by
  26   have hh : μ.mass ω * X ω ≤ 0 := by
  27     rw [← hzero]
  28     exact Finset.single_le_sum (fun x _ => mul_nonneg (μ.nonneg x) (hX x)) (Finset.mem_univ ω)
  29   exact le_antisymm (nonpos_of_mul_nonpos_right hh hω) (hX ω)

  31 theorem finiteLaw_independent_chernoff
  32     {I : Type*} [Fintype I] [DecidableEq I] {Ω : I → Type*} [∀ i, Fintype (Ω i)]
  33     (μ : ∀ i, FiniteLaw (Ω i)) (X : ∀ i, Ω i → ℝ)
  34     (hX : ∀ i ω, X i ω ∈ Set.Icc (0 : ℝ) 1)
  35     {ε : ℝ} (he : 0 ≤ ε) (he1 : ε ≤ 1) :
  36     let a := (FiniteLaw.pi μ).expectation (fun σ => ∑ i, X i (σ i))
  37     (FiniteLaw.pi μ).probability (fun σ => ε * a ≤ |(∑ i, X i (σ i)) - a|) ≤
  38       2 * Real.exp (-ε ^ 2 * a / 3) := by
  39   dsimp only
  40   rw [finiteLaw_pi_expectation_sum]
  41   apply finiteLaw_poisson_mgf_absolute_tail _ _ _
  42     (finiteLaw_independent_unit_interval_mgf μ X hX) he he1
  43   exact Finset.sum_nonneg (fun i _ => Finset.sum_nonneg
  44     (fun ω _ => mul_nonneg ((μ i).nonneg ω) (hX i ω).1))
