  12 theorem removed_split_block_singleton (H:SimpleGraph V) (R:Finset V)
  13     (hisol:∀x,x∉R→∀y,¬H.Adj x y) (xs block:List V) (hxs:xs.Nodup)
  14     (hb:block∈splitPathByGraph H xs) (hnot:¬block.toFinset⊆R):
  15     ∃x,x∉R ∧ block=[x] := by
  16   have hex:∃x,x∈block.toFinset ∧ x∉R := by
  17     by_contra h
  18     apply hnot
  19     intro x hx
  20     by_contra hn
  21     exact h ⟨x,hx,hn⟩

  26 theorem retained_path_blocks_cover (H:SimpleGraph V) (R:Finset V)
  27     (hisol:∀x,x∉R→∀y,¬H.Adj x y) (xs:List V) (hxs:xs.Nodup) (x:V):
  28     x∈(retainedPathBlocks H R xs).flatten ↔ x∈xs ∧ x∈R := by
  29   constructor
  30   · intro hx

  34     have hflat:=split_ordered_path_flatten (fun x y=>decide (H.Adj x y)) xs
  35     refine ⟨?_,(of_decide_eq_true hh.2) (List.mem_toFinset.mpr hx)⟩
  36     simpa [splitPathByGraph,hflat] using hxs'
  37   · rintro ⟨hx,hxR⟩
  38     have hflat:=split_ordered_path_flatten (fun x y=>decide (H.Adj x y)) xs
