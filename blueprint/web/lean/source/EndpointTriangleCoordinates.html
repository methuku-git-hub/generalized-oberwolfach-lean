   3 /-! An endpoint triangle has a unique index and a unique unordered row
   4 edge. These coordinates identify the actual triangle without choosing an
   5 orientation of its row edge. -/

  11 attribute [local instance] Classical.propDecidable
  12 variable {V : Type*} [Fintype V] [DecidableEq V]
  13   {Q F : SimpleGraph V} {S T : Finset V}

  15 def endpointTriangleIndex (C : EndpointTriangle Q F S T) : V :=
  16   Classical.choose (Finset.card_eq_one.mp C.property.2.2.1)
  17 
  18 theorem endpointTriangleIndex_spec (C : EndpointTriangle Q F S T) :
  19     C.val∩T={endpointTriangleIndex C} :=
  20   Classical.choose_spec (Finset.card_eq_one.mp C.property.2.2.1)

  22 theorem endpointTriangleRowEdge_exists (C : EndpointTriangle Q F S T) :
  23     ∃ e : Sym2 V,¬e.IsDiag ∧ e.toFinset=C.val∩S := by
  24   obtain ⟨v,w,hne,he⟩:=Finset.card_eq_two.mp C.property.2.1
  25   refine ⟨s(v,w),?_,?_⟩
  26   · simpa using hne
  27   · rw [Sym2.toFinset_mk_eq,he]
  28 
  29 def endpointTriangleRowEdge (C : EndpointTriangle Q F S T) : Sym2 V :=
  30   Classical.choose (endpointTriangleRowEdge_exists C)
  31 
  32 theorem endpointTriangleRowEdge_spec (C : EndpointTriangle Q F S T) :
  33     ¬(endpointTriangleRowEdge C).IsDiag ∧ (endpointTriangleRowEdge C).toFinset=C.val∩S :=
  34   Classical.choose_spec (endpointTriangleRowEdge_exists C)

  36 theorem endpointTriangleIndex_mem (C : EndpointTriangle Q F S T) :
  37     endpointTriangleIndex C∈C.val ∧ endpointTriangleIndex C∈T := by
  38   apply Finset.mem_inter.mp
  39   rw [endpointTriangleIndex_spec]
  40   exact Finset.mem_singleton_self _
  41 
  42 theorem endpointTriangleRowEdge_mem (C : EndpointTriangle Q F S T) (v : V) :
  43     v∈endpointTriangleRowEdge C ↔ v∈C.val ∧ v∈S := by
  44   rw [←Sym2.mem_toFinset,(endpointTriangleRowEdge_spec C).2,Finset.mem_inter]

  46 theorem endpointTriangle_vertices_coordinates (C : EndpointTriangle Q F S T) :
  47     C.val=insert (endpointTriangleIndex C) (endpointTriangleRowEdge C).toFinset := by
  48   rw [(endpointTriangleRowEdge_spec C).2]
  49   ext v
  50   constructor
  51   · intro hv
  52     rcases Finset.mem_union.mp (C.property.1 hv) with hS|hT
  53     · exact Finset.mem_insert_of_mem (Finset.mem_inter.mpr ⟨hv,hS⟩)
  54     · have hh : v∈C.val∩T:=Finset.mem_inter.mpr ⟨hv,hT⟩
  55       rw [endpointTriangleIndex_spec] at hh
  56       exact Finset.mem_insert.mpr (Or.inl (Finset.mem_singleton.mp hh))
  57   · intro hv
  58     rcases Finset.mem_insert.mp hv with rfl|hv
  59     · exact (endpointTriangleIndex_mem C).1
  60     · exact (Finset.mem_inter.mp hv).1

  62 theorem endpointTriangle_coordinates_injective :
  63     Function.Injective (fun C : EndpointTriangle Q F S T=>(endpointTriangleIndex C,endpointTriangleRowEdge C)) := by
  64   intro C D h
  65   have hi : endpointTriangleIndex C=endpointTriangleIndex D:=congrArg Prod.fst h
  66   have he : endpointTriangleRowEdge C=endpointTriangleRowEdge D:=congrArg Prod.snd h
  67   apply Subtype.ext
  68   rw [endpointTriangle_vertices_coordinates C,endpointTriangle_vertices_coordinates D,hi,he]

  70 theorem endpointTriangleRowEdge_in_graph (C : EndpointTriangle Q F S T) :
  71     endpointTriangleRowEdge C∈Q.edgeSet ∧ endpointTriangleRowEdge C∉F.edgeSet := by
  72   generalize he : endpointTriangleRowEdge C=e
  73   induction e using Sym2.ind with
  74   | _ v w =>
  75     have hv : v∈C.val := ((endpointTriangleRowEdge_mem C v).mp (by rw [he];simp)).1
  76     have hw : w∈C.val := ((endpointTriangleRowEdge_mem C w).mp (by rw [he];simp)).1
  77     have hne : v≠w := by simpa only [he,Sym2.mk_isDiag_iff] using (endpointTriangleRowEdge_spec C).1
  78     exact ⟨C.property.2.2.2.1 v hv w hw hne,C.property.2.2.2.2 v hv w hw⟩

  80 theorem endpointTriangle_fixed_row_coordinates (hST : Disjoint S T)
  81     (C : EndpointTriangle Q F S T) (v : V) (hvS : v∈S) (hvC : v∈C.val) :
  82     ∃ w : V,v≠w ∧ w∈S ∧ endpointTriangleRowEdge C=s(v,w) ∧
  83       C.val={endpointTriangleIndex C,v,w} := by
  84   have hv := (endpointTriangleRowEdge_mem C v).mpr ⟨hvC,hvS⟩
  85   obtain ⟨w,he⟩:=Sym2.mem_iff_exists.mp hv
  86   have hw := ((endpointTriangleRowEdge_mem C w).mp (by rw [he];simp)).2
  87   have hne : v≠w := by simpa only [he,Sym2.mk_isDiag_iff] using (endpointTriangleRowEdge_spec C).1
  88   refine ⟨w,hne,hw,he,?_⟩
  89   rw [endpointTriangle_vertices_coordinates C,he,Sym2.toFinset_mk_eq]

  91 theorem endpointTriangle_coordinates_of_vertices (hST : Disjoint S T)
  92     (C : EndpointTriangle Q F S T) (u v w : V) (hu : u∈T) (hv : v∈S) (hw : w∈S)
  93     (hC : C.val={u,v,w}) :
  94     endpointTriangleIndex C=u ∧ endpointTriangleRowEdge C=s(v,w) := by
  95   have huS : u∉S:=fun h=>Finset.disjoint_left.mp hST h hu
  96   have hvT : v∉T:=fun h=>Finset.disjoint_left.mp hST hv h
  97   have hwT : w∉T:=fun h=>Finset.disjoint_left.mp hST hw h
  98   have hT : C.val∩T={u} := by
  99     rw [hC,Finset.insert_inter_of_mem hu,Finset.insert_inter_of_notMem hvT]
 100     simp [hwT]
 101   have hS : C.val∩S={v,w} := by
 102     rw [hC,Finset.insert_inter_of_notMem huS,Finset.insert_inter_of_mem hv]
 103     simp [hw]
 104   constructor
 105   · have hh:=(endpointTriangleIndex_spec C).symm.trans hT
 106     exact Finset.singleton_injective hh
 107   · apply Sym2.ext
 108     intro z
