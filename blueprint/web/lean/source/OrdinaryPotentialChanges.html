  17 theorem ordinaryPotentialLoad_stability_ordered
  18     (w : OrdinaryWheel P F Hbase r → ℝ) (hw : ∀ W,0≤w W)
  19     (root : Sym2 V⊕(V×U)) (Q Q' : SimpleGraph V) (H H' : V → U → Prop) :
  20     |ordinaryPotentialLoad w root Q' H'-ordinaryPotentialLoad w root Q H|≤
  21       (∑ x,∑ y,if ¬(Q.Adj x y↔Q'.Adj x y) then ordinaryOmittedPairLoad w root (Sum.inl s(x,y)) else 0)+
  22       (∑ v,∑ u,if ¬(H v u↔H' v u) then ordinaryOmittedPairLoad w root (Sum.inr (v,u)) else 0) := by
  23   let D := Finset.univ.filter (fun e=>ordinaryResourceState Q H e≠ordinaryResourceState Q' H' e)
  24   have hb := ordinaryPotentialLoad_stability w hw root Q Q' H H' D
  25     (by intro e he;simpa only [D,Finset.mem_filter,Finset.mem_univ,true_and,not_not] using he)
  26   change |ordinaryPotentialLoad w root Q' H'-ordinaryPotentialLoad w root Q H|≤
  27     ∑ e∈D,ordinaryOmittedPairLoad w root e at hb
  28   have he : (∑ e∈D,ordinaryOmittedPairLoad w root e)=
  29       (∑ e : Sym2 V,if ¬(e∈Q.edgeSet↔e∈Q'.edgeSet) then ordinaryOmittedPairLoad w root (Sum.inl e) else 0)+
  30       (∑ v,∑ u,if ¬(H v u↔H' v u) then ordinaryOmittedPairLoad w root (Sum.inr (v,u)) else 0) := by
  31     dsimp only [D]
  32     rw [Finset.sum_filter,Fintype.sum_sum_type]
  33     simp only [ordinaryResourceState,Sum.elim_inl,Sum.elim_inr,ne_eq,decide_eq_decide,Fintype.sum_prod_type]
  34   rw [he] at hb
  35   apply hb.trans
  36   refine add_le_add ?_ (le_refl _)
  37   have hs := sum_sym2_le_ordered
  38     (fun e=>if ¬(e∈Q.edgeSet↔e∈Q'.edgeSet) then ordinaryOmittedPairLoad w root (Sum.inl e) else 0)
  39     (fun e=>by
  40       by_cases h : ¬(e∈Q.edgeSet↔e∈Q'.edgeSet)
  41       · rw [if_pos h];exact ordinaryOmittedPairLoad_nonneg w hw root _
  42       · simp only [if_neg h,le_refl])
  43   have ht : (∑ x,∑ y,if ¬(s(x,y)∈Q.edgeSet↔s(x,y)∈Q'.edgeSet) then ordinaryOmittedPairLoad w root (Sum.inl s(x,y)) else 0)=
  44       (∑ x,∑ y,if ¬(Q.Adj x y↔Q'.Adj x y) then ordinaryOmittedPairLoad w root (Sum.inl s(x,y)) else 0) := by
  45     apply Finset.sum_congr rfl
  46     intro x _
  47     apply Finset.sum_congr rfl
  48     intro y _
  49     simp only [SimpleGraph.mem_edgeSet]
  50   exact hs.trans_eq ht
