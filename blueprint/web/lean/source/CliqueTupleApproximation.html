  17 theorem clique_tuple_approximation
  18     {V C J : Type*} [Fintype V] [Fintype C] [Fintype J]
  19     (G : SimpleGraph V) (T : J → Finset V)
  20     (hT : ∀ j,∀ u∈T j,∀ v∈T j,u≠v → G.Adj u v)
  21     (μ : FiniteLaw (V → C×Bool)) (c : C) {s A : ℕ}
  22     (hsize : ∀ j,(T j).card≤s)
  23     (hcross : ∀ i j,i≠j → ∀ v∈T i,((T j).filter (G.Adj v)).card≤A)
  24     {b ε : ℝ} (hb : 0≤b) (hε : 0≤ε)
  25     (hgood : ∀ f : (j : J) → {v // v∈T j},
  26       (∀ i j,¬G.Adj (f i).val (f j).val) →
  27       b≤μ.probability (fun σ=>∀ j,coinRetainedColour G σ (f j).val=some c) ∧
  28       μ.probability (fun σ=>∀ j,coinRetainedColour G σ (f j).val=some c)≤(1+ε)*b) :
  29     |μ.probability (fun σ=>∀ j,∃ v∈T j,coinRetainedColour G σ v=some c)-
  30       (∏ j,((T j).card : ℝ))*b|≤
  31       (∏ j,((T j).card : ℝ))*ε*b+
  32         (((Fintype.card J)^2*A*s^(Fintype.card J-1) : ℕ) : ℝ)*b := by
  33   classical
  34   let X := fun j : J=>{v // v∈T j}
  35   let bad := fun f : ∀ j,X j=>∃ i j,i≠j ∧ G.Adj (f i).val (f j).val
  36   letI : DecidablePred bad := fun f=>Classical.propDecidable (bad f)
  37   let p := fun f : ∀ j,X j=>μ.probability (fun σ=>∀ j,coinRetainedColour G σ (f j).val=some c)
  38   have hp (f : ∀ j,X j) (hf : ¬bad f) : b≤p f ∧ p f≤(1+ε)*b := by
  39     apply hgood f
  40     intro i j hadj
  41     exact hf ⟨i,j,fun he=>G.irrefl (he ▸ hadj),hadj⟩
  42   have hz (f : ∀ j,X j) (hf : bad f) : p f=0 := by
  43     obtain ⟨i,j,_,hij⟩ := hf
  44     exact conflicting_retention_probability_zero G μ (fun j=>(f j).val) c ⟨i,j,hij⟩
  45   have hX (j : J) : Fintype.card (X j)≤s := by simpa only [X,Fintype.card_coe] using hsize j
  46   have hpairCount (i j : J) (hij : i≠j) (a : X i) :
  47       (Finset.univ.filter (fun b : X j=>G.Adj a.val b.val)).card≤A := by
  48     apply (Finset.card_le_card_of_injOn
  49       (s:=Finset.univ.filter (fun b : X j=>G.Adj a.val b.val))
  50       (t:=(T j).filter (G.Adj a.val)) (fun x : X j=>x.val) ?_ ?_).trans
  51       (hcross i j hij a.val a.property)
  52     · intro x hx
  53       exact Finset.mem_filter.mpr ⟨x.property,(Finset.mem_filter.mp hx).2⟩
  54     · intro x _ y _ he
  55       exact Subtype.ext he
  56   have hB : (Finset.univ.filter bad).card≤(Fintype.card J)^2*A*s^(Fintype.card J-1) := by
  57     have hh := finite_tuple_conflict_count (fun i j a b=>G.Adj a.val b.val) hpairCount hX
  58     convert hh using 1
  59     apply congrArg Finset.card
  60     ext f
  61     simp only [Finset.mem_filter,bad]
  62   have hN : (Fintype.card (∀ j,X j) : ℝ)=∏ j,((T j).card : ℝ) := by
  63     simp only [Fintype.card_pi,X,Fintype.card_coe,Nat.cast_prod]
  64   have hh := finite_sum_good_approximation bad p hb hε hp hz
  65   rw [hN] at hh
  66   have hBR : ((Finset.univ.filter bad).card : ℝ)≤
  67       (((Fintype.card J)^2*A*s^(Fintype.card J-1) : ℕ) : ℝ) := by exact_mod_cast hB
  68   rw [clique_joint_retention_expansion G T hT μ c]
  69   exact hh.trans (add_le_add (le_refl _) (mul_le_mul_of_nonneg_right hBR hb))
