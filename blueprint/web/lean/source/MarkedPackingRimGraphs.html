  37 theorem graph_adj_of_physical_pair_eq (D:SimpleGraph V) (a c x y:V)
  38     (he:({a,c}:Finset V)={x,y}) (hxy:D.Adj x y):D.Adj a c:=by
  39   have ha:a=x∨a=y:=by have hh:a∈({x,y}:Finset V):=he ▸ (by simp);simpa using hh
  40   rcases ha with ha|ha
  41   · rw [ha] at he ⊢
  42     have hy:y=x∨y=c:=by have hh:y∈({x,c}:Finset V):=he.symm ▸ (by simp);simpa using hh
  43     rcases hy with h|h
  44     · exact (hxy.ne h.symm).elim
  45     · exact h ▸ hxy
  46   · rw [ha] at he ⊢
  47     have hx:x=y∨x=c:=by have hh:x∈({y,c}:Finset V):=he.symm ▸ (by simp);simpa using hh
  48     rcases hx with h|h
  49     · exact (hxy.ne h).elim
  50     · exact h ▸ hxy.symm

  67 theorem marked_packing_distinct_hubs_disjoint
  68     (M N:Finset (MarkedWheel tail head ktail khead m π b H₀ Hₖ)) (u v:U) (huv:u≠v)
  69     (hM:∀ W∈M,W.2.val=u) (hN:∀ W∈N,W.2.val=v)
  70     (hpairs:∀ e∈markedPackingRims M,e.card=2)
  71     (hgraphs:Disjoint (markedPackingRimGraph M) (markedPackingRimGraph N)):
  72     Disjoint (M.biUnion markedWheelPhysicalResources) (N.biUnion markedWheelPhysicalResources):=by
  73   apply Finset.disjoint_left.mpr
  74   intro e he hf
  75   obtain ⟨W,hW,he⟩:=Finset.mem_biUnion.mp he
  76   obtain ⟨Z,hZ,hf⟩:=Finset.mem_biUnion.mp hf
  77   cases e with
  78   | inr e=>
  79     rw [markedWheelPhysicalResources_spoke] at he hf
  80     exact huv ((hM W hW).symm.trans (he.1.trans (hf.1.symm.trans (hN Z hZ))))
  81   | inl e=>
  82     rw [markedWheelPhysicalResources_rim] at he hf
  83     have hMe:e∈markedPackingRims M:=Finset.mem_biUnion.mpr ⟨W,hW,he⟩
  84     have hNe:e∈markedPackingRims N:=Finset.mem_biUnion.mpr ⟨Z,hZ,hf⟩
  85     obtain ⟨x,y,hxy,rfl⟩:=Finset.card_eq_two.mp (hpairs e hMe)
  86     exact disjoint_graphs_no_common_edge _ _ hgraphs x y ⟨hxy,hMe⟩ ⟨hxy,hNe⟩
