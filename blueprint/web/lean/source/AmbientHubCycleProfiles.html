  17 theorem embedded_class_image_iff (e : (Σ i,V i) ↪ A) (i : Bool) (z : Σ i,V i) :
  18     (∃ v : V i,e ⟨i,v⟩=e z) ↔ z.1=i := by
  19   constructor
  20   · rintro ⟨v,hv⟩
  21     exact (congrArg Sigma.fst (e.injective hv)).symm
  22   · rcases z with ⟨j,v⟩
  23     intro he
  24     change j=i at he
  25     subst j
  26     exact ⟨v,rfl⟩

  28 theorem hubMarkedCycle_supported (e : (Σ i,V i) ↪ A) (w : U)
  29     (hK : ∀ v,Hₖ v w ↔ ∃ z : V true,e ⟨true,z⟩=v)
  30     (hO : ∀ v,H₀ v w ↔ ∃ z : V false,e ⟨false,z⟩=v)
  31     (C : HubMarkedCycle tail head ktail khead m π b H₀ Hₖ w) :
  32     ∀ v∈hubMarkedCycleVertices w C,∃ z,e z=v := by
  33   intro v hv
  34   obtain ⟨i,_,rfl⟩ := Finset.mem_image.mp hv
  35   refine Fin.cases ?_ (fun i => ?_) i
  36   · obtain ⟨z,hz⟩ := (hK _).mp C.property.2.1

  38   · obtain ⟨z,hz⟩ := (hO _).mp (C.property.2.2 i)

  41 theorem ambient_hubMarkedCycle_profile (e : (Σ i,V i) ↪ A) (w : U)
  42     (hK : ∀ v,Hₖ v w ↔ ∃ z : V true,e ⟨true,z⟩=v)
  43     (hO : ∀ v,H₀ v w ↔ ∃ z : V false,e ⟨false,z⟩=v)
  44     (C : HubMarkedCycle tail head ktail khead m π b H₀ Hₖ w) (i : Bool) :
  45     (Finset.univ.filter (fun v : V i => e ⟨i,v⟩∈hubMarkedCycleVertices w C)).card=twoClassProfile m i := by
  46   have hs := hubMarkedCycle_supported e w hK hO C
  47   have hf (j : Fin (m+1)) : ∃ z,e z=C.val j := hs _ (Finset.mem_image.mpr ⟨j,Finset.mem_univ _,rfl⟩)
  48   let f (j : Fin (m+1)) : Σ i,V i := Classical.choose (hf j)
  49   have he (j) : e (f j)=C.val j := Classical.choose_spec (hf j)
  50   have hfi : Function.Injective f := by
  51     intro j k hjk
  52     apply permutationSeparated_injective π b C.val C.property.1.1
  53     rw [← he j,← he k,hjk]
  54   have hfK : (f 0).1=true := by
  55     apply (embedded_class_image_iff e true (f 0)).mp
  56     rw [he 0]
  57     exact (hK _).mp C.property.2.1
  58   have hfO (j : Fin m) : (f j.succ).1=false := by
  59     apply (embedded_class_image_iff e false (f j.succ)).mp
  60     rw [he j.succ]
  61     exact (hO _).mp (C.property.2.2 j)
  62   have hmem (v : Σ i,V i) : e v∈hubMarkedCycleVertices w C ↔ v∈Finset.univ.image f := by
  63     constructor
  64     · intro hv
  65       obtain ⟨j,_,hj⟩ := Finset.mem_image.mp hv
  66       exact Finset.mem_image.mpr ⟨j,Finset.mem_univ _,e.injective ((he j).trans hj)⟩
  67     · rintro hv
  68       obtain ⟨j,_,rfl⟩ := Finset.mem_image.mp hv
  69       exact Finset.mem_image.mpr ⟨j,Finset.mem_univ _,(he j).symm⟩
  70   have hh := twoClass_tuple_profile f hfi hfK hfO i
  71   simpa only [hmem] using hh
