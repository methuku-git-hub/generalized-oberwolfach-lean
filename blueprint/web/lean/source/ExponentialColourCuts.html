  14 theorem sampledColourCut_quadratic_exponential_event (k : ℕ) {q : ℝ} (hq : 0<q) :
  15     ∃ N : ℕ,1≤N ∧ ∀ (V : Type) [Fintype V] [LinearOrder V],N≤Fintype.card V →
  16     ∀ (G : SimpleGraph V) (ρ : Fin k→ℝ) (hρ : ∀ a,0≤ρ a) (hsum : ∑ a,ρ a=1),
  17     1-2*k*Real.exp (-(q^2/4)*Fintype.card V)≤
  18     (FiniteLaw.pi (fun _ : Sym2 V=>FiniteLaw.categoricalDirection ρ hρ hsum)).probability
  19       (fun σ=>∀ a S T,
  20         |((∑ x∈S,∑ y∈T,sampledColourOrientation G σ a x y:ℕ):ℝ)-
  21           ρ a/2*graphCut G S T|<q*(Fintype.card V:ℝ)^2) := by
  22   obtain ⟨N,hN1,hN⟩ := eventually_allCuts_failure_exponential (c:=q^2/2) (by positivity)
  23   refine ⟨N,hN1,?_⟩
  24   intro V _ _ hn G ρ hρ hsum
  25   let n := Fintype.card V
  26   have hnpos : 0<n := by dsimp [n];omega
  27   have hnreal : (0:ℝ)<n := by exact_mod_cast hnpos
  28   have hh := sampledColourCut_all_cuts G hnpos ρ hρ hsum
  29     (fun _=>q*(n:ℝ)^2) (fun _=>by positivity)
  30   have heq : -(q*(n:ℝ)^2)^2/(2*(n:ℝ)^2)=-(q^2/2)*(n:ℝ)^2 := by
  31     field_simp
  32   change 1-(4:ℝ)^n*(∑ _a : Fin k,2*Real.exp (-(q*(n:ℝ)^2)^2/(2*(n:ℝ)^2)))≤_ at hh
  33   simp only [heq,Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul] at hh
  34   have hs := hN n hn (2*k) (by positivity)
  35   have heq' : (4:ℝ)^n*((k:ℝ)*(2*Real.exp (-(q^2/2)*(n:ℝ)^2)))=
  36       2*k*(4:ℝ)^n*Real.exp (-(q^2/2)*(n:ℝ)^2) := by ring
  37   rw [heq'] at hh
  38   have he : (q^2/2)/2=q^2/4 := by ring
