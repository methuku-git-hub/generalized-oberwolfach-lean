  17 theorem endpoint_paired_absorber_count (Q F : SimpleGraph V) (S T : Finset V)
  18     (hST : Disjoint S T) (γ L : ℝ) (hγ : 0<γ) (hL : 0≤L)
  19     (hF : ∀ v,(F.degree v:ℝ)≤L)
  20     (hcommon : EndpointNeighbourCounts Q F S T (γ*Fintype.card V))
  21     (hbudget : 6*(L+1)≤γ/2*Fintype.card V)
  22     (hnb : 18≤ordinaryLocalCountCoeff 2 γ*Fintype.card V)
  23     (hnp : 48≤ordinaryLocalCountCoeff 2 γ*Fintype.card V)
  24     (x : Fin 3→V) (hx : Function.Injective x)
  25     (hxclass : ∀ i,x i∈endpointBaseClass S T i.val) :
  26     (ordinaryLocalCountCoeff 2 γ/2)^4*(Fintype.card V:ℝ)^9≤
  27       Fintype.card (LabelledCycleRecord (endpointTriangleVertices (Q:=Q) (F:=F) (S:=S) (T:=T))
  28         (pairedAbsorberTemplate 3).val (Finset.univ.image x)) := by
  29   let n : ℝ := Fintype.card V
  30   let a := γ/2*n
  31   let β := ordinaryLocalCountCoeff 2 γ
  32   let W := Finset.univ.image x
  33   have hW : W.card=3 := by rw [Finset.card_image_of_injective _ hx];simp
  34   let Base := fun f : Fin 3→V=>ClassSparseGreedyGood Q F (endpointBaseClass S T) W allPreviousLabels f
  35   let Path := fun y : Fin 3→V=>fun i : ℕ=>fun f : Fin 2→V=>
  36     ∃ hi : i<3,ClassSparseGreedyGood Q F (endpointPathClass S T (decide (i=0))) {x ⟨i,hi⟩,y ⟨i,hi⟩}
  37       (endpointPathRequirements (x ⟨i,hi⟩) (y ⟨i,hi⟩)) f
  38   have ha : 0≤a := by dsimp [a,n];positivity
  39   have hbase0 := endpoint_base_count Q F S T W hST L (γ*n) a hL ha hF hcommon (by
  40     rw [hW];dsimp [a];norm_num;linarith only [hbudget])
  41   have hbase : β*n^3≤Fintype.card {f // Base f} := by
  42     have hh := mul_le_mul_of_nonneg_right (min_le_left ((γ/2)^3) ((γ/2)^2))
  43       (pow_nonneg (show 0≤n by dsimp [n];positivity) 3)
  44     have he : a^3=(γ/2)^3*n^3 := mul_pow _ _ _
  45     exact hh.trans (he ▸ hbase0)
  46   have hpath (y : Fin 3→V) (hy : Base y) (i : ℕ) (hi : i<3) :
  47       β*n^2≤Fintype.card {f // Path y i f} := by
  48     let j : Fin 3 := ⟨i,hi⟩
  49     have hxy : x j≠y j := fun h=>(hy.2.1 j).2 (Finset.mem_image.mpr ⟨j,Finset.mem_univ _,h⟩)
  50     have hFxy : ¬F.Adj (x j) (y j) := fun h=>hy.2.2.2.1 j (x j)
  51       (Finset.mem_image.mpr ⟨j,Finset.mem_univ _,rfl⟩) h.symm
  52     have hxj : x j∈if decide (i=0) then T else S := by simpa [endpointBaseClass,j] using hxclass j
  53     have hyj : y j∈if decide (i=0) then T else S := by simpa [endpointBaseClass,j] using (hy.2.1 j).1
  54     have hh := endpoint_path_count Q F S T L (γ*n) a hL ha hF hcommon
  55       (by dsimp [a];nlinarith only [hbudget,hL]) (decide (i=0)) (x j) (y j) hxy hFxy hxj hyj
  56     have hβ := mul_le_mul_of_nonneg_right (min_le_right ((γ/2)^3) ((γ/2)^2))
  57       (pow_nonneg (show 0≤n by dsimp [n];positivity) 2)
  58     have he : a^2=(γ/2)^2*n^2 := mul_pow _ _ _
  59     have hc := hβ.trans (he ▸ hh)
  60     have hiff (f : Fin 2→V) : Path y i f ↔ ClassSparseGreedyGood Q F
  61         (endpointPathClass S T (decide (i=0))) {x j,y j} (endpointPathRequirements (x j) (y j)) f := by
  62       constructor
  63       · rintro ⟨hi',hf⟩;exact hf
  64       · intro hf;exact ⟨hi,hf⟩
  65     exact hc.trans_eq (by exact_mod_cast (Fintype.card_congr (Equiv.subtypeEquivRight hiff)).symm)
  66   apply paired_absorber_abundance (endpointTriangleVertices (Q:=Q) (F:=F) (S:=S) (T:=T)) (m:=2) (by norm_num)
  67     x hx Base Path β (ordinaryLocalCountCoeff_pos 2 γ hγ)
  68     (fun y hy=>hy.1) (fun y hy=>endpoint_base_tuple Q F S T W hST y hy)
  69     (fun y hy _ i f hf=>?_) (fun y _ _ i _ f hf=>?_) hbase
  70     (fun y hy _ i hi=>by convert hpath y hy i hi using 1 <;> congr!) (by norm_num at *;exact hnb) (by norm_num at *;exact hnp)
  71   · obtain ⟨hi,hf⟩ := hf
  72     have hxj : x i∈if decide (i.val=0) then T else S := by simpa [endpointBaseClass] using hxclass i
  73     have hyj : y i∈if decide (i.val=0) then T else S := by simpa [endpointBaseClass] using (hy.2.1 i).1
  74     exact ⟨endpoint_path_triangle Q F S T hST _ (x i) (y i) (y i) (by simp) hyj f hf,
  75       endpoint_path_triangle Q F S T hST _ (x i) (y i) (x i) (by simp) hxj f hf⟩
  76   · obtain ⟨hi,hf⟩ := hf
  77     exact hf.1
