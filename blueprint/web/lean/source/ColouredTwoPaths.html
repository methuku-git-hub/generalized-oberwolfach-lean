  14 theorem finiteLaw_fixed_size_patterns_tail_at_scale
  15     {E I : Type*} [Fintype E] [DecidableEq E]
  16     {State : E → Type*} [∀ e,Fintype (State e)]
  17     (μ : ∀ e,FiniteLaw (State e)) (C : Finset I) (F : I → Finset E)
  18     (s : I → ∀ e,State e) (k : ℕ) (hk : 0<k)
  19     (hsize : ∀ i∈C,(F i).card=k)
  20     (hdisj : ∀ i∈C,∀ j∈C,i≠j → Disjoint (F i) (F j))
  21     {n t : ℝ} (hcard : (C.card:ℝ)≤n) (ht : 0≤t) :
  22     (FiniteLaw.pi μ).probability (fun σ =>
  23       t≤|(∑ i∈C,patternValue (F i) (s i) σ)-(∑ i∈C,∏ e∈F i,(μ e).mass (s i e))|)≤
  24       2*Real.exp (-2*t^2/(n*k)) := by
  25   classical
  26   by_cases hC : C=∅
  27   · subst C
  28     by_cases ht0 : t=0
  29     · subst t;norm_num [FiniteLaw.probability,(FiniteLaw.pi μ).total]
  30     · have htp : 0<t := lt_of_le_of_ne ht (Ne.symm ht0)
  31       simp only [Finset.sum_empty,sub_self,abs_zero,not_le.mpr htp]
  32       rw [finiteLaw_probability_false]
  33       positivity
  34   · have hh := finiteLaw_disjoint_patterns_tail μ C F s hdisj ht
  35     have hsum : (∑ i∈C,((F i).card:ℝ))=(C.card:ℝ)*k := by
  36       simp_rw [Finset.sum_congr rfl (fun i hi => congrArg (fun j : ℕ => (j:ℝ)) (hsize i hi))]
  37       simp
  38     rw [hsum] at hh
  39     apply hh.trans
  40     apply mul_le_mul_of_nonneg_left _ (by norm_num)
  41     apply Real.exp_le_exp.mpr
  42     have hp : (0:ℝ)<C.card := by exact_mod_cast Finset.card_pos.mpr (Finset.nonempty_iff_ne_empty.mpr hC)
  43     have hkp : (0:ℝ)<k := Nat.cast_pos.mpr hk
  44     have hden := mul_le_mul_of_nonneg_right hcard hkp.le
  45     have hd := div_le_div_of_nonneg_left (show 0≤(2:ℝ)*t^2 by positivity) (mul_pos hp hkp) hden
  46     simpa only [neg_div,neg_mul] using neg_le_neg hd

  48 theorem common_two_path_edges_disjoint {V : Type*} [Fintype V] [DecidableEq V]
  49     (G : SimpleGraph V) (u v : V) (huv : u≠v) :
  50     ∀ x∈G.neighborFinset u∩G.neighborFinset v,
  51     ∀ y∈G.neighborFinset u∩G.neighborFinset v,x≠y →
  52       Disjoint (twoPathEdges u v x) (twoPathEdges u v y) := by
  53   intro x hx y hy hxy
  54   have hux : u≠x := ((G.mem_neighborFinset _ _).mp (Finset.mem_inter.mp hx).1).ne
  55   have hvx : v≠x := ((G.mem_neighborFinset _ _).mp (Finset.mem_inter.mp hx).2).ne
  56   have huy : u≠y := ((G.mem_neighborFinset _ _).mp (Finset.mem_inter.mp hy).1).ne
  57   have hvy : v≠y := ((G.mem_neighborFinset _ _).mp (Finset.mem_inter.mp hy).2).ne
  58   apply Finset.disjoint_left.mpr
  59   intro e he hey
  60   simp only [twoPathEdges,Finset.mem_insert,Finset.mem_singleton] at he hey
  61   rcases he with rfl | rfl <;> rcases hey with he | he <;>
  62     rw [Sym2.eq_iff] at he <;> aesop

  64 def colouredTwoPathCount {V : Type*} [Fintype V] [DecidableEq V] {R : ℕ}
  65     (G : SimpleGraph V) (σ : Sym2 V → Fin R) (u v : V) (a b : Fin R) : ℕ :=
  66   ((G.neighborFinset u∩G.neighborFinset v).filter
  67     (fun w => σ s(u,w)=a ∧ σ s(w,v)=b)).card

  69 theorem colouredTwoPathCount_patterns {V : Type*} [Fintype V] [DecidableEq V] {R : ℕ}
  70     (G : SimpleGraph V) (σ : Sym2 V → Fin R) (u v : V) (huv : u≠v) (a b : Fin R) :
  71     (colouredTwoPathCount G σ u v a b:ℝ)=
  72       ∑ w∈G.neighborFinset u∩G.neighborFinset v,
  73         patternValue (twoPathEdges u v w) (fun e => if e=s(u,w) then a else b) σ := by
  74   unfold colouredTwoPathCount
  75   rw [Finset.card_filter,Nat.cast_sum]
  76   apply Finset.sum_congr rfl
  77   intro w _
  78   have he : s(u,w)≠s(w,v) := by
  79     intro h;rcases Sym2.eq_iff.mp h with ⟨h1,h2⟩ | ⟨h1,h2⟩ <;> exact huv (by aesop)
  80   simp only [patternValue,twoPathEdges,Finset.prod_insert,Finset.prod_singleton,Finset.mem_singleton,he,Ne.symm he,not_false_eq_true,ite_true,ite_false,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  81   split_ifs <;> simp_all

  83 theorem colouredTwoPathCount_tail {V : Type*} [Fintype V] [DecidableEq V] {R : ℕ}
  84     (G : SimpleGraph V) (ρ : Fin R → ℝ) (hρ : ∀ a,0≤ρ a) (hsum : ∑ a,ρ a=1)
  85     (u v : V) (huv : u≠v) (a b : Fin R) {t : ℝ} (ht : 0≤t) :
  86     (FiniteLaw.pi (fun _ : Sym2 V => FiniteLaw.categorical ρ hρ hsum)).probability
  87       (fun σ => t≤|(colouredTwoPathCount G σ u v a b:ℝ)-
  88         ((G.neighborFinset u∩G.neighborFinset v).card:ℝ)*ρ a*ρ b|)≤
  89       2*Real.exp (-t^2/Fintype.card V) := by
  90   let C := G.neighborFinset u∩G.neighborFinset v
  91   have he (w : V) : s(u,w)≠s(w,v) := by
  92     intro h;rcases Sym2.eq_iff.mp h with ⟨h1,h2⟩ | ⟨h1,h2⟩ <;> exact huv (by aesop)
  93   have hsize (w : V) (_ : w∈C) : (twoPathEdges u v w).card=2 := by
  94     simp [twoPathEdges,he w]
  95   have hh := finiteLaw_fixed_size_patterns_tail_at_scale
  96     (fun _ : Sym2 V => FiniteLaw.categorical ρ hρ hsum) C (twoPathEdges u v)
  97     (fun w e => if e=s(u,w) then a else b) 2 (by omega) hsize
  98     (common_two_path_edges_disjoint G u v huv)
  99     (n:=(Fintype.card V:ℝ)) (by exact_mod_cast Finset.card_le_univ C) ht
 100   have hmean : (∑ w∈C,∏ e∈twoPathEdges u v w,
 101       (FiniteLaw.categorical ρ hρ hsum).mass (if e=s(u,w) then a else b))=
 102       (C.card:ℝ)*ρ a*ρ b := by
 103     simp only [FiniteLaw.categorical]
 104     have hprod (w : V) : (∏ e∈twoPathEdges u v w,ρ (if e=s(u,w) then a else b))=ρ a*ρ b := by
 105       simp [twoPathEdges,he w,Ne.symm (he w)]
 106     simp only [hprod,Finset.sum_const,nsmul_eq_mul,mul_assoc]
 107   rw [hmean] at hh
 108   dsimp only [C] at hh
 109   have hexp : -2*t^2/((Fintype.card V:ℝ)*2) = -t^2/Fintype.card V := by ring
 110   simpa only [← colouredTwoPathCount_patterns G _ u v huv a b,hexp,Nat.cast_ofNat] using hh
