  22     canonicalMarkedRotation k (Fin.last m)=k:=by

  41  theorem marked_cycle_from_cyclic_embedding {V E K:Type*}
  42     (tail head:E→V) (ktail khead:K→V) (m:ℕ) (π:Equiv.Perm V) (b:ℕ)
  43     (z:Fin (m+1)↪V) (k:Fin (m+1))
  44     (hsep:PermutationSeparated π b z)
  45     (hO:∀ i,i≠k→∃ e,tail e=z i∧head e=z (finRotate (m+1) i))
  46     (hK:∃ e,ktail e=z k∧khead e=z (finRotate (m+1) k)):
  47     ∃ C:MarkedCycleTuple tail head ktail khead m π b,
  48       ∀ i,C.val i=z (canonicalMarkedRotation k i):=by
  49   let ρ:=canonicalMarkedRotation k
  50   let C:Fin (m+1)→V:=fun i=>z (ρ i)
  51   refine ⟨⟨C,?_,?_,?_⟩,fun _=>rfl⟩
  52   · intro i j hij
  53     exact hsep (ρ i) (ρ j) (fun h=>hij (ρ.injective h))
  54   · intro i
  55     obtain ⟨e,he₁,he₂⟩:=hO (ρ i.castSucc) (canonicalMarkedRotation_ordinary k i)
  56     refine ⟨e,he₁,?_⟩

  59   · obtain ⟨e,he₁,he₂⟩:=hK
  60     refine ⟨e,?_,?_⟩
  61     · simpa only [C,ρ,canonicalMarkedRotation_last] using he₁
  62     · simpa only [C,ρ,canonicalMarkedRotation_zero] using he₂
