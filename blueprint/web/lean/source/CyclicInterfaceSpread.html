  13 theorem cyclic_interfaces_fractional_weighting_bounded {k : ℕ} {X Y : Type*}
  14     [Fintype X] [Fintype Y] [Nonempty X]
  15     (L : Fin (k + 2) → X → Y → Prop) (R : Fin (k + 2) → Y → X → Prop)
  16     (δ α : ℝ) (hn : 0 < Fintype.card Y) (hα : 0 < α)
  17     (hcard : Fintype.card X = Fintype.card Y)
  18     (hthreshold : ((k + 2) + (2 * (k + 2) - 1) * α) * Fintype.card Y ≤
  19       (2 * (k + 2) - 1) * δ)
  20     (hdL : ∀ i x, δ ≤ (Finset.univ.filter (L i x)).card)
  21     (hdY : ∀ i y, δ ≤ (Finset.univ.filter (fun x => L i x y)).card)
  22     (hY : ∀ i y, (Finset.univ.filter (fun x => L i x y)).card =
  23       (Finset.univ.filter (R i y)).card)
  24     (hX : ∀ i z, (Finset.univ.filter (fun y => R i y z)).card =
  25       (Finset.univ.filter (L (finRotate (k + 2) i) z)).card) :
  26     ∃ ω : (Fin (k + 2) → X × Y × X) → ℝ,
  27       (∀ p, 0 ≤ ω p) ∧
  28       (∑ p, ω p) = (∑ x, cyclicInterfaceDegree L 0 x) ∧
  29       (∀ i x y, (∑ p, if ((p i).1, (p i).2.1) = (x, y) then ω p else 0) =
  30         interfaceCapacity (L i) x y) ∧
  31       (∀ i y z, (∑ p, if ((p i).2.1, (p i).2.2) = (y, z) then ω p else 0) =
  32         interfaceCapacity (R i) y z) ∧
  33       (∀ p, 0 < ω p → ∀ i, (p i).2.2 = (p (finRotate (k + 2) i)).1) ∧
  34       (∀ p, (∀ i, (p i).2.2 = (p (finRotate (k + 2) i)).1) →
  35         (∀ i, L i (p i).1 (p i).2.1 ∧ R i (p i).2.1 (p i).2.2) →
  36         ((k + 2 : ℝ) * cyclicSectorLower k α) / (Fintype.card Y : ℝ) ^ (2 * k + 2) ≤ ω p ∧
  37         ω p ≤ ((k + 2 : ℝ) * cyclicSectorUpper k α) /
  38           (Fintype.card Y : ℝ) ^ (2 * k + 2)) := by
  39   have hk : (2 : ℝ) ≤ k + 2 := by linarith [Nat.cast_nonneg (α := ℝ) k]
  40   have hn' : (0 : ℝ) < Fintype.card Y := by exact_mod_cast hn
  41   obtain ⟨_, hbase⟩ := positive_margin_base hk hn' hα hthreshold
  42   obtain ⟨Q, ω, hω, ht, hl, hr, hc, hf, hformula⟩ :=
  43     cyclic_interfaces_fractional_weighting_with_formula L R δ hn hbase hdL hY hX
  44   refine ⟨ω, hω, ht, hl, hr, hc, ?_⟩
  45   intro p hp hpath
  46   rw [hformula p hp]
  47   have hb (c : Fin (k + 2)) := cyclicSectorWeight_bounds L R δ α hn hα hcard
  48     hthreshold hdL hdY hY hX Q hf p hp hpath c
  49   have hlo := Finset.sum_le_sum (s := Finset.univ) (fun c _ => (hb c).1)
  50   have hhi := Finset.sum_le_sum (s := Finset.univ) (fun c _ => (hb c).2)
  51   simpa only [Finset.sum_const, Finset.card_univ, Fintype.card_fin, nsmul_eq_mul,
  52     Nat.cast_add, Nat.cast_ofNat, ← mul_div_assoc] using And.intro hlo hhi
