   8 theorem second_order_perturbation {X : Type*} [NormedAddCommGroup X] [NormedSpace ℝ X]
   9     (S : Submodule ℝ X) (T K : X →ₗ[ℝ] X)
  10     (a b c : ℝ) (ha : 0≤a) (hb : 0≤b)
  11     (hT : ∀ z,z∈S → T z∈S)
  12     (hL : ∀ z,z∈S → ‖T z-z-K z‖≤a*‖z‖)
  13     (hK : ∀ z,‖K z‖≤b*‖z‖)
  14     (hK2 : ∀ z,z∈S → ‖K (K z)‖≤c*‖z‖) (z : X) (hz : z∈S) :
  15     ‖z-T z‖≤(a+b)*‖z‖ ∧
  16       ‖(z-T z)-T (z-T z)‖≤(c+b*a+a*(a+b))*‖z‖ := by
  17   let R (x : X) := T x-x-K x
  18   have hE (x : X) (hx : x∈S) : ‖x-T x‖≤(a+b)*‖x‖ := by
  19     have he : x-T x=-(K x+R x) := by dsimp [R];abel
  20     rw [he,norm_neg]
  21     calc
  22       _ ≤ ‖K x‖+‖R x‖ := norm_add_le _ _
  23       _ ≤ b*‖x‖+a*‖x‖ := add_le_add (hK x) (hL x hx)
  24       _ = _ := by ring
  25   refine ⟨hE z hz,?_⟩
  26   have hzE : z-T z∈S := S.sub_mem hz (hT z hz)
  27   have he2 : (z-T z)-T (z-T z)=K (K z)+K (R z)-R (z-T z) := by
  28     dsimp [R]
  29     simp only [map_sub]
  30     abel
  31   rw [he2]
  32   calc
  33     _ ≤ ‖K (K z)+K (R z)‖+‖R (z-T z)‖ := norm_sub_le _ _
  34     _ ≤ (‖K (K z)‖+‖K (R z)‖)+‖R (z-T z)‖ := add_le_add (norm_add_le _ _) le_rfl
  35     _ ≤ (c*‖z‖+b*‖R z‖)+a*‖z-T z‖ := add_le_add (add_le_add (hK2 z hz) (hK (R z))) (hL _ hzE)
  36     _ ≤ (c*‖z‖+b*(a*‖z‖))+a*((a+b)*‖z‖) := by
  37       gcongr
  38       · exact hL z hz
  39       · exact hE z hz
  40     _ = _ := by ring
  41 
  42 end Oberwolfach
