  13 def bipAdj (H : A → B → Prop) (x : A) (u : B) : ℝ := if H x u then 1 else 0

  15 def bipColumnDegree (H : A → B → Prop) (u : B) : ℝ := ∑ x,bipAdj H x u

  17 def bipPairIntersection (H : A → B → Prop) (u v : B) : ℝ :=
  18   ∑ x,bipAdj H x u*bipAdj H x v

  21   ∑ x,bipAdj H x u*bipAdj H x v*bipAdj H x t
