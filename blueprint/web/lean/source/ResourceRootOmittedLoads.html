  20 theorem root_omitted_indicator_change (R : I → Finset A) (w : I → ℝ)
  21     (hw : ∀ i,0≤w i) (keep keep' : A → Prop) (e : A) (i : I) :
  22     |(if e∈R i ∧ (∀ f∈R i,f≠e → keep' f) then w i else 0)-
  23       (if e∈R i ∧ (∀ f∈R i,f≠e → keep f) then w i else 0)|≤
  24     ∑ f,if ¬(keep f↔keep' f) ∧ e∈R i ∧ f∈R i ∧ f≠e then w i else 0 := by
  25   have hnon : 0≤∑ f,if ¬(keep f↔keep' f) ∧ e∈R i ∧ f∈R i ∧ f≠e then w i else 0 :=
  26     Finset.sum_nonneg (fun f _=>by split_ifs <;> first | exact hw i | rfl)
  27   by_cases he : e∈R i
  28   swap
  29   · simpa only [he,false_and,ite_false,sub_self,abs_zero] using hnon
  30   have hcost (f : A) (hf : f∈R i) (hne : f≠e) (hchange : ¬(keep f↔keep' f)) :
  31       w i≤∑ f,if ¬(keep f↔keep' f) ∧ e∈R i ∧ f∈R i ∧ f≠e then w i else 0 := by
  32     have hh := Finset.single_le_sum (s:=Finset.univ)
  33       (f:=fun f=>if ¬(keep f↔keep' f) ∧ e∈R i ∧ f∈R i ∧ f≠e then w i else 0)
  34       (fun f _=>by split_ifs <;> first | exact hw i | rfl) (Finset.mem_univ f)
  35     rw [if_pos ⟨hchange,he,hf,hne⟩] at hh
  36     exact hh
  37   by_cases hk : ∀ f∈R i,f≠e → keep f
  38   · by_cases hk' : ∀ f∈R i,f≠e → keep' f
  39     · rw [if_pos ⟨he,hk'⟩,if_pos ⟨he,hk⟩,sub_self,abs_zero]
  40       exact hnon
  41     · have hb := hk'
  42       push_neg at hb
  43       obtain ⟨f,hf,hne,hf'⟩ := hb
  44       have hh := hcost f hf hne (fun h=>hf' (h.mp (hk f hf hne)))
  45       rw [if_neg (fun h=>hk' h.2),if_pos ⟨he,hk⟩,zero_sub,abs_neg,abs_of_nonneg (hw i)]
  46       exact hh
  47   · by_cases hk' : ∀ f∈R i,f≠e → keep' f
  48     · have hb := hk
  49       push_neg at hb
  50       obtain ⟨f,hf,hne,hf'⟩ := hb
  51       have hh := hcost f hf hne (fun h=>hf' (h.mpr (hk' f hf hne)))
  52       rw [if_pos ⟨he,hk'⟩,if_neg (fun h=>hk h.2),sub_zero,abs_of_nonneg (hw i)]
  53       exact hh
  54     · rw [if_neg (fun h=>hk' h.2),if_neg (fun h=>hk h.2),sub_self,abs_zero]
  55       exact hnon

  58     (hw : ∀ i,0≤w i) (keep keep' : A → Prop) (e : A) :
  59     |resourceRootOmittedLoad R w keep' e-resourceRootOmittedLoad R w keep e|≤
  60       ∑ f,if ¬(keep f↔keep' f) then resourceOmittedPairWeight R w e f else 0 := by
  61   unfold resourceRootOmittedLoad

  64     _ ≤ ∑ i,|(if e∈R i ∧ (∀ f∈R i,f≠e → keep' f) then w i else 0)-
  65       (if e∈R i ∧ (∀ f∈R i,f≠e → keep f) then w i else 0)| := Finset.abs_sum_le_sum_abs _ _
  66     _ ≤ ∑ i,∑ f,if ¬(keep f↔keep' f) ∧ e∈R i ∧ f∈R i ∧ f≠e then w i else 0 :=
