  12 def averagedWeight {Ω T : Type*} [Fintype Ω]
  13     (μ : FiniteLaw Ω) (w : Ω → T → ℝ) (ρ : ℝ) (t : T) : ℝ :=
  14   μ.expectation (fun ω => w ω t) / ρ

  16 theorem averagedWeight_nonneg {Ω T : Type*} [Fintype Ω]
  17     (μ : FiniteLaw Ω) (w : Ω → T → ℝ) {ρ : ℝ} (hρ : 0 ≤ ρ)
  18     (hw : ∀ ω t, 0 ≤ w ω t) (t : T) : 0 ≤ averagedWeight μ w ρ t := by
  19   apply div_nonneg _ hρ
  20   exact Finset.sum_nonneg (fun ω _ => mul_nonneg (μ.nonneg ω) (hw ω t))

  22 theorem averagedWeight_incidence {Ω T : Type*} [Fintype Ω] [Fintype T]
  23     (μ : FiniteLaw Ω) (w : Ω → T → ℝ) (ρ : ℝ) (I : T → Prop) :
  24     (∑ t, if I t then averagedWeight μ w ρ t else 0) =
  25       μ.expectation (fun ω => ∑ t, if I t then w ω t else 0) / ρ := by
  26   unfold averagedWeight FiniteLaw.expectation
  27   have hterm (t : T) : (if I t then (∑ ω, μ.mass ω * w ω t) / ρ else 0) =
  28       (∑ ω, μ.mass ω * (if I t then w ω t else 0)) / ρ := by
  29     by_cases hi : I t <;> simp [hi]
  30   simp_rw [hterm]
  31   rw [← Finset.sum_div, Finset.sum_comm]
  32   simp only [Finset.mul_sum]

  34 theorem averagedWeight_exact_load {Ω T : Type*} [Fintype Ω] [Fintype T]
  35     (μ : FiniteLaw Ω) (w : Ω → T → ℝ) (ρ : ℝ)
  36     (I : T → Prop) (A : Ω → Prop) (k : ℝ)
  37     (hload : ∀ ω, (∑ t, if I t then w ω t else 0) = if A ω then k else 0) :
  38     (∑ t, if I t then averagedWeight μ w ρ t else 0) = k * μ.probability A / ρ := by
  39   rw [averagedWeight_incidence]
  40   simp_rw [hload]
  41   unfold FiniteLaw.expectation FiniteLaw.probability
  42   congr 1
  43   rw [Finset.mul_sum]
  44   apply Finset.sum_congr rfl
  45   intro ω _
  46   by_cases h : A ω <;> simp [h, mul_comm]

  48 theorem averagedWeight_load_interval {Ω T : Type*} [Fintype Ω] [Fintype T]
  49     (μ : FiniteLaw Ω) (w : Ω → T → ℝ) {ρ δ : ℝ} (hρ : 0 < ρ)
  50     (I : T → Prop) (A : Ω → Prop)
  51     (hload : ∀ ω, (∑ t, if I t then w ω t else 0) = if A ω then 1 else 0)
  52     (hp : (1 - δ) * ρ ≤ μ.probability A ∧ μ.probability A ≤ ρ) :
  53     1 - δ ≤ (∑ t, if I t then averagedWeight μ w ρ t else 0) ∧
  54       (∑ t, if I t then averagedWeight μ w ρ t else 0) ≤ 1 := by
  55   rw [averagedWeight_exact_load μ w ρ I A 1 hload, one_mul]
  56   exact ⟨(le_div_iff₀ hρ).mpr hp.1, (div_le_one hρ).mpr hp.2⟩

  58 theorem averagedWeight_pointwise_bounds {Ω T : Type*} [Fintype Ω]
  59     (μ : FiniteLaw Ω) (w : Ω → T → ℝ) {ρ a C : ℝ} (hρ : 0 < ρ)
  60     (S : Ω → T → Prop) (hw : ∀ ω t, 0 ≤ w ω t)
  61     (hlo : ∀ ω t, S ω t → a ≤ w ω t) (hhi : ∀ ω t, w ω t ≤ C) (t : T) :
  62     a * μ.probability (fun ω => S ω t) / ρ ≤ averagedWeight μ w ρ t ∧
  63       averagedWeight μ w ρ t ≤ C / ρ := by
  64   have hlow := finiteLaw_expectation_mono μ
  65     (fun ω => if S ω t then a else 0) (fun ω => w ω t)
  66     (by intro ω; split_ifs with h; exact hlo ω t h; exact hw ω t)
  67   have hhigh := finiteLaw_expectation_mono μ (fun ω => w ω t) (fun _ => C)
  68     (fun ω => hhi ω t)
  69   rw [finiteLaw_expectation_const] at hhigh
  70   have hex : μ.expectation (fun ω => if S ω t then a else 0) =
  71       a * μ.probability (fun ω => S ω t) := by
  72     unfold FiniteLaw.expectation FiniteLaw.probability
  73     rw [Finset.mul_sum]
  74     apply Finset.sum_congr rfl
  75     intro ω _
  76     by_cases h : S ω t <;> simp [h, mul_comm]
  77   rw [hex] at hlow
  78   exact ⟨div_le_div_of_nonneg_right hlow hρ.le,
  79     div_le_div_of_nonneg_right hhigh hρ.le⟩
