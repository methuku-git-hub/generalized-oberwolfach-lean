  34   hubRoots:∀ s,u s∈B∧u' s∈B

  45   hubBalance:∀ z,Fintype.card {s:S // u s=z∧js s=false}=m*Fintype.card {s:S // u s=z∧js s=true}
  46   sourceRimInjective:Function.Injective (fun e=>if jr e then ({src e,π (dst e)}:Finset V) else {src e,dst e})
  47   imageRimInjective:Function.Injective (fun e=>if jr e then ({φ (src e),π (φ (dst e))}:Finset V) else {φ (src e),φ (dst e)})
