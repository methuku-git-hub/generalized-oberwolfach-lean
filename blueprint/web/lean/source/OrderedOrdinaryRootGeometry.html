  14 theorem orderedOrdinary_rim_endpoints (W:OrderedOrdinaryWheel (U:=U) π b r) (hr:3≤r) (x y:V)
  15     (h:Sum.inl (x,y)∈orderedOrdinaryResources W):
  16     x≠y ∧ x∈Set.range W.1.val ∧ y∈Set.range W.1.val:=by
  17   obtain ⟨i,hi⟩:=(orderedOrdinaryResources_rim W (x,y)).mp h
  18   have hx:W.1.val i=x:=congrArg Prod.fst hi
  19   have hy:W.1.val (cycleNext i)=y:=congrArg Prod.snd hi
  20   refine ⟨?_,⟨i,hx⟩,⟨cycleNext i,hy⟩⟩
  21   intro heq
  22   apply cycleNext_ne_self hr i
  23   exact (permutationSeparated_injective π b W.1.val W.1.property) (by rw [hx,hy,heq]) |>.symm

  31 theorem orderedOrdinary_distinct_rims_physical (W:OrderedOrdinaryWheel (U:=U) π b r) (hr:3≤r)
  32     (x y a c:V) (hxy:Sum.inl (x,y)∈orderedOrdinaryResources W)
  33     (hac:Sum.inl (a,c)∈orderedOrdinaryResources W) (hne:(x,y)≠(a,c)):
  34     s(x,y)≠s(a,c):=by
  35   obtain ⟨i,hi⟩:=(orderedOrdinaryResources_rim W (x,y)).mp hxy
  36   obtain ⟨j,hj⟩:=(orderedOrdinaryResources_rim W (a,c)).mp hac
  37   intro he
  38   have he':s(W.1.val i,W.1.val (cycleNext i))=s(W.1.val j,W.1.val (cycleNext j)):=by
  39     have hx:=congrArg Prod.fst hi
  40     have hy:=congrArg Prod.snd hi
  41     have ha:=congrArg Prod.fst hj
  42     have hc:=congrArg Prod.snd hj
  43     simp only [Prod.fst,Prod.snd] at hx hy ha hc
  44     rw [hx,hy,ha,hc]
  45     exact he
  46   have hij:=orderedOrdinaryPhysical_edge_injective hr W he'
  47   apply hne
  48   exact hi.symm.trans (hij ▸ hj)
