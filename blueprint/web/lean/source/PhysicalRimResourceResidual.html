  56     (φ:W↪Sym2 V⊕(V×U)) (covered:Finset W) (x:V):
  57     (physicalRimResidualGraph φ covered).degree x≤(rimIncidentResources φ x\covered).card := by
  58   let G:=physicalRimResidualGraph φ covered
  59   let f : {y:V // G.Adj x y} → {w:W // w∈rimIncidentResources φ x\covered} := fun y=>
  60     ⟨Classical.choose y.property.2,by
  61       have hw:=Classical.choose_spec y.property.2
  62       apply Finset.mem_sdiff.mpr
  63       refine ⟨Finset.mem_filter.mpr ⟨Finset.mem_univ _,⟨y.val,Ne.symm y.property.1,hw.2⟩⟩,hw.1⟩⟩
  64   have hf:Function.Injective f := by
  65     intro a b hab
  66     have heq:φ (f a).val=φ (f b).val:=congrArg φ (congrArg Subtype.val hab)
  67     have ha:= (Classical.choose_spec a.property.2).2
  68     have hb:= (Classical.choose_spec b.property.2).2
  69     have hp:s(x,a.val)=s(x,b.val):=Sum.inl.inj (ha.symm.trans (heq.trans hb))
  70     rcases Sym2.eq_iff.mp hp with ⟨_,h⟩|⟨h,_⟩
  71     · exact Subtype.ext h
  72     · exact (b.property.1 h).elim
  73   have hh:=Fintype.card_le_of_injective f hf
  74   have hh':Fintype.card (G.neighborSet x)≤(rimIncidentResources φ x\covered).card := by
  75     convert hh using 1 <;> congr! 4
  76     simp only [Fintype.card_subtype,Finset.sdiff_eq_filter]
  77     congr 1
  78     ext w
  79     simp

  83     (φ:W↪Sym2 V⊕(V×U)) (x:V):
  84     (rimIncidentResources φ x).card≤Fintype.card V := by

  89     apply Subtype.ext

  92     have hb:φ b.val=Sum.inl s(x,f b):=(Classical.choose_spec ((Finset.mem_filter.mp b.property).2)).2
  93     rw [ha,hb,hab]
  94   have hh:=Fintype.card_le_of_injective f hf
