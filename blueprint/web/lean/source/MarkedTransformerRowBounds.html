  16  theorem marked_row_density_product_lower (S:Finset (Bool×Bool×V)) (J:Finset (Bool×U))
  17     (p p₀ pK:ℝ) (s:Bool→U→ℝ) (hp:0<p) (hp1:p≤1)
  18     (hp₀:p≤p₀) (hpK:p≤pK) (hs:∀ j u,p≤s j u)
  19     (hS:S.card≤2) (hJ:J.card≤2):
  20     p^4≤(∏ a∈S,if a.1 then pK else p₀)*(∏ a∈J,s a.1 a.2):=by
  21   have h₁:p^S.card≤∏ a∈S,if a.1 then pK else p₀:=by
  22     calc
  23       _ = ∏ _a∈S,p:=by simp
  24       _ ≤ _:=Finset.prod_le_prod₀ (fun _ _=>hp.le) (by intro a ha;cases a.1 <;> simp [hp₀,hpK])
  25   have h₂:p^J.card≤∏ a∈J,s a.1 a.2:=by
  26     calc
  27       _ = ∏ _a∈J,p:=by simp
  28       _ ≤ _:=Finset.prod_le_prod₀ (fun _ _=>hp.le) (fun a _=>hs a.1 a.2)
  29   calc
  30     _ ≤ p^(S.card+J.card):=pow_le_pow_of_le_one hp.le hp1 (by omega)
  31     _ = p^S.card*p^J.card:=pow_add _ _ _
  32     _ ≤ _:=mul_le_mul h₁ h₂ (pow_nonneg hp.le _) ((pow_nonneg hp.le _).trans h₁)
