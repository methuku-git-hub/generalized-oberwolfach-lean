  28     (e f : E) (j : Fin (k-(R f).card)) :
  29     Sum.inr ⟨f,j⟩∈uniformPaddingIncidence R k e ↔ e=f := by

  43 theorem uniformPadding_card {V E : Type*} (R : E → Finset V) (k : ℕ)
  44     (hsize : ∀ e,(R e).card≤k) (e : E) : (uniformPaddingIncidence R k e).card=k := by
  45   classical
  46   have hd : Disjoint ((R e).map Function.Embedding.inl)
  47       (Finset.univ.map ⟨fun j : Fin (k-(R e).card)=>Sum.inr ⟨e,j⟩,
  48         fun _ _ h=>by simpa using Sum.inr.inj h⟩ : Finset (UniformPaddingVertex R k)) := by
  49     apply Finset.disjoint_left.mpr
  50     intro x hx hy
  51     obtain ⟨v,hv,rfl⟩ := Finset.mem_map.mp hx
  52     obtain ⟨j,hj,hji⟩ := Finset.mem_map.mp hy
  53     cases hji
  54   rw [uniformPaddingIncidence,Finset.card_union_of_disjoint hd]
  55   simp only [Finset.card_map,Finset.card_univ,Fintype.card_fin]
  56   have he := hsize e
  57   omega
