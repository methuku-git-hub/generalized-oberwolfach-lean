  18 theorem profile_extension_factor (a : I → ℕ) (c J : ℕ) (hc : 3 ≤ c)
  19     (ha : ∀ i,1 ≤ a i) (hsize : ∑ i,a i=c)
  20     (S : T → VertexAbsorberTemplate c) (a₀ b B : ℝ)
  21     (ha₀ : 0 < a₀) (hb : 0 < b) (hB : 0 ≤ B) :
  22     ∃ η : ℝ,0 < η ∧ ∀ q : ℕ,∃ k₀ : ℕ,∀ k : ℕ,k₀ ≤ k →
  23       ∀ (V : I → Type) [∀ i,Fintype (V i)] [∀ i,DecidableEq (V i)]
  24         (C : Type) [Fintype C] [DecidableEq C],
  25       ∀ (R : C → Finset (Σ i,V i)) (w : C → ℝ) (A Z : Finset (Σ i,V i)),
  26       Disjoint A Z → Z.card ≤ q →
  27       (∀ i,(Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈A∪Z)).card=a i*k) →
  28       (∀ C₀ i,(Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈R C₀)).card=a i) →
  29       (∀ U : Finset (Σ i,V i),Fintype.card {C₀ : C // R C₀=U} ≤ J) →
  30       (∀ C₀,b/(k:ℝ)^(c-1) ≤ w C₀ ∧ w C₀ ≤ B/(k:ℝ)^(c-1)) →
  31       (∀ v∈A∪Z,|1-hypergraphWeightedDegree R (inducedResourceWeight R w (insert v A)) v| ≤ η) →
  32       (∀ σ : ∀ i,FixedSubset (V i) (a i),profileRootSet σ⊆A∪Z →
  33         a₀*(k:ℝ)^(c*c) ≤ absorberGroundCount R S (profileRootSet σ) A) →
  34       ∃ D : Finset C,((∀ x∈D,∀ y∈D,x≠y → Disjoint (R x) (R y)) ∧ D.biUnion R=A∪Z) ∧
  35         ∀ C₀∈D,(R C₀∩Z).card ≤ 1 := by
  36   obtain ⟨ηf,hηf,kf,hf⟩ := profile_factor_on_subset a c J hc ha hsize S a₀ b B ha₀ hb hB
  37   refine ⟨ηf/2,by positivity,?_⟩
  38   intro q
  39   let K := (q:ℝ)*J*(c:ℝ)^(c-2)*B
  40   obtain ⟨N,hN⟩ := exists_nat_ge (2*K/ηf)
  41   refine ⟨max kf (max N 1),?_⟩
  42   intro k hk V _ _ C _ _ R w A Z hAZ hZ hY hR hJ hw herr habs
  43   have hkf : kf ≤ k := (le_max_left _ _).trans hk
  44   have hkN : N ≤ k := (le_max_left N 1).trans ((le_max_right _ _).trans hk)
  45   have hk1 : 1 ≤ k := (le_max_right N 1).trans ((le_max_right _ _).trans hk)
  46   have hkpos : (0:ℝ)<k := by exact_mod_cast (show 0<k by omega)
  47   have hsmall : K/(k:ℝ) ≤ ηf/2 := by
  48     have hh := hN.trans (show (N:ℝ) ≤ k by exact_mod_cast hkN)
  49     have hh' := (div_le_iff₀ hηf).mp hh
  50     apply (div_le_iff₀ hkpos).mpr
  51     nlinarith
  52   have hYcard : (A∪Z).card=c*k := by
  53     rw [sigma_finset_card]
  54     simp only [hY,← Finset.sum_mul,hsize]
  55   have hRC (C₀ : C) : (R C₀).card=c := by rw [sigma_finset_card];simp only [hR];exact hsize
  56   have hw0 (C₀ : C) : 0 ≤ w C₀ := (by positivity : (0:ℝ) ≤ b/(k:ℝ)^(c-1)).trans (hw C₀).1
  57   let P := fun C₀ => (R C₀∩Z).card ≤ 1
  58   let C' := {C₀ : C // P C₀}
  59   let R' := candidateResourceSet R P
  60   let w' := fun D : C' => w D.val
  61   have hdegree (v) : hypergraphWeightedDegree R' (inducedResourceWeight R' w' (A∪Z)) v=
  62       hypergraphWeightedDegree R (extensionResourceWeight R w A Z) v := by
  63     change hypergraphWeightedDegree (candidateResourceSet R P)
  64       (fun D => inducedResourceWeight R w (A∪Z) D.val) v=_
  65     rw [candidate_resource_degree]
  66     congr 1
  67     funext C₀
  68     by_cases hp : P C₀ <;> by_cases hs : R C₀⊆A∪Z <;>
  69       simp [hp,hs,inducedResourceWeight,extensionResourceWeight,P] at *
  70   have herr' (v) (hv : v∈A∪Z) :
  71       |1-hypergraphWeightedDegree R' (inducedResourceWeight R' w' (A∪Z)) v| ≤ ηf := by
  72     rw [hdegree]
  73     rcases Finset.mem_union.mp hv with hvA | hvZ
  74     · have hbase := herr v (Finset.mem_union_left _ hvA)
  75       rw [Finset.insert_eq_of_mem hvA] at hbase
  76       obtain ⟨hlo,hhi⟩ := extension_resource_degree_change R c J (by omega) hRC hJ w (B/(k:ℝ)^(c-1))
  77         (by positivity) (fun C₀ => ⟨hw0 C₀,(hw C₀).2⟩) A Z hAZ v hvA
  78       have hbound : (Z.card:ℝ)*((J:ℝ)*((A∪Z).card:ℝ)^(c-2)*(B/(k:ℝ)^(c-1))) ≤ K/k := by
  79         calc
  80           _ ≤ (q:ℝ)*((J:ℝ)*((A∪Z).card:ℝ)^(c-2)*(B/(k:ℝ)^(c-1))) :=
  81             mul_le_mul_of_nonneg_right (by exact_mod_cast hZ) (by positivity)
  82           _ = _ := by
  83             rw [hYcard,Nat.cast_mul,mul_pow]
  84             have hpow : (k:ℝ)^(c-1)=(k:ℝ)^(c-2)*k := by rw [← pow_succ];congr 1;omega
  85             rw [hpow]
  86             dsimp [K]
  87             field_simp
  88       have hdelta := hhi.trans (hbound.trans hsmall)
  89       have hbds := abs_le.mp hbase
  90       apply abs_le.mpr
  91       constructor <;> linarith
  92     · rw [extension_resource_root_degree R w A Z hAZ v hvZ]
  93       exact (herr v (Finset.mem_union_right _ hvZ)).trans (by linarith)
  94   have habs' (σ : ∀ i,FixedSubset (V i) (a i)) (hσ : profileRootSet σ⊆A∪Z) :
  95       a₀*(k:ℝ)^(c*c) ≤ absorberGroundCount R' S (profileRootSet σ) (A∪Z) :=
  96     (habs σ hσ).trans (absorber_extension_ground_count R S (profileRootSet σ) A (A∪Z) Z hAZ Finset.subset_union_left)
  97   have hfactor := hf k hkf V C' R' w' (A∪Z) hY (fun D i => hR D.val i)
  98     (candidate_resource_multiplicity R P J hJ) (fun D => hw D.val) herr' habs'
  99   exact candidate_resource_decomposition_lift R P (A∪Z) hfactor
