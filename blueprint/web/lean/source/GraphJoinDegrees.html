  10 def joinOnFin {a b : ℕ} (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b)) :
  11     SimpleGraph (Fin (a + b)) := (disjointSumOnFin Gᶜ Hᶜ)ᶜ

  13 theorem disjointSumOnFin_degree_left {a b : ℕ}
  14     (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b)) (x : Fin a) :
  15     (disjointSumOnFin G H).degree (finSumFinEquiv (Sum.inl x)) = G.degree x := by
  16   have hh := SimpleGraph.degree_map_apply (G.sum H)
  17     (finSumFinEquiv : Fin a ⊕ Fin b ≃ Fin (a + b)).toEmbedding (Sum.inl x)
  18   rw [sum_degree_inl] at hh
  19   simpa only [disjointSumOnFin, Equiv.toEmbedding_apply, ← SimpleGraph.ncard_neighborSet] using hh
  20 
  21 theorem disjointSumOnFin_degree_right {a b : ℕ}
  22     (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b)) (x : Fin b) :
  23     (disjointSumOnFin G H).degree (finSumFinEquiv (Sum.inr x)) = H.degree x := by
  24   have hh := SimpleGraph.degree_map_apply (G.sum H)
  25     (finSumFinEquiv : Fin a ⊕ Fin b ≃ Fin (a + b)).toEmbedding (Sum.inr x)
  26   rw [sum_degree_inr] at hh

  29 theorem joinOnFin_adj {a b : ℕ}
  30     (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b)) (u v : Fin a ⊕ Fin b) :
  31     (joinOnFin G H).Adj (finSumFinEquiv u) (finSumFinEquiv v) ↔
  32       u ≠ v ∧ ¬ (Gᶜ.sum Hᶜ).Adj u v := by
  33   rw [joinOnFin, SimpleGraph.compl_adj, disjointSumOnFin_adj]
  34   exact and_congr_left (fun _ => not_congr finSumFinEquiv.injective.eq_iff)

  36 theorem joinOnFin_adj_left {a b : ℕ}
  37     (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b)) (u v : Fin a) :
  38     (joinOnFin G H).Adj (finSumFinEquiv (Sum.inl u)) (finSumFinEquiv (Sum.inl v)) ↔
  39       G.Adj u v := by
  40   rw [joinOnFin_adj]
  41   simp only [SimpleGraph.sum_adj, SimpleGraph.compl_adj, ne_eq, Sum.inl.injEq]
  42   have hne := @SimpleGraph.Adj.ne _ G u v
  43   tauto

  60 theorem joinOnFin_degree_left {a b : ℕ}
  61     (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b)) (x : Fin a) :
  62     (joinOnFin G H).degree (finSumFinEquiv (Sum.inl x)) = b + G.degree x := by
  63   have hsum := disjointSumOnFin_degree_left Gᶜ Hᶜ x
  64   have hh := (disjointSumOnFin Gᶜ Hᶜ).degree_compl (v := finSumFinEquiv (Sum.inl x))
  65   have hcomp := G.degree_compl (v := x)
  66   simp only [← SimpleGraph.ncard_neighborSet] at hsum hcomp hh ⊢
  67   rw [hsum, hcomp] at hh
  68   simp only [Fintype.card_fin] at hh
  69   have hlt : (G.neighborSet x).ncard < a := by
  70     simpa only [← SimpleGraph.ncard_neighborSet, Fintype.card_fin] using G.degree_lt_card_verts x
  71   change ((disjointSumOnFin Gᶜ Hᶜ)ᶜ.neighborSet (finSumFinEquiv (Sum.inl x))).ncard = _
  72   rw [hh]
  73   omega
  74 
  75 theorem joinOnFin_degree_right {a b : ℕ}
  76     (G : SimpleGraph (Fin a)) (H : SimpleGraph (Fin b)) (x : Fin b) :
  77     (joinOnFin G H).degree (finSumFinEquiv (Sum.inr x)) = a + H.degree x := by
  78   have hsum := disjointSumOnFin_degree_right Gᶜ Hᶜ x
  79   have hh := (disjointSumOnFin Gᶜ Hᶜ).degree_compl (v := finSumFinEquiv (Sum.inr x))
  80   have hcomp := H.degree_compl (v := x)
  81   simp only [← SimpleGraph.ncard_neighborSet] at hsum hcomp hh ⊢
  82   rw [hsum, hcomp] at hh
  83   simp only [Fintype.card_fin] at hh
  84   have hlt : (H.neighborSet x).ncard < b := by
  85     simpa only [← SimpleGraph.ncard_neighborSet, Fintype.card_fin] using H.degree_lt_card_verts x
  86   change ((disjointSumOnFin Gᶜ Hᶜ)ᶜ.neighborSet (finSumFinEquiv (Sum.inr x))).ncard = _
  87   rw [hh]
  88   omega
