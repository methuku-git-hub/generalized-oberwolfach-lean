  10 theorem prescribed_embedding_zip_update (as:List X) (bs:List V)
  11     (hlen:as.length=bs.length) (ha:as.Nodup) (hb:bs.Nodup)
  12     (A:Finset X) (f:X→V) (hinj:Set.InjOn f A)
  13     (hsource:∀ x∈as,x∉A) (htarget:∀ z∈bs,z∉A.image f):
  14     let g:=updatePairedImages f (as.zip bs);
  15     Set.InjOn g (↑(A∪as.toFinset):Set X) ∧
  16     (∀ a∈A,g a=f a) ∧ (∀ p∈as.zip bs,g p.1=p.2) := by
  17   let ps:=as.zip bs
  18   have hfst:ps.map Prod.fst=as:=List.map_fst_zip hlen.le
  19   have hsnd:ps.map Prod.snd=bs:=List.map_snd_zip hlen.ge
  20   have hsrc:(ps.map Prod.fst).Nodup:=by rw [hfst];exact ha
  21   have htgt:(ps.map Prod.snd).Nodup:=by rw [hsnd];exact hb
  22   have hdom:∀ p∈ps,p.1∉A:=by
  23     intro p hp
  24     exact hsource p.1 (List.of_mem_zip hp).1
  25   have himg:∀ p∈ps,p.2∉A.image f:=by
  26     intro p hp
  27     exact htarget p.2 (List.of_mem_zip hp).2
  28   have hh:=prescribed_embedding_batch_update ps A f hsrc htgt hdom himg hinj
