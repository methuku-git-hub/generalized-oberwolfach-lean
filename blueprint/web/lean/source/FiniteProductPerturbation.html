  13 theorem finite_product_perturbation {J : Type*} (S : Finset J) (x y : J → ℝ)
  14     (h : ∀ j∈S,0≤y j ∧ y j≤x j ∧ x j≤1) :
  15     0≤(∏ j∈S,x j)-(∏ j∈S,y j) ∧
  16       (∏ j∈S,x j)-(∏ j∈S,y j)≤∑ j∈S,(x j-y j) := by
  17   classical
  18   have hnonneg : 0≤(∏ j∈S,x j)-(∏ j∈S,y j) :=
  19     sub_nonneg.mpr (Finset.prod_le_prod₀ (fun j hj=>(h j hj).1) (fun j hj=>(h j hj).2.1))
  20   refine ⟨hnonneg,?_⟩
  21   induction S using Finset.induction_on with
  22   | empty => simp
  23   | @insert j S hj ih =>
  24     have hS : ∀ i∈S,0≤y i ∧ y i≤x i ∧ x i≤1 := fun i hi=>h i (Finset.mem_insert_of_mem hi)
  25     have hh := h j (Finset.mem_insert_self _ _)
  26     have hprod0 : 0≤∏ i∈S,y i := Finset.prod_nonneg (fun i hi=>(hS i hi).1)
  27     have hprod1 : (∏ i∈S,x i)≤1 := Finset.prod_le_one₀ (fun i hi=>(hS i hi).1.trans (hS i hi).2.1)
  28       (fun i hi=>(hS i hi).2.2)
  29     have hdiff : 0≤(∏ i∈S,x i)-(∏ i∈S,y i) :=
  30       sub_nonneg.mpr (Finset.prod_le_prod₀ (fun i hi=>(hS i hi).1) (fun i hi=>(hS i hi).2.1))
  31     have hind := ih hS hdiff
  32     rw [Finset.prod_insert hj,Finset.prod_insert hj,Finset.sum_insert hj]
  33     have h₁ := mul_le_mul_of_nonneg_left hprod1 (sub_nonneg.mpr hh.2.1)
  34     have h₂ := mul_le_mul_of_nonneg_right (hh.2.1.trans hh.2.2) hdiff
  35     nlinarith

  37 theorem normalized_tuple_mass_le_one {n s l : ℕ} (hl : 0<l) (hs : s≤l)
  38     {ρ : ℝ} (hρ : 0≤ρ ∧ ρ≤1) :
  39     (s : ℝ)^n*(ρ/(l : ℝ))^n≤1 := by
  40   rw [← mul_pow]
  41   apply pow_le_one₀
  42   · exact mul_nonneg (Nat.cast_nonneg _) (div_nonneg hρ.1 (Nat.cast_nonneg _))
  43   · have hlr : (0 : ℝ)<l := by exact_mod_cast hl
  44     have hsr : (s : ℝ)≤l := by exact_mod_cast hs
  45     have hsr0 : (0 : ℝ)≤s := Nat.cast_nonneg _
  46     rw [← mul_div_assoc]
  47     apply (div_le_iff₀ hlr).mpr
  48     calc
  49       (s : ℝ)*ρ ≤ (s : ℝ)*1 := mul_le_mul_of_nonneg_left hρ.2 hsr0
  50       _ ≤ 1*(l : ℝ) := by simpa using hsr

  52 theorem normalized_tuple_conflict_le {n s l A : ℕ} (hl : 0<l) (hs : s≤l)
  53     {ρ : ℝ} (hρ : 0≤ρ ∧ ρ≤1) :
  54     ((n^2*A*s^(n-1) : ℕ) : ℝ)*(ρ/(l : ℝ))^n≤
  55       ((n^2*A : ℕ) : ℝ)/(l : ℝ) := by
  56   by_cases hn : n=0
  57   · simp [hn]
  58   have hn1 : 1≤n := by omega
  59   have hexp : n=(n-1)+1 := by omega
  60   have hp := normalized_tuple_mass_le_one (n:=n-1) hl hs hρ
  61   have hlr : (0 : ℝ)<l := by exact_mod_cast hl
  62   have hratio : ρ/(l : ℝ)≤1/(l : ℝ) := div_le_div_of_nonneg_right hρ.2 hlr.le
  63   have hratio0 : 0≤ρ/(l : ℝ) := div_nonneg hρ.1 hlr.le
  64   have hsmall : (s : ℝ)^(n-1)*(ρ/(l : ℝ))^n≤1/(l : ℝ) := by
  65     have hpow : (ρ/(l : ℝ))^n=(ρ/(l : ℝ))^(n-1)*(ρ/(l : ℝ)) := by
  66       calc
  67         _ = (ρ/(l : ℝ))^((n-1)+1) := congrArg (fun a=>(ρ/(l : ℝ))^a) hexp
  68         _ = _ := pow_succ _ _
  69     rw [hpow]
