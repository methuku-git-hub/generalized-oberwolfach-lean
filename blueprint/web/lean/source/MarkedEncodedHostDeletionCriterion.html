  16 theorem encoded_host_rim_deletion_iff
  17     (tail head:E→V) (ktail khead:K→V) (π:Equiv.Perm V)
  18     (σ:RawMarkedOutcome V U Γ Color) (I:Finset V):
  19     (∃ D:Finset (Sym2 V),rawMarkedEncodeHost tail head ktail khead π σ I=rawMarkedDeleteRims σ D) ↔
  20     (∀ e:Sym2 V,¬e.toFinset⊆I→
  21       markedHostRimEncoding tail head ktail khead π e=none ∨
  22       markedHostRimEncoding tail head ktail khead π e=σ (Sum.inl e)) := by
  23   classical
  24   constructor
  25   · rintro ⟨D,hD⟩ e hout
  26     have hh:=congrFun hD (Sum.inl e)
  27     rw [rawMarkedEncodeHost,ite_eq_right hout,rawMarkedDeleteRims] at hh
  28     by_cases he:e∈D
  29     · left;simpa [he] using hh
  30     · right;simpa [he] using hh
  31   · intro hcomp
  32     let D:Finset (Sym2 V):=Finset.univ.filter (fun e=>¬e.toFinset⊆I ∧
  33       markedHostRimEncoding tail head ktail khead π e=none)
  34     refine ⟨D,?_⟩
  35     funext a
  36     cases a with
  37     | inr z=>rfl
  38     | inl e=>
  39       simp only [rawMarkedEncodeHost,rawMarkedDeleteRims,D,Finset.mem_filter,Finset.mem_univ,true_and,
  40         Finset.subset_iff,Sym2.mem_toFinset]
  41       have hc:=hcomp e
  42       simp only [Finset.subset_iff,Sym2.mem_toFinset] at hc
  43       by_cases hi:∀ ⦃x:V⦄,x∈e→x∈I
  44       · split_ifs <;> simp_all
  45       · by_cases he:markedHostRimEncoding tail head ktail khead π e=none
  46         · simp [hi,he]
  47         · have hs:= (hc hi).resolve_left he
  48           simp [hi,he,hs]
