  12 theorem endpoint_triangle_factor_on_domain (γ b B : ℝ) (hγ : 0<γ) (hb : 0<b) (hB : 0≤B) :
  13     ∃ ε η : ℝ,0<ε ∧ 0<η ∧ ∃ N : ℕ,
  14       ∀ (V : Type) [Fintype V] [DecidableEq V],
  15       ∀ (Q F : SimpleGraph V) (Ω S T : Finset V),N≤Ω.card →
  16       S⊆Ω → T⊆Ω → Disjoint S T → S.card=2*T.card →
  17       (∀ v∈Ω,(F.degree v:ℝ)≤ε*Ω.card) →
  18       EndpointNeighbourCounts Q F S T (γ*Ω.card) →
  19       ∀ wt : EndpointTriangle Q F S T→ℝ,
  20       (∀ C,b/(Ω.card:ℝ)^2≤wt C ∧ wt C≤B/(Ω.card:ℝ)^2) →
  21       (∀ v∈S∪T,|1-hypergraphWeightedDegree endpointTriangleVertices wt v|≤η) →
  22       HasResourceDecomposition (endpointTriangleVertices (Q:=Q) (F:=F) (S:=S) (T:=T)) (S∪T) := by
  23   obtain ⟨ε,η,hε,hη,N,hfactor⟩:=endpoint_triangle_factor γ b B hγ hb hB
  24   refine ⟨ε,η,hε,hη,N,?_⟩
  25   intro V _ _ Q F Ω S T hn hS hT hST hratio hF hcounts wt hw herr
  26   let W:=↥Ω
  27   let e:W↪V:=⟨Subtype.val,Subtype.val_injective⟩
  28   have hScov:∀ v∈S,∃ w,e w=v:=fun v hv=>⟨⟨v,hS hv⟩,rfl⟩
  29   have hTcov:∀ v∈T,∃ w,e w=v:=fun v hv=>⟨⟨v,hT hv⟩,rfl⟩
  30   have hcard:Fintype.card W=Ω.card:=Fintype.card_coe Ω
  31   have hST':Disjoint (embeddingPullFinset e S) (embeddingPullFinset e T):=by
  32     apply Finset.disjoint_left.mpr
  33     intro v hv ht
  34     exact Finset.disjoint_left.mp hST ((embeddingPullFinset_mem _ _ _).mp hv)
  35       ((embeddingPullFinset_mem _ _ _).mp ht)
  36   have hcounts':EndpointNeighbourCounts (Q.comap e) (F.comap e)
  37       (embeddingPullFinset e S) (embeddingPullFinset e T) (γ*Fintype.card W):=by
  38     rw [hcard]
  39     exact endpoint_counts_embedding e Q F S T hScov hTcov _ hcounts
  40   have hresult:=hfactor W (by simpa only [hcard] using hn) (Q.comap e) (F.comap e)
  41     (embeddingPullFinset e S) (embeddingPullFinset e T) hST'
  42     (by simpa only [embeddingPullFinset_card e S hScov,embeddingPullFinset_card e T hTcov] using hratio)
  43     (by intro v;rw [hcard];exact (Nat.cast_le.mpr (embedding_comap_degree_le e F v)).trans (hF (e v) v.property))
  44     hcounts' (fun C=>wt (endpointTriangleLift e Q F S T C))
  45     (by intro C;simpa only [hcard] using hw (endpointTriangleLift e Q F S T C))
  46     (by
  47       intro v hv
  48       rw [endpointTriangleLift_weighted_degree e Q F S T hScov hTcov]
  49       apply herr
  50       simpa only [Finset.mem_union,embeddingPullFinset_mem] using hv)
  51   exact endpointTriangleLift_decomposition e Q F S T hScov hTcov hresult
