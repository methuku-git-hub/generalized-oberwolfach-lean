  23 theorem occurrenceRootNeighbours_restriction_loss (f g:E→V)
  24     (hinj:Function.Injective (fun e=>(f e,g e))) (P:E→Prop) (v:V):
  25     (occurrenceRootNeighbours f g v\
  26       occurrenceRootNeighbours (fun e:{e // P e}=>f e.val) (fun e=>g e.val) v).card=
  27       (Finset.univ.filter (fun e=>¬P e∧g e=v)).card:=by
  28   have hi:Function.Injective (fun e:{e // P e}=>(f e.val,g e.val)):=hinj.comp Subtype.val_injective
  29   have hs:=occurrenceRootNeighbours_restriction_subset f g P v
  30   rw [Finset.card_sdiff_of_subset hs,occurrenceRootNeighbours_card f g hinj,
  31     occurrenceRootNeighbours_card (fun e:{e // P e}=>f e.val) (fun e=>g e.val) hi]
  32   have hc:(Finset.univ.filter (fun e:{e // P e}=>g e.val=v)).card=
  33       (Finset.univ.filter (fun e=>P e∧g e=v)).card:=by
  34     convert subtype_filter_predicate_card P (fun e=>g e=v) using 1 <;> congr!
  35   rw [hc]
  36   have hsplit:(Finset.univ.filter (fun e=>P e∧g e=v)).card+
  37       (Finset.univ.filter (fun e=>¬P e∧g e=v)).card=
  38         (Finset.univ.filter (fun e=>g e=v)).card:=by
  39     simp only [Finset.card_filter,←Finset.sum_add_distrib]
  40     apply Finset.sum_congr rfl
  41     intro e _
  42     by_cases hp:P e <;> by_cases hg:g e=v <;> simp [hp,hg]
  43   omega

  45 theorem marked_neighbour_physical_deletion_loss
  46     (f g:E→V) (π:Equiv.Perm V)
  47     (hp:Function.Injective (fun e=>({f e,π (g e)}:Finset V))) (D:SimpleGraph V) (v:V):
  48     ((occurrenceRootNeighbours f g v\
  49       occurrenceRootNeighbours (fun e:{e // ¬D.Adj (f e) (π (g e))}=>f e.val) (fun e=>g e.val) v).card:ℝ)
  50       ≤D.degree (π v):=by
  51   have hi:Function.Injective (fun e=>(f e,g e)):=by
  52     intro e z he
  53     apply hp
  54     have hf:f e=f z:=congrArg Prod.fst he
  55     have hg:g e=g z:=congrArg Prod.snd he
  56     change ({f e,π (g e)}:Finset V)={f z,π (g z)}
  57     rw [hf,hg]
  58   have he:(occurrenceRootNeighbours f g v\
  59       occurrenceRootNeighbours (fun e:{e // ¬D.Adj (f e) (π (g e))}=>f e.val) (fun e=>g e.val) v).card=
  60       (Finset.univ.filter (fun e=>¬¬D.Adj (f e) (π (g e))∧g e=v)).card:=by
  61     convert occurrenceRootNeighbours_restriction_loss f g hi (fun e=>¬D.Adj (f e) (π (g e))) v using 1 <;> congr!
  62   rw [he]
  63   have hc:((Finset.univ.filter (fun e=>¬¬D.Adj (f e) (π (g e))∧g e=v)).card:ℝ)=
  64       ∑ e,if D.Adj (f e) (π (g e))∧π (g e)=π v then (1:ℝ) else 0:=by
  65     simp only [not_not,Equiv.apply_eq_iff_eq,Finset.card_filter,Nat.cast_sum,Nat.cast_ite,Nat.cast_one,Nat.cast_zero]
  66   rw [hc]
  67   exact deleted_pair_indegree f (fun e=>π (g e)) hp D (π v)
