  11 theorem finiteLaw_product_exchange_mgf
  12     {n : ℕ} {State : Fin n → Type*} [∀ i, Fintype (State i)]
  13     (μ : ∀ i, FiniteLaw (State i)) (R : ∀ i, State i → State i → Prop)
  14     (v : Fin n → ℝ)
  15     (hlaw : ∀ i (f : State i → ℝ) (b : ℝ), 0 ≤ b →
  16       (∀ x y, R i x y → |f x - f y| ≤ b) → ∀ u,
  17       (μ i).expectation (fun x => Real.exp (u * (f x - (μ i).expectation f))) ≤
  18         Real.exp (u ^ 2 * (v i * b ^ 2) / 8))
  19     (X : (∀ i, State i) → ℝ) (c : Fin n → ℝ) (hc : ∀ i, 0 ≤ c i)
  20     (hchange : ∀ (i : Fin n) (σ τ : ∀ j, State j),
  21       (∀ j, j ≠ i → σ j = τ j) → R i (σ i) (τ i) → |X σ - X τ| ≤ c i)
  22     (t : ℝ) :
  23     (FiniteLaw.pi μ).expectation (fun σ =>
  24       Real.exp (t * (X σ - (FiniteLaw.pi μ).expectation X))) ≤
  25       Real.exp (t ^ 2 * (∑ i, v i * (c i) ^ 2) / 8) := by
  26   classical
  27   induction n with
  28   | zero =>
  29     let σ₀ : ∀ i : Fin 0, State i := fun i => Fin.elim0 i
  30     have hX : X = fun _ => X σ₀ := by
  31       funext σ
  32       exact congrArg X (Subsingleton.elim σ σ₀)
  33     rw [hX, finiteLaw_expectation_const]
  34     simp [finiteLaw_expectation_const]
  35   | succ n ih =>
  36     let ν := FiniteLaw.pi (fun i : Fin n => μ i.succ)
  37     let g (x : State 0) := ν.expectation (fun τ => X (Fin.cons x τ))
  38     have hmean : (FiniteLaw.pi μ).expectation X = (μ 0).expectation g :=
  39       finiteLaw_pi_expectation_succ μ X
  40     have htail (x : State 0) :
  41         ν.expectation (fun τ => Real.exp (t * (X (Fin.cons x τ) - g x))) ≤
  42           Real.exp (t ^ 2 * (∑ i : Fin n, v i.succ * (c i.succ) ^ 2) / 8) := by
  43       apply ih (fun i => μ i.succ) (fun i => R i.succ) (fun i => v i.succ)
  44         (fun i => hlaw i.succ) (fun τ => X (Fin.cons x τ))
  45         (fun i => c i.succ) (fun i => hc i.succ)
  46       intro i σ τ hagree hrel
  47       apply hchange i.succ (Fin.cons x σ) (Fin.cons x τ)
  48       · intro j hj
  49         cases j using Fin.cases with
  50         | zero => rfl
  51         | succ j =>
  52           simp only [Fin.cons_succ]
  53           exact hagree j (fun h => hj (congrArg Fin.succ h))
  54       · simpa only [Fin.cons_succ] using hrel
  55     have hedge (x y : State 0) (hrel : R 0 x y) : |g x - g y| ≤ c 0 := by
  56       apply finiteLaw_expectation_difference ν
  57       intro τ
  58       apply hchange 0 (Fin.cons x τ) (Fin.cons y τ)
  59       · intro j hj
  60         cases j using Fin.cases with
  61         | zero => exact (hj rfl).elim
  62         | succ j => rfl
  63       · simpa only [Fin.cons_zero] using hrel
  64     have hhead := hlaw 0 g (c 0) (hc 0) hedge t
  65     let A := Real.exp (t ^ 2 * (∑ i : Fin n, v i.succ * (c i.succ) ^ 2) / 8)
  66     have hfactor (x : State 0) :
  67         ν.expectation (fun τ => Real.exp (t * (X (Fin.cons x τ) -
  68           (FiniteLaw.pi μ).expectation X))) =
  69         Real.exp (t * (g x - (μ 0).expectation g)) *
  70           ν.expectation (fun τ => Real.exp (t * (X (Fin.cons x τ) - g x))) := by
  71       rw [← finiteLaw_expectation_const_mul]
  72       apply congrArg ν.expectation
  73       funext τ
  74       rw [← Real.exp_add, hmean]
  75       congr 1
  76       ring
  77     rw [finiteLaw_pi_expectation_succ]
  78     change (μ 0).expectation (fun x =>
  79       ν.expectation (fun τ => Real.exp (t * (X (Fin.cons x τ) -
  80         (FiniteLaw.pi μ).expectation X)))) ≤ _
  81     simp_rw [hfactor]
  82     calc
  83       _ ≤ (μ 0).expectation (fun x => A *
  84           Real.exp (t * (g x - (μ 0).expectation g))) := by
  85         apply finiteLaw_expectation_mono
  86         intro x
  87         have hh := mul_le_mul_of_nonneg_left (htail x)
  88           (Real.exp_pos (t * (g x - (μ 0).expectation g))).le
  89         simpa only [mul_comm] using hh
  90       _ = A * (μ 0).expectation (fun x =>
  91           Real.exp (t * (g x - (μ 0).expectation g))) :=
  92         finiteLaw_expectation_const_mul _ _ _
  93       _ ≤ A * Real.exp (t ^ 2 * (v 0 * (c 0) ^ 2) / 8) :=
  94         mul_le_mul_of_nonneg_left hhead (Real.exp_pos _).le
  95       _ = _ := by
  96         dsimp [A]
  97         rw [← Real.exp_add, Fin.sum_univ_succ]
  98         congr 1
  99         ring
