  17 structure VertexVortex {V : Type a} [Fintype V] (G : SimpleGraph V)
  18     (D : ℕ) (η ε : ℝ) (m : ℕ) where
  19   length : ℕ
  20   length_pos : 0 < length
  21   vertices : ℕ → Finset V
  22   initial : vertices 0 = Finset.univ
  23   sizes : ∀ i ≤ length, (vertices i).card = floorOrbit ε (Fintype.card V) i
  24   nested : ∀ i < length, vertices (i + 1) ⊆ vertices i
  25   terminal_lower : ε * m / 2 ≤ ((vertices length).card : ℝ)
  26   terminal_upper : (vertices length).card < m
  27   degrees : ∀ i ≤ length, 0 < i → ∀ x ∈ vertices (i - 2),
  28     |((G.neighborFinset x ∩ vertices i).card : ℝ) -
  29       ((D : ℝ) / Fintype.card V) * (vertices i).card| ≤ η * (vertices i).card

  31 theorem vertexVortex_exists {η ε : ℝ} (hη : 0 < η) (he : 0 < ε) (he1 : ε < 1) :
  32     ∃ m₀ : ℕ, 1 ≤ m₀ ∧ ∀ (V : Type a) [Fintype V] (G : SimpleGraph V) (D m : ℕ),
  33       G.IsRegularOfDegree D → m₀ ≤ m → m ≤ Fintype.card V →
  34       Nonempty (VertexVortex G D η ε m) := by
  35   obtain ⟨N₁, hN₁2, hN₁⟩ := vortex_set_sequence.{a} he he1
  36   obtain ⟨N₂, hN₂1, hN₂⟩ := eventually_vortex_error_budget (β := 2 * η)
  37     (ρ := 1) (by positivity) he he1 (by norm_num)
  38   refine ⟨max N₁ N₂, by omega, ?_⟩
  39   intro V _ G D m hreg hm hmn
  40   have hmpos : 0 < m := by omega
  41   obtain ⟨hlarge, hbudget, _⟩ := hN₂ m (by omega)
  42   have hbudget' : (m : ℝ) ^ (-1 / 3 : ℝ) / (1 - ε ^ (1 / 3 : ℝ)) ≤ η := by
  43     nlinarith [hbudget]
  44   obtain ⟨ℓ, hlpos, hlupper, hlbefore, hllower⟩ := floorOrbit_stopping he he1 hmpos hmn hlarge
  45   obtain ⟨B, hB0, hBcard, hBstep⟩ := hN₁ V Finset.univ (fun x => G.neighborFinset x) ℓ
  46     (by intro i hi; simpa only [Finset.card_univ] using (show N₁ ≤ floorOrbit ε (Fintype.card V) i from (by have := hlbefore i hi; omega)))
  47   have hcard : ∀ i ≤ ℓ, (B i).card = floorOrbit ε (Fintype.card V) i := by
  48     simpa only [Finset.card_univ] using hBcard
  49   have hsub : ∀ i < ℓ, B (i + 1) ⊆ B i := fun i hi => (hBstep i hi).1
  50   refine ⟨⟨ℓ, hlpos, B, hB0, hcard, hsub, ?_, ?_, ?_⟩⟩
  51   · rw [hcard ℓ (by omega)]
  52     exact hllower ℓ (by omega)
  53   · rw [hcard ℓ (by omega)]
  54     exact hlupper
  55   · intro i hi hipos x hx
  56     have htel := nested_normalized_count_estimate B (fun x => G.neighborFinset x)
  57       (fun j => ((B j).card : ℝ) ^ (-1 / 3 : ℝ)) ℓ hsub
  58       (fun j hj y hy => ((hBstep j hj).2 y hy).le) hi hx
  59     have hsum : ∑ j ∈ Finset.range i, ((B j).card : ℝ) ^ (-1 / 3 : ℝ) ≤ η := by
  60       have hisucc : i - 1 + 1 = i := by omega
  61       have hs := vortex_accumulated_error he he1 (show (0 : ℝ) < m by exact_mod_cast hmpos)
  62         (fun j => (floorOrbit ε (Fintype.card V) j : ℝ)) (i - 1)
  63         (by
  64           intro j hj
  65           have hh := hlbefore j (by omega)
  66           exact_mod_cast (show 0 < floorOrbit ε (Fintype.card V) j by omega))
  67         (fun j _ => floorOrbit_shrinks he.le _ j)
  68         (by exact_mod_cast hlbefore (i - 1) (by omega))
  69       rw [hisucc] at hs
  70       have heq : (∑ j ∈ Finset.range i, ((B j).card : ℝ) ^ (-1 / 3 : ℝ)) =
  71           ∑ j ∈ Finset.range i, (floorOrbit ε (Fintype.card V) j : ℝ) ^ (-1 / 3 : ℝ) := by
  72         apply Finset.sum_congr rfl
  73         intro j hj
  74         rw [hcard j (by have := Finset.mem_range.mp hj; omega)]
  75       rw [heq]
  76       exact hs.trans hbudget'
  77     have hnorm := htel.trans hsum
  78     rw [hB0, Finset.inter_univ, Finset.card_univ,
  79       SimpleGraph.card_neighborFinset_eq_degree, hreg x] at hnorm
  80     have hpos : (0 : ℝ) < (B i).card := by
  81       have hh : ε * m / 2 ≤ ((B i).card : ℝ) := by rw [hcard i hi]; exact hllower i hi
  82       linarith
  83     have heq : (((G.neighborFinset x ∩ B i).card : ℝ) -
  84         ((D : ℝ) / Fintype.card V) * (B i).card) / (B i).card =
  85         ((G.neighborFinset x ∩ B i).card : ℝ) / (B i).card - (D : ℝ) / Fintype.card V := by
  86       field_simp
  87     rw [← heq, abs_div, abs_of_pos hpos] at hnorm
  88     exact (div_le_iff₀ hpos).mp hnorm
