  16 theorem retainedIntervalCovered_unique (π : Equiv.Perm V) (d : ℕ)
  17     (σ : RawIntervalOutcome V Γ Color) (R : V→Γ→Prop)
  18     (hR : ∀ a g,R a g → rawIntervalColouredStart π d σ a g) (x : V) (g h : Γ)
  19     (hg : retainedIntervalCovered π d R x g) (hh : retainedIntervalCovered π d R x h) : g=h := by
  20   obtain ⟨a,ha,hxa⟩ := hg
  21   obtain ⟨b,hb,hxb⟩ := hh
  22   exact rawIntervalCovered_unique π d σ x g h ⟨a,hR a g ha,hxa⟩ ⟨b,hR b h hb,hxb⟩

  24 theorem retainedIntervalState_interval_iff (π : Equiv.Perm V) (d : ℕ)
  25     (σ : RawIntervalOutcome V Γ Color) (R : V→Γ→Prop)
  26     (hR : ∀ a g,R a g → rawIntervalColouredStart π d σ a g) (x : V) (g : Γ) :
  27     retainedIntervalState π d σ R x=some (Sum.inl g) ↔ retainedIntervalCovered π d R x g := by
  28   unfold retainedIntervalState
  29   split_ifs with hg hb
  30   · have hh := Classical.choose_spec hg
  31     constructor
  32     · intro he
  33       have he' : Classical.choose hg=g := Sum.inl.inj (Option.some.inj he)
  34       exact he' ▸ hh
  35     · intro he
  36       rw [retainedIntervalCovered_unique π d σ R hR x _ _ hh he]
  37   · constructor
  38     · intro he;cases he
  39     · intro he;exact (hg ⟨g,he⟩).elim
  40   · constructor
  41     · intro he
  42       cases hc : σ (Sum.inr (Sum.inr x)) <;> simp [hc] at he
  43     · intro he;exact (hg ⟨g,he⟩).elim

  77 theorem retainedRawFamily_covered_card (π : Equiv.Perm V) (d : ℕ) (hd : 1≤d)
  78     (hnoshort : ∀ x l,0<l → l<d → (π^l) x≠x)
  79     (σ : RawIntervalFamilyOutcome V U Γ Color) (g : Γ) (t : ℕ)
  80     (J : StartRetentionOutcome (rawStartHubSet π d σ g) t) (x : V) :
  81     (Finset.univ.filter (fun u=>retainedRawFamilyState π d σ g t J (x,u)=some (Sum.inl g))).card=d*t := by
  82   rw [retainedRawFamily_covered_hubs]
  83   have hdisj : ((permutationWindow π (d-1) 0 x : Finset V):Set V).PairwiseDisjoint
  84       (fun a=>startRetainedHubs (rawStartHubSet π d σ g) t J a) := by
  85     intro a ha b hb hne
  86     apply Finset.disjoint_left.mpr
  87     intro u hua hub
  88     have hra := (retainedRawStart_selected_iff π d σ g t J u a).mpr hua
  89     have hrb := (retainedRawStart_selected_iff π d σ g t J u b).mpr hub
  90     have hxa := (permutationWindow_dual π (d-1) 0 x a).mp ha
  91     have hxb := (permutationWindow_dual π (d-1) 0 x b).mp hb
  92     exact hne (rawInterval_coloured_start_unique π d (fun e=>σ (u,e)) x a b g g hra.1 hrb.1
  93       (permutationWindow_mono π (le_refl 0) (Nat.sub_le d 1) a hxa)
  94       (permutationWindow_mono π (le_refl 0) (Nat.sub_le d 1) b hxb)).1
  95   rw [Finset.card_biUnion hdisj]
  96   simp only [startRetainedHubs_card,Finset.sum_const,nsmul_eq_mul]
