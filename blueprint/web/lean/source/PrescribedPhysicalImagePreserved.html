  11 theorem prescribed_physical_image_preserved (A:Finset X) (f g:X→V)
  12     (hpres:∀ x∈A,g x=f x) (T:Finset (Sym2 X))
  13     (hends:∀ e∈T,∀ x∈e.toFinset,x∈A):
  14     T.image (Sym2.map g)=T.image (Sym2.map f) := by
  15   apply Finset.image_congr
  16   intro e he
  17   induction e using Sym2.inductionOn with
  18   | _ x y =>
  19     have hx:x∈A:=hends _ he x (by simp)
  20     have hy:y∈A:=hends _ he y (by simp)
  21     simp [hpres x hx,hpres y hy]
  22 end Oberwolfach
