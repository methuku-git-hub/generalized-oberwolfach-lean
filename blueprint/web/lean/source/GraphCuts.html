  12 def graphCut {V : Type*} [Fintype V] (G : SimpleGraph V) (S T : Finset V) : ℕ :=
  13   ∑ x ∈ S, (G.neighborFinset x ∩ T).card
  14 
  15 theorem graphCut_eq_sum
  16     {V : Type*} [Fintype V] (G : SimpleGraph V) (S T : Finset V) :
  17     graphCut G S T = ∑ x ∈ S, ∑ y ∈ T, if G.Adj x y then 1 else 0 := by
  18   unfold graphCut
  19   apply Finset.sum_congr rfl
  20   intro x _
  21   have hh : G.neighborFinset x ∩ T = T.filter (G.Adj x) := by
  22     ext y
  23     simp only [Finset.mem_inter, SimpleGraph.mem_neighborFinset, Finset.mem_filter]
  24     tauto
  25   rw [hh, Finset.card_filter]
  26 
  27 theorem graphCut_symm
  28     {V : Type*} [Fintype V] (G : SimpleGraph V) (S T : Finset V) :
  29     graphCut G S T = graphCut G T S := by
  30   rw [graphCut_eq_sum, graphCut_eq_sum, Finset.sum_comm]
  31   apply Finset.sum_congr rfl
  32   intro y _
  33   apply Finset.sum_congr rfl
  34   intro x _
  35   rw [G.adj_comm]

  37 theorem graphCut_lower_of_degrees
  38     {V : Type*} [Fintype V] (G : SimpleGraph V) (S T : Finset V) {d : ℝ}
  39     (hdegree : ∀ x ∈ S, d ≤ (G.degree x : ℝ)) :
  40     (S.card : ℝ) * (d + T.card - Fintype.card V) ≤ (graphCut G S T : ℝ) := by
  41   have hx (x : V) (hx : x ∈ S) : d + T.card - Fintype.card V ≤
  42       ((G.neighborFinset x ∩ T).card : ℝ) := by
  43     have hc := Finset.card_inter_add_card_union (G.neighborFinset x) T
  44     have hu := Finset.card_le_univ (G.neighborFinset x ∪ T)
  45     have hcr : ((G.neighborFinset x ∩ T).card : ℝ) +
  46         ((G.neighborFinset x ∪ T).card : ℝ) = (G.degree x : ℝ) + T.card := by
  47       exact_mod_cast (by simpa only [SimpleGraph.card_neighborFinset_eq_degree] using hc)
  48     have hur : ((G.neighborFinset x ∪ T).card : ℝ) ≤ Fintype.card V := by exact_mod_cast hu
  49     linarith [hdegree x hx]
  50   have hh := Finset.sum_le_sum hx
  51   simpa only [graphCut, Nat.cast_sum, Finset.sum_const, nsmul_eq_mul] using hh

  53 theorem graphCut_large_of_minDegree
  54     {V : Type*} [Fintype V] (G : SimpleGraph V) {α δ : ℝ}
  55     (hdegree : ∀ x, (1 / 2 + α) * Fintype.card V ≤ (G.degree x : ℝ))
  56     (S T : Finset V)
  57     (hlarge : (1 - 2 * δ) * Fintype.card V < (S.card : ℝ) + T.card) :
  58     (α - δ) * Fintype.card V * min (S.card : ℝ) T.card ≤ (graphCut G S T : ℝ) := by
  59   have hsmallLarge (A B : Finset V)
  60       (hab : A.card ≤ B.card)
  61       (hlarge : (1 - 2 * δ) * Fintype.card V < (A.card : ℝ) + B.card) :
  62       (α - δ) * Fintype.card V * A.card ≤ (graphCut G A B : ℝ) := by
  63     have hr : (A.card : ℝ) ≤ B.card := by exact_mod_cast hab
  64     have hb : (1 / 2 - δ) * Fintype.card V ≤ (B.card : ℝ) := by linarith
  65     have hh := graphCut_lower_of_degrees G A B (fun x _ => hdegree x)
  66     have hm := mul_le_mul_of_nonneg_left hb (show (0 : ℝ) ≤ A.card by positivity)
  67     nlinarith
  68   rcases le_total S.card T.card with hST | hTS
  69   · have hr : (S.card : ℝ) ≤ T.card := by exact_mod_cast hST
  70     rw [min_eq_left hr]
  71     exact hsmallLarge S T hST hlarge
  72   · have hr : (T.card : ℝ) ≤ S.card := by exact_mod_cast hTS
  73     rw [min_eq_right hr, graphCut_symm]
  74     exact hsmallLarge T S hTS (by linarith)
