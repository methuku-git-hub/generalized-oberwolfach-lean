  12 theorem permutationConflictGraph_degree_le {V : Type} [Fintype V] [DecidableEq V]
  13     (π : Equiv.Perm V) (b : ℕ) (x : V) : (permutationConflictGraph π b).degree x≤2*b := by
  14   let S := Finset.univ.filter (fun y=>PermutationClose π b x y)
  15   have hx : x∉(permutationConflictGraph π b).neighborFinset x := by simp
  16   have hsub : insert x ((permutationConflictGraph π b).neighborFinset x)⊆S := by
  17     intro y hy
  18     apply Finset.mem_filter.mpr
  19     refine ⟨Finset.mem_univ _,?_⟩
  20     rcases Finset.mem_insert.mp hy with rfl|hy
  21     · exact Or.inl rfl
  22     · exact (((permutationConflictGraph π b).mem_neighborFinset x y).mp hy).2
  23   have hc := Finset.card_le_card hsub
  24   rw [Finset.card_insert_of_notMem hx,SimpleGraph.card_neighborFinset_eq_degree] at hc
  25   have hb := permutationClose_neighbour_card π b x
  26   have he : S.card=Fintype.card {y // PermutationClose π b x y} := by
  27     simp only [S,Fintype.card_subtype]
  28   rw [he] at hc
  29   omega
