   3 /-! The graph of a separately typed source factor is mapped exactly into its
   4 fiber in the common dependent-sum source universe. -/
   5 set_option autoImplicit false
   6 noncomputable section
   7 namespace Oberwolfach
   8 attribute [local instance] Classical.propDecidable
   9 variable {I:Type} [Fintype I] [DecidableEq I] {X:I→Type} [∀ i,Fintype (X i)] [∀ i,DecidableEq (X i)]

  11 theorem prescribed_lifted_source_graph_eq_map (pieces:∀ i,List (PrescribedSourcePiece (X i))) (i:I):
  12     prescribedSourceGraph (prescribedLiftDependentPieces Finset.univ.toList pieces) i =
  13       (prescribedDependentGraph pieces i).map (Sigma.mk i) := by
  14   ext u v
  15   constructor
  16   · intro huv
  17     have hu:=prescribed_source_graph_endpoint _ i u v huv
  18     have hv:=prescribed_source_graph_endpoint _ i v u huv.symm
  19     have hu':u.1=i:=prescribed_lifted_source_vertex_fiber pieces i u hu
  20     have hv':v.1=i:=prescribed_lifted_source_vertex_fiber pieces i v hv
  21     rcases u with ⟨j,x⟩
  22     rcases v with ⟨k,y⟩
  23     simp only at hu' hv'
  24     subst j
  25     subst k
  26     apply (SimpleGraph.map_adj ⟨Sigma.mk i,sigma_mk_injective⟩ (prescribedDependentGraph pieces i) _ _).mpr
  27     exact ⟨x,y,(prescribed_lifted_source_adjacency pieces i x y).mp huv,rfl,rfl⟩
  28   · intro huv
  29     obtain ⟨x,y,hxy,rfl,rfl⟩:=(SimpleGraph.map_adj ⟨Sigma.mk i,sigma_mk_injective⟩ (prescribedDependentGraph pieces i) u v).mp huv
  30     exact (prescribed_lifted_source_adjacency pieces i x y).mpr hxy
  31 end Oberwolfach
