   9 theorem long_matching_tolerances_exist
  10     (K:ℕ) (hK:0<K) (θ η C:ℝ) (hθ:0<θ) (hη:0<η) (hC:0≤C):
  11     ∃t τ Δ:ℝ,0<t ∧ 0<τ ∧ 0<Δ ∧
  12       ∀x:ℝ,x≤Δ →
  13         ((K:ℝ)+1)*(t+C*x)+τ≤η ∧ t+C*x≤θ/(800*(K:ℝ)) :=by
  14   have hK':0<(K:ℝ):=by exact_mod_cast hK
  15   let r:ℝ:=min (η/(8*((K:ℝ)+1))) (θ/(1600*(K:ℝ)))
  16   have hr:0<r:=by dsimp [r]; positivity
  17   have hrη:r≤η/(8*((K:ℝ)+1)):=min_le_left _ _
  18   have hrθ:r≤θ/(1600*(K:ℝ)):=min_le_right _ _
  19   have hden:0<8*((K:ℝ)+1):=by positivity
  20   have hηbound:8*((K:ℝ)+1)*r≤η:=by simpa only [mul_comm] using (le_div_iff₀ hden).mp hrη
  21   have hθden:0<1600*(K:ℝ):=by positivity
  22   have hθbound:1600*(K:ℝ)*r≤θ:=by simpa only [mul_comm] using (le_div_iff₀ hθden).mp hrθ
  23   refine ⟨r,η/4,r/(C+1),hr,by positivity,by positivity,?_⟩
  24   intro x hx
  25   have hC1:0<C+1:=by linarith
  26   have hCx:C*x≤r:=by
  27     have hh:=mul_le_mul_of_nonneg_left hx hC
  28     have hfrac:C*(r/(C+1))≤r:=by
  29       rw [←mul_div_assoc]
  30       apply (div_le_iff₀ hC1).2
  31       nlinarith
  32     exact hh.trans hfrac
  33   constructor
  34   · have hKnonneg:0≤(K:ℝ)+1:=by positivity
  35     nlinarith [mul_nonneg hKnonneg (sub_nonneg.mpr hCx)]
  36   · apply (le_div_iff₀ (by positivity : 0<800*(K:ℝ))).mpr
  37     nlinarith [mul_nonneg hK'.le (sub_nonneg.mpr hCx)]
  38 end Oberwolfach
