  18     (a i:ℝ)*k-F ≤ Fintype.card (ProfileRemainingClass Z i) := by

  21   have hzZ : z ≤ Z.card := by
  22     rw [show z=(Z.filter (fun v => v.1=i)).card from (sigma_finset_class_card Z i).symm]

  24   have hzR : (z:ℝ) ≤ F := (by exact_mod_cast hzZ : (z:ℝ) ≤ Z.card).trans hZ
  25   rw [profileRemainingClass_card,Nat.cast_sub hzN,hn]
