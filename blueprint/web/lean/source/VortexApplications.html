  10 theorem Vortex.estimates_including_zero {V U : Type*} [Fintype V] [Fintype U]
  11     {G : SimpleGraph V} {D m : ℕ} {β ε : ℝ} (Q : Vortex U G D β ε m)
  12     (hreg : G.IsRegularOfDegree D) (hn : 0 < Fintype.card V)
  13     (hρ : 0 < initialIndexDensity U D) (hb : 0 ≤ β)
  14     (i : ℕ) (hi : i ≤ Q.length) :
  15     |2 * ((Q.indices i).card : ℝ) / initialIndexDensity U D -
  16       ((D : ℝ) / Fintype.card V) * (Q.vertices i).card| ≤ β * (Q.vertices i).card ∧
  17     ∀ x ∈ Q.vertices (i - 2),
  18       |((G.neighborFinset x ∩ Q.vertices i).card : ℝ) -
  19         2 * (Q.indices i).card / initialIndexDensity U D| ≤ β * (Q.vertices i).card := by
  20   by_cases hz : i = 0
  21   · subst i
  22     have hD : (D : ℝ) ≠ 0 := by
  23       intro h
  24       simp [initialIndexDensity, h] at hρ
  25     have hU : (Fintype.card U : ℝ) ≠ 0 := by
  26       intro h
  27       simp [initialIndexDensity, h] at hρ
  28     have hcentre : 2 * (Fintype.card U : ℝ) / initialIndexDensity U D = D := by
  29       dsimp only [initialIndexDensity]
  30       field_simp
  31     have hn' : (Fintype.card V : ℝ) ≠ 0 := by exact_mod_cast hn.ne'
  32     rw [Q.initial, Q.indices_initial, Finset.card_univ, Finset.card_univ, hcentre,
  33       div_mul_cancel₀ _ hn', sub_self, abs_zero]
  34     refine ⟨by positivity, ?_⟩
  35     intro x hx
  36     rw [Finset.inter_univ, SimpleGraph.card_neighborFinset_eq_degree, hreg x,
  37       sub_self, abs_zero]
  38     positivity
  39   · exact Q.centre_estimates i hi (by omega)

  41 theorem Vortex.later_size_le_quarter {V U : Type*} [Fintype V] [Fintype U]
  42     {G : SimpleGraph V} {D m : ℕ} {β ε : ℝ} (Q : Vortex U G D β ε m)
  43     (he : 0 ≤ ε) (heq : ε ≤ 1 / 4) {j k : ℕ} (hjk : j < k) (hk : k ≤ Q.length) :
  44     ((Q.vertices k).card : ℝ) ≤ (Q.vertices j).card / 4 := by
  45   have hsub := finset_chain_subset Q.vertices Q.length Q.nested
  46     (by omega : j + 1 ≤ k) hk
  47   have hsmall : ((Q.vertices k).card : ℝ) ≤ (Q.vertices (j + 1)).card := by
  48     exact_mod_cast Finset.card_le_card hsub
  49   have hf : ((Q.vertices (j + 1)).card : ℝ) ≤ ε * (Q.vertices j).card := by
  50     rw [Q.sizes (j + 1) (by omega), floorOrbit_succ, ← Q.sizes j (by omega)]
  51     exact Nat.floor_le (mul_nonneg he (Nat.cast_nonneg _))
  52   have hh := mul_le_mul_of_nonneg_right heq (Nat.cast_nonneg (Q.vertices j).card)
  53   linarith

  55 theorem Vortex.difference_degree_estimates {V U : Type*} [Fintype V] [Fintype U]
  56     {G : SimpleGraph V} {D m : ℕ} {β ε : ℝ} (Q : Vortex U G D β ε m)
  57     (hreg : G.IsRegularOfDegree D) (hn : 0 < Fintype.card V)
  58     (hρ : 0 < initialIndexDensity U D) (hb : 0 ≤ β)
  59     (he : 0 ≤ ε) (heq : ε ≤ 1 / 4) {j k : ℕ} (hjk : j < k) (hk : k ≤ Q.length)
  60     (x : V) (hx : x ∈ Q.vertices (k - 2)) :
  61     |2 * ((Q.indices j \ Q.indices k).card : ℝ) / initialIndexDensity U D -
  62         ((D : ℝ) / Fintype.card V) * (Q.vertices j \ Q.vertices k).card| ≤
  63       (5 / 3 : ℝ) * β * (Q.vertices j \ Q.vertices k).card ∧
  64     |((G.neighborFinset x ∩ (Q.vertices j \ Q.vertices k)).card : ℝ) -
  65         2 * (Q.indices j \ Q.indices k).card / initialIndexDensity U D| ≤
  66       (5 / 3 : ℝ) * β * (Q.vertices j \ Q.vertices k).card := by
  67   classical
  68   have hj : j ≤ Q.length := by omega
  69   have hxj : x ∈ Q.vertices (j - 2) :=
  70     finset_chain_subset Q.vertices Q.length Q.nested (by omega : j - 2 ≤ k - 2)
  71       (by omega) hx
  72   have hA := Q.estimates_including_zero hreg hn hρ hb j hj
  73   have hB := Q.estimates_including_zero hreg hn hρ hb k hk
  74   exact nested_domains_degree_subtraction G
  75     (finset_chain_subset Q.vertices Q.length Q.nested hjk.le hk)
  76     (finset_chain_subset Q.indices k (fun i _ => Q.indices_nested i) hjk.le (le_refl k))
  77     β (initialIndexDensity U D) ((D : ℝ) / Fintype.card V) hb
  78     (Q.later_size_le_quarter he heq hjk hk) hA.1 hB.1 x (hA.2 x hxj) (hB.2 x hx)
