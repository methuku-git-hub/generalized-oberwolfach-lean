   9 def markedPhysicalEdge {V E K : Type*} [DecidableEq V]
  10     (tail head : E → V) (ktail khead : K → V) (π : Equiv.Perm V) : E⊕K → Finset V :=
  11   Sum.elim (fun e => {tail e,head e}) (fun e => {ktail e,π (khead e)})

  13 theorem markedPhysicalEdge_endpoint_injections {V E K : Type*} [DecidableEq V]
  14     (tail head : E → V) (ktail khead : K → V) (π : Equiv.Perm V)
  15     (hp : Function.Injective (markedPhysicalEdge tail head ktail khead π)) :
  16     Function.Injective (fun e => (tail e,head e)) ∧
  17       Function.Injective (fun e => (ktail e,khead e)) := by
  18   constructor
  19   · intro e f hef
  20     change (tail e,head e)=(tail f,head f) at hef
  21     have ht : tail e=tail f := congrArg Prod.fst hef
  22     have hh : head e=head f := congrArg Prod.snd hef
  23     apply Sum.inl.inj
  24     apply hp
  25     simp only [markedPhysicalEdge,Sum.elim_inl,ht,hh]
  26   · intro e f hef
  27     change (ktail e,khead e)=(ktail f,khead f) at hef
  28     have ht : ktail e=ktail f := congrArg Prod.fst hef
  29     have hh : khead e=khead f := congrArg Prod.snd hef
  30     apply Sum.inr.inj
  31     apply hp
  32     simp only [markedPhysicalEdge,Sum.elim_inr,ht,hh]
