  11 theorem fixedSubset_normalized_count_mean {V : Type*} [Fintype V] {r : ℕ}
  12     [Nonempty (FixedSubset V r)] (hr : r ≤ Fintype.card V) (hrpos : 0 < r) (A : Finset V) :
  13     (FiniteLaw.uniform (FixedSubset V r)).expectation
  14       (fun S => ((A ∩ S.1).card : ℝ) / r) = (A.card : ℝ) / Fintype.card V := by
  15   have hrreal : (r : ℝ) ≠ 0 := by exact_mod_cast hrpos.ne'
  16   have he : (FiniteLaw.uniform (FixedSubset V r)).expectation
  17       (fun S => ((A ∩ S.1).card : ℝ) / r) =
  18       (FiniteLaw.uniform (FixedSubset V r)).expectation
  19       (fun S => ((A ∩ S.1).card : ℝ)) / r := by
  20     simp only [FiniteLaw.expectation, div_eq_mul_inv, ← mul_assoc, Finset.sum_mul]
  21   rw [he, fixedSubset_intersection_mean hr]
  22   field_simp

  24 theorem fixedSubset_normalized_count_tail {V : Type*} [Fintype V] {r : ℕ}
  25     [Nonempty (FixedSubset V r)] (hr : r ≤ Fintype.card V) (hrpos : 0 < r) (A : Finset V)
  26     {t : ℝ} (ht : 0 ≤ t) :
  27     (FiniteLaw.uniform (FixedSubset V r)).probability
  28       (fun S => t ≤ |((A ∩ S.1).card : ℝ) / r - (A.card : ℝ) / Fintype.card V|) ≤
  29       2 * Real.exp (-2 * (r : ℝ) * t ^ 2) := by
  30   have hrreal : (0 : ℝ) < r := by exact_mod_cast hrpos
  31   let f (S : FixedSubset V r) := ((A ∩ S.1).card : ℝ) / r
  32   have hex : ∀ S u v, |f S - f (fixedSubsetMap (Equiv.swap u v) S)| ≤ 1 / r := by
  33     intro S u v
  34     dsimp only [f, fixedSubsetMap]
  35     rw [← sub_div, abs_div, abs_of_pos hrreal]
  36     exact (div_le_div_iff_of_pos_right hrreal).mpr (swap_subset_intersection_change A S.1 u v)
  37   have hm (s : ℝ) : (FiniteLaw.uniform (FixedSubset V r)).expectation
  38       (fun S => Real.exp (s * (f S - (FiniteLaw.uniform (FixedSubset V r)).expectation f))) ≤
  39       Real.exp (s ^ 2 * (1 / (r : ℝ)) / 8) := by
  40     convert fixedSubset_mgf hr f (show 0 ≤ 1 / (r : ℝ) by positivity) hex s using 1
  41     congr 1
  42     field_simp
  43   have hh := finiteLaw_mgf_absolute_tail (FiniteLaw.uniform (FixedSubset V r)) f
  44     (show 0 ≤ 1 / (r : ℝ) by positivity) hm ht
  45   change (FiniteLaw.uniform (FixedSubset V r)).probability
  46       (fun S => t ≤ |((A ∩ S.1).card : ℝ) / r -
  47         (FiniteLaw.uniform (FixedSubset V r)).expectation (fun S => ((A ∩ S.1).card : ℝ) / r)|) ≤ _ at hh
  48   rw [fixedSubset_normalized_count_mean hr hrpos] at hh
  49   convert hh using 1
  50   congr 2
  51   field_simp

  53 theorem fixedSubset_simultaneous_normalized_counts
  54     {V J : Type*} [Fintype V] {r : ℕ} [Nonempty (FixedSubset V r)]
  55     (hr : r ≤ Fintype.card V) (hrpos : 0 < r) (tests : Finset J) (A : J → Finset V)
  56     {t : ℝ} (ht : 0 ≤ t) :
  57     1 - 2 * tests.card * Real.exp (-2 * (r : ℝ) * t ^ 2) ≤
  58       (FiniteLaw.uniform (FixedSubset V r)).probability (fun S => ∀ j ∈ tests,
  59         |((A j ∩ S.1).card : ℝ) / r - ((A j).card : ℝ) / Fintype.card V| < t) := by
  60   have hh := finiteLaw_common_event_lower (FiniteLaw.uniform (FixedSubset V r)) tests
  61     (fun j S => t ≤ |((A j ∩ S.1).card : ℝ) / r - ((A j).card : ℝ) / Fintype.card V|)
  62     (fun _ => 2 * Real.exp (-2 * (r : ℝ) * t ^ 2))
  63     (fun j _ => fixedSubset_normalized_count_tail hr hrpos (A j) ht)
  64   simp only [not_le, Finset.sum_const, nsmul_eq_mul] at hh
  65   convert hh using 1; ring

  67 theorem fixedSubset_normalized_counts_exists
  68     {V J : Type*} [Fintype V] {r : ℕ} (hr : r ≤ Fintype.card V)
  69     (hrpos : 0 < r) (tests : Finset J) (A : J → Finset V)
  70     {t : ℝ} (ht : 0 ≤ t)
  71     (hsmall : 2 * tests.card * Real.exp (-2 * (r : ℝ) * t ^ 2) < 1) :
  72     ∃ S : FixedSubset V r, ∀ j ∈ tests,
  73       |((A j ∩ S.1).card : ℝ) / r - ((A j).card : ℝ) / Fintype.card V| < t := by
  74   let : Nonempty (FixedSubset V r) := fixedSubset_nonempty hr
  75   have hh := finiteLaw_common_event_exists (FiniteLaw.uniform (FixedSubset V r)) tests
  76     (fun j S => t ≤ |((A j ∩ S.1).card : ℝ) / r - ((A j).card : ℝ) / Fintype.card V|)
  77     (fun _ => 2 * Real.exp (-2 * (r : ℝ) * t ^ 2))
  78     (fun j _ => fixedSubset_normalized_count_tail hr hrpos (A j) ht)
  79     (by simpa only [Finset.sum_const, nsmul_eq_mul, mul_left_comm, mul_assoc] using hsmall)
