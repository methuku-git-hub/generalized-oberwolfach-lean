  13 def optional {A:Type} [Fintype A] (p:ℝ) (hp:0≤p) (hp1:p≤1) (μ:FiniteLaw A):FiniteLaw (Option A) where
  14   mass x:=match x with | none=>1-p | some a=>p*μ.mass a
  15   nonneg x:=by
  16     cases x with
  17     | none => exact sub_nonneg.mpr hp1
  18     | some a => exact mul_nonneg hp (μ.nonneg a)
  19   total:=by
  20     rw [Fintype.sum_option]
  21     simp only [←Finset.mul_sum,μ.total,mul_one]
  22     ring

  28 theorem categorical_refinement_class_marginal {k:ℕ} {A:Type} [Fintype A]
  29     (ρ:Fin k→ℝ) (hρ:∀ a,0≤ρ a) (htotal:∑ a,ρ a=1)
  30     (ν:Fin k→FiniteLaw A) (a:Fin k):
  31     let μ:=(FiniteLaw.categorical ρ hρ htotal).sigma ν;
  32     μ.map (classRefinedView (A:=A) a)=FiniteLaw.optional (ρ a) (hρ a)
  33       (by have h:=Finset.single_le_sum (fun b _=>hρ b) (Finset.mem_univ a);linarith) (ν a):=by
  34   intro μ
  35   apply finiteLaw_ext
  36   intro x
  37   cases x with
  38   | none =>
  39     simp only [FiniteLaw.map,FiniteLaw.optional]
  40     rw [Fintype.sum_sigma]
  41     have hsum:(∑ c:Fin k,((if c=a then ρ c else 0)+(if c=a then 0 else ρ c)))=1:=by
  42       calc
  43         _ = ∑ c:Fin k,ρ c:=by
  44           apply Finset.sum_congr rfl
  45           intro c _
  46           by_cases h:c=a <;> simp [h]
  47         _ = 1:=htotal
  48     have hs0:(∑ c:Fin k,if c=a then ρ c else 0)=ρ a:=by simp
  49     have hs:(∑ c:Fin k,if c=a then 0 else ρ c)=1-ρ a:=by
  50       rw [Finset.sum_add_distrib,hs0] at hsum
  51       linarith
  52     calc
  53       _ = ∑ c:Fin k,if c=a then 0 else ρ c:=by
  54         apply Finset.sum_congr rfl
  55         intro c _
  56         by_cases hc:c=a
  57         · simp [classRefinedView,hc]
  58         · simp [classRefinedView,hc,μ,FiniteLaw.sigma,FiniteLaw.categorical,←Finset.mul_sum,(ν c).total]
  59       _ = _:=hs
  60   | some x =>
  61     simp only [FiniteLaw.map,FiniteLaw.optional]
  62     rw [Fintype.sum_sigma]
  63     have heach (c:Fin k):
  64         (∑ z:A,if classRefinedView a ⟨c,z⟩=some x then μ.mass ⟨c,z⟩ else 0)=
  65           if c=a then ρ a*(ν a).mass x else 0:=by
  66       by_cases hc:c=a
  67       · subst c
  68         simp [classRefinedView,μ,FiniteLaw.sigma,FiniteLaw.categorical]
  69       · simp [classRefinedView,hc]
  70     calc
  71       _ = ∑ c:Fin k,if c=a then ρ a*(ν a).mass x else 0:=by
  72         apply Finset.sum_congr rfl
  73         intro c _
  74         convert heach c using 1
  75         apply Finset.sum_congr rfl
  76         intro y _
  77         by_cases h:classRefinedView a ⟨c,y⟩=some x <;> simp [h]
  78       _ = _:=by simp
