  17 def indexedMarkedWheelExtension (G:IndexedMarkedGraph V U A S) (m:ℕ):
  18     IndexedMarkedGraph (V⊕(I×Fin (m+1))) (U⊕I) (A⊕(I×Fin (m+1))) (S⊕(I×Fin (m+1))):=
  19   {src:=paddingRimTail (Sum.inl∘G.src) (fun i k=>Sum.inr (i,k))
  20    dst:=paddingRimHead (Sum.inl∘G.dst) (fun i k=>Sum.inr (i,k))
  21    row:=paddingSpokeRow (Sum.inl∘G.row) (fun i k=>Sum.inr (i,k))
  22    hub:=paddingSpokeHub G.hub m
  23    jr:=paddingRimType G.jr m
  24    js:=paddingSpokeType G.js m}

  26 theorem indexedMarkedWheelExtension_balanced (G:IndexedMarkedGraph V U A S) (m:ℕ)
  27     (hG:G.IsBalanced m):(indexedMarkedWheelExtension (I:=I) G m).IsBalanced m:=by
  28   let f:V→V⊕(I×Fin (m+1)):=Sum.inl
  29   have hmap:=G.map_balanced m hG f id
  30   have hmrow:∀ v:V⊕(I×Fin (m+1)),(Finset.univ.filter (fun a:{a:A // f (G.dst a)=v}=>G.jr a.val=true)).card=
  31       (Finset.univ.filter (fun s:{s:S // f (G.row s)=v}=>G.js s.val=true)).card:=by
  32     intro v
  33     calc
  34       _ = Fintype.card {a:A // f (G.dst a)=v∧G.jr a=true} := by
  35         exact finite_typed_fiber_filter_card (fun a=>f (G.dst a)) G.jr v true
  36       _ = Fintype.card {s:S // f (G.row s)=v∧G.js s=true} := hmap.marked v
  37       _ = _ := (finite_typed_fiber_filter_card (fun s=>f (G.row s)) G.js v true).symm
  38   have hbase:=padding_row_balance (f∘G.src) (f∘G.dst) (f∘G.row) G.jr G.js m
  39     (fun i k=>Sum.inr (i,k):I→Fin (m+1)→V⊕(I×Fin (m+1)))
  40     (fun v=>⟨hmap.directed v,hmap.row v⟩) hmrow
  41   refine ⟨fun v=>(hbase.1 v).1,fun v=>(hbase.1 v).2,?_,?_⟩
  42   · intro v
  43     have hh:=hbase.2 v
  44     rw [finite_typed_fiber_filter_card,finite_typed_fiber_filter_card] at hh
  45     exact hh
  46   · exact padding_hub_balance G.hub G.js m hG.hub
