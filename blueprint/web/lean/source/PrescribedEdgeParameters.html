  11 theorem prescribed_edge_cap_parameters (r:ℕ) (β:ℝ) (hβ:0<β) :
  12     ∃ α₀:ℝ,0<α₀ ∧ α₀≤1 ∧ ∀ α L:ℝ,0<α → α≤α₀ → 0≤L →
  13       ∃ m₀:ℕ,∀ m N D:ℕ,m₀≤m → (N:ℝ)≤α^2*m^2 → (D:ℝ)≤α*m →
  14       let K:=⌈α*m⌉₊
  15       0<K ∧ (K:ℝ)≤2*α*m ∧
  16       (N*r:ℕ)/(K:ℝ)+(K*r:ℕ)+((r:ℝ)+2)*(L+1)+4*(K+D:ℕ)<β*m ∧
  17       (N:ℝ)/K+2*(K+D:ℕ)<β*m ∧
  18       (N*r:ℕ)/(K:ℝ)+(D*r:ℕ)+((r:ℝ)+2)*(L+1)+8*(K+D:ℕ)<β*m :=by
  19   let c:ℝ:=3*r+24
  20   have hc:0<c:=by dsimp [c];positivity
  21   let α₀:=min 1 (β/(4*c))
  22   have hα₀:0<α₀:=by dsimp [α₀];positivity
  23   refine ⟨α₀,hα₀,min_le_left _ _,?_⟩
  24   intro α L hα hαsmall hL
  25   obtain ⟨M₁,hM₁⟩:=exists_nat_gt (1/α)
  26   obtain ⟨M₂,hM₂⟩:=exists_nat_gt (4*((r:ℝ)+2)*(L+1)/β)
  27   refine ⟨max M₁ M₂,?_⟩
  28   intro m N D hm hN hD
  29   dsimp only
  30   let K:=⌈α*m⌉₊
  31   have hαm:1<α*m:=by
  32     have h:1/α<(m:ℝ):=hM₁.trans_le (Nat.cast_le.mpr (show M₁≤m by omega))
  33     have hh:=(div_lt_iff₀ hα).mp h
  34     nlinarith only [hh]
  35   have hmpos:(0:ℝ)<m:=by nlinarith only [hαm,hα.le]
  36   have hKlo:α*m≤(K:ℝ):=Nat.le_ceil _
  37   have hKpos:(0:ℝ)<K:=lt_of_lt_of_le (by linarith only [hαm]) hKlo
  38   have hKhi:(K:ℝ)≤2*α*m:=by
  39     have h:=Nat.ceil_lt_add_one (show 0≤α*m by linarith only [hαm])
  40     change (K:ℝ)<α*m+1 at h
  41     linarith only [h,hαm]
  42   have hNs:(N:ℝ)/K≤α*m:=by
  43     apply (div_le_iff₀ hKpos).mpr
  44     have hh:=mul_le_mul_of_nonneg_left hKlo (show 0≤α*m by positivity)
  45     nlinarith only [hN,hh]
  46   have hNr:(N*r:ℕ)/(K:ℝ)≤r*(α*m):=by
  47     have hh:=mul_le_mul_of_nonneg_right hNs (Nat.cast_nonneg r)
  48     simpa only [Nat.cast_mul,mul_div_assoc,mul_comm,mul_left_comm,mul_assoc] using hh
  49   have hαbound:α*c≤β/4:=by
  50     have ha:α≤β/(4*c):=hαsmall.trans (min_le_right _ _)
  51     have hh:=(le_div_iff₀ (by positivity:0<4*c)).mp ha
  52     linarith only [hh]
  53   have hsmall:c*(α*m)≤β*m/4:=by
  54     have hh:=mul_le_mul_of_nonneg_right hαbound hmpos.le
  55     nlinarith only [hh]
  56   have hconstant:((r:ℝ)+2)*(L+1)<β*m/4:=by
  57     have hh:=(div_lt_iff₀ hβ).mp (hM₂.trans_le (Nat.cast_le.mpr (show M₂≤m by omega)))
  58     nlinarith only [hh]
  59   have hKd:((K+D:ℕ):ℝ)≤3*(α*m):=by simp only [Nat.cast_add];linarith only [hKhi,hD]
  60   have hKr:((K*r:ℕ):ℝ)≤2*r*(α*m):=by
  61     rw [Nat.cast_mul]
  62     have hh:=mul_le_mul_of_nonneg_right hKhi (Nat.cast_nonneg r)
  63     nlinarith only [hh]
  64   have hDr:((D*r:ℕ):ℝ)≤r*(α*m):=by
  65     rw [Nat.cast_mul]
  66     have hh:=mul_le_mul_of_nonneg_right hD (Nat.cast_nonneg r)
  67     nlinarith only [hh]
  68   have hβm:0<β*m:=mul_pos hβ hmpos
  69   have hnon:0≤(r:ℝ)*(α*m):=by positivity
  70   refine ⟨Nat.cast_pos.mp hKpos,hKhi,?_,?_,?_⟩
  71   · dsimp [c] at hsmall
  72     nlinarith only [hNr,hKr,hKd,hsmall,hconstant,hβm,hαm]
  73   · have hseven:7*(α*m)≤c*(α*m):=by
  74       dsimp [c]
  75       nlinarith only [hnon,hαm]
  76     nlinarith only [hNs,hKd,hseven,hsmall,hβm]
  77   · dsimp [c] at hsmall
  78     nlinarith only [hNr,hDr,hKd,hsmall,hconstant,hβm,hnon,hαm]
