  12 theorem cyclicChainOrder_cast_ne {k : ℕ} (c : Fin (k + 2)) (e : Fin (k + 1)) :
  13     cyclicChainOrder c e.castSucc ≠ c := by
  14   intro h
  15   have hh := (cyclicChainOrder c).injective (h.trans (cyclicChainOrder_last c).symm)
  16   have hv := congrArg Fin.val hh
  17   simp only [Fin.val_castSucc, Fin.val_last] at hv
  18   omega

  29 theorem cyclic_complement_prod {k : ℕ} (c : Fin (k + 2)) (f : Fin (k + 2) → ℝ) :
  30     (∏ e : Fin (k + 1), f (cyclicChainOrder c e.castSucc)) =
  31       ∏ i ∈ Finset.univ.erase c, f i := by
  32   classical
  33   apply Finset.prod_bij (fun e _ => cyclicChainOrder c e.castSucc)
  34   · intro e _
  35     simp only [Finset.mem_erase, Finset.mem_univ, and_true]
  36     exact cyclicChainOrder_cast_ne c e
  37   · intro a _ b _ hab
  38     have hh := (cyclicChainOrder c).injective hab
  39     exact Fin.ext (congrArg (fun z : Fin (k + 2) => z.val) hh)
  40   · intro i hi
  41     obtain ⟨e, he, _, _⟩ := cyclicChainOrder_positions c i (Finset.mem_erase.mp hi).1
  42     refine ⟨e, Finset.mem_univ _, ?_⟩
  43     rw [← he, Equiv.apply_symm_apply]
  44   · intro e _
  45     rfl

  47 theorem cyclic_internal_prod {k : ℕ} (c : Fin (k + 2)) (f : Fin (k + 2) → ℝ) :
  48     (∏ e : Fin k, f (cyclicChainOrder c e.succ.castSucc)) =
  49       ∏ i ∈ (Finset.univ.erase c).erase (finRotate (k + 2) c), f i := by
  50   classical
  51   apply Finset.prod_bij (fun e _ => cyclicChainOrder c e.succ.castSucc)
  52   · intro e _
  53     simp only [Finset.mem_erase, Finset.mem_univ, and_true]
  54     exact ⟨cyclicChainOrder_internal_ne_first c e, cyclicChainOrder_cast_ne c e.succ⟩
  55   · intro a _ b _ hab
  56     have hh := congrArg Fin.val ((cyclicChainOrder c).injective hab)
  57     apply Fin.ext
  58     simpa only [Fin.val_castSucc, Fin.val_succ, Nat.add_right_cancel_iff] using hh
  59   · intro i hi
  60     obtain ⟨hi0, hi⟩ := Finset.mem_erase.mp hi
  61     obtain ⟨e, he, _, _⟩ := cyclicChainOrder_positions c i (Finset.mem_erase.mp hi).1
  62     have he0 : e ≠ 0 := by
  63       intro hz
  64       have hh := congrArg (cyclicChainOrder c) he
  65       simp only [Equiv.apply_symm_apply, hz, Fin.castSucc_zero, cyclicChainOrder_zero] at hh
  66       exact hi0 hh
  67     obtain ⟨a, ha⟩ := Fin.exists_succ_eq_of_ne_zero he0
  68     refine ⟨a, Finset.mem_univ _, ?_⟩
  69     rw [ha, ← he, Equiv.apply_symm_apply]
  70   · intro e _
  71     rfl

  73 theorem complementary_chain_product {k : ℕ} {X : Type*}
  74     (m : Fin (k + 2) → X → ℝ) (K : Fin (k + 2) → X → X → ℝ)
  75     (hm : ∀ i x, 0 < m i x) (c : Fin (k + 2)) (x : Fin (k + 2) → X) :
  76     chainJoint m K (finRotate (k + 2)) (finRotate (k + 2) c) (k + 1)
  77       (fun j => x (cyclicChainOrder c j)) =
  78       (∏ i ∈ Finset.univ.erase c, K i (x i) (x (finRotate (k + 2) i))) /
  79         ∏ j ∈ (Finset.univ.erase c).erase (finRotate (k + 2) c), m j (x j) := by
  80   rw [chainJoint_product m K (finRotate (k + 2)) hm]
  81   have hnum :
  82       (∏ t : Fin (k + 1),
  83         K ((finRotate (k + 2))^[t.val] (finRotate (k + 2) c))
  84           (x (cyclicChainOrder c t.castSucc)) (x (cyclicChainOrder c t.succ))) =
  85         ∏ t : Fin (k + 1), K (cyclicChainOrder c t.castSucc)
  86           (x (cyclicChainOrder c t.castSucc))
  87           (x (finRotate (k + 2) (cyclicChainOrder c t.castSucc))) := by
  88     apply Finset.prod_congr rfl
  89     intro t _
  90     have ht := cyclicChainOrder_iterate c t.castSucc
  91     simp only [Fin.val_castSucc] at ht
  92     rw [← ht, cyclicChainOrder_succ]
  93   have hden :
  94       (∏ t : Fin k,
  95         m ((finRotate (k + 2))^[t.val + 1] (finRotate (k + 2) c))
  96           (x (cyclicChainOrder c t.succ.castSucc))) =
  97         ∏ t : Fin k, m (cyclicChainOrder c t.succ.castSucc)
  98           (x (cyclicChainOrder c t.succ.castSucc)) := by
  99     apply Finset.prod_congr rfl
 100     intro t _
 101     have ht := cyclicChainOrder_iterate c t.succ.castSucc
 102     simp only [Fin.val_castSucc, Fin.val_succ] at ht
 103     rw [← ht]
 104   rw [hnum, hden,
 105     cyclic_complement_prod c (fun i => K i (x i) (x (finRotate (k + 2) i))),
 106     cyclic_internal_prod c (fun j => m j (x j))]
