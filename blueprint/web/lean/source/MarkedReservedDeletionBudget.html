  16 theorem marked_reserved_packing_deletion_bounds
  17     (P:SimpleGraph V) (π:Equiv.Perm V) (d m:ℕ) (σ:RawMarkedOutcome V U Γ Color)
  18     (H:Bool→V→U→Prop) (I:Finset V) (J:Finset U) (hI:∀ x,π x∈I↔x∈I):
  19     let E₀:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) false
  20     let E₁:=MarkedSampleArc P π (3*d) (fun e=>σ (Sum.inl e)) true
  21     let tail:=fun e:E₀=>e.val.1
  22     let head:=fun e:E₀=>e.val.2
  23     let ktail:=fun e:E₁=>e.val.1
  24     let khead:=fun e:E₁=>e.val.2
  25     ∀ (KA:Finset (E₀⊕E₁)) (KS:Finset (V×U)),
  26     (∀ a∈KA,Sum.elim tail ktail a∈I∧Sum.elim head khead a∈I)→
  27     (∀ z∈KS,z.1∈I∧z.2∈J)→
  28     ∀ M:Finset (MarkedWheel tail head ktail khead m π (3*d) (H false) (H true)),
  29     ∀ cR cV cU:ℝ,
  30     (∀ x,((markedPackingRimGraph M).degree x:ℝ)≤cR)→
  31     (∀ x,((Finset.univ.filter (markedPackingSpokes M x)).card:ℝ)≤cV)→
  32     (∀ u,((Finset.univ.filter (fun x=>markedPackingSpokes M x u)).card:ℝ)≤cU)→
  33     let DR:=markedSampleArcSetGraph P π d σ (KA∪M.biUnion markedWheelArcs)
  34     let DS:=markedFiniteSpokeGraph (KS∪M.biUnion markedWheelSpokePairs)
  35     (∀ x,(DR.degree x:ℝ)≤I.card+cR) ∧
  36     (∀ x,(DS.degree (Sum.inl x):ℝ)≤J.card+cV) ∧
  37     (∀ u,(DS.degree (Sum.inr u):ℝ)≤I.card+cU):=by
  38   intro E₀ E₁ tail head ktail khead KA KS hKA hKS M cR cV cU hR hV hU
  39   have hphys (a:E₀⊕E₁) (ha:a∈KA):markedPhysicalEdge tail head ktail khead π a⊆I:=by
  40     have hh:=hKA a ha
  41     cases a with
  42     | inl a=>exact by simpa only [markedPhysicalEdge,Sum.elim_inl,Sum.elim_inr,Finset.insert_subset_iff,Finset.singleton_subset_iff] using hh
  43     | inr a=>exact by simpa only [markedPhysicalEdge,Sum.elim_inl,Sum.elim_inr,Finset.insert_subset_iff,Finset.singleton_subset_iff] using And.intro hh.1 ((hI _).mpr hh.2)
  44   have hKR (x:V):(markedSampleArcSetGraph P π d σ KA).degree x≤I.card:=by
  45     rw [←SimpleGraph.card_neighborFinset_eq_degree]
  46     apply Finset.card_le_card
  47     intro y hy
  48     obtain ⟨a,ha,he⟩:=Finset.mem_image.mp (((markedSampleArcSetGraph P π d σ KA).mem_neighborFinset x y).mp hy).2
  49     apply hphys a ha
  50     rw [he]
  51     simp
  52   have hKV (x:V):(markedFiniteSpokeGraph KS).degree (Sum.inl x)≤J.card:=by
  53     rw [markedFiniteSpokeGraph_degree_row]
  54     exact Finset.card_le_card (fun u hu=>(hKS (x,u) (Finset.mem_filter.mp hu).2).2)
  55   have hKU (u:U):(markedFiniteSpokeGraph KS).degree (Sum.inr u)≤I.card:=by
  56     rw [markedFiniteSpokeGraph_degree_col]
  57     exact Finset.card_le_card (fun x hx=>(hKS (x,u) (Finset.mem_filter.mp hx).2).1)
  58   dsimp only
  59   rw [markedSampleArcSetGraph_union,markedSampleArcSetGraph_packing,markedFiniteSpokeGraph_union]
  60   obtain ⟨hMR,hMC⟩:=markedFiniteSpokeGraph_packing_degrees M
  61   refine ⟨?_,?_,?_⟩
  62   · intro x
  63     have hg:=(Nat.cast_le (α:=ℝ)).mpr (graph_degree_sup_le (markedSampleArcSetGraph P π d σ KA) (markedPackingRimGraph M) x)
  64     have hk:(↑((markedSampleArcSetGraph P π d σ KA).degree x):ℝ)≤I.card:=Nat.cast_le.mpr (hKR x)
  65     push_cast at hg
  66     simpa only [←SimpleGraph.ncard_neighborSet] using hg.trans (add_le_add hk (hR x))
  67   · intro x
  68     have hg:=(Nat.cast_le (α:=ℝ)).mpr (graph_degree_sup_le (markedFiniteSpokeGraph KS)
  69       (markedFiniteSpokeGraph (M.biUnion markedWheelSpokePairs)) (Sum.inl x))
  70     have hk:(↑((markedFiniteSpokeGraph KS).degree (Sum.inl x)):ℝ)≤J.card:=Nat.cast_le.mpr (hKV x)

  78     rw [hMC] at hg
