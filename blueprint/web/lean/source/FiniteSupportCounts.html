  14 theorem finite_support_loss_count (F : D → Finset V) (A B : Finset V) (K : ℕ)
  15     (hK : ∀ v,Fintype.card {d : D // F d⊆A∪B ∧ v∈F d} ≤ K) :
  16     Fintype.card {d : D // F d⊆A ∧ ¬F d⊆B} ≤ (A\B).card*K := by
  17   classical
  18   let Q (v : V) := Finset.univ.filter (fun d : D => F d⊆A∪B ∧ v∈F d)
  19   have hs : Finset.univ.filter (fun d : D => F d⊆A ∧ ¬F d⊆B) ⊆ (A\B).biUnion Q := by
  20     intro d hd
  21     have hd' := (Finset.mem_filter.mp hd).2
  22     obtain ⟨v,hv,hvB⟩ := Finset.not_subset.mp hd'.2
  23     exact Finset.mem_biUnion.mpr ⟨v,Finset.mem_sdiff.mpr ⟨hd'.1 hv,hvB⟩,
  24       Finset.mem_filter.mpr ⟨Finset.mem_univ _,hd'.1.trans Finset.subset_union_left,hv⟩⟩
  25   simp only [Fintype.card_subtype]
  26   calc
  27     _ ≤ ((A\B).biUnion Q).card := Finset.card_le_card hs
  28     _ ≤ ∑ v∈A\B,(Q v).card := Finset.card_biUnion_le
  29     _ ≤ ∑ _v∈A\B,K := Finset.sum_le_sum (fun v _ => by simpa only [Fintype.card_subtype,Q] using hK v)
  30     _ = _ := by simp

  32 theorem finite_support_count_difference (F : D → Finset V) (A B : Finset V) (K : ℕ)
  33     (hK : ∀ v,Fintype.card {d : D // F d⊆A∪B ∧ v∈F d} ≤ K) :
  34     |(Fintype.card {d : D // F d⊆A}:ℝ)-Fintype.card {d : D // F d⊆B}| ≤
  35       ((A\B).card+(B\A).card:ℕ)*(K:ℝ) := by
  36   let SA := Finset.univ.filter (fun d : D => F d⊆A)
  37   let SB := Finset.univ.filter (fun d : D => F d⊆B)
  38   have ha : (SA\SB).card ≤ (A\B).card*K := by
  39     have hh := finite_support_loss_count F A B K hK
  40     have he : SA\SB=Finset.univ.filter (fun d : D => F d⊆A ∧ ¬F d⊆B) := by ext d;simp [SA,SB]
  41     simpa only [Fintype.card_subtype,← he] using hh
  42   have hb : (SB\SA).card ≤ (B\A).card*K := by
  43     have hh := finite_support_loss_count F B A K (fun v => by simpa only [Finset.union_comm] using hK v)
  44     have he : SB\SA=Finset.univ.filter (fun d : D => F d⊆B ∧ ¬F d⊆A) := by ext d;simp [SA,SB]
  45     simpa only [Fintype.card_subtype,← he] using hh
  46   have h₁ : SA.card ≤ SB.card+(A\B).card*K := by
  47     have hh := Finset.card_le_card_sdiff_add_card (s:=SA) (t:=SB)
  48     omega
  49   have h₂ : SB.card ≤ SA.card+(B\A).card*K := by
  50     have hh := Finset.card_le_card_sdiff_add_card (s:=SB) (t:=SA)
  51     omega
  52   have h₁r : (SA.card:ℝ) ≤ SB.card+(A\B).card*K := by exact_mod_cast h₁
  53   have h₂r : (SB.card:ℝ) ≤ SA.card+(B\A).card*K := by exact_mod_cast h₂
  54   have hn₁ : 0 ≤ ((A\B).card:ℝ)*K := by positivity
  55   have hn₂ : 0 ≤ ((B\A).card:ℝ)*K := by positivity
  56   simp only [Fintype.card_subtype]
  57   change |(SA.card:ℝ)-SB.card| ≤ _
  58   push_cast
  59   apply abs_le.mpr
  60   constructor <;> nlinarith
