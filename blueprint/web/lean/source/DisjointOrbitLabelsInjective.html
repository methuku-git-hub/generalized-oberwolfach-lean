   9 theorem disjoint_extended_orbit_labels_injective
  10     {V I:Type} [Fintype V] [DecidableEq V] (π:Equiv.Perm V) (d:ℕ) (root:I→V)
  11     (hinj:∀i,Function.Injective (fun j:Fin (d+1)=>(π^j.val) (root i)))
  12     (hdis:∀i j,i≠j→Disjoint (permutationWindow π 0 d (root i)) (permutationWindow π 0 d (root j))):
  13     Function.Injective (fun z:I×Fin (d+1)=>(π^z.2.val) (root z.1)):=by
  14   rintro ⟨i,j⟩ ⟨i',j'⟩ he
  15   change (π^j.val) (root i)=(π^j'.val) (root i') at he
  16   by_cases hi:i=i'
  17   · subst i'
  18     exact Prod.ext rfl (hinj i he)
  19   · have hleft:(π^j.val) (root i)∈permutationWindow π 0 d (root i):=by
  20       rw [permutationWindow_mem_iff]
  21       simp only [zero_add,pow_zero,Equiv.refl_apply]
  22       exact ⟨⟨j.val,by have hh:=j.isLt;omega⟩,rfl⟩
  23     have hright:(π^j.val) (root i)∈permutationWindow π 0 d (root i'):=by
  24       rw [he,permutationWindow_mem_iff]
  25       simp only [zero_add,pow_zero,Equiv.refl_apply]
  26       exact ⟨⟨j'.val,by have hh:=j'.isLt;omega⟩,rfl⟩
  27     exact False.elim (Finset.disjoint_left.mp (hdis i i' hi) hleft hright)
  28 end Oberwolfach
