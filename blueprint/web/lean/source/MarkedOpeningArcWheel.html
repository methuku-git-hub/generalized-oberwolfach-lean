  16     (v : V) (u : U) (p : Fin (n+1) → V)

  27     intro i hi
  28     refine Fin.cases (fun h => (h rfl).elim) (fun j _ => ?_) i hi
  29     refine Fin.lastCases ?_ (fun k => ?_) j
  30     · have he : (Fin.last n).succ=Fin.last (n+1) := rfl
  31       simpa only [z,Function.Embedding.coeFn_mk,he,finRotate_last,Fin.cons_last,Fin.cons_zero] using hclose
  32     · have he : k.castSucc.succ=k.succ.castSucc := rfl
  33       have hv : Fin.cons (α:=fun _=>V) v p k.succ.castSucc=p k.castSucc := by
  34         rw [←he,Fin.cons_succ]

  36   have hK : ∃ e, ktail e=z 0 ∧ khead e=z (finRotate (n+2) 0) := by

  41     have he : finRotate (n+2) (0 : Fin (n+2))=(0 : Fin (n+1)).succ :=
  42       finRotate_castSucc_eq_succ (n+1) 0

  44   have hsO : ∀ i, i≠finRotate (n+2) 0 → H₀ (z i) u := by
