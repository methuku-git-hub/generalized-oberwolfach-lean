  10 theorem residual_subtype_model (R:Finset V) (H:SimpleGraph V) (D:V→V→Prop)
  11     (ps:List (PrescribedSourcePiece V))
  12     (hp:ps.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset))
  13     (hv:∀p∈ps,p.Shape ∧ p.OriginalValid D)
  14     (hR:prescribedDependentVertices (fun _ : Unit=>ps) ()=R)
  15     (hH:prescribedDependentGraph (fun _ : Unit=>ps) ()=H)
  16     (hD:∀x y,prescribedDependentDigraph (fun _ : Unit=>ps) () x y ↔ D x y ∧ H.Adj x y):
  17     ∃ qs:List (PrescribedSourcePiece {x // x∈R}),
  18       qs.map (fun q=>q.map Subtype.val)=ps ∧
  19       qs.Pairwise (fun p q=>Disjoint p.vertices.toFinset q.vertices.toFinset) ∧
  20       (∀q∈qs,q.Shape ∧ q.OriginalValid (fun x y=>D x.1 y.1)) ∧
  21       prescribedDependentVertices (fun _ : Unit=>qs) ()=Finset.univ ∧
  22       prescribedDependentGraph (fun _ : Unit=>qs) ()=H.induce (↑R:Set V) ∧
  23       ∀x y,prescribedDependentDigraph (fun _ : Unit=>qs) () x y ↔ D x.1 y.1 ∧ H.Adj x.1 y.1 := by
  24   have hR':sourcePieceVerticesUnion ps=R:=by rwa [←dependent_source_vertices_of_list]
  25   obtain ⟨qs,hmap,hpair,hvalid,hvertices⟩:=source_pieces_on_surviving_subtype R D ps hp hv hR'
  26   refine ⟨qs,hmap,hpair,hvalid,?_,?_,?_⟩
  27   · rw [dependent_source_vertices_of_list,hvertices]
  28   · ext x y
  29     have hmem:=source_subtype_edges_membership R qs ps hmap s(x,y)
  30     have hpq:(s(x,y)∈sourcePieceEdgesUnion qs) ↔ s(x.1,y.1)∈sourcePieceEdgesUnion ps:=by simpa using hmem
  31     have hadj: (prescribedDependentGraph (fun _ : Unit=>ps) ()).Adj x.1 y.1 ↔ H.Adj x.1 y.1 := by rw [hH]
  32     simp only [prescribedDependentGraph,SimpleGraph.fromEdgeSet_adj] at hadj ⊢
  33     simp only [sourcePieceEdgesUnion,Finset.mem_biUnion,
  34       List.mem_toFinset] at hpq
  35     simp only [prescribedDependentPhysical,Set.mem_ofPred_eq] at hadj ⊢
  36     have hne:x≠y ↔ x.1≠y.1:=by
  37       constructor
  38       · intro h hxy;exact h (Subtype.ext hxy)
  39       · intro h hxy;exact h (congrArg Subtype.val hxy)
  40     rw [hne]

  42   · intro x y
  43     rw [dependent_source_arcs_of_list,source_subtype_arcs_membership R qs ps hmap,
  44       ←dependent_source_arcs_of_list,hD]
