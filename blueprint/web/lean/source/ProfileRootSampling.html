  22 theorem sigma_finset_card (S : Finset (Σ i,V i)) :
  23     S.card=∑ i,(Finset.univ.filter (fun v : V i => (⟨i,v⟩ : Σ i,V i)∈S)).card := by
  24   let e : S ≃ Σ i,{v : V i // (⟨i,v⟩ : Σ i,V i)∈S} := {
  25     toFun v := ⟨v.val.1,⟨v.val.2,v.property⟩⟩
  26     invFun v := ⟨⟨v.1,v.2.val⟩,v.2.property⟩
  27     left_inv := by rintro ⟨⟨i,v⟩,h⟩;rfl
  28     right_inv := by rintro ⟨i,⟨v,h⟩⟩;rfl }
  29   have hh := Fintype.card_congr e
  30   simpa only [Fintype.card_coe,Fintype.card_sigma,Fintype.card_subtype,Finset.filter_mem_eq_inter,Finset.univ_inter] using hh

  42     [∀ i,Nonempty (FixedSubset (V i) (a i))]
  43     (k : ℝ) (hk : 1 < k) (ha : ∀ i,1 ≤ a i)
  44     (hn : ∀ i,(Fintype.card (V i):ℝ)=(a i:ℝ)*k) (v : Σ i,V i) :
  45     (FiniteLaw.pi (fun i => FiniteLaw.uniform (FixedSubset (V i) (a i)))).probability
  46       (fun σ => v∈profileRootSet σ)=1/k := by
  47   rcases v with ⟨i,v⟩
  48   let μ := fun j => FiniteLaw.uniform (FixedSubset (V j) (a j))
  49   let : ∀ j,MeasurableSpace (FixedSubset (V j) (a j)) := fun _ => ⊤
  50   let : ∀ j,MeasurableSingletonClass (FixedSubset (V j) (a j)) := fun _ => ⟨fun _ => trivial⟩
  51   have hai : (0:ℝ)<a i := by exact_mod_cast (show 0<a i by have hh:=ha i;omega)
  52   have hr : a i ≤ Fintype.card (V i) := by
  53     have hh : (a i:ℝ) ≤ Fintype.card (V i) := by rw [hn i];nlinarith
  54     exact_mod_cast hh
  55   have hh := finiteLaw_coordinate_expectation μ i (fun S => if v∈S.val then (1:ℝ) else 0)
  56   have hm := fixedSubset_vertex_probability (V:=V i) hr v
  57   have he : (FiniteLaw.pi μ).probability (fun σ => v∈(σ i).val)=(μ i).probability (fun S => v∈S.val) := by
  58     simp only [FiniteLaw.expectation,FiniteLaw.probability,mul_ite,mul_one,mul_zero] at hh ⊢
  59     convert hh using 1 <;> congr!
  60   simp only [profileRootSet_mem]
  61   rw [he,hm,hn i]
