  25     (himage:B.image g=(residualSubtypeSet a.remaining a.prescribed).image (fun x=>ψ x.1)∪C∪Z)
  26     (hOldC:Disjoint (a.oldDomain.image ψ) C)
  27     (hOldZ:Disjoint (a.oldDomain.image ψ) Z)
  28     (hfree:∀x,x∉B→Φ x∈L) (hL:Disjoint L (a.oldDomain.image ψ)):
  29     ∃h:X→V,Function.Injective h ∧ (∀x∈a.oldDomain,h x=ψ x) ∧
  30       (∀x:{x // x∈a.remaining},h x.1=Φ x):=by
  31   let embed:{x // x∈a.remaining}↪X:=⟨Subtype.val,Subtype.val_injective⟩

  33       x∈residualSubtypeSet a.remaining a.prescribed↔embed x∈a.oldDomain := by

  39     simp only [Finset.mem_image,Finset.mem_univ,true_and]
  40     change (∃y:{y // y∈a.remaining},y.1=x) ↔ x∈a.remaining
