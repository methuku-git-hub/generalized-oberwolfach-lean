  16 theorem marked_cover_remainder_fixed_inner_complement
  17     (tail head : E→V) (ktail khead : K→V) (π : Equiv.Perm V)
  18     (S : Finset (V×U)) (I : Finset V) (J : Finset U) (hI : ∀ v,π v∈I↔v∈I)
  19     (AR : Finset (E⊕K)) (SR : Finset (V×U)) (hSR : SR⊆S)
  20     (hA : ∀ a∈AR,Sum.elim tail ktail a∈I∧Sum.elim head khead a∈I)
  21     (hS : ∀ s∈SR,s.1∈I∧s.2∈J) :
  22     let B := markedFiniteResources tail head ktail khead π Finset.univ S
  23     let L := markedFiniteResources tail head ktail khead π AR SR
  24     let F := B.filter (markedInnerResource I J)\L;
  25     L⊆B.filter (markedInnerResource I J) ∧ F⊆B.filter (markedInnerResource I J) ∧
  26     L=B.filter (markedInnerResource I J)\F := by
  27   intro B L F
  28   have hLB : L⊆B := markedFiniteResources_mono (Finset.subset_univ _) hSR
  29   have hIn : ∀ e∈L,markedInnerResource I J e :=
  30     (markedFiniteResources_support_iff tail head ktail khead π AR SR I J hI).mpr ⟨hA,hS⟩
  31   have hLI : L⊆B.filter (markedInnerResource I J) :=
  32     fun e he=>Finset.mem_filter.mpr ⟨hLB he,hIn e he⟩
  33   refine ⟨hLI,Finset.sdiff_subset,?_⟩
  34   ext e
  35   have hh := fun he=>hLI (x:=e) he
